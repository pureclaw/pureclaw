   20 -- | Create a delegate_task tool that spawns an isolated sub-agent
   21 -- to complete a goal. The sub-agent runs with a restricted toolset
   22 -- (optionally filtered), a bounded turn count, and returns a summary
   23 -- of its work.
   24 --
   25 -- The sub-agent uses the same provider and model as the parent but
   26 -- gets its own context (no conversation history leakage). It cannot
   27 -- access the parent's session, vault, or harnesses.
   28 delegateTaskTool :: AgentEnv -> (ToolDefinition, ToolHandler)
   29 delegateTaskTool parentEnv = (def, handler)
   30   where
   31     def = ToolDefinition
   32       { _td_name        = "delegate_task"
   33       , _td_description = T.unlines
   34           [ "Spawn an isolated sub-agent to complete a focused task."
   35           , "The sub-agent has its own context (no conversation history leakage)"
   36           , "and a restricted toolset. It runs for up to max_turns (default: 10)"
   37           , "and returns a summary of its work."
   38           , "Use for: parallel subtasks, research, code generation, analysis."
   39           ]
   40       , _td_inputSchema = object
   41           [ "type" .= ("object" :: Text)
   42           , "properties" .= object
   43               [ "goal" .= object
   44                   [ "type" .= ("string" :: Text)
   45                   , "description" .= ("What the sub-agent should accomplish" :: Text)
   46                   ]
   47               , "context" .= object
   48                   [ "type" .= ("string" :: Text)
   49                   , "description" .= ("Additional context to include in the sub-agent's system prompt" :: Text)
   50                   ]
   51               , "tools" .= object
   52                   [ "type" .= ("array" :: Text)
   53                   , "items" .= object ["type" .= ("string" :: Text)]
   54                   , "description" .= ("Restrict sub-agent to these tool names (default: all parent tools except delegate_task)" :: Text)
   55                   ]
   56               , "max_turns" .= object
   57                   [ "type" .= ("integer" :: Text)
   58                   , "description" .= ("Maximum conversation turns (default: 10, max: 30)" :: Text)
   59                   ]
   60               ]
   61           , "required" .= (["goal"] :: [Text])
   62           ]
   63       }
   64 
   65     handler = ToolHandler $ \input ->
   66       case parseEither parseDelegateInput input of
   67         Left err -> pure (T.pack err, True)
   68         Right di -> runDelegate di
   69 
   70     runDelegate :: DelegateInput -> IO (Text, Bool)
   71     runDelegate di = do
   72       -- Check provider is available
   73       mProvider <- readIORef (_env_provider parentEnv)
   74       case mProvider of
   75         Nothing -> pure ("Cannot delegate: no provider configured", True)
   76         Just provider -> do
   77           mModel <- readIORef (_env_model parentEnv)
   78           case mModel of
   79             Nothing -> pure ("Cannot delegate: no model configured", True)
   80             Just model -> do
   81               let maxTurns = min 30 (fromMaybe 10 (_di_maxTurns di))
   82                   subRegistry = buildSubRegistry (_di_tools di)
   83                   sysPrompt = buildSubSystemPrompt (_di_goal di) (_di_context di)
   84               result <- try @SomeException $
   85                 runSubAgent provider model subRegistry sysPrompt (_di_goal di) maxTurns
   86               case result of
   87                 Left e -> pure ("delegate_task error: " <> T.pack (show e), True)
   88                 Right output -> pure (output, False)
   89 
   90     buildSubRegistry :: Maybe [Text] -> ToolRegistry
   91     buildSubRegistry Nothing =
   92       -- All parent tools except delegate_task (prevent recursive delegation)
   93       filterRegistry (/= "delegate_task") (_env_registry parentEnv)
   94     buildSubRegistry (Just names) =
   95       filterRegistry (\n -> n `elem` names && n /= "delegate_task") (_env_registry parentEnv)
   96 
   97     buildSubSystemPrompt :: Text -> Maybe Text -> Maybe Text
   98     buildSubSystemPrompt goal mContext =
   99       Just $ T.unlines $ filter (not . T.null)
  100         [ "You are a focused sub-agent. Complete the following task and report your results concisely."
  101         , ""
  102         , "TASK: " <> goal
  103         , maybe "" ("\nCONTEXT: " <>) mContext
  104         , ""
  105         , "When you have completed the task, provide a clear summary of what you did and the results."
  106         , "Do not ask clarifying questions — work with what you have."
  107         ]

  109 -- | Run a bounded sub-agent conversation.
  110 -- Returns the final assistant response text.
  111 runSubAgent
  112   :: SomeProvider
  113   -> ModelId
  114   -> ToolRegistry
  115   -> Maybe Text
  116   -> Text
  117   -> Int
  118   -> IO Text
  119 runSubAgent provider model registry sysPrompt goal maxTurns = do
  120   let ctx0 = addMessage (textMessage User goal) (emptyContext sysPrompt)
  121       tools = registryDefinitions registry
  122   go ctx0 tools maxTurns
  123   where
  124     go _ctx _tools 0 = pure "[Sub-agent reached maximum turn limit]"
  125     go ctx tools turnsLeft = do
  126       let req = CompletionRequest
  127             { _cr_model        = model
  128             , _cr_messages     = contextMessages ctx
  129             , _cr_systemPrompt = contextSystemPrompt ctx
  130             , _cr_maxTokens    = Just 4096
  131             , _cr_tools        = tools
  132             , _cr_toolChoice   = Nothing
  133             }
  134       resp <- complete provider req
  135       let calls = toolUseCalls resp
  136           text = responseText resp
  137           ctx' = addMessage (Message Assistant (_crsp_content resp)) ctx
  138       if null calls
  139         then pure text  -- Sub-agent is done
  140         else do
  141           -- Execute tool calls
  142           results <- mapM (executeSubCall registry) calls
  143           let resultMsg = toolResultMessage results
  144               ctx'' = addMessage resultMsg ctx'
  145           go ctx'' tools (turnsLeft - 1)
  146 
  147     executeSubCall :: ToolRegistry -> (ToolCallId, Text, Value) -> IO (ToolCallId, [ToolResultPart], Bool)
  148     executeSubCall reg (callId, name, input) = do
  149       result <- executeTool reg name input
  150       case result of
  151         Nothing -> pure (callId, [TRPText ("Unknown tool: " <> name)], True)
  152         Just (parts, isErr) -> pure (callId, parts, isErr)

  161 parseDelegateInput :: Value -> Parser DelegateInput
  162 parseDelegateInput = withObject "DelegateInput" $ \o ->
  163   DelegateInput
  164     <$> o .:  "goal"
  165     <*> o .:? "context"
  166     <*> o .:? "tools"
  167     <*> o .:? "max_turns"
