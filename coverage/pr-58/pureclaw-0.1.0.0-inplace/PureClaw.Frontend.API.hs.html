  140 -- | Per-origin WS subscriber counter. Lives at the same lifetime as the
  141 -- broker (constructed once in @startWithChannel@). The WS handler calls
  142 -- 'tryClaim' on upgrade (rejecting with 503 if the cap is reached) and
  143 -- 'release' on disconnect. Key is the normalized Origin string (see
  144 -- 'normalizeOrigin' in "PureClaw.Frontend.Stream").
  145 data StreamGuard = StreamGuard
  146   { _streamGuard_perOrigin    :: !(TVar (Map Text Int))
  147     -- ^ Live count per normalized Origin string.
  148   , _streamGuard_maxPerOrigin :: !Int
  149     -- ^ Maximum simultaneous WS subscriptions per origin.
  150   }

  152 -- | Construct a fresh 'StreamGuard' with the given per-origin cap.
  153 mkStreamGuard :: Int -> IO StreamGuard
  154 mkStreamGuard maxPer = do
  155   ref <- newTVarIO Map.empty
  156   pure StreamGuard
  157     { _streamGuard_perOrigin    = ref
  158     , _streamGuard_maxPerOrigin = maxPer
  159     }

  161 -- | Shared session-ID validation used by every endpoint that consumes a
  162 -- caller-supplied session id (HTTP @\/transcript@, @\/send@, @\/prompt@
  163 -- and the WS @focus@ op). Rejects @..@ and @\/@ to foreclose path
  164 -- traversal; rejects the empty string. Behavioural surface is intentionally
  165 -- the same across the HTTP and WS paths so D26's "shared helper" property
  166 -- holds: changing the rule here changes it everywhere.
  167 isValidSessionId :: Text -> Bool
  168 isValidSessionId sid
  169   | T.null sid = False
  170   | T.isInfixOf ".." sid = False
  171   | T.isInfixOf "/" sid = False
  172   | otherwise = True

  272 -- | WAI application handling @\/api\/*@ routes.
  273 apiApp :: FrontendEnv -> Application
  274 apiApp env req respond = do
  275   let method = requestMethod req
  276       path   = pathInfo req
  277   case (method, path) of
  278     ("GET", ["api", "harnesses"])            -> handleHarnesses env respond
  279     ("GET", ["api", "tabs"])                 -> handleListTabs env respond
  280     ("GET", ["api", "sessions", "recent"])   -> handleRecentSessions env respond
  281     ("GET", ["api", "sessions", "archived"]) -> handleArchivedSessions env respond
  282     ("GET", ["api", "sessions", sid, "transcript"]) ->
  283       handleTranscript env sid respond
  284     ("POST", ["api", "tabs", tidx, "close"]) ->
  285       handleCloseTab env tidx respond
  286     ("POST", ["api", "tabs", "new"]) ->
  287       handleNewTab env req respond
  288     ("POST", ["api", "sessions", "new"]) ->
  289       handleNewSessionGone respond
  290     ("POST", ["api", "sessions", sid, "send"]) ->
  291       handleSend env sid req respond
  292     ("PUT", ["api", "sessions", sid, "prompt"]) ->
  293       handleSetPrompt env sid req respond
  294     ("POST", ["api", "sessions", sid, "archive"]) ->
  295       handleSetArchived env sid True respond
  296     ("POST", ["api", "sessions", sid, "unarchive"]) ->
  297       handleSetArchived env sid False respond
  298     ("PUT", ["api", "sessions", sid, "description"]) ->
  299       handleSetDescription env sid req respond
  300     ("GET", ["api", "agents"])               -> handleAgents env respond
  301     ("GET", ["api", "providers"])            -> handleListProviders env respond
  302     ("GET", ["api", "providers", name, "models"]) ->
  303       handleListProviderModels env name respond
  304     _                                        -> respondNotFound respond

  312 -- | Probe a single harness to determine its activity.
  313 -- Checks process status first, then does a quick screen capture
  314 -- to distinguish thinking from idle.
  315 probeHarness :: (Text, HarnessHandle) -> IO HarnessInfo
  316 probeHarness (name, hh) = do
  317   st <- _hh_status hh
  318   activity <- case st of
  319     HarnessExited _ -> pure HarnessStopped
  320     HarnessRunning  -> probeActivity (_hh_session hh) name
  321   pure HarnessInfo
  322     { _hi_name     = name
  323     , _hi_activity = activity
  324     }

  326 -- | Capture the tmux window and check if the harness is idle or thinking.
  327 probeActivity :: Text -> Text -> IO HarnessActivity
  328 probeActivity tmuxSession windowName = do
  329   let target = tmuxSession <> ":" <> extractWindowIdx windowName
  330   result <- try @SomeException $ captureWindow target 50
  331   case result of
  332     Left _        -> pure HarnessIdle
  333     Right capture -> do
  334       let screenText = TE.decodeUtf8Lenient capture
  335       pure $ if isIdle screenText then HarnessIdle else HarnessThinking
  336 
  337 -- | Extract the window index suffix from a harness name like "claude-code-0".
  338 extractWindowIdx :: Text -> Text
  339 extractWindowIdx name =
  340   case T.splitOn "-" name of
  341     parts | length parts >= 2 -> last parts
  342     _                         -> "0"

  365 handleRecentSessions :: FrontendEnv -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  366 handleRecentSessions env respond = do
  367   let limit   = _fe_recentLimit env
  368       baseDir = _fe_sessionsDir env
  369   -- Over-fetch so the empty-transcript filter below can't shrink the
  370   -- visible count: listSessions already does a full directory scan and
  371   -- sorts the whole metadata set, so a generous bound is essentially
  372   -- free. Empty / archived sessions are dropped here, not erased on
  373   -- disk — archiving is purely a UI hint controlled by the user.
  374   metas    <- listSessions baseDir Nothing (limit * 3)
  375   -- Collect session IDs that are currently shown in an active tab so
  376   -- the same session does not appear in both "Active Tabs" and
  377   -- "Recent Sessions" simultaneously.
  378   tabs <- _fe_listTabs env
  379   let activeTabSids = [s | TabSnapshot { _ts_sessionId = Just s } <- tabs]
  380       notInTab m    = unSessionId (_sm_id m) `notElem` activeTabSids
  381       visible       = filter (\m -> not (_sm_archived m) && notInTab m) metas
  382   nonEmpty <- filterM (hasTranscriptEntries baseDir) visible
  383   let chosen = take limit nonEmpty
  384   -- Read a first-message snippet per session (bounded read per file).
  385   -- This is the cheap display fallback when no user description and no
  386   -- model-generated summary exist yet.
  387   snippets <- traverse (firstMessageSnippet baseDir) chosen
  388   let infos = zipWith toSessionInfo chosen snippets
  389   respond $ jsonResponse status200 infos

  391 -- | Return all archived sessions, sorted by last-active descending.
  392 handleArchivedSessions :: FrontendEnv -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  393 handleArchivedSessions env respond = do
  394   let baseDir = _fe_sessionsDir env
  395   -- Load all sessions with a generous limit; filter to archived-only.
  396   metas <- listSessions baseDir Nothing 1000
  397   let archived = filter _sm_archived metas
  398       infos = map (`toSessionInfo` Nothing) archived
  399   respond $ jsonResponse status200 infos

  405 -- | Compute the same sidebar payloads (tabs, recent sessions, archived
  406 -- sessions) that the three HTTP endpoints return, pre-serialized as a
  407 -- single JSON 'Value'. The WS handler sends this on connect and the
  408 -- mutation handlers call 'broadcastLists' to push it after every change.
  409 computeListsSnapshot :: FrontendEnv -> IO Aeson.Value
  410 computeListsSnapshot env = do
  411   let limit   = _fe_recentLimit env
  412       baseDir = _fe_sessionsDir env
  413   tabs <- _fe_listTabs env
  414   -- Recent sessions: exclude archived, exclude active-tab-backed, exclude
  415   -- empty transcripts, enrich with first-message snippet — same pipeline
  416   -- as handleRecentSessions.
  417   allMetas <- listSessions baseDir Nothing (limit * 3)
  418   let activeTabSids = [s | TabSnapshot { _ts_sessionId = Just s } <- tabs]
  419       notInTab m    = unSessionId (_sm_id m) `notElem` activeTabSids
  420       visible       = filter (\m -> not (_sm_archived m) && notInTab m) allMetas
  421   nonEmpty <- filterM (hasTranscriptEntries baseDir) visible
  422   let chosen = take limit nonEmpty
  423   snippets <- traverse (firstMessageSnippet baseDir) chosen
  424   let recentInfos = zipWith toSessionInfo chosen snippets
  425   -- Archived: all archived sessions, sorted by lastActive descending
  426   -- (listSessions already sorts; we just filter).
  427   archivedMetas <- listSessions baseDir Nothing 1000
  428   let archivedChosen = filter _sm_archived archivedMetas
  429   archivedSnippets <- traverse (firstMessageSnippet baseDir) archivedChosen
  430   let archivedInfos = zipWith toSessionInfo archivedChosen archivedSnippets
  431   pure $ object
  432     [ "type"              .= ("lists" :: Text)
  433     , "tabs"              .= tabs
  434     , "recentSessions"    .= recentInfos
  435     , "archivedSessions"  .= archivedInfos
  436     ]

  452 -- the **target** value (True for /archive, False for /unarchive), so

  465   case result of
  466     Right () -> do
  467       broadcastLists env

  470       respondNotFound respond

  474 -- | Update the user-provided description on a session.
  475 -- Body shape: @{"description": "..."}@ or @{"description": null}@ (clears).
  476 -- Trims whitespace; an all-whitespace string clears the field.
  477 handleSetDescription
  478   :: FrontendEnv
  479   -> Text
  480   -> Request
  481   -> (Response -> IO ResponseReceived)
  482   -> IO ResponseReceived
  483 handleSetDescription env sidText req respond = do
  484   body <- strictRequestBody req
  485   case Aeson.eitherDecode' body of
  486     Left err -> respond $ jsonResponse status400 (object ["error" .= T.pack err])
  487     Right (DescBody mDesc) -> do
  488       result <- setDescription (_fe_sessionsDir env) (SessionId sidText) mDesc
  489       case result of
  490         Right () -> do
  491           broadcastLists env
  492           respond $ jsonResponse status200 (object ["description" .= mDesc])
  493         Left SetDescriptionSessionMissing ->
  494           respondNotFound respond
  495         Left (SetDescriptionParseFailed msg) ->
  496           respond $ jsonResponse status500 (object ["error" .= msg])

  498 -- | Body parser that accepts @{"description": <string|null>}@; absent

  500 newtype DescBody = DescBody (Maybe Text)
  501 instance FromJSON DescBody where
  502   parseJSON = Aeson.withObject "DescBody" $ \o -> DescBody <$> o .:? "description"

  504 -- | Check whether a session has at least one transcript entry.
  505 hasTranscriptEntries :: FilePath -> SessionMeta -> IO Bool
  506 hasTranscriptEntries baseDir meta = do
  507   let path = baseDir </> T.unpack (unSessionId (_sm_id meta)) </> "transcript.jsonl"
  508   exists <- doesFileExist path
  509   if not exists
  510     then pure False
  511     else do
  512       size <- getFileSize path
  513       pure (size > 0)

  528 -- | Cheap fallback for display: read just the first line of the
  529 -- session's @transcript.jsonl@, decode it as a 'TranscriptEntry', and
  530 -- extract a short snippet of the first user message. Returns 'Nothing'
  531 -- when there's no transcript, the first entry isn't a request, or any
  532 -- decoding step fails. The returned string is at most
  533 -- 'snippetCharBudget' chars with newlines normalised to spaces.
  534 --
  535 -- Cost is one bounded read per call (we don't load the whole transcript).
  536 firstMessageSnippet :: FilePath -> SessionMeta -> IO (Maybe Text)
  537 firstMessageSnippet baseDir meta = do
  538   let path = baseDir </> T.unpack (unSessionId (_sm_id meta)) </> "transcript.jsonl"
  539   result <- try @IOException $ withFile path ReadMode BSC.hGetLine
  540   case result of
  541     Left _    -> pure Nothing
  542     Right line -> case Aeson.eitherDecodeStrict' line :: Either String TranscriptEntry of
  543       Left _ -> pure Nothing
  544       Right entry
  545         | _te_direction entry /= Request -> pure Nothing
  546         | otherwise -> pure (snippetFromPayload (_te_payload entry))
  547 
  548 snippetCharBudget :: Int
  549 snippetCharBudget = 120

  551 -- | Extract a display snippet from a request payload. Two shapes:
  552 -- (1) JSON object with a "messages" array (provider request) — pull
  553 --     text out of the first message; (2) plain text (harness send) —
  554 --     use the payload directly. Trimmed, newline-normalised, and
  555 --     truncated to 'snippetCharBudget' characters.
  556 snippetFromPayload :: Text -> Maybe Text
  557 snippetFromPayload raw = trimAndTruncate <$>
  558   case Aeson.decodeStrict (TE.encodeUtf8 raw) of
  559     Just (Aeson.Object o)
  560       | Just (Aeson.Array msgs) <- KM.lookup "messages" o
  561       , not (V.null msgs)
  562       -> messageText (V.unsafeHead msgs)
  563     _ -> Just raw
  564   where
  565     -- Anthropic-style message content: either a plain string or an
  566     -- array of {type:"text", text:"..."} blocks. We just want a
  567     -- human-readable lead.
  568     messageText :: Aeson.Value -> Maybe Text
  569     messageText (Aeson.Object m) = case KM.lookup "content" m of
  570       Just (Aeson.String s) -> Just s
  571       Just (Aeson.Array bs) -> Just (T.intercalate " " [t | Aeson.Object b <- V.toList bs
  572                                                           , Just (Aeson.String t) <- [KM.lookup "text" b]])
  573       _                     -> Nothing
  574     messageText _ = Nothing
  575 
  576     trimAndTruncate :: Text -> Text
  577     trimAndTruncate t =
  578       let normalized = T.unwords (T.words t)  -- collapse whitespace incl. newlines
  579       in  if T.length normalized > snippetCharBudget
  580             then T.take (snippetCharBudget - 1) normalized <> "\x2026"
  581             else normalized

  686 -- | Request body for creating a new tab.
  687 --
  688 -- @
  689 -- { "kind": { "tag": "session",
  690 --             "session_kind": { "tag": "provider",
  691 --                               "provider": "anthropic",

  695 -- OR:
  696 --
  697 -- @
  698 -- { "kind": { "tag": "raw_shell",
  699 --             "backend": { "tag": "local" } } }

  701 newtype NewTabRequest = NewTabRequest
  702   { _ntr_kind :: TabKind
  703   }

  705 instance FromJSON NewTabRequest where
  706   parseJSON = Aeson.withObject "NewTabRequest" $ \o -> do
  707     kindObj <- o .: "kind"
  708     tabKind <- parseTabKind kindObj
  709     pure (NewTabRequest tabKind)

  711 -- | Parse a 'TabKind' from the JSON envelope used by POST /api/tabs/new.
  712 --
  713 -- Two top-level tags:
  714 --
  715 --   * @\"session\"@ → expects a @\"session_kind\"@ sub-object whose
  716 --     shape is the existing 'SessionKind' JSON (tag-discriminated:
  717 --     @\"provider\"@ or @\"harness\"@).
  718 --   * @\"raw_shell\"@ → expects a @\"backend\"@ sub-object whose shape
  719 --     is the existing 'TerminalBackend' JSON (tag-discriminated).
  720 parseTabKind :: Aeson.Value -> AesonTypes.Parser TabKind
  721 parseTabKind = Aeson.withObject "TabKind" $ \o -> do
  722   tag <- o .: "tag" :: AesonTypes.Parser Text
  723   case tag of
  724     "session" -> do
  725       sk <- o .: "session_kind"
  726       TkSession <$> Aeson.parseJSON sk
  727     "raw_shell" -> do
  728       be <- o .: "backend"
  729       TkRawShell <$> Aeson.parseJSON be
  730     other -> fail ("unknown TabKind tag: " ++ T.unpack other)

  746 -- | Handle POST /api/tabs/new.
  747 --
  748 -- Creates a new tab (session-backed or raw shell), enforces the maxTabs
  749 -- limit (A8), and returns a JSON response with the tab index, session
  750 -- ID (if any), and kind description.
  751 handleNewTab :: FrontendEnv -> Request -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  752 handleNewTab env req respond = do
  753   body <- consumeBody req
  754   case Aeson.eitherDecode body of
  755     Left err ->
  756       respond $ jsonResponse status400
  757         (object ["error" .= ("Invalid JSON: " <> T.pack err)])
  758     Right (NewTabRequest tabKind) -> do
  759       -- A8: enforce maxTabs limit
  760       curCount <- readIORef (_fe_tabCount env)
  761       if curCount >= _fe_maxTabs env
  762         then respond $ jsonResponse status409
  763                (object ["error" .= ("maximum tab count reached" :: Text)])
  764         else createTab env tabKind respond
  765 
  766 -- | Actually create the tab and return a response.
  767 createTab :: FrontendEnv -> TabKind -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  768 createTab env tabKind respond = do
  769   curCount <- readIORef (_fe_tabCount env)
  770   -- Bump tab count
  771   writeIORef (_fe_tabCount env) (curCount + 1)
  772   case tabKind of
  773     TkSession sk -> do
  774       -- Create a session for session-backed tabs
  775       now <- getCurrentTime
  776       mModel <- readIORef (_fe_model env)
  777       let modelText = maybe "" unModelId mModel
  778           sid = newSessionId Nothing now
  779           -- Carry the agent name through from the request: the
  780           -- frontend uses it as the display name for the session
  781           -- (e.g. the chat input placeholder), and the agent's
  782           -- definition was already used to build the system prompt.
  783           -- Without this, the metadata loses the association and
  784           -- recent-sessions UI falls back to the session id.
  785           agentName = case sk of
  786             SkProvider ps -> _ps_agent ps
  787             SkHarness  _  -> Nothing
  788           meta = SessionMeta
  789             { _sm_id                = sid
  790             , _sm_agent             = agentName
  791             , _sm_kind              = sk
  792             , _sm_model             = modelText
  793             , _sm_channel           = "web"
  794             , _sm_createdAt         = now
  795             , _sm_lastActive        = now
  796             , _sm_bootstrapConsumed = True
  797             , _sm_archived          = False
  798             , _sm_description       = Nothing
  799             , _sm_autoSummary       = Nothing
  800             }
  801       sh <- mkSessionHandle (_fe_broker env) (_fe_logger env) (_fe_sessionsDir env) meta
  802       _sh_save sh
  803       -- Publish the new-session signal to the live stream broker (D18). The
  804       -- sidebar uses this to render the session row without polling. The
  805       -- no-broker path is intentional ('Nothing' preserves the legacy
  806       -- behaviour for one-off scripts and tests). Previously this lived in
  807       -- the now-410'd POST /api/sessions/new handler; it moved here when
  808       -- the session-tab unification merged on main.
  809       case _fe_broker env of
  810         Just broker ->
  811           _streamBroker_publish broker
  812             (ActivityChanged (_sm_id meta) (SaSessionCreated meta))
  813         Nothing -> pure ()
  814       broadcastLists env
  815       respond $ jsonResponse status200 NewTabResponse
  816         { _ntresp_tabIndex  = curCount
  817         , _ntresp_sessionId = Just (unSessionId sid)
  818         , _ntresp_kind      = tabKindLabel tabKind
  819         }
  820     TkRawShell _backend -> do
  821       respond $ jsonResponse status200 NewTabResponse
  822         { _ntresp_tabIndex  = curCount
  823         , _ntresp_sessionId = Nothing
  824         , _ntresp_kind      = tabKindLabel tabKind
  825         }

  830 tabKindLabel (TkSession (SkHarness _))  = "harness"
  831 tabKindLabel (TkRawShell _)             = "raw_shell"

  851 -- | Set or replace the custom prompt for a session, optionally updating
  852 -- the agent name in @session.json@.
  853 handleSetPrompt :: FrontendEnv -> Text -> Request -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  854 handleSetPrompt env sid req respond = do
  855   if not (isValidSessionId sid)
  856     then respond $ jsonResponse status400 (object ["error" .= ("Invalid session ID" :: Text)])
  857     else do
  858       let sessionDir = _fe_sessionsDir env </> T.unpack sid
  859           promptPath = sessionDir </> "custom-prompt.md"
  860           metaPath   = sessionDir </> "session.json"
  861       body <- consumeBody req
  862       case Aeson.eitherDecode body of
  863         Left _ ->
  864           respond $ jsonResponse status400 (object ["error" .= ("Invalid JSON: expected {\"prompt\": \"...\"}" :: Text)])
  865         Right obj -> case Map.lookup ("prompt" :: Text) (obj :: Map.Map Text Text) of
  866           Nothing ->
  867             respond $ jsonResponse status400 (object ["error" .= ("Missing 'prompt' field" :: Text)])
  868           Just prompt -> do
  869             TIO.writeFile promptPath prompt
  870             -- Update agent name in session metadata if a name was provided
  871             let mName = Map.lookup "name" obj
  872             case mName of
  873               Just name | not (T.null name) -> do
  874                 raw <- LBS.readFile metaPath
  875                 case Aeson.eitherDecode' raw of
  876                   Right meta -> do
  877                     let agentName = either (const Nothing) Just (mkAgentName name)
  878                         updated = (meta :: SessionMeta) { _sm_agent = agentName }
  879                         tmpP = metaPath <> ".tmp"
  880                     LBS.writeFile tmpP (Aeson.encode updated)
  881                     renameFile tmpP metaPath
  882                   Left _ -> pure ()
  883               _ -> pure ()
  884             respond $ jsonResponse status200 (object ["ok" .= True])

  893 -- | Send a user message to a session and get a completion.
  894 handleSend :: FrontendEnv -> Text -> Request -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  895 handleSend env sid req respond = do
  896   -- Validate session ID (shared helper — D26)
  897   if not (isValidSessionId sid)
  898     then respond $ jsonResponse status400 (object ["error" .= ("Invalid session ID" :: Text)])
  899     else do
  900       let sessionDir = _fe_sessionsDir env </> T.unpack sid
  901           transcriptPath = sessionDir </> "transcript.jsonl"
  902       exists <- doesFileExist transcriptPath
  903       if not exists
  904         then respond $ jsonResponse status404 (object ["error" .= ("Session not found" :: Text)])
  905         else do
  906           -- Parse request body
  907           body <- consumeBody req
  908           case Aeson.eitherDecode body of
  909             Left _ ->
  910               respond $ jsonResponse status400 (object ["error" .= ("Invalid JSON: expected {\"message\": \"...\"}" :: Text)])
  911             Right (SendRequest userText) -> do
  912               mProvider <- readIORef (_fe_provider env)
  913               mModel <- readIORef (_fe_model env)
  914               case (mProvider, mModel) of
  915                 (Nothing, _) ->
  916                   respond $ jsonResponse status503 (object ["error" .= ("No provider configured" :: Text)])
  917                 (_, Nothing) ->
  918                   respond $ jsonResponse status503 (object ["error" .= ("No model configured" :: Text)])
  919                 (Just provider, Just model) -> do
  920                   result <- try @SomeException $
  921                     doCompletion env (SessionId sid) provider model userText transcriptPath
  922                   case result of
  923                     Left e -> do
  924                       _lh_logError (_fe_logger env) $ "Send error: " <> T.pack (show e)
  925                       respond $ jsonResponse status500 (object ["error" .= ("Completion failed" :: Text)])
  926                     Right respText ->
  927                       respond $ jsonResponse status200 (object ["response" .= respText])

  929 -- | Run a completion: load context, send to provider, record to transcript.
  930 -- The transcript handle is opened via 'mkBroadcastingFileTranscriptHandle'
  931 -- so the provider-wrapper Request\/Response entries reach the broker (D25)
  932 -- when one is configured on the 'FrontendEnv'.
  933 doCompletion
  934   :: FrontendEnv
  935   -> SessionId
  936   -> SomeProvider
  937   -> ModelId
  938   -> Text
  939   -> FilePath
  940   -> IO Text
  941 doCompletion env sid provider model userText transcriptPath = do
  942   -- Open a (possibly broadcasting) transcript handle for recording.
  943   th <- mkBroadcastingFileTranscriptHandle
  944           (_fe_broker env) sid (_fe_logger env) transcriptPath
  945   -- Load recent messages for context
  946   history <- loadRecentMessages th 50 100000
  947   -- Check for per-session custom prompt, falling back to global
  948   let sessionDir = takeDirectory transcriptPath
  949       customPromptPath = sessionDir </> "custom-prompt.md"
  950   customExists <- doesFileExist customPromptPath
  951   systemPrompt <- if customExists
  952     then Just <$> TIO.readFile customPromptPath
  953     else pure (_fe_systemPrompt env)
  954   -- Build context with system prompt and history
  955   let ctx0 = foldl (flip addMessage) (emptyContext systemPrompt) history
  956       userMsg = textMessage User userText
  957       ctx = addMessage userMsg ctx0
  958       -- Wrap provider with transcript logging
  959       provider' = mkTranscriptProvider th (unModelId model) provider
  960       req = CompletionRequest
  961         { _cr_model        = model
  962         , _cr_messages     = contextMessages ctx
  963         , _cr_systemPrompt = contextSystemPrompt ctx
  964         , _cr_maxTokens    = Just 4096
  965         , _cr_tools        = []
  966         , _cr_toolChoice   = Nothing
  967         }
  968   -- Publish session-thinking activity events so the FE sidebar / chat
  969   -- area can show a live indicator while the provider call is in flight.
  970   -- For harness-backed sessions, the equivalent signal comes from the
  971   -- 2s tmux probe loop in 'PureClaw.Frontend.ActivityProbe'; provider
  972   -- sessions don't go through the harness map, so the only place we
  973   -- can hook this is at the request boundary in 'doCompletion'.
  974   -- 'bracket_' guarantees the Idle event fires even if 'complete' throws.
  975   let publishStatus s = case _fe_broker env of
  976         Just broker -> _streamBroker_publish broker
  977           (ActivityChanged sid (SaHarnessStatus s))
  978         Nothing -> pure ()
  979   resp <- bracket_
  980             (publishStatus HarnessThinking)
  981             (publishStatus HarnessIdle)
  982             (complete provider' req)
  983   _th_flush th
  984   _th_close th
  985   pure (responseText resp)

  987 -- | Consume the full request body.
  988 consumeBody :: Request -> IO LBS.ByteString

  990   where

  993       if BS.null chunk then pure [] else (chunk :) <$> go
