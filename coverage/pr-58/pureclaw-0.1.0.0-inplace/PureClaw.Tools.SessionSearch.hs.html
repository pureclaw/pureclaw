   25 -- | Create a session_search tool that searches across past session
   26 -- transcripts. Performs substring matching on transcript entries'
   27 -- payload text, returning matching entries grouped by session.
   28 sessionSearchTool :: LogHandle -> FilePath -> (ToolDefinition, ToolHandler)
   29 sessionSearchTool logger sessionsDir = (def, handler)
   30   where
   31     def = ToolDefinition
   32       { _td_name        = "session_search"
   33       , _td_description = T.unlines
   34           [ "Search across past session transcripts."
   35           , "Performs substring matching on message content."
   36           , "Returns matching entries grouped by session, most recent first."
   37           ]
   38       , _td_inputSchema = object
   39           [ "type" .= ("object" :: Text)
   40           , "properties" .= object
   41               [ "query" .= object
   42                   [ "type" .= ("string" :: Text)
   43                   , "description" .= ("Search query (substring match, case-insensitive)" :: Text)
   44                   ]
   45               , "limit" .= object
   46                   [ "type" .= ("integer" :: Text)
   47                   , "description" .= ("Maximum number of matching entries to return (default: 10)" :: Text)
   48                   ]
   49               ]
   50           , "required" .= (["query"] :: [Text])
   51           ]
   52       }
   53 
   54     handler = ToolHandler $ \input ->
   55       case parseEither parseSearchInput input of
   56         Left err -> pure (T.pack err, True)
   57         Right (query, limit) -> searchSessions query limit
   58 
   59     searchSessions :: Text -> Int -> IO (Text, Bool)
   60     searchSessions query limit = do
   61       result <- try @SomeException $ do
   62         sessionDirs <- listSessionDirs sessionsDir
   63         allMatches <- concat <$> mapM (searchOneSession query) sessionDirs
   64         let sorted = take limit (sortOn (Down . _sr_timestamp) allMatches)
   65         pure sorted
   66       case result of
   67         Left e -> pure ("session_search error: " <> T.pack (show e), True)
   68         Right [] -> pure ("No matches found for: " <> query, False)
   69         Right matches -> pure (formatMatches matches, False)
   70 
   71     searchOneSession :: Text -> FilePath -> IO [SearchResult]
   72     searchOneSession query sessionDir = do
   73       let transcriptPath = sessionDir </> "transcript.jsonl"
   74           metaPath = sessionDir </> "session.json"
   75       hasTranscript <- doesFileExist transcriptPath
   76       if not hasTranscript
   77         then pure []
   78         else do
   79           sessionId <- readSessionId metaPath
   80           th <- mkFileTranscriptHandle logger transcriptPath
   81           entries <- _th_query th emptyFilter
   82           _th_close th
   83           let queryLower = T.toLower query
   84               matching = filter (matchesQuery queryLower) entries
   85           pure [SearchResult
   86             { _sr_sessionId = sessionId
   87             , _sr_timestamp = _te_timestamp entry
   88             , _sr_direction = _te_direction entry
   89             , _sr_snippet   = extractSnippet query (_te_payload entry)
   90             } | entry <- matching]
   91 
   92     matchesQuery :: Text -> TranscriptEntry -> Bool
   93     matchesQuery queryLower entry =
   94       T.isInfixOf queryLower (T.toLower (_te_payload entry))
   95 
   96     readSessionId :: FilePath -> IO Text
   97     readSessionId metaPath = do
   98       exists <- doesFileExist metaPath
   99       if not exists
  100         then pure "(unknown)"
  101         else do
  102           result <- try @SomeException (decodeFileStrict metaPath)
  103           case result of
  104             Right (Just meta) -> pure (unSessionId (_sm_id (meta :: SessionMeta)))
  105             _ -> pure "(unknown)"
  106 
  107     listSessionDirs :: FilePath -> IO [FilePath]
  108     listSessionDirs dir = do
  109       entries <- try @SomeException (listDirectory dir)
  110       case entries of
  111         Left _ -> pure []
  112         Right names -> pure [dir </> n | n <- names]
  113 
  114 data SearchResult = SearchResult
  115   { _sr_sessionId :: Text
  116   , _sr_timestamp :: UTCTime
  117   , _sr_direction :: Direction
  118   , _sr_snippet   :: Text
  119   }
  120 
  121 formatMatches :: [SearchResult] -> Text
  122 formatMatches = T.intercalate "\n\n" . map formatMatch
  123 
  124 formatMatch :: SearchResult -> Text
  125 formatMatch sr =
  126   let dir = case _sr_direction sr of
  127         Request  -> "→"
  128         Response -> "←"
  129   in "[" <> _sr_sessionId sr <> "] " <> dir <> " " <> _sr_snippet sr
  130 
  131 extractSnippet :: Text -> Text -> Text
  132 extractSnippet query payload =
  133   let queryLower = T.toLower query
  134       payloadLower = T.toLower payload
  135       (before, _) = T.breakOn queryLower payloadLower
  136       pos = T.length before
  137       start = max 0 (pos - 50)
  138       snippet = T.take 250 (T.drop start payload)
  139       prefix = if start > 0 then "..." else ""
  140       suffix = if T.length payload > start + 250 then "..." else ""
  141   in prefix <> snippet <> suffix
  142 
  143 parseSearchInput :: Value -> Parser (Text, Int)
  144 parseSearchInput = withObject "SessionSearchInput" $ \o ->
  145   (,) <$> o .: "query" <*> o .:? "limit" .!= 10
