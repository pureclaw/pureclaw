    2 -- Module      : PureClaw.Agent.SlashDispatch
    3 -- Description : Pure classification of web-frontend slash input.
    4 --
    5 -- This module owns the PURE half of slash dispatch for the web
    6 -- frontend: 'classifyInput' maps raw user input to a 'SlashClass' by
    7 -- reusing the SAME routing parser the TUI and channels use
    8 -- ('PureClaw.Routing.Parse.parseInput'). Slash input is therefore
    9 -- classified identically across every front end, and recognised
   10 -- commands / short-circuit messages never reach the LLM.
   11 --
   12 -- The IO half ('runSlashInput') classifies input and executes a
   13 -- recognised command against a lazily-built scoped env, returning the
   14 -- captured output.
   15 module PureClaw.Agent.SlashDispatch
   16   ( SlashResult (..)
   17   , SlashClass (..)
   18   , classifyInput
   19   , runSlashInput
   20   , deferralMessage
   21   ) where

   35 -- | Outcome the transport acts on. 'SlashHandled' must NOT reach
   36 -- inference.
   37 data SlashResult
   38   = SlashHandled !Text      -- ^ short-circuit; this is the user-facing output
   39   | SlashPassThrough !Text  -- ^ ordinary chat; caller proceeds to inference
   40   deriving stock (Show, Eq)
   41 
   42 -- | Pure classification of raw input.
   43 data SlashClass
   44   = ClassPass !Text            -- ^ 'RT.Default': passthrough to inference
   45   | ClassMessage !Text         -- ^ Switch \/ Inject \/ ParseError: short-circuit text
   46   | ClassCommand !SlashCommand -- ^ a recognised command to execute
   47   deriving stock (Show, Eq)
   48 
   49 -- | Classify input using the SAME parser the TUI \/ channels use.
   50 --
   51 -- The 'RT.Switch' and 'RT.Inject' forms are tab-routing shapes that
   52 -- have no meaning in the web client, so they short-circuit to a
   53 -- friendly message rather than being executed. Parser failures are
   54 -- rendered with 'renderParseError'.
   55 classifyInput :: RT.RoutingConfig -> Text -> SlashClass
   56 classifyInput rc raw =
   57   case Parse.parseInput rc raw of
   58     Right (RT.Default t)        -> ClassPass t
   59     Right (RT.ParsedSlashCmd c) -> ClassCommand c
   60     Right (RT.Switch {})        -> ClassMessage switchMsg
   61     Right (RT.Inject {})        -> ClassMessage injectMsg
   62     Left err                    -> ClassMessage (renderParseError raw err)
   63   where
   64     switchMsg = "Tab switching isn't typed as /N in the web client — use the tab controls."
   65     injectMsg = "Cross-tab send isn't available from the web client yet."
   66 
   67 -- | User-friendly rendering of a routing 'RT.ParseError' for a browser
   68 -- bubble. Total over all 'RT.ParseError' constructors via the
   69 -- catch-all arm, which doubles as the "unknown command" message.
   70 renderParseError :: Text -> RT.ParseError -> Text
   71 renderParseError raw err = case err of
   72   RT.ParseErrorInvalidSessionId -> "That doesn't look like a valid session id. Try /tabs to list."
   73   RT.ParseErrorEmptyInput       -> "Empty command. Try /help."
   74   RT.ParseErrorIndexOutOfRange n ->
   75     "Tab " <> T.pack (show n) <> " doesn't exist. Try /tabs to list."
   76   _ -> "Unknown command: " <> firstWord <> ". Try /help."
   77   where
   78     firstWord = T.takeWhile (/= ' ') (T.stripStart raw)

   80 -- | Tracking issue for interactive-command support in the web UI. Embedded in
   81 -- the deferral message shown when a prompting command is invoked over the web.
   82 interactiveIssueUrl :: Text
   83 interactiveIssueUrl = "https://github.com/pureclaw/pureclaw/issues/84"
   84 
   85 -- | Execute one line of user input against a lazily-built scoped env.
   86 --
   87 -- @mkScoped@ builds (scoped env, output reader): the env's '_env_channel' is a
   88 -- capture channel and '_env_session' the target session. It is invoked ONLY for
   89 -- a recognized command, so passthrough chat and short-circuit messages pay no
   90 -- scoping cost.
   91 runSlashInput :: AgentEnv -> IO (AgentEnv, IO Text) -> Text -> IO SlashResult
   92 runSlashInput base mkScoped raw =
   93   case classifyInput (_env_routingConfig base) raw of
   94     ClassPass t      -> pure (SlashPassThrough t)
   95     ClassMessage t   -> pure (SlashHandled t)
   96     ClassCommand cmd -> do
   97       (scoped, readOut) <- mkScoped
   98       ctx <- buildContext scoped
   99       outcome <- try (executeSlashCommand scoped cmd ctx)
  100         :: IO (Either InteractiveUnsupported Context)
  101       case outcome of
  102         Right _ -> SlashHandled <$> readOut
  103         Left (InteractiveUnsupported _) ->
  104           SlashHandled . deferralMessage cmd <$> readOut
  105 
  106 -- | Build a Context from the scoped session's transcript so read-only commands
  107 -- (e.g. /status) report accurate counts. The Context returned by the command is
  108 -- discarded (output is transient).
  109 buildContext :: AgentEnv -> IO Context
  110 buildContext env = do
  111   tx <- envTranscript env
  112   history <- loadRecentMessages tx 50 100000
  113   pure (foldl (flip addMessage) (emptyContext Nothing) history)
  114 
  115 -- | Render the user-facing deferral text for an interactive command, preserving
  116 -- any output the command buffered before it tried to prompt.
  117 deferralMessage :: SlashCommand -> Text -> Text
  118 deferralMessage _cmd buffered =
  119   let note = "This command needs interactive input, which the web UI doesn't \
  120              \support yet (tracking: " <> interactiveIssueUrl <> "). Use the CLI for now."
  121    in if T.null buffered then note else buffered <> "\n" <> note
