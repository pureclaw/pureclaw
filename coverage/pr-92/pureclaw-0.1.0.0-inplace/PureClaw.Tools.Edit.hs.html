   25 -- | Create an edit tool that performs string replacement in files.
   26 -- Supports exact matching (default), fuzzy matching (fallback when exact
   27 -- match fails), and replace_all mode.
   28 editTool :: WorkspaceRoot -> FileHandle -> (ToolDefinition, ToolHandler)
   29 editTool root fh = (def, handler)
   30   where
   31     def = ToolDefinition
   32       { _td_name        = "edit"
   33       , _td_description = T.unlines
   34           [ "Replace a string in a file. By default, old_string must appear"
   35           , "exactly once (unique match). Set replace_all to true to replace"
   36           , "every occurrence. If an exact match fails, fuzzy matching is"
   37           , "attempted (normalizing whitespace and line endings)."
   38           ]
   39       , _td_inputSchema = object
   40           [ "type" .= ("object" :: Text)
   41           , "properties" .= object
   42               [ "path" .= object
   43                   [ "type" .= ("string" :: Text)
   44                   , "description" .= ("The file path relative to the workspace root" :: Text)
   45                   ]
   46               , "old_string" .= object
   47                   [ "type" .= ("string" :: Text)
   48                   , "description" .= ("The string to find and replace" :: Text)
   49                   ]
   50               , "new_string" .= object
   51                   [ "type" .= ("string" :: Text)
   52                   , "description" .= ("The replacement string" :: Text)
   53                   ]
   54               , "replace_all" .= object
   55                   [ "type" .= ("boolean" :: Text)
   56                   , "description" .= ("Replace all occurrences instead of requiring a unique match (default: false)" :: Text)
   57                   ]
   58               ]
   59           , "required" .= (["path", "old_string", "new_string"] :: [Text])
   60           ]
   61       }
   62 
   63     handler = ToolHandler $ \input ->
   64       case parseEither parseInput input of
   65         Left err -> pure (T.pack err, True)
   66         Right ei -> do
   67           pathResult <- mkSafePath root (T.unpack (_ei_path ei))
   68           case pathResult of
   69             Left pe -> pure (T.pack (show pe), True)
   70             Right sp -> do
   71               readResult <- try @SomeException (_fh_readFile fh sp)
   72               case readResult of
   73                 Left e -> pure (T.pack (show e), True)
   74                 Right bs -> case TE.decodeUtf8' bs of
   75                   Left _ -> pure ("Cannot edit binary file", True)
   76                   Right content -> applyEdit ei sp content
   77 
   78     applyEdit :: EditInput -> SafePath -> Text -> IO (Text, Bool)
   79     applyEdit ei sp content
   80       | _ei_replaceAll ei = doReplaceAll ei sp content
   81       | otherwise         = doUnique ei sp content
   82 
   83     doReplaceAll :: EditInput -> SafePath -> Text -> IO (Text, Bool)
   84     doReplaceAll ei sp content = do
   85       let count = countOccurrences (_ei_oldString ei) content
   86       if count == 0
   87         then tryFuzzyReplaceAll ei sp content
   88         else do
   89           let newContent = replaceAll (_ei_oldString ei) (_ei_newString ei) content
   90           writeResult <- try @SomeException
   91             (_fh_writeFile fh sp (TE.encodeUtf8 newContent))
   92           case writeResult of
   93             Left e -> pure (T.pack (show e), True)
   94             Right () -> pure ("Replaced " <> T.pack (show count) <> " occurrence(s) in " <> _ei_path ei, False)
   95 
   96     doUnique :: EditInput -> SafePath -> Text -> IO (Text, Bool)
   97     doUnique ei sp content = do
   98       let count = countOccurrences (_ei_oldString ei) content
   99       case count of
  100         0 -> tryFuzzy ei sp content
  101         1 -> do
  102           let newContent = replaceFirst (_ei_oldString ei) (_ei_newString ei) content
  103           writeResult <- try @SomeException
  104             (_fh_writeFile fh sp (TE.encodeUtf8 newContent))
  105           case writeResult of
  106             Left e -> pure (T.pack (show e), True)
  107             Right () -> pure ("Edited " <> _ei_path ei, False)
  108         n -> pure ("old_string not unique in " <> _ei_path ei
  109                    <> " (" <> T.pack (show n) <> " occurrences). Use replace_all: true to replace all.", True)
  110 
  111     tryFuzzy :: EditInput -> SafePath -> Text -> IO (Text, Bool)
  112     tryFuzzy ei sp content =
  113       case fuzzyFind (_ei_oldString ei) content of
  114         NoFuzzyMatch ->
  115           pure ("old_string not found in " <> _ei_path ei, True)
  116         AmbiguousFuzzy n strategy ->
  117           pure ("old_string not unique via " <> strategy <> " matching in " <> _ei_path ei
  118                 <> " (" <> T.pack (show n) <> " occurrences)", True)
  119         UniqueFuzzy matched strategy -> do
  120           let newContent = replaceFirst matched (_ei_newString ei) content
  121           writeResult <- try @SomeException
  122             (_fh_writeFile fh sp (TE.encodeUtf8 newContent))
  123           case writeResult of
  124             Left e -> pure (T.pack (show e), True)
  125             Right () -> pure ("Edited " <> _ei_path ei <> " (matched via " <> strategy <> ")", False)
  126 
  127     tryFuzzyReplaceAll :: EditInput -> SafePath -> Text -> IO (Text, Bool)
  128     tryFuzzyReplaceAll ei sp content =
  129       case fuzzyFind (_ei_oldString ei) content of
  130         NoFuzzyMatch ->
  131           pure ("old_string not found in " <> _ei_path ei, True)
  132         AmbiguousFuzzy n strategy -> do
  133           -- For replace_all with fuzzy, ambiguous is fine — replace them all
  134           let matched = fuzzyFindPattern (_ei_oldString ei) strategy
  135           case matched of
  136             Nothing -> pure ("old_string not found in " <> _ei_path ei, True)
  137             Just pat -> do
  138               let newContent = replaceAll pat (_ei_newString ei) content
  139               writeResult <- try @SomeException
  140                 (_fh_writeFile fh sp (TE.encodeUtf8 newContent))
  141               case writeResult of
  142                 Left e -> pure (T.pack (show e), True)
  143                 Right () -> pure ("Replaced " <> T.pack (show n)
  144                                   <> " occurrence(s) in " <> _ei_path ei
  145                                   <> " (matched via " <> strategy <> ")", False)
  146         UniqueFuzzy matched strategy -> do
  147           let newContent = replaceFirst matched (_ei_newString ei) content
  148           writeResult <- try @SomeException
  149             (_fh_writeFile fh sp (TE.encodeUtf8 newContent))
  150           case writeResult of
  151             Left e -> pure (T.pack (show e), True)
  152             Right () -> pure ("Replaced 1 occurrence(s) in " <> _ei_path ei
  153                               <> " (matched via " <> strategy <> ")", False)
  154 
  155     parseInput :: Value -> Parser EditInput
  156     parseInput = withObject "EditInput" $ \o ->
  157       EditInput
  158         <$> o .:  "path"
  159         <*> o .:  "old_string"
  160         <*> o .:  "new_string"
  161         <*> o .:? "replace_all" .!= False

  163 data EditInput = EditInput
  164   { _ei_path       :: Text
  165   , _ei_oldString  :: Text
  166   , _ei_newString  :: Text
  167   , _ei_replaceAll :: Bool
  168   }

  170 -- | Result of a fuzzy search.
  171 data FuzzyMatch
  172   = NoFuzzyMatch
  173     -- ^ No match found by any strategy.
  174   | UniqueFuzzy Text Text
  175     -- ^ Unique match found. Fields: matched text, strategy name.
  176   | AmbiguousFuzzy Int Text
  177     -- ^ Multiple matches found. Fields: count, strategy name.
  178   deriving stock (Show, Eq)
  179 
  180 -- | Try fuzzy matching strategies in priority order.
  181 -- Each strategy normalizes the needle and haystack differently,
  182 -- then searches for the normalized needle in the normalized haystack.
  183 fuzzyFind :: Text -> Text -> FuzzyMatch
  184 fuzzyFind needle haystack = tryStrategies strategies
  185   where
  186     strategies =
  187       [ ("whitespace-normalized", normalizeWhitespace)
  188       , ("trimmed-lines", trimLines)
  189       , ("line-ending-normalized", normalizeLineEndings)
  190       , ("indentation-normalized", normalizeIndentation)
  191       , ("case-insensitive", T.toLower)
  192       ]
  193 
  194     tryStrategies :: [(Text, Text -> Text)] -> FuzzyMatch
  195     tryStrategies [] = NoFuzzyMatch
  196     tryStrategies ((name, normalize):rest) =
  197       let normNeedle   = normalize needle
  198           normHaystack = normalize haystack
  199           count = countOccurrences normNeedle normHaystack
  200       in case count of
  201         0 -> tryStrategies rest
  202         1 -> case findOriginal normalize needle haystack of
  203           Nothing   -> tryStrategies rest
  204           Just orig -> UniqueFuzzy orig name
  205         n -> AmbiguousFuzzy n name
  206 
  207 -- | Given a normalization function and the original needle and haystack,
  208 -- find the original text in the haystack that corresponds to the
  209 -- normalized match.
  210 findOriginal :: (Text -> Text) -> Text -> Text -> Maybe Text
  211 findOriginal normalize needle haystack =
  212   let normNeedle = normalize needle
  213       normLen = T.length normNeedle
  214       normH = normalize haystack
  215       -- Find the position in the normalized haystack
  216       (before, _) = T.breakOn normNeedle normH
  217       normPos = T.length before
  218       -- Map back to original: scan original haystack, tracking
  219       -- normalized character count to find the start position
  220   in mapBackToOriginal normalize normPos normLen haystack
  221 
  222 -- | Map a position in the normalized text back to the original text.
  223 -- Returns the original substring that normalizes to the match.
  224 mapBackToOriginal :: (Text -> Text) -> Int -> Int -> Text -> Maybe Text
  225 mapBackToOriginal normalize normPos normLen original =
  226   -- Strategy: try progressively longer substrings at each starting
  227   -- position until we find one whose normalization contains our match.
  228   -- This is O(n²) in the worst case but files are bounded in practice.
  229   let origLen = T.length original
  230       candidates =
  231         [ T.take len (T.drop start original)
  232         | start <- [0..origLen - 1]
  233         , len   <- [1..origLen - start]
  234         , let normed = normalize (T.take len (T.drop start original))
  235         , T.length normed >= normLen
  236         , let normBefore = normalize (T.take start original)
  237         , T.length normBefore == normPos
  238         , normed == T.take normLen (T.drop normPos (normalize original))
  239         ]
  240   in case candidates of
  241     (c:_) -> Just c
  242     []    -> Nothing

  244 -- | Get the fuzzy pattern text for a strategy (used by replace_all).
  245 fuzzyFindPattern :: Text -> Text -> Maybe Text
  246 fuzzyFindPattern needle strategy =
  247   let normalize = case strategy of
  248         "whitespace-normalized"   -> normalizeWhitespace
  249         "trimmed-lines"           -> trimLines
  250         "line-ending-normalized"  -> normalizeLineEndings
  251         "indentation-normalized"  -> normalizeIndentation
  252         "case-insensitive"        -> T.toLower
  253         _                         -> id
  254   in Just (normalize needle)
  255 
  256 -- | Normalize all whitespace runs to single spaces, trim.
  257 normalizeWhitespace :: Text -> Text
  258 normalizeWhitespace = T.unwords . T.words
  259 
  260 -- | Trim leading/trailing whitespace from each line.
  261 trimLines :: Text -> Text
  262 trimLines = T.unlines . map T.strip . T.lines
  263 
  264 -- | Normalize line endings: \r\n → \n
  265 normalizeLineEndings :: Text -> Text
  266 normalizeLineEndings = T.replace "\r\n" "\n"
  267 
  268 -- | Normalize indentation: strip all leading whitespace from each line.
  269 -- This handles mismatched tabs-vs-spaces and indent-level differences.
  270 normalizeIndentation :: Text -> Text
  271 normalizeIndentation = T.unlines . map T.stripStart . T.lines
  272 
  273 -- | Count non-overlapping occurrences of a needle in a haystack.
  274 countOccurrences :: Text -> Text -> Int
  275 countOccurrences needle haystack
  276   | T.null needle = 0
  277   | otherwise     = go 0 haystack
  278   where
  279     go !n remaining =
  280       case T.breakOn needle remaining of
  281         (_, after)
  282           | T.null after -> n
  283           | otherwise    -> go (n + 1) (T.drop (T.length needle) after)
  284 
  285 -- | Replace the first occurrence of needle with replacement.
  286 replaceFirst :: Text -> Text -> Text -> Text
  287 replaceFirst needle replacement haystack =
  288   let (before, after) = T.breakOn needle haystack
  289   in before <> replacement <> T.drop (T.length needle) after
  290 
  291 -- | Replace all non-overlapping occurrences of needle with replacement.
  292 replaceAll :: Text -> Text -> Text -> Text
  293 replaceAll needle replacement haystack
  294   | T.null needle = haystack
  295   | otherwise     = T.intercalate replacement (T.splitOn needle haystack)
