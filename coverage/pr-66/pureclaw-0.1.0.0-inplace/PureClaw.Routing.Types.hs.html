   76 -- | Routing-layer configuration, loaded once at process start.
   77 --
   78 -- All caps are enforced by various WUs: '_rc_maxTabs' by the parser
   79 -- (WU2) and the spawn machinery (WU5\/WU9); '_rc_inputQueueBound' by
   80 -- the AI tab factory (WU6); '_rc_channelOutQBound' by AgentEnv
   81 -- construction (WU3); '_rc_spawnRateLimit' \/ '_rc_maxConcurrentActive'
   82 -- \/ '_rc_maxNameLen' by their respective S-series enforcement points.
   83 --
   84 -- Runtime mutation via @\/config@ is v1.5+.
   85 data RoutingConfig = RoutingConfig
   86   { _rc_defaultKind         :: !Tab.TabKind
   87     -- ^ Pre-shipped as @'Tab.TkSession' ('SkProvider' ...)@.
   88   , _rc_defaultAi           :: !AiDefaults
   89   , _rc_defaultShell        :: !ShellDefaults
   90   , _rc_switchRecap         :: !Int
   91     -- ^ Default 3 recent messages emitted on @\/N@ switch.
   92   , _rc_maxTabs             :: !Int
   93     -- ^ Default 36, matching the single-char @[0-9a-z]@ index
   94     --   alphabet of the @\/N@ routing grammar (10 digits + 26
   95     --   lowercase letters). Parser bounds-checks against this;
   96     --   dispatcher refuses to spawn beyond this cap with
   97     --   'Tab.TabLimitExceeded'. Configuring a value above 36 is
   98     --   permitted but only the first 36 slots are addressable
   99     --   through the @\/N@ shortcut; the higher slots remain
  100     --   reachable through @\/tab focus N@ and the dashboard.
  101   , _rc_inputQueueBound     :: !Int
  102     -- ^ Default 64. Per-tab @TBQueue InputEvent@ capacity.
  103   , _rc_channelOutQBound    :: !Int
  104     -- ^ Default 1024. Process-wide @TBQueue (OutputSource, ChannelEvent)@
  105     --   capacity.
  106   , _rc_spawnRateLimit      :: !Int
  107     -- ^ Default 10 spawns per minute per chat user.
  108   , _rc_maxConcurrentActive :: !Int
  109     -- ^ Default 4. S9 cap on tabs in 'Tab.Active' status simultaneously.
  110   , _rc_maxNameLen          :: !Int
  111     -- ^ Default 32. S10 length cap on tab names.
  112   , _rc_sshIdentityKey      :: !Text
  113     -- ^ Default @"default-ssh-key"@. Vault slot from which the ssh
  114     --   identity is loaded (no inline identities permitted per S4).
  115   , _rc_maxPureClawDepth   :: !Int
  116     -- ^ Maximum recursion depth for HPureClaw harness sessions.
  117     --   Default 2. Spawning is refused when
  118     --   @'_rc_pureClawDepth' >= '_rc_maxPureClawDepth'@.
  119   , _rc_pureClawDepth      :: !Int
  120     -- ^ Current recursion depth. 0 at top level. Incremented via
  121     --   the @--depth@ CLI flag when PureClaw spawns a child process.
  122   }
  123   deriving stock (Eq, Show)

  126 -- ---------------------------------------------------------------------------
  127 -- Spawn errors (depth limiting)
  128 -- ---------------------------------------------------------------------------
  129 
  130 -- | Errors that prevent spawning a new HPureClaw harness session.
  131 data SpawnError
  132   = MaxPureClawDepthExceeded !Int !Int
  133     -- ^ @MaxPureClawDepthExceeded current max@ — the current recursion
  134     --   depth equals or exceeds the configured maximum.
  135   deriving stock (Show, Eq)
  136 
  137 -- | Pure guard that checks whether the current recursion depth allows
  138 -- spawning another HPureClaw session. Returns @'Left'
  139 -- 'MaxPureClawDepthExceeded'@ when @'_rc_pureClawDepth' >=
  140 -- '_rc_maxPureClawDepth'@.
  141 checkPureClawDepth :: RoutingConfig -> Either SpawnError ()
  142 checkPureClawDepth cfg
  143   | _rc_pureClawDepth cfg >= _rc_maxPureClawDepth cfg =
  144       Left (MaxPureClawDepthExceeded (_rc_pureClawDepth cfg) (_rc_maxPureClawDepth cfg))
  145   | otherwise = Right ()

  148 -- ---------------------------------------------------------------------------
  149 -- Errors
  150 -- ---------------------------------------------------------------------------
  151 
  152 -- | Parser-level errors produced by @parseInput@ (WU2).
  153 --
  154 -- Constructors carry only bounded primitives — no free 'Text', no raw
  155 -- input — so the redacted-error contract is preserved when these
  156 -- bubble up into 'RoutingError'.
  157 --
  158 -- The legacy 'ParseErrorLeadingZero' code was removed when the
  159 -- routing grammar switched from multi-digit indices to single-char
  160 -- @[0-9a-z]@ indices — multi-char shapes (@\/01@, @\/12@, @\/1a@)
  161 -- now surface as 'ParseErrorMalformed'.
  162 data ParseError
  163   = ParseErrorIndexOutOfRange !Int
  164     -- ^ Parsed index @>= _rc_maxTabs@.
  165   | ParseErrorInvalidSessionId
  166     -- ^ @\/tab resume \<id\>@ where @\<id\>@ contains @\/@, @\\@,
  167     --   @..@, NUL, or any character outside @[a-zA-Z0-9_-]@.
  168   | ParseErrorEmptyInput
  169     -- ^ Empty or whitespace-only input.
  170   | ParseErrorMalformed
  171     -- ^ Catch-all for shapes that match no production in the grammar.
  172   deriving stock (Eq, Show)
  173 
  174 -- | Routing-layer errors. Wraps 'ParseError' for parser failures and
  175 -- carries higher-level dispatcher\/registry failures.
  176 data RoutingError
  177   = RoutingParseError !ParseError
  178   | RoutingTabError !Tab.TabError
  179     -- ^ Bubbled up from the tab layer (e.g. spawn failure).
  180   deriving stock (Eq, Show)

  183 -- ---------------------------------------------------------------------------
  184 -- ParsedInput
  185 -- ---------------------------------------------------------------------------
  186 
  187 -- | The classification @parseInput@ produces (WU2).
  188 --
  189 -- The 'Switch' \/ 'Inject' \/ 'ParsedSlashCmd' branches do NOT reach
  190 -- a provider — that's the LLM-free invariant P18. Only 'Default' text
  191 -- ever feeds a provider call.
  192 --
  193 -- Note on the constructor name 'ParsedSlashCmd' (vs the design doc's
  194 -- @SlashCmd@): Haskell does not permit two constructors with the same
  195 -- name in the same module, and 'InputEvent' already owns 'SlashCmd'.
  196 -- Both carry the same 'Slash.SlashCommand' payload; the dispatcher
  197 -- maps @ParsedSlashCmd cmd@ to @SlashCmd cmd@ when enqueueing onto a
  198 -- tab's input queue.
  199 data ParsedInput
  200   = Switch !Tab.TabIndex
  201     -- ^ @\/N@ — focus the tab at index @N@ (or auto-spawn per A-series).
  202   | Inject !Tab.TabIndex !Text
  203     -- ^ @\/N \<payload\>@ — enqueue payload on tab @N@ without
  204     --   changing focus.
  205   | Default !Text
  206     -- ^ Plain text — routed to the current focused tab.
  207   | ParsedSlashCmd !Slash.SlashCommand
  208     -- ^ Existing slash command (or the new @\/tab*@ family) parsed by
  209     --   the shared slash-command grammar.
  210   deriving stock (Eq, Show)

  213 -- ---------------------------------------------------------------------------
  214 -- InputEvent
  215 -- ---------------------------------------------------------------------------
  216 
  217 -- | One event in a tab's per-tab input queue.
  218 --
  219 -- The dispatcher (WU5) enqueues 'UserText' for free-form text and
  220 -- 'SlashCmd' for Context-mutating commands (per E5). Each tab loop
  221 -- (WU6 for 'Tab.KindAi') is the sole consumer.
  222 data InputEvent
  223   = UserText !Text
  224   | SlashCmd !Slash.SlashCommand
  225   deriving stock (Eq, Show)

  228 -- ---------------------------------------------------------------------------
  229 -- Channel output
  230 -- ---------------------------------------------------------------------------
  231 
  232 -- | Origin of a 'ChannelEvent' on the channel-out queue.
  233 --
  234 -- The single writer thread reads 'OutputSource' on dequeue and uses
  235 -- it as the focus gate: 'SrcDispatcher' events emit unconditionally
  236 -- ('PureClaw.Handles.Channel.ChannelHandle' always sees them), while
  237 -- 'SrcTab' events are dropped when '_env_focus' is not @Just n@.
  238 data OutputSource
  239   = SrcDispatcher
  240   | SrcTab !Tab.TabIndex
  241   deriving stock (Eq, Show)
  242 
  243 -- | Identifier for one logical multi-chunk message on the channel.
  244 --
  245 -- The data constructor is intentionally NOT exported; obtain values
  246 -- via 'mkStreamId' (a monotonically-increasing allocator lands in WU4
  247 -- alongside the writer thread). The 'Word64' carrier is large enough
  248 -- that wrap-around is not a practical concern.
  249 newtype StreamId = StreamId { unStreamId :: Word64 }
  250   deriving stock (Eq, Ord, Show)
  251 
  252 -- | The only way to construct a 'StreamId'. Callers (typically the
  253 -- 'PureClaw.Routing.ChannelOut' allocator) bump a 'Word64' counter and
  254 -- pass the result here.
  255 mkStreamId :: Word64 -> StreamId
  256 mkStreamId = StreamId
  257 
  258 -- | Events that flow through @_env_channelOutQ@ (the single bounded
  259 -- queue feeding the writer thread).
  260 --
  261 -- AI tab loops emit @'StreamStart' sid n@ once per logical message,
  262 -- then a sequence of @'ChunkOf' sid chunk@, then @'StreamEnd' sid@.
  263 -- Non-AI tab loops (KindShell\/Ssh\/Tmux\/Harness) emit one
  264 -- @'FullMsg' n payload@ per @_bh_recv@\/harness-recv return — no
  265 -- streaming, no breadcrumb (per D5 backend rule).
  266 --
  267 -- 'BannerLine' is reserved for one-shot dispatcher messages: switch
  268 -- confirmations, dashboards, command errors, and mid-stream
  269 -- breadcrumbs.
  270 data ChannelEvent
  271   = StreamStart !StreamId !Tab.TabIndex
  272   | ChunkOf !StreamId !Text
  273   | StreamEnd !StreamId
  274   | FullMsg !Tab.TabIndex !Text
  275   | BannerLine !Text
  276   deriving stock (Eq, Show)
