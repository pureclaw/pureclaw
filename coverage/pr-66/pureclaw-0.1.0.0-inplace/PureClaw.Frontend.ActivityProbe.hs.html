    1 -- | Activity probe loop — WU4 of the live-transcript-streaming feature.
    2 --
    3 -- The loop walks the process-wide @IORef (Map Text HarnessHandle)@ every
    4 -- two seconds, derives each harness's current 'HarnessActivity' via the
    5 -- same 'probeHarness' logic the HTTP API uses, and publishes one
    6 -- 'ActivityChanged' event per state /transition/ (not per probe tick) to
    7 -- the 'StreamBroker'.
    8 --
    9 -- /First-tick semantics./ The very first probe establishes the baseline;
   10 -- zero events are emitted on that tick. From the second tick onward only
   11 -- harnesses whose state differs from the previous snapshot produce events.
   12 -- This is the v1 source of truth for 'SaHarnessStatus' events — see
   13 -- @docs\/transcript-streaming.md@ §"Activity event source" and DoD D17.
   14 --
   15 -- /Lifecycle./ The loop is forked under 'Async.withAsync' alongside the WAI
   16 -- server in @startWithChannel@ ("PureClaw.CLI.Commands"). When the
   17 -- enclosing 'Async' scope exits, 'AsyncCancelled' is delivered; the
   18 -- exception handler /re-raises/ it (per project memory) so bracket-style
   19 -- cleanup runs. Any other 'SomeException' is logged via '_lh_logError' and
   20 -- the loop terminates — v1 prefers "loud failure" to silent restart (D24,
   21 -- @docs\/transcript-streaming.md@ §"Lifecycle and Shutdown").
   22 --
   23 -- /Test seam./ 'runActivityProbeLoopWith' exposes the tick interval and the
   24 -- probe action so unit tests can drive the loop deterministically. The
   25 -- production entry point 'runActivityProbeLoop' supplies the standard wiring
   26 -- — a 2-second tick and the harness-map snapshot via 'probeHarness'.
   27 module PureClaw.Frontend.ActivityProbe
   28   ( runActivityProbeLoop
   29   , runActivityProbeLoopWith
   30   ) where

   59 -- | Run the activity probe loop with the production tick interval (2 s) and
   60 -- the standard harness-snapshot action. The harness snapshot mirrors the
   61 -- semantics of 'PureClaw.Frontend.API.probeHarness': for each harness in
   62 -- the map, it derives 'HarnessActivity' from process status + a tmux
   63 -- capture. See 'runActivityProbeLoopWith' for the test seam.
   64 runActivityProbeLoop
   65   :: StreamBroker
   66   -> IORef (Map Text HarnessHandle)
   67   -> LogHandle
   68   -> IO ()
   69 runActivityProbeLoop broker harnessesRef =
   70   runActivityProbeLoopWith
   71     defaultTickMicros
   72     (snapshotHarnesses harnessesRef)
   73     broker
   74 
   75 -- | Production tick interval: 2 seconds, matching @docs\/transcript-streaming.md@
   76 -- §"Activity event source" (sole source of 'SaHarnessStatus' events; the
   77 -- 2 s lag is the documented v1 tradeoff).
   78 defaultTickMicros :: Int
   79 defaultTickMicros = 2_000_000
   80 
   81 -- ---------------------------------------------------------------------------
   82 -- Test-friendly entry point
   83 -- ---------------------------------------------------------------------------
   84 
   85 -- | The probe loop with explicit tick interval and a caller-supplied probe
   86 -- action. Unit tests pass a controllable probe (e.g. reading a mutable
   87 -- 'IORef') so the loop is deterministic without forking tmux. The
   88 -- @probe@ action MUST be exception-safe; the loop catches any
   89 -- 'SomeException' except 'AsyncCancelled', logs it, and terminates (no
   90 -- restart in v1).
   91 runActivityProbeLoopWith
   92   :: Int
   93   -- ^ Tick interval in microseconds.
   94   -> IO (Map Text HarnessActivity)
   95   -- ^ Probe action: returns the current per-harness activity snapshot.
   96   -> StreamBroker
   97   -> LogHandle
   98   -> IO ()
   99 runActivityProbeLoopWith tickMicros probe broker logger = handle outer $ do
  100   -- First tick: establish the baseline, emit nothing. The sleep happens
  101   -- BEFORE the probe so a test that cancels immediately doesn't race the
  102   -- first probe (and so D17's "no events on the first tick" holds even if
  103   -- the harness map is changing concurrently).
  104   threadDelay tickMicros
  105   initial <- probe
  106   lastStatesRef <- newIORef initial
  107   forever $ do
  108     threadDelay tickMicros
  109     next <- probe
  110     prev <- readIORef lastStatesRef
  111     -- Emit one event per harness whose state changed since the previous
  112     -- snapshot. @Map.differenceWith f m1 m2@ keeps every key of @m1@: if
  113     -- @m2@ also has the key, @f new old@ decides whether to keep
  114     -- (@Just new@) or drop (@Nothing@); if @m2@ does not, the entry is
  115     -- kept as-is. So this single call captures both "transitioned"
  116     -- harnesses and newly-appeared harnesses (first-observed state).
  117     -- Harnesses that /disappear/ between ticks emit no event — their
  118     -- absence is a separable v1.5 concern.
  119     let transitions :: Map Text HarnessActivity
  120         transitions =
  121           Map.differenceWith
  122             (\new old -> if new == old then Nothing else Just new)
  123             next
  124             prev
  125     forM_ (Map.toList transitions) $ \(name, status) ->
  126       _streamBroker_publish broker
  127         (ActivityChanged (sessionIdFromHarnessName name) (SaHarnessStatus status))
  128     writeIORef lastStatesRef next
  129   where
  130     -- AsyncCancelled MUST be re-raised so 'Async.withAsync' / 'Async.cancel'
  131     -- can complete cleanly (project-wide invariant; see
  132     -- 'PureClaw.Frontend.BroadcastingTranscript' for the same pattern).
  133     outer :: SomeException -> IO ()
  134     outer e
  135       | Just AsyncCancelled <- fromException e = throwIO e
  136       | otherwise = _lh_logError logger
  137           ("activity-probe crashed: " <> T.pack (show e))

  143 -- | Convert a harness-map key (the harness "name", e.g. @claude-code-0@) to
  144 -- a 'SessionId'. The activity events flow over the wire keyed by
  145 -- 'SessionId'; the harness map is keyed by 'Text'. v1 uses the harness
  146 -- name as the session id verbatim — frontends that need to map "harness
  147 -- name" back to "session id" can do so client-side. A richer mapping (e.g.
  148 -- looking up '_hh_session' on the handle) is a follow-up if real session
  149 -- ids ever diverge from harness names.
  150 sessionIdFromHarnessName :: Text -> SessionId
  151 sessionIdFromHarnessName = SessionId

  153 -- ---------------------------------------------------------------------------
  154 -- Production probe (mirrors Frontend.API.probeHarness, sans HarnessInfo wrapper)
  155 -- ---------------------------------------------------------------------------
  156 
  157 -- | Snapshot the current per-harness 'HarnessActivity'. Exception-safe per
  158 -- harness: a probe failure on one entry yields 'HarnessIdle' rather than
  159 -- propagating, matching the HTTP @\/harnesses@ probe's behaviour and
  160 -- preserving the loop's "best-effort" contract.
  161 snapshotHarnesses
  162   :: IORef (Map Text HarnessHandle)
  163   -> IO (Map Text HarnessActivity)
  164 snapshotHarnesses ref = do
  165   harnesses <- readIORef ref
  166   pairs <- traverse probeOne (Map.toList harnesses)
  167   pure (Map.fromList pairs)
  168   where
  169     probeOne :: (Text, HarnessHandle) -> IO (Text, HarnessActivity)
  170     probeOne (name, hh) = do
  171       st <- _hh_status hh
  172       activity <- case st of
  173         HarnessExited _ -> pure HarnessStopped
  174         HarnessRunning  -> probeRunning (_hh_session hh) name
  175       pure (name, activity)
  176 
  177 -- | Same shape as 'PureClaw.Frontend.API.probeActivity': tmux capture
  178 -- distinguishes idle vs thinking; capture failures degrade to idle so the
  179 -- loop never crashes on a transient tmux hiccup.
  180 probeRunning :: Text -> Text -> IO HarnessActivity
  181 probeRunning tmuxSession windowName = do
  182   let target = tmuxSession <> ":" <> extractWindowIdx windowName
  183   result <- try @SomeException (captureWindow target 50)
  184   case result of
  185     Left _        -> pure HarnessIdle
  186     Right capture ->
  187       pure $
  188         if isIdle (TE.decodeUtf8Lenient capture)
  189           then HarnessIdle
  190           else HarnessThinking
  191 
  192 -- | Extract the window-index suffix from a harness name like
  193 -- @"claude-code-0"@. Mirrors 'PureClaw.Frontend.API.extractWindowIdx' so
  194 -- the probe loop produces the same captures the HTTP @\/harnesses@
  195 -- endpoint does.
  196 extractWindowIdx :: Text -> Text
  197 extractWindowIdx name =
  198   case T.splitOn "-" name of
  199     parts | length parts >= 2 -> last parts
  200     _                         -> "0"
