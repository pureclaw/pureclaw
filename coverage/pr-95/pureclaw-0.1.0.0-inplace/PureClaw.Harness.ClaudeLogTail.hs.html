    1 -- | Bounded JSONL tail step + injectable IO seam for the claude-code session
    2 -- log content source.
    3 --
    4 -- This module is the boundary between the pure logic of 'PureClaw.Harness.JsonlTail'
    5 -- and 'PureClaw.Harness.ClaudeLogProse' and the real filesystem IO. All IO
    6 -- operations are injected via 'JsonlTailDeps', so the step function
    7 -- 'tailStep' can be fully unit-tested with a fake implementation backed by an
    8 -- 'IORef ByteString'.
    9 --
   10 -- == Design
   11 --
   12 -- 'tailStep' performs ONE bounded read-fold step:
   13 --
   14 --   1. Probe the file size via '_jt_size'.
   15 --   2. If @size < offset@ the file shrank\/rotated — reset state to the
   16 --      beginning and return immediately (the caller should re-validate the
   17 --      path and re-call).
   18 --   3. Backfill guard: if @offset == 0@ and @size > '_tc_backfill'@ return
   19 --      @['TailUnavailable']@ without advancing.
   20 --   4. Read @min(size - offset, '_tc_chunk')@ bytes via '_jt_readFrom'.
   21 --   5. Split into complete lines via 'splitLinesBounded' with '_tc_buffer' as
   22 --      the partial-line cap; on 'OverCap' return @['TailUnavailable']@ without
   23 --      advancing.
   24 --   6. Enforce '_tc_line' on each complete line; a single line longer than the
   25 --      cap yields @['TailUnavailable']@ without advancing.
   26 --   7. Fold each complete line via 'foldProseLine', collecting a 'TailProse'
   27 --      per yielded 'ProseTurn'.
   28 --   8. Advance the offset by the number of bytes consumed.
   29 --
   30 -- NO loop and NO async live here. The driving loop + async lifecycle live in
   31 -- Task 5 (wiring), keeping this module pure-over-deps and fully unit-coverable.

   80 -- ---------------------------------------------------------------------------
   81 -- Injectable IO seam
   82 -- ---------------------------------------------------------------------------

   84 -- | All filesystem IO needed by 'tailStep' and 'seekStart', injectable for
   85 -- testing.  The production implementation opens files with @O_NOFOLLOW@ and
   86 -- bounds reads to the requested byte count.
   87 --
   88 -- The 'SafeClaudeLogPath' parameter is passed through to every operation so
   89 -- the production implementation can re-open the file per-read without holding
   90 -- an fd between steps.
   91 data JsonlTailDeps = JsonlTailDeps
   92   { _jt_size     :: SafeClaudeLogPath -> IO Integer
   93     -- ^ Return the current byte length of the log file.
   94   , _jt_readFrom :: SafeClaudeLogPath -> Offset -> Int -> IO (ByteString, Offset)
   95     -- ^ @_jt_readFrom path offset n@: read UP TO @n@ bytes starting at
   96     -- @offset@, returning the chunk and the new offset.
   97   , _jt_now      :: IO UTCTime
   98     -- ^ Current UTC time (used downstream; injected for testability).
   99   }
  100 
  101 -- | Production 'JsonlTailDeps' over the real filesystem.
  102 --
  103 -- Every operation re-opens the leaf with @O_NOFOLLOW@ (so a leaf-swap to a
  104 -- symlink between steps fails the open — the steady-state half of the TOCTOU
  105 -- guard described in the spec Security section; the validate-on-rotation half
  106 -- is the caller's via 'tailStep' shrink-reset). Reads are positional (@fdSeek@
  107 -- + @fdReadBuf@) and bounded to the requested byte count, so a rapidly-growing
  108 -- file never delivers an unbounded single chunk.
  109 defaultJsonlTailDeps :: JsonlTailDeps
  110 defaultJsonlTailDeps = JsonlTailDeps
  111   { _jt_size = \path ->
  112       withNoFollowFd path $ \fd -> do
  113         status <- PF.getFdStatus fd
  114         pure (fromIntegral (PF.fileSize status))
  115   , _jt_readFrom = \path (Offset off) n ->
  116       withNoFollowFd path $ \fd -> do
  117         _ <- PIO.fdSeek fd AbsoluteSeek (fromIntegral off :: FileOffset)
  118         bs <- readUpTo fd n
  119         pure (bs, Offset (off + fromIntegral (BS.length bs)))
  120   , _jt_now = getCurrentTime
  121   }
  122   where
  123     -- Open the validated leaf with @O_NOFOLLOW@, run the action, always close.
  124     withNoFollowFd :: SafeClaudeLogPath -> (Fd -> IO a) -> IO a
  125     withNoFollowFd path action = do
  126       let flags = PIO.defaultFileFlags { PIO.nofollow = True }
  127       Exc.bracket
  128         (PIO.openFd (getSafeClaudeLogPath path) PIO.ReadOnly flags)
  129         PIO.closeFd
  130         action
  131 
  132     -- Read UP TO @n@ bytes from the current fd position into a fresh
  133     -- 'ByteString'. @fdReadBuf@ returns the count actually read (0 at EOF).
  134     readUpTo :: Fd -> Int -> IO ByteString
  135     readUpTo fd n
  136       | n <= 0 = pure BS.empty
  137       | otherwise =
  138           BSI.createAndTrim n $ \ptr -> do
  139             got <- PIO.fdReadBuf fd (castPtr (ptr :: Ptr Word8))
  140                      (fromIntegral n :: ByteCount)
  141             pure (fromIntegral got)

  144 -- DoS caps
  145 -- ---------------------------------------------------------------------------

  147 -- | Caps controlling the bounded tail step. All sizes are in bytes.
  148 data TailCaps = TailCaps
  149   { _tc_backfill :: !Int
  150     -- ^ Maximum total file size for an initial (from-0) backfill.
  151     -- If @offset == 0@ and @fileSize > _tc_backfill@, yield 'TailUnavailable'.
  152   , _tc_line     :: !Int
  153     -- ^ Maximum length of a single complete line.
  154     -- A complete line whose byte count exceeds this yields 'TailUnavailable'.
  155   , _tc_buffer   :: !Int
  156     -- ^ Maximum buffered partial-line size.
  157     -- Passed as the cap to 'splitLinesBounded'.
  158   , _tc_chunk    :: !Int
  159     -- ^ Maximum bytes read per step: @min(fileSize - offset, _tc_chunk)@.
  160   }
  161 
  162 -- | Production-safe defaults: 32 MiB backfill, 1 MiB line, 1 MiB buffer,
  163 -- 256 KiB chunk.
  164 defaultTailCaps :: TailCaps
  165 defaultTailCaps = TailCaps
  166   { _tc_backfill = 32 * 1024 * 1024
  167   , _tc_line     = 1024 * 1024
  168   , _tc_buffer   = 1024 * 1024
  169   , _tc_chunk    = 256 * 1024
  170   }

  176 -- | A single event emitted by 'tailStep'.
  177 data TailEvent
  178   = TailProse !ProseTurn
  179     -- ^ A prose turn was updated or finalized.
  180   | TailUnavailable
  181     -- ^ A DoS cap was exceeded; the caller should fall back to the tmux path.
  182   deriving stock (Eq, Show)

  184 -- ---------------------------------------------------------------------------
  185 -- seekStart
  186 -- ---------------------------------------------------------------------------
  187 
  188 -- | Determine the starting offset for a tailer.
  189 --
  190 -- @seekStart deps path mPersisted@:
  191 --
  192 --   * @Nothing@: seek to EOF (the current file size) — record nothing
  193 --     historical.
  194 --   * @Just off@: resume from the persisted offset, but if it exceeds the
  195 --     current size (e.g. after a rotation) fall back to the current size.
  196 seekStart
  197   :: JsonlTailDeps
  198   -> SafeClaudeLogPath
  199   -> Maybe Offset
  200   -> IO Offset
  201 seekStart deps path Nothing = do
  202   sz <- _jt_size deps path
  203   pure (Offset sz)
  204 seekStart deps path (Just saved@(Offset off)) = do
  205   sz <- _jt_size deps path
  206   if off > sz
  207     then pure (Offset sz)
  208     else pure saved
  209 
  210 -- ---------------------------------------------------------------------------
  211 -- tailStep
  212 -- ---------------------------------------------------------------------------
  213 
  214 -- | One bounded read-fold step of the JSONL tailer.
  215 --
  216 -- Returns the new @(Offset, Buffer, ProseState)@ and a (possibly empty) list
  217 -- of 'TailEvent's.  On any cap violation, returns @[TailUnavailable]@ and
  218 -- does NOT advance the offset (the offset in the returned triple is
  219 -- unchanged).
  220 --
  221 -- Rotation\/shrink detection: if the current file size is strictly less than
  222 -- the current offset, reset state to @(Offset 0, emptyBuffer, emptyProseState)@
  223 -- and return no events (the caller should re-validate the path and re-call).
  224 tailStep
  225   :: JsonlTailDeps
  226   -> TailCaps
  227   -> SafeClaudeLogPath
  228   -> (Offset, Buffer, ProseState)
  229   -> IO ((Offset, Buffer, ProseState), [TailEvent])
  230 tailStep deps caps path (off@(Offset offI), buf, prose) = do
  231   sz <- _jt_size deps path
  232   if sz < offI
  233     -- Shrink / rotation: reset state, return no events.
  234     then pure ((Offset 0, emptyBuffer, emptyProseState), [])
  235     else do
  236       -- Backfill guard: if we are at offset 0 and the file is too large,
  237       -- signal unavailability without advancing.
  238       if offI == 0 && sz > fromIntegral (_tc_backfill caps)
  239         then pure ((off, buf, prose), [TailUnavailable])
  240         else do
  241           let toRead = fromIntegral (min (sz - offI) (fromIntegral (_tc_chunk caps))) :: Int
  242           if toRead == 0
  243             -- Nothing new to read.
  244             then pure ((off, buf, prose), [])
  245             else do
  246               (chunk, newOff) <- _jt_readFrom deps path off toRead
  247               case splitLinesBounded (_tc_buffer caps) chunk buf of
  248                 Left OverCap ->
  249                   -- Buffer cap exceeded: signal unavailability, no advance.
  250                   pure ((off, buf, prose), [TailUnavailable])
  251                 Right (lines_, newBuf) ->
  252                   -- Check each complete line against the line-length cap.
  253                   case enforceLineCap (_tc_line caps) lines_ of
  254                     Left () ->
  255                       -- A single line was too long: signal unavailability, no advance.
  256                       pure ((off, buf, prose), [TailUnavailable])
  257                     Right goodLines ->
  258                       -- Fold all complete lines into the prose state.
  259                       let (finalProse, evs) = foldLines goodLines prose
  260                       in pure ((newOff, newBuf, finalProse), evs)
  261 
  262 -- | Check that every 'CompleteLine' is within the line-length cap.
  263 -- Returns @Left ()@ as soon as one is too long (no advance should happen).
  264 enforceLineCap :: Int -> [CompleteLine] -> Either () [CompleteLine]
  265 enforceLineCap maxLine = go
  266   where
  267     go [] = Right []
  268     go (l : ls)
  269       | BS.length (unCompleteLine l) > maxLine = Left ()
  270       | otherwise = fmap (l :) (go ls)
  271 
  272 -- | Fold a list of complete JSONL lines through 'foldProseLine', collecting
  273 -- 'TailProse' events for each yielded 'ProseTurn'.
  274 foldLines :: [CompleteLine] -> ProseState -> (ProseState, [TailEvent])
  275 foldLines [] prose = (prose, [])
  276 foldLines (l : ls) prose =
  277   let (prose', mTurn) = foldProseLine (unCompleteLine l) prose
  278       ev = maybe [] (\t -> [TailProse t]) mTurn
  279       (finalProse, restEvs) = foldLines ls prose'
  280   in (finalProse, ev <> restEvs)
