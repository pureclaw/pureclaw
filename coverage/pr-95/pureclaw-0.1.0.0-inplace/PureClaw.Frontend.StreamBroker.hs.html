    1 -- | In-process pub/sub broker for the live transcript streaming feature.
    2 --
    3 -- The broker fans out 'BrokerEvent's to a bounded set of subscribers via
    4 -- per-subscriber 'TBQueue's. Publish is non-blocking and STM-atomic: under
    5 -- contention a full subscriber queue has its oldest entry dropped, its
    6 -- overflow flag set, and the new entry written — all in a single
    7 -- transaction.
    8 --
    9 -- The broker is origin-agnostic; per-origin caps are enforced at the WS
   10 -- handler layer (WU3) by a separate 'StreamGuard'. The broker only enforces
   11 -- the global subscriber cap configured by '_bc_maxSubscribers'.
   12 --
   13 -- See "docs/transcript-streaming.md" §Broker for the full design.
   14 module PureClaw.Frontend.StreamBroker
   15   ( -- * Broker handle
   16     StreamBroker (..)
   17     -- * Events
   18   , BrokerEvent (..)
   19   , SessionActivity (..)
   20     -- * Subscription
   21   , Subscription (..)
   22   , SubscriberId (..)
   23     -- * Errors
   24   , BrokerError (..)
   25     -- * Introspection
   26   , BrokerStats (..)
   27     -- * Configuration

   62 -- ---------------------------------------------------------------------------
   63 -- Events
   64 -- ---------------------------------------------------------------------------
   65 
   66 -- | Tagged event published by writers and consumed by subscribers (WS
   67 -- handlers). 'EntryRecorded' carries the full transcript entry; subscribers
   68 -- filter by 'SessionId' when delivering to their peer. 'ActivityChanged'
   69 -- carries lightweight per-session signals that the frontend uses for the
   70 -- sidebar.
   71 data BrokerEvent
   72   = EntryRecorded   !SessionId !TranscriptEntry
   73     -- | An in-progress (ephemeral) harness message was updated in-place.
   74     -- Subscribers that are currently replaying this session drop the event
   75     -- (it will be re-sent on the next tick); all other focused subscribers
   76     -- forward it immediately as a 'SeEntryUpdate' wire event.
   77   | EntryUpdated    !SessionId !TranscriptEntry
   78   | ActivityChanged !SessionId !SessionActivity
   79     -- | Full sidebar list snapshot (tabs + recent + archived) pushed on
   80     -- every mutation and once on WS connect. Carries pre-serialized JSON
   81     -- so the WS handler can send it without knowing the domain types.
   82   | ListsSnapshot   !Value
   83   deriving stock (Show, Eq)
   84 
   85 -- | Per-session live signals for sidebar/tab indicators.
   86 data SessionActivity
   87   = -- | A new entry was recorded at this timestamp; carries no payload (the
   88     -- entry itself rides on the companion 'EntryRecorded' publish).
   89     SaEntryAt !UTCTime
   90     -- | The session's harness transitioned to a new activity state.
   91   | SaHarnessStatus !HarnessActivity
   92     -- | A new session appeared (carries its meta so the sidebar can render
   93     -- it immediately).
   94   | SaSessionCreated !SessionMeta
   95   deriving stock (Show, Eq)

   98 -- Subscription
   99 -- ---------------------------------------------------------------------------

  101 -- | Opaque identifier for a subscriber, allocated by the broker via a
  102 -- monotonically increasing STM counter.
  103 newtype SubscriberId = SubscriberId { unSubscriberId :: Int }
  104   deriving stock (Show, Eq, Ord)

  106 -- | Subscription handle returned by '_streamBroker_subscribe'. The WS
  107 -- handler reads from '_sub_queue' for events and from '_sub_overflow' for
  108 -- the overflow signal (via STM @orElse@). On disconnect the handler must
  109 -- invoke '_sub_cancel' to atomically remove the subscriber from the broker.
  110 data Subscription = Subscription
  111   { _sub_queue    :: !(TBQueue BrokerEvent)
  112   , _sub_overflow :: !(TVar Bool)
  113   , _sub_cancel   :: !(IO ())
  114   }

  138 -- ---------------------------------------------------------------------------
  139 -- Configuration
  140 -- ---------------------------------------------------------------------------
  141 
  142 -- | Static configuration for an in-process broker. All caps are enforced at
  143 -- broker boundaries; per-origin enforcement is the WS handler's
  144 -- responsibility ('_bc_maxSubsPerOrigin' is provided here so that handler
  145 -- code can read a single source of truth via '_streamBroker_config').
  146 data BrokerConfig = BrokerConfig
  147   { -- | Per-subscriber 'TBQueue' depth.
  148     _bc_queueDepth       :: !Int
  149     -- | Per-process subscriber cap.
  150   , _bc_maxSubscribers   :: !Int
  151     -- | Per-origin subscriber cap (enforced by the WS handler's
  152     -- 'StreamGuard', not by the broker itself).
  153   , _bc_maxSubsPerOrigin :: !Int
  154     -- | Per-event payload byte cap. Larger entries are truncated upstream
  155     -- by 'BroadcastingTranscriptHandle' (WU2) before publish.
  156   , _bc_maxEventBytes    :: !Int
  157   }
  158   deriving stock (Show, Eq)
  159 
  160 -- | Default config: queue depth 256, 32 subscribers, 8 per origin, 256 KB
  161 -- per event.
  162 defaultBrokerConfig :: BrokerConfig
  163 defaultBrokerConfig =
  164   BrokerConfig
  165     { _bc_queueDepth       = 256
  166     , _bc_maxSubscribers   = 32
  167     , _bc_maxSubsPerOrigin = 8
  168     , _bc_maxEventBytes    = 256 * 1024
  169     }

  171 -- ---------------------------------------------------------------------------
  172 -- Broker handle
  173 -- ---------------------------------------------------------------------------
  174 
  175 -- | Pub/sub broker capability. All operations are 'IO' so the broker can
  176 -- be passed across thread boundaries (it lives at the lifetime of the
  177 -- process).
  178 data StreamBroker = StreamBroker
  179   { -- | Non-blocking publish to all current subscribers. STM-atomic:
  180     -- enqueue or (on full queue) pop oldest + enqueue + set overflow flag,
  181     -- in a single transaction across every subscriber.
  182     _streamBroker_publish    :: BrokerEvent -> IO ()
  183     -- | Allocate a new subscription. Returns 'Left' 'SubscriberCapReached'
  184     -- when the per-process subscriber cap is reached.
  185   , _streamBroker_subscribe  :: IO (Either BrokerError Subscription)
  186     -- | Snapshot of broker state (subscriber count + per-subscriber queue
  187     -- depths). Cheap; STM-atomic.
  188   , _streamBroker_introspect :: IO BrokerStats
  189     -- | Read-only accessor exposing the configuration. WU2's
  190     -- 'BroadcastingTranscriptHandle' reads '_bc_maxEventBytes' through
  191     -- this; the WS handler reads '_bc_maxSubsPerOrigin'.
  192   , _streamBroker_config     :: BrokerConfig
  193     -- | Most-recent 'HarnessActivity' observed for a session via an
  194     -- 'ActivityChanged' / 'SaHarnessStatus' publish, or 'Nothing' if no
  195     -- such event has been seen. The WS handler reads this on a fresh
  196     -- focus so a client connecting mid-request sees the in-flight
  197     -- thinking state immediately, rather than waiting for the next
  198     -- transition.
  199   , _streamBroker_currentActivity :: SessionId -> IO (Maybe HarnessActivity)
  200   }
  201 
  202 -- ---------------------------------------------------------------------------
  203 -- In-process broker implementation
  204 -- ---------------------------------------------------------------------------
  205 
  206 -- | The mutable state of an in-process broker.
  207 data BrokerState = BrokerState
  208   { _bst_nextId           :: !(TVar Int)
  209   , _bst_subscribers      :: !(TVar (Map SubscriberId Subscription))
  210     -- | Latest 'SaHarnessStatus' seen for each session. Updated inside the
  211     -- same STM transaction as the publish so a focus snapshot taken just
  212     -- after a publish observes the post-publish state.
  213   , _bst_currentActivity  :: !(TVar (Map SessionId HarnessActivity))
  214   }
  215 
  216 -- | Construct an in-process broker. The broker exists for the lifetime of
  217 -- the process and is shared across every write site, the activity probe,
  218 -- and every WS connection. There is exactly one broker per binary main
  219 -- process (see the design's lifecycle section).
  220 mkInProcessBroker :: BrokerConfig -> IO StreamBroker
  221 mkInProcessBroker cfg = do
  222   state <- BrokerState
  223     <$> newTVarIO 0
  224     <*> newTVarIO Map.empty
  225     <*> newTVarIO Map.empty
  226   pure StreamBroker
  227     { _streamBroker_publish          = publishEvent state
  228     , _streamBroker_subscribe        = subscribeNew cfg state
  229     , _streamBroker_introspect       = introspect state
  230     , _streamBroker_config           = cfg
  231     , _streamBroker_currentActivity  = currentActivity state
  232     }
  233 
  234 -- ---------------------------------------------------------------------------
  235 -- Publish
  236 -- ---------------------------------------------------------------------------
  237 
  238 -- | Publish to every current subscriber in a single STM transaction.
  239 --
  240 -- For each subscriber, if its queue is full: drop the oldest entry, set its
  241 -- overflow 'TVar', then write the new entry. Otherwise just write the new
  242 -- entry. Performing all subscriber updates inside one 'atomically' block
  243 -- gives the design's linearizability guarantee (D3): any later
  244 -- 'atomically' read from a subscriber observes either the pre- or
  245 -- post-publish state for every subscriber, never a mix.
  246 publishEvent :: BrokerState -> BrokerEvent -> IO ()
  247 publishEvent state ev = atomically $ do
  248   -- Record the latest harness status for the affected session before fanning
  249   -- out so a subsequent 'currentActivity' read observes the post-publish
  250   -- value. Only 'SaHarnessStatus' contributes to the snapshot; entry/created
  251   -- events carry no persistent state we need to replay on focus.
  252   case ev of
  253     ActivityChanged sid (SaHarnessStatus s) ->
  254       modifyTVar' (_bst_currentActivity state) (Map.insert sid s)
  255     _ -> pure ()
  256   subs <- readTVar (_bst_subscribers state)
  257   forM_ (Map.elems subs) $ \sub -> publishOne sub ev
  258 
  259 -- | Read the most recent 'HarnessActivity' the broker has observed for a
  260 -- session. STM-atomic and cheap.
  261 currentActivity :: BrokerState -> SessionId -> IO (Maybe HarnessActivity)
  262 currentActivity state sid =
  263   Map.lookup sid <$> readTVarIO (_bst_currentActivity state)
  264 
  265 -- | Enqueue one event to one subscriber under the overflow protocol.
  266 -- Pure-STM so multiple invocations compose inside the publish transaction.
  267 publishOne :: Subscription -> BrokerEvent -> STM ()
  268 publishOne sub ev = do
  269   full <- isFullTBQueue (_sub_queue sub)
  270   when full $ do
  271     _ <- readTBQueue (_sub_queue sub)
  272     writeTVar (_sub_overflow sub) True
  273   writeTBQueue (_sub_queue sub) ev
  274 
  275 -- ---------------------------------------------------------------------------
  276 -- Subscribe
  277 -- ---------------------------------------------------------------------------
  278 
  279 -- | Allocate a new subscriber slot. Enforces the per-process subscriber cap
  280 -- ('_bc_maxSubscribers'). Returns 'Left' 'SubscriberCapReached' when the
  281 -- cap is reached; the WS handler maps this to a 503.
  282 subscribeNew
  283   :: BrokerConfig
  284   -> BrokerState
  285   -> IO (Either BrokerError Subscription)
  286 subscribeNew cfg state = atomically $ do
  287   subs <- readTVar (_bst_subscribers state)
  288   if Map.size subs >= _bc_maxSubscribers cfg
  289     then pure (Left SubscriberCapReached)
  290     else do
  291       nextId <- readTVar (_bst_nextId state)
  292       writeTVar (_bst_nextId state) (nextId + 1)
  293       let sid = SubscriberId nextId
  294       queue    <- newTBQueue (fromIntegral (_bc_queueDepth cfg))
  295       overflow <- newTVar False
  296       let sub = Subscription
  297             { _sub_queue    = queue
  298             , _sub_overflow = overflow
  299             , _sub_cancel   = cancelSubscriber state sid
  300             }
  301       modifyTVar' (_bst_subscribers state) (Map.insert sid sub)
  302       pure (Right sub)
  303 
  304 -- | Remove a subscriber from the broker map atomically. Idempotent — a
  305 -- second call is a no-op (the WS handler may invoke this from both the
  306 -- bracket cleanup and an error path; the no-op is intentional).
  307 cancelSubscriber :: BrokerState -> SubscriberId -> IO ()
  308 cancelSubscriber state sid =
  309   atomically $ modifyTVar' (_bst_subscribers state) (Map.delete sid)
  310 
  311 -- ---------------------------------------------------------------------------
  312 -- Introspect
  313 -- ---------------------------------------------------------------------------
  314 
  315 -- | Snapshot the broker's state. The per-subscriber queue depths are read
  316 -- atomically alongside the subscriber count so the resulting 'BrokerStats'
  317 -- reflects a single point in time.
  318 introspect :: BrokerState -> IO BrokerStats
  319 introspect state = do
  320   subs <- readTVarIO (_bst_subscribers state)
  321   let elems = Map.elems subs
  322   depths <- atomically $ forM elems $ \sub ->
  323     fromIntegral <$> lengthTBQueue (_sub_queue sub)
  324   pure BrokerStats
  325     { _bs_subscriberCount = Map.size subs
  326     , _bs_queueDepths     = depths
  327     }
