    1 -- | Pure prose fold over claude-code JSONL session-log lines.
    2 --
    3 -- This module extracts ONLY the assistant @text@ blocks from a claude-code
    4 -- JSONL log, accumulating them turn-by-turn into a sanitized prose string.
    5 -- It is the pure heart of the claude-log content source — no IO, fully
    6 -- unit-testable.
    7 --
    8 -- == Fold rules (prose-only)
    9 --
   10 -- * An @assistant@ event appends its @text@ content blocks (each one
   11 --   sanitized via 'sanitizeHarnessOutput') to the current turn's prose.
   12 --   The @_pt_sourceUuid@ is pinned to the FIRST assistant line's @uuid@
   13 --   and MUST NOT change as later assistant lines arrive (stable derived
   14 --   turn-id).
   15 --
   16 -- * A terminal @stop_reason@ (@end_turn@, @stop_sequence@, @max_tokens@)
   17 --   on an assistant event sets @_pt_finalized = True@.  A @stop_reason@
   18 --   of @tool_use@ does NOT finalize (the turn will continue after the
   19 --   tool call returns).
   20 --
   21 -- * A @user@ event whose @message.content@ is NOT solely @tool_result@
   22 --   blocks ends the current turn (finalizing it if it had prose) and
   23 --   starts a fresh empty turn.
   24 --
   25 -- * A @user@ event whose content IS solely @tool_result@ blocks does NOT
   26 --   end the turn (it is the tool-call return, not a real user message).
   27 --
   28 -- * @thinking@, @tool_use@, and @tool_result@ assistant content blocks
   29 --   are silently ignored.
   30 --
   31 -- * Unknown event types, malformed JSON, and missing\/wrong-typed fields
   32 --   leave the state unchanged and yield 'Nothing'.  Total — never throws.
   33 module PureClaw.Harness.ClaudeLogProse
   34   ( -- * State
   35     ProseState
   36   , emptyProseState
   37     -- * Per-turn result
   38   , ProseTurn (..)
   39     -- * Fold step
   40   , foldProseLine
   41     -- * State accessor
   42   , currentProseTurn
   43     -- * Stable turn-id derivation
   44   , deriveTurnId
   45   ) where

   72 -- | The result of one updated or finalized prose turn.
   73 data ProseTurn = ProseTurn
   74   { _pt_sourceUuid :: !Text   -- ^ Pinned to the FIRST assistant line's @uuid@.
   75   , _pt_text       :: !Text   -- ^ Accumulated sanitized assistant prose.
   76   , _pt_finalized  :: !Bool   -- ^ True iff a terminal @stop_reason@ was seen.
   77   } deriving stock (Eq, Show)
   78 
   79 -- | Opaque accumulator for 'foldProseLine'.  Callers obtain the initial
   80 -- value via 'emptyProseState' and inspect it via 'currentProseTurn' or
   81 -- the 'Maybe ProseTurn' returned by each 'foldProseLine' call.
   82 newtype ProseState = ProseState
   83   { _ps_turn :: Maybe ActiveTurn
   84   } deriving stock (Eq, Show)
   85 
   86 -- | The in-progress turn being accumulated (internal).
   87 data ActiveTurn = ActiveTurn
   88   { _at_sourceUuid :: !Text
   89   , _at_text       :: !Text
   90   , _at_finalized  :: !Bool
   91   } deriving stock (Eq, Show)
   92 
   93 -- ---------------------------------------------------------------------------
   94 -- Smart constructors / accessors
   95 -- ---------------------------------------------------------------------------
   96 
   97 -- | The initial prose accumulator: no active turn.
   98 emptyProseState :: ProseState
   99 emptyProseState = ProseState { _ps_turn = Nothing }
  100 
  101 -- | Return the current 'ProseTurn' only when it has visible prose or is
  102 -- finalized.  An empty un-finalized turn is invisible.
  103 currentProseTurn :: ProseState -> Maybe ProseTurn
  104 currentProseTurn ps = _ps_turn ps >>= turnIfVisible
  105 
  106 -- | Yield the turn only when it has visible prose or is finalized.
  107 turnIfVisible :: ActiveTurn -> Maybe ProseTurn
  108 turnIfVisible at
  109   | T.null (_at_text at) && not (_at_finalized at) = Nothing
  110   | otherwise = Just ProseTurn
  111       { _pt_sourceUuid = _at_sourceUuid at
  112       , _pt_text       = _at_text at
  113       , _pt_finalized  = _at_finalized at
  114       }
  115 
  116 -- ---------------------------------------------------------------------------
  117 -- Fold step
  118 -- ---------------------------------------------------------------------------
  119 
  120 -- | Fold one complete JSONL 'ByteString' line into 'ProseState'.
  121 --
  122 -- Returns the updated state and, when the turn's visible content changes
  123 -- (prose grew or finalized flag flipped), the current 'ProseTurn'.
  124 -- Total: any malformed\/unknown\/meta line leaves state unchanged and
  125 -- returns 'Nothing'.  Never throws.
  126 foldProseLine :: ByteString -> ProseState -> (ProseState, Maybe ProseTurn)
  127 foldProseLine raw ps =
  128   case decodeObject raw of
  129     Nothing  -> noChange ps
  130     Just obj -> dispatchEvent obj ps
  131 
  132 -- | Leave state unchanged.
  133 noChange :: ProseState -> (ProseState, Maybe ProseTurn)
  134 noChange ps = (ps, Nothing)
  135 
  136 -- | Route by @type@ field; unknown\/meta types → no change.
  137 dispatchEvent :: KeyMap Value -> ProseState -> (ProseState, Maybe ProseTurn)
  138 dispatchEvent obj ps =
  139   case lookupText "type" obj of
  140     Just "assistant" -> handleAssistant obj ps
  141     Just "user"      -> handleUser obj ps
  142     _                -> noChange ps
  143 
  144 -- ---------------------------------------------------------------------------
  145 -- Assistant event
  146 -- ---------------------------------------------------------------------------
  147 
  148 -- | Process an @assistant@ event: append sanitized @text@ blocks, pin uuid
  149 -- to the first line, finalize on a terminal @stop_reason@.
  150 handleAssistant :: KeyMap Value -> ProseState -> (ProseState, Maybe ProseTurn)
  151 handleAssistant obj ps =
  152   case (lookupText "uuid" obj, lookupObject "message" obj) of
  153     (Just uuid, Just msg) ->
  154       let prose     = extractProse msg
  155           stopR     = lookupText "stop_reason" msg
  156           finalized = isTerminal stopR
  157       in  if T.null prose && not finalized
  158             then noChange ps
  159             else
  160               let at' = applyToTurn uuid prose finalized (_ps_turn ps)
  161                   ps' = ps { _ps_turn = Just at' }
  162               in  (ps', turnIfVisible at')
  163     _ -> noChange ps
  164 
  165 -- | Incorporate new prose + finalized flag into the current 'ActiveTurn',
  166 -- pinning @uuid@ only on the FIRST assistant line.
  167 applyToTurn :: Text -> Text -> Bool -> Maybe ActiveTurn -> ActiveTurn
  168 applyToTurn uuid prose finalized Nothing =
  169   ActiveTurn
  170     { _at_sourceUuid = uuid
  171     , _at_text       = prose
  172     , _at_finalized  = finalized
  173     }
  174 applyToTurn _uuid prose finalized (Just at) =
  175   at { _at_text     = _at_text at <> prose
  176      , _at_finalized = _at_finalized at || finalized
  177      }
  178 
  179 -- | True iff the @stop_reason@ is a turn-terminal reason.
  180 -- @tool_use@ is intentionally NOT terminal.
  181 isTerminal :: Maybe Text -> Bool
  182 isTerminal (Just "end_turn")      = True
  183 isTerminal (Just "stop_sequence") = True
  184 isTerminal (Just "max_tokens")    = True
  185 isTerminal _                      = False
  186 
  187 -- | Collect all @text@ block values from @message.content@, sanitized via
  188 -- 'sanitizeHarnessOutput'.  Non-text block types are silently skipped.
  189 extractProse :: KeyMap Value -> Text
  190 extractProse msg =
  191   case KeyMap.lookup (Key.fromText "content") msg of
  192     Just (Array blocks) -> T.concat (map extractTextBlock (V.toList blocks))
  193     _                   -> ""
  194 
  195 -- | Extract the sanitized text from a single content block.
  196 -- Returns empty text for non-text blocks (@thinking@, @tool_use@, etc.).
  197 extractTextBlock :: Value -> Text
  198 extractTextBlock (Object km) =
  199   case lookupText "type" km of
  200     Just "text" -> maybe "" sanitizeHarnessOutput (lookupText "text" km)
  201     _ -> ""
  202 extractTextBlock _ = ""
  203

  205 -- User event
  206 -- ---------------------------------------------------------------------------

  208 -- | Process a @user@ event.
  209 --
  210 -- If the content is NOT solely @tool_result@ blocks, this is a real user
  211 -- message and ends the current turn (resetting to no active turn).
  212 --
  213 -- Backstop finalize: if the active turn has non-empty content and was NOT
  214 -- already finalized (i.e. it never received a terminal @stop_reason@), emit
  215 -- it now as a finalized 'ProseTurn'.  If the turn was already finalized it
  216 -- was already emitted on the earlier tick — do NOT re-emit it (avoid a
  217 -- duplicate finalized emission of the same turn id).
  218 --
  219 -- If the content is solely @tool_result@ blocks, we treat it as a tool-call
  220 -- return and leave the current turn open.
  221 handleUser :: KeyMap Value -> ProseState -> (ProseState, Maybe ProseTurn)
  222 handleUser obj ps =
  223   case lookupObject "message" obj of
  224     Nothing  -> noChange ps
  225     Just msg ->
  226       if isSolelyToolResult msg
  227         then noChange ps          -- tool-call return — do not end the turn
  228         else                      -- real user msg — end turn, backstop finalize
  229           let ps' = ps { _ps_turn = Nothing }
  230               emit = case _ps_turn ps of
  231                 Nothing -> Nothing
  232                 Just at ->
  233                   if T.null (_at_text at) || _at_finalized at
  234                     then Nothing   -- empty, or already finalized on an earlier tick
  235                     else Just ProseTurn
  236                            { _pt_sourceUuid = _at_sourceUuid at
  237                            , _pt_text       = _at_text at
  238                            , _pt_finalized  = True
  239                            }
  240           in  (ps', emit)
  241 
  242 -- | True iff every block in @message.content@ has @type = "tool_result"@.
  243 -- An empty block array or non-array content is treated as NOT solely
  244 -- tool_result (safe fallback — treat as a real user message).
  245 isSolelyToolResult :: KeyMap Value -> Bool
  246 isSolelyToolResult msg =
  247   case KeyMap.lookup (Key.fromText "content") msg of
  248     Just (Array blocks) ->
  249       not (V.null blocks) && V.all isToolResultBlock blocks
  250     _ -> False
  251 
  252 -- | True iff a content block has @type = "tool_result"@.
  253 isToolResultBlock :: Value -> Bool
  254 isToolResultBlock (Object km) = lookupText "type" km == Just "tool_result"
  255 isToolResultBlock _           = False

  261 -- | Derive a stable, collision-free UUIDv5 turn identifier.
  262 --
  263 -- @deriveTurnId sessionId firstAssistantUuid@ hashes the namespaced bytes
  264 -- @sessionId ++ \":\" ++ firstAssistantUuid@ under the DNS namespace UUID,
  265 -- producing a deterministic Text that is:
  266 --
  267 --   * stable across the lifetime of a turn (first-assistant uuid is
  268 --     pinned and never changes);
  269 --   * distinct across sessions (namespace includes @sessionId@);
  270 --   * reproducible after a crash\/restart (same inputs → same output).
  271 deriveTurnId :: Text -> Text -> Text
  272 deriveTurnId sessionId firstUuid =
  273   UUID.toText (V5.generateNamed V5.namespaceDNS nameBytes)
  274   where
  275     nameBytes :: [Word8]
  276     nameBytes = BS.unpack (TE.encodeUtf8 (sessionId <> ":" <> firstUuid))
