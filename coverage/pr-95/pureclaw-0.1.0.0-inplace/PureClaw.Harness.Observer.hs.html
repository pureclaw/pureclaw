    3 -- | Per-flavour, PURE detection and extraction of harness terminal output.
    4 -- Each harness TUI (Claude Code, Codex, …) draws its screen differently, so
    5 -- working/idle/awaiting-input detection and response extraction are
    6 -- flavour-specific. The reconcile loop and the @/harness output@ command both
    7 -- select an observer via 'observerFor'. Heuristics are facts about each tool's
    8 -- terminal output; they were validated against live captures.
    9 module PureClaw.Harness.Observer
   10   ( HarnessActivityState (..)
   11   , HarnessObserver (..)
   12   , observerFor
   13   , claudeObserver
   14   , genericObserver
   15   , isClaudeUserLine
   16   ) where
   17

   26 data HarnessActivityState = HasWorking | HasAwaitingInput | HasIdle
   27   deriving stock (Eq, Show)
   28 
   29 data HarnessObserver = HarnessObserver
   30   { _ho_classify        :: Text -> HarnessActivityState
   31   , _ho_extractResponse :: Int -> ByteString -> Text
   32   , _ho_relevantTail    :: Int -> ByteString -> Text
   33   , _ho_extractTurn     :: Int -> ByteString -> Text
   34   }
   35 
   36 observerFor :: HarnessFlavour -> HarnessObserver
   37 observerFor HClaudeCode = claudeObserver
   38 observerFor _           = genericObserver

   42 -- | Spinner glyphs Claude rotates while working (Dingbats range + a few extras).
   43 -- NOTE: the response-marker characters ⏺ (U+23FA), ● (U+25CF), and ⬤ (U+2B24)
   44 -- are intentionally NOT included here — they introduce those glyph characters as
   45 -- actual content (e.g. "⏺ Let me check the other call sites…") and must NOT be
   46 -- classified as working-spinner lines.  'isResponseMarkerLine' handles them for
   47 -- extraction; classification sees them only via the idle-prompt path.
   48 claudeSpinnerGlyphs :: [Char]
   49 claudeSpinnerGlyphs =
   50   "\x2022"   -- •  BULLET
   51   <> "\x2019"   -- '  RIGHT SINGLE QUOTATION MARK (used in some spinner patterns)
   52   <> "\x2726"   -- ✦  BLACK FOUR POINTED STAR
   53   <> "\x2732"   -- ✲  OPEN CENTRE ASTERISK
   54   <> "\x2733"   -- ✳  EIGHT SPOKED ASTERISK
   55   <> "\x2734"   -- ✴  EIGHT POINTED BLACK STAR
   56   <> "\x2735"   -- ✵  EIGHT POINTED PINWHEEL STAR
   57   <> "\x2736"   -- ✶  SIX POINTED BLACK STAR
   58   <> "\x2737"   -- ✷  EIGHT POINTED RECTILINEAR BLACK STAR
   59   <> "\x2738"   -- ✸  HEAVY EIGHT POINTED RECTILINEAR BLACK STAR
   60   <> "\x2739"   -- ✹  TWELVE POINTED BLACK STAR
   61   <> "\x273A"   -- ✺  SIXTEEN POINTED ASTERISK
   62   <> "\x273B"   -- ✻  TEARDROP-SPOKED ASTERISK
   63   <> "\x273C"   -- ✼  OPEN CENTRE TEARDROP-SPOKED ASTERISK
   64   <> "\x273D"   -- ✽  HEAVY TEARDROP-SPOKED ASTERISK
   65   <> "\x273E"   -- ✾  SIX PETALLED BLACK AND WHITE FLORETTE
   66   <> "\x273F"   -- ✿  BLACK FLORETTE
   67   <> "\x2605"   -- ★  BLACK STAR
   68   <> "\x22C6"   -- ⋆  STAR OPERATOR
   69   <> "\x22C7"   -- ⋇  DIVISION TIMES
   70   <> "\x2642"   -- ♂  MALE SIGN (used in some Hermes spinner patterns)
   71   <> "\x2666"   -- ♦  BLACK DIAMOND SUIT
   72   <> "\x00B7"   -- ·  MIDDLE DOT
   73   <> "\x2026"   -- …  HORIZONTAL ELLIPSIS (shouldn't start a line, belt-and-suspenders)
   74 
   75 -- Additional spinner chars observed in the wild (Hermes, faryo patterns).
   76 -- Claude Code uses the ✶ (U+2736) glyph as the primary working-status marker.
   77 -- The status line for "working" looks like: ✶ Smooshing… (4m 55s · ↓ 16.6k tokens)
   78 
   79 -- | A working/status line: starts with a spinner/star glyph followed by
   80 -- a description containing ellipsis or a token counter in parens, or the
   81 -- @esc to interrupt@ hint.
   82 isClaudeWorkingLine :: Text -> Bool
   83 isClaudeWorkingLine raw =
   84   let l = T.stripStart raw
   85   in (not (T.null l) && isSpinnerGlyph (T.head l)
   86        && (T.isInfixOf "\x2026" l || T.isInfixOf "..." l))   -- … or ...
   87      || hasTokenCounter l
   88      || T.isInfixOf "esc to interrupt" (T.toLower l)
   89   where
   90     -- A spinner glyph is any character in 'claudeSpinnerGlyphs'.
   91     isSpinnerGlyph c = c `elem` claudeSpinnerGlyphs
   92     hasTokenCounter t =
   93       T.isInfixOf "tokens" (T.toLower t)
   94       && T.isInfixOf "(" t && T.isInfixOf ")" t
   95 
   96 -- | An approval/menu prompt: Claude is blocked on the user.
   97 isClaudeApprovalLine :: Text -> Bool
   98 isClaudeApprovalLine raw =
   99   let l = T.toLower (T.stripStart raw)
  100   in T.isInfixOf "do you want to proceed?" l
  101      || T.isInfixOf "yes, and don't ask again" l
  102      || T.isInfixOf "yes, and don\x2019t ask again" l
  103      || T.isInfixOf "enter to confirm" l
  104      || T.isInfixOf "esc to cancel" l
  105      || isNumberedYesNo (T.stripStart raw)
  106 
  107 -- | A numbered option line, optionally led by the @❯@ menu cursor: @❯ 1. Yes@.
  108 -- Matches lines like "❯ 1. Yes", "  2. No", "1. Something".
  109 isNumberedYesNo :: Text -> Bool
  110 isNumberedYesNo l0 =
  111   -- strip leading ❯ ›  and spaces
  112   let l = T.dropWhile (\c -> c == '\x276F' || c == '\x203A' || c == ' ') l0
  113   in case T.span (\c -> c >= '0' && c <= '9') l of
  114        (ds, rest) -> not (T.null ds) && T.isPrefixOf "." rest
  115 
  116 -- | The idle input prompt: a bare @❯@/@›@ NOT followed by a number+dot
  117 -- (which would be a menu-selection cursor, not the prompt).
  118 isIdlePromptLine :: Text -> Bool
  119 isIdlePromptLine raw =
  120   let l = T.stripStart raw
  121   in (T.isPrefixOf "\x276F" l || T.isPrefixOf "\x203A" l)  -- ❯ ›
  122      && not (isNumberedYesNo l)
  123      && T.null (T.strip (T.drop 1 l))  -- nothing after the prompt glyph
  124 
  125 classifyClaude :: Text -> HarnessActivityState
  126 classifyClaude screen =
  127   let ls = T.lines screen
  128   in if any isClaudeWorkingLine ls then HasWorking
  129      else if any isClaudeApprovalLine ls then HasAwaitingInput
  130      else HasIdle
  131 
  132 -- | True for chrome lines that must never appear in extracted output.
  133 isClaudeChrome :: Text -> Bool
  134 isClaudeChrome raw =
  135   let l = T.stripStart raw
  136   in T.null (T.strip l)
  137      || isClaudeWorkingLine raw
  138      || (isIdlePromptLine raw)
  139      || T.isInfixOf "for shortcuts" (T.toLower l)
  140      || T.isInfixOf "ctrl+o to expand" (T.toLower l)
  141      || isBoxDrawingLine l  -- ─ ▀ ▄ lines (horizontal rules)
  142 
  143 isBoxDrawingLine :: Text -> Bool
  144 isBoxDrawingLine l =
  145   not (T.null l)
  146   && T.all (\c -> c == '\x2500' || c == '\x2580' || c == '\x2584' || isSpace c) l
  147 
  148 isResponseMarkerLine :: Text -> Bool
  149 isResponseMarkerLine line =
  150   let l = T.stripStart line
  151   in T.isPrefixOf "\x23FA" l    -- ⏺
  152      || T.isPrefixOf "\x25CF" l -- ● (alternate, faryo)
  153      || T.isPrefixOf "\x2B24" l -- ⬤ (alternate)
  154 
  155 stripResponseMarker :: Text -> Text
  156 stripResponseMarker line =
  157   T.stripStart (T.dropWhile (`elem` ("\x23FA\x25CF\x2B24 " :: String)) (T.stripStart line))
  158 
  159 -- | Extract the latest assistant response. @baseline@ is the number of lines
  160 -- already seen (used to skip already-delivered content). When awaiting input,
  161 -- return the approval/menu prompt block so the user sees the question.
  162 extractClaude :: Int -> ByteString -> Text
  163 extractClaude baseline capture =
  164   let body = dropBaseline baseline capture
  165       ls   = T.lines (TE.decodeUtf8Lenient body)
  166   in if any isClaudeApprovalLine ls
  167        then T.strip . T.unlines $ dropWhile (not . isClaudeApprovalLine) ls
  168        else case reverse [ i | (i, l) <- zip [0 :: Int ..] ls, isResponseMarkerLine l ] of
  169               []      -> ""
  170               (i : _) ->
  171                 let block   = takeWhile (not . isIdlePromptLine) (drop i ls)
  172                     cleaned = case block of
  173                       (h : rest) -> stripResponseMarker h : filter (not . isClaudeChrome) rest
  174                       []         -> []
  175                 in T.strip (T.intercalate "\n" (filter (not . T.null) cleaned))
  176 
  177 relevantTailClaude :: Int -> ByteString -> Text
  178 relevantTailClaude n capture =
  179   let ls = filter (not . isClaudeChrome) (T.lines (TE.decodeUtf8Lenient capture))
  180   in T.intercalate "\n" (lastN n ls)
  181 
  182 -- | A user-submitted message line: ❯/› with non-empty text, NOT a numbered
  183 -- menu (❯ 1.) and NOT the bare idle input box (❯ alone).
  184 isClaudeUserLine :: Text -> Bool
  185 isClaudeUserLine raw =
  186   let l = T.stripStart raw
  187   in (T.isPrefixOf "\x276F" l || T.isPrefixOf "\x203A" l)  -- ❯ ›
  188      && not (isNumberedYesNo l)
  189      && not (T.null (T.strip (T.drop 1 l)))
  190 
  191 -- | Tool/process activity lines the TUI shows inside a turn (not assistant prose).
  192 isClaudeToolLine :: Text -> Bool
  193 isClaudeToolLine raw =
  194   let l = T.stripStart raw
  195       toolPrefixes = [ "Update(", "Edit(", "Write(", "Create(", "MultiEdit("
  196                      , "Read(", "Bash(", "Grep(", "Glob(", "LS(", "Task(", "TodoWrite(" ]
  197   in T.isPrefixOf "\x239C" l   -- ⎜ (rare)
  198      || T.isPrefixOf "\x23BD" l   -- ⎽ scan-line/box-drawing
  199      || T.isPrefixOf "\x2514" l
  200      || T.isPrefixOf "\x251C" l   -- └ ├ tree
  201      || T.isPrefixOf "\x2387" l   -- ⎇
  202      || T.isPrefixOf "\x23BF" l   -- ⎿ result line
  203      || (not (isResponseMarkerLine l) && any (`T.isPrefixOf` l) toolPrefixes)
  204 
  205 -- | The whole assistant turn since the last user prompt: drop everything up to
  206 -- and including the last user line, then keep the response blocks (markers
  207 -- stripped) and drop chrome + tool/process lines.
  208 extractTurnClaude :: Int -> ByteString -> Text
  209 extractTurnClaude baseline capture =
  210   let ls   = T.lines (TE.decodeUtf8Lenient (dropBaseline baseline capture))
  211       turn = case reverse [ i | (i, l) <- zip [0 :: Int ..] ls, isClaudeUserLine l ] of
  212                (i : _) -> drop (i + 1) ls
  213                []      -> ls
  214   in if any isClaudeApprovalLine turn
  215        then T.strip (T.unlines (dropWhile (not . isClaudeApprovalLine) turn))
  216        else
  217          let kept =
  218                [ if isResponseMarkerLine l then stripResponseMarker l else l
  219                | l <- turn
  220                , not (isClaudeChrome l)
  221                , not (isClaudeToolLine l)
  222                ]
  223          in T.strip (T.intercalate "\n" (filter (not . T.null . T.strip) kept))
  224 
  225 claudeObserver :: HarnessObserver
  226 claudeObserver = HarnessObserver
  227   { _ho_classify        = classifyClaude
  228   , _ho_extractResponse = extractClaude
  229   , _ho_relevantTail    = relevantTailClaude
  230   , _ho_extractTurn     = extractTurnClaude
  231   }

  233 -- ── Generic fallback (Codex/OpenCode/Hermes/PureClaw/Custom) ─────────────────
  234 
  235 -- | Generic observer for non-Claude harnesses. Classification is always
  236 -- 'HasIdle' — stability over time (in the reconcile loop) is the only signal.
  237 genericObserver :: HarnessObserver
  238 genericObserver = HarnessObserver
  239   { _ho_classify        = const HasIdle
  240   , _ho_extractResponse = \n cap ->
  241       let ls = cleanLines cap
  242       in if n <= 0
  243            then T.intercalate "\n" ls
  244            else T.intercalate "\n" (lastN n ls)
  245   , _ho_relevantTail    = \n cap -> T.intercalate "\n" (lastN n (cleanLines cap))
  246   , _ho_extractTurn     = \n cap ->
  247       let ls = cleanLines cap
  248       in if n <= 0
  249            then T.intercalate "\n" ls
  250            else T.intercalate "\n" (lastN n ls)
  251   }
  252   where
  253     cleanLines cap = filter (not . T.null . T.strip) (T.lines (TE.decodeUtf8Lenient cap))
  254 
  255 -- ── Shared helpers ───────────────────────────────────────────────────────────
  256 
  257 dropBaseline :: Int -> ByteString -> ByteString
  258 dropBaseline n cap
  259   | n <= 0    = cap
  260   | otherwise = BS.intercalate (BS.singleton 0x0A) (drop n (BS.split 0x0A cap))
  261 
  262 lastN :: Int -> [a] -> [a]
  263 lastN n xs = drop (max 0 (length xs - n)) xs
