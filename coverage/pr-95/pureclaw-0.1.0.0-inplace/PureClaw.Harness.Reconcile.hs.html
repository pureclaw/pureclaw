    1 -- | Registry-based reconcile loop (Harness Registry & Lifecycle, Phase 1 — WU5).
    2 --
    3 -- This module is the registry's source of truth for harness /health/ and the
    4 -- single authoritative updater of each entry's coordinate + liveness. It
    5 -- replaces the legacy harness-map activity probe ("PureClaw.Frontend.ActivityProbe")
    6 -- with a loop that reconciles the durable registry against tmux reality.
    7 --
    8 -- == What the loop does, each tick
    9 --
   10 --   1. ONE server sweep per session ('_rd_sweep') — never per-I\/O.
   11 --   2. /Ownership filter/ (§8 C4 \/ D5.5\/D5.7): retain only rows whose
   12 --      @\@pcl_id@ matches a registry entry AND is PID-corroborated against the
   13 --      entry's recorded shell\/harness PID. A spoofed marker (collision or PID
   14 --      mismatch) is logged and treated as "not ours" — never captured, never
   15 --      marked live. A row with no recorded shell PID falls back to a second
   16 --      signal (the window-name prefix) to defend against PID reuse.
   17 --   3. Liveness classification (ours only): @Exited@ on @pane_dead@ or a dead
   18 --      harness PID; @Idle@\/@Thinking@\/@AwaitingInput@ via a screen capture
   19 --      classified by the entry's per-flavour observer (with a stability gate); an
   20 --      entry whose corroborated window is absent from the sweep → @Orphaned@.
   21 --   4. 'Reg.mergeReconcile' the observed fields into the registry atomically.
   22 --   5. /Symmetric diff/ (D5.2): emit one 'ActivityChanged' per entry whose
   23 --      liveness changed since the previous tick — INCLUDING entries that
   24 --      disappeared (now @Orphaned@\/@Exited@), fixing the disappearance gap.
   25 --      The first tick establishes the baseline and emits nothing.
   26 --
   27 -- == Resilience (D5.3)
   28 --
   29 -- A transient sweep\/capture failure marks the affected entries '_he_stale'
   30 -- and the loop CONTINUES — it never dies except on 'AsyncCancelled', which is
   31 -- re-raised so 'Async.withAsync' cleanup runs (the project-wide invariant; cf.
   32 -- "PureClaw.Frontend.ActivityProbe").
   33 --
   34 -- == Boot reconstruction (D5.6)
   35 --
   36 -- 'bootReconstruct' runs one sweep at startup and registers an entry for every
   37 -- window carrying our @\@pcl_id@ (the PCL-restart reconnect path), and lazily
   38 -- stamps a fresh 'Reg.HarnessId' onto legacy @claude-code-\<idx\>@ windows that
   39 -- carry no marker. The legacy harness map is still seeded in parallel by the
   40 -- unchanged @discoverHarnesses@; the registry is the new parallel path this
   41 -- loop owns.

  118 -- Dependencies (IO seam — tests inject deterministic fakes)

  121 -- | The injectable tmux operations the reconcile loop needs. Tests supply
  122 -- fakes (a queue of sweeps, a capture recorder) so the loop is deterministic
  123 -- and never forks a real tmux server. The production set is
  124 -- 'defaultReconcileDeps'.
  125 data ReconcileDeps = ReconcileDeps
  126   { _rd_sessions :: IO [Text]
  127     -- ^ Enumerate the sessions to sweep (production: the @\"pureclaw\"@ session
  128     --   plus any other session on the server, via 'listTmuxSessions').
  129   , _rd_sweep :: Text -> IO [TmuxWindowRow]
  130     -- ^ One server sweep of a session (production: 'readMarkers'). May throw on
  131     --   a transient tmux failure; the loop catches it and marks entries stale.
  132   , _rd_capture :: Text -> Text -> IO (Maybe Text)
  133     -- ^ Capture @session windowName@ and return the raw screen text, or
  134     --   'Nothing' if the capture failed (production: 'captureWindowNamed'
  135     --   decoded leniently, wrapped in 'try'). The classifier (per-flavour
  136     --   observer) decides liveness from the raw text; @Nothing@ disables the
  137     --   stability gate for the tick. Only called for corroborated, live (not
  138     --   pane_dead) windows.
  139   , _rd_harnessAlive :: Int -> IO Bool
  140     -- ^ Whether a recorded harness PID is still alive (production: a
  141     --   'harnessPidOf'-style liveness probe). Distinguishes @Exited@
  142     --   (harness gone, shell present) from @Idle@\/@Thinking@.
  143   , _rd_stampLegacy :: Text -> Text -> IO (Maybe Reg.HarnessId)
  144     -- ^ Lazily stamp a legacy @claude-code-\<idx\>@ window (no @\@pcl_id@) with
  145     --   a fresh marker + id, returning the new id (production: generate an id,
  146     --   'setWindowMarker'). 'Nothing' for a window we do not own.
  147   , _rd_evict :: Reg.HarnessId -> Text -> IO ()
  148     -- ^ Auto-eviction seam (design §5\/§10 Q2). Called with the
  149     --   @(harnessId, label)@ of an entry that has been Orphaned for
  150     --   'defaultOrphanGraceTicks' consecutive ticks. Production wires this to
  151     --   drop the entry from the LEGACY @_env_harnesses@ map (the registry
  152     --   delete is done by the loop itself, in 'reconcileTick'); it must NOT
  153     --   touch @session.json@ so the session reappears in Recent Sessions.
  154     --   Tests inject a recorder to assert the @(id, label)@ that was evicted.
  155   , _rd_recordResponse :: SessionId -> TranscriptEntry -> IO ()
  156     -- ^ Finalize a turn (Task 6). Records ONE harness @Response@ transcript
  157     --   entry for a session on a working→settle transition. The entry carries
  158     --   the SAME turn id + start timestamp that the live updates used and the
  159     --   whole-turn text last pushed during the turn (a structural dedup: settle
  160     --   fires once per Thinking→Idle edge and the turn id is retired at settle,
  161     --   so no content-compare is needed). Production wires this to append a
  162     --   @Response@ entry to the session transcript (via a broadcasting file
  163     --   transcript handle); tests inject a recorder.
  164   , _rd_publishUpdate :: SessionId -> TranscriptEntry -> IO ()
  165     -- ^ Live in-place turn update (Task 6). Publishes an EPHEMERAL @Response@
  166     --   entry (same stable turn id) each time the in-progress whole turn grows,
  167     --   so subscribers can render the message growing in place before it
  168     --   finalizes. Production wires this to an 'EntryUpdated' broker publish (not
  169     --   persisted); tests inject a recorder. Default is a no-op.
  170   , _rd_mintTurn :: IO (Text, UTCTime)
  171     -- ^ Mint a fresh turn id + start timestamp for a new turn (Task 6).
  172     --   Production: a random UUID + 'getCurrentTime'; tests inject fixed values.
  173   , _rd_providerFor :: Reg.HarnessEntry -> IO TurnProvider
  174     -- ^ Select the turn-content provider for an entry (Harness log content
  175     --   source, Task 4). The content-driven turn watcher sources a turn's text,
  176     --   finalize signal, and optional stable id from this provider instead of
  177     --   always from tmux's '_hh_snapshotTurn'. Default:
  178     --   @\\e -> pure (maybe nullProvider tmuxProvider (Reg._he_handle e))@ —
  179     --   a handle-attached entry gets the verbatim tmux provider; a handle-less
  180     --   entry gets the null provider. The production log provider (the
  181     --   spawned-claude-with-uuid path) is injected here in @CLI.Commands@. The
  182     --   selection is a deterministic function of the entry, so re-selecting it
  183     --   each tick latches the same provider for the duration of a turn.
  184   }

  186 -- | Production dependency set. Sweeps the @\"pureclaw\"@ session plus any other
  187 -- session on the server; capture via 'captureWindowNamed' (classified by the
  188 -- per-flavour observer in 'classifyFromObserver');
  189 -- harness liveness via 'harnessPidOf'; legacy stamping via 'setWindowMarker'.
  190 defaultReconcileDeps :: ReconcileDeps
  191 defaultReconcileDeps = ReconcileDeps
  192   { _rd_sessions = do
  193       others <- listTmuxSessions
  194       -- Always include the default session even if it has no windows yet.
  195       pure (dedup ("pureclaw" : others))
  196   , _rd_sweep = readMarkers
  197   , _rd_capture = \session windowName -> do
  198       r <- try @SomeException (captureWindowNamed session windowName 50)
  199       pure $ case r of
  200         Left _      -> Nothing
  201         Right bytes -> Just (TE.decodeUtf8Lenient bytes)
  202   , _rd_harnessAlive = \shellPid -> do
  203       -- Re-derive the harness PID by descending from the recorded shell PID and
  204       -- matching the flavour comm. If the descent still finds the agent binary
  205       -- the harness is alive; if it has exited, the descent returns 'Nothing'
  206       -- (the shell may still be present — that is exactly the @Exited@ case the
  207       -- @pane_dead@ flag also covers under @remain-on-exit on@).
  208       mPid <- harnessPidOf shellPid "claude"
  209       pure (isJust mPid)
  210   , _rd_stampLegacy = \session windowName ->
  211       if isLegacyWindowName windowName
  212         then do
  213           hid <- Reg.newHarnessId
  214           setWindowMarker session windowName (Reg.harnessIdToText hid)
  215           pure (Just hid)
  216         else pure Nothing
  217     -- Default eviction is a no-op: the loop itself deletes the evicted entry
  218     -- from the registry; the legacy-map drop is wired in CLI.Commands (which
  219     -- overrides this seam via 'runReconcileLoopWith'). Crucially this never
  220     -- deletes @session.json@.
  221   , _rd_evict = \_hid _label -> pure ()
  222     -- Default recorder is a no-op: the production transcript wiring is injected
  223     -- in CLI.Commands (which has 'sessionsDir' + the broker in scope).
  224   , _rd_recordResponse = \_sid _entry -> pure ()
  225     -- Default publishUpdate is a no-op: the production broker publish is injected
  226     -- in CLI.Commands (which has the broker in scope).
  227   , _rd_publishUpdate = \_sid _entry -> pure ()
  228     -- Default turn minting: a fresh random UUID + the current wall-clock time.
  229   , _rd_mintTurn = (,) . UUID.toText <$> UUID.nextRandom <*> getCurrentTime
  230     -- Default provider selection: the verbatim tmux provider for a
  231     -- handle-attached entry, the null provider for a handle-less one.
  232   , _rd_providerFor = pure . maybe nullProvider tmuxProvider . Reg._he_handle
  233   }
  234   where
  235     dedup = go []
  236       where
  237         go seen [] = reverse seen
  238         go seen (x : xs)
  239           | x `elem` seen = go seen xs
  240           | otherwise     = go (x : seen) xs

  246 -- | The result of corroborating a marker row against a registry entry.
  247 data CorroborationResult
  248   = Corroborated
  249     -- ^ The row is genuinely this entry's window: PID-provenance (or, for a
  250     --   provenance-less entry, the window-name prefix) confirms it.
  251   | PidMismatch
  252     -- ^ The marker matches but the recorded shell PID does not (or the second
  253     --   signal fails) — a spoof\/PID-reuse: treat as NOT ours, never trust.
  254   deriving stock (Eq, Show)
  255 
  256 -- | Corroborate a marker row (already known to carry this entry's @\@pcl_id@)
  257 -- against the entry's recorded provenance (§8 C4):
  258 --
  259 --   * If we recorded a shell PID, the row's @#{pane_pid}@ MUST match it.
  260 --   * If we recorded NO shell PID (legacy\/adopted), require a SECOND signal —
  261 --     the @claude-code-\<idx\>@-style window-name prefix matching the entry's
  262 --     label — to defend against PID reuse.
  263 corroborate :: Reg.HarnessEntry -> TmuxWindowRow -> CorroborationResult
  264 corroborate e row =
  265   case Reg._he_shellPid e of
  266     Just recordedPid
  267       | _twr_panePid row == Just recordedPid -> Corroborated
  268       | otherwise                            -> PidMismatch
  269     Nothing
  270       -- No recorded provenance: fall back to the window-name prefix as the
  271       -- second corroborating signal.
  272       | _twr_windowName row == Reg._he_label e -> Corroborated
  273       | otherwise                              -> PidMismatch

  275 -- ---------------------------------------------------------------------------
  276 -- Pure liveness classification (D5.4)
  277 -- ---------------------------------------------------------------------------
  278 
  279 -- | Classify a corroborated window's liveness from @(paneDead, harnessAlive,
  280 -- stable, screen)@ via the entry's per-flavour 'Obs.HarnessObserver':
  281 --
  282 --   * @pane_dead@ OR a dead harness PID ⇒ 'Reg.LivenessExited' (the window is
  283 --     present but the agent is gone).
  284 --   * otherwise the observer's 3-state screen classifier decides:
  285 --     working ⇒ 'Reg.LivenessThinking', awaiting-input ⇒
  286 --     'Reg.LivenessAwaitingInput', idle ⇒ 'Reg.LivenessIdle' only once the
  287 --     capture is /stable/ across ticks (an unstable idle frame stays
  288 --     'Reg.LivenessThinking', defending against mid-spinner frames).
  289 --
  290 -- 'Reg.LivenessOrphaned' is NOT produced here — it is assigned to entries whose
  291 -- corroborated window is /absent/ from the sweep (see 'reconcileTick').
  292 classifyFromObserver
  293   :: Obs.HarnessObserver
  294   -> Bool  -- ^ pane_dead
  295   -> Bool  -- ^ harness PID alive
  296   -> Bool  -- ^ the capture is stable (unchanged since the previous tick)
  297   -> Text  -- ^ raw screen capture
  298   -> Reg.Liveness
  299 classifyFromObserver obs paneDead harnessAlive stable screen
  300   | paneDead || not harnessAlive = Reg.LivenessExited
  301   | otherwise = case Obs._ho_classify obs screen of
  302       Obs.HasWorking       -> Reg.LivenessThinking
  303       Obs.HasAwaitingInput -> Reg.LivenessAwaitingInput
  304       -- An idle-looking screen is only trusted once it is STABLE across ticks:
  305       -- a single idle-marker frame mid-stream is treated as still Thinking
  306       -- (the spinner may simply be between frames).
  307       Obs.HasIdle          -> if stable then Reg.LivenessIdle else Reg.LivenessThinking
  308 
  309 -- | Map a registry 'Reg.Liveness' to the broker's 'HarnessActivity' vocabulary.
  310 -- Both @Exited@ and @Orphaned@ collapse to 'HarnessStopped' (the frontend's
  311 -- richer state→glyph mapping is Phase 2).
  312 livenessToActivity :: Reg.Liveness -> HarnessActivity
  313 livenessToActivity Reg.LivenessIdle          = HarnessIdle
  314 livenessToActivity Reg.LivenessThinking      = HarnessThinking
  315 livenessToActivity Reg.LivenessAwaitingInput = HarnessNeedsInput
  316 livenessToActivity Reg.LivenessExited        = HarnessStopped
  317 livenessToActivity Reg.LivenessOrphaned      = HarnessStopped

  342 -- Content-driven turn watcher state
  343 -- ---------------------------------------------------------------------------

  345 -- | Per-live-entry turn state threaded by the reconcile loop. The watcher is
  346 -- driven by turn-content change + stabilization (not the liveness edge), so each
  347 -- entry carries the stable minted id of the turn currently streaming, the last
  348 -- whole-turn text observed, how many consecutive ticks that text has been
  349 -- unchanged, and whether the turn is still active (streaming, not yet
  350 -- finalized). After finalize the state is RETAINED (inactive) so the same
  351 -- already-recorded content is neither re-streamed nor re-recorded; a later change
  352 -- to different non-empty content mints a NEW turn.
  353 data TurnState = TurnState
  354   { _tsTurnId   :: !Text     -- ^ stable id: same across every update + the final.
  355   , _tsStarted  :: !UTCTime  -- ^ the turn's start timestamp (reused on finalize).
  356   , _tsLastText :: !Text     -- ^ last whole-turn text streamed\/observed.
  357   , _tsStable   :: !Int      -- ^ consecutive ticks '_tsLastText' was unchanged.
  358   , _tsActive   :: !Bool     -- ^ 'True' = streaming, not yet finalized.
  359   }
  360   deriving stock (Eq, Show)
  361 
  362 -- | Consecutive ticks a turn's whole-turn text must be unchanged (while the
  363 -- harness is Idle\/AwaitingInput) before the watcher finalizes it. At the
  364 -- production 2-second cadence this lands the final ~1 tick after the last content
  365 -- change; the two-capture stability gate in 'classifyRow' is a second guard.
  366 defaultSettleStableTicks :: Int
  367 defaultSettleStableTicks = 1

  369 -- ---------------------------------------------------------------------------
  370 -- Pure symmetric diff (D5.2)
  371 -- ---------------------------------------------------------------------------
  372 
  373 -- | Compute the @(sessionId, liveness)@ transitions between two ticks' liveness
  374 -- snapshots, keyed by 'Reg.HarnessId' (rendered as 'Text' for test ergonomics).
  375 -- Each value carries the SessionId the event should be published under plus the
  376 -- current liveness.
  377 --
  378 -- An entry whose liveness CHANGED (or that is newly present) emits its new
  379 -- liveness. Crucially, an entry that went 'Reg.LivenessOrphaned'\/'Reg.LivenessExited'
  380 -- (because its window vanished from the sweep) is still present in @next@ — the
  381 -- registry retains it — so the transition surfaces, fixing the disappearance
  382 -- gap (@ActivityProbe.hs:117@). A dropped key (registry entry removed entirely)
  383 -- emits nothing; eviction is a Phase-2 concern.
  384 diffLiveness
  385   :: Map Text (Text, Reg.Liveness)  -- ^ previous tick
  386   -> Map Text (Text, Reg.Liveness)  -- ^ current tick
  387   -> [(Text, Reg.Liveness)]         -- ^ @(sessionId, newLiveness)@ to publish
  388 diffLiveness prev next =
  389   [ (sid, liveness)
  390   | (key, (sid, liveness)) <- Map.toList next
  391   , case Map.lookup key prev of
  392       Nothing            -> True                  -- newly observed
  393       Just (_, oldLive)  -> oldLive /= liveness   -- transitioned
  394   ]

  396 -- ---------------------------------------------------------------------------
  397 -- Reconcile tick (IO)
  398 -- ---------------------------------------------------------------------------
  399 
  400 -- | The per-entry observation a single 'reconcileTick' produces, keyed by the
  401 -- entry's 'Reg.HarnessId' text in the returned map. It carries the SessionId the
  402 -- loop publishes under, the classified liveness for the diff, and the raw screen
  403 -- capture taken this tick (so the loop can feed it back as next tick's @prevCap@
  404 -- for the stability gate). The held\/orphaned paths set '_to_capture' to
  405 -- 'Nothing' (no fresh capture was taken).
  406 data TickObservation = TickObservation
  407   { _to_sessionId :: !Text            -- ^ @sidOf e@ (the recorded sessionId, or the label fallback).
  408   , _to_liveness  :: !Reg.Liveness    -- ^ classified liveness this tick.
  409   , _to_capture   :: !(Maybe Text)    -- ^ raw capture taken this tick, if any.
  410   }
  411 
  412 -- | A single reconcile pass: sweep, corroborate, classify, merge, and return the
  413 -- per-entry observation map keyed by 'Reg.HarnessId' text (for the loop's diff
  414 -- + stability gate). Takes the previous tick's raw captures (id-text → last raw)
  415 -- so the per-flavour observer can apply its stability gate. On a transient sweep
  416 -- failure the registry's entries are marked '_he_stale' (holding last-known
  417 -- liveness) and the observation reflects the held state; the tick never throws
  418 -- on a sweep failure.
  419 reconcileTick
  420   :: ReconcileDeps
  421   -> Reg.HarnessRegistry
  422   -> LogHandle
  423   -> Map Text Text                          -- ^ previous captures (id-text → last raw)
  424   -> IO (Map Text TickObservation, [Text])
  425   -- ^ @(per-id observation, sessionIds of entries auto-evicted this tick)@.
  426   --   Evicted entries are removed from the observation map (so the next diff
  427   --   does not re-report them) and the loop emits one final disappearance event
  428   --   per returned sessionId.
  429 reconcileTick deps reg logger prevCaps = do
  430   sessions <- _rd_sessions deps
  431   -- Sweep every session; a failed sweep yields Nothing (we then hold stale).
  432   sweepResults <- forM sessions $ \session -> do
  433     r <- try @SomeException (_rd_sweep deps session)
  434     case r of
  435       Left err -> do
  436         _lh_logWarn logger
  437           ("reconcile: sweep of session " <> session <> " failed: " <> T.pack (show err))
  438         pure Nothing
  439       Right rows -> pure (Just rows)
  440   let sweepFailed = Nothing `elem` sweepResults
  441       allRows     = concat (catMaybes sweepResults)
  442   entries <- Reg.snapshot reg
  443   -- Build, per entry, its observed fields (or the stale/orphaned fallback).
  444   observedPairs <- forM entries $ \e -> do
  445     let idText  = Reg.harnessIdToText (Reg._he_id e)
  446         matches = [ row | row <- allRows, _twr_pclId row == idText ]
  447     case matches of
  448       [] | sweepFailed ->
  449             -- No row AND a sweep failed: we cannot conclude the window is gone.
  450             -- Hold last-known liveness; mark stale.
  451             pure (mkObservedHeld e)
  452          | otherwise ->
  453             -- Sweep succeeded and our window is absent ⇒ Orphaned.
  454             pure (mkObservedOrphaned e)
  455       _ -> do
  456         -- Corroborate each candidate; trust only the first corroborated row.
  457         let corroboratedRows =
  458               [ row | row <- matches, corroborate e row == Corroborated ]
  459         case corroboratedRows of
  460           [] -> do
  461             -- A marker matched but corroboration failed (spoof / PID mismatch):
  462             -- log + treat as not ours; the entry is Orphaned, never captured.
  463             _lh_logWarn logger
  464               ("reconcile: @pcl_id " <> idText
  465                  <> " matched a window with no corroborating PID — treating as not ours (§8 C4)")
  466             pure (mkObservedOrphaned e)
  467           (row : _) -> do
  468             -- The per-window capture + harness-alive probe is the only other
  469             -- tmux IO in the tick (besides the sweep, handled above). A
  470             -- transient failure here must NOT kill the loop (D5.3): hold this
  471             -- entry's last-known liveness and mark it stale, exactly as a sweep
  472             -- failure does — never repaint it from a capture error, never let
  473             -- the exception propagate to the loop's crash handler.
  474             r <- try @SomeException (classifyRow deps e row (Map.lookup idText prevCaps))
  475             case r of
  476               Left err -> do
  477                 _lh_logWarn logger
  478                   ("reconcile: capture/liveness probe for @pcl_id " <> idText
  479                      <> " failed: " <> T.pack (show err) <> " — holding last-known liveness")
  480                 pure (mkObservedHeld e)
  481               Right (liveness, mCap) ->
  482                 pure (mkObservedLive e row liveness mCap)
  483   let observed = map snd observedPairs
  484   Reg.mergeReconcile reg observed
  485   let snapMap = Map.fromList [ (k, v) | (k, v) <- map fst observedPairs ]
  486   -- Orphan grace policy (design §5 / §10 Q2): after the merge, evict any entry
  487   -- that has now been Orphaned for 'defaultOrphanGraceTicks' consecutive ticks.
  488   -- Eviction drops the entry from the registry (here) and from the legacy map
  489   -- (via the injectable '_rd_evict' seam, wired in CLI.Commands) — but never
  490   -- touches @session.json@. The evicted ids are removed from the returned
  491   -- snapshot so the next tick's diff does not re-report them; their sessionIds
  492   -- are returned so the loop can emit one final disappearance event.
  493   merged <- Reg.snapshot reg
  494   let toEvict =
  495         [ e | e <- merged, Reg._he_orphanedTicks e >= defaultOrphanGraceTicks ]
  496   evictedSids <- forM toEvict $ \e -> do
  497     _lh_logInfo logger
  498       ("reconcile: evicting harness " <> Reg.harnessIdToText (Reg._he_id e)
  499          <> " (label " <> Reg._he_label e <> ") after "
  500          <> T.pack (show (Reg._he_orphanedTicks e))
  501          <> " orphaned ticks; session.json retained")
  502     _rd_evict deps (Reg._he_id e) (Reg._he_label e)
  503     Reg.deleteEntry reg (Reg._he_id e)
  504     pure (Reg.harnessIdToText (Reg._he_id e), sidOf e)
  505   let evictedKeys = map fst evictedSids
  506       snapMap'    = foldr Map.delete snapMap evictedKeys
  507   pure (snapMap', map snd evictedSids)
  508   where
  509     -- Each result is ((keyText, TickObservation), ObservedHarness). The held and
  510     -- orphaned paths take no fresh capture, so '_to_capture' is 'Nothing'.
  511     mkObservedHeld :: Reg.HarnessEntry -> ((Text, TickObservation), Reg.ObservedHarness)
  512     mkObservedHeld e =
  513       let live = Reg._he_liveness e
  514       in ( (Reg.harnessIdToText (Reg._he_id e), tickObs e live Nothing)
  515          , (baseObserved e live) { Reg._oh_stale = True } )
  516 
  517     mkObservedOrphaned :: Reg.HarnessEntry -> ((Text, TickObservation), Reg.ObservedHarness)
  518     mkObservedOrphaned e =
  519       ( (Reg.harnessIdToText (Reg._he_id e), tickObs e Reg.LivenessOrphaned Nothing)
  520       , (baseObserved e Reg.LivenessOrphaned)
  521           { Reg._oh_liveness = Reg.LivenessOrphaned
  522             -- The corroborated window is absent this tick: advance the
  523             -- consecutive-orphaned counter so the grace policy can fire.
  524           , Reg._oh_orphanedTicks = Reg._he_orphanedTicks e + 1
  525           } )
  526 
  527     mkObservedLive
  528       :: Reg.HarnessEntry -> TmuxWindowRow -> Reg.Liveness -> Maybe Text
  529       -> ((Text, TickObservation), Reg.ObservedHarness)
  530     mkObservedLive e row liveness mCap =
  531       let extMod = _twr_windowName row /= Reg._he_windowName e
  532       in ( (Reg.harnessIdToText (Reg._he_id e), tickObs e liveness mCap)
  533          , Reg.ObservedHarness
  534              { Reg._oh_id          = Reg._he_id e
  535              , Reg._oh_session     = Reg._he_session e
  536              , Reg._oh_windowName  = _twr_windowName row
  537              , Reg._oh_shellPid    = _twr_panePid row
  538              , Reg._oh_harnessPid  = Reg._he_harnessPid e
  539              , Reg._oh_liveness    = liveness
  540              , Reg._oh_extModified = Reg._he_extModified e || extMod
  541              , Reg._oh_stale       = False
  542                -- The window is present + corroborated this tick: the entry is
  543                -- live again, so reset the consecutive-orphaned counter.
  544              , Reg._oh_orphanedTicks = 0
  545              } )
  546 
  547     -- A held-state observed record reusing the entry's current coordinate.
  548     baseObserved :: Reg.HarnessEntry -> Reg.Liveness -> Reg.ObservedHarness
  549     baseObserved e liveness = Reg.ObservedHarness
  550       { Reg._oh_id          = Reg._he_id e
  551       , Reg._oh_session     = Reg._he_session e
  552       , Reg._oh_windowName  = Reg._he_windowName e
  553       , Reg._oh_shellPid    = Reg._he_shellPid e
  554       , Reg._oh_harnessPid  = Reg._he_harnessPid e
  555       , Reg._oh_liveness    = liveness
  556       , Reg._oh_extModified = Reg._he_extModified e
  557       , Reg._oh_stale       = False
  558         -- Hold the current counter: a held (stale) observation neither advances
  559         -- nor resets the grace clock; only a definite Orphaned/live tick does.
  560       , Reg._oh_orphanedTicks = Reg._he_orphanedTicks e
  561       }
  562 
  563     tickObs :: Reg.HarnessEntry -> Reg.Liveness -> Maybe Text -> TickObservation
  564     tickObs e live mCap = TickObservation
  565       { _to_sessionId = sidOf e
  566       , _to_liveness  = live
  567       , _to_capture   = mCap
  568       }
  569 
  570     sidOf :: Reg.HarnessEntry -> Text
  571     sidOf e = fromMaybe (Reg._he_label e) (Reg._he_sessionId e)
  572 
  573 -- | Classify a corroborated window's liveness, capturing the screen only when
  574 -- the window is alive (not pane_dead and the harness PID is present). A
  575 -- @pane_dead@ window is NEVER captured (D5.4\/D5.5).
  576 --
  577 -- Takes the entry's PREVIOUS raw capture (the stability gate input) and returns
  578 -- @(liveness, capture-taken-this-tick)@ so the loop can carry the capture
  579 -- forward as the next tick's @prevCap@. A failed/skipped capture returns
  580 -- 'Nothing', which disables the stability gate for that tick.
  581 classifyRow
  582   :: ReconcileDeps
  583   -> Reg.HarnessEntry
  584   -> TmuxWindowRow
  585   -> Maybe Text                    -- ^ previous capture for this entry
  586   -> IO (Reg.Liveness, Maybe Text)
  587 classifyRow deps e row prevCap
  588   | _twr_paneDead row = pure (Reg.LivenessExited, Nothing)
  589   | otherwise = do
  590       -- Liveness of the agent process: re-derive it by descending from the live
  591       -- shell PID (the row's @#{pane_pid}@). We only bother when we recorded a
  592       -- harness PID at spawn (a provenance-less/legacy entry has none, so we
  593       -- assume alive and rely on @pane_dead@ for Exited).
  594       alive <- case (Reg._he_harnessPid e, _twr_panePid row) of
  595         (Just _, Just shellPid) -> _rd_harnessAlive deps shellPid
  596         _                       -> pure True
  597       if not alive
  598         then pure (Reg.LivenessExited, Nothing)
  599         else do
  600           mCap <- _rd_capture deps (Reg._he_session e) (_twr_windowName row)
  601           let cap    = fromMaybe "" mCap
  602               -- The screen is "stable" only when we got a capture this tick AND
  603               -- it is byte-identical to the previous tick's capture.
  604               stable = isJust mCap && mCap == prevCap
  605               obs    = Obs.observerFor (Reg._he_flavour e)
  606           pure (classifyFromObserver obs False True stable cap, mCap)

  612 -- | Production reconcile loop: 2-second cadence, real tmux deps.
  613 runReconcileLoop :: StreamBroker -> Reg.HarnessRegistry -> LogHandle -> IO ()
  614 runReconcileLoop broker reg =
  615   runReconcileLoopWith defaultTickMicros defaultReconcileDeps reg broker
  616 
  617 -- | Production tick interval: 2 seconds (matches the legacy probe cadence).
  618 defaultTickMicros :: Int
  619 defaultTickMicros = 2_000_000
  620 
  621 -- | The orphan grace window, in consecutive reconcile ticks (design §5\/§10
  622 -- Q2). Once an entry has been classified 'Reg.LivenessOrphaned' for this many
  623 -- ticks in a row (i.e. '_he_orphanedTicks' reaches this value) the reconcile
  624 -- loop auto-evicts it from the registry and the legacy harness map and emits a
  625 -- final disappearance event — but never deletes @session.json@.
  626 --
  627 -- We count ticks rather than wall-clock so the policy is deterministic and
  628 -- test-injectable. At the production 2-second 'defaultTickMicros' cadence, 15
  629 -- ticks ≈ 30 seconds of continuous orphaning before eviction — long enough to
  630 -- ride out a transient tmux-server blip (which is held '_he_stale' and does
  631 -- NOT advance the counter), short enough that a genuinely gone harness clears
  632 -- from Active Tabs promptly.
  633 defaultOrphanGraceTicks :: Int
  634 defaultOrphanGraceTicks = 15
  635 
  636 -- | The reconcile loop with an explicit tick interval + injected deps. The
  637 -- first tick establishes the baseline (emits nothing); from the second tick on,
  638 -- one 'ActivityChanged' is published per entry whose liveness changed
  639 -- (including disappearances). Resilient: a transient sweep failure is absorbed
  640 -- by 'reconcileTick' (entries held stale); the loop only exits on
  641 -- 'AsyncCancelled', which it re-raises.
  642 runReconcileLoopWith
  643   :: Int
  644   -> ReconcileDeps
  645   -> Reg.HarnessRegistry
  646   -> StreamBroker
  647   -> LogHandle
  648   -> IO ()
  649 runReconcileLoopWith tickMicros deps reg broker logger = handle outer $ do
  650   threadDelay tickMicros
  651   -- Baseline tick: emits no transitions. Any eviction on this first tick still
  652   -- emits its final disappearance event (an eviction is not a diff transition).
  653   -- The first tick has no previous captures, so the stability gate sees nothing
  654   -- stable (a fresh idle frame reads Thinking until a second identical capture).
  655   (initial, evicted0) <- reconcileTick deps reg logger Map.empty
  656   publishEvictions evicted0
  657   -- The baseline tick has no previous liveness, so it is not a "transition"
  658   -- and never records a Response; turn tracking begins on the second tick.
  659   loop (livenessSnap initial) (capSnap initial) Map.empty
  660   where
  661     loop prev prevCaps turnMap = do
  662       threadDelay tickMicros
  663       (obs, evicted) <- reconcileTick deps reg logger prevCaps
  664       let next = livenessSnap obs
  665       forM_ (diffLiveness prev next) $ \(sid, liveness) ->
  666         _streamBroker_publish broker
  667           (ActivityChanged (SessionId sid) (SaHarnessStatus (livenessToActivity liveness)))
  668       -- One final disappearance event per auto-evicted entry. The entry was
  669       -- already Orphaned, so the diff above does NOT re-emit it; this guarantees
  670       -- the frontend sees it leave even though no liveness transition occurred.
  671       publishEvictions evicted
  672       -- Output watcher (WU1): a SINGLE content-driven pass over each entry's
  673       -- whole-turn snapshot streams 'EntryUpdated's as the turn grows and
  674       -- finalizes (records ONE Response) on content stabilization or a terminal
  675       -- transition — independent of whether a Thinking frame was ever sampled.
  676       turnMap' <- stepTurns obs turnMap
  677       loop next (capSnap obs) turnMap'
  678 
  679     -- One content-driven turn-watcher pass. For every live entry (a bound id
  680     -- whose liveness this tick is Thinking\/Idle\/AwaitingInput) we snapshot the
  681     -- whole turn and drive streaming + finalize off the CONTENT changing and
  682     -- stabilizing, using liveness only as a guard against premature finalize.
  683     -- Then, for any id with an ACTIVE turn that has gone terminal
  684     -- (Exited\/Orphaned) or vanished from this tick's observation, we finalize
  685     -- from the last streamed text. Each id's step is wrapped in
  686     -- 'try @SomeException' (D5.3 resilience): 'AsyncCancelled' is re-raised; any
  687     -- other failure is logged + skipped so siblings proceed and the loop lives.
  688     stepTurns obs turnMap = do
  689       tm1 <- foldM stepLive turnMap (streamableIds obs)
  690       foldM stepTerminal tm1 (terminalIds obs tm1)
  691       where
  692         stepLive tm (idText, liveness) =
  693           withBoundEntry tm idText $ \e realSid _hh -> do
  694             -- Latch: select the content provider for this entry. The selection
  695             -- is a deterministic function of the entry, so it is the same
  696             -- provider for every tick of the active turn (no mid-turn switch).
  697             provider <- _rd_providerFor deps e
  698             r <- try @SomeException (stepLiveEntry tm idText realSid provider liveness)
  699             resolveTry "update" idText tm r
  700 
  701         stepTerminal tm (idText, st) =
  702           withBoundEntry tm idText $ \_e realSid _hh ->
  703             stepTerminalEntry tm idText realSid st
  704 
  705     -- The content step for one live, bound entry. Snapshot the whole turn from
  706     -- the selected provider — @(turnText, finalized)@ — then:
  707     --   * empty\/whitespace → no-op (between turns);
  708     --   * new content (no state) → start turn (provider id or mint), publish, active;
  709     --   * changed + active → publish under the SAME id, stable 0 (then, if the
  710     --     provider authoritatively finalized this snapshot, record + retire);
  711     --   * changed + inactive (previous turn done) → start a NEW turn;
  712     --   * unchanged → stable+1; finalize the active turn when the provider says
  713     --     @finalized@ (AUTHORITATIVE — bypasses the idle-stability guard) OR when
  714     --     it is idle-stable (liveness ∈ {Idle,AwaitingInput} && stable ≥ N). The
  715     --     tmux provider never finalizes, so its idle-stability behavior is
  716     --     unchanged.
  717     stepLiveEntry tm idText realSid provider liveness = do
  718       (turn, finalized) <- _tp_snapshot provider
  719       if T.null (T.strip turn)
  720         then pure tm
  721         else case Map.lookup idText tm of
  722           Nothing -> startTurn tm idText realSid provider turn finalized
  723           Just st
  724             | turn /= _tsLastText st ->
  725                 if _tsActive st
  726                   then do
  727                     -- Changed mid-turn: stream under the same stable id.
  728                     _rd_publishUpdate deps (SessionId realSid)
  729                       (mkTurnEntry (_tsTurnId st) (_tsStarted st) turn)
  730                     let st' = st { _tsLastText = turn, _tsStable = 0 }
  731                     if finalized
  732                       then finalizeRecord realSid idText
  733                              (Map.insert idText st' { _tsActive = False } tm)
  734                              (mkTurnEntry (_tsTurnId st) (_tsStarted st) turn)
  735                       else pure (Map.insert idText st' tm)
  736                   -- A previous turn finalized; different content is a NEW turn.
  737                   else startTurn tm idText realSid provider turn finalized
  738             | _tsActive st
  739             , finalized
  740               || ( liveness `elem` [Reg.LivenessIdle, Reg.LivenessAwaitingInput]
  741                    && _tsStable st + 1 >= defaultSettleStableTicks ) ->
  742                 -- Authoritative finalize (provider @finalized@, any liveness) OR
  743                 -- unchanged + idle-stable: finalize the active turn exactly once.
  744                 -- Always mark inactive (success OR record-failure) so we never
  745                 -- re-record / log-spam on the next tick.
  746                 finalizeRecord realSid idText
  747                   (Map.insert idText st { _tsStable = _tsStable st + 1, _tsActive = False } tm)
  748                   (mkTurnEntry (_tsTurnId st) (_tsStarted st) turn)
  749             | otherwise ->
  750                 -- Unchanged but not yet finalizable (still Thinking, or not yet
  751                 -- stable enough, or already retired): just advance the counter.
  752                 pure (Map.insert idText st { _tsStable = _tsStable st + 1 } tm)
  753 
  754     -- Record a turn's final 'Response', wrapped in 'try @SomeException'
  755     -- (D5.3 resilience). The caller passes the ALREADY-retired (inactive) turn
  756     -- map so that, whether the record succeeds or throws a non-async error, the
  757     -- turn is not retried on the next tick (no log spam). 'AsyncCancelled' is
  758     -- always re-raised; any other failure is logged under the
  759     -- "settle snapshot\/record" label and the inactive map is returned.
  760     finalizeRecord realSid idText retiredTm entry = do
  761       r <- try @SomeException (_rd_recordResponse deps (SessionId realSid) entry)
  762       case r of
  763         Left err
  764           | Just AsyncCancelled <- fromException err -> throwIO err
  765           | otherwise -> do
  766               _lh_logWarn logger
  767                 ("reconcile: settle snapshot/record for " <> idText
  768                    <> " failed: " <> T.pack (show err) <> " — skipping")
  769               pure retiredTm
  770         Right () -> pure retiredTm
  771 
  772     -- Start a turn: take the stable id+timestamp from the provider's
  773     -- '_tp_turnId' if it supplies one (the log provider derives both from the
  774     -- JSONL event), else mint via '_rd_mintTurn' (the tmux path, unchanged).
  775     -- Publish the first update. If the provider already finalized this first
  776     -- snapshot, record it once and retire; otherwise insert it active.
  777     startTurn tm idText realSid provider turn finalized = do
  778       mDerived <- _tp_turnId provider
  779       (turnId, ts) <- maybe (_rd_mintTurn deps) pure mDerived
  780       _rd_publishUpdate deps (SessionId realSid) (mkTurnEntry turnId ts turn)
  781       if finalized
  782         then finalizeRecord realSid idText
  783                (Map.insert idText (TurnState turnId ts turn 0 False) tm)
  784                (mkTurnEntry turnId ts turn)
  785         else pure (Map.insert idText (TurnState turnId ts turn 0 True) tm)
  786 
  787     -- Terminal finalize for one bound entry whose ACTIVE turn went terminal or
  788     -- vanished. 'terminalIds' only selects active turns, and a turn is only ever
  789     -- made active via 'startTurn' (which requires non-empty content), so an
  790     -- active turn always has a non-empty '_tsLastText' to finalize from. Record
  791     -- from that last streamed text and retire. (A harness that exits having
  792     -- streamed nothing has NO active turn state at all, so it is not selected by
  793     -- 'terminalIds' and records nothing — see DoD #6.)
  794     stepTerminalEntry tm idText realSid st =
  795       finalizeRecord realSid idText
  796         (Map.insert idText st { _tsActive = False } tm)
  797         (mkTurnEntry (_tsTurnId st) (_tsStarted st) (_tsLastText st))
  798 
  799     -- Look up a bound entry (a real sessionId + an attached handle) and run the
  800     -- step; an unbound\/handle-less\/unparseable id is skipped (returns @tm@).
  801     withBoundEntry tm idText k =
  802       case Reg.parseHarnessId idText of
  803         Nothing  -> pure tm
  804         Just hid -> do
  805           mEntry <- Reg.lookupById reg hid
  806           case mEntry of
  807             Just e
  808               | Just realSid <- Reg._he_sessionId e
  809               , Just hh      <- Reg._he_handle e -> k e realSid hh
  810             _ -> pure tm
  811 
  812     -- Resolve a per-entry 'try': re-raise 'AsyncCancelled', log + fall back to
  813     -- the unchanged turn map on any other error, otherwise take the new map.
  814     resolveTry
  815       :: Text
  816       -> Text
  817       -> Map Text TurnState
  818       -> Either SomeException (Map Text TurnState)
  819       -> IO (Map Text TurnState)
  820     resolveTry what idText fallback r = case r of
  821       Left err
  822         | Just AsyncCancelled <- fromException err -> throwIO err
  823         | otherwise -> do
  824             _lh_logWarn logger
  825               ("reconcile: " <> what <> " for " <> idText
  826                  <> " failed: " <> T.pack (show err) <> " — skipping")
  827             pure fallback
  828       Right tm' -> pure tm'
  829 
  830     -- Ids whose liveness THIS tick is one we can stream\/finalize from:
  831     -- Thinking\/Idle\/AwaitingInput, paired with that liveness. Exited\/Orphaned
  832     -- are handled by the terminal pass.
  833     streamableIds obs =
  834       [ (idText, _to_liveness o)
  835       | (idText, o) <- Map.toList obs
  836       , _to_liveness o `elem`
  837           [Reg.LivenessThinking, Reg.LivenessIdle, Reg.LivenessAwaitingInput]
  838       ]
  839 
  840     -- Ids (paired with their state) with an ACTIVE turn (after the live pass)
  841     -- whose current liveness is terminal (Exited\/Orphaned) OR which are absent
  842     -- from this tick's observation entirely (evicted\/orphaned-then-removed).
  843     terminalIds obs tm =
  844       [ (idText, st)
  845       | (idText, st) <- Map.toList tm
  846       , _tsActive st
  847       , case Map.lookup idText obs of
  848           Nothing -> True
  849           Just o  -> _to_liveness o `elem` [Reg.LivenessExited, Reg.LivenessOrphaned]
  850       ]
  851 
  852     -- Project the observation map into the (sessionId, liveness) snapshot the
  853     -- 'diffLiveness' publish path consumes, and into the carry-forward captures.
  854     livenessSnap :: Map Text TickObservation -> Map Text (Text, Reg.Liveness)
  855     livenessSnap = Map.map (\o -> (_to_sessionId o, _to_liveness o))
  856 
  857     capSnap :: Map Text TickObservation -> Map Text Text
  858     capSnap = Map.mapMaybe _to_capture
  859 
  860     publishEvictions = mapM_ $ \sid ->
  861       _streamBroker_publish broker
  862         (ActivityChanged (SessionId sid) (SaHarnessStatus (livenessToActivity Reg.LivenessOrphaned)))
  863 
  864     -- AsyncCancelled MUST be re-raised so 'Async.withAsync'/'Async.cancel'
  865     -- complete cleanly (project-wide invariant).
  866     outer :: SomeException -> IO ()
  867     outer ex
  868       | Just AsyncCancelled <- fromException ex = throwIO ex
  869       | otherwise = _lh_logError logger
  870           ("reconcile loop crashed: " <> T.pack (show ex))

  872 -- ---------------------------------------------------------------------------
  873 -- Boot reconstruction (D5.6)
  874 -- ---------------------------------------------------------------------------
  875 
  876 -- | At startup, run one sweep and register an entry for every window we own:
  877 --
  878 --   * a window carrying our @\@pcl_id@ → an 'Reg.OriginDiscovered' entry (the
  879 --     PCL-restart reconnect path);
  880 --   * a legacy @claude-code-\<idx\>@ window with no marker → lazily stamped a
  881 --     fresh id (via '_rd_stampLegacy') and registered as 'Reg.OriginDiscovered'.
  882 --
  883 -- The legacy harness map is seeded separately by the unchanged
  884 -- @discoverHarnesses@; this builds the parallel registry the reconcile loop
  885 -- owns. Idempotent-ish: an already-registered id is overwritten with the
  886 -- freshly observed coordinate (boot runs once).
  887 bootReconstruct :: ReconcileDeps -> Reg.HarnessRegistry -> LogHandle -> IO ()
  888 bootReconstruct deps reg logger = do
  889   sessions <- _rd_sessions deps
  890   forM_ sessions $ \session -> do
  891     r <- try @SomeException (_rd_sweep deps session)
  892     case r of
  893       Left err ->
  894         _lh_logWarn logger
  895           ("bootReconstruct: sweep of " <> session <> " failed: " <> T.pack (show err))
  896       Right rows -> forM_ rows $ \row -> do
  897         case Reg.parseHarnessId (_twr_pclId row) of
  898           Just hid ->
  899             registerEntry session row hid
  900           Nothing -> do
  901             mHid <- _rd_stampLegacy deps session (_twr_windowName row)
  902             for_ mHid (registerEntry session row)  -- 'Nothing' ⇒ not ours
  903   where
  904     registerEntry session row hid = do
  905       let live = if _twr_paneDead row then Reg.LivenessExited else Reg.LivenessIdle
  906       Reg.insertEntry reg Reg.HarnessEntry
  907         { Reg._he_id          = hid
  908         , Reg._he_session     = session
  909         , Reg._he_windowName  = _twr_windowName row
  910         , Reg._he_shellPid    = _twr_panePid row
  911         , Reg._he_harnessPid  = Nothing
  912         , Reg._he_origin      = Reg.OriginDiscovered
  913         , Reg._he_flavour     = Kind.HClaudeCode
  914         , Reg._he_liveness    = live
  915         , Reg._he_extModified = False
  916         , Reg._he_stale       = False
  917         , Reg._he_sessionId   = Nothing
  918         , Reg._he_label       = _twr_windowName row
  919         , Reg._he_orphanedTicks = 0
  920         , Reg._he_handle      = Nothing
  921         }

  927 -- | A legacy harness window name is @\<canonical\>-\<idx\>@ for a known flavour
  928 -- (Phase 1: only @claude-code@). Used by the default 'ReconcileDeps' to decide
  929 -- whether an unmarked window should be lazily stamped.
  930 isLegacyWindowName :: Text -> Bool
  931 isLegacyWindowName name =
  932   case T.stripPrefix "claude-code-" name of
  933     Just suffix -> not (T.null suffix) && T.all isDigit suffix
  934     Nothing     -> False
