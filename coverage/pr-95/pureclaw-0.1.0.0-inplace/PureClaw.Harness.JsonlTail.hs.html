    1 -- | Pure, dependency-free core of the JSONL session-log tail.
    2 --
    3 -- A claude-code session log is an append-only file of LF-terminated JSON
    4 -- records. A tailer reads it in arbitrary byte chunks as it grows; chunk
    5 -- boundaries fall wherever the OS happens to return, so a single logical line
    6 -- may be split across two (or more) reads. This module provides the pure
    7 -- splitting logic that turns a stream of byte chunks into a stream of complete
    8 -- lines, buffering any trailing partial line until its terminating LF arrives.
    9 --
   10 -- This is the pure foundation only: there is no IO, no dependency record, and
   11 -- no tailer loop here. WU5 extends this same module with @JsonlTailDeps@ and
   12 -- the loop that drives 'splitLines' over a growing file.
   13 module PureClaw.Harness.JsonlTail
   14   ( -- * Offsets
   15     Offset (..)
   16 
   17     -- * Complete lines
   18   , CompleteLine
   19   , unCompleteLine
   20 
   21     -- * Incremental line buffer
   22   , Buffer
   23   , emptyBuffer
   24   , unBuffer
   25 
   26     -- * Splitting
   27   , splitLines

   31   , splitLinesBounded
   32   ) where

   34 import Data.ByteString (ByteString)
   35 import Data.ByteString qualified as BS

   37 -- | A byte offset into a growing log file.
   38 --
   39 -- 'Integer' (not 'Int') because log files may in principle exceed a 64-bit
   40 -- offset, and because the tailer arithmetic should never silently overflow.
   41 -- The value constructor is exported so the (future) IO tailer can construct
   42 -- and pattern-match offsets directly; an offset carries no invariant beyond
   43 -- "a number of bytes".
   44 newtype Offset = Offset Integer
   45   deriving stock (Eq, Ord, Show)
   46 
   47 -- | One complete, LF-delimited line — the bytes /without/ the terminating
   48 -- @\\n@. A line may be empty (two consecutive newlines, or a leading newline).
   49 --
   50 -- A carriage return (@\\r@) is ordinary content: claude-code logs are
   51 -- LF-terminated, so a @\\r@ immediately before the @\\n@ stays inside these
   52 -- bytes (CRLF is /not/ treated specially). The value constructor is
   53 -- intentionally hidden — 'CompleteLine' values are only ever produced by
   54 -- 'splitLines', and consumers read the bytes back with 'unCompleteLine'.
   55 newtype CompleteLine = CompleteLine ByteString
   56   deriving stock (Eq, Ord, Show)
   57 
   58 -- | The raw bytes of a complete line, without its terminating LF.
   59 unCompleteLine :: CompleteLine -> ByteString
   60 unCompleteLine (CompleteLine bs) = bs
   61 
   62 -- | Carries the trailing partial (un-terminated) bytes between successive
   63 -- reads. After a chunk is processed, everything after the last @\\n@ is held
   64 -- here and prepended to the next incoming chunk.
   65 --
   66 -- The value constructor is hidden: a 'Buffer' is only ever produced by
   67 -- 'emptyBuffer' or threaded out of 'splitLines'. Use 'unBuffer' to inspect the
   68 -- pending bytes (primarily for tests and diagnostics).
   69 newtype Buffer = Buffer ByteString
   70   deriving stock (Eq, Ord, Show)
   71 
   72 -- | The initial, empty buffer — no pending partial line.
   73 emptyBuffer :: Buffer
   74 emptyBuffer = Buffer BS.empty
   75 
   76 -- | The pending (un-terminated) bytes currently held in the buffer.
   77 unBuffer :: Buffer -> ByteString
   78 unBuffer (Buffer bs) = bs
   79 
   80 -- | Split an incoming byte chunk into complete LF-delimited lines plus a
   81 -- residual buffer.
   82 --
   83 -- The incoming 'Buffer' (the partial line left over from previous reads) is
   84 -- prepended to the chunk, so a line split across reads reassembles once its
   85 -- @\\n@ arrives. Every complete line (the bytes up to, but not including, each
   86 -- @\\n@) is emitted in order; the bytes after the final @\\n@ — which have no
   87 -- terminator yet — are carried in the returned 'Buffer' and are /not/ emitted.
   88 --
   89 -- Edge-case semantics (all pinned down by tests):
   90 --
   91 --   * __Empty input__: no lines; the buffer is returned unchanged.
   92 --   * __No LF__: the whole (buffer @<>@ chunk) is buffered; no lines.
   93 --   * __Trailing LF__ (@"a\\n"@): emits @["a"]@, residual empty.
   94 --   * __Leading LF__ (@"\\nx\\n"@): emits @["", "x"]@ (leading empty line).
   95 --   * __Consecutive LFs__ (@"a\\n\\nb\\n"@): emits @["a", "", "b"]@.
   96 --   * __Lone LF__ (@"\\n"@): emits a single empty line @[""]@.
   97 --   * __CR preserved__ (@"a\\r\\nb\\n"@): emits @["a\\r", "b"]@ — the @\\r@
   98 --     stays on the first line; CRLF is not special-cased.
   99 --
  100 -- Total function: no partial functions, no exceptions, for any input.
  101 splitLines :: ByteString -> Buffer -> ([CompleteLine], Buffer)
  102 splitLines chunk (Buffer pending) = (map CompleteLine completed, Buffer residual)
  103   where
  104     -- Reassemble: the buffered partial precedes the fresh bytes.
  105     combined :: ByteString
  106     combined = pending <> chunk
  107 
  108     -- 'BS.split' on a stream containing @n@ LF separators yields @n + 1@
  109     -- pieces (one per inter-separator gap, including empties). The trailing
  110     -- piece is everything after the final LF — un-terminated — so it becomes
  111     -- the residual; the earlier pieces are the complete lines.
  112     --
  113     -- The one exception is /empty/ input: 'BS.split' returns @[]@ (zero
  114     -- pieces) for the empty 'ByteString', which 'splitLast' maps to no lines
  115     -- and an empty residual. (For any non-empty input 'BS.split' is non-empty.)
  116     (completed, residual) = splitLast (BS.split 0x0A combined)
  117 
  118 -- | Separate the pieces of a 'BS.split' into the complete lines (all but the
  119 -- last piece) and the residual partial (the last piece). For the @[]@ result
  120 -- that 'BS.split' produces on empty input, there are no lines and the residual
  121 -- is empty. Total in all cases.
  122 splitLast :: [ByteString] -> ([ByteString], ByteString)
  123 splitLast [] = ([], BS.empty)
  124 splitLast [x] = ([], x)
  125 splitLast (x : xs) =
  126   let (initLines, lst) = splitLast xs
  127    in (x : initLines, lst)
  128 
  129 -- | The only cap-exceeded outcome from 'splitLinesBounded'.
  130 data SplitCap = OverCap
  131   deriving stock (Eq, Show)
  132 
  133 -- | Like 'splitLines', but fails closed if the residual partial line would
  134 -- exceed @maxBuffer@ bytes (a no-LF flood). The cap is checked BEFORE the split
  135 -- so an over-cap chunk never transiently allocates the concatenation.
  136 splitLinesBounded :: Int -> ByteString -> Buffer -> Either SplitCap ([CompleteLine], Buffer)
  137 splitLinesBounded maxBuffer chunk buf@(Buffer pending)
  138   -- If there is no LF anywhere in chunk and pending<>chunk already exceeds the
  139   -- cap, reject without building the full residual.
  140   | not (0x0A `BS.elem` chunk)
  141   , BS.length pending + BS.length chunk > maxBuffer = Left OverCap
  142   | otherwise =
  143       let (ls, resid@(Buffer r)) = splitLines chunk buf
  144        in if BS.length r > maxBuffer then Left OverCap else Right (ls, resid)
