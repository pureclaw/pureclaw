    5 -- Description : Live wiring of the tab subsystem (Tabs-as-View 8c.2 flip).
    6 --
    7 -- This module is the 8c.2 /wiring flip/ (GitHub #79; execution-binding spike
    8 -- §3–§6). It binds the additive 8b\/8c.1 pieces — the per-'TabRef' runtime
    9 -- registry ("PureClaw.Tabs.Exec"), the real runtime constructors
   10 -- ("PureClaw.Tabs.Runtimes"), the output-side relay writer
   11 -- ("PureClaw.Tabs.RelayWriter"), and the per-conversation router
   12 -- ("PureClaw.Routing.TabDispatch") — to the live 'AgentEnv', and exposes
   13 -- 'runTabbedLoop', the new production entry point.
   14 --
   15 -- 'runTabbedLoop':
   16 --
   17 --   1. Forks the __relay-writer thread__ draining '_env_tabOutQ' through
   18 --      'RelayWriter.processOutput'.
   19 --   2. Registers the CLI conversation's __output sink__ (its
   20 --      'ConversationKey' -> '_env_channel') in '_env_sinks'.

   26 -- 'Runtimes.mkProviderRuntime' for a @BoundSession@ (pooled 'SessionHandle',
   27 -- context seeded from the transcript, transcript-recording provider stream,
   28 -- tool exec over the env registry) and a 'Runtimes.mkHarnessRuntime' for a
   29 -- @BoundHarness@ (the 'HarnessHandle' resolved from '_env_harnessRegistry').

  130 -- | A tiny @'Core.SessionId' -> 'Session.SessionHandle'@ pool, owned by one
  131 -- 'runTabbedLoop' invocation. The default-session minter ('mkNewDefaultSession')
  132 -- inserts a freshly-opened handle keyed by its id; the provider runtime
  133 -- constructor ('mkExecDeps') reads it back to seed + record against the same
  134 -- on-disk session. This is the 8c.2 stand-in for the full
  135 -- "PureClaw.Tabs.SessionPool" wiring (which lands with detach\/refcount work);
  136 -- here it is a write-on-create, read-on-bind map — adequate for the CLI flip,
  137 -- where every bound session was minted by '_td_newDefaultSession'.
  138 type SessionStore = IORef (Map Core.SessionId Session.SessionHandle)

  144 -- | The Tabs-as-View production loop (replaces 'runAgentLoopWith' at the 8c.2
  145 -- flip). Forks the relay-writer thread, registers the CLI sink, then reads the
  146 -- channel and dispatches each inbound message per conversation.
  147 runTabbedLoop
  148   :: AgentEnv
  149   -> SessionStore
  150   -> (Core.SessionId -> Maybe Text -> IO (Either Text ()))
  151      -- ^ The session-description seam (built per 'ServerMode' in
  152      -- "PureClaw.CLI.Commands"); threaded into the dispatcher deps so @\/rename@
  153      -- on a session tab sets the session's canonical, cross-process name.
  154   -> IO ()
  155 runTabbedLoop env store setSessionDescription = do
  156   _lh_logInfo logger "Tabbed agent loop started"
  157   let execDeps   = mkExecDeps env store
  158       relayDeps  = RelayWriterDeps
  159         { _rw_sinks   = _env_sinks env
  160         , _rw_cursors = readIORef (_env_cursors env)
  161         , _rw_tabs    = Tabs.readTabs (_env_tabRegistry env)
  162         , _rw_default = defaultRelayMode
  163         , _rw_modelOf = modelOf store
  164         }
  165   -- (a) Fork the relay-writer thread draining the ref-tagged tab-output queue.
  166   _ <- _env_fork env $ forever $ do
  167     (ref, ev) <- atomically (readTBQueue (_env_tabOutQ env))
  168     processOutput relayDeps (_env_relayWriter env) ref ev
  169   -- The dispatcher deps are built once; they close over the live env + store.
  170   let dispatchDeps = mkTabDispatchDeps env execDeps store setSessionDescription
  171   -- Read the first inbound message to learn the CLI conversation key, register
  172   -- its sink, then loop. (b) + (c).
  173   loop dispatchDeps
  174   where
  175     channel = _env_channel env
  176     logger  = _env_logger env
  177 
  178     loop dispatchDeps = do
  179       receiveResult <- try @E.IOException (_ch_receive channel)
  180       case receiveResult of
  181         Left _    -> _lh_logInfo logger "Session ended"
  182         Right msg -> do
  183           let src     = _im_source msg
  184               convKey = conversationKeyOf src
  185           -- (b) Register this conversation's output sink (idempotent).
  186           registerSink (_env_sinks env) convKey channel
  187           -- Capture the session origin (set-once) on the FOREGROUND session,
  188           -- mirroring the legacy loop (the no-active-tab case relies on this —
  189           -- Tabs.WiringSpec).
  190           fgSh <- readIORef (_env_session env)
  191           Session.setSourceIfAbsent fgSh src
  192           -- Also capture it (set-once) on the conversation's ACTIVE BOUND
  193           -- session, so provenance lands on the session the turn actually runs
  194           -- in (pureclaw-opr; #79). No active tab / non-session ref → no-op.
  195           captureBoundSource env store convKey src
  196           -- (c) Dispatch per conversation — skip empty/whitespace input silently,
  197           -- mirroring legacy dispatchMsg's T.null(T.strip) guard (pureclaw-z9h).
  198           -- Source capture above is preserved even for empty content.
  199           if T.null (T.strip (_im_content msg))
  200             then loop dispatchDeps
  201             else do
  202               Dispatch.handleInbound dispatchDeps convKey (_im_content msg)
  203               loop dispatchDeps

  205 -- | The 'ConversationKey' for a message source: its channel + conversation id
  206 -- (invariant I3 — the dispatcher keys cursors\/relay by this pair).
  207 conversationKeyOf :: Core.MessageSource -> ConversationKey
  208 conversationKeyOf src = (Core._ms_channel src, Core._ms_conversation src)
  209 
  210 -- | Capture the inbound message source (set-once) onto the conversation's
  211 -- ACTIVE BOUND session — the session the turn actually runs in. Resolves the
  212 -- active tab exactly as 'TabDispatch.doDefault' does (cursor -> ref -> tab) and,
  213 -- when the ref is a @BoundSession@, resolves its 'Session.SessionHandle' from
  214 -- the loop's @store@ and applies 'Session.setSourceIfAbsent'. No active tab (no
  215 -- cursor / dangling cursor) or a non-session ref (@BoundHarness@) is a safe
  216 -- no-op. This restores the provenance the 8c cutover dropped (pureclaw-opr; #79)
  217 -- without disturbing the foreground @_env_session@ capture the no-tab case needs.
  218 captureBoundSource
  219   :: AgentEnv -> SessionStore -> ConversationKey -> Core.MessageSource -> IO ()
  220 captureBoundSource env store convKey src = do
  221   mSh <- activeBoundSession env store convKey
  222   case mSh of
  223     Just sh -> Session.setSourceIfAbsent sh src
  224     Nothing -> pure ()
  225 
  226 -- | Resolve the 'Session.SessionHandle' bound to a conversation's ACTIVE tab,
  227 -- if any. Follows the same cursor -> slot -> tab -> '_tab_ref' chain
  228 -- 'TabDispatch.doDefault' uses to pick the session a turn runs in, then — when
  229 -- the ref is a @BoundSession@ — resolves its handle from the loop's @store@. No
  230 -- active tab (no cursor / dangling cursor) or a non-session ref
  231 -- (@BoundHarness@) yields 'Nothing'. Shared by 'captureBoundSource' (provenance
  232 -- set-once) and 'fallthrough' (seeding the @\/status@ \/ @\/compact@ context
  233 -- from the live conversation's transcript — pureclaw-25k).
  234 activeBoundSession
  235   :: AgentEnv -> SessionStore -> ConversationKey -> IO (Maybe Session.SessionHandle)
  236 activeBoundSession env store convKey = do
  237   cs <- readIORef (_env_cursors env)
  238   tl <- Tabs.readTabs (_env_tabRegistry env)
  239   case resolveCursorSlot convKey cs tl of
  240     Nothing   -> pure Nothing
  241     Just slot -> do
  242       mTab <- Tabs.registryLookupSlot (_env_tabRegistry env) slot
  243       case _tab_ref <$> mTab of
  244         Just (BoundSession sid) -> Just <$> resolveSession env store sid
  245         _                       -> pure Nothing

  247 -- | The global default 'RelayMode' for conversations with no override.
  248 -- 'FocusedOnly' matches the design default (only the focused tab is relayed).
  249 defaultRelayMode :: RelayMode
  250 defaultRelayMode = FocusedOnly

  252 -- | Resolve a tab's model name for the relay's focused speaker prefix
  253 -- (@\/N \<model\>: @). A @BoundSession@ resolves its pooled 'Session.SessionHandle'
  254 -- from @store@ (the running tab's session is always cached there) and reads its
  255 -- current @_sm_model@; an empty model string and a @BoundHarness@ (no model)
  256 -- both yield 'Nothing', so the prefix renders slot-only (@\/N: @). This reads
  257 -- only the in-memory store + meta 'IORef' — no disk I/O on the relay path.
  258 modelOf :: SessionStore -> TabRef -> IO (Maybe Text)
  259 modelOf store = \case
  260   BoundSession sid -> do
  261     m <- readIORef store
  262     case Map.lookup sid m of
  263       Nothing -> pure Nothing
  264       Just sh -> do
  265         meta <- readIORef (Session._sh_meta sh)
  266         let model = SessionTypes._sm_model meta
  267         pure (if T.null model then Nothing else Just model)
  268   BoundHarness _ -> pure Nothing

  274 -- | Build the production 'ExecDeps': the '_ex_startRuntime' closure that
  275 -- constructs a provider or harness runtime for a 'TabRef' against the live
  276 -- 'AgentEnv'.
  277 --
  278 --   * @BoundSession sid@ — 'mkProviderRuntime' over the session's pooled
  279 --     'SessionHandle' (resolved from @store@), seeding 'Ctx.Context' from the
  280 --     transcript via 'Session.loadRecentMessages', streaming through a
  281 --     transcript-recording provider, executing tools against the env registry,
  282 --     and emitting @('TabRef','ChannelEvent')@ onto '_env_tabOutQ'.
  283 --   * @BoundHarness hid@ — 'mkHarnessRuntime' over the 'HarnessHandle' resolved
  284 --     from '_env_harnessRegistry'.
  285 mkExecDeps :: AgentEnv -> SessionStore -> ExecDeps
  286 mkExecDeps env store = ExecDeps { _ex_startRuntime = startRuntime }
  287   where
  288     startRuntime :: TabRef -> IO Runtime
  289     startRuntime = \case
  290       BoundSession sid -> startProvider env store sid
  291       BoundHarness hid -> startHarness env hid

  293 -- | Boot-time runtime restore. 'PureClaw.Tabs.Persist.loadTabs' restores the
  294 -- persisted 'TabList' into the registry ('PureClaw.Tabs.overwriteTabs'), but a
  295 -- restored tab is only /bound/ — the bind-time 'ensure' that live creation
  296 -- performs ('PureClaw.Routing.TabDispatch.bindNewTab' \/ @routeToNew@) never ran
  297 -- for it, so its 'Exec' runtime is absent. The first message routed to such a
  298 -- tab then fails with 'PureClaw.Handles.Tab.TabNotFound' (\"tab: not found\")
  299 -- instead of resuming the session — the symptom seen after a gateway restart.
  300 --
  301 -- This re-establishes the invariant by 'ensure'-ing each restored ref exactly
  302 -- once (balancing the one 'release' per @\/close@), so a 'BoundSession'
  303 -- reconnects with its transcript context reseeded ('startProvider') and a live
  304 -- 'BoundHarness' re-attaches. 'Dead' tombstones are skipped — they carry no
  305 -- runtime and drop input by design (Tabs-as-View §8); a dead harness is already
  306 -- pruned at load ('PureClaw.Tabs.Persist.reconcileTabs'), so its conversation
  307 -- falls through to a fresh default session instead.
  308 ensureRestoredRuntimes :: ExecDeps -> Exec -> TabList -> IO ()
  309 ensureRestoredRuntimes deps ex =
  310   mapM_ (ensure deps ex . _tab_ref) . filter ((/= Dead) . _tab_status) . toList

  318 -- | Build + start a provider-session runtime for a 'Core.SessionId'.
  319 startProvider :: AgentEnv -> SessionStore -> Core.SessionId -> IO Runtime
  320 startProvider env store sid = do
  321   sh <- resolveSession env store sid
  322   let th = Session._sh_transcript sh
  323   mProvider <- readIORef (_env_provider env)
  324   mModel    <- readIORef (_env_model env)
  325   let model     = fromMaybe (ModelId "") mModel
  326       modelName = unModelId model
  327   -- Wrap the provider with the per-session transcript recorder (records
  328   -- request/response entries to transcript.jsonl + broker), exactly as the
  329   -- legacy loop did. A 'Nothing' provider yields a stream that records nothing
  330   -- and produces no StreamDone (the turn returns the seeded context).
  331   --
  332   -- The transcript's Request @metadata.source@ is the BOUND session's current
  333   -- @_sm_source@, read per turn: 'runTabbedLoop' captures the inbound source
  334   -- set-once onto this session before the worker runs, so this restores the
  335   -- legacy loop's @(Just (_im_source msg))@ semantics (pureclaw-opr; #79).
  336   let streamFn req cb = case mProvider of
  337         Nothing -> pure ()
  338         Just provider -> do
  339           meta <- readIORef (Session._sh_meta sh)
  340           let provider' =
  341                 mkTranscriptProvider th modelName (SessionTypes._sm_source meta) provider
  342           P.completeStream provider' req cb
  343   let deps = ProviderRuntimeDeps
  344         { _prd_ref          = BoundSession sid
  345         , _prd_emit         = enqueueTabOut env
  346         , _prd_stream       = streamFn
  347         , _prd_execTool     = execOneTool env
  348         , _prd_record       = \_ -> pure ()
  349             -- Transcript append is performed by the transcript-recording
  350             -- provider wrapper above (request/response entries); the
  351             -- message-level append is a no-op, matching the legacy Tab.Ai path.
  352         , _prd_seedCtx      = seedContextFrom env th
  353         , _prd_model        = model
  354         , _prd_systemPrompt = _env_systemPrompt env
  355         , _prd_tools        = registryDefinitions <$> effectiveRegistry env
  356         , _prd_maxTokens    = Just 4096
  357           -- Restore the legacy @Loop.handleCompletion@ one-shot: fire (and
  358           -- clear) @_env_onFirstStreamDone@ on the FIRST provider StreamDone, so
  359           -- @markBootstrapConsumed@ runs once and an agent's BOOTSTRAP.md is not
  360           -- re-injected on every resume (pureclaw-8g4; #79). The tabbed path
  361           -- never fired this before the fix.
  362         , _prd_onStreamDone = fireOnce (_env_onFirstStreamDone env)
  363         , _prd_fork         = _env_fork env
  364         , _prd_inputBound   = RT._rc_inputQueueBound (_env_routingConfig env)
  365         }
  366   mkProviderRuntime deps
  367 
  368 -- | Run a one-shot action stored in an 'IORef' at most once: atomically
  369 -- read-and-clear the 'Maybe' action, then run it if present. Mirrors the
  370 -- legacy @Loop.handleCompletion@ StreamDone handler
  371 -- (@atomicModifyIORef' ref (Nothing,); for_ mAction id@) so the armed
  372 -- @markBootstrapConsumed@ fires exactly once even though '_turn_onStreamDone'
  373 -- is invoked on every 'P.StreamDone' (pureclaw-8g4).
  374 fireOnce :: IORef (Maybe (IO ())) -> IO ()
  375 fireOnce ref = do
  376   mAction <- atomicModifyIORef' ref (Nothing,)
  377   fromMaybe (pure ()) mAction

  379 -- | Build + start a harness runtime for a 'Registry.HarnessId'. The
  380 -- 'HarnessHandle' is resolved from '_env_harnessRegistry'; when the entry has
  381 -- no live handle yet (boot-discovered, not attached), a no-op handle is used so
  382 -- the runtime starts cleanly and simply produces no output until reconcile
  383 -- attaches one.
  384 startHarness :: AgentEnv -> Registry.HarnessId -> IO Runtime
  385 startHarness env hid = do
  386   mEntry <- Registry.lookupById (_env_harnessRegistry env) hid
  387   let mHandle = mEntry >>= Registry._he_handle
  388   case mHandle of
  389     Just h  -> startHarnessWith env hid h
  390     Nothing ->
  391       -- TODO(8c.3): resolve a freshly-attached handle once detach/re-attach
  392       -- lifecycle lands; for now an unattached harness ref produces no output.
  393       startHarnessWith env hid noOutputHarness

  395 -- | The harness-runtime build over a resolved 'HarnessHandle'.

  398   mkHarnessRuntime HarnessRuntimeDeps

  402     , _hrd_fork       = _env_fork env
  403     , _hrd_pollMicros = 100000  -- 100ms drainer poll, matching the legacy loop

  418 -- | Seed a fresh 'Ctx.Context' for a provider runtime from its session's
  419 -- transcript: reload a bounded window of recent messages
  420 -- ('Session.loadRecentMessages') and replace them into an empty context
  421 -- carrying the env system prompt (mirrors the CLI resume seam, spike §2).
  422 seedContextFrom :: AgentEnv -> TranscriptHandle -> IO Ctx.Context
  423 seedContextFrom env th = do
  424   recent <- Session.loadRecentMessages th 50 100000
  425   pure (Ctx.replaceMessages recent (Ctx.emptyContext (_env_systemPrompt env)))
  426 
  427 -- | Seed a 'Ctx.Context' for a slash command (the 'fallthrough' path) from a
  428 -- session's transcript: the recent-message window ('seedContextFrom') PLUS the
  429 -- cumulative provider token usage recovered from the transcript's Response
  430 -- entries ('transcriptUsage'). This restores both the message count \/ context
  431 -- estimate AND the @Total input\/output tokens@ that @\/status@ reports — which
  432 -- the live runtime worker accumulates in its private 'Ctx.Context' that the
  433 -- slash path could never see (pureclaw-25k).
  434 seedSlashContext :: AgentEnv -> TranscriptHandle -> IO Ctx.Context
  435 seedSlashContext env th = do
  436   base  <- seedContextFrom env th
  437   usage <- transcriptUsage th
  438   pure (Ctx.recordUsage (Just usage) base)
  439 
  440 -- | Sum the provider-reported token usage across every Response entry recorded
  441 -- in a session's transcript. Each Response entry carries its
  442 -- @input_tokens@ \/ @output_tokens@ in '_te_metadata' (written by
  443 -- 'PureClaw.Transcript.Provider.buildResponseMetadata'); missing or
  444 -- non-numeric values contribute zero. The result mirrors what the live worker's
  445 -- 'Ctx.recordUsage' accumulator would hold, so a seeded @\/status@ reports the
  446 -- real cumulative totals rather than zeros.
  447 transcriptUsage :: TranscriptHandle -> IO P.Usage
  448 transcriptUsage th = do
  449   entries <- _th_query th TT.emptyFilter
  450   let responses = [ e | e <- entries, TT._te_direction e == TT.Response ]
  451       inp = sum [ metaInt "input_tokens"  (TT._te_metadata e) | e <- responses ]
  452       out = sum [ metaInt "output_tokens" (TT._te_metadata e) | e <- responses ]
  453   pure (P.Usage inp out)
  454 
  455 -- | Read a non-negative 'Int' from a metadata map under @key@; absent or
  456 -- non-numeric yields 0.
  457 metaInt :: Text -> Map.Map Text Value -> Int
  458 metaInt key m = case Map.lookup key m of
  459   Just v  -> case Aeson.fromJSON v of
  460     Aeson.Success n -> n
  461     Aeson.Error _   -> 0
  462   Nothing -> 0

  464 -- | The effective tool registry for a per-tab runtime: the env's built-in
  465 -- registry merged with any connected MCP server tools (read per call, so tools
  466 -- connected after a tab is created are still seen). Mirrors the still-live
  467 -- @Loop.backgroundRegistry@ \/ legacy @effectiveRegistry@ (pureclaw-2u4; #79).
  468 effectiveRegistry :: AgentEnv -> IO ToolRegistry
  469 effectiveRegistry env = do
  470   servers <- readIORef (_env_mcpServers env)
  471   let base = _env_registry env
  472   pure $
  473     if Map.null servers
  474       then base
  475       else mergeRegistries base (mcpRegistry (Map.elems servers))

  477 -- | Run ONE tool call against the effective registry, returning a tool-result
  478 -- 'P.Message'. Mirrors @Loop.executeCall@ + @toolResultMessage@ for a single
  479 -- call: an unknown tool yields a one-line error result rather than throwing.
  480 execOneTool :: AgentEnv -> ToolCallId -> Text -> Value -> IO P.Message
  481 execOneTool env callId name input = do
  482   reg    <- effectiveRegistry env
  483   result <- executeTool reg name input
  484   let (parts, isErr) = case result of
  485         Nothing       -> ([P.TRPText ("Unknown tool: " <> name)], True)
  486         Just (ps, er) -> (ps, er)
  487   pure (P.toolResultMessage [(callId, parts, isErr)])

  489 -- | Resolve the 'Session.SessionHandle' for a 'Core.SessionId': prefer the
  490 -- @store@ entry (minted by '_td_newDefaultSession'); fall back to the live
  491 -- '_env_session' if its id matches; otherwise open a fresh handle from disk and
  492 -- cache it. The store guarantees the common case (every bound session was
  493 -- minted by the dispatcher) never touches disk twice.
  494 resolveSession :: AgentEnv -> SessionStore -> Core.SessionId -> IO Session.SessionHandle
  495 resolveSession env store sid = do
  496   m <- readIORef store
  497   case Map.lookup sid m of
  498     Just sh -> pure sh
  499     Nothing -> do
  500       cur <- readIORef (_env_session env)
  501       curMeta <- readIORef (Session._sh_meta cur)
  502       if SessionTypes._sm_id curMeta == sid
  503         then do
  504           cacheSession store sid cur
  505           pure cur
  506         else do
  507           -- TODO(8c.3): a wizard-reopened session whose handle was not minted
  508           -- here. Open it from disk via the same sessions dir and cache it.
  509           sh <- openSessionFromDisk env sid
  510           cacheSession store sid sh
  511           pure sh

  513 -- | Cache a resolved handle in the store under its id.
  514 cacheSession :: SessionStore -> Core.SessionId -> Session.SessionHandle -> IO ()
  515 cacheSession store sid sh =
  516   atomicModifyIORef' store (\m -> (Map.insert sid sh m, ()))

  518 -- | Open a 'Session.SessionHandle' for an existing on-disk session id, rooted
  519 -- under the foreground session's sessions directory. Used only on the
  520 -- wizard-reopen fallback path (a 'BoundSession' whose handle was not minted by
  521 -- '_td_newDefaultSession' and is not the foreground session).
  522 --
  523 -- Resolves the REAL on-disk session via 'Session.resumeSession' — the same
  524 -- loader the CLI @--session@ resume and the @\/resume@ slash command use — so
  525 -- the handle carries the actual @session.json@ metadata (model, description,
  526 -- created\/last-active) and reopens the real @transcript.jsonl@ for append.
  527 -- This replaces the prior fabrication (which set @_sm_model = unSessionId sid@
  528 -- and @createdAt\/lastActive = now@ via 'baseProviderMeta'); that synthesized
  529 -- meta was then persisted by 'Session.setSourceIfAbsent' \/ 'saveMeta',
  530 -- corrupting the real @session.json@ (pureclaw-apv; #79).
  531 --
  532 -- On a resume failure (missing or corrupted @session.json@ — the session no
  533 -- longer exists or its metadata is unreadable) we log a warning and fall back
  534 -- to a freshly-minted handle. This does not corrupt a healthy session: resume
  535 -- only fails when there is no valid @session.json@ to preserve.
  536 openSessionFromDisk :: AgentEnv -> Core.SessionId -> IO Session.SessionHandle
  537 openSessionFromDisk env sid = do
  538   dir <- sessionsDirOf env
  539   result <- Session.resumeSession (_env_broker env) (_env_logger env) dir sid
  540   case result of
  541     Right sh -> pure sh
  542     Left err -> do
  543       _lh_logWarn (_env_logger env) $
  544         "Tabs.Wiring: could not resume wizard-reopened session "
  545           <> Core.unSessionId sid <> ": " <> T.pack (show err)
  546           <> " — falling back to a fresh session handle"
  547       now <- getCurrentTime
  548       let meta = baseProviderMeta env now sid (ModelId (Core.unSessionId sid)) "session"
  549       Session.mkSessionHandle (_env_broker env) (_env_logger env) dir meta

  603 -- | Mint a fresh default-provider session, cache its handle in @store@, and
  604 -- return its 'BoundSession' ref. @'Left' msg@ when no default provider or
  605 -- model is configured; the message carries actionable setup guidance.
  606 mkNewDefaultSession :: AgentEnv -> SessionStore -> IO (Either Text TabRef)
  607 mkNewDefaultSession env store = do
  608   mProvider <- readIORef (_env_provider env)
  609   mModel    <- readIORef (_env_model env)
  610   case (mProvider, mModel) of
  611     (Nothing, _) -> pure (Left noProviderMsg)
  612     (_, Nothing) -> pure (Left noModelMsg)
  613     (Just _, Just model) -> do
  614       now <- getCurrentTime
  615       dir <- sessionsDirOf env
  616       let sid  = SessionTypes.newSessionId Nothing now
  617           meta = baseProviderMeta env now sid model "session"
  618       sh <- Session.mkSessionHandle (_env_broker env) (_env_logger env) dir meta
  619       cacheSession store sid sh
  620       pure (Right (BoundSession sid))

  705 -- | Handle a non-tab slash command (the routing grammar's
  706 -- 'RT.ParsedSlashCmd'): the legacy slash-command surface (@\/help@, @\/status@,
  707 -- @\/provider@, @\/vault@, …) still runs through 'Slash.executeSlashCommand',
  708 -- which emits to '_env_channel' — the CLI conversation's registered sink — so
  709 -- all non-tab commands behave exactly as before the flip.
  710 --
  711 -- @\/bg@ is special-cased: 'Slash.executeSlashCommand' only stubs it (it has no
  712 -- dispatcher), so the tabbed loop forks a self-contained background turn
  713 -- ('runBg') instead, matching the legacy loop's @\/bg@ handling.
  714 --
  715 -- The per-call 'Ctx.Context' is SEEDED from the conversation's ACTIVE bound
  716 -- session transcript ('seedSlashContext' over 'activeBoundSession'): the
  717 -- recent-message window plus the cumulative token usage. This is what
  718 -- @\/status@ (message count, context estimate, total tokens) and @\/compact@
  719 -- read — the live runtime worker keeps those in its own private 'Ctx.Context'
  720 -- the slash path cannot reach, so without seeding every count reads zero
  721 -- (pureclaw-25k). When there is no active session\/tab (or a harness ref) the
  722 -- context falls back to empty, the pre-fix behaviour. Most other fallthrough
  723 -- commands (help\/config\/provider\/…) ignore the context entirely, so seeding
  724 -- is harmless for them.
  725 fallthrough
  726   :: AgentEnv -> SessionStore -> ConversationKey -> Slash.SlashCommand -> IO ()
  727 fallthrough env store convKey cmd = case cmd of
  728   Slash.CmdBg prompt -> do
  729     _ch_send (_env_channel env)
  730       (OutgoingMessage
  731         "\x1F504 /bg: running in the background \x2014 the result will appear here when ready.")
  732     _ <- _env_fork env (runBg env prompt)
  733     pure ()
  734   _ -> do
  735     ctx <- slashContext env store convKey
  736     _ <- Slash.executeSlashCommand env cmd ctx
  737     pure ()
  738 
  739 -- | The 'Ctx.Context' a fallthrough slash command runs against: seeded from the
  740 -- conversation's active bound session transcript when one exists, else an empty
  741 -- context carrying the env system prompt (the pre-fix fallback). See
  742 -- 'fallthrough' and 'seedSlashContext' (pureclaw-25k).
  743 slashContext :: AgentEnv -> SessionStore -> ConversationKey -> IO Ctx.Context
  744 slashContext env store convKey = do
  745   mSh <- activeBoundSession env store convKey
  746   case mSh of
  747     Just sh -> seedSlashContext env (Session._sh_transcript sh)
  748     Nothing -> pure (Ctx.emptyContext (_env_systemPrompt env))
  749 
  750 -- | Fork a @\/bg@ background turn. Re-exported from the loop so the tabbed
  751 -- fallthrough preserves @\/bg@ semantics without importing the whole loop's
  752 -- internals. The result is pushed directly to the conversation channel by
  753 -- 'runBackgroundTurn'.
  754 runBg :: AgentEnv -> Text -> IO ()
  755 runBg = Loop.runBackgroundTurn
