    1 -- | Turn-content provider seam (Harness log content source — Task 4) plus the
    2 -- recorded-id dedup helpers + the tailer driving loop (Task 5).
    3 --
    4 -- A 'TurnProvider' is the source of a harness turn's text, a finalize signal,
    5 -- and an optional stable turn id, consumed by the reconcile loop's
    6 -- content-driven turn watcher ('PureClaw.Harness.Reconcile.stepTurns'). It
    7 -- decouples /where/ a turn's text comes from (tmux screen capture today, the
    8 -- claude JSONL log later) from the loop that streams + finalizes it.
    9 --
   10 -- This module is a LEAF below "PureClaw.Harness.Reconcile" for the provider
   11 -- seam — it must NOT import Reconcile for that purpose (Reconcile imports this
   12 -- for 'TurnProvider'). The Task-5 helpers ('recordOnce', 'seedRecordedIds',
   13 -- 'runLogTailer') live here too; 'recordOnce' wraps the record closure and only
   14 -- references 'mkTurnEntry'/'TranscriptEntry' types, not the loop, so no cycle is
   15 -- introduced.
   16 --
   17 -- == The two built-in providers
   18 --
   19 --   * 'tmuxProvider' — preserves today's behavior VERBATIM: the turn text is
   20 --     the handle's '_hh_snapshotTurn', it never finalizes itself
   21 --     (@finalized = False@), and it derives no id (@_tp_turnId = pure Nothing@,
   22 --     so the loop falls back to its '_rd_mintTurn'). The tmux path stays
   23 --     byte-identical.
   24 --   * 'nullProvider' — a handle-less entry: empty text, never finalizes, no
   25 --     derived id.
   26 --
   27 -- == Idempotent replay (Task 5)
   28 --
   29 -- The durable transcript write path is append-only with no upsert, so dedup
   30 -- must happen in front of the record call. 'seedRecordedIds' loads the set of
   31 -- '_te_id's already on disk at tailer startup (via the UNTRIMMED
   32 -- @_th_query emptyFilter@ — not 'loadRecentMessages', which trims at compaction
   33 -- boundaries and could miss a pre-boundary id). 'recordOnce' refuses to record
   34 -- a turn whose derived id is already in that set and adds each newly-recorded id
   35 -- to it. Because the log provider's turn id is deterministic from durable JSONL
   36 -- bytes, a turn re-folded after a crash/restart derives the SAME id, which is
   37 -- already disk-seeded → skipped (no duplicate).
   38 --
   39 -- == The tailer driver (Task 5)
   40 --
   41 -- 'runLogTailer' is the driving loop around 'tailStep': poll and forward each
   42 -- 'ProseTurn' to a sink. No tail offset is persisted across (re)starts — on
   43 -- every (re)start it seeks to EOF ('seekStart' … 'Nothing'); replay correctness
   44 -- rests entirely on the disk-seeded recorded-id set ('seedRecordedIds' +
   45 -- 'recordOnce'), not on a saved byte offset. It
   46 -- treats 'TailUnavailable' as TERMINAL — a loud WARN + STOP so the harness falls
   47 -- back to the tmux provider — because 'tailStep' returns the UNCHANGED offset on
   48 -- a cap trip, so re-looping would spin forever. It runs the loop under
   49 -- 'Async.withAsync' and re-raises 'SomeAsyncException' (not only
   50 -- 'AsyncCancelled') on teardown per the recurring darwin-CI-hang invariant.

  112 -- ---------------------------------------------------------------------------
  113 -- Turn-content provider seam
  114 -- ---------------------------------------------------------------------------
  115 
  116 -- | The source of a harness turn's content for the reconcile loop.
  117 --
  118 --   * '_tp_snapshot' returns @(currentTurnText, finalized?)@. @finalized == True@
  119 --     is an authoritative finalize that bypasses the loop's idle-stability guard.
  120 --   * '_tp_turnId' returns @Just (derivedId, ts)@ to pin a stable id + the
  121 --     turn's timestamp (the log provider derives both from the JSONL event), or
  122 --     'Nothing' to fall back to the loop's @_rd_mintTurn :: IO (Text, UTCTime)@.
  123 --     Carrying the timestamp avoids a second clock call — 'mkTurnEntry' needs
  124 --     both the id AND a 'UTCTime'.
  125 data TurnProvider = TurnProvider
  126   { _tp_snapshot :: IO (Text, Bool)
  127     -- ^ @(currentTurnText, finalized?)@
  128   , _tp_turnId   :: IO (Maybe (Text, UTCTime))
  129     -- ^ @Just (derivedId, ts)@ for the log provider; 'Nothing' → fall back to
  130     --   @_rd_mintTurn@.
  131   }
  132 
  133 -- | The tmux content provider: text from '_hh_snapshotTurn', never finalizes,
  134 -- derives no id. Preserves today's tmux behavior verbatim.
  135 tmuxProvider :: HarnessHandle -> TurnProvider
  136 tmuxProvider hh = TurnProvider
  137   { _tp_snapshot = (,) <$> _hh_snapshotTurn hh <*> pure False
  138   , _tp_turnId   = pure Nothing
  139   }
  140 
  141 -- | A provider for a handle-less entry: empty text, never finalizes, no id.
  142 nullProvider :: TurnProvider
  143 nullProvider = TurnProvider
  144   { _tp_snapshot = pure ("", False)
  145   , _tp_turnId   = pure Nothing
  146   }

  148 -- ---------------------------------------------------------------------------
  149 -- Log provider over a shared tail state
  150 -- ---------------------------------------------------------------------------
  151 
  152 -- | The mutable state shared between the tailer (writer) and the log
  153 -- 'TurnProvider' (reader): the current turn's text, its finalize flag, and its
  154 -- pinned @(derivedId, timestamp)@ (Nothing until the first assistant line of a
  155 -- turn arrives).
  156 data LogTurnState = LogTurnState
  157   { _lts_text      :: !Text
  158   , _lts_finalized :: !Bool
  159   , _lts_turnId    :: !(Maybe (Text, UTCTime))
  160   } deriving stock (Eq, Show)
  161 
  162 -- | The initial empty turn state.
  163 emptyLogTurnState :: LogTurnState
  164 emptyLogTurnState = LogTurnState
  165   { _lts_text      = ""
  166   , _lts_finalized = False
  167   , _lts_turnId    = Nothing
  168   }
  169 
  170 -- | Fold a freshly-yielded 'ProseTurn' into the shared state. The derived id is
  171 -- @deriveTurnId sessionId (_pt_sourceUuid)@ — deterministic from durable JSONL
  172 -- bytes (the first assistant line's uuid), pinned with the supplied event
  173 -- timestamp @ts@. The text + finalize flag track the latest 'ProseTurn'.
  174 applyProseTurn :: Text -> UTCTime -> ProseTurn -> LogTurnState -> LogTurnState
  175 applyProseTurn sessionId ts pt _prev = LogTurnState
  176   { _lts_text      = _pt_text pt
  177   , _lts_finalized = _pt_finalized pt
  178   , _lts_turnId    = Just (deriveTurnId sessionId (_pt_sourceUuid pt), ts)
  179   }
  180 
  181 -- | A log 'TurnProvider' that reads the shared 'LogTurnState' IORef. The tailer
  182 -- sink writes the IORef (via 'applyProseTurn'); the reconcile loop reads it.
  183 mkLogTurnProvider :: IORef LogTurnState -> TurnProvider
  184 mkLogTurnProvider ref = TurnProvider
  185   { _tp_snapshot = do
  186       s <- readIORef ref
  187       pure (_lts_text s, _lts_finalized s)
  188   , _tp_turnId = _lts_turnId <$> readIORef ref
  189   }

  191 -- ---------------------------------------------------------------------------
  192 -- Cached provider selection (no per-tick decode/WARN spam)
  193 -- ---------------------------------------------------------------------------
  194 
  195 -- | Per-harness cached selection state. The reconcile loop runs the selector
  196 -- every tick (~2s) for every bound entry; without a cache a freshly-spawned
  197 -- claude harness — which has NO JSONL file until its first output — would
  198 -- re-decode @session.json@, re-resolve the base, and re-attempt (and re-WARN)
  199 -- on every tick for the entire pre-first-output window. These three states make
  200 -- that cheap and quiet:
  201 --
  202 --   * 'LpResolved' — success: the log tailer has engaged; return the cached
  203 --     log 'TurnProvider' as-is.
  204 --   * 'LpPending' — a valid @claude-code@ harness whose log file is not present
  205 --     YET (a 'ClaudeLogNotFound', the NORMAL pre-output state). Caches a CHEAP
  206 --     re-attempt closure (only 'mkSafeClaudeLogPath' — the validated uuid\/base
  207 --     are already closed over), so subsequent ticks do NOT re-decode
  208 --     @session.json@ and do NOT WARN, yet the provider still ENGAGES the moment
  209 --     the JSONL appears (a later 'Right' transitions to 'LpResolved').
  210 --   * 'LpFallback' — permanently the tmux\/null fallback: not @claude-code@, no
  211 --     or invalid uuid, OR a genuine (non-not-found) resolution error. WARNed at
  212 --     most once; the input resolver is never re-run.
  213 data LogProviderState
  214   = LpResolved !TurnProvider
  215   | LpPending !(IO (Either ClaudeLogPathError SafeClaudeLogPath))
  216   | LpFallback
  217 
  218 -- | Outcome of the one-time \"resolve inputs\" phase (decode @session.json@,
  219 -- resolve the claude base). Keeps the expensive\/fragile input resolution OUT
  220 -- of the per-tick path: it runs once, and its result is cached as either a
  221 -- permanent 'LpFallback' or a cheap re-attemptable 'LpPending'\/'LpResolved'.
  222 data ResolveInputs
  223   = -- | Not a resolvable @claude-code@ log source (not @claude-code@, no uuid,
  224     -- invalid uuid). Carries the one-shot WARN text. Cached as 'LpFallback'.
  225     RiFallback !Text
  226   | -- | The inputs are valid; carries a CHEAP re-attempt closure that only
  227     -- re-runs 'mkSafeClaudeLogPath' (no @session.json@ decode, no base
  228     -- re-resolution). Its first result is classified by 'selectLogProvider':
  229     -- 'Right' → engage ('LpResolved'); 'Left' 'ClaudeLogNotFound' → 'LpPending'
  230     -- (re-attempt next tick, no WARN); any other 'Left' → 'LpFallback' (WARN
  231     -- once).
  232     RiReady !(IO (Either ClaudeLogPathError SafeClaudeLogPath))
  233 
  234 -- | True for the one error that means \"the log just is not on disk yet\" — the
  235 -- expected, transient pre-first-output state. Every OTHER error
  236 -- (ambiguity, symlink escape, owner\/mode) is a genuine fault that must NOT be
  237 -- treated as Pending (it would re-attempt forever); those become 'LpFallback'.
  238 isNotFound :: ClaudeLogPathError -> Bool
  239 isNotFound (ClaudeLogNotFound _) = True
  240 isNotFound _                     = False
  241 
  242 -- | Choose the turn-content provider for one entry, caching the decision per
  243 -- key so the per-tick path neither re-decodes inputs nor WARN-spams.
  244 --
  245 -- On a cache HIT: 'LpResolved' returns the cached provider; 'LpFallback'
  246 -- returns @fallback@ (no work, no WARN); 'LpPending' cheaply RE-ATTEMPTS via the
  247 -- cached closure — on 'Right' it transitions to 'LpResolved' (engage), on a
  248 -- still-not-found 'Left' it stays Pending and returns @fallback@ WITHOUT
  249 -- WARNing, and on a genuine non-not-found 'Left' it transitions to 'LpFallback'
  250 -- (WARN once).
  251 --
  252 -- On a cache MISS: run the one-time @resolveInputs@. 'RiFallback' caches
  253 -- 'LpFallback' and WARNs once; 'RiReady' runs the cheap attempt once and
  254 -- classifies it exactly as the Pending re-attempt does.
  255 --
  256 -- @engage@ is the IO that seeds the recorded-id set, starts the tailer Async,
  257 -- and builds the log 'TurnProvider' (run at most once, on the first 'Right').
  258 selectLogProvider
  259   :: Ord k
  260   => IORef (Map k LogProviderState)
  261   -> k
  262   -> TurnProvider
  263   -- ^ tmux\/null fallback provider for this entry
  264   -> IO ResolveInputs
  265   -- ^ one-time input resolution (decode @session.json@, resolve base)
  266   -> (SafeClaudeLogPath -> IO TurnProvider)
  267   -- ^ engage on the first 'Right': seed + start tailer + build provider
  268   -> (Text -> IO ())
  269   -- ^ WARN sink (fired at most once per key)
  270   -> IO TurnProvider
  271 selectLogProvider cacheRef key fallback resolveInputs engage warn = do
  272   cached <- Map.lookup key <$> readIORef cacheRef
  273   case cached of
  274     Just (LpResolved p) -> pure p
  275     Just LpFallback     -> pure fallback
  276     Just (LpPending reattempt) -> reattempt >>= classify reattempt
  277     Nothing -> do
  278       ri <- resolveInputs
  279       case ri of
  280         RiFallback warnTxt -> do
  281           modifyIORef' cacheRef (Map.insert key LpFallback)
  282           warn warnTxt
  283           pure fallback
  284         RiReady reattempt ->
  285           reattempt >>= classify reattempt
  286   where
  287     -- Classify one attempt result, transitioning the cache + (at most once)
  288     -- WARNing. A genuine fault transitions to 'LpFallback' exactly once per key
  289     -- — whether reached on the first attempt (cache miss) or on a later Pending
  290     -- re-attempt — so the WARN here fires at most once per key.
  291     classify reattempt = \case
  292       Right safePath -> do
  293         prov <- engage safePath
  294         modifyIORef' cacheRef (Map.insert key (LpResolved prov))
  295         pure prov
  296       Left err
  297         | isNotFound err -> do
  298             -- Expected pre-output state: stay Pending, re-attempt next tick,
  299             -- NO WARN. (Idempotent re-insert keeps the closure cached.)
  300             modifyIORef' cacheRef (Map.insert key (LpPending reattempt))
  301             pure fallback
  302         | otherwise -> do
  303             modifyIORef' cacheRef (Map.insert key LpFallback)
  304             warn (faultWarn err)
  305             pure fallback
  306 
  307     faultWarn err =
  308       "harness log unavailable (" <> textShow err <> ") — falling back to tmux"

  310 -- ---------------------------------------------------------------------------
  311 -- Recorded-id dedup (idempotent replay)
  312 -- ---------------------------------------------------------------------------
  313 
  314 -- | Seed the recorded-id set from this session's on-disk transcript at tailer
  315 -- startup. Reads EVERY entry via the UNTRIMMED @_th_query emptyFilter@ (NOT
  316 -- 'loadRecentMessages', which trims to the last compaction boundary and could
  317 -- miss a pre-boundary id, re-introducing a duplicate on replay).
  318 seedRecordedIds :: TranscriptHandle -> IO (Set Text)
  319 seedRecordedIds th =
  320   Set.fromList . map _te_id <$> _th_query th emptyFilter
  321 
  322 -- | Record a turn at most once. If the entry's '_te_id' is already in the
  323 -- (disk-seeded, mutable) set, skip the record entirely; otherwise record via
  324 -- the supplied append action and insert the id. The reconcile loop is
  325 -- single-threaded, so check-set → record → add-id is atomic w.r.t. itself.
  326 recordOnce
  327   :: IORef (Set Text)
  328   -> (SessionId -> TranscriptEntry -> IO ())
  329   -> SessionId
  330   -> TranscriptEntry
  331   -> IO ()
  332 recordOnce ref record sid entry = do
  333   let tid = _te_id entry
  334   -- atomicModifyIORef' returns whether the id was newly inserted; only then do
  335   -- we record. Insert-before-record so a re-entrant call for the same id during
  336   -- the record (impossible in the single-threaded loop, but cheap to be safe)
  337   -- still dedups.
  338   isNew <- atomicModifyIORef' ref $ \s ->
  339     if Set.member tid s
  340       then (s, False)
  341       else (Set.insert tid s, True)
  342   Monad.when isNew (record sid entry)

  344 -- ---------------------------------------------------------------------------
  345 -- Tailer driver
  346 -- ---------------------------------------------------------------------------

  348 -- | Poll interval between tail steps (250 ms).
  349 tailerPollMicros :: Int
  350 tailerPollMicros = 250 * 1000
  351 
  352 -- | Drive 'tailStep' in a loop, forwarding each yielded 'ProseTurn' to @sink@.
  353 --
  354 -- Runs the loop body under 'Async.withAsync' so a teardown cancellation is
  355 -- delivered as an async exception; 'SomeAsyncException' is RE-RAISED (not only
  356 -- 'AsyncCancelled') per the darwin-CI-hang invariant, while a logged
  357 -- non-async exception terminates the loop cleanly.
  358 --
  359 -- 'TailUnavailable' is TERMINAL: a loud WARN + STOP (so the harness falls back
  360 -- to the tmux provider). 'tailStep' returns the UNCHANGED offset on a cap trip,
  361 -- so re-looping on the same bytes would spin forever — hence we never re-loop
  362 -- on it.
  363 runLogTailer
  364   :: JsonlTailDeps
  365   -> TailCaps
  366   -> LogHandle
  367   -> SafeClaudeLogPath
  368   -> (ProseTurn -> IO ())
  369   -> IO ()
  370 runLogTailer deps caps logH path sink =
  371   Async.withAsync loop $ \a -> do
  372     r <- try (Async.wait a)
  373     case r of
  374       Right () -> pure ()
  375       Left e
  376         -- Re-raise async exceptions (cancellation/shutdown) so teardown is not
  377         -- swallowed (darwin-CI-hang invariant).
  378         | Just ae <- fromException e -> throwIO (ae :: SomeAsyncException)
  379         -- A non-async exception terminates the loop; log it and stop.
  380         | otherwise -> _lh_logWarn logH
  381             ("claude log tailer error: " <> textShow (e :: SomeException))
  382   where
  383     loop = do
  384       -- No persisted offset: 'Nothing' seeks to EOF on every (re)start. Replay
  385       -- correctness rests on the disk-seeded recorded-id set, not a saved
  386       -- offset (see this module's header).
  387       off0 <- seekStart deps path Nothing
  388       go (off0, emptyBuffer, emptyProseState)
  389 
  390     go st = do
  391       (st', evs) <- tailStep deps caps path st
  392       if any isUnavailable evs
  393         then _lh_logWarn logH
  394                "claude log unavailable for harness — falling back to tmux"
  395         else do
  396           mapM_ emit evs
  397           threadDelay tailerPollMicros
  398           go st'
  399 
  400     emit (TailProse pt) = sink pt
  401     emit TailUnavailable = pure ()
  402 
  403     isUnavailable TailUnavailable = True
  404     isUnavailable (TailProse _)   = False

  406 -- Local helper to render an exception as 'Text'.
  407 textShow :: Show a => a -> Text
  408 textShow = T.pack . show
