    1 -- | A PURE converter from @claude-code@ JSONL session-log lines to
    2 -- 'TranscriptEntry' values (WU3).
    3 --
    4 -- A spawned @claude-code@ harness records its conversation to an on-disk
    5 -- @<uuid>.jsonl@ file (one JSON object per line; see
    6 -- @docs/harness-jsonl-capture-spike.md@ §D0.5). 'convertLine' turns ONE such
    7 -- line into the 'TranscriptEntry' shape PureClaw's transcript/UI already
    8 -- understands, so a harness session can be surfaced exactly like a native
    9 -- provider session. This module is the data source for WU5's live tailer.
   10 --
   11 -- == Purity & totality
   12 --
   13 -- There is NO IO here — just a total decode-and-map. 'convertLine' NEVER
   14 -- throws on ANY input: malformed JSON, wrong-typed fields, missing fields,
   15 -- unknown event @type@, or an unparseable timestamp all yield 'Nothing'
   16 -- (the line is skipped). Lookups go through 'KeyMap.lookup' / 'fromJSON';
   17 -- no partial functions (@head@, @fromJust@, partial record access) appear.
   18 --
   19 -- == Mapping (matches @frontend\/src\/App.tsx@ — D3.1)
   20 --
   21 --   * An @assistant@ event becomes a 'Response' entry whose payload is an
   22 --     object with top-level @content@ (lifted from @message.content@),
   23 --     @usage@ (from @message.usage@) and @model@ (from @message.model@), and
   24 --     whose '_te_model' column is @Just message.model@. This is exactly what
   25 --     the @App.tsx@ Response parser reads (@parsed.content@ / @parsed.usage@ /
   26 --     @e.model@).
   27 --   * A @user@ event with text\/string content becomes a 'Request' entry whose
   28 --     payload is @{messages:[{role:"user",content:[{type:"text",text}]}]}@.
   29 --     String content is normalized to a single-element @text@ block so the
   30 --     @App.tsx@ @extractTextFromContent@ (which expects an array) reads it.
   31 --   * A @user@ event whose @message.content[]@ holds @tool_result@ blocks
   32 --     becomes a 'Request' entry that preserves those blocks
   33 --     (@type@\/@tool_use_id@\/@content@\/@is_error@) inside
   34 --     @messages[].content[]@, so @App.tsx@ @buildToolResultIndex@ joins them
   35 --     by @tool_use_id@. (Text and tool_result blocks share one code path.)
   36 --   * @thinking@ blocks are carried through into the payload @content@ so
   37 --     WU8's renderer can show them (collapsed).
   38 --   * Every other event @type@ (mode, permission-mode, system,
   39 --     file-history-snapshot, …) and every malformed line yields 'Nothing'.
   40 --
   41 -- == De-dup obligation (D3.4)
   42 --
   43 -- '_te_id' and '_te_correlationId' are both set to the JSONL @uuid@, and
   44 -- conversion is a pure function of the line, so converting the same event
   45 -- twice yields the same '_te_id'. WU5\/WU7 MUST de-duplicate emitted entries
   46 -- by '_te_id' (the on-disk log is read from offset 0 on every view-open, so
   47 -- the same line will be seen more than once).
   48 --
   49 -- == Sanitization (D3.5)
   50 --
   51 -- All text destined for the payload (assistant @text@\/@thinking@, user text,
   52 -- and @tool_result@ @content@) is passed through the SAME
   53 -- 'PureClaw.Handles.Harness.sanitizeHarnessOutput' used for tmux scrollback
   54 -- broadcast — imported, never re-implemented — so control sequences are
   55 -- stripped identically on both paths.
   56 module PureClaw.Harness.ClaudeLogConvert
   57   ( -- * Conversion
   58     convertLine
   59     -- * Payload encoder (exported for golden tests)
   60   , encodeValuePayload
   61     -- * Total JSONL helpers (re-used by ClaudeLogProse — do not re-implement)
   62   , decodeObject
   63   , lookupText
   64   , lookupObject

   66   ) where

   88 -- | The harness label stamped on every converted entry's '_te_harness'.
   89 harnessLabel :: Text
   90 harnessLabel = "claude-code"
   91 
   92 -- | Convert ONE claude-code JSONL line to a 'TranscriptEntry', or 'Nothing'
   93 -- if the line is malformed, an unknown\/ignored event type, or is missing a
   94 -- parseable timestamp. Total: never throws on any input. (D3.1 / D3.2)
   95 convertLine :: ByteString -> Maybe TranscriptEntry
   96 convertLine raw = do
   97   obj <- decodeObject raw
   98   typ <- lookupText "type" obj
   99   case typ of
  100     "assistant" -> convertAssistant obj
  101     "user"      -> convertUser obj
  102     _           -> Nothing  -- mode, permission-mode, system, snapshots, … (D3.2)
  103 
  104 -- | An @assistant@ event → 'Response' entry. (D3.1)
  105 convertAssistant :: KeyMap Value -> Maybe TranscriptEntry
  106 convertAssistant obj = do
  107   uuid <- lookupText "uuid" obj
  108   tstamp <- lookupTimestamp obj
  109   message <- lookupObject "message" obj
  110   let model = lookupText "model" message
  111       contentBlocks = case KeyMap.lookup "content" message of
  112         Just (Array blocks) -> V.toList (V.map sanitizeBlock blocks)
  113         _                   -> []
  114       usageField = maybe [] (\u -> [("usage", u)]) (KeyMap.lookup "usage" message)
  115       modelField = maybe [] (\m -> [("model", String m)]) model
  116       payload = objectOf $
  117         [("content", Array (V.fromList contentBlocks))]
  118           <> usageField
  119           <> modelField
  120   pure (mkEntry uuid tstamp model Response payload)
  121 
  122 -- | A @user@ event → 'Request' entry. Handles both plain text\/string content
  123 -- and @tool_result@ content through one normalization path. (D3.1)
  124 convertUser :: KeyMap Value -> Maybe TranscriptEntry
  125 convertUser obj = do
  126   uuid <- lookupText "uuid" obj
  127   tstamp <- lookupTimestamp obj
  128   message <- lookupObject "message" obj
  129   let content = KeyMap.lookup "content" message
  130       normalized = normalizeUserContent content
  131       payload = objectOf
  132         [ ("messages", Array (V.singleton (objectOf
  133             [ ("role", String "user")
  134             , ("content", Array (V.fromList normalized))
  135             ])))
  136         ]
  137   pure (mkEntry uuid tstamp Nothing Request payload)
  138 
  139 -- | Normalize a @user@ event's @message.content@ into a list of content
  140 -- blocks suitable for the payload:
  141 --
  142 --   * a JSON string becomes a single @{type:"text",text:<sanitized>}@ block;
  143 --   * an array is mapped block-by-block through 'sanitizeBlock' (which
  144 --     sanitizes @text@\/@thinking@\/@content@ string fields and preserves
  145 --     @tool_result@ structure so @buildToolResultIndex@ can join it);
  146 --   * anything else (missing\/null\/number) becomes an empty list.
  147 normalizeUserContent :: Maybe Value -> [Value]
  148 normalizeUserContent = \case
  149   Just (String s) ->
  150     [ objectOf
  151         [ ("type", String "text")
  152         , ("text", String (sanitizeHarnessOutput s))
  153         ]
  154     ]
  155   Just (Array blocks) -> V.toList (V.map sanitizeBlock blocks)
  156   _ -> []
  157 
  158 -- | Sanitize the human-visible string fields of one content block, preserving
  159 -- all other fields (and non-object blocks) untouched. Applies the SAME
  160 -- 'sanitizeHarnessOutput' used for tmux broadcast (D3.5) to @text@,
  161 -- @thinking@, and @content@ string fields. @tool_result@ structure
  162 -- (@type@\/@tool_use_id@\/@is_error@\/@content@) is preserved so
  163 -- @App.tsx buildToolResultIndex@ can join by @tool_use_id@.
  164 sanitizeBlock :: Value -> Value
  165 sanitizeBlock (Object km) = Object (KeyMap.mapWithKey sanitizeField km)
  166   where
  167     sanitizeField k v
  168       | Key.toText k `elem` sanitizedFields = sanitizeStringValue v
  169       | otherwise                           = v
  170 sanitizeBlock other = other
  171 
  172 -- | Field names whose string values are human-visible and must be sanitized.
  173 sanitizedFields :: [Text]
  174 sanitizedFields = ["text", "thinking", "content"]
  175 
  176 -- | Apply 'sanitizeHarnessOutput' to a JSON string; leave non-strings as-is.
  177 sanitizeStringValue :: Value -> Value
  178 sanitizeStringValue (String s) = String (sanitizeHarnessOutput s)
  179 sanitizeStringValue v          = v
  180 
  181 -- | Assemble a 'TranscriptEntry' with the WU3-fixed columns: '_te_harness' is
  182 -- always @Just "claude-code"@, '_te_id' and '_te_correlationId' are both the
  183 -- JSONL @uuid@ (de-dup key — see module header, D3.4), no duration, empty
  184 -- metadata.
  185 mkEntry :: Text -> UTCTime -> Maybe Text -> Direction -> Value -> TranscriptEntry
  186 mkEntry uuid tstamp model dir payload = TranscriptEntry
  187   { _te_id            = uuid
  188   , _te_timestamp     = tstamp
  189   , _te_harness       = Just harnessLabel
  190   , _te_model         = model
  191   , _te_direction     = dir
  192   , _te_payload       = encodeValuePayload payload
  193   , _te_durationMs    = Nothing
  194   , _te_correlationId = uuid
  195   , _te_metadata      = Map.empty
  196   }

  200 -- | Decode a line into a top-level JSON object, or 'Nothing' if it is not
  201 -- valid JSON or not an object (e.g. an array, bare string, number, or the
  202 -- intentionally-malformed fixture line).
  203 decodeObject :: ByteString -> Maybe (KeyMap Value)
  204 decodeObject raw =
  205   case Aeson.decodeStrict raw of
  206     Just (Object km) -> Just km
  207     _                -> Nothing
  208 
  209 -- | Total lookup of a string-valued field.
  210 lookupText :: Text -> KeyMap Value -> Maybe Text
  211 lookupText k km =
  212   case KeyMap.lookup (Key.fromText k) km of
  213     Just (String t) -> Just t
  214     _               -> Nothing
  215 
  216 -- | Total lookup of an object-valued field.
  217 lookupObject :: Text -> KeyMap Value -> Maybe (KeyMap Value)
  218 lookupObject k km =
  219   case KeyMap.lookup (Key.fromText k) km of
  220     Just (Object o) -> Just o
  221     _               -> Nothing
  222 
  223 -- | Parse the event @timestamp@ as an ISO-8601 'UTCTime' via aeson's own
  224 -- 'FromJSON' instance (a total, format-checked path). A missing or
  225 -- unparseable timestamp yields 'Nothing' — a real claude log always carries
  226 -- a parseable timestamp, so an absent one means the line is not a turn we can
  227 -- faithfully place in the transcript and is deliberately skipped. (D3.1)
  228 lookupTimestamp :: KeyMap Value -> Maybe UTCTime
  229 lookupTimestamp km = do
  230   v <- KeyMap.lookup "timestamp" km
  231   case Aeson.fromJSON v of
  232     Aeson.Success u -> Just u
  233     Aeson.Error _   -> Nothing
  234 
  235 -- | Build a JSON object from key\/value pairs. Centralizes the
  236 -- 'Key.fromText' conversion so call sites read in plain 'Text'.
  237 objectOf :: [(Text, Value)] -> Value
  238 objectOf = Object . KeyMap.fromList . map (first Key.fromText)
  239 
  240 -- | The production payload encoder: serialize a JSON 'Value' with aeson's
  241 -- 'Aeson.encode' (deterministic, lexicographically-keyed objects) and decode
  242 -- the bytes to '_te_payload' 'Text' via the SAME 'encodePayload' the
  243 -- transcript layer uses. Exported so golden tests can build the EXPECTED
  244 -- payload with this exact encoder, making the comparison byte-for-byte. (D3.3)
  245 encodeValuePayload :: Value -> Text
  246 encodeValuePayload = encodePayload . BL.toStrict . Aeson.encode
