   98 -- | The kind of channel a message originated from. Known channels get a
   99 -- typed constructor; 'CkOther' is an open escape hatch for channels that
  100 -- do not (yet) have a dedicated tag.
  101 data ChannelKind
  102   = CkCli
  103   | CkWeb
  104   | CkSignal
  105   | CkTelegram
  106   | CkBackground
  107   | CkOther !Text
  108   deriving stock (Show, Eq, Generic)
  109 
  110 -- | Render a 'ChannelKind' as a flat lowercase tag. 'CkOther' carries its
  111 -- own name verbatim. Mirrors the @flavourToText@ precedent.
  112 channelKindToText :: ChannelKind -> Text
  113 channelKindToText CkCli        = "cli"
  114 channelKindToText CkWeb        = "web"
  115 channelKindToText CkSignal     = "signal"
  116 channelKindToText CkTelegram   = "telegram"
  117 channelKindToText CkBackground = "background"
  118 channelKindToText (CkOther n)  = n
  119 
  120 -- | Parse a flat tag back into a 'ChannelKind'. Known names map to their
  121 -- typed constructor; everything else becomes 'CkOther'. Critically,
  122 -- @"signal"@ maps to 'CkSignal' (never @CkOther "signal"@), so a
  123 -- 'CkOther' naming a known channel cannot round-trip into existence.
  124 channelKindFromText :: Text -> ChannelKind
  125 channelKindFromText "cli"        = CkCli
  126 channelKindFromText "web"        = CkWeb
  127 channelKindFromText "signal"     = CkSignal
  128 channelKindFromText "telegram"   = CkTelegram
  129 channelKindFromText "background" = CkBackground
  130 channelKindFromText t            = CkOther t
  131 
  132 instance Aeson.ToJSON ChannelKind where
  133   toJSON = Aeson.String . channelKindToText
  134 
  135 instance Aeson.FromJSON ChannelKind where
  136   parseJSON = Aeson.withText "ChannelKind" (pure . channelKindFromText)

  138 -- | Maximum length (in characters) for any attacker-controlled string
  139 -- stored inside a 'MessageSource' after normalization.
  140 maxSourceLen :: Int
  141 maxSourceLen = 512
  142 
  143 -- | Where an inbound message came from. Structured and extensible: a typed
  144 -- 'ChannelKind', the channel's user id (when one exists), and an open
  145 -- field map for supplementary, channel-specific data (which may carry
  146 -- nested JSON).
  147 --
  148 -- Construct via 'mkMessageSource', never the raw record — the smart
  149 -- constructor normalizes attacker-controlled input.
  150 --
  151 -- Invariants (enforced by convention, not the type): '_ms_fields' must not
  152 -- duplicate '_ms_userId' and must never contain credentials/secrets.
  153 data MessageSource = MessageSource
  154   { _ms_channel :: !ChannelKind
  155   , _ms_userId  :: !(Maybe UserId)
  156   , _ms_fields  :: !(Map Text Aeson.Value)
  157   } deriving stock (Show, Eq, Generic)
  158 
  159 -- | Strip ASCII control characters (covers newlines, tabs, carriage
  160 -- returns and other control bytes) and bound the result to 'maxSourceLen'
  161 -- characters.
  162 normalizeText :: Text -> Text
  163 normalizeText = T.take maxSourceLen . T.filter (not . Char.isControl)
  164 
  165 -- | Recursively normalize every 'Aeson.String' leaf in a JSON value,
  166 -- leaving the structure otherwise intact.
  167 normalizeValue :: Aeson.Value -> Aeson.Value
  168 normalizeValue (Aeson.String t) = Aeson.String (normalizeText t)
  169 normalizeValue (Aeson.Array xs) = Aeson.Array (fmap normalizeValue xs)
  170 normalizeValue (Aeson.Object o) = Aeson.Object (fmap normalizeValue o)
  171 normalizeValue v                = v
  172 
  173 -- | Smart constructor for 'MessageSource'. Normalizes attacker-controlled
  174 -- input: strips control characters / newlines and bounds length on the
  175 -- user id and on every string leaf inside the field map, and folds a
  176 -- 'CkOther' naming a known channel (e.g. @CkOther "signal"@) down to its
  177 -- typed constructor.
  178 mkMessageSource :: ChannelKind -> Maybe UserId -> Map Text Aeson.Value -> MessageSource
  179 mkMessageSource ch uid fields = MessageSource
  180   { _ms_channel = foldChannel ch
  181   , _ms_userId  = fmap (UserId . normalizeText . unUserId) uid
  182   , _ms_fields  = fmap normalizeValue fields
  183   }
  184   where
  185     foldChannel (CkOther n) = channelKindFromText n
  186     foldChannel c           = c
  187 
  188 instance Aeson.ToJSON MessageSource where
  189   toJSON s = Aeson.object $
  190     ["channel" .= _ms_channel s]
  191       <> case _ms_userId s of
  192         Just u  -> ["user_id" .= unUserId u]
  193         Nothing -> []
  194       <> ["fields" .= _ms_fields s | not (Map.null (_ms_fields s))]
  195 
  196 instance Aeson.FromJSON MessageSource where
  197   parseJSON = Aeson.withObject "MessageSource" $ \o -> do
  198     ch     <- o .:  "channel"
  199     uid    <- o .:? "user_id"
  200     fields <- o .:? "fields" .!= Map.empty
  201     pure MessageSource
  202       { _ms_channel = ch
  203       , _ms_userId  = fmap UserId uid
  204       , _ms_fields  = fields
  205       }

  218 -- | Typed allow-list. @AllowAll@ explicitly opts in to allowing everything.
  219 -- @AllowList s@ restricts to the given set.
  220 data AllowList a
  221   = AllowAll
  222   | AllowList (Set a)
  223   deriving stock (Show, Eq, Generic)
  224 
  225 -- | Check whether a value is permitted by the allow-list.
  226 isAllowed :: Ord a => AllowList a -> a -> Bool
  227 isAllowed AllowAll      _ = True
  228 isAllowed (AllowList s) x = Set.member x s
