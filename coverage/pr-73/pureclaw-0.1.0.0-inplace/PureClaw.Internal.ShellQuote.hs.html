    1 -- |
    2 -- Module      : PureClaw.Internal.ShellQuote
    3 -- Description : Canonical single-quote shell quoter (WU4).
    4 --
    5 -- Single source of truth for POSIX single-quote-style shell quoting
    6 -- across PureClaw. The output is safe to splice into a command line
    7 -- that will be re-parsed by a POSIX-compatible shell ('sh', 'bash',
    8 -- 'dash', the remote login shell sshd hands off to, etc.).
    9 --
   10 -- Quoting rules:
   11 --
   12 --   * The empty string returns the literal two-character output @''@
   13 --     (so an empty positional arg survives shell re-tokenisation).
   14 --
   15 --   * Inputs whose characters are all in the \"safe\" set
   16 --     @[A-Za-z0-9_./=:@-]@ are returned unchanged — quoting them
   17 --     would only add noise to logs and command-line traces.
   18 --
   19 --   * All other inputs are wrapped in single quotes. Embedded single
   20 --     quotes are emitted as the four-character sequence @'\\''@
   21 --     (close-quote, backslash-escaped quote, reopen-quote), which is
   22 --     the canonical idiom for embedding a literal single quote inside
   23 --     a single-quoted POSIX string.
   24 --
   25 -- The byte-for-byte semantics match the original
   26 -- 'PureClaw.Harness.Tmux.shellEscape' / 'shellEscapeStr' helpers, which
   27 -- have been in production use; this module is now the canonical
   28 -- location for new code, and the 'Harness.Tmux' names delegate here.
   29 --
   30 -- See @docs\/terminal-backend-abstractions.md@ § \"Remote arg quoting\"
   31 -- and § \"Scope of Harness.Tmux migration\".
   32 module PureClaw.Internal.ShellQuote
   33   ( shellQuote
   34   , shellQuoteString
   35   ) where
   36 
   37 import Data.Text (Text)
   38 import Data.Text qualified as T
   39 
   40 -- | Quote a 'Text' for inclusion as a single argument in a POSIX
   41 -- shell command line.
   42 --
   43 -- See the module haddock for the full rules. Examples:
   44 --
   45 -- >>> shellQuote ""
   46 -- "''"
   47 --
   48 -- >>> shellQuote "hello"
   49 -- "hello"
   50 --
   51 -- >>> shellQuote "hello world"
   52 -- "'hello world'"
   53 --
   54 -- >>> shellQuote "it's"
   55 -- "'it'\\''s'"
   56 shellQuote :: Text -> Text
   57 shellQuote t
   58   | T.null t = "''"
   59   | T.all isSafe t = t
   60   | otherwise = "'" <> T.replace "'" "'\\''" t <> "'"
   61   where
   62     isSafe c = c `elem` safeChars
   63     safeChars = ['a'..'z'] <> ['A'..'Z'] <> ['0'..'9'] <> "-_./=:@"
   64 
   65 -- | 'String'-typed convenience wrapper around 'shellQuote' for callers
   66 -- working with 'FilePath' or other @String@-shaped argv values.
   67 shellQuoteString :: String -> String
   68 shellQuoteString = T.unpack . shellQuote . T.pack
