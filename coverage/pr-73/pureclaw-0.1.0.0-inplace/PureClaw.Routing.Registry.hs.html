    1 -- |
    2 -- Module      : PureClaw.Routing.Registry
    3 -- Description : Pure tab CRUD over the AgentEnv's tab map (Tabbed Chat WU3).
    4 --
    5 -- This module owns the in-memory tab registry — an
    6 -- @'IORef' ('IntMap' 'TabHandle')@ stored as
    7 -- 'PureClaw.Agent.Env._env_tabs'. Functions here are deliberately
    8 -- minimal: lookup, insert (rejecting the in-use case), remove, and a
    9 -- @lowestFreeIndex@ helper used by the auto-spawn path.
   10 --
   11 -- No factory dispatch (that belongs to 'PureClaw.Routing.Dispatcher',
   12 -- WU5), no spawning, no async exception discipline, no side effects
   13 -- beyond reading\/writing the supplied 'IORef'.
   14 --
   15 -- See @docs\/tabbed-chat.md@ §\"Registry & AgentEnv (E-series)\" for
   16 -- the design context.
   17 module PureClaw.Routing.Registry
   18   ( -- * Pure tab CRUD
   19     lookupTab
   20   , insertTab
   21   , removeTab
   22   , lowestFreeIndex
   23     -- * tmux-style packing helper
   24   , packAfterRemove
   25   ) where
   26 
   27 import Data.IORef (IORef, atomicModifyIORef', readIORef)
   28 import Data.IntMap.Strict (IntMap)
   29 import Data.IntMap.Strict qualified as IntMap
   30 
   31 import PureClaw.Handles.Tab (TabError (..), TabHandle, TabIndex, mkTabIndex, unTabIndex)
   32 
   33 -- | Look up a tab by index. Returns 'Nothing' if no tab is registered
   34 -- at that slot.
   35 lookupTab :: IORef (IntMap TabHandle) -> TabIndex -> IO (Maybe TabHandle)
   36 lookupTab ref idx = do
   37   m <- readIORef ref
   38   pure (IntMap.lookup (unTabIndex idx) m)
   39 
   40 -- | Insert a tab handle at the given index. Returns
   41 -- @'Left' ('TabIndexInUse' idx)@ if the slot is already occupied
   42 -- (callers must explicitly call 'removeTab' first if they want to
   43 -- replace an existing tab).
   44 insertTab :: IORef (IntMap TabHandle) -> TabIndex -> TabHandle
   45           -> IO (Either TabError ())
   46 insertTab ref idx h =
   47   atomicModifyIORef' ref $ \m ->
   48     case IntMap.lookup (unTabIndex idx) m of
   49       Just _  -> (m, Left (TabIndexInUse idx))
   50       Nothing -> (IntMap.insert (unTabIndex idx) h m, Right ())
   51 
   52 -- | Remove the tab at the given index. Returns the removed handle (or
   53 -- 'Nothing' if no tab was present). The caller is responsible for
   54 -- running any kind-specific close logic on the returned handle.
   55 removeTab :: IORef (IntMap TabHandle) -> TabIndex -> IO (Maybe TabHandle)
   56 removeTab ref idx =
   57   atomicModifyIORef' ref $ \m ->
   58     case IntMap.lookup (unTabIndex idx) m of
   59       Nothing -> (m, Nothing)
   60       Just h  -> (IntMap.delete (unTabIndex idx) m, Just h)
   61 
   62 -- | Find the smallest non-negative 'TabIndex' that is strictly less
   63 -- than the supplied @maxN@ ceiling AND is not currently occupied.
   64 -- Returns 'Nothing' when every slot @0 .. maxN-1@ is taken (or when
   65 -- @maxN <= 0@).
   66 --
   67 -- The expected caller is the auto-spawn path (WU9): given the
   68 -- routing config's @_rc_maxTabs@, find a slot for a new tab.
   69 lowestFreeIndex :: IORef (IntMap TabHandle) -> Int -> IO (Maybe TabIndex)
   70 lowestFreeIndex ref maxN
   71   | maxN <= 0 = pure Nothing
   72   | otherwise = do
   73       m <- readIORef ref
   74       pure (firstFree m 0 maxN)
   75   where
   76     firstFree m n limit
   77       | n >= limit         = Nothing
   78       | IntMap.member n m  = firstFree m (n + 1) limit
   79       | otherwise          = mkTabIndex n
   80 
   81 -- | Pure variant of the tmux-style \"renumber-windows\" pass: given a
   82 -- map whose entry at index @k@ has already been removed, shift every
   83 -- entry strictly greater than @k@ down by one key. Keys @\< k@ are
   84 -- left untouched.
   85 --
   86 -- This is a no-op when no key exceeds @k@. The pass is value-preserving
   87 -- (only keys are remapped); duplicates cannot arise because the
   88 -- pre-shift map has unique keys and the shift is monotone.
   89 packAfterRemove :: Int -> IntMap a -> IntMap a
   90 packAfterRemove k =
   91   -- Walk in ascending order so the destination key for an entry is
   92   -- always free by the time we write it (the previous entry at that
   93   -- key was either removed or also shifted down).
   94   IntMap.foldlWithKey'
   95     (\acc i v ->
   96        if i > k
   97          then IntMap.insert (i - 1) v acc
   98          else IntMap.insert i v acc)
   99     IntMap.empty
