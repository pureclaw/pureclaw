   12 -- | When the list is open, write the banner lines to @h@ and emit the WARN
   13 --   log line via the LogHandle. No-op when senders are restricted. @h@ is
   14 --   injectable so tests can capture output to a temp-file Handle.
   15 emitAllowListWarning :: Handle -> LogHandle -> AllowListContext -> AllowList a -> IO ()
   16 emitAllowListWarning h lh ctx al = case allowListWarning ctx al of
   17   Nothing -> pure ()
   18   Just (banner, logMsg) -> do
   19     mapM_ (TIO.hPutStrLn h) banner
   20     _lh_logWarn lh logMsg
   21 
   22 -- | Convenience for live call sites: banner to stdout (matching the existing
   23 --   @PureClaw 0.1.0@ startup banner), log line to stderr via the LogHandle.
   24 warnIfOpenAllowList :: LogHandle -> AllowListContext -> AllowList a -> IO ()
   25 warnIfOpenAllowList = emitAllowListWarning stdout
