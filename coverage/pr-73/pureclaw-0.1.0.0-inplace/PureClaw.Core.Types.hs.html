  101 -- | The kind of channel a message originated from. Known channels get a
  102 -- typed constructor; 'CkOther' is an open escape hatch for channels that
  103 -- do not (yet) have a dedicated tag.
  104 data ChannelKind
  105   = CkCli
  106   | CkWeb
  107   | CkSignal
  108   | CkTelegram
  109   | CkBackground
  110   | CkOther !Text
  111   deriving stock (Show, Eq, Generic)
  112 
  113 -- | Render a 'ChannelKind' as a flat lowercase tag. 'CkOther' carries its
  114 -- own name verbatim. Mirrors the @flavourToText@ precedent.
  115 channelKindToText :: ChannelKind -> Text
  116 channelKindToText CkCli        = "cli"
  117 channelKindToText CkWeb        = "web"
  118 channelKindToText CkSignal     = "signal"
  119 channelKindToText CkTelegram   = "telegram"
  120 channelKindToText CkBackground = "background"
  121 channelKindToText (CkOther n)  = n
  122 
  123 -- | Parse a flat tag back into a 'ChannelKind'. Known names map to their
  124 -- typed constructor; everything else becomes 'CkOther'. Critically,
  125 -- @"signal"@ maps to 'CkSignal' (never @CkOther "signal"@), so a
  126 -- 'CkOther' naming a known channel cannot round-trip into existence.
  127 channelKindFromText :: Text -> ChannelKind
  128 channelKindFromText "cli"        = CkCli
  129 channelKindFromText "web"        = CkWeb
  130 channelKindFromText "signal"     = CkSignal
  131 channelKindFromText "telegram"   = CkTelegram
  132 channelKindFromText "background" = CkBackground
  133 channelKindFromText t            = CkOther t
  134 
  135 instance Aeson.ToJSON ChannelKind where
  136   toJSON = Aeson.String . channelKindToText
  137 
  138 instance Aeson.FromJSON ChannelKind where
  139   parseJSON = Aeson.withText "ChannelKind" (pure . channelKindFromText)

  141 -- | Maximum length (in characters) for any attacker-controlled string
  142 -- stored inside a 'MessageSource' after normalization.
  143 maxSourceLen :: Int
  144 maxSourceLen = 512
  145 
  146 -- | Where an inbound message came from. Structured and extensible: a typed
  147 -- 'ChannelKind', the channel's user id (when one exists), and an open
  148 -- field map for supplementary, channel-specific data (which may carry
  149 -- nested JSON).
  150 --
  151 -- Construct via 'mkMessageSource', never the raw record — the smart
  152 -- constructor normalizes attacker-controlled input.
  153 --
  154 -- Invariants (enforced by convention, not the type): '_ms_fields' must not
  155 -- duplicate '_ms_userId' and must never contain credentials/secrets.
  156 data MessageSource = MessageSource
  157   { _ms_channel :: !ChannelKind
  158   , _ms_userId  :: !(Maybe UserId)
  159   , _ms_fields  :: !(Map Text Aeson.Value)
  160   } deriving stock (Show, Eq, Generic)
  161 
  162 -- | Strip ASCII control characters (covers newlines, tabs, carriage
  163 -- returns and other control bytes) and bound the result to 'maxSourceLen'
  164 -- characters.
  165 normalizeText :: Text -> Text
  166 normalizeText = T.take maxSourceLen . T.filter (not . Char.isControl)
  167 
  168 -- | Recursively normalize every 'Aeson.String' leaf in a JSON value,
  169 -- leaving the structure otherwise intact.
  170 normalizeValue :: Aeson.Value -> Aeson.Value
  171 normalizeValue (Aeson.String t) = Aeson.String (normalizeText t)
  172 normalizeValue (Aeson.Array xs) = Aeson.Array (fmap normalizeValue xs)
  173 normalizeValue (Aeson.Object o) = Aeson.Object (fmap normalizeValue o)
  174 normalizeValue v                = v
  175 
  176 -- | Smart constructor for 'MessageSource'. Normalizes attacker-controlled
  177 -- input: strips control characters / newlines and bounds length on the
  178 -- user id and on every string leaf inside the field map, and folds a
  179 -- 'CkOther' naming a known channel (e.g. @CkOther "signal"@) down to its
  180 -- typed constructor.
  181 mkMessageSource :: ChannelKind -> Maybe UserId -> Map Text Aeson.Value -> MessageSource
  182 mkMessageSource ch uid fields = MessageSource
  183   { _ms_channel = foldChannel ch
  184   , _ms_userId  = fmap (UserId . normalizeText . unUserId) uid
  185   , _ms_fields  = fmap normalizeValue fields
  186   }
  187   where
  188     foldChannel (CkOther n) = channelKindFromText n
  189     foldChannel c           = c
  190 
  191 instance Aeson.ToJSON MessageSource where
  192   toJSON s = Aeson.object $
  193     ["channel" .= _ms_channel s]
  194       <> case _ms_userId s of
  195         Just u  -> ["user_id" .= unUserId u]
  196         Nothing -> []
  197       <> ["fields" .= _ms_fields s | not (Map.null (_ms_fields s))]
  198 
  199 instance Aeson.FromJSON MessageSource where
  200   parseJSON = Aeson.withObject "MessageSource" $ \o -> do
  201     ch     <- o .:  "channel"
  202     uid    <- o .:? "user_id"
  203     fields <- o .:? "fields" .!= Map.empty
  204     pure MessageSource
  205       { _ms_channel = ch
  206       , _ms_userId  = fmap UserId uid
  207       , _ms_fields  = fields
  208       }

  221 -- | Typed allow-list. @AllowAll@ explicitly opts in to allowing everything.
  222 -- @AllowList s@ restricts to the given set.
  223 data AllowList a
  224   = AllowAll
  225   | AllowList (Set a)
  226   deriving stock (Show, Eq, Generic)
  227 
  228 -- | Check whether a value is permitted by the allow-list.
  229 isAllowed :: Ord a => AllowList a -> a -> Bool
  230 isAllowed AllowAll      _ = True
  231 isAllowed (AllowList s) x = Set.member x s
  232 
  233 -- | True when the list permits every sender (no restriction configured).
  234 allowListOpen :: AllowList a -> Bool
  235 allowListOpen AllowAll      = True
  236 allowListOpen (AllowList _) = False

  238 -- | Context needed to render an actionable open-allow-list warning. Carries
  239 --   both the human-readable channel name (for prose) and the lowercase TOML
  240 --   table key (for the copy-pasteable config example) so the two never get
  241 --   conflated.
  242 data AllowListContext = AllowListContext
  243   { _alc_channelName :: Text  -- ^ Human-readable channel name, e.g. "Signal"
  244   , _alc_configTable :: Text  -- ^ TOML table key, e.g. "signal"
  245   , _alc_exampleId   :: Text  -- ^ Example allow_from entry, e.g. a Signal user UUID or a numeric Telegram id
  246   }
  247   deriving stock (Show, Eq)

  249 -- | Banner lines plus a WARN log line describing an OPEN allow-list on the
  250 --   named channel, or Nothing when senders are restricted. Pure.
  251 --
  252 --   When open, the banner is prominent AND actionable: it tells the operator
  253 --   exactly which config file to edit, which (lowercase) TOML table to add an
  254 --   @allow_from@ list under, and shows a concrete copy-pasteable example. The
  255 --   config table key (never the display name) is used everywhere a TOML table
  256 --   is shown.
  257 allowListWarning :: AllowListContext -> AllowList a -> Maybe ([Text], Text)
  258 allowListWarning ctx al
  259   | not (allowListOpen al) = Nothing
  260   | otherwise = Just (banner, logLine)
  261   where
  262     name  = _alc_channelName ctx
  263     table = _alc_configTable ctx
  264     exId  = _alc_exampleId ctx
  265     rule  = T.replicate 64 "="
  266     banner =
  267       [ rule
  268       , "  SECURITY WARNING: channel \"" <> name <> "\" has NO allow-list configured."
  269       , "  It will accept messages from ANY sender, which is a security risk."
  270       , ""
  271       , "  To restrict who may message this agent, edit your config file:"
  272       , "      ~/.pureclaw/config.toml"
  273       , "  and add an `allow_from` list under the [" <> table <> "] section, e.g.:"
  274       , ""
  275       , "      [" <> table <> "]"
  276       , "      allow_from = [\"" <> exId <> "\"]"
  277       , ""
  278       , "  List one ID per allowed sender, then restart PureClaw."
  279       , rule
  280       ]
  281     logLine =
  282       "channel \"" <> name <> "\" has no allow-list configured; "
  283         <> "accepting messages from any sender "
  284         <> "(set allow_from under [" <> table <> "] in config.toml to restrict)"
