    1 -- | Registry-based reconcile loop (Harness Registry & Lifecycle, Phase 1 — WU5).
    2 --
    3 -- This module is the registry's source of truth for harness /health/ and the
    4 -- single authoritative updater of each entry's coordinate + liveness. It
    5 -- replaces the legacy harness-map activity probe ("PureClaw.Frontend.ActivityProbe")
    6 -- with a loop that reconciles the durable registry against tmux reality.
    7 --
    8 -- == What the loop does, each tick
    9 --
   10 --   1. ONE server sweep per session ('_rd_sweep') — never per-I\/O.
   11 --   2. /Ownership filter/ (§8 C4 \/ D5.5\/D5.7): retain only rows whose
   12 --      @\@pcl_id@ matches a registry entry AND is PID-corroborated against the
   13 --      entry's recorded shell\/harness PID. A spoofed marker (collision or PID
   14 --      mismatch) is logged and treated as "not ours" — never captured, never
   15 --      marked live. A row with no recorded shell PID falls back to a second
   16 --      signal (the window-name prefix) to defend against PID reuse.
   17 --   3. Liveness classification (ours only): @Exited@ on @pane_dead@ or a dead
   18 --      harness PID; @Idle@\/@Thinking@ via a screen capture + 'isIdle'; an
   19 --      entry whose corroborated window is absent from the sweep → @Orphaned@.
   20 --   4. 'Reg.mergeReconcile' the observed fields into the registry atomically.
   21 --   5. /Symmetric diff/ (D5.2): emit one 'ActivityChanged' per entry whose
   22 --      liveness changed since the previous tick — INCLUDING entries that
   23 --      disappeared (now @Orphaned@\/@Exited@), fixing the disappearance gap.
   24 --      The first tick establishes the baseline and emits nothing.
   25 --
   26 -- == Resilience (D5.3)
   27 --
   28 -- A transient sweep\/capture failure marks the affected entries '_he_stale'
   29 -- and the loop CONTINUES — it never dies except on 'AsyncCancelled', which is
   30 -- re-raised so 'Async.withAsync' cleanup runs (the project-wide invariant; cf.
   31 -- "PureClaw.Frontend.ActivityProbe").
   32 --
   33 -- == Boot reconstruction (D5.6)
   34 --
   35 -- 'bootReconstruct' runs one sweep at startup and registers an entry for every
   36 -- window carrying our @\@pcl_id@ (the PCL-restart reconnect path), and lazily
   37 -- stamps a fresh 'Reg.HarnessId' onto legacy @claude-code-\<idx\>@ windows that
   38 -- carry no marker. The legacy harness map is still seeded in parallel by the
   39 -- unchanged @discoverHarnesses@; the registry is the new parallel path this
   40 -- loop owns.

   96 -- ---------------------------------------------------------------------------
   97 -- Dependencies (IO seam — tests inject deterministic fakes)

  100 -- | The injectable tmux operations the reconcile loop needs. Tests supply
  101 -- fakes (a queue of sweeps, a capture recorder) so the loop is deterministic
  102 -- and never forks a real tmux server. The production set is
  103 -- 'defaultReconcileDeps'.
  104 data ReconcileDeps = ReconcileDeps
  105   { _rd_sessions :: IO [Text]
  106     -- ^ Enumerate the sessions to sweep (production: the @\"pureclaw\"@ session
  107     --   plus any other session on the server, via 'listTmuxSessions').
  108   , _rd_sweep :: Text -> IO [TmuxWindowRow]
  109     -- ^ One server sweep of a session (production: 'readMarkers'). May throw on
  110     --   a transient tmux failure; the loop catches it and marks entries stale.
  111   , _rd_capture :: Text -> Text -> IO Bool
  112     -- ^ Capture @session windowName@ and report whether the screen reads idle
  113     --   (production: 'captureWindowNamed' + 'isIdle'). Only called for
  114     --   corroborated, live (not pane_dead) windows.
  115   , _rd_harnessAlive :: Int -> IO Bool
  116     -- ^ Whether a recorded harness PID is still alive (production: a
  117     --   'harnessPidOf'-style liveness probe). Distinguishes @Exited@
  118     --   (harness gone, shell present) from @Idle@\/@Thinking@.
  119   , _rd_stampLegacy :: Text -> Text -> IO (Maybe Reg.HarnessId)
  120     -- ^ Lazily stamp a legacy @claude-code-\<idx\>@ window (no @\@pcl_id@) with
  121     --   a fresh marker + id, returning the new id (production: generate an id,
  122     --   'setWindowMarker'). 'Nothing' for a window we do not own.
  123   , _rd_evict :: Reg.HarnessId -> Text -> IO ()
  124     -- ^ Auto-eviction seam (design §5\/§10 Q2). Called with the
  125     --   @(harnessId, label)@ of an entry that has been Orphaned for
  126     --   'defaultOrphanGraceTicks' consecutive ticks. Production wires this to
  127     --   drop the entry from the LEGACY @_env_harnesses@ map (the registry
  128     --   delete is done by the loop itself, in 'reconcileTick'); it must NOT
  129     --   touch @session.json@ so the session reappears in Recent Sessions.
  130     --   Tests inject a recorder to assert the @(id, label)@ that was evicted.
  131   }
  132 
  133 -- | Production dependency set. Sweeps the @\"pureclaw\"@ session plus any other
  134 -- session on the server; capture\/idle via 'captureWindowNamed' + 'isIdle';
  135 -- harness liveness via 'harnessPidOf'; legacy stamping via 'setWindowMarker'.
  136 defaultReconcileDeps :: ReconcileDeps
  137 defaultReconcileDeps = ReconcileDeps
  138   { _rd_sessions = do
  139       others <- listTmuxSessions
  140       -- Always include the default session even if it has no windows yet.
  141       pure (dedup ("pureclaw" : others))
  142   , _rd_sweep = readMarkers
  143   , _rd_capture = \session windowName -> do
  144       bytes <- captureWindowNamed session windowName 50
  145       pure (isIdle (TE.decodeUtf8Lenient bytes))
  146   , _rd_harnessAlive = \shellPid -> do
  147       -- Re-derive the harness PID by descending from the recorded shell PID and
  148       -- matching the flavour comm. If the descent still finds the agent binary
  149       -- the harness is alive; if it has exited, the descent returns 'Nothing'
  150       -- (the shell may still be present — that is exactly the @Exited@ case the
  151       -- @pane_dead@ flag also covers under @remain-on-exit on@).
  152       mPid <- harnessPidOf shellPid "claude"
  153       pure (isJust mPid)
  154   , _rd_stampLegacy = \session windowName ->
  155       if isLegacyWindowName windowName
  156         then do
  157           hid <- Reg.newHarnessId
  158           setWindowMarker session windowName (Reg.harnessIdToText hid)
  159           pure (Just hid)
  160         else pure Nothing
  161     -- Default eviction is a no-op: the loop itself deletes the evicted entry
  162     -- from the registry; the legacy-map drop is wired in CLI.Commands (which
  163     -- overrides this seam via 'runReconcileLoopWith'). Crucially this never
  164     -- deletes @session.json@.
  165   , _rd_evict = \_hid _label -> pure ()
  166   }
  167   where
  168     dedup = go []
  169       where
  170         go seen [] = reverse seen
  171         go seen (x : xs)
  172           | x `elem` seen = go seen xs
  173           | otherwise     = go (x : seen) xs

  179 -- | The result of corroborating a marker row against a registry entry.
  180 data CorroborationResult
  181   = Corroborated
  182     -- ^ The row is genuinely this entry's window: PID-provenance (or, for a
  183     --   provenance-less entry, the window-name prefix) confirms it.
  184   | PidMismatch
  185     -- ^ The marker matches but the recorded shell PID does not (or the second
  186     --   signal fails) — a spoof\/PID-reuse: treat as NOT ours, never trust.
  187   deriving stock (Eq, Show)
  188 
  189 -- | Corroborate a marker row (already known to carry this entry's @\@pcl_id@)
  190 -- against the entry's recorded provenance (§8 C4):
  191 --
  192 --   * If we recorded a shell PID, the row's @#{pane_pid}@ MUST match it.
  193 --   * If we recorded NO shell PID (legacy\/adopted), require a SECOND signal —
  194 --     the @claude-code-\<idx\>@-style window-name prefix matching the entry's
  195 --     label — to defend against PID reuse.
  196 corroborate :: Reg.HarnessEntry -> TmuxWindowRow -> CorroborationResult
  197 corroborate e row =
  198   case Reg._he_shellPid e of
  199     Just recordedPid
  200       | _twr_panePid row == Just recordedPid -> Corroborated
  201       | otherwise                            -> PidMismatch
  202     Nothing
  203       -- No recorded provenance: fall back to the window-name prefix as the
  204       -- second corroborating signal.
  205       | _twr_windowName row == Reg._he_label e -> Corroborated
  206       | otherwise                              -> PidMismatch

  209 -- Pure liveness classification (D5.4)

  212 -- | Classify a corroborated window's liveness from @(paneDead, harnessAlive,
  213 -- isIdle)@:
  214 --
  215 --   * @pane_dead@ OR a dead harness PID ⇒ 'Reg.LivenessExited' (the window is
  216 --     present but the agent is gone).
  217 --   * otherwise the screen capture decides: idle ⇒ 'Reg.LivenessIdle',
  218 --     busy ⇒ 'Reg.LivenessThinking'.
  219 --
  220 -- 'Reg.LivenessOrphaned' is NOT produced here — it is assigned to entries whose
  221 -- corroborated window is /absent/ from the sweep (see 'reconcileTick').
  222 classifyLiveness :: Bool -> Bool -> Bool -> Reg.Liveness
  223 classifyLiveness paneDead harnessAlive idle
  224   | paneDead || not harnessAlive = Reg.LivenessExited
  225   | idle                         = Reg.LivenessIdle
  226   | otherwise                    = Reg.LivenessThinking
  227 
  228 -- | Map a registry 'Reg.Liveness' to the broker's 'HarnessActivity' vocabulary.
  229 -- Both @Exited@ and @Orphaned@ collapse to 'HarnessStopped' (the frontend's
  230 -- richer state→glyph mapping is Phase 2).
  231 livenessToActivity :: Reg.Liveness -> HarnessActivity
  232 livenessToActivity Reg.LivenessIdle     = HarnessIdle
  233 livenessToActivity Reg.LivenessThinking = HarnessThinking
  234 livenessToActivity Reg.LivenessExited   = HarnessStopped
  235 livenessToActivity Reg.LivenessOrphaned = HarnessStopped

  241 -- | Compute the @(sessionId, liveness)@ transitions between two ticks' liveness
  242 -- snapshots, keyed by 'Reg.HarnessId' (rendered as 'Text' for test ergonomics).
  243 -- Each value carries the SessionId the event should be published under plus the
  244 -- current liveness.
  245 --
  246 -- An entry whose liveness CHANGED (or that is newly present) emits its new
  247 -- liveness. Crucially, an entry that went 'Reg.LivenessOrphaned'\/'Reg.LivenessExited'
  248 -- (because its window vanished from the sweep) is still present in @next@ — the
  249 -- registry retains it — so the transition surfaces, fixing the disappearance
  250 -- gap (@ActivityProbe.hs:117@). A dropped key (registry entry removed entirely)
  251 -- emits nothing; eviction is a Phase-2 concern.
  252 diffLiveness
  253   :: Map Text (Text, Reg.Liveness)  -- ^ previous tick
  254   -> Map Text (Text, Reg.Liveness)  -- ^ current tick
  255   -> [(Text, Reg.Liveness)]         -- ^ @(sessionId, newLiveness)@ to publish
  256 diffLiveness prev next =
  257   [ (sid, liveness)
  258   | (key, (sid, liveness)) <- Map.toList next
  259   , case Map.lookup key prev of
  260       Nothing            -> True                  -- newly observed
  261       Just (_, oldLive)  -> oldLive /= liveness   -- transitioned
  262   ]

  264 -- ---------------------------------------------------------------------------
  265 -- Reconcile tick (IO)
  266 -- ---------------------------------------------------------------------------

  268 -- | A single reconcile pass: sweep, corroborate, classify, merge, and return
  269 -- the per-entry liveness snapshot keyed by 'Reg.HarnessId' text (for the loop's
  270 -- diff). On a transient sweep failure the registry's entries are marked
  271 -- '_he_stale' (holding last-known liveness) and the snapshot reflects the held
  272 -- state; the tick never throws on a sweep failure.
  273 reconcileTick
  274   :: ReconcileDeps
  275   -> Reg.HarnessRegistry
  276   -> LogHandle
  277   -> IO (Map Text (Text, Reg.Liveness), [Text])
  278   -- ^ @(liveness snapshot keyed by id-text, sessionIds of entries auto-evicted
  279   --   this tick)@. Evicted entries are removed from the snapshot map (so the
  280   --   next diff does not re-report them) and the loop emits one final
  281   --   disappearance event per returned sessionId.
  282 reconcileTick deps reg logger = do
  283   sessions <- _rd_sessions deps
  284   -- Sweep every session; a failed sweep yields Nothing (we then hold stale).
  285   sweepResults <- forM sessions $ \session -> do
  286     r <- try @SomeException (_rd_sweep deps session)
  287     case r of
  288       Left err -> do
  289         _lh_logWarn logger
  290           ("reconcile: sweep of session " <> session <> " failed: " <> T.pack (show err))
  291         pure Nothing
  292       Right rows -> pure (Just rows)
  293   let sweepFailed = Nothing `elem` sweepResults
  294       allRows     = concat (catMaybes sweepResults)
  295   entries <- Reg.snapshot reg
  296   -- Build, per entry, its observed fields (or the stale/orphaned fallback).
  297   observedPairs <- forM entries $ \e -> do
  298     let idText  = Reg.harnessIdToText (Reg._he_id e)
  299         matches = [ row | row <- allRows, _twr_pclId row == idText ]
  300     case matches of
  301       [] | sweepFailed ->
  302             -- No row AND a sweep failed: we cannot conclude the window is gone.
  303             -- Hold last-known liveness; mark stale.
  304             pure (mkObservedHeld e)
  305          | otherwise ->
  306             -- Sweep succeeded and our window is absent ⇒ Orphaned.
  307             pure (mkObservedOrphaned e)
  308       _ -> do
  309         -- Corroborate each candidate; trust only the first corroborated row.
  310         let corroboratedRows =
  311               [ row | row <- matches, corroborate e row == Corroborated ]
  312         case corroboratedRows of
  313           [] -> do
  314             -- A marker matched but corroboration failed (spoof / PID mismatch):
  315             -- log + treat as not ours; the entry is Orphaned, never captured.
  316             _lh_logWarn logger
  317               ("reconcile: @pcl_id " <> idText
  318                  <> " matched a window with no corroborating PID — treating as not ours (§8 C4)")
  319             pure (mkObservedOrphaned e)
  320           (row : _) -> do
  321             -- The per-window capture + harness-alive probe is the only other
  322             -- tmux IO in the tick (besides the sweep, handled above). A
  323             -- transient failure here must NOT kill the loop (D5.3): hold this
  324             -- entry's last-known liveness and mark it stale, exactly as a sweep
  325             -- failure does — never repaint it from a capture error, never let
  326             -- the exception propagate to the loop's crash handler.
  327             r <- try @SomeException (classifyRow deps e row)
  328             case r of
  329               Left err -> do
  330                 _lh_logWarn logger
  331                   ("reconcile: capture/liveness probe for @pcl_id " <> idText
  332                      <> " failed: " <> T.pack (show err) <> " — holding last-known liveness")
  333                 pure (mkObservedHeld e)
  334               Right liveness ->
  335                 pure (mkObservedLive e row liveness)
  336   let observed = map snd observedPairs
  337   Reg.mergeReconcile reg observed
  338   let snapMap = Map.fromList [ (k, v) | (k, v) <- map fst observedPairs ]
  339   -- Orphan grace policy (design §5 / §10 Q2): after the merge, evict any entry
  340   -- that has now been Orphaned for 'defaultOrphanGraceTicks' consecutive ticks.
  341   -- Eviction drops the entry from the registry (here) and from the legacy map
  342   -- (via the injectable '_rd_evict' seam, wired in CLI.Commands) — but never
  343   -- touches @session.json@. The evicted ids are removed from the returned
  344   -- snapshot so the next tick's diff does not re-report them; their sessionIds
  345   -- are returned so the loop can emit one final disappearance event.
  346   merged <- Reg.snapshot reg
  347   let toEvict =
  348         [ e | e <- merged, Reg._he_orphanedTicks e >= defaultOrphanGraceTicks ]
  349   evictedSids <- forM toEvict $ \e -> do
  350     _lh_logInfo logger
  351       ("reconcile: evicting harness " <> Reg.harnessIdToText (Reg._he_id e)
  352          <> " (label " <> Reg._he_label e <> ") after "
  353          <> T.pack (show (Reg._he_orphanedTicks e))
  354          <> " orphaned ticks; session.json retained")
  355     _rd_evict deps (Reg._he_id e) (Reg._he_label e)
  356     Reg.deleteEntry reg (Reg._he_id e)
  357     pure (Reg.harnessIdToText (Reg._he_id e), sidOf e)
  358   let evictedKeys = map fst evictedSids
  359       snapMap'    = foldr Map.delete snapMap evictedKeys
  360   pure (snapMap', map snd evictedSids)
  361   where
  362     -- Each result is ((keyText, (sessionId, liveness)), ObservedHarness).
  363     mkObservedHeld :: Reg.HarnessEntry -> ((Text, (Text, Reg.Liveness)), Reg.ObservedHarness)
  364     mkObservedHeld e =
  365       let live = Reg._he_liveness e
  366       in ( (Reg.harnessIdToText (Reg._he_id e), (sidOf e, live))
  367          , (baseObserved e live) { Reg._oh_stale = True } )
  368 
  369     mkObservedOrphaned :: Reg.HarnessEntry -> ((Text, (Text, Reg.Liveness)), Reg.ObservedHarness)
  370     mkObservedOrphaned e =
  371       ( (Reg.harnessIdToText (Reg._he_id e), (sidOf e, Reg.LivenessOrphaned))
  372       , (baseObserved e Reg.LivenessOrphaned)
  373           { Reg._oh_liveness = Reg.LivenessOrphaned
  374             -- The corroborated window is absent this tick: advance the
  375             -- consecutive-orphaned counter so the grace policy can fire.
  376           , Reg._oh_orphanedTicks = Reg._he_orphanedTicks e + 1
  377           } )
  378 
  379     mkObservedLive
  380       :: Reg.HarnessEntry -> TmuxWindowRow -> Reg.Liveness
  381       -> ((Text, (Text, Reg.Liveness)), Reg.ObservedHarness)
  382     mkObservedLive e row liveness =
  383       let extMod = _twr_windowName row /= Reg._he_windowName e
  384       in ( (Reg.harnessIdToText (Reg._he_id e), (sidOf e, liveness))
  385          , Reg.ObservedHarness
  386              { Reg._oh_id          = Reg._he_id e
  387              , Reg._oh_session     = Reg._he_session e
  388              , Reg._oh_windowName  = _twr_windowName row
  389              , Reg._oh_shellPid    = _twr_panePid row
  390              , Reg._oh_harnessPid  = Reg._he_harnessPid e
  391              , Reg._oh_liveness    = liveness
  392              , Reg._oh_extModified = Reg._he_extModified e || extMod
  393              , Reg._oh_stale       = False
  394                -- The window is present + corroborated this tick: the entry is
  395                -- live again, so reset the consecutive-orphaned counter.
  396              , Reg._oh_orphanedTicks = 0
  397              } )
  398 
  399     -- A held-state observed record reusing the entry's current coordinate.
  400     baseObserved :: Reg.HarnessEntry -> Reg.Liveness -> Reg.ObservedHarness
  401     baseObserved e liveness = Reg.ObservedHarness
  402       { Reg._oh_id          = Reg._he_id e
  403       , Reg._oh_session     = Reg._he_session e
  404       , Reg._oh_windowName  = Reg._he_windowName e
  405       , Reg._oh_shellPid    = Reg._he_shellPid e
  406       , Reg._oh_harnessPid  = Reg._he_harnessPid e
  407       , Reg._oh_liveness    = liveness
  408       , Reg._oh_extModified = Reg._he_extModified e
  409       , Reg._oh_stale       = False
  410         -- Hold the current counter: a held (stale) observation neither advances
  411         -- nor resets the grace clock; only a definite Orphaned/live tick does.
  412       , Reg._oh_orphanedTicks = Reg._he_orphanedTicks e
  413       }
  414 
  415     sidOf :: Reg.HarnessEntry -> Text
  416     sidOf e = fromMaybe (Reg._he_label e) (Reg._he_sessionId e)

  418 -- | Classify a corroborated window's liveness, capturing the screen only when
  419 -- the window is alive (not pane_dead and the harness PID is present). A
  420 -- @pane_dead@ window is NEVER captured (D5.4\/D5.5).
  421 classifyRow :: ReconcileDeps -> Reg.HarnessEntry -> TmuxWindowRow -> IO Reg.Liveness
  422 classifyRow deps e row
  423   | _twr_paneDead row = pure Reg.LivenessExited
  424   | otherwise = do
  425       -- Liveness of the agent process: re-derive it by descending from the live
  426       -- shell PID (the row's @#{pane_pid}@). We only bother when we recorded a
  427       -- harness PID at spawn (a provenance-less/legacy entry has none, so we
  428       -- assume alive and rely on @pane_dead@ for Exited).
  429       alive <- case (Reg._he_harnessPid e, _twr_panePid row) of
  430         (Just _, Just shellPid) -> _rd_harnessAlive deps shellPid
  431         _                       -> pure True
  432       if not alive
  433         then pure Reg.LivenessExited
  434         else do
  435           idle <- _rd_capture deps (Reg._he_session e) (_twr_windowName row)
  436           pure (classifyLiveness False True idle)

  442 -- | Production reconcile loop: 2-second cadence, real tmux deps.
  443 runReconcileLoop :: StreamBroker -> Reg.HarnessRegistry -> LogHandle -> IO ()
  444 runReconcileLoop broker reg =
  445   runReconcileLoopWith defaultTickMicros defaultReconcileDeps reg broker
  446 
  447 -- | Production tick interval: 2 seconds (matches the legacy probe cadence).
  448 defaultTickMicros :: Int
  449 defaultTickMicros = 2_000_000
  450 
  451 -- | The orphan grace window, in consecutive reconcile ticks (design §5\/§10
  452 -- Q2). Once an entry has been classified 'Reg.LivenessOrphaned' for this many
  453 -- ticks in a row (i.e. '_he_orphanedTicks' reaches this value) the reconcile
  454 -- loop auto-evicts it from the registry and the legacy harness map and emits a
  455 -- final disappearance event — but never deletes @session.json@.
  456 --
  457 -- We count ticks rather than wall-clock so the policy is deterministic and
  458 -- test-injectable. At the production 2-second 'defaultTickMicros' cadence, 15
  459 -- ticks ≈ 30 seconds of continuous orphaning before eviction — long enough to
  460 -- ride out a transient tmux-server blip (which is held '_he_stale' and does
  461 -- NOT advance the counter), short enough that a genuinely gone harness clears
  462 -- from Active Tabs promptly.
  463 defaultOrphanGraceTicks :: Int
  464 defaultOrphanGraceTicks = 15
  465 
  466 -- | The reconcile loop with an explicit tick interval + injected deps. The
  467 -- first tick establishes the baseline (emits nothing); from the second tick on,
  468 -- one 'ActivityChanged' is published per entry whose liveness changed
  469 -- (including disappearances). Resilient: a transient sweep failure is absorbed
  470 -- by 'reconcileTick' (entries held stale); the loop only exits on
  471 -- 'AsyncCancelled', which it re-raises.
  472 runReconcileLoopWith
  473   :: Int
  474   -> ReconcileDeps
  475   -> Reg.HarnessRegistry
  476   -> StreamBroker
  477   -> LogHandle
  478   -> IO ()
  479 runReconcileLoopWith tickMicros deps reg broker logger = handle outer $ do
  480   threadDelay tickMicros
  481   -- Baseline tick: emits no transitions. Any eviction on this first tick still
  482   -- emits its final disappearance event (an eviction is not a diff transition).
  483   (initial, evicted0) <- reconcileTick deps reg logger
  484   publishEvictions evicted0
  485   loop initial
  486   where
  487     loop prev = do
  488       threadDelay tickMicros
  489       (next, evicted) <- reconcileTick deps reg logger
  490       forM_ (diffLiveness prev next) $ \(sid, liveness) ->
  491         _streamBroker_publish broker
  492           (ActivityChanged (SessionId sid) (SaHarnessStatus (livenessToActivity liveness)))
  493       -- One final disappearance event per auto-evicted entry. The entry was
  494       -- already Orphaned, so the diff above does NOT re-emit it; this guarantees
  495       -- the frontend sees it leave even though no liveness transition occurred.
  496       publishEvictions evicted
  497       loop next
  498 
  499     publishEvictions = mapM_ $ \sid ->
  500       _streamBroker_publish broker
  501         (ActivityChanged (SessionId sid) (SaHarnessStatus (livenessToActivity Reg.LivenessOrphaned)))
  502 
  503     -- AsyncCancelled MUST be re-raised so 'Async.withAsync'/'Async.cancel'
  504     -- complete cleanly (project-wide invariant).
  505     outer :: SomeException -> IO ()
  506     outer ex
  507       | Just AsyncCancelled <- fromException ex = throwIO ex
  508       | otherwise = _lh_logError logger
  509           ("reconcile loop crashed: " <> T.pack (show ex))
  510 
  511 -- ---------------------------------------------------------------------------
  512 -- Boot reconstruction (D5.6)
  513 -- ---------------------------------------------------------------------------
  514 
  515 -- | At startup, run one sweep and register an entry for every window we own:
  516 --
  517 --   * a window carrying our @\@pcl_id@ → an 'Reg.OriginDiscovered' entry (the
  518 --     PCL-restart reconnect path);
  519 --   * a legacy @claude-code-\<idx\>@ window with no marker → lazily stamped a
  520 --     fresh id (via '_rd_stampLegacy') and registered as 'Reg.OriginDiscovered'.
  521 --
  522 -- The legacy harness map is seeded separately by the unchanged
  523 -- @discoverHarnesses@; this builds the parallel registry the reconcile loop
  524 -- owns. Idempotent-ish: an already-registered id is overwritten with the
  525 -- freshly observed coordinate (boot runs once).
  526 bootReconstruct :: ReconcileDeps -> Reg.HarnessRegistry -> LogHandle -> IO ()
  527 bootReconstruct deps reg logger = do
  528   sessions <- _rd_sessions deps
  529   forM_ sessions $ \session -> do
  530     r <- try @SomeException (_rd_sweep deps session)
  531     case r of
  532       Left err ->
  533         _lh_logWarn logger
  534           ("bootReconstruct: sweep of " <> session <> " failed: " <> T.pack (show err))
  535       Right rows -> forM_ rows $ \row -> do
  536         case Reg.parseHarnessId (_twr_pclId row) of
  537           Just hid ->
  538             registerEntry session row hid
  539           Nothing -> do
  540             mHid <- _rd_stampLegacy deps session (_twr_windowName row)
  541             for_ mHid (registerEntry session row)  -- 'Nothing' ⇒ not ours
  542   where
  543     registerEntry session row hid = do
  544       let live = if _twr_paneDead row then Reg.LivenessExited else Reg.LivenessIdle
  545       Reg.insertEntry reg Reg.HarnessEntry
  546         { Reg._he_id          = hid
  547         , Reg._he_session     = session
  548         , Reg._he_windowName  = _twr_windowName row
  549         , Reg._he_shellPid    = _twr_panePid row
  550         , Reg._he_harnessPid  = Nothing
  551         , Reg._he_origin      = Reg.OriginDiscovered
  552         , Reg._he_liveness    = live
  553         , Reg._he_extModified = False
  554         , Reg._he_stale       = False
  555         , Reg._he_sessionId   = Nothing
  556         , Reg._he_label       = _twr_windowName row
  557         , Reg._he_orphanedTicks = 0
  558         , Reg._he_handle      = Nothing
  559         }
