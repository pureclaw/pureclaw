  108 -- | WAI application handling @\/api\/*@ routes.
  109 apiApp :: FrontendEnv -> Application
  110 apiApp env req respond = do
  111   let method = requestMethod req
  112       path   = pathInfo req
  113   case (method, path) of
  114     ("GET", ["api", "harnesses"])            -> handleHarnesses env respond
  115     ("GET", ["api", "sessions", "recent"])   -> handleRecentSessions env respond
  116     ("GET", ["api", "sessions", sid, "transcript"]) ->
  117       handleTranscript env sid respond
  118     ("POST", ["api", "sessions", sid, "send"]) ->
  119       handleSend env sid req respond
  120     _                                        -> respondNotFound respond

  122 handleHarnesses :: FrontendEnv -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  123 handleHarnesses env respond = do
  124   harnesses <- readIORef (_fe_harnesses env)
  125   infos <- traverse probeHarness (Map.toList harnesses)
  126   respond $ jsonResponse status200 infos
  127 
  128 -- | Probe a single harness to determine its activity.
  129 -- Checks process status first, then does a quick screen capture
  130 -- to distinguish thinking from idle.
  131 probeHarness :: (Text, HarnessHandle) -> IO HarnessInfo
  132 probeHarness (name, hh) = do
  133   st <- _hh_status hh
  134   activity <- case st of
  135     HarnessExited _ -> pure HarnessStopped
  136     HarnessRunning  -> probeActivity (_hh_session hh) name
  137   pure HarnessInfo
  138     { _hi_name     = name
  139     , _hi_activity = activity
  140     }
  141 
  142 -- | Capture the tmux window and check if the harness is idle or thinking.
  143 probeActivity :: Text -> Text -> IO HarnessActivity
  144 probeActivity tmuxSession windowName = do
  145   let target = tmuxSession <> ":" <> extractWindowIdx windowName
  146   result <- try @SomeException $ captureWindow target 50
  147   case result of
  148     Left _        -> pure HarnessIdle
  149     Right capture -> do
  150       let screenText = TE.decodeUtf8Lenient capture
  151       pure $ if isIdle screenText then HarnessIdle else HarnessThinking
  152 
  153 -- | Extract the window index suffix from a harness name like "claude-code-0".
  154 extractWindowIdx :: Text -> Text
  155 extractWindowIdx name =
  156   case T.splitOn "-" name of
  157     parts | length parts >= 2 -> last parts
  158     _                         -> "0"

  160 handleRecentSessions :: FrontendEnv -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  161 handleRecentSessions env respond = do
  162   metas <- listSessions (_fe_sessionsDir env) Nothing (_fe_recentLimit env)
  163   nonEmpty <- filterM (hasTranscriptEntries (_fe_sessionsDir env)) metas
  164   let infos = map toSessionInfo nonEmpty
  165   respond $ jsonResponse status200 infos

  167 -- | Check whether a session has at least one transcript entry.
  168 hasTranscriptEntries :: FilePath -> SessionMeta -> IO Bool
  169 hasTranscriptEntries baseDir meta = do
  170   let path = baseDir </> T.unpack (unSessionId (_sm_id meta)) </> "transcript.jsonl"
  171   exists <- doesFileExist path
  172   if not exists
  173     then pure False
  174     else do
  175       size <- getFileSize path
  176       pure (size > 0)
  177 
  178 toSessionInfo :: SessionMeta -> SessionInfo
  179 toSessionInfo m = SessionInfo
  180   { _si_id         = unSessionId (_sm_id m)
  181   , _si_agent      = fmap unAgentName (_sm_agent m)
  182   , _si_runtime    = runtimeToText (_sm_runtime m)
  183   , _si_model      = _sm_model m
  184   , _si_lastActive = _sm_lastActive m
  185   , _si_createdAt  = _sm_createdAt m
  186   }

  188 runtimeToText :: RuntimeType -> Text
  189 runtimeToText RTProvider      = "provider"
  190 runtimeToText (RTHarness name) = "harness:" <> name

  225 -- | Read transcript entries from a session's @transcript.jsonl@ file.
  226 handleTranscript :: FrontendEnv -> Text -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  227 handleTranscript env sid respond = do
  228   -- Reject path traversal
  229   if T.isInfixOf ".." sid || T.isInfixOf "/" sid
  230     then respond $ jsonResponse status400 (object ["error" .= ("Invalid session ID" :: Text)])
  231     else do
  232       let path = _fe_sessionsDir env </> T.unpack sid </> "transcript.jsonl"
  233       exists <- doesFileExist path
  234       if not exists
  235         then respond $ jsonResponse status404 (object ["error" .= ("Session not found" :: Text)])
  236         else do
  237           result <- try @SomeException $ readTranscriptFile path
  238           case result of
  239             Left _ ->
  240               respond $ jsonResponse status500 (object ["error" .= ("Failed to read transcript" :: Text)])
  241             Right entries ->
  242               respond $ jsonResponse status200 (map toTranscriptEntryInfo entries)
  243 
  244 -- | Read and parse a JSONL transcript file.
  245 readTranscriptFile :: FilePath -> IO [TranscriptEntry]
  246 readTranscriptFile path = do
  247   contents <- LBS.readFile path
  248   let linesBS = filter (not . LBS.null) (LBS.split 0x0A contents)
  249   pure [e | l <- linesBS, Just e <- [Aeson.decode l]]

  251 -- | Request body for sending a message.
  252 newtype SendRequest = SendRequest { _sr_message :: Text }
  253 
  254 instance FromJSON SendRequest where
  255   parseJSON = Aeson.withObject "SendRequest" $ \o ->
  256     SendRequest <$> o .: "message"

  258 -- | Send a user message to a session and get a completion.
  259 handleSend :: FrontendEnv -> Text -> Request -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  260 handleSend env sid req respond = do
  261   -- Validate session ID
  262   if T.isInfixOf ".." sid || T.isInfixOf "/" sid
  263     then respond $ jsonResponse status400 (object ["error" .= ("Invalid session ID" :: Text)])
  264     else do
  265       let sessionDir = _fe_sessionsDir env </> T.unpack sid
  266           transcriptPath = sessionDir </> "transcript.jsonl"
  267       exists <- doesFileExist transcriptPath
  268       if not exists
  269         then respond $ jsonResponse status404 (object ["error" .= ("Session not found" :: Text)])
  270         else do
  271           -- Parse request body
  272           body <- consumeBody req
  273           case Aeson.eitherDecode body of
  274             Left _ ->
  275               respond $ jsonResponse status400 (object ["error" .= ("Invalid JSON: expected {\"message\": \"...\"}" :: Text)])
  276             Right (SendRequest userText) -> do
  277               mProvider <- readIORef (_fe_provider env)
  278               mModel <- readIORef (_fe_model env)
  279               case (mProvider, mModel) of
  280                 (Nothing, _) ->
  281                   respond $ jsonResponse status503 (object ["error" .= ("No provider configured" :: Text)])
  282                 (_, Nothing) ->
  283                   respond $ jsonResponse status503 (object ["error" .= ("No model configured" :: Text)])
  284                 (Just provider, Just model) -> do
  285                   result <- try @SomeException $ doCompletion env provider model userText transcriptPath
  286                   case result of
  287                     Left e -> do
  288                       _lh_logError (_fe_logger env) $ "Send error: " <> T.pack (show e)
  289                       respond $ jsonResponse status500 (object ["error" .= ("Completion failed" :: Text)])
  290                     Right respText ->
  291                       respond $ jsonResponse status200 (object ["response" .= respText])
  292 
  293 -- | Run a completion: load context, send to provider, record to transcript.
  294 doCompletion :: FrontendEnv -> SomeProvider -> ModelId -> Text -> FilePath -> IO Text
  295 doCompletion env provider model userText transcriptPath = do
  296   -- Open a transcript handle for recording
  297   th <- mkFileTranscriptHandle (_fe_logger env) transcriptPath
  298   -- Load recent messages for context
  299   history <- loadRecentMessages th 50 100000
  300   -- Build context with system prompt and history
  301   let ctx0 = foldl (flip addMessage) (emptyContext (_fe_systemPrompt env)) history
  302       userMsg = textMessage User userText
  303       ctx = addMessage userMsg ctx0
  304       -- Wrap provider with transcript logging
  305       provider' = mkTranscriptProvider th (unModelId model) provider
  306       req = CompletionRequest
  307         { _cr_model        = model
  308         , _cr_messages     = contextMessages ctx
  309         , _cr_systemPrompt = contextSystemPrompt ctx
  310         , _cr_maxTokens    = Just 4096
  311         , _cr_tools        = []
  312         , _cr_toolChoice   = Nothing
  313         }
  314   resp <- complete provider' req
  315   _th_flush th
  316   _th_close th
  317   pure (responseText resp)
  318 
  319 -- | Consume the full request body.
  320 consumeBody :: Request -> IO LBS.ByteString
  321 consumeBody req = LBS.fromChunks <$> go
  322   where
  323     go = do
  324       chunk <- getRequestBodyChunk req
  325       if BS.null chunk then pure [] else (chunk :) <$> go

  332 -- | Build a JSON response.

  334 jsonResponse st body =
