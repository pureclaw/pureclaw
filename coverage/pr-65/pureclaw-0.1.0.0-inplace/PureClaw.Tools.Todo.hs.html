   17 -- | A single todo item.
   18 data TodoItem = TodoItem
   19   { _ti_id      :: Text
   20   , _ti_content :: Text
   21   , _ti_status  :: Text  -- pending | in_progress | completed | cancelled
   22   }
   23   deriving stock (Show, Eq)
   24 
   25 instance ToJSON TodoItem where
   26   toJSON ti = object
   27     [ "id"      .= _ti_id ti
   28     , "content" .= _ti_content ti
   29     , "status"  .= _ti_status ti
   30     ]
   31 
   32 instance FromJSON TodoItem where
   33   parseJSON = withObject "TodoItem" $ \o ->
   34     TodoItem <$> o .: "id" <*> o .: "content" <*> o .: "status"
   35 
   36 -- | Valid todo statuses.
   37 validStatuses :: [Text]
   38 validStatuses = ["pending", "in_progress", "completed", "cancelled"]
   39 
   40 -- | Create an in-memory todo tool for lightweight task tracking.
   41 -- State is per-session (not persisted across restarts).
   42 -- Omit the @todos@ field to read the current list; provide it to write.
   43 todoTool :: IO (ToolDefinition, ToolHandler)
   44 todoTool = do
   45   stateRef <- newIORef (Map.empty :: Map Text TodoItem)
   46   let def = ToolDefinition
   47         { _td_name        = "todo"
   48         , _td_description = T.unlines
   49             [ "In-memory task list for decomposing complex work."
   50             , "Omit 'todos' to read the current list."
   51             , "Provide 'todos' array to write/update items."
   52             , "Set merge=true to merge with existing items (default: replace all)."
   53             , "Statuses: pending, in_progress, completed, cancelled."
   54             ]
   55         , _td_inputSchema = object
   56             [ "type" .= ("object" :: Text)
   57             , "properties" .= object
   58                 [ "todos" .= object
   59                     [ "type" .= ("array" :: Text)
   60                     , "items" .= object
   61                         [ "type" .= ("object" :: Text)
   62                         , "properties" .= object
   63                             [ "id" .= object ["type" .= ("string" :: Text)]
   64                             , "content" .= object ["type" .= ("string" :: Text)]
   65                             , "status" .= object
   66                                 [ "type" .= ("string" :: Text)
   67                                 , "enum" .= validStatuses
   68                                 ]
   69                             ]
   70                         , "required" .= (["id", "content", "status"] :: [Text])
   71                         ]
   72                     , "description" .= ("Todo items to set. Omit to read current list." :: Text)
   73                     ]
   74                 , "merge" .= object
   75                     [ "type" .= ("boolean" :: Text)
   76                     , "description" .= ("Merge with existing items instead of replacing (default: false)" :: Text)
   77                     ]
   78                 ]
   79             ]
   80         }
   81 
   82       handler = ToolHandler $ \input ->
   83         case parseEither parseTodoInput input of
   84           Left err -> pure (T.pack err, True)
   85           Right (Nothing, _) -> do
   86             -- Read mode
   87             items <- readIORef stateRef
   88             pure (formatTodos (Map.elems items), False)
   89           Right (Just todos, merge) -> do
   90             -- Validate statuses
   91             let invalid = filter (\ti -> _ti_status ti `notElem` validStatuses) todos
   92             if not (null invalid)
   93               then pure ("Invalid status in: " <> T.intercalate ", " (map _ti_id invalid)
   94                         <> ". Valid: " <> T.intercalate ", " validStatuses, True)
   95               else do
   96                 let newMap = Map.fromList [(_ti_id ti, ti) | ti <- todos]
   97                 if merge
   98                   then atomicModifyIORef' stateRef $ \old ->
   99                     (Map.union newMap old, ())
  100                   else writeIORef stateRef newMap
  101                 items <- readIORef stateRef
  102                 pure (formatTodos (Map.elems items), False)
  103 
  104   pure (def, handler)
  105 
  106 formatTodos :: [TodoItem] -> Text
  107 formatTodos [] = "No todos."
  108 formatTodos items =
  109   let grouped = Map.fromListWith (++) [(statusOrder (_ti_status ti), [ti]) | ti <- items]
  110       sections = map formatSection (Map.toAscList grouped)
  111   in T.intercalate "\n\n" sections
  112 
  113 statusOrder :: Text -> Int
  114 statusOrder "in_progress" = 0
  115 statusOrder "pending"     = 1
  116 statusOrder "completed"   = 2
  117 statusOrder "cancelled"   = 3
  118 statusOrder _             = 4
  119 
  120 formatSection :: (Int, [TodoItem]) -> Text
  121 formatSection (_, []) = ""
  122 formatSection (_, items@(first:_)) =
  123   let header = T.toUpper (_ti_status first)
  124       lines' = map formatItem items
  125   in header <> ":\n" <> T.intercalate "\n" lines'
  126 
  127 formatItem :: TodoItem -> Text
  128 formatItem ti = "  [" <> _ti_id ti <> "] " <> _ti_content ti
  129 
  130 parseTodoInput :: Value -> Parser (Maybe [TodoItem], Bool)
  131 parseTodoInput = withObject "TodoInput" $ \o ->
  132   (,) <$> o .:? "todos" <*> o .:? "merge" .!= False
