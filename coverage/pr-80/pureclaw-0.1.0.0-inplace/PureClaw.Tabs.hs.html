    1 -- |
    2 -- Module      : PureClaw.Tabs
    3 -- Description : Thin IORef handle over the pure 'TabList' registry.
    4 --
    5 -- 'TabRegistry' is the single mutable cell holding the ordered 'TabList'
    6 -- (invariants I1\/I2 — see "PureClaw.Tabs.Types"). All mutation is delegated
    7 -- to the pure operations in that module; this layer only owns the 'IORef'
    8 -- and re-exports the value types for convenience. There is no global state —
    9 -- the registry is constructed explicitly and passed by the caller (Handle
   10 -- pattern).
   11 --
   12 -- See the Tabs-as-View refactor (GitHub #79) for the design context.
   13 module PureClaw.Tabs
   14   ( -- * Registry handle
   15     TabRegistry (..)
   16   , newTabRegistry
   17   , readTabs
   18   , overwriteTabs
   19   , registryAppend
   20   , registryRemove
   21   , registryLookupSlot
   22   , registryLookupRef
   23   , registrySetStatus
   24     -- * Re-exports
   25   , module PureClaw.Tabs.Types
   26   ) where
   27 
   28 import Data.IORef (IORef, atomicModifyIORef', atomicWriteIORef, newIORef, readIORef)
   29 import Data.Text (Text)
   30 
   31 import PureClaw.Handles.Tab (TabIndex)
   32 import PureClaw.Tabs.Types
   33 
   34 -- | The mutable tab registry: a single 'IORef' over a pure 'TabList'.
   35 newtype TabRegistry = TabRegistry (IORef TabList)
   36 
   37 -- | Create an empty registry.
   38 newTabRegistry :: IO TabRegistry
   39 newTabRegistry = TabRegistry <$> newIORef emptyTabs
   40 
   41 -- | Read the current 'TabList' snapshot.
   42 readTabs :: TabRegistry -> IO TabList
   43 readTabs (TabRegistry ref) = readIORef ref
   44 
   45 -- | Replace the entire registry contents with @tl@, atomically. Used at boot
   46 -- (WU8) to SEED the live shared registry from the reconciled 'TabList' that
   47 -- 'PureClaw.Tabs.Persist.loadTabs' returns, BEFORE the frontend server or the
   48 -- tabbed loop observe it. The supplied 'TabList' already satisfies the slot
   49 -- invariants (it comes from 'loadTabs'\'s @rebuildTabs@), so this is a direct
   50 -- swap — no per-tab re-append.
   51 overwriteTabs :: TabRegistry -> TabList -> IO ()
   52 overwriteTabs (TabRegistry ref) = atomicWriteIORef ref
   53 
   54 -- | Append a tab binding @ref@ with label @name@, atomically. Returns the new
   55 -- slot, or the pure 'TabsError' (dedup\/cap) without mutating on rejection.
   56 registryAppend :: TabRegistry -> TabRef -> Text -> IO (Either TabsError TabIndex)
   57 registryAppend (TabRegistry ref) tabRef name =
   58   atomicModifyIORef' ref $ \tl ->
   59     case appendTab tabRef name tl of
   60       Left err          -> (tl, Left err)
   61       Right (slot, tl') -> (tl', Right slot)
   62 
   63 -- | Remove the tab at @slot@ and compact, atomically (no-op if absent).
   64 registryRemove :: TabRegistry -> TabIndex -> IO ()
   65 registryRemove (TabRegistry ref) slot =
   66   atomicModifyIORef' ref $ \tl -> (removeSlot slot tl, ())
   67 
   68 -- | Look up the tab currently at @slot@.
   69 registryLookupSlot :: TabRegistry -> TabIndex -> IO (Maybe Tab)
   70 registryLookupSlot (TabRegistry ref) slot =
   71   lookupSlot slot <$> readIORef ref
   72 
   73 -- | Look up the slot currently occupied by @ref@.
   74 registryLookupRef :: TabRegistry -> TabRef -> IO (Maybe TabIndex)
   75 registryLookupRef (TabRegistry ref) tabRef =
   76   lookupRef tabRef <$> readIORef ref
   77 
   78 -- | Set the status of the tab bound to @ref@, atomically (no-op if absent).
   79 registrySetStatus :: TabRegistry -> TabRef -> TabStatus -> IO ()
   80 registrySetStatus (TabRegistry ref) tabRef status =
   81   atomicModifyIORef' ref $ \tl -> (setStatus tabRef status tl, ())
