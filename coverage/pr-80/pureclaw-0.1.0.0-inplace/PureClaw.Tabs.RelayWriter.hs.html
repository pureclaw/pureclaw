    1 -- |
    2 -- Module      : PureClaw.Tabs.RelayWriter
    3 -- Description : Output-side relay writer + sink registry (Tabs-as-View, #79).
    4 --
    5 -- This is the __output side__ of the Tabs-as-View model (design spike §4\/§5).
    6 -- It replaces the old global-focus @ChannelOut@ writer: instead of one focus
    7 -- gate deciding whether a whole message reaches /the/ channel, each tab-output
    8 -- 'ChannelEvent' is fanned out __per conversation__ via
    9 -- 'PureClaw.Tabs.Relay.relayEvent', and the resulting per-conversation events
   10 -- are delivered to each conversation's own 'ChannelHandle'.
   11 --
   12 -- Two pieces:
   13 --
   14 --   * __Sink registry__ ('SinkRegistry') — a @'ConversationKey' ->
   15 --     'ChannelHandle'@ map, registered by each channel as a conversation
   16 --     becomes active. The relay sends a conversation's output to its
   17 --     registered sink; a /missing/ sink is a safe drop (§5).
   18 --   * __Relay writer__ ('RelayWriter' + 'processOutput') — drives
   19 --     'relayEvent' for one @('TabRef', 'ChannelEvent')@ at a time. It reads a
   20 --     snapshot of the 'CursorState' and 'TabList' (the dispatcher owns those
   21 --     IORefs), carries the burst-dedup pinged-set across events in its own
   22 --     IORef, and routes each per-conversation event the relay produces into the
   23 --     sink registry via 'emitToChannel'.
   24 --
   25 -- The 'emitToChannel' mapping is a direct port of @ChannelOut.emitEvent@
   26 -- (which is deleted in 8c): @StreamStart@ has no on-channel representation;
   27 -- @ChunkOf@\/@StreamEnd@ become '_ch_sendChunk' calls; @FullMsg@\/@BannerLine@
   28 -- become '_ch_send' calls.
   29 --
   30 -- This module is __additive__ in 8b: it is wired into the live loop in 8c.
   31 -- Every seam is injected, so it is fully unit-testable with fake
   32 -- 'ChannelHandle's.

   72 -- ---------------------------------------------------------------------------
   73 -- Sink registry
   74 -- ---------------------------------------------------------------------------

   76 -- | Where each conversation's output goes: a @'ConversationKey' ->
   77 -- 'ChannelHandle'@ map, registered by each channel as a conversation becomes
   78 -- active. A conversation with no entry is a safe drop in 'processOutput'.
   79 newtype SinkRegistry = SinkRegistry (IORef (Map ConversationKey ChannelHandle))
   80 
   81 -- | A fresh, empty sink registry.
   82 newSinkRegistry :: IO SinkRegistry
   83 newSinkRegistry = SinkRegistry <$> newIORef Map.empty
   84 
   85 -- | Register (or replace) the 'ChannelHandle' a conversation's output is sent
   86 -- to.
   87 registerSink :: SinkRegistry -> ConversationKey -> ChannelHandle -> IO ()
   88 registerSink (SinkRegistry ref) k ch =
   89   atomicModifyIORef' ref (\m -> (Map.insert k ch m, ()))
   90 
   91 -- | Drop a conversation's sink (a no-op if it had none).
   92 unregisterSink :: SinkRegistry -> ConversationKey -> IO ()
   93 unregisterSink (SinkRegistry ref) k =
   94   atomicModifyIORef' ref (\m -> (Map.delete k m, ()))
   95 
   96 -- | Look up a conversation's registered sink.
   97 lookupSink :: SinkRegistry -> ConversationKey -> IO (Maybe ChannelHandle)
   98 lookupSink (SinkRegistry ref) k = Map.lookup k <$> readIORef ref
   99

  101 -- Relay writer

  104 -- | Everything 'processOutput' needs to fan one tab-output event out.
  105 --
  106 -- The cursor and tab snapshots are read fresh on every 'processOutput' call —
  107 -- the dispatcher owns those IORefs; the writer only reads them — so focus and
  108 -- slot changes between events take effect immediately.
  109 data RelayWriterDeps = RelayWriterDeps
  110   { _rw_sinks   :: !SinkRegistry
  111     -- ^ The conversation -> 'ChannelHandle' registry.
  112   , _rw_cursors :: !(IO CursorState)
  113     -- ^ Snapshot the current 'CursorState' (dispatcher-owned).
  114   , _rw_tabs    :: !(IO TabList)
  115     -- ^ Snapshot the current 'TabList' (tab registry).
  116   , _rw_default :: !RelayMode
  117     -- ^ The global default 'RelayMode' for conversations with no override.
  118   }
  119 
  120 -- | The mutable state carried across 'processOutput' calls.
  121 --
  122 -- '_rw_pinged' is the burst-dedup set: the @('ConversationKey', 'BurstKey')@
  123 -- pairs already activity-pinged. The relay's 'ActivityDigest' decision keys on
  124 -- this so a multi-chunk stream pings once, and refocus clears a conversation's
  125 -- entries (so a later background burst pings again).
  126 --
  127 -- '_rw_buffer' is the per-stream accumulation buffer for __non-streaming__
  128 -- channels (Signal\/Telegram, @'_ch_streaming' = False@). Those channels
  129 -- discard '_ch_sendChunk', so a streamed provider reply would be lost entirely.
  130 -- Instead the writer accumulates each stream's 'ChunkOf' text keyed by
  131 -- @('ConversationKey', 'StreamId')@ and flushes it as ONE '_ch_send' on
  132 -- 'StreamEnd' — restoring the legacy full-send. Streaming channels never touch
  133 -- this buffer.
  134 data RelayWriter = RelayWriter
  135   { _rw_pinged :: !(IORef (Set (ConversationKey, BurstKey)))
  136   , _rw_buffer :: !(IORef (Map (ConversationKey, StreamId) Text))
  137   }
  138 
  139 -- | A fresh relay writer with an empty pinged-set and an empty stream buffer.
  140 newRelayWriter :: IO RelayWriter
  141 newRelayWriter =
  142   RelayWriter <$> newIORef Set.empty <*> newIORef Map.empty

  144 -- | Process ONE tab-output event: fan it out per conversation via
  145 -- 'relayEvent', delivering each resulting per-conversation event to that
  146 -- conversation's registered sink (via 'emitToChannel'). A conversation with no
  147 -- registered sink is a safe drop.
  148 --
  149 -- Reads the cursor + tab snapshots and the writer's carried pinged-set, runs
  150 -- the relay decision, then writes the updated pinged-set back so burst dedup
  151 -- persists across calls.
  152 processOutput :: RelayWriterDeps -> RelayWriter -> TabRef -> ChannelEvent -> IO ()
  153 processOutput deps rw src event = do
  154   cursors <- _rw_cursors deps
  155   tabs    <- _rw_tabs deps
  156   pinged  <- readIORef (_rw_pinged rw)
  157   let sink = RelayDeps $ \convKey ev ->
  158         lookupSink (_rw_sinks deps) convKey
  159           >>= maybe (pure ()) (\ch -> emitToConversation (_rw_buffer rw) convKey ch ev)
  160   pinged' <- relayEvent sink cursors (_rw_default deps) tabs pinged src event
  161   writeIORef (_rw_pinged rw) pinged'
  162 
  163 -- | Deliver one per-conversation 'ChannelEvent' to its sink, honoring the
  164 -- channel's '_ch_streaming' capability.
  165 --
  166 --   * __Streaming channel__ (@'_ch_streaming' = True@, e.g. CLI): unchanged —
  167 --     defer to 'emitToChannel', which streams 'ChunkOf'\/'StreamEnd' via
  168 --     '_ch_sendChunk'.
  169 --   * __Non-streaming channel__ (@'_ch_streaming' = False@, e.g.
  170 --     Signal\/Telegram): '_ch_sendChunk' is a no-op on the real channel, so
  171 --     buffer each stream's 'ChunkOf' text keyed by @('ConversationKey',
  172 --     'StreamId')@ and flush it as ONE '_ch_send' on 'StreamEnd' (dropping an
  173 --     empty stream). 'StreamStart' just primes an empty buffer entry;
  174 --     'FullMsg'\/'BannerLine' send a full message directly.
  175 emitToConversation
  176   :: IORef (Map (ConversationKey, StreamId) Text)
  177   -> ConversationKey
  178   -> ChannelHandle
  179   -> ChannelEvent
  180   -> IO ()
  181 emitToConversation bufRef convKey ch ev
  182   | _ch_streaming ch = emitToChannel ch ev
  183   | otherwise = case ev of
  184       StreamStart sid _ ->
  185         atomicModifyIORef' bufRef
  186           (\m -> (Map.insertWith (\_ old -> old) (convKey, sid) "" m, ()))
  187       ChunkOf sid t ->
  188         -- 'flip (<>)' so the existing buffered text precedes the new chunk:
  189         -- Map.insertWith f applies @f new old@, so @flip (<>) new old = old <> new@.
  190         atomicModifyIORef' bufRef
  191           (\m -> (Map.insertWith (flip (<>)) (convKey, sid) t m, ()))
  192       StreamEnd sid -> do
  193         buffered <- atomicModifyIORef' bufRef
  194           (\m -> (Map.delete (convKey, sid) m, Map.lookup (convKey, sid) m))
  195         case buffered of
  196           Just full | not (T.null full) -> _ch_send ch (OutgoingMessage full)
  197           _                                      -> pure ()
  198       FullMsg _ t  -> _ch_send ch (OutgoingMessage t)
  199       BannerLine t -> _ch_send ch (OutgoingMessage t)
  200 
  201 -- ---------------------------------------------------------------------------
  202 -- Event -> channel mapping
  203 -- ---------------------------------------------------------------------------
  204 
  205 -- | Translate a 'ChannelEvent' into the underlying 'ChannelHandle' calls. A
  206 -- direct port of @ChannelOut.emitEvent@ (deleted in 8c): 'StreamStart' is
  207 -- writer-state metadata with no on-channel representation; 'ChunkOf' and
  208 -- 'StreamEnd' stream via '_ch_sendChunk'; 'FullMsg' and 'BannerLine' send a
  209 -- full message via '_ch_send'.
  210 emitToChannel :: ChannelHandle -> ChannelEvent -> IO ()
  211 emitToChannel ch ev = case ev of
  212   StreamStart{} -> pure ()
  213   ChunkOf _ t   -> _ch_sendChunk ch (ChunkText t)
  214   StreamEnd _   -> _ch_sendChunk ch ChunkDone
  215   FullMsg _ t   -> _ch_send ch (OutgoingMessage t)
  216   BannerLine t  -> _ch_send ch (OutgoingMessage t)
