    1 -- |
    2 -- Module      : PureClaw.Tabs.Relay
    3 -- Description : Streaming-aware per-conversation output relay (Tabs-as-View, #79).
    4 --
    5 -- When a tab (named by its 'TabRef') emits a 'ChannelEvent', 'relayEvent' fans
    6 -- that event out to every conversation according to /that conversation's/
    7 -- effective 'RelayMode' (its per-conversation override, else the global
    8 -- default). This replaces the single-focus @ChannelOut@ gate (@shouldEmit@): a
    9 -- tab can be foreground for conversation A and background for B at the same
   10 -- time, so the decision is taken per conversation rather than once for the
   11 -- whole output (design §9.3).
   12 --
   13 -- Unlike the earlier whole-@Text@ relay, this engine is __streaming-aware__:
   14 -- it sinks a 'ChannelEvent', so a focused conversation receives every
   15 -- @StreamStart@\/@ChunkOf@\/@StreamEnd@ verbatim (full provider streaming),
   16 -- while a background 'ActivityDigest' conversation gets __at most one__
   17 -- breadcrumb ping per /burst/ — one logical message. The burst is identified
   18 -- by a 'BurstKey': all framing events of one stream share @'BurstStream' sid@,
   19 -- and a one-shot @FullMsg@ is its own @'BurstFull'@ burst. The pinged-set is
   20 -- keyed on @('ConversationKey', 'BurstKey')@ so a 100-chunk stream pings once,
   21 -- not 100 times (§4).
   22 --
   23 -- The engine is a __single writer__: the dispatcher invokes it from one
   24 -- thread, and it iterates conversations in a deterministic (sorted-key) order,
   25 -- calling the injected sink in turn. There is no concurrency inside the
   26 -- engine — ordering across the per-conversation sinks is preserved exactly as
   27 -- @ChannelOut@ guaranteed.
   28 --
   29 -- == Per-conversation decision (effective mode @m@ for key @k@)
   30 --
   31 -- Let @focused@ mean the conversation's cursor names the source tab.
   32 --
   33 --   * __Focused__ (any mode) — forward the event verbatim, and clear /every/
   34 --     @(k, _)@ entry from the pinged-set: the conversation is looking at the
   35 --     tab, so a future background burst should ping it again.
   36 --   * __FocusedOnly__ background — nothing.
   37 --   * __Firehose__ background — forward the event verbatim.
   38 --   * __ActivityDigest__ background — emit at most one ping per burst. A
   39 --     'BannerLine' source event is skipped entirely (banners are
   40 --     dispatcher-class, never relayed here): no ping, set untouched. For a
   41 --     stream\/full event, if @(k, burstKey)@ is not already pinged AND the
   42 --     source ref resolves in the 'TabList', deliver a 'BannerLine' ping
   43 --     (@\"\<name\> (\/N) has new output\"@, where @\<name\>@ is the source
   44 --     tab's friendly name and @N@ its /current/ display slot) and remember
   45 --     @(k, burstKey)@. A 'Dead' tombstone source still pings (§8) — only a
   46 --     source ref /absent/ from the list yields no ping (no name\/slot).
   47 --
   48 -- The set of conversations considered is every key present in the
   49 -- 'CursorState' — the union of the cursor-map keys and the relay-override-map
   50 -- keys — so a conversation that has only set a 'RelayMode' (no cursor yet) is
   51 -- still routed.
   52 module PureClaw.Tabs.Relay
   53   ( BurstKey (..)
   54   , RelayDeps (..)
   55   , relayEvent
   56   ) where

   79 -- | The /burst/ a 'ChannelEvent' belongs to — one logical message. All framing
   80 -- events of a single provider stream (@StreamStart@\/@ChunkOf@\/@StreamEnd@)
   81 -- share the same @'BurstStream' sid@; a one-shot @FullMsg@ is its own
   82 -- @'BurstFull'@ burst. 'ActivityDigest' de-dups its background ping on
   83 -- @('ConversationKey', 'BurstKey')@, so a multi-chunk stream pings once.
   84 data BurstKey
   85   = BurstStream !StreamId
   86   | BurstFull
   87   deriving stock (Eq, Ord, Show)
   88 
   89 -- | The relay engine's only side-effect seam: a conversation-addressed output
   90 -- sink. @'_rl_sink' k e@ delivers event @e@ to conversation @k@. Injected so
   91 -- the engine can be unit-tested against a recording sink.
   92 newtype RelayDeps = RelayDeps
   93   { _rl_sink :: ConversationKey -> ChannelEvent -> IO ()
   94   }
   95 
   96 -- | The burst a 'ChannelEvent' belongs to, or 'Nothing' for a 'BannerLine'
   97 -- (banners are dispatcher-class and never relayed to background conversations).
   98 burstKeyOf :: ChannelEvent -> Maybe BurstKey
   99 burstKeyOf = \case
  100   StreamStart sid _ -> Just (BurstStream sid)
  101   ChunkOf sid _     -> Just (BurstStream sid)
  102   StreamEnd sid     -> Just (BurstStream sid)
  103   FullMsg _ _       -> Just BurstFull
  104   BannerLine _      -> Nothing
  105 
  106 -- | Fan one tab's 'ChannelEvent' out to every conversation per its effective
  107 -- 'RelayMode'. See the module header for the full per-conversation decision.
  108 --
  109 -- @globalDefault@ is the fallback 'RelayMode' for conversations with no
  110 -- override. @pinged@ is the set of @('ConversationKey', 'BurstKey')@ pairs
  111 -- already activity-pinged (burst de-dup for 'ActivityDigest'); the updated set
  112 -- is returned.
  113 relayEvent
  114   :: RelayDeps
  115   -> CursorState
  116   -> RelayMode
  117   -> TabList
  118   -> Set (ConversationKey, BurstKey)
  119   -> TabRef
  120   -> ChannelEvent
  121   -> IO (Set (ConversationKey, BurstKey))
  122 relayEvent deps cs globalDefault tl pinged src event =
  123     go pinged conversationKeys
  124   where
  125     -- Every conversation the dispatcher knows about: keys with a cursor or a
  126     -- relay override. Sorted so the single-writer emission order is testable.
  127     conversationKeys :: [ConversationKey]
  128     conversationKeys =
  129       Set.toAscList
  130         (Set.fromList (Map.keys (_cs_cursors cs))
  131            `Set.union` Set.fromList (Map.keys (_cs_relay cs)))
  132 
  133     -- The source tab's current display slot (0-based), if it is in the list.
  134     srcSlot :: Maybe Int
  135     srcSlot = unTabIndex <$> lookupRef src tl
  136 
  137     -- The source tab's friendly name, if it is in the list.
  138     srcName :: Maybe Text
  139     srcName = do
  140       i <- lookupRef src tl
  141       _tab_name <$> lookupSlot i tl
  142 
  143     -- A name-first activity ping for the source tab at its current slot, as a
  144     -- 'BannerLine'. 'Nothing' when the source ref is absent (no name/slot).
  145     activityPing :: Maybe ChannelEvent
  146     activityPing = do
  147       name <- srcName
  148       slot <- srcSlot
  149       pure (BannerLine (name <> " (/" <> T.pack (show slot) <> ") has new output"))
  150 
  151     -- How many tabs exist right now. A single-tab CLI stays unlabelled; a
  152     -- multi-tab session labels focused bursts so the speaker is identifiable.
  153     tabCount :: Int
  154     tabCount = length (toList tl)
  155 
  156     -- A concise speaker label for the source tab at its current slot, as a
  157     -- 'BannerLine' (tab identity only — NO "has new output" activity suffix).
  158     -- 'Nothing' when the source ref is absent (no name/slot).
  159     focusedLabel :: Maybe ChannelEvent
  160     focusedLabel = do
  161       name <- srcName
  162       slot <- srcSlot
  163       pure (BannerLine (name <> " (/" <> T.pack (show slot) <> ")"))
  164 
  165     deliver :: ConversationKey -> ChannelEvent -> IO ()
  166     deliver = _rl_sink deps
  167 
  168     -- Forward the event to a focused conversation, prefixing a one-shot speaker
  169     -- label at the START of a burst when more than one tab exists. The label is
  170     -- emitted before a 'StreamStart' (once per stream) and before a 'FullMsg'
  171     -- (once per whole-message capture); 'ChunkOf'\/'StreamEnd'\/'BannerLine'
  172     -- carry no label, so a multi-chunk reply is labelled exactly once. A
  173     -- single-tab session is never labelled (clean CLI, the gb7 decision).
  174     deliverFocused :: ConversationKey -> IO ()
  175     deliverFocused k = do
  176       mapM_ (deliver k) (focusedLabelFor event)
  177       deliver k event
  178 
  179     -- The speaker label to prefix to a focused delivery of @ev@, if any: only
  180     -- when more than one tab exists, the source tab resolves to a name\/slot,
  181     -- AND @ev@ is a burst-start ('StreamStart' or 'FullMsg'). 'Nothing' for a
  182     -- single-tab session, an unresolved source, or a non-burst-start event.
  183     focusedLabelFor :: ChannelEvent -> Maybe ChannelEvent
  184     focusedLabelFor ev
  185       | tabCount > 1 && labelsBurstStart ev = focusedLabel
  186       | otherwise                           = Nothing
  187 
  188     -- Whether @event@ begins a burst that should carry a focused speaker label:
  189     -- a 'StreamStart' (start of a streamed reply) or a 'FullMsg' (a whole
  190     -- message capture). Mid-burst and framing-tail events do not.
  191     labelsBurstStart :: ChannelEvent -> Bool
  192     labelsBurstStart = \case
  193       StreamStart {} -> True
  194       FullMsg {}     -> True
  195       ChunkOf {}     -> False
  196       StreamEnd {}   -> False
  197       BannerLine {}  -> False
  198 
  199     go
  200       :: Set (ConversationKey, BurstKey)
  201       -> [ConversationKey]
  202       -> IO (Set (ConversationKey, BurstKey))
  203     go acc []       = pure acc
  204     go acc (k : ks) = do
  205       acc' <-
  206         if Map.lookup k (_cs_cursors cs) == Just src
  207           -- Focused on the source tab: forward the event verbatim in every
  208           -- mode, and clear ALL of this conversation's burst entries (it is
  209           -- now looking at the tab, so a future background burst should ping).
  210           then deliverFocused k >> pure (clearKey k acc)
  211           -- Background: behaviour depends on the conversation's effective mode.
  212           else case relayModeFor k globalDefault cs of
  213             FocusedOnly    -> pure acc
  214             Firehose       -> deliver k event >> pure acc
  215             ActivityDigest -> backgroundDigest acc k
  216       go acc' ks
  217 
  218     -- Drop every @(k, _)@ burst entry for conversation @k@.
  219     clearKey
  220       :: ConversationKey
  221       -> Set (ConversationKey, BurstKey)
  222       -> Set (ConversationKey, BurstKey)
  223     clearKey k = Set.filter ((/= k) . fst)
  224 
  225     -- ActivityDigest for a background conversation: one name-first ping per
  226     -- burst. A 'BannerLine' source is skipped (no burst, set untouched).
  227     -- Otherwise suppressed when @(k, burstKey)@ was already pinged, or when the
  228     -- source ref is absent from the list (no name/slot to render).
  229     backgroundDigest
  230       :: Set (ConversationKey, BurstKey)
  231       -> ConversationKey
  232       -> IO (Set (ConversationKey, BurstKey))
  233     backgroundDigest acc k = case burstKeyOf event of
  234       Nothing -> pure acc
  235       Just bk
  236         | (k, bk) `Set.member` acc -> pure acc
  237         | otherwise -> case activityPing of
  238             Just p  -> deliver k p >> pure (Set.insert (k, bk) acc)
  239             Nothing -> pure acc
