    1 -- |
    2 -- Module      : PureClaw.Tabs.Types
    3 -- Description : First-class tab registry — core value types & pure operations.
    4 --
    5 -- A /tab/ is a pure binding over ground truth: it names a 'TabRef' (a live
    6 -- 'SessionId' or 'HarnessId'), the slot it currently occupies, a friendly
    7 -- label, and a liveness 'TabStatus'. The ordered collection of tabs is a
    8 -- 'TabList' that maintains two invariants:
    9 --
   10 --   * __I1 (contiguity)__ — the slots of a 'TabList' are exactly
   11 --     @[0 .. n-1]@ at all times. 'removeSlot' compacts (tmux-window style):
   12 --     every tab above the removed slot shifts down by one.
   13 --   * __I2 (uniqueness)__ — no two tabs bind the same 'TabRef'. 'appendTab'
   14 --     of an already-bound ref is rejected with @'Left' ('AlreadyBound' i)@,
   15 --     reporting the ref's /current/ slot.
   16 --
   17 -- There is also a hard cap of 36 slots (@0..35@), matching the validated
   18 -- 'TabIndex' range. Appending past it returns @'Left' 'SlotsFull'@.
   19 --
   20 -- This module is pure (no IO). The thin IORef handle that mutates a 'TabList'
   21 -- lives in "PureClaw.Tabs". The 0–35 'TabIndex' arithmetic is reused from
   22 -- "PureClaw.Handles.Tab" ('TabIndex'\/'mkTabIndex'\/'unTabIndex') rather than
   23 -- redefined, so there is a single validated index type across the codebase.
   24 --
   25 -- See the Tabs-as-View refactor (GitHub #79) for the design context.
   26 module PureClaw.Tabs.Types

   68 -- TabRef / TabStatus / Tab

   71 -- | The ground-truth entity a tab binds to: either an LLM-backed
   72 -- 'SessionId' or an external 'HarnessId'. A tab is a /view/ over one of
   73 -- these; it owns no lifecycle of its own.
   74 data TabRef
   75   = BoundSession !SessionId
   76   | BoundHarness !HarnessId
   77   deriving stock (Eq, Ord, Show)
   78 
   79 -- | Liveness of a tab's bound ground truth.
   80 --
   81 -- * 'Live' — the bound session\/harness is believed alive; messages route.
   82 -- * 'Dead' — the bound harness has exited (a tombstone). The tab persists
   83 --   in the list (visible in @\/tabs@) until it is explicitly removed; routing
   84 --   to a 'Dead' tab is handled by the dispatcher (deferred warning + drop).
   85 data TabStatus
   86   = Live
   87   | Dead
   88   deriving stock (Eq, Show)
   89 
   90 -- | A single tab: a pure binding over ground truth.
   91 data Tab = Tab
   92   { _tab_slot   :: !TabIndex
   93     -- ^ The tab's position in the 'TabList'. Maintained contiguous (I1);
   94     --   never set directly by callers — derived by 'appendTab'\/'removeSlot'.
   95   , _tab_ref    :: !TabRef
   96     -- ^ The bound ground-truth entity (unique across the list, I2).
   97   , _tab_name   :: !Text
   98     -- ^ Friendly label. (Sanitisation is the caller's responsibility — this
   99     --   module stores the text verbatim.)
  100   , _tab_status :: !TabStatus
  101     -- ^ Liveness of the binding.
  102   }
  103   deriving stock (Eq, Show)
  104 
  105 -- ---------------------------------------------------------------------------
  106 -- TabList
  107 -- ---------------------------------------------------------------------------
  108 
  109 -- | An ordered list of tabs maintaining invariants I1 (contiguous slots
  110 -- @[0..n-1]@) and I2 (unique 'TabRef's). The constructor is not exported;
  111 -- all mutation goes through the smart operations below, which preserve the
  112 -- invariants by construction.
  113 newtype TabList = TabList [Tab]
  114   deriving stock (Eq, Show)
  115 
  116 -- | Why a 'TabList' mutation was rejected.
  117 data TabsError
  118   = SlotsFull
  119     -- ^ All 36 slots are occupied; the next append would be slot 36.
  120   | AlreadyBound !TabIndex
  121     -- ^ The 'TabRef' is already bound; the payload is its /current/ slot.
  122   deriving stock (Eq, Show)
  123 
  124 -- | The hard slot cap. Slots are @0 .. 'maxTabSlots' - 1@ (i.e. 36 slots,
  125 -- matching the validated 'TabIndex' range used elsewhere).
  126 maxTabSlots :: Int
  127 maxTabSlots = 36
  128 
  129 -- | The empty tab list.
  130 emptyTabs :: TabList
  131 emptyTabs = TabList []
  132 
  133 -- | The tabs in slot order (slot @0@ first). Total; never partial.
  134 toList :: TabList -> [Tab]
  135 toList (TabList ts) = ts
  136 
  137 -- | Append a new tab binding @ref@ with label @name@ at the next free slot.
  138 --
  139 -- * If @ref@ is already bound, returns @'Left' ('AlreadyBound' i)@ where @i@
  140 --   is its current slot (I2 — no duplicate refs).
  141 -- * If all 36 slots are taken, returns @'Left' 'SlotsFull'@.
  142 -- * Otherwise the new tab lands at slot @length@ (preserving I1) with status
  143 --   'Live', and the new slot index is returned alongside the updated list.
  144 appendTab :: TabRef -> Text -> TabList -> Either TabsError (TabIndex, TabList)
  145 appendTab ref name tl@(TabList ts) =
  146   case lookupRef ref tl of
  147     Just i  -> Left (AlreadyBound i)
  148     Nothing ->
  149       let n = length ts
  150       in if n >= maxTabSlots
  151            then Left SlotsFull
  152            else
  153              -- @0 <= n < 36@ here, so slot @n@ is valid. Append the new tab
  154              -- (with a provisional slot 'reindex' overwrites) and 'reindex' to
  155              -- stamp contiguous slots @0..n@ (I1). The new tab is last, so its
  156              -- slot is the last stamped index, read back via 'NE.last'.
  157              let placed = NE.fromList (reindex (ts ++ [Tab firstSlot ref name Live]))
  158              in Right (_tab_slot (NE.last placed), TabList (NE.toList placed))
  159 
  160 -- | The infinite stream of valid slots @0, 1, 2, …@ as a 'NonEmpty'. Built
  161 -- via the comprehension-filter trick so the impossible @'mkTabIndex' n ==
  162 -- Nothing@ (n negative) case introduces no uncoverable alternative; the
  163 -- 'NonEmpty' shape makes 'firstSlot' total (no partial 'head').
  164 slotStream :: NonEmpty TabIndex
  165 slotStream = NE.fromList [ s | n <- [0 :: Int ..], Just s <- [mkTabIndex n] ]
  166 
  167 -- | The first slot (@0@), used as the provisional slot for a freshly-appended
  168 -- tab before 'reindex' stamps the real one.
  169 firstSlot :: TabIndex
  170 firstSlot = NE.head slotStream
  171 
  172 -- | Remove the tab at @slot@ (if present) and compact: every tab above the
  173 -- removed slot shifts down by one, so the result still satisfies I1. A no-op
  174 -- when no tab occupies @slot@.
  175 removeSlot :: TabIndex -> TabList -> TabList
  176 removeSlot slot (TabList ts) =
  177   TabList (reindex (filter ((/= unTabIndex slot) . unTabIndex . _tab_slot) ts))
  178 
  179 -- | Re-stamp a list of tabs with contiguous slots @0..n-1@ in their current
  180 -- order. Used after a removal to restore I1. Total: 'slotStream' supplies a
  181 -- valid 'TabIndex' for every position, so 'zipWith' stamps every tab.
  182 reindex :: [Tab] -> [Tab]
  183 reindex = zipWith (\s t -> t { _tab_slot = s }) (NE.toList slotStream)
  184 
  185 -- | Look up the tab currently at @slot@.
  186 lookupSlot :: TabIndex -> TabList -> Maybe Tab
  187 lookupSlot slot (TabList ts) =
  188   case filter ((== unTabIndex slot) . unTabIndex . _tab_slot) ts of
  189     (t : _) -> Just t
  190     []      -> Nothing
  191 
  192 -- | Look up the slot currently occupied by @ref@ (I2 guarantees at most one).
  193 lookupRef :: TabRef -> TabList -> Maybe TabIndex
  194 lookupRef ref (TabList ts) =
  195   case filter ((== ref) . _tab_ref) ts of
  196     (t : _) -> Just (_tab_slot t)
  197     []      -> Nothing
  198 
  199 -- | Set the 'TabStatus' of the tab bound to @ref@. A no-op when @ref@ is not
  200 -- bound. Slots are unaffected.
  201 setStatus :: TabRef -> TabStatus -> TabList -> TabList
  202 setStatus ref status (TabList ts) =
  203   TabList (map upd ts)
  204   where
  205     upd t =
  206       if _tab_ref t == ref
  207         then t { _tab_status = status }
  208         else t
  209 
  210 -- | Rebind the tab at @slot@ to a new @ref@ and @name@, /in place/ — the
  211 -- slot, and the slots of every other tab, are unchanged (I1 preserved). The
  212 -- rebound tab's status is reset to 'Live'. This backs @\/new@'s "reset the
  213 -- active tab to a fresh session" semantics (design §6.1).
  214 --
  215 -- The new @ref@ is dedup-checked against I2:
  216 --
  217 --   * If @ref@ is already bound to a /different/ slot, the rebind is rejected
  218 --     with @'Left' ('AlreadyBound' i)@ (@i@ = that other slot) and the list is
  219 --     unchanged.
  220 --   * Rebinding a slot to the ref it /already/ holds is allowed — it is just a
  221 --     relabel (and a status reset) in place.
  222 --
  223 -- If no tab occupies @slot@, this is a no-op returning the list unchanged
  224 -- (@'Right'@) — callers that already verified an active tab never hit this,
  225 -- but it keeps the function total.
  226 rebindSlot :: TabIndex -> TabRef -> Text -> TabList -> Either TabsError TabList
  227 rebindSlot slot ref name tl@(TabList ts) =
  228   case lookupSlot slot tl of
  229     Nothing -> Right tl
  230     Just _  ->
  231       case lookupRef ref tl of
  232         Just other | other /= slot -> Left (AlreadyBound other)
  233         _ -> Right (TabList (map upd ts))
  234   where
  235     upd t =
  236       if unTabIndex (_tab_slot t) == unTabIndex slot
  237         then t { _tab_ref = ref, _tab_name = name, _tab_status = Live }
  238         else t

  240 -- ---------------------------------------------------------------------------
  241 -- Per-conversation cursors & relay (I3)
  242 -- ---------------------------------------------------------------------------

  244 -- | The identity of a conversation: its originating 'ChannelKind' paired with
  245 -- a server-derived 'ConversationId'. Two messages sharing a key belong to the
  246 -- same conversation and therefore share a tab cursor and a 'RelayMode'.
  247 type ConversationKey = (ChannelKind, ConversationId)
  248 
  249 -- | How a conversation's output relay fans tab output out to it. Persisted
  250 -- per 'ConversationKey'; the global default lives in config (see the
  251 -- Tabs-as-View design §9.3).
  252 --
  253 -- * 'FocusedOnly' (default) — only the conversation's focused tab is relayed.
  254 -- * 'ActivityDigest' — focused tab full; other live tabs get an activity ping.
  255 -- * 'Firehose' — full content from all live tabs.
  256 data RelayMode
  257   = FocusedOnly
  258   | ActivityDigest
  259   | Firehose
  260   deriving stock (Eq, Show)
  261 
  262 -- | The per-conversation routing state the dispatcher owns: which 'TabRef'
  263 -- each conversation is focused on ('_cs_cursors'), and any per-conversation
  264 -- 'RelayMode' override ('_cs_relay'). Both maps are keyed by 'ConversationKey'.
  265 --
  266 -- Cursors key by 'TabRef', not by slot — this is invariant __I3__: a cursor
  267 -- survives slot compaction ('removeSlot') because it names ground truth, not a
  268 -- position. 'resolveCursorSlot' resolves the ref to its /current/ slot at read
  269 -- time.
  270 data CursorState = CursorState
  271   { _cs_cursors :: !(Map ConversationKey TabRef)
  272     -- ^ The 'TabRef' each conversation is currently focused on.
  273   , _cs_relay   :: !(Map ConversationKey RelayMode)
  274     -- ^ Per-conversation 'RelayMode' overrides (absent ⇒ use the global default).
  275   }
  276   deriving stock (Eq, Show)
  277 
  278 -- | The empty cursor state: no cursors, no relay overrides.
  279 emptyCursors :: CursorState
  280 emptyCursors = CursorState Map.empty Map.empty
  281 
  282 -- | Focus a conversation on a 'TabRef', replacing any prior cursor for that key.
  283 -- The relay map is untouched.
  284 setCursor :: ConversationKey -> TabRef -> CursorState -> CursorState
  285 setCursor k ref cs = cs { _cs_cursors = Map.insert k ref (_cs_cursors cs) }
  286 
  287 -- | Drop a conversation's cursor (a no-op if it had none). The relay map is
  288 -- untouched.
  289 clearCursor :: ConversationKey -> CursorState -> CursorState
  290 clearCursor k cs = cs { _cs_cursors = Map.delete k (_cs_cursors cs) }
  291 
  292 -- | Resolve a conversation's cursor to the slot its bound 'TabRef' currently
  293 -- occupies (__I3__). Returns 'Nothing' when the key has no cursor or when the
  294 -- cursor's ref is no longer present in the 'TabList' (a dangling cursor).
  295 resolveCursorSlot :: ConversationKey -> CursorState -> TabList -> Maybe TabIndex
  296 resolveCursorSlot k cs tl =
  297   case Map.lookup k (_cs_cursors cs) of
  298     Nothing  -> Nothing
  299     Just ref -> lookupRef ref tl
  300 
  301 -- | All conversation keys whose cursor is focused on the given 'TabRef'.
  302 conversationsOn :: TabRef -> CursorState -> [ConversationKey]
  303 conversationsOn ref cs =
  304   [ k | (k, r) <- Map.toList (_cs_cursors cs), r == ref ]
  305 
  306 -- | Drop every cursor whose 'TabRef' is no longer present in the 'TabList'.
  307 -- Valid cursors and the entire relay map are preserved.
  308 pruneDangling :: TabList -> CursorState -> CursorState
  309 pruneDangling tl cs =
  310   cs { _cs_cursors = Map.filter present (_cs_cursors cs) }
  311   where
  312     present ref = case lookupRef ref tl of
  313       Just _  -> True
  314       Nothing -> False
  315 
  316 -- | The 'RelayMode' for a conversation: its per-conversation override if one is
  317 -- set, otherwise the supplied global default.
  318 relayModeFor :: ConversationKey -> RelayMode -> CursorState -> RelayMode
  319 relayModeFor k def cs = Map.findWithDefault def k (_cs_relay cs)
