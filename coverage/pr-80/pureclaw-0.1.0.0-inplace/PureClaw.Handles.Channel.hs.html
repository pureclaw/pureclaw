   21 -- | A message received from a channel user, tagged with the source that
   22 -- describes where it came from (channel + optional user id + extensible
   23 -- fields). The source is captured at the channel boundary via
   24 -- 'mkMessageSource'.
   25 data IncomingMessage = IncomingMessage
   26   { _im_source  :: MessageSource
   27   , _im_content :: Text
   28   }
   29   deriving stock (Show, Eq)
   30 
   31 -- | Derive the channel user id from a message's source, preserving the
   32 -- legacy '_im_userId' behavior. A source with no user id (@Nothing@)
   33 -- yields the @UserId ""@ sentinel — the same value the no-op channel
   34 -- produced before the 'MessageSource' migration — so dispatcher routing
   35 -- against the allow-list is unchanged.
   36 imUserId :: IncomingMessage -> UserId
   37 imUserId m = fromMaybe (UserId "") (_ms_userId (_im_source m))

   51 -- | Channel communication capability interface. Concrete implementations
   52 -- (CLI, Telegram, Signal) live in @PureClaw.Channels.*@ modules.
   53 --
   54 -- 'sendError' only accepts 'PublicError' — internal errors with stack
   55 -- traces or model names cannot be sent to channel users. This is enforced
   56 -- at the type level.
   57 data ChannelHandle = ChannelHandle
   58   { _ch_receive      :: IO IncomingMessage
   59   , _ch_send         :: OutgoingMessage -> IO ()
   60   , _ch_sendError    :: PublicError -> IO ()
   61   , _ch_sendChunk    :: StreamChunk -> IO ()
   62   , _ch_streaming    :: Bool
   63     -- ^ Whether this channel supports streaming output. When 'True',
   64     -- the agent loop sends text via '_ch_sendChunk' during generation
   65     -- and skips the full '_ch_send'. When 'False', only '_ch_send'
   66     -- is used for the final complete response.
   67   , _ch_readSecret   :: IO Text    -- ^ Read a line without echo (CLI only)
   68   , _ch_prompt       :: Text -> IO Text
   69     -- ^ Display a prompt and read input on the same line (no trailing
   70     -- newline after the prompt text). For non-interactive channels this
   71     -- falls back to send-then-receive.
   72   , _ch_promptSecret :: Text -> IO Text
   73     -- ^ Like '_ch_prompt' but with echo disabled (for passwords / API keys).
   74     -- For non-interactive channels this falls back to '_ch_readSecret'.
   75   }
   76 
   77 -- | No-op channel handle. Receive returns an empty message, send and
   78 -- sendError are silent. readSecret returns empty text.
   79 mkNoOpChannelHandle :: ChannelHandle
   80 mkNoOpChannelHandle = ChannelHandle
   81   { _ch_receive      = pure (IncomingMessage (mkMessageSource (CkOther "noop") noopConversationId Nothing mempty) "")
   82   , _ch_send         = \_ -> pure ()
   83   , _ch_sendError    = \_ -> pure ()
   84   , _ch_sendChunk    = \_ -> pure ()
   85   , _ch_streaming    = False
   86   , _ch_readSecret   = pure ""
   87   , _ch_prompt       = \_ -> pure ""
   88   , _ch_promptSecret = \_ -> pure ""
   89   }
   90 
   91 -- | The conversation id for the no-op channel handle. A constant, since the
   92 -- no-op handle has no real transport and never carries a routable
   93 -- conversation.
   94 noopConversationId :: ConversationId
   95 noopConversationId = ConversationId "noop"
