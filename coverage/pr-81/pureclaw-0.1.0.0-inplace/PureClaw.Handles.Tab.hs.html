    1 {-# LANGUAGE PatternSynonyms #-}
    2 -- |
    3 -- Module      : PureClaw.Handles.Tab
    4 -- Description : Tabbed Chat type layer (WU1) — user-facing primitive: a tab.
    5 --
    6 -- A 'TabHandle' is the universal Handle-pattern carrier for one tab in
    7 -- the Tabbed Chat surface. Each tab carries a 'TabKind' discriminator
    8 -- that classifies it as either a /session-backed/ tab
    9 -- (@'TkSession' sk@, where @sk@ is a 'PureClaw.Session.Kind.SessionKind')
   10 -- or a /raw terminal-backend/ tab
   11 -- (@'TkRawShell' tb@, where @tb@ is a 'PureClaw.Session.Kind.TerminalBackend').
   12 -- The dispatcher routes user input to the focused tab and per-tab
   13 -- output to the channel (focus-gated).
   14 --
   15 -- == Spec-vs-runtime layering (T6)
   16 --
   17 -- 'PureClaw.Session.Kind.SessionKind' and its component types
   18 -- ('ProviderSpec', 'HarnessSpec', 'TerminalBackend', etc.) are
   19 -- /serialisation-safe specs/ — they carry configuration data that can
   20 -- be round-tripped through JSON or TOML without IO.
   21 --
   22 -- Runtime-validated types ('SafePath', 'SshTarget', 'AuthorizedCommand',
   23 -- etc.) live in the factory and transport layers
   24 -- ('PureClaw.Tab.Ai', 'PureClaw.Tab.Backend', 'PureClaw.Tab.Harness')
   25 -- and are never stored in 'TabKind'. This module sits at the boundary:
   26 -- 'TabKind' references the spec layer but never exposes the runtime
   27 -- layer.
   28 --
   29 -- == Scope of WU1
   30 --
   31 -- This module currently lands the type layer only: 'TabHandle' record
   32 -- shape, smart constructors, error ADTs (with the manually-written
   33 -- redacted 'Show' instance for 'TabError' required by H14), and the
   34 -- per-kind factory signatures. Factory bodies are stubbed with
   35 -- @'error' \"not implemented\"@ — the real bodies land in WU6 ('Tab.Ai'),
   36 -- WU7 ('Tab.Harness'), and WU8 ('Tab.Backend').
   37 --
   38 -- == Field-naming convention
   39 --
   40 -- The handle record uses the @_tabHandle_*@ field-name prefix (NOT the
   41 -- terser @_th_*@ used in some older draft notes). This was decided
   42 -- during WU0 planning to keep the prefix self-describing when the field
   43 -- name is grepped in isolation.
   44 --
   45 -- See @docs\/tabbed-chat.md@ for the full design (sections
   46 -- \"TabHandle abstraction (H-series)\" and \"Channel output\").

  106 -- TabIndex
  107 -- ---------------------------------------------------------------------------
  108 
  109 -- | Validated tab index. The data constructor is intentionally NOT
  110 -- exported; obtain values via 'mkTabIndex' (which bounds-checks against
  111 -- the configured @_rc_maxTabs@ ceiling) or by pattern-matching on the
  112 -- newtype via 'unTabIndex'.
  113 --
  114 -- The underlying 'Int' is non-negative on every value produced by
  115 -- 'mkTabIndex' (@0 <= n@). Bounds-checking against @_rc_maxTabs@ is
  116 -- performed at the parser layer (WU2) using the 'RoutingConfig' in
  117 -- scope; the smart constructor here enforces the floor only.
  118 newtype TabIndex = TabIndex { unTabIndex :: Int }
  119   deriving stock (Eq, Ord, Show)
  120 
  121 -- | The only way to construct a 'TabIndex'. Returns 'Nothing' for
  122 -- negative inputs.
  123 mkTabIndex :: Int -> Maybe TabIndex
  124 mkTabIndex n
  125   | n < 0     = Nothing
  126   | otherwise = Just (TabIndex n)

  133 -- | Kind of tab — two-level discriminator (WU-5 refactor, T4).
  134 --
  135 -- * @'TkSession' sk@ — session-backed tab. The 'SessionKind' payload
  136 --   ('SkProvider' or 'SkHarness') is a serialisation-safe /spec/
  137 --   (declared in "PureClaw.Session.Kind"); runtime-validated types
  138 --   ('SafePath', 'SshTarget', etc.) live in factory\/transport layers.
  139 --   This module sits at the boundary: 'TabKind' references the spec
  140 --   layer but never exposes the runtime layer (T6).
  141 --
  142 -- * @'TkRawShell' tb@ — raw terminal-backend tab (local shell, SSH,
  143 --   or tmux). The 'TerminalBackend' payload carries connection
  144 --   coordinates; the backend factory validates and connects.
  145 data TabKind
  146   = TkSession  !SessionKind
  147   | TkRawShell !TerminalBackend
  148   deriving stock (Eq, Show)

  235 -- ---------------------------------------------------------------------------
  236 -- TabStatus
  237 -- ---------------------------------------------------------------------------
  238 
  239 -- | Runtime status of a tab.
  240 --
  241 -- * 'Active' — a provider call or tool execution is in flight, OR a
  242 --   backend recv is waiting for output. The dashboard does not
  243 --   distinguish these sub-cases.
  244 -- * 'Idle' — the tab is waiting for input; carries the timestamp of
  245 --   the last observed input (for sort / display).
  246 -- * 'Crashed' — the tab loop hit a synchronous exception and exited;
  247 --   the wrapped 'PublicTabError' is the channel-safe summary. The
  248 --   registry entry persists until @\/tab close N@ or
  249 --   @\/tab resume \<session\>@.
  250 data TabStatus
  251   = Active
  252   | Idle !UTCTime
  253   | Crashed !PublicTabError
  254   deriving stock (Eq, Show)

  257 -- ---------------------------------------------------------------------------
  258 -- CloseMode
  259 -- ---------------------------------------------------------------------------
  260 
  261 -- | Close semantics requested by the caller of '_tabHandle_close'.
  262 --
  263 -- * 'CloseGraceful' — kind-specific graceful close: for AI sessions
  264 --   ('TkSession (SkProvider _)') archives the session via @_sh_save@;
  265 --   for non-AI kinds runs the underlying destructive close (e.g.
  266 --   @_bh_close@, @_hh_stop@).
  267 -- * 'CloseForce' — for AI sessions skips the archive (transcript
  268 --   deleted from disk); for non-AI kinds is a no-op distinct from
  269 --   'CloseGraceful' because close is already destructive there.
  270 data CloseMode
  271   = CloseGraceful
  272   | CloseForce
  273   deriving stock (Eq, Show)

  276 -- ---------------------------------------------------------------------------
  277 -- TabRunner
  278 -- ---------------------------------------------------------------------------
  279 
  280 -- | The cancel + wait pair returned by the @_env_fork@ test seam.
  281 --
  282 -- Cannot be an 'Control.Concurrent.Async.Async' directly because:
  283 -- (a) the synchronous test variant runs the body inline and has no
  284 -- live async to wrap; (b) the IORef-Bool cancel-observability primitive
  285 -- used by tests needs a uniform shape across sync and async variants.
  286 --
  287 -- See the @TabRunner bootstrapping during spawn@ section of the design
  288 -- doc and the WU0 scaffold in @Test.Fake.TabFactory@.
  289 data TabRunner = TabRunner
  290   { _trun_cancel :: IO ()
  291     -- ^ Idempotent cancel. Safe to call multiple times; safe to call
  292     --   on a runner whose body has already exited.
  293   , _trun_wait   :: IO ()
  294     -- ^ Block until the tab loop's body exits. In the synchronous
  295     --   test variant returns immediately because the body has
  296     --   already run inline.
  297   }

  300 -- ---------------------------------------------------------------------------
  301 -- TabHandle
  302 -- ---------------------------------------------------------------------------
  303 
  304 -- | Universal Handle-pattern carrier for one tab.
  305 --
  306 -- Pure fields ('_tabHandle_index', '_tabHandle_name', '_tabHandle_kind')
  307 -- are bound at construction time. IO-action fields are kind-specific:
  308 -- 'TkSession (SkProvider _)' enqueues into a per-tab @TBQueue InputEvent@; non-AI kinds
  309 -- write to a 'PureClaw.Handles.Backend.BackendHandle' or
  310 -- 'PureClaw.Handles.Harness.HarnessHandle'.
  311 --
  312 -- '_tabHandle_close' is idempotent and never throws (parallel to the
  313 -- @_bh_close@ contract on 'PureClaw.Handles.Backend.BackendHandle').
  314 data TabHandle = TabHandle
  315   { _tabHandle_index         :: !TabIndex
  316     -- ^ Pure: tab's slot in the registry (H12-adjacent).
  317   , _tabHandle_name          :: !TabName
  318     -- ^ Pure: redacted friendly label (H11). Constructed via
  319     --   @sanitizeTabName@ (WU2) at the factory site.
  320   , _tabHandle_kind          :: !TabKind
  321     -- ^ Pure: H12. Read without IO.
  322   , _tabHandle_status        :: IO TabStatus
  323     -- ^ H5: returns 'Active' \/ 'Idle' \/ 'Crashed'.
  324   , _tabHandle_send          :: Text -> IO (Either TabError ())
  325     -- ^ H4: enqueue a 'UserText' input event. Bounded by
  326     --   @_rc_inputQueueBound@; overflow returns
  327     --   'Left' (e.g. 'TabConcurrencyLimit') so the dispatcher never
  328     --   blocks.
  329   , _tabHandle_enqueueSlash  :: Slash.SlashCommand -> IO (Either TabError ())
  330     -- ^ H13: enqueue a 'SlashCmd' input event. For AI sessions the tab
  331     --   loop runs @executeSlashCommand@ against its own per-tab
  332     --   context; for non-AI kinds the implementation returns
  333     --   @Left ('TabUnsupportedCommand' cmd)@ immediately without
  334     --   enqueueing.
  335   , _tabHandle_close         :: CloseMode -> IO ()
  336     -- ^ H6 \/ H7 \/ H8 \/ H9 \/ H10: idempotent, never throws,
  337     --   kind-specific semantics. The 'CloseMode' selects graceful
  338     --   vs forced archive behaviour for AI sessions.
  339   }

  381 -- ---------------------------------------------------------------------------
  382 -- TabError
  383 -- ---------------------------------------------------------------------------
  384 
  385 -- | Construction-time and runtime errors on a tab.
  386 --
  387 -- Every constructor's payload is itself a redaction-safe ADT or a
  388 -- bounded primitive — no constructor carries a free 'Data.Text.Text',
  389 -- a raw 'Control.Exception.SomeException' message, or a raw filesystem
  390 -- path. The hand-written 'Show' below (H14) elides every payload value
  391 -- so accidental @show@ of a 'TabError' in a log line cannot leak
  392 -- hostnames, paths, or stderr fragments.
  393 --
  394 -- The exhaustive constructor list is enforced by H3.
  395 data TabError
  396   = TabIndexInUse !TabIndex
  397   | TabIndexOutOfRange !Int
  398   | TabLimitExceeded !Int
  399   | TabBackendConstructFailed !Backend.BackendError
  400   | TabSessionCreateFailed !SessionError
  401   | TabSpawnAuthDenied !PublicAuthError
  402   | TabNotFound !Int
  403   | TabConcurrencyLimit !Int
  404   | TabInvalidName !NameError
  405   | TabUnsupportedCommand !Slash.SlashCommand
  406   deriving stock (Eq)
  407 
  408 -- | Hand-written 'Show' for 'TabError' per H14: constructor names only,
  409 -- payload values elided. Adding a new constructor here requires adding
  410 -- a new branch in this 'Show' instance (H3 enumerates the exhaustive
  411 -- set), so accidentally regressing to @deriving Show@ would surface as
  412 -- a build error from the @-Wincomplete-patterns@ flag.
  413 instance Show TabError where
  414   show e = case e of
  415     TabIndexInUse{}             -> "TabIndexInUse"
  416     TabIndexOutOfRange{}        -> "TabIndexOutOfRange"
  417     TabLimitExceeded{}          -> "TabLimitExceeded"
  418     TabBackendConstructFailed{} -> "TabBackendConstructFailed"
  419     TabSessionCreateFailed{}    -> "TabSessionCreateFailed"
  420     TabSpawnAuthDenied{}        -> "TabSpawnAuthDenied"
  421     TabNotFound{}               -> "TabNotFound"
  422     TabConcurrencyLimit{}       -> "TabConcurrencyLimit"
  423     TabInvalidName{}            -> "TabInvalidName"
  424     TabUnsupportedCommand{}     -> "TabUnsupportedCommand"
  425

  427 -- ---------------------------------------------------------------------------
  428 -- PublicTabError
  429 -- ---------------------------------------------------------------------------
  430 
  431 -- | Channel-safe projection of a 'TabError'.
  432 --
  433 -- The wrapped 'Data.Text.Text' is short, fixed-vocabulary, and
  434 -- contains no hostnames, paths, or secrets. This is what reaches the
  435 -- user via 'PureClaw.Handles.Channel.ChannelHandle' when a tab
  436 -- operation fails.
  437 newtype PublicTabError = PublicTabError { unPublicTabError :: Text }
  438   deriving stock (Eq, Show)
  439 
  440 -- | Project a 'TabError' to its user-visible form.
  441 --
  442 -- The mapping is intentionally coarse: similar failure modes collapse
  443 -- to the same short label so a future audit of channel-bound strings
  444 -- has a small surface to grep.
  445 toPublicTabError :: TabError -> PublicTabError
  446 toPublicTabError e = PublicTabError $ case e of
  447   TabIndexInUse{}             -> "tab: index already in use"
  448   TabIndexOutOfRange{}        -> "tab: index out of range"
  449   TabLimitExceeded{}          -> "tab: maximum tab count reached"
  450   TabBackendConstructFailed{} -> "tab: backend construction failed"
  451   TabSessionCreateFailed{}    -> "tab: session creation failed"
  452   TabSpawnAuthDenied{}        -> "tab: spawn authorization denied"
  453   TabNotFound{}               -> "tab: not found"
  454   TabConcurrencyLimit{}       -> "tab: input queue full"
  455   TabInvalidName{}            -> "tab: invalid name"
  456   TabUnsupportedCommand{}     -> "tab: command not supported on this tab kind"
