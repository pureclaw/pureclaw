   64 -- | Configuration for a harness-backed session (an external CLI tool managed
   65 -- via a terminal backend).
   66 data HarnessSpec = HarnessSpec
   67   { _h_flavour   :: !HarnessFlavour
   68   , _h_backend   :: !TerminalBackend
   69   , _h_cwd       :: !(Maybe Text)
   70   , _h_args      :: ![Text]
   71   , _h_harnessId :: !(Maybe HarnessId)
   72     -- ^ The durable, UUID-backed harness identity (Harness Registry Phase 1).
   73     --
   74     -- ADDITIVE and OPTIONAL (design @docs\/harness-registry.md@ K4): existing
   75     -- @session.json@ files written before this field decode with
   76     -- @_h_harnessId == Nothing@ (tolerant @.:? "harnessId"@), and the key is
   77     -- emitted ONLY when 'Just' (emit-when-Just, like '_h_cwd'). The tmux
   78     -- window name in '_h_backend' is /dual-written/ alongside it for one
   79     -- release so a back-out path and the legacy name-keyed routing fallback
   80     -- both keep working until the registry is the sole key.
   81   , _h_claudeSessionUuid :: !(Maybe Text)
   82     -- ^ The canonical @claude-code@ session UUID minted at spawn time and
   83     -- injected as @--session-id \<uuid\>@ into the spawned @claude@ argv
   84     -- (Harness JSONL Capture, WU6 / @docs\/harness-jsonl-capture.md@). It
   85     -- correlates this harness with its on-disk JSONL session log so a restart
   86     -- can re-derive the log path (WU2).
   87     --
   88     -- NOTE — this is NOT the same identifier as @Registry._he_sessionId@:
   89     -- @_he_sessionId@ is the PureClaw 'PureClaw.Session.Types.SessionId' (which
   90     -- session this harness belongs to). '_h_claudeSessionUuid' is the
   91     -- /claude-code/ tool's own session UUID, used purely for log-file path
   92     -- derivation. The two are deliberately disambiguated.
   93     --
   94     -- ADDITIVE and OPTIONAL: minted ONLY for spawned @claude-code@ harnesses.
   95     -- Adopted harnesses and non-@claude-code@ flavours carry 'Nothing'. Legacy
   96     -- @session.json@ files written before this field decode with 'Nothing'
   97     -- (tolerant @.:? "claudeSessionUuid"@); the key is emitted ONLY when 'Just'.
   98   , _h_canonicalCwd :: !(Maybe Text)
   99     -- ^ The canonicalized spawn working directory used to derive the
  100     -- @claude-code@ JSONL log path (WU2 cross-check\/fallback). This is the
  101     -- @canonicalizePath@ of the resolved spawn cwd at spawn time, persisted so
  102     -- a restart can re-derive the log directory without re-resolving symlinks.
  103     --
  104     -- ADDITIVE and OPTIONAL, mirroring '_h_claudeSessionUuid': set ONLY for
  105     -- spawned @claude-code@ harnesses; 'Nothing' for adopted\/other flavours
  106     -- and for legacy @session.json@ files (tolerant @.:? "canonicalCwd"@,
  107     -- emit-when-'Just').
  108   } deriving stock (Show, Eq)

  114 -- | Known harness flavours. 'HCustom' is only constructible via 'mkHCustom'
  115 -- to reject path separators.
  116 data HarnessFlavour
  117   = HClaudeCode
  118   | HCodex
  119   | HOpenCode
  120   | HHermes
  121   | HPureClaw
  122   | HCustom !Text
  123   deriving stock (Show, Eq, Ord)
  124 
  125 -- | Reasons a raw 'Text' cannot become an 'HCustom' flavour.
  126 data HCustomError
  127   = HCustomEmpty
  128   | HCustomPathSeparator Text
  129   deriving stock (Show, Eq)
  130 
  131 -- | Smart constructor for 'HCustom'. Accepts bare command names; rejects
  132 -- empty strings and strings containing path separators (@\/@ or @\\@).
  133 mkHCustom :: Text -> Either HCustomError HarnessFlavour
  134 mkHCustom raw
  135   | T.null raw = Left HCustomEmpty
  136   | T.any (\c -> c == '/' || c == '\\') raw = Left (HCustomPathSeparator raw)
  137   | otherwise = Right (HCustom raw)
  138 
  139 -- | Map a well-known harness name to its 'HarnessFlavour'. Unknown names
  140 -- fall through to 'HCustom' via 'mkHCustom' (which is infallible for
  141 -- non-empty, separator-free bare names).
  142 fixedFlavourLookup :: Text -> HarnessFlavour
  143 fixedFlavourLookup t = case t of
  144   "claude-code" -> HClaudeCode
  145   "codex"       -> HCodex
  146   "opencode"    -> HOpenCode
  147   "hermes"      -> HHermes
  148   "pureclaw"    -> HPureClaw
  149   other         -> case mkHCustom other of
  150     Right f -> f
  151     -- Should not happen for non-empty, separator-free names passed from
  152     -- config. Defensively fall back to HCustom with the raw text.
  153     Left _  -> HCustom other

  193   deriving stock (Show, Eq, Ord, Enum, Bounded)

  195 -- | A validated container target name. Only constructible via
  196 -- 'mkContainerTarget'.
  197 newtype ContainerTarget = ContainerTarget { unContainerTarget :: Text }
  198   deriving stock (Show, Eq, Ord)
  199 
  200 -- | Reasons a raw 'Text' cannot become a 'ContainerTarget'.
  201 data ContainerTargetError
  202   = TargetEmpty
  203   | TargetInvalidChars Text
  204   deriving stock (Show, Eq)
  205 
  206 -- | Accepts @[a-zA-Z0-9_\\-.\\/:]+@ (supports kubectl @pod\/name:container@
  207 -- format). Rejects empty strings and strings containing shell
  208 -- metacharacters: @; & | \` $ ( ) { } < > ! \\ \" \' @ space/tab/newline.
  209 mkContainerTarget :: Text -> Either ContainerTargetError ContainerTarget
  210 mkContainerTarget raw
  211   | T.null raw = Left TargetEmpty
  212   | T.all isValidTargetChar raw = Right (ContainerTarget raw)
  213   | otherwise = Left (TargetInvalidChars raw)
  214 
  215 -- | Character predicate for container target names. Allows alphanumerics,
  216 -- @_@, @-@, @.@, @/@, @:@ only.
  217 isValidTargetChar :: Char -> Bool
  218 isValidTargetChar c =
  219   Char.isAsciiUpper c
  220     || Char.isAsciiLower c
  221     || Char.isDigit c
  222     || c == '_'
  223     || c == '-'
  224     || c == '.'
  225     || c == '/'
  226     || c == ':'

  302 instance Aeson.ToJSON TerminalBackend where
  303   toJSON TbLocal = Aeson.object ["tag" .= ("local" :: Text)]
  304   toJSON (TbTmux tc) = Aeson.object $
  305     [ "tag"     .= ("tmux" :: Text)
  306     , "session" .= _tc_session tc
  307     , "window"  .= _tc_window tc
  308     ] ++ maybe [] (\p -> ["pane" .= p]) (_tc_pane tc)
  309   toJSON (TbSsh sc) = Aeson.object $
  310     [ "tag"  .= ("ssh" :: Text)
  311     , "user" .= _sc_user sc
  312     , "host" .= _sc_host sc
  313     ] ++ maybe [] (\p -> ["port" .= p]) (_sc_port sc)
  314   toJSON (TbContainer cs) = Aeson.object
  315     [ "tag"    .= ("container" :: Text)
  316     , "engine" .= _cs_engine cs
  317     , "target" .= _cs_target cs
  318     ]
  319 
  320 instance Aeson.FromJSON TerminalBackend where
  321   parseJSON = Aeson.withObject "TerminalBackend" $ \o -> do
  322     tag <- o .: "tag" :: Parser Text
  323     case tag of
  324       "local"     -> pure TbLocal
  325       "tmux"      -> TbTmux <$>
  326         -- 'session' optional (default "") — see the 'TmuxConfig' FromJSON note.
  327         (TmuxConfig <$> (o .:? "session" .!= "") <*> o .: "window" <*> o .:? "pane")
  328       "ssh"       -> TbSsh <$>
  329         (SshConfig <$> o .: "user" <*> o .: "host" <*> o .:? "port")
  330       "container" -> TbContainer <$>
  331         (ContainerSpec <$> o .: "engine" <*> o .: "target")
  332       other -> fail ("unknown TerminalBackend tag: " ++ show other)

  334 -- HarnessFlavour — string; custom: prefix for HCustom
  335 -- FromJSON routes custom:* through mkHCustom to reject path separators (S12)
  336 
  337 instance Aeson.ToJSON HarnessFlavour where
  338   toJSON HClaudeCode = Aeson.String "claude-code"
  339   toJSON HCodex      = Aeson.String "codex"
  340   toJSON HOpenCode   = Aeson.String "opencode"
  341   toJSON HHermes     = Aeson.String "hermes"
  342   toJSON HPureClaw   = Aeson.String "pureclaw"
  343   toJSON (HCustom n) = Aeson.String ("custom:" <> n)
  344 
  345 instance Aeson.FromJSON HarnessFlavour where
  346   parseJSON = Aeson.withText "HarnessFlavour" $ \case
  347     "claude-code" -> pure HClaudeCode
  348     "codex"       -> pure HCodex
  349     "opencode"    -> pure HOpenCode
  350     "hermes"      -> pure HHermes
  351     "pureclaw"    -> pure HPureClaw
  352     other
  353       | Just rest <- T.stripPrefix "custom:" other ->
  354           case mkHCustom rest of
  355             Right hf -> pure hf
  356             Left HCustomEmpty -> fail "HarnessFlavour: custom name is empty"
  357             Left (HCustomPathSeparator raw) ->
  358               fail ("HarnessFlavour: path separator in custom name: " ++ show raw)
  359       | otherwise -> fail ("unknown HarnessFlavour: " ++ show other)

  375 -- HarnessSpec — flat object; cwd optional (omitted when Nothing),
  376 -- args defaults to [], harnessId optional (emitted ONLY when Just; absent on
  377 -- decode -> Nothing — the back-compat guarantee for legacy session.json, K4).
  378 
  379 instance Aeson.ToJSON HarnessSpec where
  380   toJSON hs = Aeson.object $
  381     [ "flavour" .= _h_flavour hs
  382     , "backend" .= _h_backend hs
  383     ] ++ maybe [] (\c -> ["cwd" .= c]) (_h_cwd hs)
  384       ++ ["args" .= _h_args hs | not (null (_h_args hs))]
  385       ++ maybe [] (\hid -> ["harnessId" .= hid]) (_h_harnessId hs)
  386       ++ maybe [] (\u -> ["claudeSessionUuid" .= u]) (_h_claudeSessionUuid hs)
  387       ++ maybe [] (\c -> ["canonicalCwd" .= c]) (_h_canonicalCwd hs)

  389 instance Aeson.FromJSON HarnessSpec where
  390   parseJSON = Aeson.withObject "HarnessSpec" $ \o ->
  391     HarnessSpec
  392       <$> o .: "flavour"
  393       <*> o .: "backend"
  394       <*> o .:? "cwd"
  395       <*> o .:? "args" .!= []
  396       <*> o .:? "harnessId"
  397       <*> o .:? "claudeSessionUuid"
  398       <*> o .:? "canonicalCwd"

  400 -- SessionKind — tag-discriminated: "provider" or "harness"
  401 
  402 instance Aeson.ToJSON SessionKind where
  403   toJSON (SkProvider ps) = case Aeson.toJSON ps of
  404     Aeson.Object o -> Aeson.Object (KM.insert (AesonKey.fromText "tag") (Aeson.String "provider") o)
  405     other -> other  -- should not happen
  406   toJSON (SkHarness hs) = case Aeson.toJSON hs of
  407     Aeson.Object o -> Aeson.Object (KM.insert (AesonKey.fromText "tag") (Aeson.String "harness") o)
  408     other -> other  -- should not happen
  409 
  410 instance Aeson.FromJSON SessionKind where
  411   parseJSON = Aeson.withObject "SessionKind" $ \o -> do
  412     tag <- o .: "tag" :: Parser Text
  413     case tag of
  414       "provider" -> SkProvider <$> Aeson.parseJSON (Aeson.Object o)
  415       "harness"  -> SkHarness <$> Aeson.parseJSON (Aeson.Object o)
  416       other -> fail ("unknown SessionKind tag: " ++ show other)

  422 -- | Infer a 'ProviderId' from a model name prefix.
  423 --
  424 -- * @claude-*@, @opus-*@, @sonnet-*@, @haiku-*@ -> @\"anthropic\"@
  425 -- * @gpt-*@, @o1-*@, @o3-*@, @o4-*@ -> @\"openai\"@
  426 -- * @gemini-*@ -> @\"google\"@
  427 -- * anything else -> @\"anthropic\"@ (default)
  428 inferProviderId :: Text -> ProviderId
  429 inferProviderId model
  430   | matchesAny ["claude-", "opus-", "sonnet-", "haiku-"] = ProviderId "anthropic"
  431   | matchesAny ["gpt-", "o1-", "o3-", "o4-"]            = ProviderId "openai"
  432   | T.isPrefixOf "gemini-" model                         = ProviderId "google"
  433   | otherwise                                            = ProviderId "anthropic"
  434   where
  435     matchesAny :: [Text] -> Bool
  436     matchesAny = any (`T.isPrefixOf` model)
