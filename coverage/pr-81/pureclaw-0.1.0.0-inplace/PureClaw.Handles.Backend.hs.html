    1 -- |
    2 -- Module      : PureClaw.Handles.Backend
    3 -- Description : Unified terminal backend type layer (WU1).
    4 --
    5 -- == Choose-a-kind decision tree
    6 --
    7 -- Use this decision tree to pick a 'BackendKind':
    8 --
    9 -- * 'Pipe' — one-shot, non-conversational. Write stdin, close stdin,
   10 --   read stdout until EOF. Stderr is merged into stdout at the OS
   11 --   level. Use for: @hostname@, @nix build@, one-shot @ssh host hostname@,
   12 --   any program that does its work and exits.
   13 -- * 'Pty' — conversational, PTY-backed, idle-detected. The child sees a
   14 --   real terminal; stdout and stderr are merged (this is how every
   15 --   terminal works). Use for: local @bash@, @ghci@, remote @ssh -tt@
   16 --   shells, and @ssh -tt … tmux attach@ when attaching to a
   17 --   pre-existing tmux window.
   18 -- * @TmuxRpc@ (future) — non-PTY tmux RPC via @send-keys@ /
   19 --   @capture-pane@. Not yet implemented. Will be additive: a new
   20 --   'BackendKind' constructor and a new factory. The decision tree
   21 --   gains: \"if you need to control a tmux window someone else owns
   22 --   without disturbing them, use TmuxRpc.\"
   23 --
   24 -- The trade-off in one line: 'Pipe' is the safest one-shot, 'Pty' is
   25 -- the only conversational kind in v1, and @TmuxRpc@ (future) is the
   26 -- only non-intrusive controller.
   27 --
   28 -- == Field-naming convention
   29 --
   30 -- The handle record uses @_bh_*@; per-kind option records use prefixes
   31 -- that match the project's universal Handle convention
   32 -- (@_pto_*@ for 'PtyOpts', @_po_*@ for 'PipeOpts').

  154 -- | The mechanism a 'BackendHandle' uses to talk to its subprocess.
  155 --
  156 -- A future @TmuxRpc@ constructor will be added when non-PTY tmux RPC
  157 -- (send-keys / capture-pane) lands; pattern matches on 'BackendKind'
  158 -- in dependent code should remain exhaustive across the existing
  159 -- constructors and explicitly handle (or @-Wall@-fail-shut) any new
  160 -- one.
  161 data BackendKind
  162   = Pipe
  163     -- ^ One-shot, non-conversational: write stdin, close, read until EOF.
  164   | Pty
  165     -- ^ Conversational, PTY-backed, idle-detected.
  166   deriving stock (Eq, Show)

  168 -- | Result of a single recv call.
  169 --
  170 -- Idle-timeout and EOF are reported, not thrown, so callers can decide
  171 -- whether to retry, keep waiting, or move on. 'RecvTruncated' latches
  172 -- across recv calls until '_bh_close'.
  173 data RecvResult a
  174   = RecvSettled !a
  175     -- ^ Output settled within @idleQuietMs@.
  176   | RecvTimedOut !a
  177     -- ^ @idleTimeoutMs@ reached; bytes accumulated so far returned.
  178   | RecvEof !a
  179     -- ^ Subprocess / PTY closed; bytes accumulated so far returned.
  180   | RecvTruncated !a
  181     -- ^ Per-backend read-buffer cap reached; subsequent recvs continue
  182     -- to return 'RecvTruncated' until close.
  183   deriving stock (Eq, Ord, Show, Functor, Foldable, Traversable)
  184 
  185 -- | Project a 'RecvResult' to the accumulated bytes regardless of outcome.
  186 recvBytes :: RecvResult a -> a
  187 recvBytes r = case r of
  188   RecvSettled   a -> a
  189   RecvTimedOut  a -> a
  190   RecvEof       a -> a
  191   RecvTruncated a -> a
  192 
  193 -- | Pure projection of a 'RecvResult'\'s outcome flag.
  194 --
  195 -- Useful when callers only need to switch on the outcome without
  196 -- pattern-matching on the payload.
  197 data RecvOutcome
  198   = Settled
  199   | TimedOut
  200   | Eof
  201   | Truncated
  202   deriving stock (Eq, Show)
  203 
  204 -- | Map a 'RecvResult' to its 'RecvOutcome'.
  205 recvOutcome :: RecvResult a -> RecvOutcome
  206 recvOutcome r = case r of
  207   RecvSettled   _ -> Settled
  208   RecvTimedOut  _ -> TimedOut
  209   RecvEof       _ -> Eof
  210   RecvTruncated _ -> Truncated

  220 -- | Construction-time policy for \"when has output settled?\".
  221 --
  222 -- Constructor is intentionally NOT exported; obtain values via
  223 -- 'mkIdleSpec' or one of the tiered defaults
  224 -- ('localIdle' \/ 'sshIdle' \/ 'tmuxIdle' \/ 'testIdleSpec').
  225 --
  226 -- Invariants enforced by 'mkIdleSpec':
  227 --
  228 -- * Every field is non-negative.
  229 -- * @idleMinFirstByte <= idleTimeoutMs@.
  230 data IdleSpec = IdleSpec
  231   { _idle_quietMs       :: !Int
  232   , _idle_timeoutMs     :: !Int
  233   , _idle_minFirstByte  :: !Int
  234   }
  235   deriving stock (Eq, Show)
  236 
  237 -- | Quiet-window in milliseconds, measured from the latest byte received.
  238 idleQuietMs :: IdleSpec -> Int
  239 idleQuietMs = _idle_quietMs
  240 
  241 -- | Hard ceiling in milliseconds on the total wait per recv.
  242 idleTimeoutMs :: IdleSpec -> Int
  243 idleTimeoutMs = _idle_timeoutMs
  244 
  245 -- | Minimum wait in milliseconds for the first byte before treating
  246 -- \"no bytes\" as settled.
  247 idleMinFirstByte :: IdleSpec -> Int
  248 idleMinFirstByte = _idle_minFirstByte
  249 
  250 -- | The only way to construct an 'IdleSpec'.
  251 --
  252 -- Arguments are @idleQuietMs@, @idleTimeoutMs@, @idleMinFirstByte@ in
  253 -- that order. Returns @Left 'BackendInvalidOption'@ if any argument is
  254 -- negative or if @idleMinFirstByte > idleTimeoutMs@.
  255 mkIdleSpec :: Int -> Int -> Int -> Either BackendError IdleSpec
  256 mkIdleSpec quietMs timeoutMs minFirstByte
  257   | quietMs < 0 =
  258       Left (BackendInvalidOption
  259               (InvalidOptionDetail "IdleSpec: idleQuietMs must be non-negative"))
  260   | timeoutMs < 0 =
  261       Left (BackendInvalidOption
  262               (InvalidOptionDetail "IdleSpec: idleTimeoutMs must be non-negative"))
  263   | minFirstByte < 0 =
  264       Left (BackendInvalidOption
  265               (InvalidOptionDetail "IdleSpec: idleMinFirstByte must be non-negative"))
  266   | minFirstByte > timeoutMs =
  267       Left (BackendInvalidOption
  268               (InvalidOptionDetail
  269                 "IdleSpec: idleMinFirstByte must be <= idleTimeoutMs"))
  270   | otherwise =
  271       Right IdleSpec
  272         { _idle_quietMs      = quietMs
  273         , _idle_timeoutMs    = timeoutMs
  274         , _idle_minFirstByte = minFirstByte
  275         }

  277 -- | Helper for building compile-time 'IdleSpec' constants. The argument
  278 -- triples are statically known and validated, so a 'mkIdleSpec' failure
  279 -- is a programming error in this module itself, not a runtime
  280 -- condition that callers can recover from.
  281 unsafeMkIdleSpec :: String -> Int -> Int -> Int -> IdleSpec
  282 unsafeMkIdleSpec label quietMs timeoutMs minFirstByte =
  283   case mkIdleSpec quietMs timeoutMs minFirstByte of
  284     Right s -> s
  285     Left _  -> error ("unsafeMkIdleSpec: invalid " ++ label)

  321 -- | Environment variable names that must never appear in a caller-supplied
  322 -- 'EnvMap'.
  323 --
  324 -- Includes loader-hijack hooks ('LD_PRELOAD' \/ 'DYLD_INSERT_LIBRARIES' \/
  325 -- the @LIBRARY_PATH@ family), ssh-agent\/askpass channels, shell init
  326 -- hooks ('BASH_ENV', 'ENV', 'PROMPT_COMMAND', 'PS4'), and Git\'s
  327 -- @GIT_SSH_COMMAND@ \/ @GIT_SSH@ overrides.
  328 forbiddenEnvVars :: Set String
  329 forbiddenEnvVars = Set.fromList
  330   [ "LD_PRELOAD"
  331   , "DYLD_INSERT_LIBRARIES"
  332   , "LD_LIBRARY_PATH"
  333   , "DYLD_LIBRARY_PATH"
  334   , "DYLD_FALLBACK_LIBRARY_PATH"
  335   , "SSH_AUTH_SOCK"
  336   , "SSH_ASKPASS"
  337   , "SSH_ASKPASS_REQUIRE"
  338   , "SSH_AGENT_PID"
  339   , "BASH_ENV"
  340   , "ENV"
  341   , "GIT_SSH_COMMAND"
  342   , "GIT_SSH"
  343   , "PROMPT_COMMAND"
  344   , "PS4"

  347 -- | Build an 'EnvMap' from a key/value list.
  348 --
  349 -- Rejects any key listed in 'forbiddenEnvVars' with a
  350 -- 'BackendInvalidOption' carrying the offending name.
  351 mkEnvMap :: [(String, String)] -> Either BackendError EnvMap
  352 mkEnvMap = go Map.empty
  353   where
  354     go acc [] = Right acc
  355     go acc ((k, v) : rest)
  356       | Set.member k forbiddenEnvVars =
  357           Left (BackendInvalidOption
  358                   (InvalidOptionDetail
  359                     (T.pack ("EnvMap: forbidden var '" ++ k ++ "'"))))
  360       | otherwise =
  361           go (Map.insert k (EnvValue (encodeUtf8Latin1 v)) acc) rest
  362 
  363     -- Latin-1 round-trip ByteString-encode for environment values without
  364     -- pulling in 'text' \<-\> 'bytestring' overhead. Environment values are
  365     -- typically ASCII; non-ASCII bytes are preserved 1:1 because the
  366     -- subprocess sees raw bytes.
  367     encodeUtf8Latin1 :: String -> ByteString
  368     encodeUtf8Latin1 s = TE.encodeUtf8 (T.pack s)

  452 -- | Construction-time backend errors.
  453 --
  454 -- Every constructor's payload is itself a redaction-safe ADT — no
  455 -- field carries a free 'Text', a raw 'SomeException' message, or a
  456 -- raw filesystem path. The hand-written 'Show' just calls 'show' on
  457 -- the inner payload, so adding a new constructor cannot accidentally
  458 -- regress the redaction guarantee.
  459 data BackendError
  460   = BackendBinaryNotFound !CommandName
  461   | BackendPtyAllocFailed !PtyAllocFailure
  462   | BackendSshConnectFailed !SshConnectFailure
  463   | BackendTmuxTargetMissing !TmuxTargetRef
  464   | BackendInvalidOption !InvalidOptionDetail
  465   | BackendBufferQuotaExceeded !Int
  466     -- ^ The 'Int' is the requested per-backend cap, in MiB.
  467   | BackendBrokenTmuxTarget !TmuxTargetRef
  468     -- ^ The pinned @\@window_id@ no longer matches a live tmux window.
  469   deriving stock (Eq)

  471 -- Hand-written: routes through 'PureClaw.Internal.Redact.redactedShowString'
  472 -- so the rendered output is always redaction-safe — no raw hostnames,
  473 -- IPv4 addresses, paths, or ssh stderr fragments.
  474 --
  475 -- The constructor-by-constructor case is here (not in @Redact@) to
  476 -- keep @Redact@ outside @Handles.Backend@\'s import cycle: each branch
  477 -- renders a fixed prefix plus 'show' on a leaf ADT, then the whole
  478 -- string is passed through the same pipeline 'redactErr' uses.
  479 instance Show BackendError where
  480   show e = T.unpack (Redact.redactedShowString (renderBackendError e))

  524 -- | Runtime exception raised from inside '_bh_send' \/ '_bh_recv' \/
  525 -- '_bh_close' when something below the abstraction goes wrong (broken
  526 -- pipe, ssh transport drop, kernel error during PTY I\/O, etc.).
  527 --
  528 -- The hand-written 'Show' deliberately does NOT include the inner
  529 -- 'SomeException'\'s message: that message may carry raw hostnames or
  530 -- file paths from a child program. The real redaction-routing
  531 -- replaces this stub in WU2.
  532 data BackendException = BackendException
  533   { _be_context :: !BackendContext
  534   , _be_cause   :: !SomeException
  535   }

  537 -- Hand-written: routes through 'PureClaw.Internal.Redact.redactedShowString'.
  538 -- The wrapped 'SomeException'\'s message is rendered first via
  539 -- 'displayException' and then fed through the redaction pipeline.
  540 instance Show BackendException where
  541   show be =
  542     T.unpack
  543       (Redact.redactedShowString
  544         ( "BackendException { _be_context = "
  545             <> show (_be_context be)
  546             <> ", _be_cause = "
  547             <> displayException (_be_cause be)
  548             <> " }"

  609 -- | Subprocess I\/O capability: a record of IO actions tagged with its
  610 -- 'BackendKind'.
  611 --
  612 -- Concurrency contract (documented in the design doc): '_bh_send' and
  613 -- '_bh_recv' MUST NOT be called concurrently from multiple caller
  614 -- threads against the same handle. Callers needing shared access wrap
  615 -- in an 'Control.Concurrent.MVar.MVar BackendHandle'.
  616 --
  617 -- '_bh_close' is idempotent and never throws on any backend kind.
  618 data BackendHandle = BackendHandle
  619   { _bh_name        :: !Text
  620     -- ^ Human-readable, for logs; not an identity.
  621   , _bh_kind        :: !BackendKind
  622     -- ^ Kind tag for introspection.
  623   , _bh_defaultIdle :: !IdleSpec
  624     -- ^ The 'IdleSpec' bound at construction.
  625   , _bh_send        :: ByteString -> IO ()
  626   , _bh_recv        :: Maybe IdleSpec -> IO (RecvResult ByteString)
  627     -- ^ Read output. @Nothing@ uses '_bh_defaultIdle';
  628     -- @Just s@ overrides for this call only.
  629   , _bh_resize      :: Cols -> Rows -> IO ()
  630     -- ^ No-op on 'Pipe'-kind backends (and future @TmuxRpc@).
  631   , _bh_close       :: IO ()
  632     -- ^ Idempotent; never throws.
  633   }
  634 
  635 -- | Send a request and receive the response with the handle's default
  636 -- idle policy. For 'Pipe' backends this is a true request\/response;
  637 -- for 'Pty' backends the response is whatever the idle policy
  638 -- captures, which may include unrelated background output.
  639 runBackend :: BackendHandle -> ByteString -> IO (RecvResult ByteString)
  640 runBackend b bs = _bh_send b bs *> _bh_recv b Nothing
  641 
  642 -- | Whether the handle's kind is conversational
  643 -- ('Pty' — @True@; 'Pipe' — @False@).
  644 isConversational :: BackendHandle -> Bool
  645 isConversational b = case _bh_kind b of
  646   Pty  -> True
  647   Pipe -> False
  648 
  649 -- | Equivalent to @_bh_recv b Nothing@ as a top-level helper.
  650 recv :: BackendHandle -> IO (RecvResult ByteString)
  651 recv b = _bh_recv b Nothing
  652 
  653 -- | Equivalent to @_bh_recv b (Just s)@ as a top-level helper, for the
  654 -- common case of overriding the default idle policy for a single call.
  655 recvWith :: BackendHandle -> IdleSpec -> IO (RecvResult ByteString)
  656 recvWith b s = _bh_recv b (Just s)
  657 
  658 -- | Run an action with an already-constructed 'BackendHandle',
  659 -- guaranteeing '_bh_close' runs on success or exception.
  660 --
  661 -- Body exceptions are re-raised after the close action finishes.
  662 -- '_bh_close' is documented as idempotent and never-throwing on every
  663 -- backend kind, so the close phase cannot mask a body exception.
  664 withBackendHandle :: BackendHandle -> (BackendHandle -> IO a) -> IO a
  665 withBackendHandle b = bracket (pure b) (\_ -> _bh_close b)
  666 
  667 -- | Combined acquire-and-bracket helper for the common factory pattern
  668 -- @IO (Either BackendError BackendHandle)@.
  669 --
  670 -- On a 'Left' result from @acquire@, no body is run and the error is
  671 -- returned as-is. On a 'Right' result, behaves like 'withBackendHandle'
  672 -- and re-raises any body exception after closing.
  673 withBackendHandleE
  674   :: IO (Either BackendError BackendHandle)
  675   -> (BackendHandle -> IO a)
  676   -> IO (Either BackendError a)
  677 withBackendHandleE acquire body = do
  678   result <- acquire
  679   case result of
  680     Left  e -> pure (Left e)
  681     Right b -> Right <$> withBackendHandle b body
  682 
  683 -- | Wrap a 'BackendHandle' so that concurrent entry into '_bh_send'
  684 -- and '_bh_recv' from multiple threads is detected and rejected.
  685 --
  686 -- The documented concurrency contract on 'BackendHandle' says callers
  687 -- MUST NOT invoke '_bh_send' or '_bh_recv' from two threads at once;
  688 -- this wrapper enforces that contract by gating both actions on a
  689 -- single 'MVar'. The second-entrant observes a 'BackendException' with
  690 -- @_be_context = 'BcConcurrentUse'@ instead of corrupting state.
  691 --
  692 -- '_bh_resize' and '_bh_close' are NOT gated: resize is documented as
  693 -- safe from any thread, and close is documented as idempotent.
  694 withConcurrentUseGuard :: BackendHandle -> IO BackendHandle
  695 withConcurrentUseGuard bh = do
  696   gate <- newMVar ()
  697   pure bh
  698     { _bh_send = guarded gate . _bh_send bh
  699     , _bh_recv = guarded gate . _bh_recv bh
  700     }
  701   where
  702     guarded :: MVar () -> IO a -> IO a
  703     guarded gate io = do
  704       m <- tryTakeMVar gate
  705       case m of
  706         Nothing ->
  707           throwIO $ BackendException
  708             BcConcurrentUse
  709             (toException
  710                (userError "concurrent _bh_send/_bh_recv on a BackendHandle"))
  711         Just () -> io `finally` putMVar gate ()
  712

  739 -- | Construction-time configuration for 'mkInMemoryBackendHandle'.
  740 --
  741 -- The caller owns @_imc_state@ — create it with 'newInMemoryState'
  742 -- ahead of time and inspect it via 'readIORef' after exercising the
  743 -- returned handle. This shape keeps the factory's signature
  744 -- @BackendKind -> InMemoryConfig -> IO BackendHandle@ (per the design
  745 -- doc) while still giving tests a way to introspect what was sent /
  746 -- received / resized / closed.
  747 data InMemoryConfig = InMemoryConfig
  748   { _imc_clock           :: !FakeClock
  749     -- ^ Deterministic clock used for idle measurement (not yet wired
  750     -- to the recv path in WU3; required by the design doc for parity
  751     -- with @fakePtyIO@ and for forward-compatibility with the
  752     -- property-test seam in WU7).
  753   , _imc_scriptedReplies :: ![ByteString]
  754     -- ^ Consumed in order on each '_bh_recv'. Once exhausted,
  755     -- subsequent recvs return @RecvEof BS.empty@.
  756   , _imc_eofAfter        :: !(Maybe Int)
  757     -- ^ Optional cap: after this many '_bh_recv' calls, every
  758     -- subsequent recv returns @RecvEof BS.empty@ regardless of any
  759     -- remaining scripted replies.
  760   , _imc_state           :: !(IORef InMemoryState)
  761     -- ^ Caller-supplied introspection state. Construct via
  762     -- 'newInMemoryState'.
  763   }
  764 
  765 -- | Mutable in-memory backend state. Updated in place by the
  766 -- 'BackendHandle' returned from 'mkInMemoryBackendHandle'.
  767 --
  768 -- Lists are stored in reverse-chronological order for cheap @cons@;
  769 -- callers that want chronological order should 'reverse' on read.
  770 data InMemoryState = InMemoryState
  771   { _ims_sentBytes   :: ![ByteString]
  772     -- ^ Every payload passed to '_bh_send', most-recent first.
  773   , _ims_recvCalls   :: !Int
  774     -- ^ Number of '_bh_recv' invocations issued against the handle.
  775   , _ims_resizeCalls :: ![(Cols, Rows)]
  776     -- ^ Every @(cols, rows)@ pair passed to '_bh_resize', most-recent
  777     -- first. For 'Pipe'-kind handles, this list always stays empty
  778     -- because '_bh_resize' is a silent no-op on pipes.
  779   , _ims_closed      :: !Bool
  780     -- ^ Set to 'True' on first '_bh_close'; remains 'True' on subsequent
  781     -- (idempotent) calls.
  782   }
  783   deriving stock (Eq, Show)
  784 
  785 -- | Allocate a fresh, empty 'InMemoryState' for 'InMemoryConfig'.
  786 newInMemoryState :: IO (IORef InMemoryState)
  787 newInMemoryState = newIORef InMemoryState
  788   { _ims_sentBytes   = []
  789   , _ims_recvCalls   = 0
  790   , _ims_resizeCalls = []
  791   , _ims_closed      = False
  792   }

  794 -- | A deterministic, scriptable 'BackendHandle' for property tests.
  795 --
  796 -- The handle:
  797 --
  798 -- * records each '_bh_send' payload into '_imc_state' (newest first);
  799 -- * pulls the next scripted reply from '_imc_scriptedReplies' on each
  800 --   '_bh_recv', wrapping it in 'RecvSettled'. Once the script is
  801 --   exhausted — or @_imc_eofAfter@ recvs have occurred, whichever
  802 --   comes first — every subsequent recv returns @RecvEof BS.empty@;
  803 -- * records each '_bh_resize' call into '_imc_state' if @kind == Pty@;
  804 --   for @kind == Pipe@, resize is a silent no-op (nothing recorded);
  805 -- * marks '_ims_closed' on '_bh_close'. Close is idempotent.
  806 mkInMemoryBackendHandle :: BackendKind -> InMemoryConfig -> IO BackendHandle
  807 mkInMemoryBackendHandle kind cfg = do
  808   scriptRef <- newIORef (_imc_scriptedReplies cfg)
  809   let stRef    = _imc_state cfg
  810       eofAfter = _imc_eofAfter cfg
  811 
  812       doRecv :: IO (RecvResult ByteString)
  813       doRecv = do
  814         -- Pre-increment so #-of-calls includes the current invocation
  815         -- when comparing against eofAfter.
  816         n <- atomicModifyIORef' stRef $ \s ->
  817           let n' = _ims_recvCalls s + 1
  818           in (s { _ims_recvCalls = n' }, n')
  819         let pastCap = case eofAfter of
  820               Nothing  -> False
  821               Just cap -> n > cap
  822         if pastCap
  823           then pure (RecvEof BS.empty)
  824           else do
  825             next <- atomicModifyIORef' scriptRef $ \case
  826               []     -> ([], Nothing)
  827               (r:tl) -> (tl, Just r)
  828             case next of
  829               Just r  -> pure (RecvSettled r)
  830               Nothing -> pure (RecvEof BS.empty)
  831 
  832       doResize :: Cols -> Rows -> IO ()
  833       doResize c r = case kind of
  834         Pipe -> pure ()
  835         Pty  -> atomicModifyIORef' stRef $ \s ->
  836           (s { _ims_resizeCalls = (c, r) : _ims_resizeCalls s }, ())
  837 
  838       doSend :: ByteString -> IO ()
  839       doSend bs = atomicModifyIORef' stRef $ \s ->
  840         (s { _ims_sentBytes = bs : _ims_sentBytes s }, ())
  841 
  842       doClose :: IO ()
  843       doClose = atomicModifyIORef' stRef $ \s ->
  844         (s { _ims_closed = True }, ())
  845 
  846   pure BackendHandle
  847     { _bh_name        = "in-memory"
  848     , _bh_kind        = kind
  849     , _bh_defaultIdle = testIdleSpec
  850     , _bh_send        = doSend
  851     , _bh_recv        = const doRecv
  852     , _bh_resize      = doResize
  853     , _bh_close       = doClose
  854     }

  857 -- Process-wide recv-buffer quota (WU7)
  858 --------------------------------------------------------------------------------
  859 
  860 -- | The default process-wide aggregate recv-buffer cap (64 MiB).
  861 --
  862 -- Override at process startup by setting the
  863 -- @PURECLAW_BACKEND_AGGREGATE_CAP_MIB@ environment variable; the
  864 -- 'globalBackendBufferQuota' singleton consults this variable once on
  865 -- first use.
  866 defaultAggregateBufferCapMiB :: Int
  867 defaultAggregateBufferCapMiB = 64
  868 
  869 -- | Process-wide recv-buffer quota, in MiB.
  870 --
  871 -- Backed by an 'IORef' counter (not a @QSem@ \/ @QSemN@): the
  872 -- semantics we want are __non-blocking__ @tryAcquire@-style, returning
  873 -- 'Left BackendBufferQuotaExceeded' on oversubscription rather than
  874 -- blocking the calling thread. @QSemN.tryWaitQSemN@ does not exist
  875 -- upstream, and a counter-based 'atomicModifyIORef'' is simpler and
  876 -- equivalent in correctness for a single-process quota.
  877 --
  878 -- The double-release defense in 'releaseBufferQuota' relies on the
  879 -- per-acquisition 'IORef' flag that backend factories thread through
  880 -- their close action; this 'BackendBufferQuota' type itself stores
  881 -- only the running total.
  882 newtype BackendBufferQuota = BackendBufferQuota (IORef Int)
  883 
  884 -- | Construct a quota with the given capacity (in MiB). Available
  885 -- units start at the capacity.
  886 newBackendBufferQuota :: Int -> IO BackendBufferQuota
  887 newBackendBufferQuota capMiB = BackendBufferQuota <$> newIORef capMiB
  888 
  889 -- | The shared, process-wide 'BackendBufferQuota' used by all real
  890 -- backend factories.
  891 --
  892 -- Sized from the @PURECLAW_BACKEND_AGGREGATE_CAP_MIB@ environment
  893 -- variable at first use; falls back to 'defaultAggregateBufferCapMiB'
  894 -- if the variable is unset or unparseable.
  895 --
  896 -- Tests that need to drive oversubscription deterministically can
  897 -- replace the singleton with 'setGlobalBackendBufferQuotaForTest'.
  898 globalBackendBufferQuota :: IORef BackendBufferQuota
  899 globalBackendBufferQuota = unsafePerformIO $ do
  900   cap <- readBufferQuotaCap
  901   q   <- newBackendBufferQuota cap
  902   newIORef q
  903 {-# NOINLINE globalBackendBufferQuota #-}
  904 
  905 readBufferQuotaCap :: IO Int
  906 readBufferQuotaCap = do
  907   mEnv <- lookupEnv "PURECLAW_BACKEND_AGGREGATE_CAP_MIB"
  908   case mEnv >>= readMaybe of
  909     Just n | n > 0 -> pure n
  910     _              -> pure defaultAggregateBufferCapMiB
  911 
  912 -- | Replace the global quota with a fresh one of the given capacity.
  913 -- Test-only helper; production code must not call this.
  914 setGlobalBackendBufferQuotaForTest :: Int -> IO ()
  915 setGlobalBackendBufferQuotaForTest capMiB = do
  916   q <- newBackendBufferQuota capMiB
  917   writeIORef globalBackendBufferQuota q
  918 
  919 -- | Inspect available MiB in a 'BackendBufferQuota'. Read-only; for
  920 -- tests and diagnostics.
  921 bufferQuotaAvailableMiB :: BackendBufferQuota -> IO Int
  922 bufferQuotaAvailableMiB (BackendBufferQuota ref) = readIORef ref
  923 
  924 -- | Try to acquire @n@ MiB. Returns 'Left BackendBufferQuotaExceeded'
  925 -- if the quota is oversubscribed; the caller is expected to surface
  926 -- this as a 'BackendError' on construction. Atomically decrements the
  927 -- running counter on success.
  928 --
  929 -- Pass a negative or zero @n@ to acquire nothing — useful for pipe-
  930 -- kind factories that don\'t carry a recv buffer.
  931 acquireBufferQuota :: BackendBufferQuota -> Int -> IO (Either BackendError ())
  932 acquireBufferQuota (BackendBufferQuota ref) n
  933   | n <= 0    = pure (Right ())
  934   | otherwise = atomicModifyIORef' ref $ \avail ->
  935       if avail >= n
  936         then (avail - n, Right ())
  937         else (avail, Left (BackendBufferQuotaExceeded n))
  938 
  939 -- | Release @n@ MiB back to the quota.
  940 --
  941 -- Designed to be called inside a backend\'s close action. The double-
  942 -- release defense lives at the call site (a per-backend 'IORef Bool'
  943 -- flag flipped on first release); this function unconditionally
  944 -- increments and is therefore safe to call from a 'bracket' cleanup.
  945 releaseBufferQuota :: BackendBufferQuota -> Int -> IO ()
  946 releaseBufferQuota (BackendBufferQuota ref) n
  947   | n <= 0    = pure ()
  948   | otherwise = atomicModifyIORef' ref $ \avail -> (avail + n, ())
