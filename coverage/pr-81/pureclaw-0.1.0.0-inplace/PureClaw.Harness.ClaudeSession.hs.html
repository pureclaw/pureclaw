    1 -- | A validated, canonical UUID used to correlate a spawned @claude-code@
    2 -- harness with its on-disk JSONL session-log file.
    3 --
    4 -- The value constructor is intentionally NOT exported. The only ways to
    5 -- obtain a 'ClaudeSessionUuid' are:
    6 --
    7 --   * 'mkClaudeSessionUuid' — the smart constructor, which accepts ONLY a
    8 --     canonical lowercase UUID (8-4-4-4-12, 36 chars).
    9 --   * 'mintClaudeSessionUuid' — generates a fresh UUIDv4 from a
   10 --     cryptographic source.
   11 --   * the 'Aeson.FromJSON' instance, which routes through
   12 --     'mkClaudeSessionUuid' so corrupted on-disk JSON cannot bypass
   13 --     validation.
   14 --
   15 -- This mirrors the 'PureClaw.Session.Types.SessionPrefix' smart-constructor
   16 -- pattern (newtype + typed error sum + validated 'FromJSON').

   19     -- The data constructor is intentionally NOT exported.
   20     ClaudeSessionUuid

   41 -- | A validated canonical UUID correlating a @claude-code@ harness with its
   42 -- JSONL session log. Always holds a canonical lowercase 36-character UUID.
   43 --
   44 -- The 'Show' instance is deliberately REDACTED: it never prints the raw uuid
   45 -- into general logs. Use 'unClaudeSessionUuid' for the raw value when a caller
   46 -- legitimately needs it (e.g. to build the log path).
   47 newtype ClaudeSessionUuid = ClaudeSessionUuid Text
   48   deriving stock (Eq, Ord)
   49 
   50 -- | Redacted 'Show'. Never leaks the raw uuid into general logs. (D1.4)
   51 instance Show ClaudeSessionUuid where
   52   show _ = "ClaudeSessionUuid <redacted>"
   53 
   54 -- | Explicit accessor for the raw canonical uuid text. Use only where the
   55 -- raw value is genuinely required (e.g. constructing the JSONL log path).
   56 unClaudeSessionUuid :: ClaudeSessionUuid -> Text
   57 unClaudeSessionUuid (ClaudeSessionUuid t) = t
   58 
   59 -- | Reasons a raw 'Text' cannot be promoted to a 'ClaudeSessionUuid'.
   60 -- Mirrors 'PureClaw.Session.Types.SessionPrefixError'.
   61 data ClaudeSessionUuidError
   62   = -- | The input was empty.
   63     UuidEmpty
   64   | -- | The input was not exactly 36 characters.
   65     UuidWrongLength
   66   | -- | The input was 36 characters but not a canonical lowercase UUID
   67     -- (wrong charset, uppercase hex, wrong hyphen positions, contains
   68     -- @\'\/\'@, @\"..\"@, NUL, etc.). Carries the offending input.
   69     UuidNotCanonical Text
   70   deriving stock (Show, Eq)
   71 
   72 -- | Canonical UUID length: 8-4-4-4-12 plus four hyphens = 36 characters.
   73 uuidLength :: Int
   74 uuidLength = 36
   75 
   76 -- | Zero-based positions of the four hyphens in a canonical UUID.
   77 hyphenPositions :: [Int]
   78 hyphenPositions = [8, 13, 18, 23]
   79 
   80 -- | True iff the character is a lowercase hex digit (@0-9@, @a-f@).
   81 -- Uppercase is intentionally rejected (reject-unless-already-canonical).
   82 isLowerHexDigit :: Char -> Bool
   83 isLowerHexDigit c = Char.isDigit c || (c >= 'a' && c <= 'f')
   84 
   85 -- | True iff the character at the given zero-based position is valid for a
   86 -- canonical UUID: a hyphen at a hyphen position, a lowercase hex digit
   87 -- everywhere else.
   88 isValidAtPos :: Int -> Char -> Bool
   89 isValidAtPos pos c =
   90   if pos `elem` hyphenPositions
   91     then c == '-'
   92     else isLowerHexDigit c
   93 
   94 -- | Smart constructor. Accepts ONLY a canonical lowercase UUID in the
   95 -- @8-4-4-4-12@ (36-character) form. Rejects uppercase, wrong length, wrong
   96 -- charset, wrong hyphen positions, and — by construction — any input
   97 -- containing @\'\/\'@, @\"..\"@, or NUL bytes (none of those are valid
   98 -- lowercase-hex-or-hyphen characters in the right positions).
   99 --
  100 -- The policy is reject-unless-already-canonical: an uppercase or otherwise
  101 -- non-canonical UUID is REJECTED, not normalized, so the stored value is a
  102 -- single canonical form that is safe to use directly in a filesystem path.
  103 mkClaudeSessionUuid :: Text -> Either ClaudeSessionUuidError ClaudeSessionUuid
  104 mkClaudeSessionUuid raw
  105   | T.null raw = Left UuidEmpty
  106   | T.length raw /= uuidLength = Left UuidWrongLength
  107   | not isCanonical = Left (UuidNotCanonical raw)
  108   | otherwise = Right (ClaudeSessionUuid raw)
  109   where
  110     isCanonical = all (uncurry isValidAtPos) (zip [0 ..] (T.unpack raw))
  111 
  112 -- | Custom 'FromJSON' routes through 'mkClaudeSessionUuid' so corrupted
  113 -- on-disk JSON cannot bypass the smart constructor. (D1.3)
  114 instance FromJSON ClaudeSessionUuid where
  115   parseJSON = Aeson.withText "ClaudeSessionUuid" $ \t ->
  116     case mkClaudeSessionUuid t of
  117       Right u -> pure u
  118       Left e -> fail ("invalid ClaudeSessionUuid: " <> show e)
  119 
  120 -- | 'ToJSON' renders the canonical uuid text. Round-trips with 'FromJSON'.
  121 -- (D1.3)
  122 instance ToJSON ClaudeSessionUuid where
  123   toJSON = toJSON . unClaudeSessionUuid
  124 
  125 -- | Generate a fresh canonical UUIDv4 from a CRYPTOGRAPHIC source. (D1.2)
  126 --
  127 -- We draw 16 random bytes from 'Crypto.Random.getRandomBytes' (the @crypton@
  128 -- CSPRNG, the same system-entropy source used elsewhere in this codebase for
  129 -- key/IV/pairing-token generation — see "PureClaw.Security.Crypto"). We set
  130 -- the RFC 4122 version (4) and variant (10xx) bits, then render the bytes
  131 -- directly as canonical lowercase hex with hyphens at the 8-4-4-4-12
  132 -- boundaries. We deliberately do NOT use @System.Random@'s default 'StdGen',
  133 -- which is not a CSPRNG.
  134 --
  135 -- Construction is total and branch-free: 16 bytes always render to exactly
  136 -- 36 canonical characters, so no error path is possible.
  137 mintClaudeSessionUuid :: IO ClaudeSessionUuid
  138 mintClaudeSessionUuid = do
  139   raw <- CR.getRandomBytes 16 :: IO ByteString
  140   let bytes = BS.pack (setVersionVariant (BS.unpack raw))
  141   pure (ClaudeSessionUuid (renderCanonical bytes))
  142 
  143 -- | Force the RFC 4122 version (4) and variant (10xx) bits on the 16-byte
  144 -- UUID payload so the result is a well-formed version-4 UUID.
  145 setVersionVariant :: [Word8] -> [Word8]
  146 setVersionVariant = zipWith adjust [0 :: Int ..]
  147   where
  148     adjust 6 b = (b .&. 0x0f) .|. 0x40 -- version 4 in the high nibble
  149     adjust 8 b = (b .&. 0x3f) .|. 0x80 -- variant 10xx in the high bits
  150     adjust _ b = b
  151 
  152 -- | Render exactly 16 bytes as a canonical lowercase UUID: 32 lowercase hex
  153 -- digits (via @base16-bytestring@) with hyphens inserted at the 8-4-4-4-12
  154 -- boundaries. Total: 16 bytes always produce 32 hex chars, which split into
  155 -- the canonical groups.
  156 renderCanonical :: ByteString -> Text
  157 renderCanonical bytes =
  158   let hex = TE.decodeLatin1 (B16.encode bytes) -- 32 lowercase hex chars
  159       (g1, r1) = T.splitAt 8 hex
  160       (g2, r2) = T.splitAt 4 r1
  161       (g3, r3) = T.splitAt 4 r2
  162       (g4, g5) = T.splitAt 4 r3
  163    in T.intercalate "-" [g1, g2, g3, g4, g5]
