   37 -- | Frontend server configuration.
   38 data FrontendConfig = FrontendConfig
   39   { _fsc_port          :: Int
   40   , _fsc_staticDir     :: FilePath
   41   , _fc_allowedOrigins :: [Text]
   42     -- ^ Exact-match allowlist of @Origin@ headers permitted to upgrade to
   43     -- the WS endpoint. Matching is case-insensitive on scheme + host per
   44     -- RFC 6454 (see "PureClaw.Frontend.Stream.normalizeOrigin"). An empty
   45     -- list denies every origin (the deny-all configuration). The default
   46     -- admits @http://localhost:8080@ and @http://127.0.0.1:8080@.
   47   }
   48   deriving stock (Show, Eq)

   50 -- | Default: port 8080, serving from @frontend\/dist@, accepting WS
   51 -- upgrades from @http://localhost:8080@ and @http://127.0.0.1:8080@.
   52 defaultFrontendConfig :: FrontendConfig
   53 defaultFrontendConfig = FrontendConfig
   54   { _fsc_port          = 8080
   55   , _fsc_staticDir     = "frontend/dist"
   56   , _fc_allowedOrigins = [ "http://localhost:8080"

   61 -- | Build the pure subset of Warp settings from the frontend config.
   62 -- Binds the port and applies @setTimeout 30@ to non-hijacked HTTP routes.
   63 -- The connection cap (@setMaxTotalConnections@ substitute) is applied
   64 -- by 'runFrontend' on top of these settings because it requires a
   65 -- runtime 'TVar' counter.
   66 mkFrontendSettings :: FrontendConfig -> Warp.Settings
   67 mkFrontendSettings cfg =
   68   Warp.setPort (_fsc_port cfg)
   69     $ Warp.setTimeout 30
   70       Warp.defaultSettings
   71 
   72 -- | CORS middleware that restricts the @Access-Control-Allow-Origin@
   73 -- header to @http:\/\/localhost:\<port\>@ and handles OPTIONS preflight
   74 -- requests with a 200 response.
   75 corsMiddleware :: FrontendConfig -> Middleware
   76 corsMiddleware cfg app req respond
   77   | requestMethod req == "OPTIONS" =
   78       respond $ responseLBS HTTP.status200 corsHeaders ""
   79   | otherwise =
   80       app req $ \resp ->
   81         respond (mapResponseHeaders (corsHeaders ++) resp)
   82   where
   83     origin :: BS.ByteString
   84     origin = "http://localhost:" <> BS8.pack (show (_fsc_port cfg))
   85 
   86     corsHeaders :: [HTTP.Header]
   87     corsHeaders =
   88       [ ("Access-Control-Allow-Origin",  origin)
   89       , ("Access-Control-Allow-Methods", "GET, POST, PUT, DELETE, OPTIONS")
   90       , ("Access-Control-Allow-Headers", "Content-Type")
   91       ]

   93 -- | Start the frontend server with API endpoints and static file serving.
   94 -- API routes (@\/api\/*@) are handled by 'apiApp'; the WS endpoint at
   95 -- @\/api\/stream@ is composed via 'WaiWS.websocketsOr' from 'streamApp';
   96 -- everything else falls through to the static file server with SPA
   97 -- fallback. CORS middleware is applied to all HTTP routes.
   98 --
   99 -- Warp hardening (D20): an accept-side connection counter caps concurrent
  100 -- HTTP connections at 1024 (Warp 3.4 dropped the dedicated
  101 -- @setMaxTotalConnections@ helper; the same effect is achieved via
  102 -- @setOnOpen@/@setOnClose@ with a TVar gate). @setTimeout 30@ closes
  103 -- idle non-WS HTTP routes after 30 s. The timeout does NOT apply to
  104 -- hijacked WS sockets — those use 'Network.WebSockets.withPingThread'
  105 -- for keepalive (per the design's §Security and 'SECURITY_PRACTICES.md'
  106 -- §9.1).
  107 runFrontend :: FrontendConfig -> Maybe FrontendEnv -> LogHandle -> IO ()
  108 runFrontend cfg mEnv logger = do
  109   let logInfo = _lh_logInfo logger
  110   logInfo "PureClaw frontend server"
  111   logInfo $ "  Serving: " <> T.pack (_fsc_staticDir cfg)
  112   logInfo $ "  URL:     http://localhost:" <> T.pack (show (_fsc_port cfg))
  113   counterTv <- newTVarIO 0
  114   let cap = 1024 :: Int
  115       settings = Warp.setOnOpen  (onOpenCounter counterTv cap)
  116                $ Warp.setOnClose (onCloseCounter counterTv)
  117                $ mkFrontendSettings cfg
  118       app = corsMiddleware cfg (combinedApp cfg mEnv (_fsc_staticDir cfg))
  119   Warp.runSettings settings app
  120 
  121 -- | Connection-open callback. Blocks (via STM @retry@) when the active
  122 -- count reaches the cap so Warp pauses 'accept' until a slot opens up.
  123 -- Returning 'True' admits the connection; 'False' would reject it (we
  124 -- never reject — we throttle by blocking).
  125 onOpenCounter :: TVar Int -> Int -> SockAddr -> IO Bool
  126 onOpenCounter tv cap _addr = do
  127   atomically $ do
  128     n <- readTVar tv
  129     if n >= cap then retry else modifyTVar' tv (+ 1)
  130   pure True
  131 
  132 -- | Connection-close callback. Decrement the counter.
  133 onCloseCounter :: TVar Int -> SockAddr -> IO ()
  134 onCloseCounter tv _addr = atomically $ modifyTVar' tv (subtract 1)

  136 -- | Combined WAI application: WS upgrade (@\/api\/stream@) is composed via
  137 -- 'WaiWS.websocketsOr' so that a single 'Application' handles both the WS
  138 -- upgrade and the JSON HTTP routes. Non-API paths fall through to
  139 -- 'staticApp'.
  140 combinedApp :: FrontendConfig -> Maybe FrontendEnv -> FilePath -> Application
  141 combinedApp cfg mEnv staticDir = WaiWS.websocketsOr
  142   WS.defaultConnectionOptions
  143   (case mEnv of
  144      Just env -> streamApp (_fc_allowedOrigins cfg) env
  145      Nothing  -> rejectStreaming)
  146   apiOrStatic
  147   where
  148     apiOrStatic req respond = case pathInfo req of
  149       ("api" : _) -> case mEnv of
  150         Just env -> apiApp env req respond
  151         Nothing  -> respond $ responseLBS HTTP.status503
  152           [(HTTP.hContentType, "application/json")]
  153           "{\"error\":\"API not available in standalone serve mode\"}"
  154       _ -> staticApp staticDir req respond
  155 
  156     rejectStreaming :: WS.ServerApp
  157     rejectStreaming pending =
  158       WS.rejectRequestWith pending
  159         (WS.defaultRejectRequest
  160           { WS.rejectCode    = 503
  161           , WS.rejectMessage = "streaming disabled"
  162           , WS.rejectBody    = "streaming disabled"
  163           })
  164 
  165 -- | WAI application that serves static files with SPA fallback.
  166 staticApp :: FilePath -> Application
  167 staticApp dir req respond = do
  168   let segments = pathInfo req
  169   -- Reject path traversal
  170   if any (\s -> s == ".." || s == ".") segments
  171     then respond $ responseLBS HTTP.status400 [] "Invalid path"
  172     else do
  173       let relPath  = T.unpack (T.intercalate "/" segments)
  174           filePath = dir </> if null relPath then "index.html" else relPath
  175       exists <- doesFileExist filePath
  176       if exists
  177         then serveFile filePath respond
  178         else do
  179           -- SPA fallback: serve index.html for client-side routing
  180           let indexPath = dir </> "index.html"
  181           indexExists <- doesFileExist indexPath
  182           if indexExists
  183             then serveFile indexPath respond
  184             else respond $ responseLBS HTTP.status404 [] "Not found"
  185 
  186 serveFile :: FilePath -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  187 serveFile path respond = do
  188   contents <- LBS.readFile path
  189   let ct = mimeType (takeExtension path)
  190   respond $ responseLBS HTTP.status200 [(HTTP.hContentType, ct)] contents
  191
