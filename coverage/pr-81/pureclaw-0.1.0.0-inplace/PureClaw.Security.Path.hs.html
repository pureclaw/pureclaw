   38 -- | A filesystem path that has been validated to be within the workspace
   39 -- and not on the blocked list. Constructor is intentionally NOT exported —
   40 -- the only way to obtain a 'SafePath' is through 'mkSafePath'.
   41 newtype SafePath = SafePath { getSafePath :: FilePath }
   42   deriving stock (Eq, Ord)
   43 
   44 instance Show SafePath where
   45   show sp = "SafePath " ++ show (getSafePath sp)

   47 -- | Errors that can occur during path validation.
   48 data PathError
   49   = PathEscapesWorkspace FilePath FilePath
   50     -- ^ requested, resolved
   51   | PathIsBlocked FilePath Text
   52     -- ^ requested, reason
   53   | PathDoesNotExist FilePath
   54     -- ^ the requested path does not exist
   55   | PathInsecureMode FilePath FileMode FileMode
   56     -- ^ path, actual mode (masked to 0o777), expected mode (masked to 0o777)
   57   deriving stock (Show, Eq)
   58 
   59 -- | Paths that must never be readable or writable, regardless of workspace.
   60 -- Checked against the first component of the relative path within the workspace.
   61 blockedPaths :: Set String
   62 blockedPaths = Set.fromList
   63   [ ".env"
   64   , ".env.local"
   65   , ".env.production"
   66   , ".ssh"
   67   , ".gnupg"
   68   , ".netrc"
   69   , ".pureclaw"  -- protects vault and config from agent file tools
   70   ]

   72 -- | The ONLY way to obtain a 'SafePath'. Canonicalizes the path (following
   73 -- symlinks), verifies it stays within the workspace, checks the blocked list,
   74 -- and verifies the path exists.
   75 --
   76 -- Checks run in order, each as an early return:
   77 --  1. Reject @..@ traversal (prevents workspace escape)
   78 --  2. Reject absolute paths outside workspace
   79 --  3. Reject blocked paths (@.env@, @.ssh@, etc.)
   80 --  4. Reject non-existent paths
   81 --  5. Reject symlinks that resolve outside workspace
   82 mkSafePath :: WorkspaceRoot -> FilePath -> IO (Either PathError SafePath)
   83 mkSafePath (WorkspaceRoot root) requested = do
   84   canonRoot <- canonicalizePath root
   85   let raw = if isAbsolute requested then requested else canonRoot </> requested
   86       relative = makeRelative canonRoot raw
   87   -- Pure checks first (no IO, no filesystem access)
   88   let pureCheck
   89         | hasParentTraversal requested = Just (PathEscapesWorkspace requested raw)
   90         | isAbsolute requested && not (canonRoot `isPrefixOf` raw) = Just (PathEscapesWorkspace requested raw)
   91         | isBlockedPath relative = Just (PathIsBlocked requested "blocked path")
   92         | otherwise = Nothing
   93   case pureCheck of
   94     Just err -> pure (Left err)
   95     Nothing -> do
   96       exists <- doesPathExist raw
   97       if not exists
   98         then pure $ Left (PathDoesNotExist requested)
   99         else do
  100           -- Canonicalize resolves symlinks — a symlink inside the workspace
  101           -- could point outside it, so we re-check containment.
  102           canonical <- canonicalizePath raw
  103           pure $ if canonRoot `isPrefixOf` canonical
  104             then Right (SafePath canonical)
  105             else Left (PathEscapesWorkspace requested canonical)
  106 
  107 -- | Check if a path contains ".." components that could traverse upward.
  108 hasParentTraversal :: FilePath -> Bool
  109 hasParentTraversal path = ".." `elem` splitDirectories path
  110 
  111 -- | Check if a relative path matches any blocked path.
  112 -- Matches on the first path component (e.g. ".env" matches ".env" and ".env/foo").
  113 isBlockedPath :: FilePath -> Bool
  114 isBlockedPath relative =
  115   let firstComponent = Prelude.takeWhile (\c -> c /= '/' && c /= '\\') relative
  116   in Set.member firstComponent blockedPaths

  154 -- | The ONLY way to obtain a 'SafeKeyPath'.
  155 --
  156 -- Validation rules (match the rigor of 'mkSafePath'):
  157 --
  158 --  1. Reject @..@ traversal in the requested path.
  159 --  2. Canonicalize the keys root; reject if the root directory does not exist.
  160 --  3. Canonicalize the requested file (when it already exists) and verify
  161 --     the result is under the canonicalized keys root. This catches
  162 --     symlinks that point outside the root.
  163 --  4. When the file already exists, @fstat@ it and verify
  164 --     @mode `intersectFileModes` 0o777 == 0o400@ and
  165 --     @owner == geteuid()@; surface 'PathInsecureMode' otherwise.
  166 --
  167 -- Note: a 'SafeKeyPath' for a file that does not yet exist is permitted —
  168 -- it is the caller's responsibility (typically the ssh-key-write path) to
  169 -- create the file with mode 0400. Re-validating after write is the
  170 -- caller's job.
  171 mkSafeKeyPath :: KeysRoot -> FilePath -> IO (Either PathError SafeKeyPath)
  172 mkSafeKeyPath (KeysRoot root) requested =
  173   validateUnderRoot root requested SafeKeyPath (Just (0o400, expectedKeyMode))
  174   where
  175     expectedKeyMode :: FileMode
  176     expectedKeyMode = 0o400
  177 
  178 -- | The ONLY way to obtain a 'SafeRuntimePath'.
  179 --
  180 -- Same validation rules as 'mkSafeKeyPath' except no mode/owner check on
  181 -- the file — sockets are typically created by ssh itself and known_hosts
  182 -- mode is enforced elsewhere.
  183 mkSafeRuntimePath :: RuntimeRoot -> FilePath -> IO (Either PathError SafeRuntimePath)
  184 mkSafeRuntimePath (RuntimeRoot root) requested =
  185   validateUnderRoot root requested SafeRuntimePath Nothing
  186 
  187 -- | Shared validation pipeline for 'mkSafeKeyPath' and 'mkSafeRuntimePath'.
  188 -- Takes a wrapping constructor and an optional @(expectedMode, expectedMode)@
  189 -- pair. The pair carries the expected mode twice because the same value is
  190 -- both the value reported in 'PathInsecureMode' (the "expected" field) and
  191 -- the value we compare against — having a single source kept the call sites
  192 -- intentional.
  193 validateUnderRoot
  194   :: FilePath
  195   -> FilePath
  196   -> (FilePath -> a)
  197   -> Maybe (FileMode, FileMode)
  198   -> IO (Either PathError a)
  199 validateUnderRoot root requested wrap mModeCheck = do
  200   if hasParentTraversal requested
  201     then pure $ Left (PathEscapesWorkspace requested requested)
  202     else do
  203       rootExists <- doesDirectoryExist root
  204       if not rootExists
  205         then pure $ Left (PathDoesNotExist root)
  206         else do
  207           canonRoot <- canonicalizePath root
  208           let raw = if isAbsolute requested
  209                       then requested
  210                       else canonRoot </> requested
  211           fileExists <- doesPathExist raw
  212           if fileExists
  213             then do
  214               canonical <- canonicalizePath raw
  215               if canonRoot `isPrefixOf` canonical
  216                 then case mModeCheck of
  217                        Nothing -> pure $ Right (wrap canonical)
  218                        Just (expected, _) -> checkModeAndOwner canonical expected wrap
  219                 else pure $ Left (PathEscapesWorkspace requested canonical)
  220             else
  221               -- File does not exist yet (e.g. ssh has not written the key).
  222               -- Verify the *requested* location resolves under the root.
  223               if canonRoot `isPrefixOf` raw
  224                 then pure $ Right (wrap raw)
  225                 else pure $ Left (PathEscapesWorkspace requested raw)
  226 
  227 -- | Verify the file's mode (masked to the standard permission bits) and
  228 -- effective-uid owner match the expected mode. Returns 'PathInsecureMode'
  229 -- on mismatch.
  230 checkModeAndOwner
  231   :: FilePath
  232   -> FileMode
  233   -> (FilePath -> a)
  234   -> IO (Either PathError a)
  235 checkModeAndOwner canonical expectedMode wrap = do
  236   status <- PF.getFileStatus canonical
  237   let actualMode = PF.fileMode status `PF.intersectFileModes` 0o777
  238       ownerUid   = PF.fileOwner status
  239   euid <- PU.getEffectiveUserID
  240   if actualMode == expectedMode && ownerUid == euid
  241     then pure $ Right (wrap canonical)
  242     else pure $ Left (PathInsecureMode canonical actualMode expectedMode)

  244 -- | Create the keys-root directory if missing (mode 0700) and return a
  245 -- 'KeysRoot' wrapping it. Idempotent — if the directory already exists, the
  246 -- mode is re-applied. Callers wire this into agent startup before any
  247 -- 'mkSafeKeyPath' call.
  248 ensureKeysRoot :: FilePath -> IO KeysRoot
  249 ensureKeysRoot dir = do
  250   createDirectoryIfMissing True dir
  251   PF.setFileMode dir 0o700
  252   pure (KeysRoot dir)
