    1 -- | Broadcasting transcript decorator — WU2.
    2 --
    3 -- Wraps a 'TranscriptHandle' so that every '_th_record' call publishes
    4 -- @EntryRecorded@ and @ActivityChanged (SaEntryAt _)@ events to a
    5 -- 'StreamBroker' in addition to writing to the inner handle. This is the
    6 -- standard helper for every transcript write-path in PureClaw — direct use
    7 -- of 'mkFileTranscriptHandle' is forbidden outside the allowlist enforced
    8 -- by @scripts/lint-transcript-handles.sh@ (D6).
    9 --
   10 -- /AsyncCancelled discipline./ On 'AsyncCancelled' from the inner handle
   11 -- or the broker publish, the exception is re-raised so that bracket-style
   12 -- cleanup runs (project-wide invariant; see "PureClaw.Tab.Harness" and
   13 -- 'PureClaw.Transcript.Combinator.safeRecord' for the canonical pattern).
   14 -- Non-async exceptions from the inner handle are caught and logged via
   15 -- '_lh_logWarn'; the wrapper still publishes to the broker so subscribers
   16 -- remain consistent with the in-memory observation stream (D34 — the
   17 -- documented divergence from the inner's silent-drop semantics).
   18 --
   19 -- /Payload cap./ Before publish, the transcript entry is passed through
   20 -- 'capPayload' which truncates the payload text if its UTF-8 byte length
   21 -- exceeds the broker's configured @_bc_maxEventBytes@. Truncated entries
   22 -- carry a @"truncated": true@ flag in their metadata so subscribers can
   23 -- render an indicator. The on-disk record is /not/ truncated — only the
   24 -- broker-published view is.
   25 module PureClaw.Frontend.BroadcastingTranscript
   26   ( mkBroadcastingFileTranscriptHandle
   27   , mkBroadcastingTranscriptHandle
   28   ) where
   29

   52 -- | Open a new file-backed transcript handle that ALSO publishes to a
   53 -- broker. The standard helper used by every transcript write-path call
   54 -- site ('mkSessionHandle', 'resumeSession', 'handleSend'). When the broker
   55 -- is 'Nothing' no decorator is installed and the result is the bare
   56 -- 'mkFileTranscriptHandle' — the no-broker path is preserved for one-off
   57 -- scripts and tests.
   58 mkBroadcastingFileTranscriptHandle
   59   :: Maybe StreamBroker
   60   -> SessionId
   61   -> LogHandle
   62   -> FilePath
   63   -> IO TranscriptHandle
   64 mkBroadcastingFileTranscriptHandle mBroker sid logger path = case mBroker of
   65   Nothing     -> mkFileTranscriptHandle logger path
   66   Just broker -> do
   67     inner <- mkFileTranscriptHandle logger path
   68     pure (mkBroadcastingTranscriptHandle broker sid logger inner)
   69 
   70 -- | Wrap an existing 'TranscriptHandle' to publish broker events on
   71 -- record. See the module Haddock for the full semantics (AsyncCancelled
   72 -- discipline, disk-failure logging, payload capping).
   73 mkBroadcastingTranscriptHandle
   74   :: StreamBroker
   75   -> SessionId
   76   -> LogHandle
   77   -> TranscriptHandle
   78   -> TranscriptHandle
   79 mkBroadcastingTranscriptHandle broker sid logger inner = inner
   80   { _th_record = \entry -> do
   81       let cap           = _bc_maxEventBytes (_streamBroker_config broker)
   82           truncatedEntry = capPayload cap entry
   83       r <- try @SomeException (_th_record inner entry)
   84       case r of
   85         Right () -> publishSafely truncatedEntry
   86         Left e
   87           | Just AsyncCancelled <- fromException e -> throwIO e
   88           | otherwise -> do
   89               _lh_logWarn logger $
   90                 "transcript disk-write failed; in-memory broker still notified "
   91                   <> "(entry="    <> _te_id entry
   92                   <> ", session=" <> unSessionId sid
   93                   <> "): "        <> T.pack (show e)
   94               publishSafely truncatedEntry
   95   }
   96   where
   97     publishSafely e = do
   98       r <- try @SomeException $ do
   99         _streamBroker_publish broker (EntryRecorded   sid e)
  100         _streamBroker_publish broker
  101           (ActivityChanged sid (SaEntryAt (_te_timestamp e)))
  102       case r of
  103         Right () -> pure ()
  104         Left ex
  105           | Just AsyncCancelled <- fromException ex -> throwIO ex
  106           | otherwise -> _lh_logWarn logger
  107               ("broker publish failed: " <> T.pack (show ex))
  108 
  109 -- | Truncate a 'TranscriptEntry' if its payload's UTF-8 byte length exceeds
  110 -- the broker's per-event cap. On truncation, marks the entry's metadata
  111 -- with @"truncated" -> true@ so subscribers can render a "[truncated]"
  112 -- badge.
  113 --
  114 -- Per-event size capping is enforced at the broker boundary (this
  115 -- decorator) rather than inside the broker itself; see the design doc
  116 -- §Broker for the rationale.
  117 capPayload :: Int -> TranscriptEntry -> TranscriptEntry
  118 capPayload cap entry
  119   | BS.length payloadBytes <= cap = entry
  120   | otherwise = entry
  121       { _te_payload  = truncatedText
  122       , _te_metadata = Map.insert "truncated" (Aeson.Bool True)
  123                                   (_te_metadata entry)
  124       }
  125   where
  126     payloadBytes  = TE.encodeUtf8 (_te_payload entry)
  127     truncatedText = TE.decodeUtf8Lenient (BS.take cap payloadBytes)
