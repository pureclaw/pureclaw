    3 -- Description : Local subprocess backend factories (WU8).
    4 --
    5 -- == Overview
    6 --
    7 -- Two factories produce a real 'BackendHandle' over an actual local
    8 -- subprocess:
    9 --
   10 -- * 'mkLocalBackendHandle' — 'Pipe' kind. One-shot: stdin is written
   11 --   from '_po_stdinBytes' and then closed; stdout (with stderr merged
   12 --   into it) is read until EOF or until the recv-buffer cap is reached.
   13 -- * 'mkLocalPtyBackendHandle' — 'Pty' kind. Conversational: a real PTY
   14 --   is allocated via the supplied 'PtyIO'; the drainer + STM idle state
   15 --   machine from "PureClaw.Backend.Pty" runs the recv loop.
   16 --
   17 -- == Why 'PtyIO' is a separate parameter
   18 --
   19 -- WU1 deliberately omitted a @_pto_io :: PtyIO@ field from 'PtyOpts' to
   20 -- keep "PureClaw.Handles.Backend" free of any dependency on
   21 -- "PureClaw.Backend.Pty" (the @posix-pty@ firewall lives there). WU7
   22 -- chose not to retrofit, so 'mkLocalPtyBackendHandle' takes the
   23 -- 'PtyIO' as a separate parameter rather than as a field on
   24 -- 'PtyOpts'. Mirrors 'mkDrainerBackendHandle'\'s shape.

   71 -- Pipe-kind factory
   72 --------------------------------------------------------------------------------
   73 
   74 -- | Construct a 'Pipe'-kind local subprocess backend.
   75 --
   76 -- Semantics:
   77 --
   78 -- * stdin is fed @_po_stdinBytes@ and then closed (via
   79 --   'P.byteStringInput'); after construction, @_bh_send@ is a silent
   80 --   no-op because the OS-level stdin is already gone.
   81 -- * stdout AND stderr are wired to the SAME pipe write-end at the OS
   82 --   level via 'System.Posix.IO.createPipe' + 'P.useHandleOpen' on both
   83 --   stream slots. The child sees fd 1 and fd 2 pointing at the same
   84 --   pipe; byte order is preserved end-to-end (this is the OS-level
   85 --   @2>&1@ pattern; see design doc § "stderr Handling").
   86 -- * @_bh_recv@ reads from the accumulator until EOF or until the
   87 --   per-backend cap @_po_recvBufferCap@ is reached. EOF within the
   88 --   cap yields 'RecvSettled'; cap reached first yields
   89 --   'RecvTruncated'.
   90 -- * @_bh_resize@ is a silent no-op (no PTY exists on a 'Pipe' backend).
   91 -- * @_bh_close@ waits for the subprocess to exit and releases the
   92 --   process-wide buffer quota; idempotent.
   93 --
   94 -- On construction failure (quota oversubscription, spawn exception)
   95 -- the quota is released and the error is propagated as a 'Left'.
   96 mkLocalBackendHandle
   97   :: AuthorizedCommand
   98   -> PipeOpts
   99   -> IO (Either BackendError BackendHandle)
  100 mkLocalBackendHandle cmd opts = do
  101   let capBytes = _po_recvBufferCap opts
  102       capMiB   = bytesToMiBCeil capBytes
  103   quotaPtr <- readIORef globalBackendBufferQuota
  104   acq <- acquireBufferQuota quotaPtr capMiB
  105   case acq of
  106     Left e   -> pure (Left e)
  107     Right () -> do
  108       releasedRef <- newIORef False
  109       let releaseQuotaOnce = do
  110             already <- atomicSwapTrue releasedRef
  111             if already
  112               then pure ()
  113               else releaseBufferQuota quotaPtr capMiB
  114 
  115       let prog = getCommandProgram cmd
  116           args = map T.unpack (getCommandArgs cmd)
  117           stdinLazy = byteStringToLazy (_po_stdinBytes opts)
  118           envList   = envMapWithDefaults (_po_env opts)
  119 
  120       -- Create a single shared pipe. Both stdout (fd 1) and stderr (fd 2)
  121       -- in the child are dup'd from the same write-end; the parent reads
  122       -- from the read-end. This is the OS-level @2>&1@ pattern — byte
  123       -- order is preserved by the kernel.
  124       pipeResult <- try @SomeException PosixIO.createPipe
  125       case pipeResult of
  126         Left _ -> do
  127           releaseQuotaOnce
  128           pure (Left (BackendBinaryNotFound (toCommandName prog)))
  129         Right (readFd, writeFd) -> do
  130           readH  <- PosixIO.fdToHandle readFd
  131           writeH <- PosixIO.fdToHandle writeFd
  132           let baseConfig =
  133                   P.setStdin (P.byteStringInput stdinLazy)
  134                 $ P.setStdout (P.useHandleOpen writeH)
  135                 $ P.setStderr (P.useHandleOpen writeH)
  136                 $ P.setEnv envList
  137                 $ P.proc prog args
  138               withCwd = maybe id P.setWorkingDir (_po_cwd opts)
  139               config = withCwd baseConfig
  140 
  141           spawnResult <- try @SomeException (P.startProcess config)
  142           case spawnResult of
  143             Left _ -> do
  144               -- Close both ends of our pipe and release quota.
  145               _ <- try @SomeException (hClose writeH)
  146               _ <- try @SomeException (hClose readH)
  147               releaseQuotaOnce
  148               pure (Left (BackendBinaryNotFound (toCommandName prog)))
  149             Right proc -> do
  150               -- After spawn, the parent doesn't need its copy of the
  151               -- write-end; close it so the reader sees EOF when the
  152               -- child exits and closes its copies. Critical for EOF
  153               -- detection.
  154               _ <- try @SomeException (hClose writeH)
  155 
  156               accRef    <- newIORef BS.empty
  157               truncRef  <- newIORef False
  158               reader    <- Async.async (drainHandleInto capBytes accRef truncRef readH)
  159 
  160               closeLock     <- newMVar ()
  161               closedRef     <- newIORef False
  162               recvDoneRef   <- newIORef False
  163 
  164               let doSend _ = pure ()
  165 
  166                   doRecv _mIdle = do
  167                     already <- readIORef recvDoneRef
  168                     if already
  169                       then do
  170                         bs <- readIORef accRef
  171                         isT <- readIORef truncRef
  172                         pure (if isT then RecvTruncated bs else RecvSettled bs)
  173                       else do
  174                         _ <- Async.waitCatch reader
  175                         writeIORef recvDoneRef True
  176                         bs  <- readIORef accRef
  177                         isT <- readIORef truncRef
  178                         pure (if isT then RecvTruncated bs else RecvSettled bs)
  179 
  180                   doResize _ _ = pure ()
  181 
  182                   doClose = modifyMVar_ closeLock $ \() -> do
  183                     wasClosed <- readIORef closedRef
  184                     if wasClosed
  185                       then pure ()
  186                       else do
  187                         writeIORef closedRef True
  188                         Async.cancel reader
  189                         _ <- try @SomeException (P.stopProcess proc)
  190                         _ <- try @SomeException (hClose readH)
  191                         releaseQuotaOnce
  192                         pure ()
  193 
  194               guarded <- withConcurrentUseGuard BackendHandle
  195                 { _bh_name        = "local-pipe"
  196                 , _bh_kind        = Pipe
  197                 , _bh_defaultIdle = localIdle
  198                 , _bh_send        = doSend
  199                 , _bh_recv        = doRecv
  200                 , _bh_resize      = doResize
  201                 , _bh_close       = doClose
  202                 }
  203               pure (Right guarded)
  204 
  205 -- | Drain a 'Handle' (stdout or stderr pipe) into the shared
  206 -- accumulator, latching @truncRef@ if the per-backend cap is exceeded.
  207 --
  208 -- Best-effort: read exceptions are treated as EOF. Once the cap is
  209 -- reached, the latch is flipped to 'True' and subsequent bytes are
  210 -- dropped (so the accumulator stops growing past the cap).
  211 drainHandleInto :: Int -> IORef ByteString -> IORef Bool -> Handle -> IO ()
  212 drainHandleInto cap accRef truncRef h = go
  213   where
  214     go = do
  215       r <- try @SomeException (BS.hGetSome h 4096)
  216       case r of
  217         Left _    -> pure ()
  218         Right bs
  219           | BS.null bs -> pure ()
  220           | otherwise -> do
  221               acc <- readIORef accRef
  222               let newSize = BS.length acc + BS.length bs
  223               if newSize > cap
  224                 then writeIORef truncRef True
  225                 else do
  226                   atomicModifyIORef' accRef $ \old -> (old <> bs, ())
  227                   go

  230 -- Pty-kind factory
  231 --------------------------------------------------------------------------------

  233 -- | Construct a 'Pty'-kind local subprocess backend.
  234 --
  235 -- Composes the 'AuthorizedCommand' + 'PtyOpts' into a 'PtyOpenSpec' and
  236 -- delegates to 'mkDrainerBackendHandle'. The drainer + STM idle state
  237 -- machine from "PureClaw.Backend.Pty" provides @_bh_recv@, @_bh_send@
  238 -- forwards to @_pio_write@, and @_bh_close@ is idempotent (cancels
  239 -- drainer, closes PTY, releases buffer quota).
  240 --
  241 -- The environment passed to the child is 'envMapWithDefaults' of
  242 -- @_pto_env@ — i.e. the caller-supplied map plus minimal defaults for
  243 -- 'TERM' (@xterm-256color@) and 'PATH' (@\/usr\/bin:\/bin:\/usr\/local\/bin@)
  244 -- only when those keys are absent. No inheritance from the parent
  245 -- process.
  246 mkLocalPtyBackendHandle
  247   :: PtyIO
  248   -> AuthorizedCommand
  249   -> PtyOpts
  250   -> IO (Either BackendError BackendHandle)
  251 mkLocalPtyBackendHandle pio cmd opts = do
  252   let spec = PtyOpenSpec
  253         { _pos_program = getCommandProgram cmd
  254         , _pos_args    = getCommandArgs cmd
  255         , _pos_env     = envMapWithBackendDefaults (_pto_env opts)
  256         , _pos_cwd     = _pto_cwd opts
  257         , _pos_cols    = _pto_cols opts
  258         , _pos_rows    = _pto_rows opts
  259         }
  260   r <- mkDrainerBackendHandle pio opts spec Pty
  261   case r of
  262     Left  e  -> pure (Left e)
  263     Right bh -> Right <$> withConcurrentUseGuard bh

  266 -- Env helpers
  267 --------------------------------------------------------------------------------

  269 -- | Add minimal 'TERM' and 'PATH' defaults to a caller-supplied 'EnvMap'
  270 -- if either is absent. Returns the result in the @[(String, String)]@
  271 -- shape that @typed-process@\'s 'P.setEnv' expects.
  272 --
  273 -- Behaviour:
  274 --
  275 -- * If @TERM@ is missing, set it to @xterm-256color@.
  276 -- * If @PATH@ is missing, set it to @\/usr\/bin:\/bin:\/usr\/local\/bin@.
  277 -- * Caller-supplied entries always win — existing keys are not
  278 --   overwritten.
  279 envMapWithDefaults :: EnvMap -> [(String, String)]
  280 envMapWithDefaults = envToTuples . envMapWithBackendDefaults
  281 
  282 -- | The 'EnvMap'-side of 'envMapWithDefaults' — used by the PTY factory
  283 -- which feeds its 'EnvMap' through 'PtyOpenSpec' rather than the
  284 -- @typed-process@ tuple list.
  285 envMapWithBackendDefaults :: EnvMap -> EnvMap
  286 envMapWithBackendDefaults =
  287     insertIfAbsent "TERM" "xterm-256color"
  288   . insertIfAbsent "PATH" "/usr/bin:/bin:/usr/local/bin"
  289   where
  290     insertIfAbsent k v m
  291       | Map.member k m = m
  292       | otherwise      = Map.insert k (EnvValue (stringToByteString v)) m
  293 
  294 -- | Render an 'EnvMap' as the @[(String, String)]@ tuple list that
  295 -- @typed-process@\'s 'P.setEnv' expects. Values are decoded
  296 -- byte-for-byte from the underlying 'ByteString' so non-ASCII bytes
  297 -- round-trip 1:1.
  298 envToTuples :: EnvMap -> [(String, String)]
  299 envToTuples = map render . Map.toAscList
  300   where
  301     render (k, EnvValue v) = (k, byteStringToString v)
  302 
  303 -- | Latin-1 round-trip ('ByteString' → 'String'). The subprocess sees
  304 -- raw bytes; this is the same transform used in
  305 -- "PureClaw.Backend.Pty".
  306 byteStringToString :: ByteString -> String
  307 byteStringToString bs = map (toEnum . fromIntegral) (BS.unpack bs)
  308 
  309 -- | Latin-1 round-trip ('String' → 'ByteString'). Inverse of
  310 -- 'byteStringToString'.
  311 stringToByteString :: String -> ByteString
  312 stringToByteString = BS.pack . map (fromIntegral . fromEnum)
  313 
  314 -- | Lift a strict 'ByteString' into the lazy 'ByteString' shape that
  315 -- 'P.byteStringInput' expects.
  316 byteStringToLazy :: ByteString -> BL.ByteString
  317 byteStringToLazy = BL.fromStrict

  323 -- | Round a byte count up to whole MiB (1 MiB == 1_048_576 bytes).
  324 -- Zero or negative byte counts produce @0@. Mirrors the helper in
  325 -- "PureClaw.Backend.Pty"; reimplemented locally to keep that module\'s
  326 -- export surface tight.
  327 bytesToMiBCeil :: Int -> Int
  328 bytesToMiBCeil n
  329   | n <= 0    = 0
  330   | otherwise = (n + mib - 1) `div` mib
  331   where mib = 1024 * 1024
  332 
  333 -- | Atomically set an 'IORef' 'Bool' to 'True', returning its previous
  334 -- value. Used to defend against double-release of the buffer quota.
  335 -- Mirrors the helper in "PureClaw.Backend.Pty".
  336 atomicSwapTrue :: IORef Bool -> IO Bool
  337 atomicSwapTrue ref = atomicModifyIORef' ref (True,)
