   22 -- | Create a web_extract tool that fetches a URL and converts the
   23 -- response to readable markdown. Handles HTML→markdown conversion,
   24 -- response size limits, and content-type detection.
   25 webExtractTool :: AllowList Text -> NetworkHandle -> (ToolDefinition, ToolHandler)
   26 webExtractTool allowList nh = (def, handler)
   27   where
   28     def = ToolDefinition
   29       { _td_name        = "web_extract"
   30       , _td_description = T.unlines
   31           [ "Fetch a URL and extract its content as readable markdown."
   32           , "HTML pages are converted to markdown with headings, links, and lists preserved."
   33           , "Non-HTML responses (JSON, plain text) are returned as-is."
   34           , "Large responses are truncated to 100K characters."
   35           ]
   36       , _td_inputSchema = object
   37           [ "type" .= ("object" :: Text)
   38           , "properties" .= object
   39               [ "url" .= object
   40                   [ "type" .= ("string" :: Text)
   41                   , "description" .= ("The URL to fetch and extract content from" :: Text)
   42                   ]
   43               , "max_length" .= object
   44                   [ "type" .= ("integer" :: Text)
   45                   , "description" .= ("Maximum response length in characters (default: 100000)" :: Text)
   46                   ]
   47               ]
   48           , "required" .= (["url"] :: [Text])
   49           ]
   50       }
   51 
   52     handler = ToolHandler $ \input ->
   53       case parseEither parseInput input of
   54         Left err -> pure (T.pack err, True)
   55         Right (url, maxLen) ->
   56           case mkAllowedUrl allowList url of
   57             Left (UrlNotAllowed u) -> pure ("URL domain not allowed: " <> u, True)
   58             Left (UrlMalformed u) -> pure ("Malformed URL: " <> u, True)
   59             Right allowed -> do
   60               result <- try @SomeException (_nh_httpGet nh allowed)
   61               case result of
   62                 Left e -> pure ("web_extract error: " <> T.pack (show e), True)
   63                 Right resp ->
   64                   let status = _hr_statusCode resp
   65                       body = T.pack (BS8.unpack (_hr_body resp))
   66                       limit = fromMaybe 100000 maxLen
   67                   in if status >= 400
   68                     then pure ("HTTP " <> T.pack (show status), True)
   69                     else
   70                       let converted = if looksLikeHtml body
   71                             then htmlToMarkdown body
   72                             else body
   73                           truncated = truncateBody limit converted
   74                       in pure (truncated, False)
   75 
   76     parseInput :: Value -> Parser (Text, Maybe Int)
   77     parseInput = withObject "WebExtractInput" $ \o ->
   78       (,) <$> o .: "url" <*> o .:? "max_length"

   80 -- | Check if the response body looks like HTML.
   81 looksLikeHtml :: Text -> Bool
   82 looksLikeHtml t =
   83   let lower = T.toLower (T.take 500 t)
   84   in T.isInfixOf "<html" lower
   85   || T.isInfixOf "<!doctype" lower
   86   || T.isInfixOf "<head" lower
   87   || T.isInfixOf "<body" lower

   89 -- | Convert HTML to readable markdown-like text.
   90 -- This is a simple tag-aware converter, not a full HTML parser.
   91 -- Handles: headings, paragraphs, links, lists, code blocks, emphasis.
   92 htmlToMarkdown :: Text -> Text
   93 htmlToMarkdown html =
   94   let -- Remove script and style blocks first
   95       noScript = removeBlocks "script" (removeBlocks "style" (removeBlocks "noscript" html))
   96       -- Process the remaining HTML
   97       processed = processHtml noScript
   98       -- Clean up excessive whitespace
   99       cleaned = collapseBlankLines (T.strip processed)
  100   in cleaned
  101 
  102 -- | Remove everything between <tag>...</tag> including the tags.
  103 removeBlocks :: Text -> Text -> Text
  104 removeBlocks tag = go
  105   where
  106     openTag = "<" <> tag
  107     closeTag = "</" <> tag <> ">"
  108 
  109     go remaining
  110       | T.null remaining = remaining
  111       | otherwise =
  112           let (before, afterOpen) = T.breakOn openTag (T.toLower remaining)
  113               -- Use the original casing for the "before" portion
  114               origBefore = T.take (T.length before) remaining
  115           in if T.null afterOpen
  116             then remaining
  117             else
  118               let rest = T.drop (T.length before) remaining
  119                   -- Find the closing tag in the rest
  120                   (_, afterClose) = T.breakOn closeTag (T.toLower rest)
  121                   origAfterClose = T.drop (T.length (T.take (T.length (T.toLower rest) - T.length afterClose) (T.toLower rest))) rest
  122               in if T.null afterClose
  123                 then remaining  -- malformed, return as-is
  124                 else
  125                   let skipClose = T.drop (T.length closeTag) origAfterClose
  126                   in origBefore <> go skipClose
  127 
  128 -- | Process HTML tags into markdown.
  129 processHtml :: Text -> Text
  130 processHtml = go ""
  131   where
  132     go acc remaining
  133       | T.null remaining = acc
  134       | "<" `T.isPrefixOf` remaining =
  135           let (tag, afterTag) = parseTag remaining
  136               (converted, rest) = handleTag tag afterTag
  137           in go (acc <> converted) rest
  138       | otherwise =
  139           let (text, rest) = T.break (== '<') remaining
  140               decoded = decodeEntities text
  141           in go (acc <> decoded) rest
  142 
  143     parseTag :: Text -> (Text, Text)
  144     parseTag t =
  145       let withoutLt = T.drop 1 t  -- drop '<'
  146           (tagContent, rest) = T.break (== '>') withoutLt
  147       in (T.strip tagContent, T.drop 1 rest)  -- drop '>'
  148 
  149     handleTag :: Text -> Text -> (Text, Text)
  150     handleTag tag rest
  151       | tagIs "h1" tag  = ("\n\n# ", rest)
  152       | tagIs "/h1" tag = ("\n\n", rest)
  153       | tagIs "h2" tag  = ("\n\n## ", rest)
  154       | tagIs "/h2" tag = ("\n\n", rest)
  155       | tagIs "h3" tag  = ("\n\n### ", rest)
  156       | tagIs "/h3" tag = ("\n\n", rest)
  157       | tagIs "h4" tag  = ("\n\n#### ", rest)
  158       | tagIs "/h4" tag = ("\n\n", rest)
  159       | tagIs "h5" tag  = ("\n\n##### ", rest)
  160       | tagIs "/h5" tag = ("\n\n", rest)
  161       | tagIs "h6" tag  = ("\n\n###### ", rest)
  162       | tagIs "/h6" tag = ("\n\n", rest)
  163       | tagIs "p" tag   = ("\n\n", rest)
  164       | tagIs "/p" tag  = ("\n\n", rest)
  165       | tagIs "br" tag || tagIs "br/" tag || tagIs "br /" tag = ("\n", rest)
  166       | tagIs "li" tag  = ("\n- ", rest)
  167       | tagIs "/li" tag = ("", rest)
  168       | tagIs "ul" tag || tagIs "ol" tag = ("\n", rest)
  169       | tagIs "/ul" tag || tagIs "/ol" tag = ("\n", rest)
  170       | tagIs "strong" tag || tagIs "b" tag = ("**", rest)
  171       | tagIs "/strong" tag || tagIs "/b" tag = ("**", rest)
  172       | tagIs "em" tag || tagIs "i" tag = ("*", rest)
  173       | tagIs "/em" tag || tagIs "/i" tag = ("*", rest)
  174       | tagIs "code" tag = ("`", rest)
  175       | tagIs "/code" tag = ("`", rest)
  176       | tagIs "pre" tag = ("\n```\n", rest)
  177       | tagIs "/pre" tag = ("\n```\n", rest)
  178       | tagIs "blockquote" tag = ("\n> ", rest)
  179       | tagIs "/blockquote" tag = ("\n", rest)
  180       | tagIs "hr" tag || tagIs "hr/" tag = ("\n---\n", rest)
  181       | tagIs "a" tag =
  182           let href = extractAttr "href" tag
  183           in case href of
  184             Just _href -> ("[", rest)
  185                 -- Simple approach: emit the link text in brackets.
  186                 -- For a full implementation, we'd buffer until </a> and append the href.
  187             Nothing -> ("", rest)
  188       | tagIs "/a" tag = ("", rest)
  189       | tagIs "div" tag = ("\n", rest)
  190       | tagIs "/div" tag = ("\n", rest)
  191       | tagIs "td" tag = (" | ", rest)
  192       | tagIs "th" tag = (" | ", rest)
  193       | tagIs "tr" tag = ("\n", rest)
  194       | tagIs "/tr" tag = (" |", rest)
  195       | otherwise = ("", rest)  -- skip unknown tags
  196 
  197     tagIs :: Text -> Text -> Bool
  198     tagIs expected full =
  199       let lower = T.toLower full
  200           name = T.takeWhile (\c -> c /= ' ' && c /= '\t') lower
  201       in name == expected
  202 
  203     extractAttr :: Text -> Text -> Maybe Text
  204     extractAttr attr tag =
  205       let lower = T.toLower tag
  206       in case T.breakOn (attr <> "=\"") lower of
  207         (_, after)
  208           | T.null after -> Nothing
  209           | otherwise ->
  210               let rest = T.drop (T.length attr + 2) after
  211                   value = T.takeWhile (/= '"') rest
  212               in Just value
  213 
  214 -- | Decode common HTML entities.
  215 decodeEntities :: Text -> Text
  216 decodeEntities = T.replace "&amp;" "&"
  217                . T.replace "&lt;" "<"
  218                . T.replace "&gt;" ">"
  219                . T.replace "&quot;" "\""
  220                . T.replace "&#39;" "'"
  221                . T.replace "&apos;" "'"
  222                . T.replace "&nbsp;" " "
  223                . T.replace "&#x27;" "'"
  224                . T.replace "&#x2F;" "/"
  225 
  226 -- | Collapse runs of more than 2 consecutive blank lines.
  227 collapseBlankLines :: Text -> Text
  228 collapseBlankLines = T.intercalate "\n\n" . filter (not . T.null) . map T.strip . T.splitOn "\n\n\n"
  229 
  230 -- | Truncate body to a maximum length, adding a marker if truncated.
  231 truncateBody :: Int -> Text -> Text
  232 truncateBody maxLen t
  233   | T.length t <= maxLen = t
  234   | otherwise = T.take maxLen t <> "\n\n[...truncated at " <> T.pack (show maxLen) <> " chars]"
