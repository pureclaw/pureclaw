    1 -- | Durable harness identity registry (Harness Registry & Lifecycle, Phase 1).
    2 --
    3 -- This is the source of truth for harness identity and health. A harness is
    4 -- keyed by a PureClaw-assigned 'HarnessId' (a UUID), NOT by its mutable tmux
    5 -- window name (which the user can rename) — see @docs\/harness-registry.md@ §3.
    6 --
    7 -- The registry itself is a 'TVar' (STM) because a background reconcile loop
    8 -- performs compound read-modify-write concurrently with HTTP\/slash handlers
    9 -- (design decision K2b). The critical invariant is that 'mergeReconcile' merges
   10 -- tmux-observed fields into existing entries /by key/ inside a single
   11 -- 'atomically', recomputing from the CURRENT 'TVar' contents — so an entry
   12 -- inserted by another transaction between a naive read and write is never
   13 -- clobbered (the lost-update-safe path, §4).
   14 --
   15 -- Leaf module: it may import 'PureClaw.Handles.Harness' (for 'HarnessHandle')
   16 -- and standard libraries only — never @Agent@\/@Frontend@\/@CLI@.
   17 module PureClaw.Harness.Registry

   19     HarnessId
   20   , newHarnessId
   21   , parseHarnessId

   23     -- * Entry types

   25   , Liveness (..)
   26   , HarnessEntry (..)
   27     -- * The registry
   28   , HarnessRegistry (..)
   29   , newRegistry
   30     -- * CRUD

   54 -- ---------------------------------------------------------------------------
   55 -- Identity
   56 -- ---------------------------------------------------------------------------
   57 
   58 -- | A PureClaw-assigned, UUID-backed harness identity. This is the canonical
   59 -- key for the registry and the durable anchor that survives tmux window
   60 -- rename\/move and PureClaw restart (persisted in @session.json@).
   61 newtype HarnessId = HarnessId { unHarnessId :: UUID }
   62   deriving stock (Eq, Ord, Show)
   63 
   64 -- | Generate a fresh random 'HarnessId' (UUID v4).
   65 newHarnessId :: IO HarnessId
   66 newHarnessId = HarnessId <$> UUIDv4.nextRandom
   67 
   68 -- | Parse a 'HarnessId' from its canonical UUID text representation.
   69 -- Returns 'Nothing' for any non-UUID input.
   70 parseHarnessId :: Text -> Maybe HarnessId
   71 parseHarnessId = fmap HarnessId . UUID.fromText
   72 
   73 -- | Render a 'HarnessId' as its canonical UUID text representation.
   74 harnessIdToText :: HarnessId -> Text
   75 harnessIdToText = UUID.toText . unHarnessId
   76 
   77 -- | Round-trippable JSON: a 'HarnessId' is encoded as the canonical UUID
   78 -- string (D2.4). We hand-write the codec rather than @deriving newtype@ so the
   79 -- on-the-wire shape (a plain string) is explicit and decode rejects malformed
   80 -- UUIDs with a clear error.
   81 instance ToJSON HarnessId where
   82   toJSON = Aeson.String . harnessIdToText
   83 
   84 instance FromJSON HarnessId where
   85   parseJSON = Aeson.withText "HarnessId" $ \t ->
   86     case parseHarnessId t of
   87       Just hid -> pure hid
   88       Nothing  -> fail ("invalid HarnessId (not a UUID): " <> show t)
   89 
   90 -- ---------------------------------------------------------------------------
   91 -- Entry types
   92 -- ---------------------------------------------------------------------------
   93 
   94 -- | How PureClaw came to manage this harness.
   95 --
   96 -- Phase 1 produces 'OriginSpawned' (we launched it) and 'OriginDiscovered'
   97 -- (boot-reconstructed from a window carrying our @\@pcl_id@). 'OriginAdopted'
   98 -- (an external window we took over) is defined here but exercised in Phase 3.
   99 data HarnessOrigin
  100   = OriginSpawned
  101   | OriginDiscovered
  102   | OriginAdopted
  103   deriving stock (Eq, Show)
  104 
  105 -- | Liveness of a harness (design §5 / decision K7).
  106 --
  107 -- Note that @ExternallyModified@ and @Unknown@ are intentionally NOT liveness
  108 -- states: the former is the orthogonal '_he_extModified' flag, and the latter
  109 -- maps to the '_he_stale' flag (hold last-known liveness).
  110 data Liveness
  111   = LivenessIdle      -- ^ Window present, harness running, not actively working.
  112   | LivenessThinking  -- ^ Harness actively working (screen-capture heuristic).
  113   | LivenessExited    -- ^ Harness PID gone or @pane_dead@, window still present.
  114   | LivenessOrphaned  -- ^ No live window+PID for this id.
  115   deriving stock (Eq, Show)
  116 
  117 -- | A registry entry: the durable identity plus the reconciled health\/coordinate
  118 -- cache (design §4). The cached @(session, windowName)@ coordinate lets handles
  119 -- target tmux without a per-I\/O @\@pcl_id@ sweep; the reconcile loop is the sole
  120 -- owner that refreshes it.
  121 --
  122 -- Note: '_he_handle' is a 'HarnessHandle', which is a record of @IO@ actions and
  123 -- therefore has no 'Eq'\/'Show'. Consequently 'HarnessEntry' does NOT derive
  124 -- 'Eq'\/'Show'; tests assert via field projections instead.
  125 data HarnessEntry = HarnessEntry
  126   { _he_id          :: !HarnessId
  127     -- ^ Canonical identity (the map key).
  128   , _he_session     :: !Text
  129     -- ^ Cached tmux session name (coordinate; revalidated by reconcile, K3).
  130   , _he_windowName  :: !Text
  131     -- ^ Cached tmux window name (coordinate; revalidated by reconcile, K3).
  132   , _he_shellPid    :: !(Maybe Int)
  133     -- ^ Recorded pane shell PID (@#{pane_pid}@) — provenance + liveness.
  134   , _he_harnessPid  :: !(Maybe Int)
  135     -- ^ Recorded harness-process PID (the agent binary) — trust anchor.
  136   , _he_origin      :: !HarnessOrigin
  137     -- ^ How PureClaw came to manage this harness.
  138   , _he_liveness    :: !Liveness
  139     -- ^ Current reconciled liveness.
  140   , _he_extModified :: !Bool
  141     -- ^ Orthogonal flag: a matched window's name\/session changed out-of-band.
  142   , _he_stale       :: !Bool
  143     -- ^ The last sweep\/capture failed; hold last-known liveness, don't repaint.
  144   , _he_sessionId   :: !(Maybe Text)
  145     -- ^ The PureClaw session this harness backs (the join to @session.json@).
  146   , _he_label       :: !Text
  147     -- ^ Legacy window-name label, used for name->id resolution ('lookupByLabel').
  148   , _he_orphanedTicks :: !Int
  149     -- ^ Consecutive reconcile ticks this entry has been classified Orphaned.
  150     --   0 whenever the entry is live (Idle\/Thinking\/Exited); the reconcile
  151     --   loop increments it each tick the corroborated window stays absent and
  152     --   auto-evicts the entry once it reaches 'PureClaw.Harness.Reconcile.defaultOrphanGraceTicks'
  153     --   (the wall-clock-free grace policy, design §5\/§10 Q2). Eviction drops
  154     --   the entry from the registry + the legacy harness map but never touches
  155     --   @session.json@ (the session reappears in Recent Sessions).
  156   , _he_handle      :: !(Maybe HarnessHandle)
  157     -- ^ The live handle, if we have one. 'Nothing' for a boot-discovered entry
  158     --   that has not yet had a handle attached.
  159   }
  160 
  161 -- ---------------------------------------------------------------------------
  162 -- The registry
  163 -- ---------------------------------------------------------------------------
  164 
  165 -- | The durable harness registry: a strict 'Map' from 'HarnessId' to
  166 -- 'HarnessEntry', held in a 'TVar' for safe concurrent read-modify-write.
  167 newtype HarnessRegistry = HarnessRegistry
  168   { unHarnessRegistry :: TVar (Map HarnessId HarnessEntry) }
  169 
  170 -- | Create a new, empty registry.
  171 newRegistry :: IO HarnessRegistry
  172 newRegistry = HarnessRegistry <$> newTVarIO Map.empty
  173 
  174 -- ---------------------------------------------------------------------------
  175 -- CRUD
  176 -- ---------------------------------------------------------------------------
  177 
  178 -- | Insert (or overwrite) an entry by its 'HarnessId'.
  179 insertEntry :: HarnessRegistry -> HarnessEntry -> IO ()
  180 insertEntry reg e =
  181   atomically (modifyTVar' (unHarnessRegistry reg) (Map.insert (_he_id e) e))
  182 
  183 -- | Look up an entry by its canonical id.
  184 lookupById :: HarnessRegistry -> HarnessId -> IO (Maybe HarnessEntry)
  185 lookupById reg hid =
  186   atomically (Map.lookup hid <$> readTVar (unHarnessRegistry reg))
  187 
  188 -- | Resolve an entry by its label (legacy window-name). Returns the first
  189 -- match (label is not guaranteed unique on a shared server; the canonical key
  190 -- is the id). Used by name->id resolution for legacy\/CLI routing surfaces.
  191 lookupByLabel :: HarnessRegistry -> Text -> IO (Maybe HarnessEntry)
  192 lookupByLabel reg lbl = do
  193   m <- readTVarIO (unHarnessRegistry reg)
  194   pure (firstMatch (Map.elems m))
  195   where
  196     firstMatch []       = Nothing
  197     firstMatch (e : es)
  198       | _he_label e == lbl = Just e
  199       | otherwise          = firstMatch es
  200 
  201 -- | Remove an entry by id (no-op if absent).
  202 deleteEntry :: HarnessRegistry -> HarnessId -> IO ()
  203 deleteEntry reg hid =
  204   atomically (modifyTVar' (unHarnessRegistry reg) (Map.delete hid))
  205 
  206 -- | Apply a pure function to the entry with the given id, in-place, as a
  207 -- SINGLE-STM read-modify-write (no-op if the id is absent). This is the
  208 -- race-safe way to flip a flag on one entry (e.g. clearing '_he_extModified'
  209 -- for Acknowledge): unlike a 'lookupById' followed by 'insertEntry' — two
  210 -- separate transactions — the whole adjust commits atomically, so a concurrent
  211 -- 'mergeReconcile' tick cannot interleave and clobber the change (a lost
  212 -- update). 'Map.adjust' touches only the named key, leaving every other entry
  213 -- intact.
  214 modifyEntry :: HarnessRegistry -> HarnessId -> (HarnessEntry -> HarnessEntry) -> IO ()
  215 modifyEntry reg hid f =
  216   atomically (modifyTVar' (unHarnessRegistry reg) (Map.adjust f hid))
  217 
  218 -- | Snapshot all current entries.
  219 snapshot :: HarnessRegistry -> IO [HarnessEntry]
  220 snapshot reg =
  221   atomically (Map.elems <$> readTVar (unHarnessRegistry reg))
  222 
  223 -- ---------------------------------------------------------------------------
  224 -- Reconciliation
  225 -- ---------------------------------------------------------------------------
  226 
  227 -- | The tmux-observed fields the reconcile loop merges into an existing entry,
  228 -- keyed by 'HarnessId'. This is deliberately a small projection of
  229 -- 'HarnessEntry' — it carries only what a server sweep observes, never the
  230 -- handle or the durable provenance/origin, which the registry owns.
  231 data ObservedHarness = ObservedHarness
  232   { _oh_id          :: !HarnessId
  233   , _oh_session     :: !Text
  234   , _oh_windowName  :: !Text
  235   , _oh_shellPid    :: !(Maybe Int)
  236   , _oh_harnessPid  :: !(Maybe Int)
  237   , _oh_liveness    :: !Liveness
  238   , _oh_extModified :: !Bool
  239   , _oh_stale       :: !Bool
  240   , _oh_orphanedTicks :: !Int
  241     -- ^ The entry's new consecutive-orphaned-tick count for this observation:
  242     --   @old + 1@ when classified Orphaned, @0@ when live. Rides the
  243     --   'mergeReconcile' path so the counter lives durably on the entry.
  244   }
  245   deriving stock (Eq, Show)
  246 
  247 -- | Merge tmux-observed fields into existing entries BY KEY, inside a single
  248 -- 'atomically'. This is the lost-update-safe reconcile path (design §4):
  249 --
  250 --   * The new map is computed from the CURRENT 'TVar' contents inside the
  251 --     transaction, so a concurrent insert that landed before this transaction
  252 --     commits is preserved.
  253 --   * Only keys present in the observed set are updated; every other key
  254 --     (including newly-inserted ones) is left intact.
  255 --   * An observed row whose id is NOT already in the registry is ignored — an
  256 --     observation alone never resurrects\/creates an entry (registration is the
  257 --     caller's job; this defends against, e.g., a spoofed marker).
  258 mergeReconcile :: HarnessRegistry -> [ObservedHarness] -> IO ()
  259 mergeReconcile reg observed =
  260   atomically (modifyTVar' (unHarnessRegistry reg) merge)
  261   where
  262     obsMap :: Map HarnessId ObservedHarness
  263     obsMap = Map.fromList [ (_oh_id o, o) | o <- observed ]
  264 
  265     -- Map over the CURRENT contents, updating only keys we observed and
  266     -- leaving all others (incl. concurrently-inserted) untouched.
  267     merge :: Map HarnessId HarnessEntry -> Map HarnessId HarnessEntry
  268     merge = Map.mapWithKey applyObserved
  269 
  270     applyObserved :: HarnessId -> HarnessEntry -> HarnessEntry
  271     applyObserved hid e =
  272       case Map.lookup hid obsMap of
  273         Nothing -> e
  274         Just o  -> e
  275           { _he_session     = _oh_session o
  276           , _he_windowName  = _oh_windowName o
  277           , _he_shellPid    = _oh_shellPid o
  278           , _he_harnessPid  = _oh_harnessPid o
  279           , _he_liveness    = _oh_liveness o
  280           , _he_extModified = _oh_extModified o
  281           , _he_stale       = _oh_stale o
  282           , _he_orphanedTicks = _oh_orphanedTicks o
  283           }
