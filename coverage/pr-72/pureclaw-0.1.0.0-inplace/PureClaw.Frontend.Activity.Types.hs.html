    1 -- | Shared activity types used by the WS-streaming broker and the HTTP API.
    2 --
    3 -- Originally these types lived in "PureClaw.Frontend.API"; they were promoted
    4 -- to this module so "PureClaw.Frontend.StreamBroker" can reference
    5 -- 'HarnessActivity' without taking a dependency on the WAI/API layer.
    6 --
    7 -- "PureClaw.Frontend.API" re-exports 'HarnessActivity' for compatibility so
    8 -- its public surface is unchanged.
    9 module PureClaw.Frontend.Activity.Types
   10   ( HarnessActivity (..)
   11   ) where
   12 
   13 import Data.Aeson qualified as Aeson
   14 import Data.Aeson (ToJSON (..))
   15 
   16 -- | Activity state of a harness, derived from tmux screen capture.
   17 data HarnessActivity
   18   = HarnessThinking
   19   | HarnessIdle
   20   | HarnessStopped
   21   deriving stock (Show, Eq)
   22 
   23 instance ToJSON HarnessActivity where
   24   toJSON HarnessThinking = Aeson.String "thinking"
   25   toJSON HarnessIdle     = Aeson.String "idle"
   26   toJSON HarnessStopped  = Aeson.String "stopped"
