   26 -- | Result of a subprocess execution.
   27 data ProcessResult = ProcessResult
   28   { _pr_exitCode :: ExitCode
   29   , _pr_stdout   :: ByteString
   30   , _pr_stderr   :: ByteString
   31   }
   32   deriving stock (Show, Eq)
   33 
   34 -- | Options controlling how a command is executed. Designed to be
   35 -- extended as new backends (Docker, SSH, etc.) are added — each
   36 -- backend interprets the fields it supports and ignores the rest.
   37 data ExecOptions = ExecOptions
   38   { _eo_timeout    :: Maybe Int
   39     -- ^ Timeout in milliseconds. 'Nothing' means no timeout.
   40   , _eo_workingDir :: Maybe FilePath
   41     -- ^ Working directory for the subprocess. 'Nothing' uses the
   42     -- agent's current directory.
   43   }
   44   deriving stock (Show, Eq)
   45 
   46 -- | Default execution options: no timeout, inherit working directory.
   47 defaultExecOptions :: ExecOptions
   48 defaultExecOptions = ExecOptions
   49   { _eo_timeout    = Nothing
   50   , _eo_workingDir = Nothing
   51   }

   53 -- | Subprocess execution capability. Only accepts 'AuthorizedCommand',
   54 -- which is proof that the command passed security policy evaluation.
   55 --
   56 -- The real implementation strips the subprocess environment to prevent
   57 -- secret leakage via inherited environment variables.
   58 --
   59 -- Backends (local, Docker, SSH, etc.) provide different implementations
   60 -- of this record. All backends receive 'ExecOptions' and interpret the
   61 -- fields they support.
   62 newtype ShellHandle = ShellHandle
   63   { _sh_execute :: ExecOptions -> AuthorizedCommand -> IO ProcessResult
   64   }
   65 
   66 -- | Minimal safe environment for subprocesses. Provides only PATH so
   67 -- commands can be resolved, but inherits nothing else from the parent.
   68 safeEnv :: [(String, String)]
   69 safeEnv = [("PATH", "/usr/bin:/bin:/usr/local/bin")]
   70 
   71 -- | Real shell handle using @typed-process@. Strips the subprocess
   72 -- environment (provides only a minimal PATH) as noted in the architecture.
   73 mkShellHandle :: LogHandle -> ShellHandle
   74 mkShellHandle logger = ShellHandle
   75   { _sh_execute = \opts cmd -> do
   76       let prog = getCommandProgram cmd
   77           args = map T.unpack (getCommandArgs cmd)
   78           config = maybe id P.setWorkingDir (_eo_workingDir opts)
   79                  $ P.setEnv safeEnv
   80                  $ P.proc prog args
   81       _lh_logInfo logger $ "Executing: " <> T.pack prog <> " " <> T.unwords (getCommandArgs cmd)
   82       case _eo_timeout opts of
   83         Nothing -> do
   84           (exitCode, outLazy, errLazy) <- P.readProcess config
   85           pure ProcessResult
   86             { _pr_exitCode = exitCode
   87             , _pr_stdout   = BL.toStrict outLazy
   88             , _pr_stderr   = BL.toStrict errLazy
   89             }
   90         Just ms -> runWithTimeout config ms
   91   }
   92 
   93 -- | Run a process with a timeout. Launches the process via
   94 -- 'readProcess' on a background thread, then races it against a
   95 -- timer. On timeout the process is killed via 'Async.cancel' and a
   96 -- synthetic exit-124 result is returned.
   97 --
   98 -- This avoids the "waitForProcess: No child processes" error that
   99 -- occurs when 'System.Timeout.timeout' wrapping 'readProcess' races
  100 -- with the child reaper on Linux — 'Async.cancel' delivers a clean
  101 -- 'AsyncCancelled' that typed-process handles correctly.
  102 runWithTimeout :: P.ProcessConfig () () () -> Int -> IO ProcessResult
  103 runWithTimeout config ms = do
  104   worker <- Async.async $ P.readProcess config
  105   result <- Timeout.timeout (ms * 1000) (Async.wait worker)
  106   case result of
  107     Just (exitCode, outLazy, errLazy) ->
  108       pure ProcessResult
  109         { _pr_exitCode = exitCode
  110         , _pr_stdout   = BL.toStrict outLazy
  111         , _pr_stderr   = BL.toStrict errLazy
  112         }
  113     Nothing -> do
  114       Async.cancel worker
  115       pure ProcessResult
  116         { _pr_exitCode = ExitFailure 124
  117         , _pr_stdout   = ""
  118         , _pr_stderr   = TE.encodeUtf8 ("Command timed out after " <> T.pack (show ms) <> "ms")
  119         }

  121 -- | No-op shell handle. Returns success with empty output.
  122 mkNoOpShellHandle :: ShellHandle
  123 mkNoOpShellHandle = ShellHandle
  124   { _sh_execute = \_ _ -> pure ProcessResult
  125       { _pr_exitCode = ExitSuccess
  126       , _pr_stdout   = ""
  127       , _pr_stderr   = ""
  128       }
