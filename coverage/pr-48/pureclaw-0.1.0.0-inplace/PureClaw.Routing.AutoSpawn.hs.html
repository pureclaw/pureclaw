    1 -- |
    2 -- Module      : PureClaw.Routing.AutoSpawn
    3 -- Description : Auto-spawn UX, close, resume, rename, crashed handling
    4 --               (Tabbed Chat WU9, A\/L\/X\/B\/S6\/S10 series).
    5 --
    6 -- The user-visible UX surface for tab creation, close, resume, rename,
    7 -- and crashed-tab handling. The dispatcher (WU5) routes parsed slash
    8 -- commands here; this module owns the @\/N@ switch UX (A-series), the
    9 -- close lifecycle on top of the per-kind factory '_tabHandle_close'
   10 -- (L-series), the crashed-tab retry prompt (X-series), and the rename
   11 -- handler (S10).
   12 --
   13 -- == @\/N@ switch UX (A-series, tmux-style packing model)
   14 --
   15 -- 'handleSwitch' implements the simple truth table:
   16 --
   17 --   * tab present → focus + recap (and Crashed observation, X1).
   18 --   * tab missing → emit an error banner; do NOT auto-spawn.
   19 --
   20 -- The previous "auto-spawn on missing /N" behaviour was retired in
   21 -- favour of the tmux packing model: tabs always occupy the lowest
   22 -- slots @\/0..K-1@, so @\/N@ with @N >= K@ is always user error.
   23 --
   24 -- == 'handleDefault' first-run UX (K3)
   25 --
   26 -- The first-run case (no tab focused, registry empty) still
   27 -- implicit-spawns '_rc_defaultKind' at index 0 and forwards the typed
   28 -- text. This is what makes a brand-new user with no tabs able to just
   29 -- type a message and start talking — the K3 first-run path is the only
   30 -- surviving implicit-spawn surface after the tmux packing refactor.
   31 --
   32 -- == @\/tab new@ — always-lowest allocation
   33 --
   34 -- @\/tab new@ no longer accepts a user-supplied index: spawns always
   35 -- land at the lowest free slot via 'Registry.lowestFreeIndex'. The
   36 -- no-kind variant force-prompts the user; the with-kind variant spawns
   37 -- directly.
   38 --
   39 -- == Max-tab cap (S6)
   40 --
   41 -- Every spawn path consults '_rc_maxTabs' BEFORE calling the factory;
   42 -- exceeding the cap surfaces 'TabLimitExceeded' as a dispatcher
   43 -- 'PublicError'. The cap check lives in the dispatcher's 'spawnTabWith'
   44 -- helper (WU5) so AutoSpawn does not duplicate the bookkeeping; we
   45 -- merely surface the redacted error to the user.
   46 --
   47 -- == Crashed UX (X-series)
   48 --
   49 -- Original spawn args are retained per-tab in a @Map Int SpawnArgs@
   50 -- carried by the dispatcher's 'DispatcherState' so the X2\/X3 retry
   51 -- path can replay the exact original args (per design OQ-6: original
   52 -- args, not current defaults).
   53 --
   54 -- See @docs\/tabbed-chat.md@ §"Auto-spawn behavior (A-series)",
   55 -- §"Lifecycle \/ close (L-series)", §"Crashed tab UX (X-series)",
   56 -- §"Dashboard (B-series)", and §"Security (S-series)" S6 \/ S10.

  152 -- ---------------------------------------------------------------------------
  153 -- handleSwitch — /N (tmux packing model)
  154 -- ---------------------------------------------------------------------------
  155 
  156 -- | Handle @\/N@ (Switch). Truth table (tmux packing model):
  157 --
  158 -- * tab present → focus, emit recap, observe Crashed status (X1).
  159 -- * tab missing → emit an error banner ('missingTabBanner'); no
  160 --   auto-spawn.
  161 --
  162 -- The previous "auto-spawn on missing /N" behaviour was retired with
  163 -- the tmux packing refactor: because tabs are always packed in the
  164 -- lowest slots, @\/N@ for any @N@ outside the open range is always
  165 -- user error. The K3 first-run UX (no focus, registry empty, plain
  166 -- text) still implicit-spawns via 'handleDefault'.
  167 handleSwitch
  168   :: AgentEnv
  169   -> BannerEmit
  170   -> TabIndex
  171   -> IO ()
  172 handleSwitch env emit idx = do
  173   mTab <- Registry.lookupTab (_env_tabs env) idx
  174   case mTab of
  175     Just h  -> focusExisting env emit idx h
  176     Nothing -> emit (missingTabBanner idx)
  177 
  178 -- | Banner emitted when @\/N@ references a tab that does not exist.
  179 --
  180 -- Phrased to nudge the user toward the (always-allocating) @\/tab new@
  181 -- command rather than the old "auto-spawn at the requested index"
  182 -- behaviour, which no longer exists.
  183 missingTabBanner :: TabIndex -> Text
  184 missingTabBanner idx =
  185   "/" <> tShowIdx idx
  186     <> ": no such tab — use /tab new to create one"

  189 -- ---------------------------------------------------------------------------
  190 -- handleDefault — Default text input with empty focus (L6 + K3)
  191 -- ---------------------------------------------------------------------------
  192 
  193 -- | Handle the case where the channel layer delivered a plain-text
  194 -- 'PureClaw.Routing.Types.Default' input AND '_env_focus' is 'Nothing'
  195 -- (no tab focused).
  196 --
  197 -- L6 \/ K3 specifies: implicitly spawn '_rc_defaultKind' at the lowest
  198 -- free index (typically 0), then enqueue the text on the new tab.
  199 --
  200 -- If the spawn fails (e.g. rate-limit, max-tabs hit), surface the
  201 -- redacted public error via the banner emitter.
  202 handleDefault
  203   :: AgentEnv
  204   -> SpawnIO
  205   -> BannerEmit
  206   -> IORef (Map Int SpawnArgs)
  207   -> Text
  208   -> IO ()
  209 handleDefault env spawnIO emit argsRef text = do
  210   let rc = _env_routingConfig env
  211       kind = _rc_defaultKind rc
  212   r <- spawnIO kind []
  213   case r of
  214     Left e ->
  215       emit ("/0: " <> unPublicTabError (toPublicTabError e))
  216     Right newIdx -> do
  217       rememberArgs argsRef newIdx kind []
  218       writeIORef (_env_focus env) (Just newIdx)
  219       enqueuePayloadOn env emit newIdx text

  222 -- ---------------------------------------------------------------------------
  223 -- handleNew — /tab new [<kind> [<args>]]
  224 -- ---------------------------------------------------------------------------
  225 
  226 -- | Handle @\/tab new [\<kind\> [\<arg-text\>]]@ (tmux packing model).
  227 --
  228 -- Truth table:
  229 --
  230 -- * @\/tab new@ (no kind)            → force-prompt at the next free
  231 --   slot.
  232 -- * @\/tab new \<kind\> [\<args\>]@  → spawn at the lowest free slot
  233 --   via 'SpawnIO' (the dispatcher's rate-limited factory), focus it,
  234 --   and emit a confirmation banner.
  235 -- * Spawn failure (rate-limit, cap, factory failure, etc.) → redacted
  236 --   'PublicError' banner.
  237 --
  238 -- The previous "user picks the slot" behaviour is gone: new tabs are
  239 -- always allocated at the lowest free index (matches tmux
  240 -- @renumber-windows on@ semantics). The slot for the force-prompt
  241 -- banner is determined by 'Registry.lowestFreeIndex' so the prompt
  242 -- text shows the user where the spawn would land.
  243 handleNew
  244   :: AgentEnv
  245   -> PromptRenderer
  246   -> SpawnIO
  247   -> BannerEmit
  248   -> IORef (Map Int SpawnArgs)
  249   -> Maybe TabKindArg  -- ^ kind keyword, if supplied
  250   -> Maybe Text        -- ^ arg text, if supplied
  251   -> IO ()
  252 handleNew env renderer spawnIO emit argsRef mKind mArgs = do
  253   let rc = _env_routingConfig env
  254   case mKind of
  255     Nothing -> do
  256       -- Force-prompt: surface the index where the spawn would land so
  257       -- the prompt UX has a target to render. If the cap is full, we
  258       -- emit a redacted PublicError instead of a prompt for a slot
  259       -- that doesn't exist.
  260       mFree <- Registry.lowestFreeIndex (_env_tabs env) (_rc_maxTabs rc)
  261       case mFree of
  262         Nothing ->
  263           emit ("/tab new: "
  264                 <> unPublicTabError
  265                      (toPublicTabError (TabLimitExceeded (_rc_maxTabs rc))))
  266         Just idx ->
  267           emit (_pr_renderSpawnPrompt renderer idx Nothing)
  268     Just kindArg -> do
  269       let kind = tabKindArgToKind kindArg
  270           args = splitArgs mArgs
  271       r <- spawnIO kind args
  272       case r of
  273         Left e ->
  274           -- We don't have a known target index at this point — the
  275           -- spawn never assigned one. Emit a generic /tab new banner.
  276           emit ("/tab new: "
  277                 <> unPublicTabError (toPublicTabError e))
  278         Right newIdx -> do
  279           rememberArgs argsRef newIdx kind args
  280           writeIORef (_env_focus env) (Just newIdx)
  281           emit ("/" <> tShowIdx newIdx <> ": spawned ("
  282                 <> kindKeyword kind <> ")")

  285 -- ---------------------------------------------------------------------------
  286 -- handleClose — /tab close N [--force] (tmux-style renumber)
  287 -- ---------------------------------------------------------------------------
  288 
  289 -- | The post-close renumber callback. The dispatcher passes in a
  290 -- closure that shifts its side-maps (spawn-args, pending-retry, etc.)
  291 -- down by one starting at @closedIdx + 1@. AutoSpawn calls this AFTER
  292 -- it has driven the registry-side renumber so the side maps stay in
  293 -- sync with the registry's view of slot ownership.
  294 type RenumberCallback = Int -> IO ()
  295 
  296 -- | Handle @\/tab close N [--force]@ with tmux-style packing.
  297 --
  298 -- Truth table:
  299 --
  300 -- * N missing  → 'TabNotFound' (L5).
  301 -- * N present  → '_tabHandle_close' (graceful or force);
  302 --   remove the registry entry; shift every remaining tab at index
  303 --   @\> N@ down by one (tmux @renumber-windows on@ model); drop the
  304 --   SpawnArgs entry for @N@ then shift remaining SpawnArgs keys;
  305 --   reconcile '_env_focus' (was N → cleared; was \> N → decremented).
  306 --
  307 -- The renumber pass is done under a single 'atomicModifyIORef'' for
  308 -- '_env_tabs' and '_env_runners' so a concurrent reader (the
  309 -- dispatcher is the only writer per E3) always observes a consistent
  310 -- contiguous slot layout. The 'RenumberCallback' lets the caller
  311 -- (Dispatcher) apply the same shift to side maps that AutoSpawn does
  312 -- not own.
  313 --
  314 -- The 'TabHandle._tabHandle_index' field of any remaining handle is
  315 -- /not/ rewritten — it reflects the creation index and is treated as
  316 -- advisory after a renumber. The authoritative current slot is the
  317 -- registry key.
  318 handleClose
  319   :: AgentEnv
  320   -> BannerEmit
  321   -> IORef (Map Int SpawnArgs)
  322   -> RenumberCallback  -- ^ side-map shift (dispatcher passes the
  323                        --   callback that shifts pendingRetry etc.)
  324   -> Int               -- ^ requested index
  325   -> Bool              -- ^ True = --force
  326   -> IO ()
  327 handleClose env emit argsRef renumber rawIdx force =
  328   case mkTabIndex rawIdx of
  329     Nothing  ->
  330       emit ("/" <> T.pack (show rawIdx) <> ": tab: invalid index")
  331     Just idx -> do
  332       removed <- Registry.removeTab (_env_tabs env) idx
  333       case removed of
  334         Nothing -> emit ("/" <> tShowIdx idx <> ": "
  335                           <> unPublicTabError
  336                                (toPublicTabError (TabNotFound rawIdx)))
  337         Just h  -> do
  338           safeIgnore (_tabHandle_close h closeMode)
  339           -- tmux packing: shift every remaining tab > closedIdx down
  340           -- by one so the registry stays contiguous starting at 0.
  341           let closedN = unTabIndex idx
  342           atomicModifyIORef' (_env_tabs env) $ \m ->
  343             (Registry.packAfterRemove closedN m, ())
  344           atomicModifyIORef' (_env_runners env) $ \m ->
  345             (Registry.packAfterRemove closedN m, ())
  346           -- Side-map shifts: drop the spawn-args for the closed slot,
  347           -- then shift remaining keys down. The dispatcher's
  348           -- 'RenumberCallback' handles its own pending-retry map and
  349           -- any other dispatcher-owned side state.
  350           atomicModifyIORef' argsRef $ \m ->
  351             ( shiftMapKeysAfter closedN (Map.delete closedN m), () )
  352           renumber closedN
  353           -- Reconcile focus: closed tab → Nothing; tabs above → -1.
  354           updateFocusOnClose env idx
  355           emit ("/" <> tShowIdx idx <> ": closed")
  356   where
  357     closeMode = if force then CloseForce else CloseGraceful
  358 
  359 -- | After a close (with tmux-style renumber), reconcile '_env_focus':
  360 --
  361 --   * focused tab was the closed one → clear focus (the dispatcher's
  362 --     K3 implicit-spawn path will pick up the next 'Default' input);
  363 --   * focused tab was strictly greater than the closed one →
  364 --     decrement the focus index by one (it just got renumbered);
  365 --   * focused tab was strictly less than the closed one → unchanged.
  366 updateFocusOnClose :: AgentEnv -> TabIndex -> IO ()
  367 updateFocusOnClose env closedIdx = do
  368   mFocus <- readIORef (_env_focus env)
  369   case mFocus of
  370     Nothing -> pure ()
  371     Just f
  372       | f == closedIdx ->
  373           writeIORef (_env_focus env) Nothing
  374       | unTabIndex f > unTabIndex closedIdx ->
  375           -- The focused tab was renumbered down by one.
  376           writeIORef (_env_focus env) (mkTabIndex (unTabIndex f - 1))
  377       | otherwise -> pure ()
  378 
  379 -- | Shift the keys of a 'Data.Map.Strict.Map' (Int-keyed) down by one
  380 -- for every key strictly greater than @k@. Mirrors
  381 -- 'Registry.packAfterRemove' for 'IntMap' but works on the generic
  382 -- 'Map' that the dispatcher uses for spawn-args and pending-retry side
  383 -- state.
  384 shiftMapKeysAfter :: Int -> Map Int v -> Map Int v
  385 shiftMapKeysAfter k = Map.foldlWithKey'
  386   (\acc i v ->
  387      if i > k
  388        then Map.insert (i - 1) v acc
  389        else Map.insert i v acc)
  390   Map.empty

  413 -- handleRename — /tab rename N <name> (S10)
  414 -- ---------------------------------------------------------------------------
  415 
  416 -- | Handle @\/tab rename N \<name\>@. The user-supplied name passes
  417 -- through the caller-supplied sanitizer (typically
  418 -- 'PureClaw.Routing.Parse.sanitizeTabName'); on success the new name
  419 -- is announced; on failure a redacted public error is emitted.
  420 --
  421 -- /WU9 scope note:/ rename in v1 is observable-only at the registry
  422 -- level — the new name is reported back to the user; persisting the
  423 -- renamed label requires mutating the 'TabHandle' record (currently
  424 -- sealed). For v1 we surface the rename intent and the sanitization
  425 -- outcome; the in-place name swap lands in a v1.5 refinement that
  426 -- replaces the 'TabHandle' field with a mutable 'IORef TabName'.
  427 handleRename
  428   :: AgentEnv
  429   -> BannerEmit
  430   -> (Text -> Either NameError Text)  -- ^ sanitizer
  431   -> Int       -- ^ requested index
  432   -> Text      -- ^ requested new name (raw)
  433   -> IO ()
  434 handleRename env emit sanitize rawIdx rawName =
  435   case mkTabIndex rawIdx of
  436     Nothing  ->
  437       emit ("/" <> T.pack (show rawIdx) <> ": tab: invalid index")
  438     Just idx -> do
  439       mTab <- Registry.lookupTab (_env_tabs env) idx
  440       case mTab of
  441         Nothing ->
  442           emit ("/" <> tShowIdx idx <> ": "
  443                 <> unPublicTabError
  444                      (toPublicTabError (TabNotFound rawIdx)))
  445         Just h -> case sanitize rawName of
  446           Left  nameErr ->
  447             emit ("/" <> tShowIdx idx <> ": "
  448                   <> unPublicTabError
  449                        (toPublicTabError (TabInvalidName nameErr)))
  450           Right safeName -> do
  451             let oldName = unTabName (_tabHandle_name h)
  452                 redactedSuffix =
  453                   if safeName == rawName
  454                     then ""
  455                     else " (redacted host/path fragment)"
  456             emit ("Renamed /" <> tShowIdx idx <> " "
  457                    <> oldName <> " to \"" <> safeName <> "\""
  458                    <> redactedSuffix)

  508 -- | Focus an existing tab + emit the recap and (when applicable) the
  509 -- Crashed banner.
  510 focusExisting :: AgentEnv -> BannerEmit -> TabIndex -> TabHandle -> IO ()
  511 focusExisting env emit idx h = do
  512   writeIORef (_env_focus env) (Just idx)
  513   st <- safeStatus h
  514   case st of
  515     Just (Crashed pe) ->
  516       -- X1: surface Crashed status as a one-line PublicError + retry
  517       -- \/ close prompt.
  518       emit ("/" <> tShowIdx idx <> " crashed: "
  519              <> unPublicTabError pe
  520              <> " — [1] retry [2] close")
  521     _ -> do
  522       let recap = recapText (_env_routingConfig env) idx
  523       emit ("/" <> tShowIdx idx <> ": focused")
  524       if T.null recap then pure () else emit recap
  525 
  526 -- | Construct the (WU9-minimal) recap banner. WU10's
  527 -- 'PureClaw.Agent.Loop' refactor will replace this with a transcript
  528 -- replay; for v1 we surface the per-tab recap budget so the user
  529 -- knows what the budget would have been.
  530 recapText :: RoutingConfig -> TabIndex -> Text
  531 recapText rc idx =
  532   let n     = _rc_switchRecap rc
  533       nTxt  = T.pack (show n)
  534   in if n <= 0
  535        then ""
  536        else "(recap: last " <> nTxt <> " messages from /"
  537               <> tShowIdx idx <> ")"

  539 -- | Enqueue a payload on a tab via '_tabHandle_send', surfacing send
  540 -- failures as a redacted dispatcher banner.
  541 enqueuePayloadOn :: AgentEnv -> BannerEmit -> TabIndex -> Text -> IO ()
  542 enqueuePayloadOn env emit idx payload = do
  543   mTab <- Registry.lookupTab (_env_tabs env) idx
  544   case mTab of
  545     Nothing ->
  546       emit ("/" <> tShowIdx idx <> ": "
  547             <> unPublicTabError
  548                  (toPublicTabError (TabNotFound (unTabIndex idx))))
  549     Just h -> do
  550       result <- safeEnqueue h payload
  551       case result of
  552         Right ()  -> pure ()
  553         Left  e   -> emit ("/" <> tShowIdx idx <> ": "
  554                            <> unPublicTabError (toPublicTabError e))
  555 
  556 -- | Tolerant '_tabHandle_send' wrapper: any synchronous exception
  557 -- collapses to a redacted 'TabConcurrencyLimit'.
  558 safeEnqueue :: TabHandle -> Text -> IO (Either TabError ())
  559 safeEnqueue h payload = do
  560   r <- try @SomeException (_tabHandle_send h payload)
  561   pure $ case r of
  562     Left  _e    -> Left (TabConcurrencyLimit 0)
  563     Right inner -> inner
  564 
  565 -- | Read '_tabHandle_status' tolerantly. A buggy factory could throw;
  566 -- in that case we collapse the result to 'Nothing' (treated as
  567 -- "unknown status" — no Crashed banner emitted).
  568 safeStatus :: TabHandle -> IO (Maybe TabStatus)
  569 safeStatus h = do
  570   r <- try @SomeException (_tabHandle_status h)
  571   pure $ case r of
  572     Left _  -> Nothing
  573     Right s -> Just s
  574 
  575 -- | Best-effort: run an IO action and swallow synchronous failures.
  576 safeIgnore :: IO () -> IO ()
  577 safeIgnore m = do
  578   _ <- try @SomeException m
  579   pure ()

  600 -- | Split the @[args-text]@ field (a single 'Text' parsed greedily
  601 -- after the kind keyword) into a list of arguments by whitespace.
  602 splitArgs :: Maybe Text -> [Text]
  603 splitArgs Nothing  = []
  604 splitArgs (Just t) = T.words t

  606 -- | Format a 'TabIndex' as the user-facing decimal string.
  607 tShowIdx :: TabIndex -> Text
  608 tShowIdx = T.pack . show . unTabIndex

  610 -- | Record the spawn args for a given tab so X2\/X3 retry can replay.
  611 rememberArgs :: IORef (Map Int SpawnArgs) -> TabIndex -> TabKind -> [Text] -> IO ()
  612 rememberArgs ref idx kind args =
  613   modifyIORef' ref
  614     (Map.insert (unTabIndex idx) SpawnArgs { _sa_kind = kind, _sa_args = args })

  620 -- on every successful spawn.
  621 rememberArgsForTest :: IORef (Map Int SpawnArgs) -> TabIndex -> TabKind -> [Text] -> IO ()
  622 rememberArgsForTest = rememberArgs
