  127 -- | WAI application handling @\/api\/*@ routes.
  128 apiApp :: FrontendEnv -> Application
  129 apiApp env req respond = do
  130   let method = requestMethod req
  131       path   = pathInfo req
  132   case (method, path) of
  133     ("GET", ["api", "harnesses"])            -> handleHarnesses env respond
  134     ("GET", ["api", "sessions", "recent"])   -> handleRecentSessions env respond
  135     ("GET", ["api", "sessions", sid, "transcript"]) ->
  136       handleTranscript env sid respond
  137     ("POST", ["api", "sessions", "new"]) ->
  138       handleNewSession env req respond
  139     ("POST", ["api", "sessions", sid, "send"]) ->
  140       handleSend env sid req respond
  141     ("PUT", ["api", "sessions", sid, "prompt"]) ->
  142       handleSetPrompt env sid req respond
  143     ("GET", ["api", "agents"])               -> handleAgents env respond
  144     _                                        -> respondNotFound respond

  146 handleHarnesses :: FrontendEnv -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  147 handleHarnesses env respond = do
  148   harnesses <- readIORef (_fe_harnesses env)
  149   infos <- traverse probeHarness (Map.toList harnesses)
  150   respond $ jsonResponse status200 infos

  152 -- | Probe a single harness to determine its activity.
  153 -- Checks process status first, then does a quick screen capture
  154 -- to distinguish thinking from idle.
  155 probeHarness :: (Text, HarnessHandle) -> IO HarnessInfo
  156 probeHarness (name, hh) = do
  157   st <- _hh_status hh
  158   activity <- case st of
  159     HarnessExited _ -> pure HarnessStopped
  160     HarnessRunning  -> probeActivity (_hh_session hh) name
  161   pure HarnessInfo
  162     { _hi_name     = name
  163     , _hi_activity = activity
  164     }
  165 
  166 -- | Capture the tmux window and check if the harness is idle or thinking.
  167 probeActivity :: Text -> Text -> IO HarnessActivity
  168 probeActivity tmuxSession windowName = do
  169   let target = tmuxSession <> ":" <> extractWindowIdx windowName
  170   result <- try @SomeException $ captureWindow target 50
  171   case result of
  172     Left _        -> pure HarnessIdle
  173     Right capture -> do
  174       let screenText = TE.decodeUtf8Lenient capture
  175       pure $ if isIdle screenText then HarnessIdle else HarnessThinking
  176 
  177 -- | Extract the window index suffix from a harness name like "claude-code-0".
  178 extractWindowIdx :: Text -> Text
  179 extractWindowIdx name =
  180   case T.splitOn "-" name of
  181     parts | length parts >= 2 -> last parts
  182     _                         -> "0"

  184 handleRecentSessions :: FrontendEnv -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  185 handleRecentSessions env respond = do
  186   metas <- listSessions (_fe_sessionsDir env) Nothing (_fe_recentLimit env)
  187   nonEmpty <- filterM (hasTranscriptEntries (_fe_sessionsDir env)) metas
  188   let infos = map toSessionInfo nonEmpty
  189   respond $ jsonResponse status200 infos

  191 -- | Check whether a session has at least one transcript entry.
  192 hasTranscriptEntries :: FilePath -> SessionMeta -> IO Bool
  193 hasTranscriptEntries baseDir meta = do
  194   let path = baseDir </> T.unpack (unSessionId (_sm_id meta)) </> "transcript.jsonl"
  195   exists <- doesFileExist path
  196   if not exists
  197     then pure False
  198     else do
  199       size <- getFileSize path
  200       pure (size > 0)

  249 -- | Read transcript entries from a session's @transcript.jsonl@ file.
  250 handleTranscript :: FrontendEnv -> Text -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  251 handleTranscript env sid respond = do
  252   -- Reject path traversal
  253   if T.isInfixOf ".." sid || T.isInfixOf "/" sid
  254     then respond $ jsonResponse status400 (object ["error" .= ("Invalid session ID" :: Text)])
  255     else do
  256       let path = _fe_sessionsDir env </> T.unpack sid </> "transcript.jsonl"
  257       exists <- doesFileExist path
  258       if not exists
  259         then respond $ jsonResponse status404 (object ["error" .= ("Session not found" :: Text)])
  260         else do
  261           result <- try @SomeException $ readTranscriptFile path
  262           case result of
  263             Left _ ->
  264               respond $ jsonResponse status500 (object ["error" .= ("Failed to read transcript" :: Text)])
  265             Right entries ->
  266               respond $ jsonResponse status200 (map toTranscriptEntryInfo entries)
  267 
  268 -- | Read and parse a JSONL transcript file.
  269 readTranscriptFile :: FilePath -> IO [TranscriptEntry]
  270 readTranscriptFile path = do
  271   contents <- LBS.readFile path
  272   let linesBS = filter (not . LBS.null) (LBS.split 0x0A contents)
  273   pure [e | l <- linesBS, Just e <- [Aeson.decode l]]

  299 -- | Create a new empty session.
  300 handleNewSession :: FrontendEnv -> Request -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  301 handleNewSession env req respond = do
  302   body <- consumeBody req
  303   let parsed = case Aeson.eitherDecode body of
  304         Right r -> r
  305         Left _  -> NewSessionRequest Nothing Nothing
  306       mAgentText = _nsr_agent parsed
  307       mAgentName = mAgentText >>= either (const Nothing) Just . mkAgentName
  308       mPrefix = mAgentText >>= either (const Nothing) Just . mkSessionPrefix
  309   now <- getCurrentTime
  310   mModel <- readIORef (_fe_model env)
  311   let modelText = maybe "" unModelId mModel
  312       sid = newSessionId mPrefix now
  313       meta = SessionMeta
  314         { _sm_id                = sid
  315         , _sm_agent             = mAgentName
  316         , _sm_runtime           = RTProvider
  317         , _sm_model             = modelText
  318         , _sm_channel           = "web"
  319         , _sm_createdAt         = now
  320         , _sm_lastActive        = now
  321         , _sm_bootstrapConsumed = True
  322         }
  323   sh <- mkSessionHandle (_fe_logger env) (_fe_sessionsDir env) meta
  324   -- Write custom prompt file if provided
  325   case _nsr_customPrompt parsed of
  326     Just prompt | not (T.null (T.strip prompt)) ->
  327       TIO.writeFile (_sh_dir sh </> "custom-prompt.md") prompt
  328     _ -> pure ()
  329   _sh_save sh
  330   respond $ jsonResponse status200 (toSessionInfo meta)

  332 -- | Set or replace the custom prompt for a session, optionally updating
  333 -- the agent name in @session.json@.
  334 handleSetPrompt :: FrontendEnv -> Text -> Request -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  335 handleSetPrompt env sid req respond = do
  336   if T.isInfixOf ".." sid || T.isInfixOf "/" sid
  337     then respond $ jsonResponse status400 (object ["error" .= ("Invalid session ID" :: Text)])
  338     else do
  339       let sessionDir = _fe_sessionsDir env </> T.unpack sid
  340           promptPath = sessionDir </> "custom-prompt.md"
  341           metaPath   = sessionDir </> "session.json"
  342       body <- consumeBody req
  343       case Aeson.eitherDecode body of
  344         Left _ ->
  345           respond $ jsonResponse status400 (object ["error" .= ("Invalid JSON: expected {\"prompt\": \"...\"}" :: Text)])
  346         Right obj -> case Map.lookup ("prompt" :: Text) (obj :: Map.Map Text Text) of
  347           Nothing ->
  348             respond $ jsonResponse status400 (object ["error" .= ("Missing 'prompt' field" :: Text)])
  349           Just prompt -> do
  350             TIO.writeFile promptPath prompt
  351             -- Update agent name in session metadata if a name was provided
  352             let mName = Map.lookup "name" obj
  353             case mName of
  354               Just name | not (T.null name) -> do
  355                 raw <- LBS.readFile metaPath
  356                 case Aeson.eitherDecode' raw of
  357                   Right meta -> do
  358                     let agentName = either (const Nothing) Just (mkAgentName name)
  359                         updated = (meta :: SessionMeta) { _sm_agent = agentName }
  360                         tmpP = metaPath <> ".tmp"
  361                     LBS.writeFile tmpP (Aeson.encode updated)
  362                     renameFile tmpP metaPath
  363                   Left _ -> pure ()
  364               _ -> pure ()
  365             respond $ jsonResponse status200 (object ["ok" .= True])

  374 -- | Send a user message to a session and get a completion.
  375 handleSend :: FrontendEnv -> Text -> Request -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  376 handleSend env sid req respond = do
  377   -- Validate session ID
  378   if T.isInfixOf ".." sid || T.isInfixOf "/" sid
  379     then respond $ jsonResponse status400 (object ["error" .= ("Invalid session ID" :: Text)])
  380     else do
  381       let sessionDir = _fe_sessionsDir env </> T.unpack sid
  382           transcriptPath = sessionDir </> "transcript.jsonl"
  383       exists <- doesFileExist transcriptPath
  384       if not exists
  385         then respond $ jsonResponse status404 (object ["error" .= ("Session not found" :: Text)])
  386         else do
  387           -- Parse request body
  388           body <- consumeBody req
  389           case Aeson.eitherDecode body of
  390             Left _ ->
  391               respond $ jsonResponse status400 (object ["error" .= ("Invalid JSON: expected {\"message\": \"...\"}" :: Text)])
  392             Right (SendRequest userText) -> do
  393               mProvider <- readIORef (_fe_provider env)
  394               mModel <- readIORef (_fe_model env)
  395               case (mProvider, mModel) of
  396                 (Nothing, _) ->
  397                   respond $ jsonResponse status503 (object ["error" .= ("No provider configured" :: Text)])
  398                 (_, Nothing) ->
  399                   respond $ jsonResponse status503 (object ["error" .= ("No model configured" :: Text)])
  400                 (Just provider, Just model) -> do
  401                   result <- try @SomeException $ doCompletion env provider model userText transcriptPath
  402                   case result of
  403                     Left e -> do
  404                       _lh_logError (_fe_logger env) $ "Send error: " <> T.pack (show e)
  405                       respond $ jsonResponse status500 (object ["error" .= ("Completion failed" :: Text)])
  406                     Right respText ->
  407                       respond $ jsonResponse status200 (object ["response" .= respText])
  408 
  409 -- | Run a completion: load context, send to provider, record to transcript.
  410 doCompletion :: FrontendEnv -> SomeProvider -> ModelId -> Text -> FilePath -> IO Text
  411 doCompletion env provider model userText transcriptPath = do
  412   -- Open a transcript handle for recording
  413   th <- mkFileTranscriptHandle (_fe_logger env) transcriptPath
  414   -- Load recent messages for context
  415   history <- loadRecentMessages th 50 100000
  416   -- Check for per-session custom prompt, falling back to global
  417   let sessionDir = takeDirectory transcriptPath
  418       customPromptPath = sessionDir </> "custom-prompt.md"
  419   customExists <- doesFileExist customPromptPath
  420   systemPrompt <- if customExists
  421     then Just <$> TIO.readFile customPromptPath
  422     else pure (_fe_systemPrompt env)
  423   -- Build context with system prompt and history
  424   let ctx0 = foldl (flip addMessage) (emptyContext systemPrompt) history
  425       userMsg = textMessage User userText
  426       ctx = addMessage userMsg ctx0
  427       -- Wrap provider with transcript logging
  428       provider' = mkTranscriptProvider th (unModelId model) provider
  429       req = CompletionRequest
  430         { _cr_model        = model
  431         , _cr_messages     = contextMessages ctx
  432         , _cr_systemPrompt = contextSystemPrompt ctx
  433         , _cr_maxTokens    = Just 4096
  434         , _cr_tools        = []
  435         , _cr_toolChoice   = Nothing
  436         }
  437   resp <- complete provider' req
  438   _th_flush th
  439   _th_close th
  440   pure (responseText resp)

  442 -- | Consume the full request body.
  443 consumeBody :: Request -> IO LBS.ByteString
  444 consumeBody req = LBS.fromChunks <$> go
  445   where
  446     go = do
  447       chunk <- getRequestBodyChunk req
  448       if BS.null chunk then pure [] else (chunk :) <$> go
