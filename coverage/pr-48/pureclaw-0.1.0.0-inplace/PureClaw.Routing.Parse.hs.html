    1 -- |
    2 -- Module      : PureClaw.Routing.Parse
    3 -- Description : Slash-input parser + shared sanitizers (Tabbed Chat WU2).
    4 --
    5 -- This module owns the slash-input grammar for Tabbed Chat (#51):
    6 --
    7 --   * @parseInput :: RoutingConfig -> Text -> Either ParseError ParsedInput@
    8 --     classifies a single user message into 'Switch' \/ 'Inject' \/
    9 --     'Default' \/ 'ParsedSlashCmd' per the grammar at
   10 --     @docs\/tabbed-chat.md@ §"Routing Grammar (v1)".
   11 --   * @parseSlashCommand :: Text -> Maybe SlashCommand@ is the
   12 --     re-exported in-tab-loop entry-point used by AI tab loops (WU6, I2
   13 --     contract) when an inbound 'UserText' starts with @\/@.
   14 --   * @mkSessionId :: Text -> Either ParseError SessionId@ enforces the
   15 --     S3 \/ P15a corpus: rejects @\/@, @\\@, @..@, @NUL@, and any
   16 --     character outside @[a-zA-Z0-9_-]@.
   17 --   * @sanitizeTabName :: Text -> Either NameError Text@ is the shared
   18 --     sanitization function used by every code path that sets
   19 --     @_tabHandle_name@ (H11) AND the @\/tab rename@ handler (S10).
   20 --
   21 -- == Grammar (single-char index, [0-9a-z])
   22 --
   23 -- @
   24 -- input        ::= switch | inject | default | slash-cmd
   25 -- switch       ::= \'\/\' IDX
   26 -- inject       ::= \'\/\' IDX WS payload
   27 -- default      ::= payload
   28 -- slash-cmd    ::= \'\/\' WORD ...
   29 --              |   \'\/tab\' WS action [WS args]
   30 --              |   \'\/tabs\'
   31 -- IDX          ::= [0-9a-z]        -- exactly one character
   32 -- WS           ::= one or more spaces\/tabs
   33 -- @
   34 --
   35 -- The index is exactly one character; digits @0-9@ map to tab indices
   36 -- @0..9@ and lowercase letters @a-z@ map to tab indices @10..35@. This
   37 -- gives a 36-tab maximum (matching @_rc_maxTabs@ default). Multi-char
   38 -- indices like @\/10@, @\/12@, @\/aa@, or @\/1a@ are rejected as
   39 -- 'ParseErrorMalformed' — there is no greedy digit run any more.
   40 --
   41 -- @\/0 0 run@ parses as @Inject 0 \"0 run\"@ — the payload character
   42 -- after the WS separator is preserved verbatim per P4.
   43 --
   44 -- The @\/0@ vs @\/0 \<text\>@ disambiguation is whitespace-driven:
   45 -- @\/0@ alone (or with trailing whitespace only) is a 'Switch';
   46 -- @\/0 \<anything-non-blank\>@ is an 'Inject'. The @\<text\>@ side of
   47 -- 'Inject' preserves all characters after the first whitespace
   48 -- separator, including embedded newlines (the channel layer is
   49 -- responsible for splitting incoming messages at newlines if it wants
   50 -- per-line semantics; we treat the message as a single input).
   51 --
   52 -- == Notes
   53 --
   54 --   * Index bounds-checking uses 'RoutingConfig._rc_maxTabs'. WU1's
   55 --     'mkTabIndex' enforces the floor only; the parser enforces the
   56 --     ceiling and emits 'ParseErrorIndexOutOfRange' on failure.
   57 --   * @\/0@ with a non-whitespace tail (e.g. @\/01@) is rejected as
   58 --     'ParseErrorMalformed' — the single-char grammar admits no
   59 --     multi-char numeric or alphanumeric indices.
   60 --   * @\/tab resume \<id\>@ runs the id through 'mkSessionId' at parse
   61 --     time per P15a (so the dispatcher never sees an unvalidated id);
   62 --     rejection surfaces as 'ParseErrorInvalidSessionId'.

   89 -- parseInput
   90 -- ---------------------------------------------------------------------------
   91 
   92 -- | Parse a single incoming message into the high-level routing
   93 -- classification.
   94 --
   95 -- See module haddock for the grammar; see @docs\/tabbed-chat.md@
   96 -- §"Parser invariants" for the per-form contract.
   97 parseInput :: RT.RoutingConfig -> Text -> Either RT.ParseError RT.ParsedInput
   98 parseInput rc raw
   99   | T.null trimmedLeft = Left RT.ParseErrorEmptyInput
  100   | not slashy        = Right (RT.Default raw)
  101   | otherwise         = classifySlashy rc trimmedLeft raw
  102   where
  103     -- We only drop leading whitespace; trailing whitespace is part of
  104     -- the payload semantics ('/0 ' alone collapses to a Switch per the
  105     -- "trailing-whitespace-only payload" rule).
  106     trimmedLeft = T.dropWhile isHSpace raw
  107     slashy      = case T.uncons trimmedLeft of
  108       Just ('/', _) -> True
  109       _             -> False
  110 
  111 -- | Dispatch on the shape after the leading @\/@.
  112 --
  113 --   * If the body is a single index char @c \in [0-9a-z]@ followed
  114 --     by end-of-input or whitespace, we have a 'Switch' (no payload)
  115 --     or 'Inject' (with payload).
  116 --   * Anything longer that starts with a digit (e.g. @\/12@,
  117 --     @\/1a@) is malformed — there is no greedy index any more.
  118 --     Bonus: @\/01@ \/ @\/02@ also fall here.
  119 --   * Anything longer that starts with a letter (e.g. @\/help@,
  120 --     @\/session new@) is routed through the slash-command parser.
  121 --
  122 -- The whitespace-or-EOI lookahead on the second character is what
  123 -- disambiguates @\/s@ (Switch to tab 28) from @\/session new@
  124 -- (slash command). The lookahead is a single 'T.uncons' on the
  125 -- tail; no second pass over the input.
  126 classifySlashy
  127   :: RT.RoutingConfig
  128   -> Text   -- ^ leading-whitespace-stripped input (starts with @\/@)
  129   -> Text   -- ^ original input (preserved for the slash-cmd fallback)
  130   -> Either RT.ParseError RT.ParsedInput
  131 classifySlashy rc t orig =
  132   let body = T.drop 1 t   -- drop the leading '/'
  133   in case T.uncons body of
  134        Nothing -> Left RT.ParseErrorMalformed
  135        Just (c, rest)
  136          | Just n  <- parseTabIndexChar c
  137          , isIndexBoundary rest         -> parseIndexCharForm rc n rest
  138          | Char.isDigit c               ->
  139              -- Digit followed by more (non-WS) characters: legacy
  140              -- multi-digit shapes (@\/12@, @\/01@, @\/1a@) are no
  141              -- longer in the grammar. They cannot be slash commands
  142              -- either (commands start with a letter), so reject.
  143              Left RT.ParseErrorMalformed
  144          | otherwise                    -> parseSlashCommandForm orig
  145 
  146 -- | True when the byte following the single index char is end-of-input
  147 -- or whitespace (space, tab, newline, carriage return). Anything else
  148 -- means the index char was the first letter of a word — punt to the
  149 -- slash-command parser.
  150 isIndexBoundary :: Text -> Bool
  151 isIndexBoundary rest = case T.uncons rest of
  152   Nothing     -> True
  153   Just (c, _) -> isHSpace c || c == '\n' || c == '\r'
  154 
  155 -- | Map a single ASCII character to its tab index.
  156 --
  157 --   * @\'0\'-\'9\'@ → @0..9@
  158 --   * @\'a\'-\'z\'@ → @10..35@
  159 --   * anything else → 'Nothing'
  160 --
  161 -- This is the only place the single-char index alphabet is encoded;
  162 -- the parser, dashboard, and any future readers should reuse this
  163 -- function rather than re-implementing the mapping.
  164 parseTabIndexChar :: Char -> Maybe Int
  165 parseTabIndexChar c
  166   | Char.isDigit c        = Just (Char.ord c - Char.ord '0')
  167   | Char.isAsciiLower c   = Just (10 + Char.ord c - Char.ord 'a')
  168   | otherwise             = Nothing
  169 
  170 -- | Parse the @\/IDX [WS payload]@ form. The caller has already
  171 -- (a) decoded the index char to an 'Int' and (b) confirmed via
  172 -- 'isIndexBoundary' that @rest@ is empty or starts with whitespace.
  173 -- We bounds-check the index against @_rc_maxTabs@ and hand off to
  174 -- 'classifyAfterIndex' to decide Switch vs Inject.
  175 parseIndexCharForm
  176   :: RT.RoutingConfig
  177   -> Int    -- ^ decoded index from the single index character
  178   -> Text   -- ^ everything after the index character (WS or empty by caller invariant)
  179   -> Either RT.ParseError RT.ParsedInput
  180 parseIndexCharForm rc n rest = do
  181   ti <- boundedTabIndex rc n
  182   classifyAfterIndex ti rest
  183 
  184 -- | Bounds-check an 'Int' against @_rc_maxTabs@ and build a
  185 -- 'Tab.TabIndex' on success.
  186 --
  187 -- WU1's 'Tab.mkTabIndex' enforces only the floor (@n >= 0@); we
  188 -- enforce the ceiling here. Caller invariant: @n >= 0@ — the single
  189 -- index character produced by 'parseTabIndexChar' is always in
  190 -- @[0..35]@, so 'Tab.mkTabIndex' always returns 'Just'. We collapse
  191 -- the structurally-unreachable 'Nothing' branch into a single 'error'
  192 -- call via 'Maybe.fromMaybe' so the HPC report has no defensive dead
  193 -- arm to flag.
  194 boundedTabIndex
  195   :: RT.RoutingConfig
  196   -> Int
  197   -> Either RT.ParseError Tab.TabIndex
  198 boundedTabIndex rc n
  199   | n >= RT._rc_maxTabs rc = Left (RT.ParseErrorIndexOutOfRange n)
  200   | otherwise =
  201       Right (Maybe.fromMaybe
  202                (error "PureClaw.Routing.Parse.boundedTabIndex: \
  203                       \mkTabIndex returned Nothing for n >= 0 — \
  204                       \caller invariant violated")
  205                (Tab.mkTabIndex n))
  206 
  207 -- | After consuming the digit run, decide between 'Switch' (no payload)
  208 -- and 'Inject' (payload preserved verbatim after the separator).
  209 --
  210 -- Newline handling per the design's multi-line rule: a newline
  211 -- immediately after the digit run counts as a separator, but the
  212 -- payload includes the newline (so @\/0\\nfoo@ → @Inject 0 \"\\nfoo\"@).
  213 -- Horizontal whitespace (space \/ tab) is consumed and the payload
  214 -- starts at the first non-blank byte (so @\/0 run tests@ →
  215 -- @Inject 0 \"run tests\"@).
  216 --
  217 -- 'shapeCheckRest' has already rejected the malformed @\/12abc@ case
  218 -- before this function is called, so the first char of @rest@ here is
  219 -- guaranteed to be a horizontal-WS, newline, or carriage return when
  220 -- @rest@ is non-empty.
  221 classifyAfterIndex :: Tab.TabIndex -> Text -> Either RT.ParseError RT.ParsedInput
  222 classifyAfterIndex ti rest
  223   | T.null rest = Right (RT.Switch ti)
  224   | otherwise   =
  225       let (_hSep, afterHSep)  = T.span isHSpace rest
  226           startsWithNewline   = case T.uncons afterHSep of
  227                                   Just (c, _) -> c == '\n' || c == '\r'
  228                                   Nothing     -> False
  229       in if startsWithNewline
  230            then
  231              -- The newline (and everything after it, including any
  232              -- trailing hSpace) IS the payload: a multi-line input
  233              -- like @\/0\\nfoo@ produces @Inject 0 \"\\nfoo\"@; an
  234              -- input ending right at the newline like @\/0\\n@
  235              -- produces @Inject 0 \"\\n\"@.
  236              Right (RT.Inject ti afterHSep)
  237            else
  238              -- Horizontal-WS-only separator. Drop the leading hSpace
  239              -- from the payload; if nothing meaningful remains, treat
  240              -- the trailing whitespace as belonging to the bare switch
  241              -- ("/0 " rule).
  242              let payload = T.dropWhile isHSpace afterHSep
  243              in if T.null payload
  244                   then Right (RT.Switch ti)
  245                   else Right (RT.Inject ti payload)
  246 
  247 -- | Whitespace-class used by the parser. Excludes newlines so that a
  248 -- newline embedded in @\/0\\nfoo@ is treated as payload (per the
  249 -- design's multi-line guidance), not as a separator that strips
  250 -- preceding whitespace.
  251 isHSpace :: Char -> Bool
  252 isHSpace c = c == ' ' || c == '\t'

  259 -- | The slash-cmd branch of 'parseInput' (e.g. @\/help@, @\/vault add foo@,
  260 -- @\/tab new 3 shell@, @\/tabs@).
  261 --
  262 -- Re-uses the existing 'PureClaw.Agent.SlashCommands.parseSlashCommand'
  263 -- so there is no parser divergence between Tabbed Chat's preprocessor
  264 -- and the in-tab-loop I2 re-parse path (AI tab loop receives
  265 -- @UserText t@ starting with @\/@ and calls 'parseSlashCommand'
  266 -- directly).
  267 --
  268 -- Special-case for P15a: @\/tab resume \<bad-id\>@ is surfaced as the
  269 -- dedicated 'RT.ParseErrorInvalidSessionId' (not the generic
  270 -- 'RT.ParseErrorMalformed') so the user sees a useful rejection
  271 -- reason. Other slash-cmd shapes fall through to the generic parser.
  272 --
  273 -- Otherwise: success becomes 'RT.ParsedSlashCmd'; failure becomes
  274 -- 'RT.ParseErrorMalformed' (a @\/@-prefixed input that matches no
  275 -- known command is an error, not a 'Default').
  276 parseSlashCommandForm :: Text -> Either RT.ParseError RT.ParsedInput
  277 parseSlashCommandForm orig
  278   -- Detect a /tab resume <id> shape with an invalid id so we can
  279   -- emit the specific ParseErrorInvalidSessionId code (P15a).
  280   | isTabResumeShape orig
  281   , Left e <- mkSessionId (extractTabResumeArg orig)
  282   = Left e
  283   | otherwise = case parseSlashCommand orig of
  284       Just cmd -> Right (RT.ParsedSlashCmd cmd)
  285       Nothing  -> Left RT.ParseErrorMalformed
  286 
  287 -- | True if the input begins with @\/tab resume @ followed by at least
  288 -- one non-blank character (case-insensitive on the keywords).
  289 --
  290 -- Intentionally permissive on what follows the prefix — including
  291 -- inputs that contain spaces, control bytes, or path separators —
  292 -- because the dedicated 'ParseErrorInvalidSessionId' code path needs
  293 -- to fire for ANY ill-formed id, not just well-shaped-but-rejected
  294 -- ones.
  295 isTabResumeShape :: Text -> Bool
  296 isTabResumeShape t =
  297   let stripped = T.strip t
  298       lower    = T.toLower stripped
  299       pfx      = "/tab resume "
  300   in pfx `T.isPrefixOf` lower
  301        && not (T.null (T.strip (T.drop (T.length pfx) stripped)))
  302 
  303 -- | Extract the everything-after-@\/tab resume @ argument blob.
  304 -- Caller ensures shape via 'isTabResumeShape'.
  305 extractTabResumeArg :: Text -> Text
  306 extractTabResumeArg t =
  307   let stripped = T.strip t
  308       pfx      = "/tab resume "
  309   in T.strip (T.drop (T.length pfx) stripped)
  310 
  311 -- | The in-tab-loop I2 re-parser. Includes the Tabbed Chat
  312 -- @\/tab*@ / @\/tabs@ family in addition to the existing slash
  313 -- commands.
  314 --
  315 -- A plain @parseSlashCommand "/foo"@ returns 'Nothing' for unknown
  316 -- commands; this wrapper preserves that contract while folding in the
  317 -- new @\/tab@ command family before delegating to
  318 -- 'Slash.parseSlashCommand'.
  319 parseSlashCommand :: Text -> Maybe Slash.SlashCommand
  320 parseSlashCommand input =
  321   let stripped = T.strip input
  322   in case parseTabFamily stripped of
  323        Just cmd -> Just cmd
  324        Nothing  -> Slash.parseSlashCommand stripped

  331 -- | Recognise the @\/tab@ / @\/tabs@ command family. Returns 'Nothing'
  332 -- for inputs outside this family so the caller can fall back to the
  333 -- existing slash-command parsers without parser divergence.
  334 parseTabFamily :: Text -> Maybe Slash.SlashCommand
  335 parseTabFamily t
  336   -- /tabs is the bare alias for /tab list.
  337   | lower == "/tabs" = Just (Slash.CmdTab Slash.TabListCmd)
  338   | lower == "/tab"  = Nothing   -- bare /tab is not a command (would need a sub)
  339   | "/tab " `T.isPrefixOf` lower =
  340       let rest = T.strip (T.drop (T.length "/tab ") t)
  341       in parseTabAction rest
  342   | otherwise = Nothing
  343   where
  344     lower = T.toLower t
  345 
  346 -- | Dispatch on the @\<action\>@ word: @list@ \/ @new\<N\>@ \/
  347 -- @close\<N\>@ \/ @focus\<N\>@ \/ @resume\<id\>@ \/ @rename\<N\>@.
  348 --
  349 -- The caller in 'parseTabFamily' has already stripped surrounding
  350 -- whitespace and ensured the @\/tab @ prefix is followed by at least
  351 -- one non-blank character; so @T.words rest@ here always produces a
  352 -- non-empty list.
  353 parseTabAction :: Text -> Maybe Slash.SlashCommand
  354 parseTabAction rest = case T.words rest of
  355   []         -> Nothing  -- defensive; caller ensures non-empty
  356   (act:args) -> case T.toLower act of
  357     "list"   -> if null args then Just (Slash.CmdTab Slash.TabListCmd) else Nothing
  358     "new"    -> parseTabNew args
  359     "close"  -> parseTabClose args
  360     "focus"  -> parseTabFocus args
  361     "resume" -> parseTabResume args
  362     "rename" -> parseTabRename args rest
  363     _        -> Nothing
  364 
  365 -- | @\/tab new [\<kind\> [\<arg-text\>]]@.
  366 --
  367 -- The grammar deliberately omits a user-specified index: new tabs are
  368 -- always allocated at the lowest free slot by the handler (tmux-style
  369 -- packing). @\/tab new@ with no kind force-prompts the user; with a
  370 -- kind it spawns directly.
  371 parseTabNew :: [Text] -> Maybe Slash.SlashCommand
  372 parseTabNew []        = Just (Slash.CmdTab (Slash.TabNewCmd Nothing Nothing))
  373 parseTabNew (k:argTs) = do
  374   kind <- parseTabKindArg k
  375   let argText = if null argTs
  376                   then Nothing
  377                   else Just (T.unwords argTs)
  378   Just (Slash.CmdTab (Slash.TabNewCmd (Just kind) argText))
  379 
  380 -- | @\/tab close \<N\> [--force]@.
  381 parseTabClose :: [Text] -> Maybe Slash.SlashCommand
  382 parseTabClose []     = Nothing
  383 parseTabClose (n:fs) = do
  384   idx <- parseDecimalIndex n
  385   force <- case fs of
  386              []           -> Just Slash.ForceNo
  387              ["--force"]  -> Just Slash.ForceYes
  388              _            -> Nothing
  389   Just (Slash.CmdTab (Slash.TabCloseCmd idx force))
  390 
  391 -- | @\/tab focus \<N\>@.
  392 parseTabFocus :: [Text] -> Maybe Slash.SlashCommand
  393 parseTabFocus [n] = do
  394   idx <- parseDecimalIndex n
  395   Just (Slash.CmdTab (Slash.TabFocusCmd idx))
  396 parseTabFocus _   = Nothing

  398 -- | @\/tab resume \<session-id\>@. The id is run through 'mkSessionId'
  399 -- so that S3 \/ P15a rejection (path traversal, NUL, non-alnum)
  400 -- happens at parser time. We return 'Nothing' on rejection here and
  401 -- rely on the @\/@-prefixed-must-parse policy in
  402 -- 'parseSlashCommandForm' to surface this as 'ParseErrorMalformed'.
  403 -- The dedicated 'ParseErrorInvalidSessionId' code is emitted by
  404 -- 'parseInput' below when it recognises a @\/tab resume@ shape with
  405 -- a bad id.
  406 parseTabResume :: [Text] -> Maybe Slash.SlashCommand
  407 parseTabResume [idText] = case mkSessionId idText of
  408   Right sid -> Just (Slash.CmdTab (Slash.TabResumeCmd sid))
  409   Left _    -> Nothing
  410 parseTabResume _        = Nothing
  411 
  412 -- | @\/tab rename \<N\> \<name\>@. The original rest-of-line is taken
  413 -- so that names with embedded whitespace (e.g. @"my shell"@) survive
  414 -- intact.
  415 --
  416 -- Caller invariant: when @args@ has >= 2 elements, the rest-of-line
  417 -- 'line' contains at least @"rename N "@ plus a non-blank tail; so
  418 -- the recovered @name@ here is never empty.
  419 parseTabRename :: [Text] -> Text -> Maybe Slash.SlashCommand
  420 parseTabRename []        _    = Nothing
  421 parseTabRename [_]       _    = Nothing
  422 parseTabRename (n:_rest) line = do
  423   idx <- parseDecimalIndex n
  424   let afterRenameKw = stripLeadingWord "rename" line
  425       afterN        = stripLeadingWord n        afterRenameKw
  426       name          = T.strip afterN
  427   -- 'name' is non-empty by caller's invariant; guard preserved
  428   -- defensively.
  429   if T.null name
  430      then Nothing
  431      else Just (Slash.CmdTab (Slash.TabRenameCmd idx name))
  432 
  433 -- | Drop a leading word (case-insensitive on alpha keywords; literal
  434 -- match for non-alpha) plus its trailing whitespace from a 'Text'.
  435 -- Returns the original 'Text' unchanged if the prefix is absent.
  436 stripLeadingWord :: Text -> Text -> Text
  437 stripLeadingWord w t =
  438   let lt   = T.toLower t
  439       lw   = T.toLower w
  440       lead = T.dropWhile isHSpace lt
  441       ofs  = T.length t - T.length lead
  442   in if lw `T.isPrefixOf` lead
  443        then T.dropWhile isHSpace (T.drop (ofs + T.length lw) t)
  444        else t
  445 
  446 -- | Parse a strictly-decimal index, rejecting leading-zero forms per
  447 -- the same disambiguation rule that 'parseDigitsForm' applies.
  448 parseDecimalIndex :: Text -> Maybe Int
  449 parseDecimalIndex t
  450   | T.null t                                  = Nothing
  451   | not (T.all Char.isDigit t)                = Nothing
  452   | T.length t > 1 && T.head t == '0'         = Nothing
  453   | otherwise = case TR.decimal t of
  454       Right (n, leftover) | T.null leftover -> Just n
  455       _                                     -> Nothing
  456 
  457 -- | Recognise the short tab-kind keywords (case-insensitive).
  458 parseTabKindArg :: Text -> Maybe Slash.TabKindArg
  459 parseTabKindArg t = case T.toLower t of
  460   "ai"      -> Just Slash.TkaAi
  461   "harness" -> Just Slash.TkaHarness
  462   "shell"   -> Just Slash.TkaShell
  463   "ssh"     -> Just Slash.TkaSsh
  464   "tmux"    -> Just Slash.TkaTmux
  465   _         -> Nothing

  472 -- | Smart constructor for 'Core.SessionId' used by the @\/tab resume@
  473 -- and @\/session resume@ parsers per S3 \/ P15a \/ L7.
  474 --
  475 -- Rejects:
  476 --
  477 --   * the empty string;
  478 --   * any occurrence of @\/@ or @\\@ (path traversal);
  479 --   * the substring @..@ (parent-dir reference);
  480 --   * any NUL byte (@\\0@);
  481 --   * any character outside the @[a-zA-Z0-9_-]@ corpus.
  482 mkSessionId :: Text -> Either RT.ParseError Core.SessionId
  483 mkSessionId t
  484   | T.null t                              = Left RT.ParseErrorInvalidSessionId
  485   | T.any (== '/')  t                     = Left RT.ParseErrorInvalidSessionId
  486   | T.any (== '\\') t                     = Left RT.ParseErrorInvalidSessionId
  487   | T.any (== '\0') t                     = Left RT.ParseErrorInvalidSessionId
  488   | ".." `T.isInfixOf` t                  = Left RT.ParseErrorInvalidSessionId
  489   | not (T.all isSessionIdChar t)         = Left RT.ParseErrorInvalidSessionId
  490   | otherwise = Right (Core.SessionId t)
  491 
  492 -- | The @[a-zA-Z0-9_-]@ predicate used by 'mkSessionId'.
  493 isSessionIdChar :: Char -> Bool
  494 isSessionIdChar c =
  495      Char.isAsciiLower c
  496   || Char.isAsciiUpper c
  497   || Char.isDigit      c
  498   || c == '_'
  499   || c == '-'

  506 -- | Default length cap used by the bare 'sanitizeTabName'. Matches
  507 -- 'RT.RoutingConfig._rc_maxNameLen' default. Exposed so test fixtures
  508 -- can reference it directly.
  509 defaultMaxNameLen :: Int
  510 defaultMaxNameLen = 32
  511 
  512 -- | Sanitize a user-supplied tab name per H11 / S10.
  513 --
  514 -- Equivalent to @'sanitizeTabNameWith' 'defaultMaxNameLen'@.
  515 sanitizeTabName :: Text -> Either Tab.NameError Text
  516 sanitizeTabName = sanitizeTabNameWith defaultMaxNameLen
  517 
  518 -- | Sanitize a user-supplied tab name against an explicit length cap.
  519 --
  520 -- Pipeline (in order):
  521 --
  522 --   1. Reject inputs that contain ANSI escape sequences
  523 --      (@\\x1b[@ \/ CSI @\\x9b@) — 'Tab.NameContainsAnsi'.
  524 --   2. Reject inputs that contain control bytes (@\<0x20@ except
  525 --      ordinary space) — 'Tab.NameContainsControlBytes'.
  526 --   3. Reject inputs whose UTF-16 length exceeds the cap —
  527 --      'Tab.NameTooLong'.
  528 --   4. Run the redaction pipeline (paths → hostnames → IPv4 → ssh
  529 --      stderr) from 'PureClaw.Internal.Redact'. If the result is empty
  530 --      after trimming, surface 'Tab.NameRedactedToEmpty'.
  531 --
  532 -- The output is always a non-empty 'Text' satisfying the four
  533 -- invariants asserted by the H11 property test.
  534 sanitizeTabNameWith :: Int -> Text -> Either Tab.NameError Text
  535 sanitizeTabNameWith cap raw
  536   | containsAnsi raw         = Left Tab.NameContainsAnsi
  537   | containsControlByte raw  = Left Tab.NameContainsControlBytes
  538   | T.length raw > cap       = Left Tab.NameTooLong
  539   | T.null trimmed           = Left Tab.NameRedactedToEmpty
  540   | otherwise                = Right trimmed
  541   where
  542     redacted = redactPipeline raw
  543     trimmed  = T.strip redacted
  544 
  545 -- | The redaction pipeline; mirrors 'Redact.redactText' which is not
  546 -- exported (only individual stage functions are). Order matches the
  547 -- canonical pipeline: paths first, hostnames second, IPv4 third, ssh
  548 -- stderr last.
  549 redactPipeline :: Text -> Text
  550 redactPipeline =
  551     Redact.redactSshStderr
  552   . Redact.redactIPv4
  553   . Redact.redactHostname
  554   . Redact.redactPath . T.unpack
  555 
  556 -- | True if the input contains an ANSI \/ CSI escape introducer
  557 -- (@ESC [@ or the 8-bit CSI byte @0x9B@).
  558 containsAnsi :: Text -> Bool
  559 containsAnsi t = "\ESC[" `T.isInfixOf` t || T.any (== '\x9B') t
  560 
  561 -- | True if the input contains any byte below @0x20@ other than a
  562 -- tab (@\\t@) or newline. We treat ordinary spaces (@0x20@) as
  563 -- permitted; the H11 spec says "control bytes \< 0x20 except space"
  564 -- which we read as: control bytes are forbidden whether or not the
  565 -- containing whitespace-class is allowed. Tabs and newlines (the
  566 -- two most common embedded controls) are rejected — a tab name is a
  567 -- single-line label.
  568 containsControlByte :: Text -> Bool
  569 containsControlByte = T.any isReject
  570   where
  571     isReject c = c < ' ' && c /= ' '
