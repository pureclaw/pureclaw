  114 -- | Handle for the current conversation session.
  115 --
  116 -- Owns the on-disk session directory (mode @0o700@), the @session.json@
  117 -- metadata file (mode @0o600@), and the per-session transcript handle
  118 -- writing to @transcript.jsonl@ (mode @0o600@).
  119 data SessionHandle = SessionHandle
  120   { _sh_meta       :: IORef SessionMeta
  121     -- ^ Mutable session metadata. Updated by callers (e.g. to bump
  122     -- @last_active@) and persisted by '_sh_save'.
  123   , _sh_transcript :: TranscriptHandle
  124     -- ^ Transcript handle owned by the session.
  125   , _sh_dir        :: FilePath
  126     -- ^ On-disk session directory, i.e. @<baseDir>/<sessionId>/@.
  127   , _sh_save       :: IO ()
  128     -- ^ Persist '_sh_meta' to @session.json@ atomically (write to
  129     -- @session.json.tmp@ then rename).
  130   }
  131 
  132 -- | Reasons a resume attempt can fail.
  133 data ResumeError
  134   = ResumeMissingMetadata FilePath
  135     -- ^ No @session.json@ at the expected path.
  136   | ResumeCorruptedMetadata FilePath String
  137     -- ^ @session.json@ exists but does not parse as 'SessionMeta'.
  138     -- The second field is a human-readable recovery hint.
  139   deriving stock (Show, Eq)

  158 -- | Create a brand-new on-disk session handle.
  159 --
  160 -- Creates @\<baseDir\>/\<sessionId\>/@ with mode @0o700@, writes the
  161 -- initial @session.json@ (mode @0o600@) via 'saveMeta', and opens
  162 -- @transcript.jsonl@ (mode @0o600@) through
  163 -- 'mkBroadcastingFileTranscriptHandle'. When the optional broker is
  164 -- 'Nothing' the helper falls back to a plain file handle; when 'Just',
  165 -- transcript writes are broadcast to the supplied 'StreamBroker' in
  166 -- addition to being persisted.
  167 mkSessionHandle
  168   :: Maybe StreamBroker
  169   -> LogHandle
  170   -> FilePath
  171   -> SessionMeta
  172   -> IO SessionHandle
  173 mkSessionHandle mBroker logger baseDir meta = do
  174   let sid  = unSessionId (_sm_id meta)
  175       dir  = baseDir </> T.unpack sid
  176       txp  = dir </> "transcript.jsonl"
  177   createDirectoryIfMissing True dir
  178   setFileMode dir 0o700
  179   metaRef <- newIORef meta
  180   saveMeta dir metaRef
  181   tx <- mkBroadcastingFileTranscriptHandle mBroker (_sm_id meta) logger txp
  182   let save = saveMeta dir metaRef
  183       tx'  = touchLastActive metaRef save tx
  184   pure SessionHandle
  185     { _sh_meta       = metaRef
  186     , _sh_transcript = tx'
  187     , _sh_dir        = dir
  188     , _sh_save       = save
  189     }
  190 
  191 -- | Atomically persist the metadata IORef to @\<dir\>/session.json@ by
  192 -- writing to a sibling @session.json.tmp@ with mode @0o600@ then
  193 -- 'renameFile'-ing it into place. The rename is atomic on POSIX, so a
  194 -- crash mid-write leaves the previous @session.json@ intact.
  195 saveMeta :: FilePath -> IORef SessionMeta -> IO ()
  196 saveMeta dir ref = do
  197   meta <- readIORef ref
  198   let finalP = dir </> "session.json"
  199       tmpP   = finalP <> ".tmp"
  200   LBS.writeFile tmpP (Aeson.encode meta)
  201   setFileMode tmpP 0o600
  202   renameFile tmpP finalP
  203 
  204 -- | Wrap a 'TranscriptHandle' so that after every successful
  205 -- '_th_record' call, @_sm_lastActive@ is bumped to 'getCurrentTime'
  206 -- and the session metadata is persisted to disk via the supplied save
  207 -- action. This keeps the sidebar's "last active" column accurate
  208 -- without requiring every transcript call site to remember the bump.
  209 touchLastActive :: IORef SessionMeta -> IO () -> TranscriptHandle -> TranscriptHandle
  210 touchLastActive metaRef save th = th
  211   { _th_record = \entry -> do
  212       _th_record th entry
  213       now <- getCurrentTime
  214       atomicModifyIORef' metaRef (\m -> (m { _sm_lastActive = now }, ()))
  215       save
  216   }

  284 -- | Reopen an existing session by ID. Reads @session.json@, validates
  285 -- the JSON, and reopens @transcript.jsonl@ for append. When the optional
  286 -- broker is 'Just', the reopened transcript handle broadcasts records to
  287 -- it; 'Nothing' yields a plain file handle.
  288 resumeSession
  289   :: Maybe StreamBroker
  290   -> LogHandle
  291   -> FilePath
  292   -> SessionId
  293   -> IO (Either ResumeError SessionHandle)
  294 resumeSession mBroker logger baseDir sid = do
  295   let dir    = baseDir </> T.unpack (unSessionId sid)
  296       metaP  = dir </> "session.json"
  297       txP    = dir </> "transcript.jsonl"
  298   exists <- doesFileExist metaP
  299   if not exists
  300     then pure (Left (ResumeMissingMetadata metaP))
  301     else do
  302       raw <- LBS.readFile metaP
  303       case Aeson.eitherDecode' raw of
  304         Left err -> pure
  305           (Left (ResumeCorruptedMetadata metaP
  306             ("failed to parse session metadata: " <> err
  307               <> " — recovery hint: inspect or remove " <> metaP)))
  308         Right meta -> do
  309           metaRef <- newIORef (meta :: SessionMeta)
  310           tx <- mkBroadcastingFileTranscriptHandle mBroker sid logger txP
  311           let save = saveMeta dir metaRef
  312               tx'  = touchLastActive metaRef save tx
  313           pure (Right SessionHandle
  314             { _sh_meta       = metaRef
  315             , _sh_transcript = tx'
  316             , _sh_dir        = dir
  317             , _sh_save       = save
  318             })

  324 -- | Reasons a branch seed cannot be resolved from a source session.
  325 -- Constructors carry diagnostic payloads for parity with the existing
  326 -- 'ResumeError' style.
  327 data BranchError
  328   = BranchInvalidSourceId Text
  329     -- ^ The supplied source session id failed the traversal-safety guard
  330     -- (empty / contains @..@ or @\/@). The 'Text' is the offending id.
  331   | BranchSourceMissing FilePath
  332     -- ^ No @session.json@ at the expected path for the source session.
  333   | BranchSourceNotProvider
  334     -- ^ The source session is harness-backed; branching is provider-only.
  335   | BranchEntryNotFound Text
  336     -- ^ No transcript entry with the requested @_te_id@ exists in the
  337     -- source transcript. The 'Text' is the requested entry id.
  338   deriving stock (Show, Eq)
  339 
  340 -- | A client-supplied branch request: the source session to copy from and
  341 -- the transcript entry id at which to cut the (inclusive) prefix.
  342 data BranchSpec = BranchSpec
  343   { _bs_sourceSessionId :: Text
  344   , _bs_upToEntryId     :: Text
  345   }
  346   deriving stock (Show, Eq)
  347 
  348 -- | Wire shape: @{ "session_id": ..., "up_to_entry_id": ... }@. Defined
  349 -- here (not in the API module) so it is not an orphan instance.
  350 instance Aeson.FromJSON BranchSpec where
  351   parseJSON = Aeson.withObject "BranchSpec" $ \o ->
  352     BranchSpec
  353       <$> o Aeson..: "session_id"
  354       <*> o Aeson..: "up_to_entry_id"
  355 
  356 -- | The resolved ingredients for seeding a branched session, read entirely
  357 -- from the on-disk source session (never from client-supplied payloads).
  358 data BranchSeed = BranchSeed
  359   { _bseed_prefix       :: [TranscriptEntry]
  360     -- ^ Source transcript entries @[0..boundary]@ (inclusive), in order.
  361   , _bseed_sourceMeta   :: SessionMeta
  362     -- ^ Source session metadata, so the branch can inherit
  363     -- @_sm_kind@ / @_sm_model@ / @_sm_agent@.
  364   }
  365   deriving stock (Show, Eq)
  366 
  367 -- | Traversal-safety guard for a branch source session id. Mirrors
  368 -- @PureClaw.Frontend.API.isValidSessionId@; replicated locally to avoid an
  369 -- import cycle (the API module imports this module). Rejects the empty
  370 -- string and anything containing @..@ or @\/@.
  371 isValidBranchSourceId :: Text -> Bool
  372 isValidBranchSourceId sid
  373   | T.null sid = False
  374   | T.isInfixOf ".." sid = False
  375   | T.isInfixOf "/" sid = False
  376   | otherwise = True
  377 
  378 -- | Resolve a 'BranchSpec' against the on-disk source session under
  379 -- @baseDir@, producing a 'BranchSeed' or a typed 'BranchError'.
  380 --
  381 -- Validates (in order): the source id is traversal-safe, the source
  382 -- @session.json@ exists and decodes, the source is provider-backed, and
  383 -- the requested entry id is present in the source @transcript.jsonl@. On
  384 -- success returns the inclusive prefix @[0..boundary]@ and the source meta.
  385 --
  386 -- The frozen system prompt and last-used model ride in the copied
  387 -- transcript prefix (see @docs/session-branching.md@ §9), so the seed does
  388 -- not carry the source's @custom-prompt.md@; the fork inherits the agent
  389 -- identity from the source meta instead.
  390 --
  391 -- The prefix is read verbatim from the source transcript on disk — never
  392 -- from any caller-supplied payload — so a branch cannot inject history.
  393 resolveBranchSeed :: FilePath -> BranchSpec -> IO (Either BranchError BranchSeed)
  394 resolveBranchSeed baseDir bs
  395   | not (isValidBranchSourceId sourceId) =
  396       pure (Left (BranchInvalidSourceId sourceId))
  397   | otherwise = do
  398       let dir   = baseDir </> T.unpack sourceId
  399           metaP = dir </> "session.json"
  400           txP   = dir </> "transcript.jsonl"
  401       exists <- doesFileExist metaP
  402       if not exists
  403         then pure (Left (BranchSourceMissing metaP))
  404         else do
  405           raw <- LBS.readFile metaP
  406           case Aeson.eitherDecode' raw of
  407             Left _ -> pure (Left (BranchSourceMissing metaP))
  408             Right meta -> case _sm_kind (meta :: SessionMeta) of
  409               SkHarness _ -> pure (Left BranchSourceNotProvider)
  410               SkProvider _ -> do
  411                 entries <- readTranscriptEntries txP
  412                 case sliceInclusivePrefix (_bs_upToEntryId bs) entries of
  413                   Nothing -> pure (Left (BranchEntryNotFound (_bs_upToEntryId bs)))
  414                   Just prefix ->
  415                     pure (Right BranchSeed
  416                       { _bseed_prefix     = prefix
  417                       , _bseed_sourceMeta = meta
  418                       })
  419   where
  420     sourceId = _bs_sourceSessionId bs
  421 
  422 -- | Read and decode a @transcript.jsonl@ file into 'TranscriptEntry'
  423 -- values, in file (oldest-first) order. Missing files and undecodable
  424 -- lines yield @[]@ / skipped lines respectively. This is a read-only path,
  425 -- so it intentionally does not open a file transcript /handle/ (those are
  426 -- reserved for write sites that must reach the broker).
  427 readTranscriptEntries :: FilePath -> IO [TranscriptEntry]
  428 readTranscriptEntries path = do
  429   exists <- doesFileExist path
  430   if not exists
  431     then pure []
  432     else do
  433       eRaw <- try (LBS.readFile path) :: IO (Either IOException LBS.ByteString)
  434       case eRaw of
  435         Left _    -> pure []
  436         Right raw -> pure (mapMaybe Aeson.decode' (splitJsonlLines raw))
  437 
  438 -- | Split a lazy 'LBS.ByteString' into non-empty newline-delimited chunks.
  439 splitJsonlLines :: LBS.ByteString -> [LBS.ByteString]
  440 splitJsonlLines = filter (not . LBS.null) . LBSC.split '\n'
  441 
  442 -- | Return the inclusive prefix @[0..idx]@ up to and including the first
  443 -- entry whose @_te_id@ matches the target, or 'Nothing' if no such entry
  444 -- exists.
  445 sliceInclusivePrefix :: Text -> [TranscriptEntry] -> Maybe [TranscriptEntry]
  446 sliceInclusivePrefix target = go []
  447   where
  448     go _   [] = Nothing
  449     go acc (e:es)
  450       | _te_id e == target = Just (reverse (e : acc))
  451       | otherwise          = go (e : acc) es

  457 -- | Read all @session.json@ files under @baseDir@, optionally filtering
  458 -- by agent name, sorted by @last_active@ descending, and capped at
  459 -- @limit@ entries.
  460 --
  461 -- Silently skips entries that fail to decode so that a single corrupted
  462 -- metadata file does not hide the remaining sessions from the user.
  463 listSessions :: FilePath -> Maybe AgentName -> Int -> IO [SessionMeta]
  464 listSessions baseDir mAgent limit = do
  465   exists <- doesDirectoryExist baseDir
  466   if not exists
  467     then pure []
  468     else do
  469       entries <- listDirectory baseDir
  470       metas <- fmap catMaybes' (traverse (tryLoad baseDir) entries)
  471       let filtered = case mAgent of
  472             Nothing -> metas
  473             Just a  -> filter (\m -> _sm_agent m == Just a) metas
  474           sorted = sortOn (Down . _sm_lastActive) filtered
  475       pure (take limit sorted)
  476   where
  477     catMaybes' = foldr (\mx acc -> case mx of
  478                           Just x  -> x : acc
  479                           Nothing -> acc) []
  480 
  481 -- | Attempt to load @\<baseDir\>/\<name\>/session.json@. Returns
  482 -- 'Nothing' if the path is not a directory, the metadata file is
  483 -- missing, unreadable, or malformed.
  484 tryLoad :: FilePath -> FilePath -> IO (Maybe SessionMeta)
  485 tryLoad baseDir name = do
  486   let dir   = baseDir </> name
  487       metaP = dir </> "session.json"
  488   isDir <- doesDirectoryExist dir
  489   if not isDir
  490     then pure Nothing
  491     else do
  492       exists <- doesFileExist metaP
  493       if not exists
  494         then pure Nothing
  495         else do
  496           eBytes <- try (LBS.readFile metaP) :: IO (Either IOException LBS.ByteString)
  497           case eBytes of
  498             Left _     -> pure Nothing
  499             Right raw  -> case Aeson.decode' raw of
  500               Just m  -> pure (Just m)
  501               Nothing -> pure Nothing

  503 -- | Resolve a user-supplied reference (exact ID or prefix) to a concrete
  504 -- 'SessionId'. Exact matches win outright; otherwise the match set is
  505 -- computed by 'T.isPrefixOf' over the full ID string.
  506 resolveSessionRef
  507   :: FilePath
  508   -> Text
  509   -> IO (Either ResolveError SessionId)
  510 resolveSessionRef baseDir ref = do
  511   exists <- doesDirectoryExist baseDir
  512   if not exists
  513     then pure (Left NotFound)
  514     else do
  515       entries <- listDirectory baseDir
  516       -- Keep only entries that actually have a session.json — avoids
  517       -- matching stray directories created by unrelated tests or users.
  518       valid <- filterM hasMetadata entries
  519       let ids = map (SessionId . T.pack) valid
  520       if ref `elem` map (\(SessionId t) -> t) ids
  521         then pure (Right (SessionId ref))
  522         else case filter (\(SessionId t) -> ref `T.isPrefixOf` t) ids of
  523           []  -> pure (Left NotFound)
  524           [x] -> pure (Right x)
  525           xs  -> pure (Left (Ambiguous xs))
  526   where
  527     hasMetadata name = do
  528       let metaP = baseDir </> name </> "session.json"
  529       doesFileExist metaP
  530 
  531     -- Local strict filterM to avoid pulling in Control.Monad just for this.
  532     filterM _ []     = pure []
  533     filterM p (x:xs) = do
  534       keep <- p x
  535       rest <- filterM p xs
  536       pure (if keep then x : rest else rest)

  539 -- Runtime validation
  540 -- ----------------------------------------------------------------------------

  542 -- | Validate a 'SessionKind' against the currently-running harnesses.
  543 --
  544 -- * 'SkProvider' always resolves to @RuntimeOk TargetProvider@.
  545 -- * @'SkHarness' spec@ resolves to @RuntimeOk (TargetHarness name)@ if
  546 --   a harness with that name is present in the map (name derived from
  547 --   the harness flavour).
  548 -- * @'SkHarness' spec@ resolves to @RuntimeFallback TargetProvider msg@
  549 --   if the harness is absent, where @msg@ explains the fallback for
  550 --   logging at warn level.
  551 validateRuntime :: Map Text HarnessHandle -> SessionKind -> ResolvedRuntime
  552 validateRuntime _ (SkProvider _) = RuntimeOk TargetProvider
  553 validateRuntime harnesses (SkHarness spec) =
  554   let name = sessionKindHarnessName spec
  555   in if Map.member name harnesses
  556        then RuntimeOk (TargetHarness name)
  557        else let msg = "harness '" <> name <> "' is not running, falling back to provider"
  558             in RuntimeFallback TargetProvider msg
  559 
  560 -- | Extract the harness name from a 'HarnessSpec' by rendering its flavour.
  561 sessionKindHarnessName :: HarnessSpec -> Text
  562 sessionKindHarnessName spec = case _h_flavour spec of
  563   HClaudeCode -> "claude-code"
  564   HCodex      -> "codex"
  565   HOpenCode   -> "opencode"
  566   HHermes     -> "hermes"
  567   HPureClaw   -> "pureclaw"
  568   HCustom n   -> n

  570 -- | Resolve a resumed session's 'SessionKind' to a concrete
  571 -- 'MessageTarget' given the currently-running harness map, logging a
  572 -- warning if the recorded runtime is no longer available.
  573 --
  574 -- Wraps 'validateRuntime': on 'RuntimeFallback' the provided warning
  575 -- message is routed to @_lh_logWarn@ and the fallback target (always
  576 -- 'TargetProvider') is returned; on 'RuntimeOk' the target is returned
  577 -- without logging.
  578 resolveResumedTarget
  579   :: LogHandle
  580   -> Map Text HarnessHandle
  581   -> SessionKind
  582   -> IO MessageTarget
  583 resolveResumedTarget logger harnesses sk = case validateRuntime harnesses sk of
  584   RuntimeOk tgt -> pure tgt
  585   RuntimeFallback tgt warning -> do
  586     _lh_logWarn logger warning
  587     pure tgt

  593 -- | Flip '_sm_bootstrapConsumed' to 'True' in the session metadata
  594 -- 'IORef' and persist the change to @session.json@ via '_sh_save'.
  595 --
  596 -- Called by the agent loop the first time a provider response completes
  597 -- ('StreamDone'), via the one-shot callback installed on
  598 -- @_env_onFirstStreamDone@. Idempotent: re-invocations re-save the same

  600 markBootstrapConsumed :: SessionHandle -> IO ()
  601 markBootstrapConsumed sh = do
  602   atomicModifyIORef' (_sh_meta sh) $ \m ->
  603     (m { _sm_bootstrapConsumed = True }, ())
  604   _sh_save sh

  610 -- | Reasons 'setArchived' may fail. The session directory and transcript
  611 -- are never touched, so failure leaves on-disk state unchanged.
  612 data SetArchivedError
  613   = SetArchivedSessionMissing
  614     -- ^ @\<baseDir\>/\<sid\>/session.json@ does not exist.
  615   | SetArchivedParseFailed Text
  616     -- ^ session.json was unreadable or did not decode as 'SessionMeta'.
  617   deriving stock (Show, Eq)
  618 
  619 -- | Toggle 'SessionMeta._sm_archived' for a session that may or may not
  620 -- have a live 'SessionHandle'. Operates directly on the on-disk
  621 -- @session.json@: reads, updates the archive flag, writes back via the
  622 -- usual tmp-file + rename atomicity pattern.
  623 --
  624 -- Crucially, this is **display state only** — neither the session
  625 -- directory nor the transcript is removed. Unarchiving restores the
  626 -- session to "Recent Sessions" with all history intact.
  627 setArchived :: FilePath -> SessionId -> Bool -> IO (Either SetArchivedError ())
  628 setArchived baseDir sid archived =
  629   updateSessionMeta baseDir sid (\m -> m { _sm_archived = archived })
  630     SetArchivedSessionMissing SetArchivedParseFailed

  632 -- | Reasons 'setDescription' may fail. Same shape as 'SetArchivedError'.
  633 data SetDescriptionError
  634   = SetDescriptionSessionMissing
  635   | SetDescriptionParseFailed Text
  636   deriving stock (Show, Eq)

  638 -- | Set or clear the user-provided session description. Passing 'Just t'
  639 -- stores the description (trimmed of surrounding whitespace; an empty
  640 -- result is treated as 'Nothing' so callers can pass through what the
  641 -- user typed without preprocessing). Passing 'Nothing' clears it,
  642 -- letting display surfaces fall back to '_sm_autoSummary' or a
  643 -- transcript-derived snippet.
  644 setDescription :: FilePath -> SessionId -> Maybe Text -> IO (Either SetDescriptionError ())
  645 setDescription baseDir sid mDesc =
  646   let normalized = mDesc >>= \t -> let s = T.strip t in if T.null s then Nothing else Just s
  647   in  updateSessionMeta baseDir sid (\m -> m { _sm_description = normalized })
  648         SetDescriptionSessionMissing SetDescriptionParseFailed

  650 -- | Shared read-modify-write helper for the disk-only meta mutators.
  651 -- Reads @session.json@, applies @f@, and writes back via tmp-file +
  652 -- rename. @missing@ and @parseFail@ are the caller's error
  653 -- constructors so each public mutator returns its own error type.
  654 updateSessionMeta
  655   :: FilePath
  656   -> SessionId
  657   -> (SessionMeta -> SessionMeta)
  658   -> e
  659   -> (Text -> e)
  660   -> IO (Either e ())
  661 updateSessionMeta baseDir sid f missing parseFail = do
  662   let dir    = baseDir </> T.unpack (unSessionId sid)
  663       finalP = dir </> "session.json"
  664       tmpP   = finalP <> ".tmp"
  665   exists <- doesFileExist finalP
  666   if not exists
  667     then pure (Left missing)
  668     else do
  669       eBytes <- try (LBS.readFile finalP) :: IO (Either IOException LBS.ByteString)
  670       case eBytes of
  671         Left e -> pure (Left (parseFail (T.pack (show e))))
  672         Right raw -> case Aeson.eitherDecode' raw of
  673           Left  err  -> pure (Left (parseFail (T.pack err)))
  674           Right meta -> do
  675             LBS.writeFile tmpP (Aeson.encode (f meta))
  676             setFileMode tmpP 0o600
  677             renameFile tmpP finalP
  678             pure (Right ())

  684 -- | Load a bounded window of recent 'Message' values from a session's
  685 -- transcript, oldest-first.
  686 --
  687 -- @loadRecentMessages th maxCount maxTokens@ reads every recorded
  688 -- transcript entry via '_th_query', converts each to a text 'Message'
  689 -- (mapping 'Request' → 'User' and 'Response' → 'Assistant'), keeps at
  690 -- most the last @maxCount@, then walks that window from newest back to
  691 -- oldest accumulating an estimated token budget (computed as
  692 -- @T.length payload `div` 4@ — a rough heuristic, not a true tokenizer).
  693 -- Once the budget is exhausted no further messages are added.
  694 --
  695 -- If the transcript contains a compaction boundary (an entry whose
  696 -- metadata includes the 'compactionMetadataKey'), only entries from the
  697 -- last such boundary onward are considered.  The compaction entry itself
  698 -- is included — its payload carries the summary text that replaces the
  699 -- earlier conversation.  This ensures that after a gateway restart the
  700 -- agent resumes with the compacted view rather than double-counting
  701 -- messages that were folded into the summary.
  702 --
  703 -- Returns the surviving messages in chronological (oldest-first) order
  704 -- so they can be replayed directly into the context.
  705 loadRecentMessages :: TranscriptHandle -> Int -> Int -> IO [Message]
  706 loadRecentMessages th maxCount maxTokens = do
  707   entries <- _th_query th TranscriptFilter
  708     { _tf_harness   = Nothing
  709     , _tf_model     = Nothing
  710     , _tf_direction = Nothing
  711     , _tf_timeRange = Nothing
  712     , _tf_limit     = Nothing
  713     }
  714   let -- Trim to entries from the last compaction boundary onward.
  715       postCompaction = trimToLastCompaction entries
  716       total     = length postCompaction
  717       countWin  = if total > maxCount
  718                     then drop (total - maxCount) postCompaction
  719                     else postCompaction
  720       -- Walk newest→oldest, adding to the budget; stop at the first
  721       -- entry that would push us OVER the limit. Always include at
  722       -- least one entry if the window is non-empty, so a single
  723       -- oversized message is not silently dropped.
  724       reversed  = reverse countWin
  725       budgeted  = goBudget 0 True reversed
  726       goBudget _    _     [] = []
  727       goBudget used first (e:es) =
  728         let cost  = T.length (_te_payload e) `div` 4
  729             used' = used + cost
  730         in if used' > maxTokens && not first
  731              then []
  732              else e : goBudget used' False es
  733   pure (map entryToMessage (reverse budgeted))
  734   where
  735     -- Extract just the NEW message text from a recorded provider
  736     -- request/response JSON payload — NOT the entire envelope. Falls back
  737     -- to the raw payload when the JSON shape doesn't match (custom-harness
  738     -- transcripts, plain-text test fixtures), so unrelated callers keep
  739     -- working.
  740     --
  741     -- Why this is non-obvious: '_te_payload' is the full provider API
  742     -- body (correct for debugging — you can replay the exact request).
  743     -- But if we feed that ENTIRE envelope back into the next turn as
  744     -- "message text", we get recursive wrapping: turn 2's request
  745     -- contains turn 1's request-as-JSON, turn 3 contains turn 2's
  746     -- request-as-JSON (which contains turn 1's request-as-JSON), and so
  747     -- on. After a few turns the LLM sees a huge nested-escape-encoded
  748     -- blob instead of a clean conversation. See the matching parse in
  749     -- 'transcriptToMessages' in 'frontend/src/App.tsx' — both sides
  750     -- need to do the same extraction.
  751     entryToMessage e =
  752       let role = case _te_direction e of
  753             Request  -> User
  754             Response -> Assistant
  755           txt = case _te_direction e of
  756             Request  -> extractNewMessageText  (_te_payload e)
  757             Response -> extractAssistantText   (_te_payload e)
  758       in Message role [TextBlock txt]
  759 
  760 -- | Read the session's frozen system prompt from the transcript.
  761 --
  762 -- The system prompt is computed once on the first turn and then \"rides in
  763 -- the transcript\": every recorded provider 'Request' entry carries the
  764 -- prompt actually used at completion time under the top-level
  765 -- @system_prompt@ field of its 'CompletionRequest' payload
  766 -- (see 'PureClaw.Providers.Class' ToJSON). This helper recovers it so
  767 -- @doCompletion@ can reuse the frozen value on turns ≥ 2 instead of
  768 -- recomputing.
  769 --
  770 -- The result distinguishes three cases:
  771 --
  772 --   * @Nothing@        — there is NO prior 'Request' entry, i.e. this is the
  773 --                        first turn. The caller should compute and freeze a
  774 --                        prompt now.
  775 --   * @Just (Just p)@  — a prior request recorded a non-null
  776 --                        @system_prompt@; reuse @p@ verbatim.
  777 --   * @Just Nothing@   — a prior request recorded @system_prompt: null@ (or
  778 --                        omitted the field entirely); the frozen prompt is
  779 --                        null and MUST stay null (do not recompute).
  780 --
  781 -- Only 'Request'-direction entries are considered (never a 'Response'), and
  782 -- the LAST such entry wins. This is distinct from 'extractNewMessageText',
  783 -- which deliberately discards @system_prompt@.
  784 frozenSystemPrompt :: TranscriptHandle -> IO (Maybe (Maybe Text))
  785 frozenSystemPrompt th = do
  786   entries <- _th_query th TranscriptFilter
  787     { _tf_harness   = Nothing
  788     , _tf_model     = Nothing
  789     , _tf_direction = Just Request
  790     , _tf_timeRange = Nothing
  791     , _tf_limit     = Nothing
  792     }
  793   pure $ case reverse entries of
  794     []      -> Nothing
  795     (e : _) -> Just (extractSystemPrompt (_te_payload e))
  796 
  797 -- | The model recorded by the most-recent provider 'Request' entry, read from
  798 -- the structured @_te_model@ column (set by 'mkTranscriptProvider' from
  799 -- @unModelId model@ at completion time). Like 'frozenSystemPrompt', the
  800 -- session's model \"rides in the transcript\": this is the per-session
  801 -- fallback model used by @doCompletion@ when a @/send@ request omits an
  802 -- explicit model (§9.2).
  803 --
  804 --   * @Nothing@   — there is NO prior 'Request' entry, or the last one
  805 --                   recorded no model (@_te_model = Nothing@). The caller
  806 --                   should fall back to the global model.
  807 --   * @Just m@    — the last 'Request' entry recorded model @m@; reuse it.
  808 --
  809 -- Only 'Request'-direction entries are considered, and the LAST such entry
  810 -- wins. 'loadRecentMessages' deliberately discards @_te_model@, so this raw
  811 -- read is the only way to recover it.
  812 lastRequestModel :: TranscriptHandle -> IO (Maybe Text)
  813 lastRequestModel th = do
  814   entries <- _th_query th TranscriptFilter
  815     { _tf_harness   = Nothing
  816     , _tf_model     = Nothing
  817     , _tf_direction = Just Request
  818     , _tf_timeRange = Nothing
  819     , _tf_limit     = Nothing
  820     }
  821   pure $ case reverse entries of
  822     []      -> Nothing
  823     (e : _) -> _te_model e
  824 
  825 -- | Parse the top-level @system_prompt@ field from a recorded provider
  826 -- 'Request' payload. A present JSON string yields @Just t@; an explicit
  827 -- @null@, a missing key, or a non-string\/non-decodable payload all yield
  828 -- @Nothing@ (frozen-null semantics — see 'frozenSystemPrompt').
  829 extractSystemPrompt :: Text -> Maybe Text
  830 extractSystemPrompt raw = do
  831   Aeson.Object o <- decodeText raw
  832   case KM.lookup "system_prompt" o of
  833     Just (Aeson.String t) -> Just t
  834     _                     -> Nothing
  835 
  836 -- | Extract the most recent message's text from a recorded provider request
  837 -- JSON payload. Handles common provider shapes:
  838 --   Anthropic: @{"messages":[..., {"role":..., "content":[{"type":"text","text":<>}, ...]}], ...}@
  839 --   OpenAI:    @{"messages":[..., {"role":..., "content":<string>}], ...}@
  840 -- Earlier messages in the array are conversation history that's already
  841 -- represented by prior transcript entries, so we want only the LAST one.
  842 -- Returns the raw payload unchanged when the JSON shape doesn't match.
  843 extractNewMessageText :: Text -> Text
  844 extractNewMessageText raw = fromMaybe raw $ do
  845   Aeson.Object o    <- decodeText raw
  846   Aeson.Array msgs  <- KM.lookup "messages" o
  847   guard (not (V.null msgs))
  848   Aeson.Object m    <- pure (V.last msgs)
  849   case KM.lookup "content" m of
  850     Just (Aeson.String t)  -> pure t
  851     Just (Aeson.Array  cs) -> pure (extractTextBlocks cs)
  852     _                      -> Nothing
  853 
  854 -- | Extract the assistant's reply text from a recorded provider response
  855 -- JSON payload. Handles:
  856 --   Anthropic: @{"content":[{"type":"text","text":<>}, ...], ...}@
  857 --   OpenAI:    @{"choices":[{"message":{"content":<string>}}], ...}@
  858 -- Tool-use / tool_call blocks are dropped from the context replay — they
  859 -- are conversation flow control, not message text the LLM needs verbatim.
  860 -- Returns the raw payload unchanged when the JSON shape doesn't match.
  861 extractAssistantText :: Text -> Text
  862 extractAssistantText raw = fromMaybe raw $ do
  863   Aeson.Object o <- decodeText raw
  864   case KM.lookup "content" o of
  865     Just (Aeson.Array  cs) -> pure (extractTextBlocks cs)
  866     Just (Aeson.String t)  -> pure t
  867     _                      -> openaiChoiceText o
  868 
  869 -- | OpenAI-style fallback: messages live under @choices[0].message.content@.
  870 openaiChoiceText :: Aeson.Object -> Maybe Text
  871 openaiChoiceText o = do
  872   Aeson.Array choices <- KM.lookup "choices" o
  873   guard (not (V.null choices))
  874   Aeson.Object c <- pure (V.head choices)
  875   Aeson.Object m <- KM.lookup "message" c
  876   case KM.lookup "content" m of
  877     Just (Aeson.String t) -> pure t
  878     _                     -> Nothing
  879 
  880 -- | Concatenate the @text@ fields of an Anthropic-style typed-content array,
  881 -- joining with newlines. Non-text blocks (tool_use, etc.) are skipped.
  882 extractTextBlocks :: V.Vector Aeson.Value -> Text
  883 extractTextBlocks = T.intercalate "\n" . mapMaybe asText . V.toList
  884   where
  885     asText (Aeson.Object o) = case KM.lookup "text" o of
  886       Just (Aeson.String t) -> Just t
  887       _                     -> Nothing
  888     asText _ = Nothing

  890 decodeText :: Text -> Maybe Aeson.Value
  891 decodeText = Aeson.decode . LBS.fromStrict . TE.encodeUtf8

  893 -- | If any entry carries the compaction metadata key, return entries
  894 -- from the last such entry onward (inclusive).  Otherwise return the
  895 -- full list unchanged.
  896 trimToLastCompaction :: [TranscriptEntry] -> [TranscriptEntry]
  897 trimToLastCompaction entries =
  898   case lastCompactionIdx of
  899     Nothing -> entries
  900     Just i  -> drop i entries
  901   where
  902     isCompaction e = Map.member compactionMetadataKey (_te_metadata e)
  903     lastCompactionIdx = go Nothing 0 entries
  904     go acc _ []     = acc
  905     go acc n (e:es)
  906       | isCompaction e = go (Just n) (n + 1) es
  907       | otherwise      = go acc      (n + 1) es
