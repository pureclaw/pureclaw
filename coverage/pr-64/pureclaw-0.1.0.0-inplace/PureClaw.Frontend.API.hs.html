  119     -- 'Just', the @POST \/sessions\/\<id\>\/send@ completion path opens its

  121     -- provider Request\/Response entries reach subscribers in real time

  136     -- real implementation; the default stub returns @Left \"not wired\"@.

  165 -- | Per-origin WS subscriber counter. Lives at the same lifetime as the
  166 -- broker (constructed once in @startWithChannel@). The WS handler calls
  167 -- 'tryClaim' on upgrade (rejecting with 503 if the cap is reached) and
  168 -- 'release' on disconnect. Key is the normalized Origin string (see
  169 -- 'normalizeOrigin' in "PureClaw.Frontend.Stream").
  170 data StreamGuard = StreamGuard
  171   { _streamGuard_perOrigin    :: !(TVar (Map Text Int))
  172     -- ^ Live count per normalized Origin string.
  173   , _streamGuard_maxPerOrigin :: !Int
  174     -- ^ Maximum simultaneous WS subscriptions per origin.
  175   }

  177 -- | Construct a fresh 'StreamGuard' with the given per-origin cap.
  178 mkStreamGuard :: Int -> IO StreamGuard
  179 mkStreamGuard maxPer = do
  180   ref <- newTVarIO Map.empty
  181   pure StreamGuard
  182     { _streamGuard_perOrigin    = ref
  183     , _streamGuard_maxPerOrigin = maxPer
  184     }

  186 -- | Shared session-ID validation used by every endpoint that consumes a
  187 -- caller-supplied session id (HTTP @\/transcript@, @\/send@, @\/prompt@
  188 -- and the WS @focus@ op). Rejects @..@ and @\/@ to foreclose path
  189 -- traversal; rejects the empty string. Behavioural surface is intentionally
  190 -- the same across the HTTP and WS paths so D26's "shared helper" property
  191 -- holds: changing the rule here changes it everywhere.
  192 isValidSessionId :: Text -> Bool
  193 isValidSessionId sid
  194   | T.null sid = False
  195   | T.isInfixOf ".." sid = False
  196   | T.isInfixOf "/" sid = False
  197   | otherwise = True

  298 -- | WAI application handling @\/api\/*@ routes.
  299 apiApp :: FrontendEnv -> Application
  300 apiApp env req respond = do
  301   let method = requestMethod req
  302       path   = pathInfo req
  303   case (method, path) of
  304     ("GET", ["api", "harnesses"])            -> handleHarnesses env respond
  305     ("GET", ["api", "tabs"])                 -> handleListTabs env respond
  306     ("GET", ["api", "sessions", "recent"])   -> handleRecentSessions env respond
  307     ("GET", ["api", "sessions", "archived"]) -> handleArchivedSessions env respond
  308     ("GET", ["api", "sessions", sid, "transcript"]) ->
  309       handleTranscript env sid respond
  310     ("POST", ["api", "tabs", tidx, "close"]) ->
  311       handleCloseTab env tidx respond
  312     ("POST", ["api", "tabs", "new"]) ->
  313       handleNewTab env req respond
  314     ("POST", ["api", "sessions", "new"]) ->
  315       handleNewSessionGone respond
  316     ("POST", ["api", "sessions", sid, "send"]) ->
  317       handleSend env sid req respond
  318     ("PUT", ["api", "sessions", sid, "prompt"]) ->
  319       handleSetPrompt env sid req respond
  320     ("POST", ["api", "sessions", sid, "archive"]) ->
  321       handleSetArchived env sid True respond
  322     ("POST", ["api", "sessions", sid, "unarchive"]) ->
  323       handleSetArchived env sid False respond
  324     ("PUT", ["api", "sessions", sid, "description"]) ->
  325       handleSetDescription env sid req respond
  326     ("GET", ["api", "agents"])               -> handleAgents env respond
  327     ("GET", ["api", "providers"])            -> handleListProviders env respond
  328     ("GET", ["api", "providers", name, "models"]) ->
  329       handleListProviderModels env name respond
  330     _                                        -> respondNotFound respond

  338 -- | Probe a single harness to determine its activity.
  339 -- Checks process status first, then does a quick screen capture
  340 -- to distinguish thinking from idle.
  341 probeHarness :: (Text, HarnessHandle) -> IO HarnessInfo
  342 probeHarness (name, hh) = do
  343   st <- _hh_status hh
  344   activity <- case st of
  345     HarnessExited _ -> pure HarnessStopped
  346     HarnessRunning  -> probeActivity (_hh_session hh) name
  347   pure HarnessInfo
  348     { _hi_name     = name
  349     , _hi_activity = activity
  350     }

  352 -- | Capture the tmux window and check if the harness is idle or thinking.
  353 probeActivity :: Text -> Text -> IO HarnessActivity
  354 probeActivity tmuxSession windowName = do
  355   let target = tmuxSession <> ":" <> extractWindowIdx windowName
  356   result <- try @SomeException $ captureWindow target 50
  357   case result of
  358     Left _        -> pure HarnessIdle
  359     Right capture -> do
  360       let screenText = TE.decodeUtf8Lenient capture
  361       pure $ if isIdle screenText then HarnessIdle else HarnessThinking

  363 -- | Extract the window index suffix from a harness name like "claude-code-0".
  364 extractWindowIdx :: Text -> Text
  365 extractWindowIdx name =
  366   case T.splitOn "-" name of
  367     parts | length parts >= 2 -> last parts
  368     _                         -> "0"

  376 -- | Close a tab by index via the '_fe_closeTab' callback.
  377 handleCloseTab :: FrontendEnv -> Text -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  378 handleCloseTab env tidxText respond =
  379   case reads (T.unpack tidxText) :: [(Int, String)] of
  380     [(idx, "")] -> do
  381       result <- _fe_closeTab env idx
  382       case result of
  383         Right () -> do
  384           broadcastLists env
  385           respond $ jsonResponse status200 (object ["closed" .= True])
  386         Left errMsg ->
  387           respond $ jsonResponse status404 (object ["error" .= errMsg])
  388     _ ->
  389       respond $ jsonResponse status400 (object ["error" .= ("Invalid tab index" :: Text)])

  391 handleRecentSessions :: FrontendEnv -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  392 handleRecentSessions env respond = do
  393   let limit   = _fe_recentLimit env
  394       baseDir = _fe_sessionsDir env
  395   -- Over-fetch so the empty-transcript filter below can't shrink the
  396   -- visible count: listSessions already does a full directory scan and
  397   -- sorts the whole metadata set, so a generous bound is essentially
  398   -- free. Empty / archived sessions are dropped here, not erased on
  399   -- disk — archiving is purely a UI hint controlled by the user.
  400   metas    <- listSessions baseDir Nothing (limit * 3)
  401   -- Collect session IDs that are currently shown in an active tab so
  402   -- the same session does not appear in both "Active Tabs" and
  403   -- "Recent Sessions" simultaneously.
  404   tabs <- _fe_listTabs env
  405   let activeTabSids = [s | TabSnapshot { _ts_sessionId = Just s } <- tabs]
  406       notInTab m    = unSessionId (_sm_id m) `notElem` activeTabSids
  407       visible       = filter (\m -> not (_sm_archived m) && notInTab m) metas
  408   nonEmpty <- filterM (hasTranscriptEntries baseDir) visible
  409   let chosen = take limit nonEmpty
  410   -- Read a first-message snippet per session (bounded read per file).
  411   -- This is the cheap display fallback when no user description and no
  412   -- model-generated summary exist yet.
  413   snippets <- traverse (firstMessageSnippet baseDir) chosen
  414   let infos = zipWith toSessionInfo chosen snippets
  415   respond $ jsonResponse status200 infos

  417 -- | Return all archived sessions, sorted by last-active descending.
  418 handleArchivedSessions :: FrontendEnv -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  419 handleArchivedSessions env respond = do
  420   let baseDir = _fe_sessionsDir env
  421   -- Load all sessions with a generous limit; filter to archived-only.
  422   metas <- listSessions baseDir Nothing 1000
  423   let archived = filter _sm_archived metas
  424       infos = map (`toSessionInfo` Nothing) archived
  425   respond $ jsonResponse status200 infos

  431 -- | Compute the same sidebar payloads (tabs, recent sessions, archived
  432 -- sessions) that the three HTTP endpoints return, pre-serialized as a
  433 -- single JSON 'Value'. The WS handler sends this on connect and the
  434 -- mutation handlers call 'broadcastLists' to push it after every change.
  435 computeListsSnapshot :: FrontendEnv -> IO Aeson.Value
  436 computeListsSnapshot env = do
  437   let limit   = _fe_recentLimit env
  438       baseDir = _fe_sessionsDir env
  439   tabs <- _fe_listTabs env
  440   -- Recent sessions: exclude archived, exclude active-tab-backed, exclude
  441   -- empty transcripts, enrich with first-message snippet — same pipeline
  442   -- as handleRecentSessions.
  443   allMetas <- listSessions baseDir Nothing (limit * 3)
  444   let activeTabSids = [s | TabSnapshot { _ts_sessionId = Just s } <- tabs]
  445       notInTab m    = unSessionId (_sm_id m) `notElem` activeTabSids
  446       visible       = filter (\m -> not (_sm_archived m) && notInTab m) allMetas
  447   nonEmpty <- filterM (hasTranscriptEntries baseDir) visible
  448   let chosen = take limit nonEmpty
  449   snippets <- traverse (firstMessageSnippet baseDir) chosen
  450   let recentInfos = zipWith toSessionInfo chosen snippets
  451   -- Archived: all archived sessions, sorted by lastActive descending
  452   -- (listSessions already sorts; we just filter).
  453   archivedMetas <- listSessions baseDir Nothing 1000
  454   let archivedChosen = filter _sm_archived archivedMetas
  455   archivedSnippets <- traverse (firstMessageSnippet baseDir) archivedChosen
  456   let archivedInfos = zipWith toSessionInfo archivedChosen archivedSnippets
  457   pure $ object
  458     [ "type"              .= ("lists" :: Text)
  459     , "tabs"              .= tabs
  460     , "recentSessions"    .= recentInfos
  461     , "archivedSessions"  .= archivedInfos
  462     ]

  464 -- | Compute and broadcast a lists snapshot to all WS subscribers.
  465 -- No-op when the broker is absent (test / legacy paths).
  466 broadcastLists :: FrontendEnv -> IO ()
  467 broadcastLists env = case _fe_broker env of
  468   Nothing     -> pure ()
  469   Just broker -> do
  470     v <- computeListsSnapshot env
  471     _streamBroker_publish broker (ListsSnapshot v)

  477 -- | Toggle the archive flag on a session. The 'archived' argument is
  478 -- the **target** value (True for /archive, False for /unarchive), so
  479 -- the routes are idempotent: archiving an already-archived session is
  480 -- a no-op success. The session directory and transcript are never
  481 -- removed — see Session.Handle.setArchived for the disk-side contract.
  482 handleSetArchived
  483   :: FrontendEnv
  484   -> Text
  485   -> Bool
  486   -> (Response -> IO ResponseReceived)
  487   -> IO ResponseReceived
  488 handleSetArchived env sidText archived respond = do
  489   let sid = SessionId sidText
  490   result <- setArchived (_fe_sessionsDir env) sid archived
  491   case result of
  492     Right () -> do
  493       broadcastLists env
  494       respond $ jsonResponse status200 (object ["archived" .= archived])
  495     Left SetArchivedSessionMissing ->
  496       respondNotFound respond
  497     Left (SetArchivedParseFailed msg) ->
  498       respond $ jsonResponse status500 (object ["error" .= msg])

  500 -- | Update the user-provided description on a session.
  501 -- Body shape: @{"description": "..."}@ or @{"description": null}@ (clears).
  502 -- Trims whitespace; an all-whitespace string clears the field.
  503 handleSetDescription
  504   :: FrontendEnv
  505   -> Text
  506   -> Request
  507   -> (Response -> IO ResponseReceived)
  508   -> IO ResponseReceived
  509 handleSetDescription env sidText req respond = do
  510   body <- strictRequestBody req
  511   case Aeson.eitherDecode' body of
  512     Left err -> respond $ jsonResponse status400 (object ["error" .= T.pack err])
  513     Right (DescBody mDesc) -> do
  514       result <- setDescription (_fe_sessionsDir env) (SessionId sidText) mDesc
  515       case result of
  516         Right () -> do
  517           broadcastLists env
  518           respond $ jsonResponse status200 (object ["description" .= mDesc])
  519         Left SetDescriptionSessionMissing ->
  520           respondNotFound respond
  521         Left (SetDescriptionParseFailed msg) ->
  522           respond $ jsonResponse status500 (object ["error" .= msg])

  524 -- | Body parser that accepts @{"description": <string|null>}@; absent
  525 -- or non-string values normalise to 'Nothing' (which clears the field).
  526 newtype DescBody = DescBody (Maybe Text)
  527 instance FromJSON DescBody where
  528   parseJSON = Aeson.withObject "DescBody" $ \o -> DescBody <$> o .:? "description"

  530 -- | Check whether a session has at least one transcript entry.
  531 hasTranscriptEntries :: FilePath -> SessionMeta -> IO Bool
  532 hasTranscriptEntries baseDir meta = do
  533   let path = baseDir </> T.unpack (unSessionId (_sm_id meta)) </> "transcript.jsonl"
  534   exists <- doesFileExist path
  535   if not exists
  536     then pure False
  537     else do
  538       size <- getFileSize path
  539       pure (size > 0)

  554 -- | Cheap fallback for display: read just the first line of the
  555 -- session's @transcript.jsonl@, decode it as a 'TranscriptEntry', and
  556 -- extract a short snippet of the first user message. Returns 'Nothing'
  557 -- when there's no transcript, the first entry isn't a request, or any
  558 -- decoding step fails. The returned string is at most
  559 -- 'snippetCharBudget' chars with newlines normalised to spaces.
  560 --
  561 -- Cost is one bounded read per call (we don't load the whole transcript).
  562 firstMessageSnippet :: FilePath -> SessionMeta -> IO (Maybe Text)
  563 firstMessageSnippet baseDir meta = do
  564   let path = baseDir </> T.unpack (unSessionId (_sm_id meta)) </> "transcript.jsonl"
  565   result <- try @IOException $ withFile path ReadMode BSC.hGetLine
  566   case result of
  567     Left _    -> pure Nothing
  568     Right line -> case Aeson.eitherDecodeStrict' line :: Either String TranscriptEntry of
  569       Left _ -> pure Nothing
  570       Right entry
  571         | _te_direction entry /= Request -> pure Nothing
  572         | otherwise -> pure (snippetFromPayload (_te_payload entry))
  573 
  574 snippetCharBudget :: Int
  575 snippetCharBudget = 120
  576 
  577 -- | Extract a display snippet from a request payload. Two shapes:
  578 -- (1) JSON object with a "messages" array (provider request) — pull
  579 --     text out of the first message; (2) plain text (harness send) —
  580 --     use the payload directly. Trimmed, newline-normalised, and
  581 --     truncated to 'snippetCharBudget' characters.
  582 snippetFromPayload :: Text -> Maybe Text
  583 snippetFromPayload raw = trimAndTruncate <$>
  584   case Aeson.decodeStrict (TE.encodeUtf8 raw) of
  585     Just (Aeson.Object o)
  586       | Just (Aeson.Array msgs) <- KM.lookup "messages" o
  587       , not (V.null msgs)
  588       -> messageText (V.unsafeHead msgs)
  589     _ -> Just raw
  590   where
  591     -- Anthropic-style message content: either a plain string or an
  592     -- array of {type:"text", text:"..."} blocks. We just want a
  593     -- human-readable lead.
  594     messageText :: Aeson.Value -> Maybe Text
  595     messageText (Aeson.Object m) = case KM.lookup "content" m of
  596       Just (Aeson.String s) -> Just s
  597       Just (Aeson.Array bs) -> Just (T.intercalate " " [t | Aeson.Object b <- V.toList bs
  598                                                           , Just (Aeson.String t) <- [KM.lookup "text" b]])
  599       _                     -> Nothing
  600     messageText _ = Nothing
  601 
  602     trimAndTruncate :: Text -> Text
  603     trimAndTruncate t =
  604       let normalized = T.unwords (T.words t)  -- collapse whitespace incl. newlines
  605       in  if T.length normalized > snippetCharBudget
  606             then T.take (snippetCharBudget - 1) normalized <> "\x2026"
  607             else normalized

  728   { _ntr_kind       :: TabKind

  742 -- | Parse a 'TabKind' from the JSON envelope used by POST /api/tabs/new.
  743 --
  744 -- Two top-level tags:
  745 --
  746 --   * @\"session\"@ → expects a @\"session_kind\"@ sub-object whose
  747 --     shape is the existing 'SessionKind' JSON (tag-discriminated:
  748 --     @\"provider\"@ or @\"harness\"@).
  749 --   * @\"raw_shell\"@ → expects a @\"backend\"@ sub-object whose shape
  750 --     is the existing 'TerminalBackend' JSON (tag-discriminated).
  751 parseTabKind :: Aeson.Value -> AesonTypes.Parser TabKind
  752 parseTabKind = Aeson.withObject "TabKind" $ \o -> do
  753   tag <- o .: "tag" :: AesonTypes.Parser Text
  754   case tag of
  755     "session" -> do
  756       sk <- o .: "session_kind"
  757       TkSession <$> Aeson.parseJSON sk
  758     "raw_shell" -> do
  759       be <- o .: "backend"
  760       TkRawShell <$> Aeson.parseJSON be
  761     other -> fail ("unknown TabKind tag: " ++ T.unpack other)

  777 -- | Handle POST /api/tabs/new.
  778 --
  779 -- Creates a new tab (session-backed or raw shell), enforces the maxTabs
  780 -- limit (A8), and returns a JSON response with the tab index, session
  781 -- ID (if any), and kind description.
  782 handleNewTab :: FrontendEnv -> Request -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  783 handleNewTab env req respond = do
  784   body <- consumeBody req
  785   case Aeson.eitherDecode body of
  786     Left err ->
  787       respond $ jsonResponse status400
  788         (object ["error" .= ("Invalid JSON: " <> T.pack err)])
  789     Right (NewTabRequest tabKind mBranch) ->
  790       case mBranch of
  791         Nothing -> do
  792           -- A8: enforce maxTabs limit
  793           curCount <- readIORef (_fe_tabCount env)
  794           if curCount >= _fe_maxTabs env
  795             then respond $ jsonResponse status409
  796                    (object ["error" .= ("maximum tab count reached" :: Text)])
  797             else createTab env tabKind Nothing respond
  798         Just branchSpec ->
  799           -- Branch path: validate the request kind and resolve the seed
  800           -- BEFORE the maxTabs gate / createTab so a failed branch never
  801           -- consumes a tab slot or creates a directory (D5).
  802           case tabKind of
  803             TkSession (SkProvider _) -> do
  804               seedResult <- resolveBranchSeed (_fe_sessionsDir env) branchSpec
  805               case seedResult of
  806                 Left err -> respond (branchErrorResponse err)
  807                 Right seed -> do
  808                   curCount <- readIORef (_fe_tabCount env)
  809                   if curCount >= _fe_maxTabs env
  810                     then respond $ jsonResponse status409
  811                            (object ["error" .= ("maximum tab count reached" :: Text)])
  812                     else createTab env tabKind (Just seed) respond
  813             _ ->
  814               respond $ jsonResponse status400
  815                 (object ["error" .= ("branch target must be a provider session" :: Text)])
  816 
  817 -- | Map a 'BranchError' to its HTTP response. Invalid/traversal source id
  818 -- and non-provider sources are client errors (400); a missing source
  819 -- session or a missing entry id are not-found (404).
  820 branchErrorResponse :: BranchError -> Response
  821 branchErrorResponse err = case err of
  822   BranchInvalidSourceId sid -> jsonResponse status400
  823     (object ["error" .= ("invalid branch source id: " <> sid)])
  824   BranchSourceNotProvider -> jsonResponse status400
  825     (object ["error" .= ("branch source is not a provider session" :: Text)])
  826   BranchSourceMissing _ -> jsonResponse status404
  827     (object ["error" .= ("branch source session not found" :: Text)])
  828   BranchEntryNotFound eid -> jsonResponse status404
  829     (object ["error" .= ("branch source entry not found: " <> eid)])

  831 -- | Actually create the tab and return a response. With @'Just' seed@ the
  832 -- session is seeded from a branch source (see 'createBranchedSession');
  833 -- with 'Nothing' the behaviour is identical to a fresh New-tab session.
  834 createTab :: FrontendEnv -> TabKind -> Maybe BranchSeed -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  835 createTab env tabKind mSeed respond = do
  836   curCount <- readIORef (_fe_tabCount env)
  837   -- Bump tab count
  838   writeIORef (_fe_tabCount env) (curCount + 1)
  839   case tabKind of
  840     TkSession sk -> do
  841       -- Create a session for session-backed tabs
  842       now <- getCurrentTime
  843       mModel <- readIORef (_fe_model env)
  844       let modelText = maybe "" unModelId mModel
  845           sid = newSessionId Nothing now
  846           -- Carry the agent name through from the request: the
  847           -- frontend uses it as the display name for the session
  848           -- (e.g. the chat input placeholder), and the agent's
  849           -- definition was already used to build the system prompt.
  850           -- Without this, the metadata loses the association and
  851           -- recent-sessions UI falls back to the session id.
  852           agentName = case sk of
  853             SkProvider ps -> _ps_agent ps
  854             SkHarness  _  -> Nothing
  855           -- For a branch, inherit _sm_kind / _sm_model / _sm_agent from
  856           -- the source meta so the branch's sidebar row matches its
  857           -- parent; otherwise use the request kind + global model.
  858           (metaKind, metaModel, metaAgent) = case mSeed of
  859             Just seed ->
  860               let srcMeta = _bseed_sourceMeta seed
  861               in ( _sm_kind srcMeta, _sm_model srcMeta, _sm_agent srcMeta )
  862             Nothing -> (sk, modelText, agentName)
  863           meta = SessionMeta
  864             { _sm_id                = sid
  865             , _sm_agent             = metaAgent
  866             , _sm_kind              = metaKind
  867             , _sm_model             = metaModel
  868             , _sm_channel           = "web"
  869             , _sm_createdAt         = now
  870             , _sm_lastActive        = now
  871             , _sm_bootstrapConsumed = True
  872             , _sm_archived          = False
  873             , _sm_description       = Nothing
  874             , _sm_autoSummary       = Nothing
  875             }
  876       sh <- mkSessionHandle (_fe_broker env) (_fe_logger env) (_fe_sessionsDir env) meta
  877       _sh_save sh
  878       -- For a branch, seed the new transcript with a verbatim copy of the
  879       -- source prefix (preserving _te_id and order). The frozen system prompt
  880       -- and last-used model ride in the copied transcript (see
  881       -- docs/session-branching.md §9), so the fork does NOT copy the source's
  882       -- custom-prompt.md; it inherits the agent identity via _sm_agent above.
  883       case mSeed of
  884         Just seed -> mapM_ (_th_record (_sh_transcript sh)) (_bseed_prefix seed)
  885         Nothing -> pure ()
  886       -- Publish the new-session signal to the live stream broker (D18). The
  887       -- sidebar uses this to render the session row without polling. The
  888       -- no-broker path is intentional ('Nothing' preserves the legacy
  889       -- behaviour for one-off scripts and tests). Previously this lived in
  890       -- the now-410'd POST /api/sessions/new handler; it moved here when
  891       -- the session-tab unification merged on main.
  892       case _fe_broker env of
  893         Just broker ->
  894           _streamBroker_publish broker
  895             (ActivityChanged (_sm_id meta) (SaSessionCreated meta))
  896         Nothing -> pure ()
  897       broadcastLists env
  898       respond $ jsonResponse status200 NewTabResponse
  899         { _ntresp_tabIndex  = curCount
  900         , _ntresp_sessionId = Just (unSessionId sid)
  901         , _ntresp_kind      = tabKindLabel tabKind
  902         }
  903     TkRawShell _backend -> do
  904       respond $ jsonResponse status200 NewTabResponse
  905         { _ntresp_tabIndex  = curCount
  906         , _ntresp_sessionId = Nothing
  907         , _ntresp_kind      = tabKindLabel tabKind
  908         }

  913 tabKindLabel (TkSession (SkHarness _))  = "harness"
  914 tabKindLabel (TkRawShell _)             = "raw_shell"

  934 -- | Set or replace the custom prompt for a session (WU6, §9.3).
  935 --
  936 -- Enforces the @custom-prompt.md ⊕ agent@ invariant: a session may have a
  937 -- per-session agent /or/ a user-supplied custom prompt, never both. Setting
  938 -- a custom prompt therefore:
  939 --
  940 --   * clears @_sm_agent@ (the custom prompt supersedes the agent);
  941 --   * stores an optionally-provided @name@ in @_sm_description@ (a friendly
  942 --     title), /not/ as the agent;
  943 --   * leaves all other metadata fields untouched.
  944 --
  945 -- __Ordering / atomicity:__ the metadata is read, updated, and written
  946 -- (atomic tmp + rename) /before/ @custom-prompt.md@ is written. If the
  947 -- existing @session.json@ is missing or undecodable, no @custom-prompt.md@
  948 -- is written, so a meta failure can never leave a custom prompt sitting
  949 -- alongside a still-set agent.
  950 handleSetPrompt :: FrontendEnv -> Text -> Request -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  951 handleSetPrompt env sid req respond = do
  952   if not (isValidSessionId sid)
  953     then respond $ jsonResponse status400 (object ["error" .= ("Invalid session ID" :: Text)])
  954     else do
  955       let sessionDir = _fe_sessionsDir env </> T.unpack sid
  956           promptPath = sessionDir </> "custom-prompt.md"
  957           metaPath   = sessionDir </> "session.json"
  958       body <- consumeBody req
  959       case Aeson.eitherDecode body of
  960         Left _ ->
  961           respond $ jsonResponse status400 (object ["error" .= ("Invalid JSON: expected {\"prompt\": \"...\"}" :: Text)])
  962         Right obj -> case Map.lookup ("prompt" :: Text) (obj :: Map.Map Text Text) of
  963           Nothing ->
  964             respond $ jsonResponse status400 (object ["error" .= ("Missing 'prompt' field" :: Text)])
  965           Just prompt -> do
  966             -- An optionally-provided friendly title, trimmed; an
  967             -- all-whitespace value is treated as "no description".
  968             let mDescription = do
  969                   name <- Map.lookup "name" obj
  970                   let trimmed = T.strip name
  971                   if T.null trimmed then Nothing else Just trimmed
  972             -- Step 1: read + update + atomically rewrite session.json FIRST.
  973             -- Clear the agent (invariant) and set the description from name.
  974             metaExists <- doesFileExist metaPath
  975             if not metaExists
  976               then respondNotFound respond
  977               else do
  978                 eBytes <- try (LBS.readFile metaPath) :: IO (Either IOException LBS.ByteString)
  979                 case eBytes of
  980                   Left e ->
  981                     respond $ jsonResponse status500
  982                       (object ["error" .= T.pack (show e)])
  983                   Right raw -> case Aeson.eitherDecode' raw of
  984                     Left err ->
  985                       respond $ jsonResponse status500
  986                         (object ["error" .= T.pack err])
  987                     Right meta -> do
  988                       let updated = (meta :: SessionMeta)
  989                             { _sm_agent       = Nothing
  990                             , _sm_description = mDescription
  991                             }
  992                           tmpP = metaPath <> ".tmp"
  993                       LBS.writeFile tmpP (Aeson.encode updated)
  994                       renameFile tmpP metaPath
  995                       -- Step 2: only after the meta update succeeds do we
  996                       -- write the custom prompt file. Now the agent is
  997                       -- guaranteed cleared, so no contradiction can exist.
  998                       TIO.writeFile promptPath prompt
  999                       respond $ jsonResponse status200 (object ["ok" .= True])

 1014 -- | Send a user message to a session and get a completion.
 1015 handleSend :: FrontendEnv -> Text -> Request -> (Response -> IO ResponseReceived) -> IO ResponseReceived
 1016 handleSend env sid req respond = do
 1017   -- Validate session ID (shared helper — D26)
 1018   if not (isValidSessionId sid)
 1019     then respond $ jsonResponse status400 (object ["error" .= ("Invalid session ID" :: Text)])
 1020     else do
 1021       let sessionDir = _fe_sessionsDir env </> T.unpack sid
 1022           transcriptPath = sessionDir </> "transcript.jsonl"
 1023       exists <- doesFileExist transcriptPath
 1024       if not exists
 1025         then respond $ jsonResponse status404 (object ["error" .= ("Session not found" :: Text)])
 1026         else do
 1027           -- Parse request body
 1028           body <- consumeBody req
 1029           case Aeson.eitherDecode body of
 1030             Left _ ->
 1031               respond $ jsonResponse status400 (object ["error" .= ("Invalid JSON: expected {\"message\": \"...\"}" :: Text)])
 1032             Right (SendRequest userText reqModel) -> do
 1033               mProvider <- readIORef (_fe_provider env)
 1034               mModel <- readIORef (_fe_model env)
 1035               case (mProvider, mModel) of
 1036                 (Nothing, _) ->
 1037                   respond $ jsonResponse status503 (object ["error" .= ("No provider configured" :: Text)])
 1038                 (_, Nothing) ->
 1039                   respond $ jsonResponse status503 (object ["error" .= ("No model configured" :: Text)])
 1040                 (Just provider, Just model) -> do
 1041                   result <- try @SomeException $
 1042                     doCompletion env (SessionId sid) provider reqModel model userText transcriptPath
 1043                   case result of
 1044                     Left e -> do
 1045                       _lh_logError (_fe_logger env) $ "Send error: " <> T.pack (show e)
 1046                       respond $ jsonResponse status500 (object ["error" .= ("Completion failed" :: Text)])
 1047                     Right respText ->
 1048                       respond $ jsonResponse status200 (object ["response" .= respText])

 1050 -- | Run a completion: load context, send to provider, execute any tool
 1051 -- calls the model emits, and loop until the model produces a turn with
 1052 -- no tool calls (or the iteration cap is hit). The transcript handle is
 1053 -- opened via 'mkBroadcastingFileTranscriptHandle' so each iteration's
 1054 -- provider-wrapper Request\/Response entries reach the broker (D25) when
 1055 -- one is configured on the 'FrontendEnv'.
 1056 doCompletion
 1057   :: FrontendEnv
 1058   -> SessionId
 1059   -> SomeProvider
 1060   -> Maybe Text  -- ^ Request-supplied model (WU5, §9.2), if any.
 1061   -> ModelId     -- ^ Global fallback model (the @_fe_model@ IORef value).
 1062   -> Text
 1063   -> FilePath
 1064   -> IO Text
 1065 doCompletion env sid provider reqModel fallbackModel userText transcriptPath = do
 1066   -- Open a (possibly broadcasting) transcript handle for recording.
 1067   th <- mkBroadcastingFileTranscriptHandle
 1068           (_fe_broker env) sid (_fe_logger env) transcriptPath
 1069   -- Resolve the per-session model with precedence (§9.2): the request's
 1070   -- explicit model, else the most-recent model recorded in the transcript
 1071   -- (the "rides in the transcript" fallback — read from the structured
 1072   -- '_te_model' column, which 'loadRecentMessages' discards), else the
 1073   -- global '_fe_model' fallback. The chosen 'ModelId' flows into
 1074   -- 'mkTranscriptProvider' below so '_te_model' is recorded automatically.
 1075   -- A blank/whitespace request model (e.g. {"model":""}) is treated as
 1076   -- absent so it falls through to the transcript/global fallback rather
 1077   -- than sending an empty model id to the provider.
 1078   model <- case reqModel of
 1079     Just m | not (T.null (T.strip m)) -> pure (ModelId m)
 1080     _ -> do
 1081       mTranscriptModel <- lastRequestModel th
 1082       pure $ maybe fallbackModel ModelId mTranscriptModel
 1083   -- Load recent messages for context
 1084   history <- loadRecentMessages th 50 100000
 1085   -- The system prompt is FROZEN per session: computed once on the first
 1086   -- turn and thereafter replayed from the transcript unchanged (§9.1).
 1087   -- 'frozenSystemPrompt' reads the last recorded Request's system_prompt:
 1088   --   * Just frozen ⇒ a prior turn exists ⇒ reuse it verbatim (including a
 1089   --     frozen-null, which must stay null — do NOT recompute, do NOT
 1090   --     re-read custom-prompt.md).
 1091   --   * Nothing      ⇒ first turn ⇒ compute with precedence
 1092   --                    custom-prompt.md → per-session agent → global.
 1093   let sessionDir = takeDirectory transcriptPath
 1094   frozen <- frozenSystemPrompt th
 1095   systemPrompt <- case frozen of
 1096     Just p  -> pure p
 1097     Nothing -> computeFirstTurnPrompt env sessionDir
 1098   -- Build context with system prompt and history
 1099   let ctx0 = foldl (flip addMessage) (emptyContext systemPrompt) history
 1100       userMsg = textMessage User userText
 1101       ctx = addMessage userMsg ctx0
 1102       -- Wrap provider with transcript logging
 1103       provider' = mkTranscriptProvider th (unModelId model) provider
 1104       reg = _fe_registry env
 1105       tools = registryDefinitions reg
 1106       cap = _fe_maxToolIterations env
 1107   -- Publish session-thinking activity events so the FE sidebar / chat
 1108   -- area can show a live indicator while the provider call is in flight.
 1109   -- For harness-backed sessions, the equivalent signal comes from the
 1110   -- 2s tmux probe loop in 'PureClaw.Frontend.ActivityProbe'; provider
 1111   -- sessions don't go through the harness map, so the only place we
 1112   -- can hook this is at the request boundary in 'doCompletion'.
 1113   -- 'bracket_' guarantees the Idle event fires even if 'complete' throws.
 1114   let publishStatus s = case _fe_broker env of
 1115         Just broker -> _streamBroker_publish broker
 1116           (ActivityChanged sid (SaHarnessStatus s))
 1117         Nothing -> pure ()
 1118   result <- bracket_
 1119               (publishStatus HarnessThinking)
 1120               (publishStatus HarnessIdle)
 1121               (runCompletionLoop env provider' model tools reg cap ctx)
 1122   _th_flush th
 1123   _th_close th
 1124   pure result

 1126 -- | Section-character limit used when rendering a per-session agent's
 1127 -- system prompt, matching the limit used elsewhere for web/session agent
 1128 -- rendering (e.g. the @/agent@ slash command).
 1129 agentPromptSectionLimit :: Int
 1130 agentPromptSectionLimit = 8000
 1131 
 1132 -- | Compute the system prompt for the FIRST turn of a session (called only
 1133 -- when 'frozenSystemPrompt' returns 'Nothing'). Precedence (§9.1):
 1134 --
 1135 --   1. @custom-prompt.md@ in the session dir (user override), if present.
 1136 --   2. The per-session agent (from @session.json@'s @_sm_agent@), rendered
 1137 --      with 'composeAgentPromptWithBootstrap' honoring the session's
 1138 --      @_sm_bootstrapConsumed@ state. A missing/undiscoverable agent (F7)
 1139 --      or an empty/whitespace render (F8) falls through to the global
 1140 --      prompt — never a 500.
 1141 --   3. The global @_fe_systemPrompt@.
 1142 --
 1143 -- The computed value is frozen into the transcript automatically because
 1144 -- 'doCompletion' records the request that carries it.
 1145 computeFirstTurnPrompt :: FrontendEnv -> FilePath -> IO (Maybe Text)
 1146 computeFirstTurnPrompt env sessionDir = do
 1147   let customPromptPath = sessionDir </> "custom-prompt.md"
 1148   customExists <- doesFileExist customPromptPath
 1149   if customExists
 1150     then Just <$> TIO.readFile customPromptPath
 1151     else do
 1152       mAgentPrompt <- renderSessionAgentPrompt env sessionDir
 1153       case mAgentPrompt of
 1154         Just p  -> pure (Just p)
 1155         Nothing -> pure (_fe_systemPrompt env)
 1156 
 1157 -- | Render the per-session agent's system prompt, or 'Nothing' if there is
 1158 -- no usable agent. Reads the session's @_sm_agent@ inline from
 1159 -- @session.json@, looks the agent up under @_fe_agentsDir@, and renders it
 1160 -- honoring @_sm_bootstrapConsumed@. Returns 'Nothing' (so the caller falls
 1161 -- back to the global prompt) when:
 1162 --
 1163 --   * @session.json@ is missing or fails to decode;
 1164 --   * @_sm_agent@ is 'Nothing';
 1165 --   * the named agent is not found under @_fe_agentsDir@ (F7); or
 1166 --   * the rendered prompt is empty/whitespace (F8).
 1167 renderSessionAgentPrompt :: FrontendEnv -> FilePath -> IO (Maybe Text)
 1168 renderSessionAgentPrompt env sessionDir = do
 1169   let metaPath = sessionDir </> "session.json"
 1170   readResult <- try @IOException (LBS.readFile metaPath)
 1171   let raw = either (Left . show) Right readResult
 1172   case raw >>= Aeson.eitherDecode' of
 1173     Left err -> do
 1174       _lh_logWarn (_fe_logger env) $
 1175         "Per-session agent: could not read session.json (" <>
 1176         T.pack metaPath <> "): " <> T.pack err <> " — falling back to global prompt"
 1177       pure Nothing
 1178     Right meta -> case _sm_agent (meta :: SessionMeta) of
 1179       Nothing   -> pure Nothing
 1180       Just name -> do
 1181         mDef <- loadAgent (_fe_agentsDir env) name
 1182         case mDef of
 1183           Nothing -> do
 1184             _lh_logWarn (_fe_logger env) $
 1185               "Per-session agent \"" <> unAgentName name <>
 1186               "\" not found under " <> T.pack (_fe_agentsDir env) <>
 1187               " — falling back to global prompt"
 1188             pure Nothing
 1189           Just def -> do
 1190             rendered <- composeAgentPromptWithBootstrap
 1191                           (_fe_logger env) def agentPromptSectionLimit
 1192                           (_sm_bootstrapConsumed meta)
 1193             if T.null (T.strip rendered)
 1194               then do
 1195                 _lh_logWarn (_fe_logger env) $
 1196                   "Per-session agent \"" <> unAgentName name <>
 1197                   "\" rendered an empty prompt — falling back to global prompt"
 1198                 pure Nothing
 1199               else pure (Just rendered)

 1201 -- | Loop: call the provider, execute any tool_use blocks, append the
 1202 -- tool_result message to context, and recurse. Terminates when the
 1203 -- model returns a turn without tool calls or when @iters@ reaches zero.
 1204 runCompletionLoop
 1205   :: FrontendEnv
 1206   -> SomeProvider
 1207   -> ModelId
 1208   -> [ToolDefinition]
 1209   -> ToolRegistry
 1210   -> Int           -- ^ remaining iterations
 1211   -> Context
 1212   -> IO Text
 1213 runCompletionLoop env provider' model tools reg iters ctx
 1214   | iters <= 0 = do
 1215       _lh_logWarn (_fe_logger env)
 1216         "Tool iteration cap reached; returning placeholder response"
 1217       pure "Tool iteration cap reached without a final response."
 1218   | otherwise = do
 1219       let req = CompletionRequest
 1220             { _cr_model        = model
 1221             , _cr_messages     = contextMessages ctx
 1222             , _cr_systemPrompt = contextSystemPrompt ctx
 1223             , _cr_maxTokens    = Just 4096
 1224             , _cr_tools        = tools
 1225             , _cr_toolChoice   = Nothing
 1226             }
 1227       resp <- complete provider' req
 1228       let calls = toolUseCalls resp
 1229           assistantMsg = Message Assistant (_crsp_content resp)
 1230           ctx' = addMessage assistantMsg ctx
 1231       if null calls
 1232         then pure (responseText resp)
 1233         else do
 1234           results <- mapM (executeOneCall env reg) calls
 1235           let ctx'' = addMessage (toolResultMessage results) ctx'
 1236           runCompletionLoop env provider' model tools reg (iters - 1) ctx''

 1238 -- | Execute a single tool call against the registry, logging any
 1239 -- unknown-tool or error condition.
 1240 executeOneCall
 1241   :: FrontendEnv
 1242   -> ToolRegistry
 1243   -> (ToolCallId, Text, Value)
 1244   -> IO (ToolCallId, [ToolResultPart], Bool)
 1245 executeOneCall env reg (callId, name, input) = do
 1246   _lh_logInfo (_fe_logger env) $ "Tool call: " <> name
 1247   result <- executeTool reg name input
 1248   case result of
 1249     Nothing -> do
 1250       _lh_logWarn (_fe_logger env) $ "Unknown tool: " <> name
 1251       pure (callId, [TRPText ("Unknown tool: " <> name)], True)
 1252     Just (parts, isErr) -> do
 1253       when isErr $ _lh_logWarn (_fe_logger env)
 1254         ("Tool error in " <> name)
 1255       pure (callId, parts, isErr)

 1257 -- | Consume the full request body.
 1258 consumeBody :: Request -> IO LBS.ByteString
 1259 consumeBody req = LBS.fromChunks <$> go
 1260   where
 1261     go = do
 1262       chunk <- getRequestBodyChunk req
 1263       if BS.null chunk then pure [] else (chunk :) <$> go
