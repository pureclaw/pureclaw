    1 -- |
    2 -- Module      : PureClaw.Routing.AutoSpawn
    3 -- Description : Auto-spawn UX, close, resume, rename, crashed handling
    4 --               (Tabbed Chat WU9, A\/L\/X\/B\/S6\/S10 series).
    5 --
    6 -- The user-visible UX surface for tab creation, close, resume, rename,
    7 -- and crashed-tab handling. The dispatcher (WU5) routes parsed slash
    8 -- commands here; this module owns the @\/N@ switch UX (A-series), the
    9 -- close lifecycle on top of the per-kind factory '_tabHandle_close'
   10 -- (L-series), the crashed-tab retry prompt (X-series), and the rename
   11 -- handler (S10).
   12 --
   13 -- == @\/N@ switch UX (A-series, tmux-style packing model)
   14 --
   15 -- 'handleSwitch' implements the simple truth table:
   16 --
   17 --   * tab present → focus + recap (and Crashed observation, X1).
   18 --   * tab missing → emit an error banner; do NOT auto-spawn.
   19 --
   20 -- The previous "auto-spawn on missing /N" behaviour was retired in
   21 -- favour of the tmux packing model: tabs always occupy the lowest
   22 -- slots @\/0..K-1@, so @\/N@ with @N >= K@ is always user error.
   23 --
   24 -- == 'handleDefault' first-run UX (K3)
   25 --
   26 -- The first-run case (no tab focused, registry empty) still
   27 -- implicit-spawns '_rc_defaultKind' at index 0 and forwards the typed
   28 -- text. This is what makes a brand-new user with no tabs able to just
   29 -- type a message and start talking — the K3 first-run path is the only
   30 -- surviving implicit-spawn surface after the tmux packing refactor.
   31 --
   32 -- == @\/tab new@ — always-lowest allocation
   33 --
   34 -- @\/tab new@ no longer accepts a user-supplied index: spawns always
   35 -- land at the lowest free slot via 'Registry.lowestFreeIndex'. The
   36 -- no-kind variant force-prompts the user; the with-kind variant spawns
   37 -- directly.
   38 --
   39 -- == Max-tab cap (S6)
   40 --
   41 -- Every spawn path consults '_rc_maxTabs' BEFORE calling the factory;
   42 -- exceeding the cap surfaces 'TabLimitExceeded' as a dispatcher
   43 -- 'PublicError'. The cap check lives in the dispatcher's 'spawnTabWith'
   44 -- helper (WU5) so AutoSpawn does not duplicate the bookkeeping; we
   45 -- merely surface the redacted error to the user.
   46 --
   47 -- == Crashed UX (X-series)
   48 --
   49 -- Original spawn args are retained per-tab in a @Map Int SpawnArgs@
   50 -- carried by the dispatcher's 'DispatcherState' so the X2\/X3 retry
   51 -- path can replay the exact original args (per design OQ-6: original
   52 -- args, not current defaults).
   53 --
   54 -- See @docs\/tabbed-chat.md@ §"Auto-spawn behavior (A-series)",
   55 -- §"Lifecycle \/ close (L-series)", §"Crashed tab UX (X-series)",
   56 -- §"Dashboard (B-series)", and §"Security (S-series)" S6 \/ S10.

  161 -- ---------------------------------------------------------------------------
  162 -- handleSwitch — /N (tmux packing model)
  163 -- ---------------------------------------------------------------------------
  164 
  165 -- | Handle @\/N@ (Switch). Truth table (tmux packing model):
  166 --
  167 -- * tab present → focus, emit recap, observe Crashed status (X1).
  168 -- * tab missing → emit an error banner ('missingTabBanner'); no
  169 --   auto-spawn.
  170 --
  171 -- The previous "auto-spawn on missing /N" behaviour was retired with
  172 -- the tmux packing refactor: because tabs are always packed in the
  173 -- lowest slots, @\/N@ for any @N@ outside the open range is always
  174 -- user error. The K3 first-run UX (no focus, registry empty, plain
  175 -- text) still implicit-spawns via 'handleDefault'.
  176 handleSwitch
  177   :: AgentEnv
  178   -> BannerEmit
  179   -> TabIndex
  180   -> IO ()
  181 handleSwitch env emit idx = do
  182   mTab <- Registry.lookupTab (_env_tabs env) idx
  183   case mTab of
  184     Just h  -> focusExisting env emit idx h
  185     Nothing -> emit (missingTabBanner idx)
  186 
  187 -- | Banner emitted when @\/N@ references a tab that does not exist.
  188 --
  189 -- Phrased to nudge the user toward the (always-allocating) @\/tab new@
  190 -- command rather than the old "auto-spawn at the requested index"
  191 -- behaviour, which no longer exists.
  192 missingTabBanner :: TabIndex -> Text
  193 missingTabBanner idx =
  194   "/" <> tShowIdx idx
  195     <> ": no such tab — use /tab new to create one"

  198 -- ---------------------------------------------------------------------------
  199 -- handleDefault — Default text input with empty focus (L6 + K3)
  200 -- ---------------------------------------------------------------------------
  201 
  202 -- | Handle the case where the channel layer delivered a plain-text
  203 -- 'PureClaw.Routing.Types.Default' input AND '_env_focus' is 'Nothing'
  204 -- (no tab focused).
  205 --
  206 -- L6 \/ K3 specifies: implicitly spawn '_rc_defaultKind' at the lowest
  207 -- free index (typically 0), then enqueue the text on the new tab.
  208 --
  209 -- If the spawn fails (e.g. rate-limit, max-tabs hit), surface the
  210 -- redacted public error via the banner emitter.
  211 handleDefault
  212   :: AgentEnv
  213   -> SpawnIO
  214   -> BannerEmit
  215   -> IORef (Map Int SpawnArgs)
  216   -> Text
  217   -> IO ()
  218 handleDefault env spawnIO emit argsRef text = do
  219   let rc = _env_routingConfig env
  220       kind = _rc_defaultKind rc
  221   r <- spawnIO kind []
  222   case r of
  223     Left e ->
  224       emit ("/0: " <> unPublicTabError (toPublicTabError e))
  225     Right newIdx -> do
  226       rememberArgs argsRef newIdx kind []
  227       writeIORef (_env_focus env) (Just newIdx)
  228       enqueuePayloadOn env emit newIdx text

  231 -- ---------------------------------------------------------------------------
  232 -- handleNew — /tab new [<kind> [<args>]]
  233 -- ---------------------------------------------------------------------------
  234 
  235 -- | Handle @\/tab new [\<kind\> [\<arg-text\>]]@ (tmux packing model).
  236 --
  237 -- Truth table:
  238 --
  239 -- * @\/tab new@ (no kind)            → force-prompt at the next free
  240 --   slot.
  241 -- * @\/tab new \<kind\> [\<args\>]@  → spawn at the lowest free slot
  242 --   via 'SpawnIO' (the dispatcher's rate-limited factory), focus it,
  243 --   and emit a confirmation banner.
  244 -- * Spawn failure (rate-limit, cap, factory failure, etc.) → redacted
  245 --   'PublicError' banner.
  246 --
  247 -- The previous "user picks the slot" behaviour is gone: new tabs are
  248 -- always allocated at the lowest free index (matches tmux
  249 -- @renumber-windows on@ semantics). The slot for the force-prompt
  250 -- banner is determined by 'Registry.lowestFreeIndex' so the prompt
  251 -- text shows the user where the spawn would land.
  252 handleNew
  253   :: AgentEnv
  254   -> PromptRenderer
  255   -> SpawnIO
  256   -> BannerEmit
  257   -> IORef (Map Int SpawnArgs)
  258   -> Maybe TabKindArg  -- ^ kind keyword, if supplied
  259   -> Maybe Text        -- ^ arg text, if supplied
  260   -> IO ()
  261 handleNew env renderer spawnIO emit argsRef mKind mArgs = do
  262   let rc = _env_routingConfig env
  263   case mKind of
  264     Nothing -> do
  265       -- Force-prompt: surface the index where the spawn would land so
  266       -- the prompt UX has a target to render. If the cap is full, we
  267       -- emit a redacted PublicError instead of a prompt for a slot
  268       -- that doesn't exist.
  269       mFree <- Registry.lowestFreeIndex (_env_tabs env) (_rc_maxTabs rc)
  270       case mFree of
  271         Nothing ->
  272           emit ("/tab new: "
  273                 <> unPublicTabError
  274                      (toPublicTabError (TabLimitExceeded (_rc_maxTabs rc))))
  275         Just idx ->
  276           emit (_pr_renderSpawnPrompt renderer idx Nothing)
  277     Just kindArg -> do
  278       let kind = tabKindArgToKind kindArg
  279           args = splitArgs mArgs
  280       r <- spawnIO kind args
  281       case r of
  282         Left e ->
  283           -- We don't have a known target index at this point — the
  284           -- spawn never assigned one. Emit a generic /tab new banner.
  285           emit ("/tab new: "
  286                 <> unPublicTabError (toPublicTabError e))
  287         Right newIdx -> do
  288           rememberArgs argsRef newIdx kind args
  289           writeIORef (_env_focus env) (Just newIdx)
  290           emit ("/" <> tShowIdx newIdx <> ": spawned ("
  291                 <> kindKeyword kind <> ")")

  294 -- ---------------------------------------------------------------------------
  295 -- handleClose — /tab close N [--force] (tmux-style renumber)
  296 -- ---------------------------------------------------------------------------
  297 
  298 -- | The post-close renumber callback. The dispatcher passes in a
  299 -- closure that shifts its side-maps (spawn-args, pending-retry, etc.)
  300 -- down by one starting at @closedIdx + 1@. AutoSpawn calls this AFTER
  301 -- it has driven the registry-side renumber so the side maps stay in
  302 -- sync with the registry's view of slot ownership.
  303 type RenumberCallback = Int -> IO ()
  304 
  305 -- | Handle @\/tab close N [--force]@ with tmux-style packing.
  306 --
  307 -- Truth table:
  308 --
  309 -- * N missing  → 'TabNotFound' (L5).
  310 -- * N present  → '_tabHandle_close' (graceful or force);
  311 --   remove the registry entry; shift every remaining tab at index
  312 --   @\> N@ down by one (tmux @renumber-windows on@ model); drop the
  313 --   SpawnArgs entry for @N@ then shift remaining SpawnArgs keys;
  314 --   reconcile '_env_focus' (was N → cleared; was \> N → decremented).
  315 --
  316 -- The renumber pass is done under a single 'atomicModifyIORef'' for
  317 -- '_env_tabs' and '_env_runners' so a concurrent reader (the
  318 -- dispatcher is the only writer per E3) always observes a consistent
  319 -- contiguous slot layout. The 'RenumberCallback' lets the caller
  320 -- (Dispatcher) apply the same shift to side maps that AutoSpawn does
  321 -- not own.
  322 --
  323 -- The 'TabHandle._tabHandle_index' field of any remaining handle is
  324 -- /not/ rewritten — it reflects the creation index and is treated as
  325 -- advisory after a renumber. The authoritative current slot is the
  326 -- registry key.
  327 handleClose
  328   :: AgentEnv
  329   -> BannerEmit
  330   -> IORef (Map Int SpawnArgs)
  331   -> RenumberCallback  -- ^ side-map shift (dispatcher passes the
  332                        --   callback that shifts pendingRetry etc.)
  333   -> Int               -- ^ requested index
  334   -> Bool              -- ^ True = --force
  335   -> IO ()
  336 handleClose env emit argsRef renumber rawIdx force =
  337   case mkTabIndex rawIdx of
  338     Nothing  ->
  339       emit ("/" <> T.pack (show rawIdx) <> ": tab: invalid index")
  340     Just idx -> do
  341       removed <- Registry.removeTab (_env_tabs env) idx
  342       case removed of
  343         Nothing -> emit ("/" <> tShowIdx idx <> ": "
  344                           <> unPublicTabError
  345                                (toPublicTabError (TabNotFound rawIdx)))
  346         Just h  -> do
  347           safeIgnore (_tabHandle_close h closeMode)
  348           -- tmux packing: shift every remaining tab > closedIdx down
  349           -- by one so the registry stays contiguous starting at 0.
  350           let closedN = unTabIndex idx
  351           atomicModifyIORef' (_env_tabs env) $ \m ->
  352             (Registry.packAfterRemove closedN m, ())
  353           atomicModifyIORef' (_env_runners env) $ \m ->
  354             (Registry.packAfterRemove closedN m, ())
  355           -- Side-map shifts: drop the spawn-args for the closed slot,
  356           -- then shift remaining keys down. The dispatcher's
  357           -- 'RenumberCallback' handles its own pending-retry map and
  358           -- any other dispatcher-owned side state.
  359           atomicModifyIORef' argsRef $ \m ->
  360             ( shiftMapKeysAfter closedN (Map.delete closedN m), () )
  361           renumber closedN
  362           -- Reconcile focus: closed tab → Nothing; tabs above → -1.
  363           updateFocusOnClose env idx
  364           emit ("/" <> tShowIdx idx <> ": closed")
  365   where
  366     closeMode = if force then CloseForce else CloseGraceful
  367 
  368 -- | After a close (with tmux-style renumber), reconcile '_env_focus':
  369 --
  370 --   * focused tab was the closed one → clear focus (the dispatcher's
  371 --     K3 implicit-spawn path will pick up the next 'Default' input);
  372 --   * focused tab was strictly greater than the closed one →
  373 --     decrement the focus index by one (it just got renumbered);
  374 --   * focused tab was strictly less than the closed one → unchanged.
  375 updateFocusOnClose :: AgentEnv -> TabIndex -> IO ()
  376 updateFocusOnClose env closedIdx = do
  377   mFocus <- readIORef (_env_focus env)
  378   case mFocus of
  379     Nothing -> pure ()
  380     Just f
  381       | f == closedIdx ->
  382           writeIORef (_env_focus env) Nothing
  383       | unTabIndex f > unTabIndex closedIdx ->
  384           -- The focused tab was renumbered down by one.
  385           writeIORef (_env_focus env) (mkTabIndex (unTabIndex f - 1))
  386       | otherwise -> pure ()
  387 
  388 -- | Shift the keys of a 'Data.Map.Strict.Map' (Int-keyed) down by one
  389 -- for every key strictly greater than @k@. Mirrors
  390 -- 'Registry.packAfterRemove' for 'IntMap' but works on the generic
  391 -- 'Map' that the dispatcher uses for spawn-args and pending-retry side
  392 -- state.
  393 shiftMapKeysAfter :: Int -> Map Int v -> Map Int v
  394 shiftMapKeysAfter k = Map.foldlWithKey'
  395   (\acc i v ->
  396      if i > k
  397        then Map.insert (i - 1) v acc
  398        else Map.insert i v acc)
  399   Map.empty

  422 -- handleRename — /tab rename N <name> (S10)
  423 -- ---------------------------------------------------------------------------
  424 
  425 -- | Handle @\/tab rename N \<name\>@. The user-supplied name passes
  426 -- through the caller-supplied sanitizer (typically
  427 -- 'PureClaw.Routing.Parse.sanitizeTabName'); on success the new name
  428 -- is announced; on failure a redacted public error is emitted.
  429 --
  430 -- /WU9 scope note:/ rename in v1 is observable-only at the registry
  431 -- level — the new name is reported back to the user; persisting the
  432 -- renamed label requires mutating the 'TabHandle' record (currently
  433 -- sealed). For v1 we surface the rename intent and the sanitization
  434 -- outcome; the in-place name swap lands in a v1.5 refinement that
  435 -- replaces the 'TabHandle' field with a mutable 'IORef TabName'.
  436 handleRename
  437   :: AgentEnv
  438   -> BannerEmit
  439   -> (Text -> Either NameError Text)  -- ^ sanitizer
  440   -> Int       -- ^ requested index
  441   -> Text      -- ^ requested new name (raw)
  442   -> IO ()
  443 handleRename env emit sanitize rawIdx rawName =
  444   case mkTabIndex rawIdx of
  445     Nothing  ->
  446       emit ("/" <> T.pack (show rawIdx) <> ": tab: invalid index")
  447     Just idx -> do
  448       mTab <- Registry.lookupTab (_env_tabs env) idx
  449       case mTab of
  450         Nothing ->
  451           emit ("/" <> tShowIdx idx <> ": "
  452                 <> unPublicTabError
  453                      (toPublicTabError (TabNotFound rawIdx)))
  454         Just h -> case sanitize rawName of
  455           Left  nameErr ->
  456             emit ("/" <> tShowIdx idx <> ": "
  457                   <> unPublicTabError
  458                        (toPublicTabError (TabInvalidName nameErr)))
  459           Right safeName -> do
  460             let oldName = unTabName (_tabHandle_name h)
  461                 redactedSuffix =
  462                   if safeName == rawName
  463                     then ""
  464                     else " (redacted host/path fragment)"
  465             emit ("Renamed /" <> tShowIdx idx <> " "
  466                    <> oldName <> " to \"" <> safeName <> "\""
  467                    <> redactedSuffix)

  517 -- | Focus an existing tab + emit the recap and (when applicable) the
  518 -- Crashed banner.
  519 focusExisting :: AgentEnv -> BannerEmit -> TabIndex -> TabHandle -> IO ()
  520 focusExisting env emit idx h = do
  521   writeIORef (_env_focus env) (Just idx)
  522   st <- safeStatus h
  523   case st of
  524     Just (Crashed pe) ->
  525       -- X1: surface Crashed status as a one-line PublicError + retry
  526       -- \/ close prompt.
  527       emit ("/" <> tShowIdx idx <> " crashed: "
  528              <> unPublicTabError pe
  529              <> " — [1] retry [2] close")
  530     _ -> do
  531       let recap = recapText (_env_routingConfig env) idx
  532       emit ("/" <> tShowIdx idx <> ": focused")
  533       if T.null recap then pure () else emit recap
  534 
  535 -- | Construct the (WU9-minimal) recap banner. WU10's
  536 -- 'PureClaw.Agent.Loop' refactor will replace this with a transcript
  537 -- replay; for v1 we surface the per-tab recap budget so the user
  538 -- knows what the budget would have been.
  539 recapText :: RoutingConfig -> TabIndex -> Text
  540 recapText rc idx =
  541   let n     = _rc_switchRecap rc
  542       nTxt  = T.pack (show n)
  543   in if n <= 0
  544        then ""
  545        else "(recap: last " <> nTxt <> " messages from /"
  546               <> tShowIdx idx <> ")"

  548 -- | Enqueue a payload on a tab via '_tabHandle_send', surfacing send
  549 -- failures as a redacted dispatcher banner.
  550 enqueuePayloadOn :: AgentEnv -> BannerEmit -> TabIndex -> Text -> IO ()
  551 enqueuePayloadOn env emit idx payload = do
  552   mTab <- Registry.lookupTab (_env_tabs env) idx
  553   case mTab of
  554     Nothing ->
  555       emit ("/" <> tShowIdx idx <> ": "
  556             <> unPublicTabError
  557                  (toPublicTabError (TabNotFound (unTabIndex idx))))
  558     Just h -> do
  559       result <- safeEnqueue h payload
  560       case result of
  561         Right ()  -> pure ()
  562         Left  e   -> emit ("/" <> tShowIdx idx <> ": "
  563                            <> unPublicTabError (toPublicTabError e))
  564 
  565 -- | Tolerant '_tabHandle_send' wrapper: any synchronous exception
  566 -- collapses to a redacted 'TabConcurrencyLimit'.
  567 safeEnqueue :: TabHandle -> Text -> IO (Either TabError ())
  568 safeEnqueue h payload = do
  569   r <- try @SomeException (_tabHandle_send h payload)
  570   pure $ case r of
  571     Left  _e    -> Left (TabConcurrencyLimit 0)
  572     Right inner -> inner
  573 
  574 -- | Read '_tabHandle_status' tolerantly. A buggy factory could throw;
  575 -- in that case we collapse the result to 'Nothing' (treated as
  576 -- "unknown status" — no Crashed banner emitted).
  577 safeStatus :: TabHandle -> IO (Maybe TabStatus)
  578 safeStatus h = do
  579   r <- try @SomeException (_tabHandle_status h)
  580   pure $ case r of
  581     Left _  -> Nothing
  582     Right s -> Just s
  583 
  584 -- | Best-effort: run an IO action and swallow synchronous failures.
  585 safeIgnore :: IO () -> IO ()
  586 safeIgnore m = do
  587   _ <- try @SomeException m
  588   pure ()

  590 -- | Convert a parser-level 'TabKindArg' to the canonical 'TabKind'.
  591 --
  592 -- The payload in the resulting 'TabKind' is a /placeholder/ — the real
  593 -- configuration (provider, model, harness flavour, SSH host, etc.) is
  594 -- constructed by the per-kind factory from its spawn-args, not from
  595 -- the 'TabKind' value. The 'TabKind' here is used only for dispatch.
  596 tabKindArgToKind :: TabKindArg -> TabKind
  597 tabKindArgToKind k = case k of
  598   TkaAi       -> TkSession (SkProvider placeholderProviderSpec)
  599   TkaProvider -> TkSession (SkProvider placeholderProviderSpec)
  600   TkaHarness  -> TkSession (SkHarness placeholderHarnessSpec)
  601   TkaShell    -> TkRawShell TbLocal
  602   TkaSsh      -> TkRawShell (TbSsh placeholderSshConfig)
  603   TkaTmux     -> TkRawShell (TbTmux placeholderTmuxConfig)

  654 -- | Split the @[args-text]@ field (a single 'Text' parsed greedily
  655 -- after the kind keyword) into a list of arguments by whitespace.
  656 splitArgs :: Maybe Text -> [Text]
  657 splitArgs Nothing  = []
  658 splitArgs (Just t) = T.words t

  660 -- | Format a 'TabIndex' as the user-facing decimal string.
  661 tShowIdx :: TabIndex -> Text
  662 tShowIdx = T.pack . show . unTabIndex

  664 -- | Record the spawn args for a given tab so X2\/X3 retry can replay.
  665 rememberArgs :: IORef (Map Int SpawnArgs) -> TabIndex -> TabKind -> [Text] -> IO ()
  666 rememberArgs ref idx kind args =
  667   modifyIORef' ref
  668     (Map.insert (unTabIndex idx) SpawnArgs { _sa_kind = kind, _sa_args = args })
  669 
  670 -- | Public alias of 'rememberArgs' exposed for tests that want to seed
  671 -- a spawn-args map without going through the full dispatchOne path.
  672 -- Production code should never call this directly; the public
  673 -- 'handleNew' \/ 'handleDefault' paths invoke 'rememberArgs' internally
  674 -- on every successful spawn.
  675 rememberArgsForTest :: IORef (Map Int SpawnArgs) -> TabIndex -> TabKind -> [Text] -> IO ()
  676 rememberArgsForTest = rememberArgs
