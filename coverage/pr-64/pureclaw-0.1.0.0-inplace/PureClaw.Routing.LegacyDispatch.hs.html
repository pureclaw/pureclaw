    1 -- |
    2 -- Module      : PureClaw.Routing.LegacyDispatch
    3 -- Description : Bridge from the legacy single-tab CLI loop into Routing.AutoSpawn.
    4 --
    5 -- The new tabbed-chat dispatcher ('PureClaw.Routing.Dispatcher.runDispatcher')
    6 -- is not yet wired as the production CLI entry point. Users running
    7 -- 'PureClaw.Agent.Loop.runAgentLoop' (the legacy single-tab loop) need
    8 -- @\/tab*@ commands to actually DO something — without this module they
    9 -- hit the @CmdTab _@ fallback in 'executeSlashCommand' which only
   10 -- emits a stub message.
   11 --
   12 -- 'dispatchLegacyTabCmd' is the bridge: it receives a parsed
   13 -- 'TabSlashCommand' and invokes the matching 'PureClaw.Routing.AutoSpawn'
   14 -- handler against the existing 'AgentEnv'. Handler callbacks are wired
   15 -- with the minimum surface needed for legacy mode:
   16 --
   17 --   * 'BannerEmit' — direct @_ch_send@ to the user's channel.
   18 --   * 'PromptRenderer' — built from the channel's streaming flag via
   19 --     'mkDefaultPromptRenderer'.
   20 --   * 'SpawnArgs' map — a fresh empty 'IORef' per dispatch. Legacy mode
   21 --     does not implement the X1 crashed-tab retry UX, so per-call
   22 --     ephemerality is fine.
   23 --   * 'RenumberCallback' — no-op (no DispatcherState to keep in sync).
   24 --   * Sanitize callback — 'PureClaw.Routing.Parse.sanitizeTabName'.
   25 --   * 'TabFactory' — dispatches by 'TabKind' to the real
   26 --     'PureClaw.Tab.Ai.mkTabAi' \/ 'Tab.Harness.mkTabHarness' \/
   27 --     'Tab.Backend.mkTabBackend' factories (all 3-arg variants that
   28 --     take 'AgentEnv').
   29 --
   30 -- @\/tab resume@ requires 'DispatcherState' (session resolver +
   31 -- pending-retry tracking) that legacy mode lacks; this dispatch emits
   32 -- a "not available in legacy mode" message rather than crashing.
   33 --
   34 -- This module sits ABOVE 'PureClaw.Agent.SlashCommands' (for the
   35 -- 'TabSlashCommand' ADT) and ABOVE 'PureClaw.Routing.AutoSpawn' (for
   36 -- the handlers), so it can import both freely. Conversely it must NOT
   37 -- be imported from 'Agent.SlashCommands' (would create a cycle through
   38 -- 'AutoSpawn' which already imports 'SlashCommands').
   39 module PureClaw.Routing.LegacyDispatch
   40   ( dispatchLegacyTabCmd
   41   ) where
   42 
   43 import Data.IORef
   44 import Data.Map.Strict qualified as Map
   45 import Data.Text (Text)

   51 import PureClaw.Routing.Dispatcher qualified as Dispatcher
   52 import PureClaw.Routing.Parse qualified as Parse
   53 import PureClaw.Routing.PromptRenderer (mkDefaultPromptRenderer)

   56 -- | Dispatch a 'TabSlashCommand' from the legacy CLI loop into the
   57 -- 'Routing.AutoSpawn' handler surface.
   58 --
   59 -- All output reaches the user via @_ch_send@; the caller does not need
   60 -- to consume any return value (handlers are 'IO ()').
   61 dispatchLegacyTabCmd :: AgentEnv -> TabSlashCommand -> IO ()
   62 dispatchLegacyTabCmd env tabCmd = do
   63   let emit     = legacyEmit env
   64       renderer = mkDefaultPromptRenderer (_env_channel env)
   65   argsRef <- newIORef (Map.empty :: Map.Map Int AutoSpawn.SpawnArgs)
   66   case tabCmd of
   67     TabListCmd ->
   68       AutoSpawn.handleListTabs env emit
   69     TabFocusCmd rawIdx ->
   70       AutoSpawn.handleFocus env emit rawIdx
   71     TabCloseCmd rawIdx force ->
   72       AutoSpawn.handleClose env emit argsRef noRenumber rawIdx
   73                             (force == ForceYes)
   74     TabRenameCmd rawIdx rawName ->
   75       AutoSpawn.handleRename env emit Parse.sanitizeTabName rawIdx rawName
   76     TabNewCmd mKind mArgs ->
   77       AutoSpawn.handleNew env renderer (Dispatcher.spawnTab env)
   78                           emit argsRef mKind mArgs
   79     TabResumeCmd _sid ->
   80       emit
   81         "Tab resume requires the tabbed-chat dispatcher\
   82         \ — not yet wired in the legacy CLI loop."
   83 
   84 -- | The 'BannerEmit' callback used by every handler — directly emits via
   85 -- @_ch_send@.
   86 legacyEmit :: AgentEnv -> Text -> IO ()
   87 legacyEmit env text =
   88   _ch_send (_env_channel env) (OutgoingMessage text)
   89 
   90 -- | Renumber callback — no-op for legacy mode. Legacy has no
   91 -- 'DispatcherState', so the pending-retry / spawn-args sidemaps that
   92 -- would otherwise need reindexing don't exist.
   93 noRenumber :: Int -> IO ()
   94 noRenumber _ = pure ()
