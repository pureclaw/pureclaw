    1 -- | WS-streaming endpoint for live transcript updates — WU3.
    2 --
    3 -- Implements the wire protocol defined in
    4 -- @docs/transcript-streaming.md §Wire Protocol@: on upgrade, the server
    5 -- sends a one-shot @hello@; then a reader/writer race forwards
    6 -- 'BrokerEvent's from the broker to the WS peer while accepting
    7 -- @focus@ ops from the client.
    8 --
    9 -- /Security/. Every WS upgrade is gated by an exact-match Origin allowlist
   10 -- ('originAllowed' against the supplied list), a per-origin subscriber
   11 -- cap (via 'StreamGuard'), and the broker's global cap. Inbound frames
   12 -- are bounded at 4 KB; malformed JSON returns an in-band error without
   13 -- closing the connection. 'Network.WebSockets.withPingThread' provides
   14 -- the keepalive (Warp's idle timeout does NOT apply to hijacked sockets).
   15 --
   16 -- /Replay/. A focus op carrying @since@ enters replay mode: incoming
   17 -- 'EntryRecorded' events for the focused session are buffered until the
   18 -- file slice is read, then the buffered set is sent (UUID-deduped against
   19 -- the file slice). The algorithm tolerates publishes-during-read and
   20 -- falls back to @replay-failed@ on any anomaly so the client can refetch
   21 -- via HTTP.

  178 -- | Encode a 'ServerEvent' to JSON.
  179 encodeServerEvent :: ServerEvent -> Value
  180 encodeServerEvent (SeHello pv started) = object
  181   [ "type"            .= ("hello" :: Text)
  182   , "protocolVersion" .= pv
  183   , "serverStartedAt" .= T.pack (iso8601Show started)

  185 encodeServerEvent (SeEntry sid entry) = object
  186   [ "type"      .= ("entry" :: Text)
  187   , "sessionId" .= unSessionId sid

  191   [ "type"      .= ("activity" :: Text)

  196   [ "type"                .= ("replay-end" :: Text)

  202   [ "type"    .= ("error" :: Text)

  210 -- | Outcome of a 'decodeClientOp' attempt. Stays a sum so callers can
  211 -- distinguish "JSON didn't parse" from "JSON parsed but the op was
  212 -- unknown" (D29 vs D36).
  213 data DecodeResult
  214   = DrOk !ClientOp
  215   | DrInvalidFrame
  216   | DrInvalidOp
  217   deriving stock (Show, Eq)
  218 
  219 -- | Two-phase parse: structural JSON decode, then op-discriminant parse.
  220 decodeClientOp :: LBS.ByteString -> DecodeResult
  221 decodeClientOp bs = case Aeson.eitherDecode bs of
  222   Left _      -> DrInvalidFrame
  223   Right value -> case Aeson.parseEither parseFocus (value :: Value) of
  224     Right v   -> DrOk v
  225     Left err
  226       | "INVALID_OP_TAG" `T.isInfixOf` T.pack err -> DrInvalidOp
  227       | otherwise                                 -> DrInvalidFrame
  228   where
  229     -- Tag failures with a non-natural sentinel that's unlikely to collide
  230     -- with Aeson's generic decode errors. The match above lifts the tag
  231     -- out of the prefixed error message.
  232     parseFocus :: Value -> Aeson.Parser ClientOp
  233     parseFocus = withObject "ClientOp" $ \o -> do
  234       op <- o .: "op" :: Aeson.Parser Text
  235       case op of
  236         "focus" -> do
  237           msid   <- o .:? "sessionId"
  238           msince <- o .:? "since"
  239           pure (CoFocus (fmap SessionId msid) msince)
  240         _ -> fail "INVALID_OP_TAG"
  241 
  242 instance FromJSON ClientOp where
  243   parseJSON = withObject "ClientOp" $ \o -> do
  244     op <- o .: "op" :: Aeson.Parser Text
  245     case op of
  246       "focus" -> do
  247         msid   <- o .:? "sessionId"
  248         msince <- o .:? "since"
  249         pure (CoFocus (fmap SessionId msid) msince)
  250       _ -> fail ("unknown-op: " <> T.unpack op)

  256 -- | Lowercase the scheme and host components of an origin string; strip a
  257 -- trailing slash. Port is left as-is.
  258 normalizeOrigin :: Text -> Text
  259 normalizeOrigin t = dropTrailingSlash (lowerSchemeHost t)
  260   where
  261     dropTrailingSlash s
  262       | T.null s          = s
  263       | T.last s == '/'   = T.init s
  264       | otherwise         = s
  265     lowerSchemeHost s =
  266       case T.breakOn "://" s of
  267         (scheme, rest) | T.isPrefixOf "://" rest ->
  268           let body = T.drop 3 rest
  269               (host, port) = T.breakOn ":" body
  270           in T.toLower scheme <> "://" <> T.toLower host <> port
  271         _ -> T.toLower s
  272 
  273 -- | Exact-match origin lookup against a /list of normalized origins/.
  274 -- Empty allowlist denies everything.
  275 originAllowed :: [Text] -> Text -> Bool
  276 originAllowed []      _      = False
  277 originAllowed allowed origin = origin `elem` map normalizeOrigin allowed
  278 
  279 -- | Accept the given (already-normalized) origin if it matches the configured
  280 -- allowlist OR matches the Host-reflected origin — i.e. the
  281 -- @scheme://Host-header@ the request itself arrived on.
  282 --
  283 -- The Host-reflection relaxation lets a user open the URL from a LAN IP,
  284 -- shared link, or alternative interface (anything that resolves to the
  285 -- bound address) without enumerating each one in @_fc_allowedOrigins@.
  286 -- It does NOT broaden the threat surface beyond what the HTTP API already
  287 -- accepts: any client able to reach the configured bind address can hit
  288 -- the HTTP routes today, and Host-reflection makes the WS endpoint match
  289 -- that posture.
  290 --
  291 -- It does still block the drive-by case the strict allowlist was added for:
  292 -- a malicious page at @attacker.com@ gets neither the allowlist match nor
  293 -- the Host-reflection match (its Origin won't equal the server's Host
  294 -- header — the server's Host header is the address the BROWSER connected
  295 -- to, which is by construction the legit server, not the attacker).
  296 originAcceptable :: Maybe Text -> [Text] -> Text -> Bool
  297 originAcceptable mHostReflected allowed origin =
  298      originAllowed allowed origin
  299   || maybe False ((== origin) . normalizeOrigin) mHostReflected

  301 -- ---------------------------------------------------------------------------
  302 -- StreamGuard
  303 -- ---------------------------------------------------------------------------
  304 
  305 -- | Atomically increment the per-origin counter, returning 'True' if the
  306 -- counter was strictly below '_streamGuard_maxPerOrigin' before the
  307 -- increment. The caller must invoke 'releaseClaim' exactly once for each
  308 -- successful claim.
  309 tryClaim :: StreamGuard -> Text -> STM Bool
  310 tryClaim g origin = do
  311   m <- readTVar (_streamGuard_perOrigin g)
  312   let current = Map.findWithDefault 0 origin m
  313   if current >= _streamGuard_maxPerOrigin g
  314     then pure False
  315     else do
  316       writeTVar (_streamGuard_perOrigin g) (Map.insert origin (current + 1) m)
  317       pure True
  318 
  319 -- | Decrement (or remove) the per-origin counter. Idempotent against a
  320 -- key with count 0.
  321 releaseClaim :: StreamGuard -> Text -> STM ()
  322 releaseClaim g origin = modifyTVar' (_streamGuard_perOrigin g) (Map.update dec origin)
  323   where
  324     dec n | n <= 1    = Nothing
  325           | otherwise = Just (n - 1)

  355 -- | Per-WS-connection mutable state.
  356 data ConnState = ConnState
  357   { _conn_focus       :: !(IORef (Maybe SessionId))
  358   , _conn_origin      :: !Text
  359   , _conn_originRaw   :: !Text
  360   , _conn_replayMode  :: !(IORef (Maybe SessionId))
  361     -- ^ When @Just sid@, the writer buffers 'EntryRecorded' events for
  362     -- @sid@ instead of forwarding them. @Nothing@ means live mode.
  363     --
  364     -- (Stored in an IORef because the WS handler is single-threaded
  365     -- per-connection; the reader thread sets it under STM only when
  366     -- aborting a replay mid-flight (D40), where the buffer drain and
  367     -- the flag flip must observe one consistent moment.)
  368   , _conn_replayBuf   :: !(IORef [TranscriptEntry])
  369   }
  370 
  371 mkConnState :: Text -> Text -> IO ConnState
  372 mkConnState origin originRaw = do
  373   focus      <- newIORef Nothing
  374   replayMode <- newIORef Nothing
  375   replayBuf  <- newIORef []
  376   pure ConnState
  377     { _conn_focus      = focus
  378     , _conn_origin     = origin
  379     , _conn_originRaw  = originRaw
  380     , _conn_replayMode = replayMode
  381     , _conn_replayBuf  = replayBuf
  382     }

  384 -- ---------------------------------------------------------------------------
  385 -- streamApp
  386 -- ---------------------------------------------------------------------------

  388 -- | The WS server-app. Returns a 'WS.ServerApp' suitable for composition
  389 -- via 'Network.Wai.Handler.WebSockets.websocketsOr'. The list of allowed
  390 -- origins is supplied by the caller (Server.hs reads it out of
  391 -- @_fc_allowedOrigins@; tests inject their own).
  392 streamApp :: [Text] -> FrontendEnv -> WS.ServerApp
  393 streamApp allowed env pending = do
  394   let logger = _fe_logger env
  395   case (_fe_broker env, _fe_streamGuard env) of
  396     (Just broker, Just guard) ->
  397       handleUpgrade allowed env broker guard pending
  398     _ -> do
  399       _lh_logWarn logger "WS upgrade rejected: streaming disabled"
  400       WS.rejectRequestWith pending (rejection 503 "streaming disabled")
  401 
  402 rejection :: Int -> BS.ByteString -> WS.RejectRequest
  403 rejection code msg = WS.defaultRejectRequest
  404   { WS.rejectCode    = code
  405   , WS.rejectMessage = msg
  406   , WS.rejectBody    = msg
  407   }

  409 -- | Handle a pending WS upgrade. Order:
  410 -- Origin allowlist → 'tryClaim' → broker subscribe → 'acceptRequest'.
  411 handleUpgrade
  412   :: [Text]
  413   -> FrontendEnv
  414   -> StreamBroker
  415   -> StreamGuard
  416   -> WS.PendingConnection
  417   -> IO ()
  418 handleUpgrade allowed env broker guard pending = do
  419   let logger        = _fe_logger env
  420       req           = WS.pendingRequest pending
  421       headers       = WS.requestHeaders req
  422       mOriginBS     = lookup "Origin" headers
  423       mHostHeader   = TE.decodeUtf8 <$> lookup "Host" headers
  424       hostRemote    = fromMaybe "<unknown>" mHostHeader
  425       -- Host-reflected origin: scheme://<Host header>, with scheme matching
  426       -- the upgrade's TLS state. Used by 'originAcceptable' to allow a
  427       -- request whose Origin matches the address the request itself hit
  428       -- (typically: localhost vs LAN IP, or any alternate interface the
  429       -- user reached the server through).
  430       scheme        = if WS.requestSecure req then "https" else "http"
  431       mHostReflected = (\h -> scheme <> "://" <> h) <$> mHostHeader
  432   case mOriginBS of
  433     Nothing -> do
  434       _lh_logWarn logger $
  435         "WS upgrade rejected: missing Origin (remote=" <> hostRemote <> ")"
  436       WS.rejectRequestWith pending (rejection 403 "missing Origin")
  437     Just rawBytes -> do
  438       let originRaw  = TE.decodeUtf8 rawBytes
  439           normalized = normalizeOrigin originRaw
  440       if not (originAcceptable mHostReflected allowed normalized)
  441         then do
  442           _lh_logWarn logger $
  443             "WS upgrade rejected: Origin not in allowlist and does not match Host (origin="
  444               <> originRaw <> ", host=" <> hostRemote <> ")"
  445           WS.rejectRequestWith pending (rejection 403 "Origin not allowed")
  446         else do
  447           claimed <- atomically (tryClaim guard normalized)
  448           if not claimed
  449             then do
  450               _lh_logWarn logger $
  451                 "WS upgrade rejected: per-origin cap reached (origin="
  452                   <> originRaw <> ", host=" <> hostRemote <> ")"
  453               WS.rejectRequestWith pending
  454                 (rejection 503 "per-origin cap reached")
  455             else do
  456               subResult <- _streamBroker_subscribe broker
  457               case subResult of
  458                 Left SubscriberCapReached -> do
  459                   atomically (releaseClaim guard normalized)
  460                   _lh_logWarn logger $
  461                     "WS upgrade rejected: broker cap reached (origin="
  462                       <> originRaw <> ")"
  463                   WS.rejectRequestWith pending
  464                     (rejection 503 "global cap reached")
  465                 Right sub ->
  466                   bracket
  467                     (do
  468                        conn <- WS.acceptRequest pending
  469                        cs   <- mkConnState normalized originRaw
  470                        pure (conn, cs))
  471                     (\_ -> do
  472                        atomically (releaseClaim guard normalized)
  473                        _sub_cancel sub)
  474                     (uncurry (runConnection env sub))

  476 -- ---------------------------------------------------------------------------
  477 -- Per-connection runner
  478 -- ---------------------------------------------------------------------------
  479 
  480 -- | Once the upgrade has succeeded, send the hello, install the ping
  481 -- thread, and run the reader/writer race. Any escaped exception is
  482 -- caught, logged at error level, surfaced as @{code:\"internal\"}@,
  483 -- and the connection is closed (D33).
  484 runConnection
  485   :: FrontendEnv
  486   -> Subscription
  487   -> WS.Connection
  488   -> ConnState
  489   -> IO ()
  490 runConnection env sub conn cs = do
  491   let logger = _fe_logger env
  492   handle (escapeHandler logger) $ do
  493     sendHello conn
  494     -- Push initial sidebar lists so the client renders without waiting
  495     -- for the first mutation or a fallback HTTP poll.
  496     case _fe_broker env of
  497       Just _broker -> do
  498         snapshot <- computeListsSnapshot env
  499         WS.sendTextData conn (Aeson.encode snapshot)
  500       Nothing -> pure ()
  501     WS.withPingThread conn pingInterval (pure ()) $
  502       Async.race_ (readerLoop env conn cs)
  503                   (writerLoop sub conn cs)
  504   where
  505     escapeHandler :: LogHandle -> SomeException -> IO ()
  506     escapeHandler logger e
  507       | Just AsyncCancelled <- fromException e = Exception.throwIO e
  508       | Just (WS.CloseRequest {}) <- fromException e = pure ()
  509       | Just WS.ConnectionClosed <- fromException e = pure ()
  510       | otherwise = do
  511           _lh_logError logger $
  512             "WS connection internal error (origin=" <> _conn_originRaw cs
  513               <> "): " <> T.pack (show e)
  514           _ <- try @SomeException
  515                  (sendEvent conn (SeError EcInternal (T.pack (show e))))
  516           pure ()
  517 
  518 -- | Ping every 25 s; the websockets lib's pong timeout (10 s) closes the
  519 -- socket if the peer goes silent (D32). Logged via the bracket exit path
  520 -- in 'handleUpgrade'.
  521 pingInterval :: Int
  522 pingInterval = 25
  523 
  524 sendHello :: WS.Connection -> IO ()
  525 sendHello conn = sendEvent conn (SeHello "v1" serverStartedAt)

  527 sendEvent :: WS.Connection -> ServerEvent -> IO ()
  528 sendEvent conn ev = WS.sendTextData conn (Aeson.encode (encodeServerEvent ev))

  531 sendError conn code msg = sendEvent conn (SeError code msg)

  533 -- ---------------------------------------------------------------------------
  534 -- Reader loop
  535 -- ---------------------------------------------------------------------------
  536 
  537 -- | Read inbound frames forever. Each frame is bounded at 4 KB; oversize
  538 -- frames close the connection. Malformed JSON or unknown ops surface as
  539 -- in-band errors without closing.
  540 readerLoop :: FrontendEnv -> WS.Connection -> ConnState -> IO ()
  541 readerLoop env conn cs = loop
  542   where
  543     logger = _fe_logger env
  544     loop = do
  545       msg <- try @SomeException (WS.receiveDataMessage conn)
  546       case msg of
  547         Left e
  548           | Just AsyncCancelled <- fromException e -> Exception.throwIO e
  549           | otherwise -> pure () -- closed
  550         Right (WS.Text bs _) -> handleData bs >> loop
  551         Right (WS.Binary _)  -> do
  552           _lh_logWarn logger "WS rejecting binary frame"
  553           sendError conn EcInvalidFrame "binary frames not supported"
  554           loop
  555 
  556     handleData :: LBS.ByteString -> IO ()
  557     handleData bs
  558       | LBS.length bs > fromIntegral maxInboundBytes = do
  559           _lh_logWarn logger "WS frame-too-large: closing"
  560           sendError conn EcFrameTooLarge "inbound frame exceeded 4 KB"
  561           WS.sendClose conn ("frame too large" :: BS.ByteString)
  562       | otherwise = case decodeClientOp bs of
  563           DrInvalidFrame -> do
  564             _lh_logWarn logger $ "WS invalid frame: "
  565               <> TE.decodeUtf8 (BS.take 100 (LBS.toStrict bs))
  566             sendError conn EcInvalidFrame "malformed JSON"
  567           DrInvalidOp ->
  568             sendError conn EcInvalidOp "unknown op"
  569           DrOk (CoFocus mSid mSince) ->
  570             handleFocus env conn cs mSid mSince

  574 -- ---------------------------------------------------------------------------

  576 -- | Handle a @focus@ op. If @since@ is present and valid, run the replay
  577 -- snapshot algorithm; otherwise just update the focus IORef.
  578 handleFocus
  579   :: FrontendEnv
  580   -> WS.Connection
  581   -> ConnState
  582   -> Maybe SessionId
  583   -> Maybe Text
  584   -> IO ()
  585 handleFocus env conn cs mSid mSince =
  586   case mSid of
  587     Just (SessionId raw) | not (isValidSessionId raw) ->
  588       sendError conn EcSessionNotFound ("invalid session id: " <> raw)
  589     _ -> case mSince of
  590       Nothing -> setLiveFocus
  591       Just s | T.length s > maxSinceLength ->
  592         sendError conn EcInvalidFrame "since exceeds 64 characters"
  593       Just sinceTok -> case mSid of
  594         Nothing  -> sendError conn EcInvalidFrame "since requires sessionId"
  595         Just sid -> do
  596           startReplay env conn cs sid sinceTok
  597           sendActivitySnapshot env conn sid
  598   where
  599     setLiveFocus = do
  600       abortInflightReplay conn cs
  601       writeIORef (_conn_focus cs) mSid
  602       forM_ mSid (sendActivitySnapshot env conn)
  603 
  604 -- | If the broker has a recorded 'HarnessActivity' for this session, push
  605 -- it to the freshly-focused client. Without this, a tab that connects
  606 -- /while/ a provider request is in flight never sees the in-progress
  607 -- 'HarnessThinking' event (it was published before the subscriber existed)
  608 -- and the thinking indicator stays dark until the request completes.
  609 sendActivitySnapshot :: FrontendEnv -> WS.Connection -> SessionId -> IO ()
  610 sendActivitySnapshot env conn sid = case _fe_broker env of
  611   Nothing     -> pure ()
  612   Just broker -> do
  613     mAct <- _streamBroker_currentActivity broker sid
  614     forM_ mAct $ \act ->
  615       sendEvent conn (SeActivity sid (AkHarnessStatus act))

  617 -- | Abort any in-flight replay. If a replay was in progress, emit
  618 -- @replay-aborted@ and clear the buffer (D40).
  619 abortInflightReplay :: WS.Connection -> ConnState -> IO ()
  620 abortInflightReplay conn cs = do
  621   prev <- readIORef (_conn_replayMode cs)
  622   case prev of
  623     Nothing -> pure ()
  624     Just _  -> do
  625       writeIORef (_conn_replayMode cs) Nothing
  626       writeIORef (_conn_replayBuf cs) []
  627       _ <- try @SomeException
  628              (sendError conn EcReplayAborted "focus changed mid-replay")
  629       pure ()
  630 
  631 -- | Snapshot-based replay algorithm:
  632 --   1. Mark replay-mode for the session; writer starts buffering.
  633 --   2. Update focus IORef.
  634 --   3. Read transcript file (slice after @since@).
  635 --   4. Send each file entry as @entry@.
  636 --   5. Clear replay-mode; drain buffer, dedup against file ids, send.
  637 --   6. Emit @replay-end@.
  638 -- On any I/O error during step 3, clear state and emit @replay-failed@
  639 -- (D28) — the client falls back to HTTP GET.
  640 startReplay
  641   :: FrontendEnv
  642   -> WS.Connection
  643   -> ConnState
  644   -> SessionId
  645   -> Text
  646   -> IO ()
  647 startReplay env conn cs sid sinceId = do
  648   abortInflightReplay conn cs
  649   writeIORef (_conn_replayMode cs) (Just sid)
  650   writeIORef (_conn_replayBuf cs) []
  651   writeIORef (_conn_focus cs) (Just sid)
  652   let logger = _fe_logger env
  653       path   = _fe_sessionsDir env <> "/" <> T.unpack (unSessionId sid) <> "/transcript.jsonl"
  654   result <- try @SomeException (readReplaySlice path sinceId)
  655   case result of
  656     Left e -> do
  657       _lh_logError logger $
  658         "replay read failed (session=" <> unSessionId sid <> "): "
  659           <> T.pack (show e)
  660       writeIORef (_conn_replayMode cs) Nothing
  661       writeIORef (_conn_replayBuf cs) []
  662       sendError conn EcReplayFailed (T.pack (show e))
  663     Right slice -> do
  664       forM_ slice $ \e -> sendEvent conn (SeEntry sid e)
  665       -- Drain buffer; if focus changed mid-replay, abortInflightReplay
  666       -- already cleared state, and replayMode is Nothing now — skip.
  667       currentMode <- readIORef (_conn_replayMode cs)
  668       case currentMode of
  669         Just sid' | sid' == sid -> do
  670           buf <- atomicModifyIORef' (_conn_replayBuf cs) ([],)
  671           writeIORef (_conn_replayMode cs) Nothing
  672           let dedup   = Set.fromList (map _te_id slice)
  673               residue = filter (\e -> not (Set.member (_te_id e) dedup)) buf
  674           forM_ residue $ \e -> sendEvent conn (SeEntry sid e)
  675           let lastId = case (reverse residue, reverse slice) of
  676                 (e:_, _) -> Just (_te_id e)
  677                 (_, e:_) -> Just (_te_id e)
  678                 _        -> Nothing
  679           sendEvent conn (SeReplayEnd sid lastId)
  680         _ -> pure ()  -- focus changed mid-replay; abort path already fired
  681 
  682 -- | Read JSONL file at @path@, returning entries strictly after @since@.
  683 readReplaySlice :: FilePath -> Text -> IO [TranscriptEntry]
  684 readReplaySlice path sinceId = do
  685   raw <- try @SomeException (LBS.readFile path)
  686   case raw of
  687     Left e
  688       | Just ioErr <- fromException e, isDoesNotExistError ioErr -> pure []
  689       | otherwise -> Exception.throwIO e
  690     Right contents -> do
  691       let raws    = filter (not . LBS.null) (LBS.split 0x0A contents)
  692           entries = [ x | l <- raws, Just x <- [Aeson.decode l] ]
  693       pure (sliceAfter sinceId entries)
  694   where
  695     sliceAfter "" es = es
  696     sliceAfter s  es = drop 1 (dropWhile (\e -> _te_id e /= s) es)

  702 -- | Drain broker events; filter & forward to the WS peer. The reader
  703 -- thread updates the focus IORef and the replay-mode IORef; the writer
  704 -- consults both for every event.
  705 writerLoop :: Subscription -> WS.Connection -> ConnState -> IO ()
  706 writerLoop sub conn cs = loop
  707   where
  708     loop = do
  709       r <- atomically nextSignal
  710       case r of
  711         Left () -> do
  712           sendEvent conn SeOverflow
  713           WS.sendClose conn ("overflow" :: BS.ByteString)
  714         Right ev -> do
  715           handleEvent ev
  716           loop
  717 
  718     nextSignal :: STM (Either () BrokerEvent)
  719     nextSignal =
  720       (do
  721         b <- readTVar (_sub_overflow sub)
  722         if b then pure (Left ()) else retry)
  723       `orElse`
  724       (Right <$> readTBQueue (_sub_queue sub))
  725 
  726     handleEvent (EntryRecorded sid entry) = do
  727       replaying <- readIORef (_conn_replayMode cs)
  728       focus     <- readIORef (_conn_focus cs)
  729       case replaying of
  730         Just rsid | rsid == sid ->
  731           atomicModifyIORef' (_conn_replayBuf cs) (\xs -> (xs ++ [entry], ()))
  732         _ -> case focus of
  733           Just fsid | fsid == sid -> sendEvent conn (SeEntry sid entry)
  734           _ -> pure ()
  735     handleEvent (ActivityChanged sid (SaEntryAt t)) =
  736       sendEvent conn (SeActivity sid (AkEntryAt t))
  737     handleEvent (ActivityChanged sid (SaHarnessStatus s)) =
  738       sendEvent conn (SeActivity sid (AkHarnessStatus s))
  739     handleEvent (ActivityChanged sid (SaSessionCreated meta)) =
  740       sendEvent conn (SeActivity sid (AkSessionCreated meta))
  741     handleEvent (ListsSnapshot v) =
  742       WS.sendTextData conn (Aeson.encode v)
