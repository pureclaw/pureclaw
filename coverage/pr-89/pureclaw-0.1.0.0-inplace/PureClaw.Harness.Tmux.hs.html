   94 -- | The SOLE chokepoint for constructing a tmux subprocess. Every tmux
   95 -- invocation in PureClaw builds its 'P.ProcessConfig' from an
   96 -- 'Command.AuthorizedCommand' obtained via 'Command.authorizeTmuxCommand'
   97 -- (the manager-owned tmux seam, §8 B1). After this helper has produced the
   98 -- base config, callers apply the appropriate @setStdin@\/@setStdout@\/
   99 -- @setStderr@ wrappers.
  100 --
  101 -- Because 'Command.AuthorizedCommand' has an un-exported value constructor and
  102 -- 'Command.authorizeTmuxCommand' is the only tmux path to one, routing every
  103 -- @tmux@ @P.proc@ through here makes an un-authorized tmux call impossible by
  104 -- construction. Bare @P.proc tmuxBin@ must appear nowhere else.
  105 --
  106 -- TODO(harness-registry phase 2): audit-log tmux seam invocations once a
  107 -- 'LogHandle' is threaded to the call sites (would change exported signatures,
  108 -- out of scope for this additive refactor — see 'Command.authorizeTmuxCommand').
  109 tmuxProc :: Command.AuthorizedCommand -> P.ProcessConfig () () ()
  110 tmuxProc ac =
  111   P.proc (Command.getCommandProgram ac) (map T.unpack (Command.getCommandArgs ac))

  113 -- | Resolve the absolute path to the tmux binary.
  114 -- First checks PATH via 'findExecutable', then tries common system locations.
  115 findTmux :: IO (Maybe FilePath)
  116 findTmux = do
  117   mPath <- Dir.findExecutable "tmux"
  118   case mPath of
  119     Just p  -> pure (Just p)
  120     Nothing -> findFirstExisting fallbackPaths
  121   where
  122     fallbackPaths =
  123       [ "/opt/homebrew/bin/tmux"
  124       , "/usr/local/bin/tmux"
  125       , "/usr/bin/tmux"
  126       ]
  127     findFirstExisting [] = pure Nothing
  128     findFirstExisting (p : ps) = do
  129       exists <- Dir.doesFileExist p
  130       if exists then pure (Just p) else findFirstExisting ps

  132 -- | Check if tmux is available on PATH.
  133 requireTmux :: IO (Either HarnessError ())
  134 requireTmux = do
  135   mPath <- findTmux
  136   pure $ case mPath of
  137     Nothing -> Left (HarnessTmuxNotAvailable "tmux not found on PATH or fallback locations")
  138     Just _  -> Right ()

  140 -- | Run a tmux command, capturing stderr for diagnostics.
  141 -- Returns the exit code and stderr output.
  142 runTmux :: [String] -> IO (ExitCode, ByteString)
  143 runTmux args = do
  144   mPath <- findTmux
  145   case mPath of
  146     Nothing -> pure (ExitFailure 127, "tmux not found")
  147     Just tmuxBin -> do
  148       let config = P.setStdin P.closed
  149                  $ P.setStdout P.nullStream
  150                  $ P.setStderr P.byteStringOutput
  151                  $ tmuxProc (Command.authorizeTmuxCommand tmuxBin (map T.pack args))
  152       (exitCode, _stdout, stderr) <- P.readProcess config
  153       pure (exitCode, LBS.toStrict stderr)
  154 
  155 -- | Run a tmux command silently. Returns just the exit code.
  156 runTmuxSilent :: [String] -> IO ExitCode
  157 runTmuxSilent args = fst <$> runTmux args

  159 -- | Build a stealth shell command string for launching a binary in tmux.
  160 -- Strips TMUX env vars and wraps with script(1) for a fresh PTY.
  161 --
  162 -- macOS: @script -q \/dev\/null command args...@
  163 -- Linux: @script -qc \"command args...\" \/dev\/null@
  164 stealthShellCommand :: FilePath -> [Text] -> String
  165 stealthShellCommand binary args =
  166   let argStr = unwords (map (T.unpack . shellEscape) args)
  167       fullCmd = binary <> if null args then "" else " " <> argStr
  168       scriptWrapped
  169         | Info.os == "darwin" =
  170             "script -q /dev/null " <> fullCmd
  171         | otherwise =
  172             "script -qc \"" <> escapeForShell fullCmd <> "\" /dev/null"
  173   in "env -u TMUX -u TMUX_PANE TERM=xterm-256color " <> scriptWrapped
  174 
  175 -- | Escape a string for embedding inside double quotes in a shell command.
  176 escapeForShell :: String -> String
  177 escapeForShell = concatMap go
  178   where
  179     go '"'  = "\\\""
  180     go '\\' = "\\\\"
  181     go '$'  = "\\$"
  182     go '`'  = "\\`"
  183     go c    = [c]

  198 -- | Start a tmux session with the given name if not already running.
  199 -- Creates a detached session with 300x100 dimensions.
  200 startTmuxSession :: Text -> IO (Either HarnessError ())
  201 startTmuxSession sessionName = do
  202   tmuxCheck <- requireTmux
  203   case tmuxCheck of
  204     Left err -> pure (Left err)
  205     Right () -> do
  206       exists <- sessionExists sessionName
  207       if exists
  208         then pure (Right ())
  209         else do
  210           (exitCode, stderr) <- runTmux
  211             [ "new-session", "-d"
  212             , "-s", T.unpack sessionName
  213             , "-x", "300"
  214             , "-y", "100"
  215             ]
  216           case exitCode of
  217             ExitSuccess   -> pure (Right ())
  218             ExitFailure c -> pure (Left (HarnessTmuxNotAvailable
  219               ("tmux new-session failed (exit " <> T.pack (show c) <> "): "
  220                 <> TE.decodeUtf8Lenient stderr)))

  222 -- | Check if a tmux session with the given name exists.
  223 sessionExists :: Text -> IO Bool
  224 sessionExists sessionName = do
  225   exitCode <- runTmuxSilent ["has-session", "-t", T.unpack sessionName]
  226   pure (exitCode == ExitSuccess)

  228 -- | Strip ANSI escape sequences from a ByteString.
  229 -- Matches ESC [ ... (letter or @) sequences.
  230 stripAnsi :: ByteString -> ByteString
  231 stripAnsi = go
  232   where
  233     go input
  234       | BC.null input = BC.empty
  235       | otherwise =
  236           let (before, rest) = BC.break (== '\ESC') input
  237           in if BC.null rest
  238              then before
  239              else before <> skipEsc (BC.drop 1 rest)
  240     skipEsc input
  241       | BC.null input = BC.empty
  242       | BC.head input == '[' = skipCsi (BC.drop 1 input)
  243       | otherwise = go (BC.drop 1 input)
  244     skipCsi input
  245       | BC.null input = BC.empty
  246       | let c = BC.head input
  247       , c >= '@' && c <= '~' = go (BC.drop 1 input)
  248       | otherwise = skipCsi (BC.drop 1 input)

  265 -- | List all windows in a tmux session, returning @(windowIndex, windowName)@ pairs.
  266 -- Returns an empty list if the session does not exist or tmux is unavailable.
  267 listSessionWindows :: Text -> IO [(Int, Text)]
  268 listSessionWindows sessionName = do
  269   mPath <- findTmux
  270   case mPath of
  271     Nothing -> pure []
  272     Just tmuxBin -> do
  273       let config = P.setStdin P.closed
  274                  $ P.setStdout P.byteStringOutput
  275                  $ P.setStderr P.nullStream
  276                  $ tmuxProc (Command.authorizeTmuxCommand tmuxBin
  277                      [ "list-windows", "-t", sessionName
  278                      , "-F", "#{window_index}\t#{window_name}"
  279                      ])
  280       (exitCode, stdout, _stderr) <- P.readProcess config
  281       case exitCode of
  282         ExitFailure _ -> pure []
  283         ExitSuccess   -> pure (parseListing (LBS.toStrict stdout))
  284   where
  285     parseListing bs =
  286       [ (idx, name)
  287       | line <- BC.lines bs
  288       , let txt = TE.decodeUtf8Lenient line
  289       , (idxStr, rest) <- [T.break (== '\t') txt]
  290       , not (T.null rest)
  291       , let name = T.drop 1 rest  -- drop the tab
  292       , Just idx <- [readIndex idxStr]
  293       ]
  294 
  295     readIndex t = case reads (T.unpack t) of
  296       [(n, "")] -> Just n
  297       _         -> Nothing

  309 -- | Render a tmux target string @\<session\>:\<windowName\>@.
  310 --
  311 -- This is the name-based analogue of the @session:index@ targets the legacy
  312 -- index-based functions build. Pure; the single source of truth for how the
  313 -- name-based ops below address a window.
  314 windowTarget :: Text -> Text -> String
  315 windowTarget sessionName windowName =
  316   T.unpack sessionName <> ":" <> T.unpack windowName
  317 
  318 -- | Validate a tmux session\/window identifier (C3 input hygiene, §8 C3,
  319 -- defense-in-depth).
  320 --
  321 -- The PRIMARY injection defense is argv-style 'P.proc' (via the
  322 -- 'tmuxProc'\/'Command.authorizeTmuxCommand' seam): every identifier is passed
  323 -- as a DISTINCT argv element, so it can never be re-tokenized into extra tmux
  324 -- arguments. 'validateTmuxIdent' is a second layer that rejects identifiers
  325 -- which — even as a single argv element — tmux could misread:
  326 --
  327 --   * a leading @-@ (would be parsed as an OPTION by @set-option@\/@send-keys@
  328 --     @-t <name>@ etc. — the @\@pcl_id@ marker path is the chief concern when a
  329 --     name is derived from an adopted\/discovered external window);
  330 --   * an empty identifier (no window\/session to target);
  331 --   * a NUL or any control character (newline\/tab corrupts the @-F@ sweep
  332 --     parsing and is never a legitimate harness name);
  333 --   * a @:@ (the 'windowTarget' @session:window@ separator — a @:@ inside a
  334 --     single component would let it spill into the next).
  335 --
  336 -- Total and pure. Valid harness names (e.g. @claude-code-0@, @pureclaw@, a bare
  337 -- index like @0@) pass.
  338 validateTmuxIdent :: Text -> Bool
  339 validateTmuxIdent ident =
  340   not (T.null ident)
  341     && T.head ident /= '-'
  342     && T.all ok ident
  343   where
  344     ok c = c /= ':' && not (Char.isControl c)
  345 
  346 -- | Parse a non-negative 'Int' from 'Text', returning 'Nothing' on any

  348 readIntMaybe :: Text -> Maybe Int
  349 readIntMaybe t = case reads (T.unpack t) of
  350   [(n, "")] -> Just n

  357 -- | @send-keys -l --@ argv for a small payload, targeting a window by name.
  358 --
  359 -- C3 input hygiene (§8 C3): the payload is delivered LITERALLY via @-l@ so tmux
  360 -- does NOT interpret it as key tokens (a payload containing @C-c@, @Enter@, @;@,
  361 -- etc. is typed verbatim, never executed as a tmux key/command). The @--@ ends
  362 -- option parsing so a payload that begins with @-@ is never mistaken for a flag.
  363 --
  364 -- This builder does NOT submit the line — 'sendEnterNamedArgs' sends a SEPARATE
  365 -- @Enter@ key. Splitting the literal text from the submit keystroke is what
  366 -- prevents key-token injection while still letting callers actually run a
  367 -- command (literal text, then Enter).
  368 sendKeysNamedArgs :: Text -> Text -> ByteString -> [String]
  369 sendKeysNamedArgs sessionName windowName input =
  370   ["send-keys", "-t", windowTarget sessionName windowName, "-l", "--", BC.unpack input]
  371 
  372 -- | @send-keys ... Enter@ argv — the SEPARATE submit keystroke that follows a
  373 -- 'sendKeysNamedArgs' (or 'pasteBufferNamedArgs') literal payload. Kept distinct
  374 -- from the literal send so injected key tokens in the payload can never be
  375 -- interpreted (C3); the only intentional key tmux sees is this @Enter@.
  376 sendEnterNamedArgs :: Text -> Text -> [String]
  377 sendEnterNamedArgs sessionName windowName =
  378   ["send-keys", "-t", windowTarget sessionName windowName, "Enter"]

  404 -- | @new-window@ argv creating a named window in a session, optionally honoring
  405 -- a working directory (@-c@). The command string is run in the new window.
  406 --
  407 -- The target is the SESSION form @\<session\>:@ (trailing colon), NOT the bare
  408 -- session name. @new-window@'s @-t@ is an /insertion-window/ target: a bare
  409 -- name is resolved by tmux as a window-name match in the CURRENT session first,
  410 -- so if the user has a window named e.g. @pureclaw@ in their attached session,
  411 -- @-t pureclaw@ lands on that window's index and tmux fails with
  412 -- @create window failed: index 0 in use@ — the harness window is never created.
  413 -- The trailing colon pins the target to the session named @\<session\>@, so
  414 -- tmux allocates the next free window index there regardless of any same-named
  415 -- windows elsewhere on the server.
  416 newWindowNamedArgs :: Text -> Text -> Maybe FilePath -> String -> [String]
  417 newWindowNamedArgs sessionName windowName mWorkDir cmd =
  418   ["new-window", "-t", T.unpack sessionName <> ":", "-n", T.unpack windowName]
  419     <> dirArgs
  420     <> [cmd]
  421   where
  422     dirArgs = case mWorkDir of
  423       Just dir -> ["-c", dir]
  424       Nothing  -> []

  451 -- | Send input to a harness window addressed by name.
  452 --
  453 -- C3 input hygiene (§8 C3): small input (<= 256 bytes) is delivered LITERALLY
  454 -- via @send-keys -l --@ (so tmux key tokens in the payload are NOT interpreted),
  455 -- followed by a SEPARATE @Enter@ to submit. Large input uses
  456 -- @load-buffer@ + @paste-buffer@ (already literal) with a separate @Enter@.
  457 --
  458 -- Identifiers are validated (defense-in-depth) before any tmux op: an invalid
  459 -- session\/window name (leading @-@, empty, control char, @:@) is a no-op
  460 -- (fail-closed) since argv targeting could otherwise be misread as options.
  461 sendToWindowNamed :: Text -> Text -> ByteString -> IO ()
  462 sendToWindowNamed sessionName windowName input
  463   | not (validateTmuxIdent sessionName && validateTmuxIdent windowName) = pure ()
  464   | BC.length input <= 256 = do
  465       _ <- runTmuxSilent (sendKeysNamedArgs sessionName windowName input)
  466       _ <- runTmuxSilent (sendEnterNamedArgs sessionName windowName)
  467       pure ()
  468   | otherwise =
  469       Temp.withSystemTempFile "pureclaw-tmux-input" $ \tmpPath tmpHandle -> do
  470         BC.hPut tmpHandle input
  471         IO.hClose tmpHandle
  472         _ <- runTmuxSilent ["load-buffer", tmpPath]
  473         _ <- runTmuxSilent (pasteBufferNamedArgs sessionName windowName)
  474         _ <- runTmuxSilent (sendEnterNamedArgs sessionName windowName)
  475         pure ()
  476 
  477 -- | Type a command LINE into a window and run it: send the bytes LITERALLY
  478 -- (@send-keys -l --@) then a SEPARATE @Enter@ to execute (C3 two-step). Used by
  479 -- the launch path ('addHarnessWindowNamed') for the @cd@ and stealth-launch
  480 -- commands, which MUST still execute. Fail-closed on an invalid identifier.
  481 sendLineNamed :: Text -> Text -> ByteString -> IO ()
  482 sendLineNamed sessionName windowName input
  483   | not (validateTmuxIdent sessionName && validateTmuxIdent windowName) = pure ()
  484   | otherwise = do
  485       _ <- runTmuxSilent (sendKeysNamedArgs sessionName windowName input)
  486       _ <- runTmuxSilent (sendEnterNamedArgs sessionName windowName)
  487       pure ()

  519 -- | Add a harness window addressed by name, given whether the tmux session was
  520 -- just created by us ('TmuxSessionCreated') or already existed
  521 -- ('TmuxSessionExisted').
  522 --
  523 -- A freshly created session still holds only its default window 0 (a bare
  524 -- shell), so we REUSE it: rename it to @windowName@, @cd@ if asked, then run
  525 -- the command. This keeps the FIRST harness in window 0 instead of stranding an
  526 -- idle shell there and opening the harness in window 1. An existing session
  527 -- already has windows, so we append a new named window.
  528 --
  529 -- The status comes from 'startTmuxSessionStatus' and is authoritative. An
  530 -- earlier version inferred freshness from window 0's NAME (@name == index@),
  531 -- but that never holds under tmux @automatic-rename@ — a new session's window 0
  532 -- is named after its shell (e.g. @zsh@), not @0@ — so the reuse branch was dead
  533 -- and every harness, including the first, landed in a new window.
  534 --
  535 -- Targets and names the window by 'Text' rather than by index.
  536 addHarnessWindowNamed :: TmuxSessionStatus -> Text -> Text -> FilePath -> [Text] -> Maybe FilePath -> IO (Either HarnessError ())
  537 addHarnessWindowNamed sessionStatus sessionName windowName binary args mWorkDir = do
  538   tmuxCheck <- requireTmux
  539   case tmuxCheck of
  540     Left err -> pure (Left err)
  541     Right () -> do
  542       let stealthCmd = stealthShellCommand binary args
  543       case sessionStatus of
  544         TmuxSessionCreated -> do
  545           -- Reuse the freshly created session's default window 0: name it, then
  546           -- run. 'renameWindowNamed' targets @session:0@, which resolves to
  547           -- window INDEX 0 (no window is named "0"), so it works regardless of
  548           -- the shell-derived default name.
  549           renameWindowNamed sessionName "0" windowName
  550           case mWorkDir of
  551             Just dir ->
  552               -- C3: send the `cd` command text LITERALLY, then a SEPARATE Enter
  553               -- so it actually executes (the launch path MUST still run).
  554               sendLineNamed sessionName windowName (BC.pack ("cd " <> shellEscapeStr dir))
  555             Nothing -> pure ()
  556           -- Same two-step for the stealth-launch command — literal text, then
  557           -- Enter to execute it.
  558           sendLineNamed sessionName windowName (BC.pack stealthCmd)
  559           pure (Right ())
  560         TmuxSessionExisted -> do
  561           (exitCode, stderr) <- runTmux (newWindowNamedArgs sessionName windowName mWorkDir stealthCmd)
  562           case exitCode of
  563             ExitSuccess   -> pure (Right ())
  564             -- Surface the failure rather than swallowing it as success: a
  565             -- dropped @new-window@ error previously left the caller believing
  566             -- the harness launched (registering a phantom entry the reconciler
  567             -- later evicts) while no window or command ever ran. Fail loud so
  568             -- the create path reports the real tmux error (everything-visible).
  569             ExitFailure c -> pure (Left (HarnessTmuxNotAvailable
  570               ("tmux new-window failed (exit " <> T.pack (show c) <> "): "
  571                 <> TE.decodeUtf8Lenient stderr)))

  616 -- @pcl_id marker, the shell PID, and the pane-dead flag. Returns an empty list
  617 -- if the session does not exist or tmux is unavailable.
  618 readMarkers :: Text -> IO [TmuxWindowRow]
  619 readMarkers sessionName = do
  620   mPath <- findTmux
  621   case mPath of
  622     Nothing -> pure []
  623     Just tmuxBin -> do

  625                  $ P.setStdout P.byteStringOutput
  626                  $ P.setStderr P.nullStream
  627                  $ tmuxProc (Command.authorizeTmuxCommand tmuxBin

  629       (exitCode, stdout, _stderr) <- P.readProcess config

  631         ExitFailure _ -> pure []

  634 -- | Parse the tab-separated 'readMarkers' output into 'TmuxWindowRow's.
  635 -- Tolerant: lines with fewer than five fields, or an unparseable window index,
  636 -- are skipped; an unparseable @pane_pid@ yields 'Nothing'; @pane_dead@ is
  637 -- @True@ iff the field is exactly @1@.
  638 parseMarkerRows :: ByteString -> [TmuxWindowRow]
  639 parseMarkerRows bs =
  640   [ TmuxWindowRow
  641       { _twr_windowIndex = idx
  642       , _twr_windowName  = name
  643       , _twr_pclId       = pclId
  644       , _twr_panePid     = readIntMaybe panePidT
  645       , _twr_paneDead    = paneDeadT == "1"

  647   | line <- BC.lines bs
  648   , let fields = T.splitOn "\t" (TE.decodeUtf8Lenient line)
  649   , (idxT : name : pclId : panePidT : paneDeadT : _) <- [fields]
  650   , Just idx <- [readIntMaybe idxT]
  651   ]

  732 -- ---------------------------------------------------------------------------
  733 -- PID provenance
  734 -- ---------------------------------------------------------------------------
  735 
  736 -- | Given a pane shell PID and the flavour binary name (the @comm@ to match),
  737 -- find the harness process PID by walking the process subtree rooted at the
  738 -- pane PID and matching @comm@.
  739 --
  740 -- NEVER a global @pgrep@\/@grep@ (spike §11 METHOD: a global match kills
  741 -- unrelated processes). On macOS the wrapper exec-collapses so the binary is a
  742 -- direct child (spike §11 E2); on Linux it may be a grandchild via @sh -c@ —
  743 -- the recursive descent handles both.
  744 --
  745 -- Pure over @(pid, ppid, comm)@ rows; chooses the shallowest matching
  746 -- descendant (BFS) so a direct child wins over a deeper one.
  747 --
  748 -- The @(pid, ppid)@ table is untrusted (parsed from @ps@ output, or in tests
  749 -- from arbitrary fixtures) and may describe a cycle — a self-parent
  750 -- (@pid == ppid@), a mutual cycle, or any longer loop. A visited-set bounds the
  751 -- descent so it is total over arbitrary rows: every PID is enqueued at most
  752 -- once, so the BFS terminates on any cycle and still returns the correct
  753 -- shallowest match (or 'Nothing' when no descendant matches).
  754 selectHarnessPid :: Int -> Text -> [(Int, Int, Text)] -> Maybe Int
  755 selectHarnessPid paneShellPid flavourComm rows =
  756   bfs (Set.singleton paneShellPid) [paneShellPid]
  757   where
  758     -- BFS over the descendant frontier; at each level prefer a comm match.
  759     -- @visited@ accumulates every PID already enqueued so a cyclic table cannot
  760     -- re-enqueue a node and loop forever.
  761     bfs :: Set Int -> [Int] -> Maybe Int
  762     bfs _       []      = Nothing
  763     bfs visited parents =
  764       let kids = [ (pid, comm) | (pid, ppid, comm) <- rows, ppid `elem` parents ]
  765       in case [ pid | (pid, comm) <- kids, comm == flavourComm ] of
  766            (p : _) -> Just p
  767            []      ->
  768              -- Descend only into children not yet visited; otherwise a cycle
  769              -- (a ppid pointing back at an ancestor) would loop forever.
  770              let nextPids = filter (`Set.notMember` visited) (map fst kids)
  771              in bfs (foldr Set.insert visited nextPids) nextPids
  772 
  773 -- | IO wrapper around 'selectHarnessPid': gather @(pid, ppid, comm)@ rows via
  774 -- @ps -axo pid=,ppid=,comm=@ and select the harness PID descending from the
  775 -- pane shell PID. Returns 'Nothing' if @ps@ is unavailable or no match is
  776 -- found.
  777 harnessPidOf :: Int -> Text -> IO (Maybe Int)
  778 harnessPidOf paneShellPid flavourComm = do
  779   let config = P.setStdin P.closed
  780              $ P.setStdout P.byteStringOutput
  781              $ P.setStderr P.nullStream
  782              $ P.proc "ps" ["-axo", "pid=,ppid=,comm="]
  783   (exitCode, stdout, _stderr) <- P.readProcess config
  784   case exitCode of
  785     ExitFailure _ -> pure Nothing
  786     ExitSuccess   -> pure (selectHarnessPid paneShellPid flavourComm (parsePsRows (LBS.toStrict stdout)))
  787 
  788 -- | Parse @ps -axo pid=,ppid=,comm=@ output into @(pid, ppid, comm)@ rows.
  789 -- Each line is whitespace-separated: pid, ppid, then the command (the basename
  790 -- of the executable path; tmux's @comm@ matching uses the basename).
  791 parsePsRows :: ByteString -> [(Int, Int, Text)]
  792 parsePsRows bs =
  793   [ (pid, ppid, comm)
  794   | line <- BC.lines bs
  795   , let ws = T.words (TE.decodeUtf8Lenient line)
  796   , (pidT : ppidT : rest) <- [ws]
  797   , not (null rest)
  798   , let comm = lastSegment (T.unwords rest)
  799   , Just pid  <- [readIntMaybe pidT]
  800   , Just ppid <- [readIntMaybe ppidT]
  801   ]
  802   where
  803     -- comm may be a full path on some ps variants; take the basename so it
  804     -- matches the flavour binary name (e.g. "claude").
  805     lastSegment = T.takeWhileEnd (/= '/')

  811 -- | Confirm the running tmux supports the formats the registry depends on:
  812 -- the @pcl_id user-option and the @#{pane_dead}@ format variable. Probes by
  813 -- asking tmux to expand a format string and checking the output is well-formed
  814 -- (tmux leaves an unknown @#{...}@ literal, or errors, when unsupported).
  815 --
  816 -- Returns @Right ()@ when supported; @Left HarnessTmuxNotAvailable@ (with a
  817 -- diagnostic the caller can warn on and then degrade) otherwise.
  818 checkTmuxCapabilities :: IO (Either HarnessError ())
  819 checkTmuxCapabilities = do
  820   tmuxCheck <- requireTmux
  821   case tmuxCheck of
  822     Left err -> pure (Left err)
  823     Right () -> do
  824       mPath <- findTmux
  825       case mPath of
  826         Nothing -> pure (Left (HarnessTmuxNotAvailable "tmux not found"))
  827         Just tmuxBin -> do
  828           -- Expand a probe format. A capable tmux substitutes #{pane_dead}
  829           -- with 0/1 and #{@pcl_id} with the (empty) marker, yielding a line
  830           -- that contains neither a literal "#{" nor "pane_dead".
  831           let config = P.setStdin P.closed
  832                      $ P.setStdout P.byteStringOutput
  833                      $ P.setStderr P.byteStringOutput
  834                      $ tmuxProc (Command.authorizeTmuxCommand tmuxBin
  835                          [ "display-message", "-p", "PCLCAP:#{pane_dead}:#{@pcl_id}:" ])
  836           (exitCode, stdout, _stderr) <- P.readProcess config
  837           let out = TE.decodeUtf8Lenient (LBS.toStrict stdout)
  838           case exitCode of
  839             ExitFailure _ ->
  840               pure (Left (HarnessTmuxNotAvailable
  841                 "tmux does not support the required @pcl_id/pane_dead formats"))
  842             ExitSuccess
  843               | "PCLCAP:" `T.isInfixOf` out
  844               , not ("#{" `T.isInfixOf` out)
  845               , not ("pane_dead" `T.isInfixOf` out) ->
  846                   pure (Right ())
  847               | otherwise ->
  848                   pure (Left (HarnessTmuxNotAvailable
  849                     "tmux does not support the required @pcl_id/pane_dead formats"))
