   99 -- | The one strict session-id predicate. Path-traversal-safe: nonempty, no
  100 -- leading dot, charset @[a-zA-Z0-9_-]@. Used at every boundary where an id
  101 -- is path-joined (parseRef, resumeSession) and at the HTTP/WS surfaces.
  102 -- Single source of truth — do not replicate.
  103 isValidSessionId :: Text -> Bool
  104 isValidSessionId t = not (T.null t) && T.head t /= '.' && T.all ok t
  105   where
  106     ok c = Char.isAsciiLower c || Char.isAsciiUpper c || Char.isDigit c
  107         || c == '_' || c == '-'

  129 -- | The kind of channel a message originated from. Known channels get a
  130 -- typed constructor; 'CkOther' is an open escape hatch for channels that
  131 -- do not (yet) have a dedicated tag.
  132 data ChannelKind
  133   = CkCli
  134   | CkWeb
  135   | CkSignal
  136   | CkTelegram
  137   | CkBackground
  138   | CkOther !Text
  139   deriving stock (Show, Eq, Ord, Generic)
  140 
  141 -- | Render a 'ChannelKind' as a flat lowercase tag. 'CkOther' carries its
  142 -- own name verbatim. Mirrors the @flavourToText@ precedent.
  143 channelKindToText :: ChannelKind -> Text
  144 channelKindToText CkCli        = "cli"
  145 channelKindToText CkWeb        = "web"
  146 channelKindToText CkSignal     = "signal"
  147 channelKindToText CkTelegram   = "telegram"
  148 channelKindToText CkBackground = "background"
  149 channelKindToText (CkOther n)  = n
  150 
  151 -- | Parse a flat tag back into a 'ChannelKind'. Known names map to their
  152 -- typed constructor; everything else becomes 'CkOther'. Critically,
  153 -- @"signal"@ maps to 'CkSignal' (never @CkOther "signal"@), so a
  154 -- 'CkOther' naming a known channel cannot round-trip into existence.
  155 channelKindFromText :: Text -> ChannelKind
  156 channelKindFromText "cli"        = CkCli
  157 channelKindFromText "web"        = CkWeb
  158 channelKindFromText "signal"     = CkSignal
  159 channelKindFromText "telegram"   = CkTelegram
  160 channelKindFromText "background" = CkBackground
  161 channelKindFromText t            = CkOther t
  162 
  163 instance Aeson.ToJSON ChannelKind where
  164   toJSON = Aeson.String . channelKindToText
  165 
  166 instance Aeson.FromJSON ChannelKind where
  167   parseJSON = Aeson.withText "ChannelKind" (pure . channelKindFromText)

  169 -- | Maximum length (in characters) for any attacker-controlled string
  170 -- stored inside a 'MessageSource' after normalization.
  171 maxSourceLen :: Int
  172 maxSourceLen = 512
  173 
  174 -- | Where an inbound message came from. Structured and extensible: a typed
  175 -- 'ChannelKind', the channel's user id (when one exists), and an open
  176 -- field map for supplementary, channel-specific data (which may carry
  177 -- nested JSON).
  178 --
  179 -- Construct via 'mkMessageSource', never the raw record — the smart
  180 -- constructor normalizes attacker-controlled input.
  181 --
  182 -- Invariants (enforced by convention, not the type): '_ms_fields' must not
  183 -- duplicate '_ms_userId' and must never contain credentials/secrets.
  184 --
  185 -- '_ms_conversation' is the server-derived, transport-scoped conversation id
  186 -- (see 'ConversationId'). It is supplied as a required argument to
  187 -- 'mkMessageSource' and is NEVER read from '_ms_fields'/the message body, so a
  188 -- sender cannot forge it to steal another conversation's tab cursor.
  189 data MessageSource = MessageSource
  190   { _ms_channel      :: !ChannelKind
  191   , _ms_conversation :: !ConversationId
  192   , _ms_userId       :: !(Maybe UserId)
  193   , _ms_fields       :: !(Map Text Aeson.Value)
  194   } deriving stock (Show, Eq, Generic)
  195 
  196 -- | Strip ASCII control characters (covers newlines, tabs, carriage
  197 -- returns and other control bytes) and bound the result to 'maxSourceLen'
  198 -- characters.
  199 normalizeText :: Text -> Text
  200 normalizeText = T.take maxSourceLen . T.filter (not . Char.isControl)
  201 
  202 -- | Recursively normalize every 'Aeson.String' leaf in a JSON value,
  203 -- leaving the structure otherwise intact.
  204 normalizeValue :: Aeson.Value -> Aeson.Value
  205 normalizeValue (Aeson.String t) = Aeson.String (normalizeText t)
  206 normalizeValue (Aeson.Array xs) = Aeson.Array (fmap normalizeValue xs)
  207 normalizeValue (Aeson.Object o) = Aeson.Object (fmap normalizeValue o)
  208 normalizeValue v                = v
  209 
  210 -- | Smart constructor for 'MessageSource'. Normalizes attacker-controlled
  211 -- input: strips control characters / newlines and bounds length on the
  212 -- user id and on every string leaf inside the field map, and folds a
  213 -- 'CkOther' naming a known channel (e.g. @CkOther "signal"@) down to its
  214 -- typed constructor.
  215 --
  216 -- The 'ConversationId' is a REQUIRED positional argument and is taken
  217 -- verbatim from the authenticated transport (after the same control-character
  218 -- normalization applied elsewhere). It is never sourced from @fields@, so a
  219 -- @conversation_id@ smuggled into the message body is ignored.
  220 mkMessageSource
  221   :: ChannelKind -> ConversationId -> Maybe UserId -> Map Text Aeson.Value -> MessageSource
  222 mkMessageSource ch (ConversationId conv) uid fields = MessageSource
  223   { _ms_channel      = foldChannel ch
  224   , _ms_conversation = ConversationId (normalizeText conv)
  225   , _ms_userId       = fmap (UserId . normalizeText . unUserId) uid
  226   , _ms_fields       = fmap normalizeValue fields
  227   }
  228   where
  229     foldChannel (CkOther n) = channelKindFromText n
  230     foldChannel c           = c
  231 
  232 instance Aeson.ToJSON MessageSource where
  233   toJSON s = Aeson.object $
  234     ["channel" .= _ms_channel s]
  235       <> ["conversation" .= unConversationId (_ms_conversation s)]
  236       <> case _ms_userId s of
  237         Just u  -> ["user_id" .= unUserId u]
  238         Nothing -> []
  239       <> ["fields" .= _ms_fields s | not (Map.null (_ms_fields s))]
  240     where
  241       unConversationId (ConversationId c) = c
  242 
  243 instance Aeson.FromJSON MessageSource where
  244   parseJSON = Aeson.withObject "MessageSource" $ \o -> do
  245     ch     <- o .:  "channel"
  246     conv   <- o .:? "conversation" .!= ""
  247     uid    <- o .:? "user_id"
  248     fields <- o .:? "fields" .!= Map.empty
  249     pure MessageSource
  250       { _ms_channel      = ch
  251       , _ms_conversation = ConversationId conv
  252       , _ms_userId       = fmap UserId uid
  253       , _ms_fields       = fields
  254       }

  267 -- | Typed allow-list. @AllowAll@ explicitly opts in to allowing everything.
  268 -- @AllowList s@ restricts to the given set.
  269 data AllowList a
  270   = AllowAll
  271   | AllowList (Set a)
  272   deriving stock (Show, Eq, Generic)
  273 
  274 -- | Check whether a value is permitted by the allow-list.
  275 isAllowed :: Ord a => AllowList a -> a -> Bool
  276 isAllowed AllowAll      _ = True
  277 isAllowed (AllowList s) x = Set.member x s
  278 
  279 -- | True when the list permits every sender (no restriction configured).
  280 allowListOpen :: AllowList a -> Bool
  281 allowListOpen AllowAll      = True
  282 allowListOpen (AllowList _) = False

  284 -- | Context needed to render an actionable open-allow-list warning. Carries
  285 --   both the human-readable channel name (for prose) and the lowercase TOML
  286 --   table key (for the copy-pasteable config example) so the two never get
  287 --   conflated.
  288 data AllowListContext = AllowListContext
  289   { _alc_channelName :: Text  -- ^ Human-readable channel name, e.g. "Signal"
  290   , _alc_configTable :: Text  -- ^ TOML table key, e.g. "signal"
  291   , _alc_exampleId   :: Text  -- ^ Example allow_from entry, e.g. a Signal user UUID or a numeric Telegram id
  292   }
  293   deriving stock (Show, Eq)

  295 -- | Banner lines plus a WARN log line describing an OPEN allow-list on the
  296 --   named channel, or Nothing when senders are restricted. Pure.
  297 --
  298 --   When open, the banner is prominent AND actionable: it tells the operator
  299 --   exactly which config file to edit, which (lowercase) TOML table to add an
  300 --   @allow_from@ list under, and shows a concrete copy-pasteable example. The
  301 --   config table key (never the display name) is used everywhere a TOML table
  302 --   is shown.
  303 allowListWarning :: AllowListContext -> AllowList a -> Maybe ([Text], Text)
  304 allowListWarning ctx al
  305   | not (allowListOpen al) = Nothing
  306   | otherwise = Just (banner, logLine)
  307   where
  308     name  = _alc_channelName ctx
  309     table = _alc_configTable ctx
  310     exId  = _alc_exampleId ctx
  311     rule  = T.replicate 64 "="
  312     banner =
  313       [ rule
  314       , "  SECURITY WARNING: channel \"" <> name <> "\" has NO allow-list configured."
  315       , "  It will accept messages from ANY sender, which is a security risk."
  316       , ""
  317       , "  To restrict who may message this agent, edit your config file:"
  318       , "      ~/.pureclaw/config.toml"
  319       , "  and add an `allow_from` list under the [" <> table <> "] section, e.g.:"
  320       , ""
  321       , "      [" <> table <> "]"
  322       , "      allow_from = [\"" <> exId <> "\"]"
  323       , ""
  324       , "  List one ID per allowed sender, then restart PureClaw."
  325       , rule
  326       ]
  327     logLine =
  328       "channel \"" <> name <> "\" has no allow-list configured; "
  329         <> "accepting messages from any sender "
  330         <> "(set allow_from under [" <> table <> "] in config.toml to restrict)"
