   23 -- | A single hunk within a patch file.
   24 data PatchHunk = PatchHunk
   25   { _ph_removals :: [Text]
   26     -- ^ Lines to remove (without the '-' prefix)
   27   , _ph_additions :: [Text]
   28     -- ^ Lines to add (without the '+' prefix)
   29   , _ph_context :: [Text]
   30     -- ^ Context lines (without the ' ' prefix) preceding the change
   31   }
   32   deriving stock (Show, Eq)

   34 -- | A patch for a single file, possibly containing multiple hunks.
   35 data PatchFile = PatchFile
   36   { _pf_path  :: Text
   37     -- ^ File path (relative to workspace root)
   38   , _pf_hunks :: [PatchHunk]
   39   }
   40   deriving stock (Show, Eq)

   42 -- | Create a patch tool that applies unified-diff-style patches
   43 -- to one or more files in a single tool call.
   44 patchTool :: WorkspaceRoot -> FileHandle -> (ToolDefinition, ToolHandler)
   45 patchTool root fh = (def, handler)
   46   where
   47     def = ToolDefinition
   48       { _td_name        = "patch"
   49       , _td_description = T.unlines
   50           [ "Apply a multi-file patch in unified diff format."
   51           , "Accepts a patch string containing one or more file diffs."
   52           , "Format:"
   53           , "  --- a/path/to/file"
   54           , "  +++ b/path/to/file"
   55           , "  @@ ... @@"
   56           , "  -removed line"
   57           , "  +added line"
   58           , "   context line"
   59           , "Multiple files can be patched in a single call."
   60           ]
   61       , _td_inputSchema = object
   62           [ "type" .= ("object" :: Text)
   63           , "properties" .= object
   64               [ "patch" .= object
   65                   [ "type" .= ("string" :: Text)
   66                   , "description" .= ("The patch in unified diff format" :: Text)
   67                   ]
   68               ]
   69           , "required" .= (["patch"] :: [Text])
   70           ]
   71       }
   72 
   73     handler = ToolHandler $ \input ->
   74       case parseEither parseInput input of
   75         Left err -> pure (T.pack err, True)
   76         Right patchText ->
   77           case parsePatch patchText of
   78             Left err -> pure ("Patch parse error: " <> err, True)
   79             Right [] -> pure ("Empty patch — no files to modify", True)
   80             Right files -> applyPatch files
   81 
   82     applyPatch :: [PatchFile] -> IO (Text, Bool)
   83     applyPatch files = do
   84       results <- mapM applyFile files
   85       let (successes, failures) = partitionResults results
   86       if null failures
   87         then pure (T.intercalate "\n" successes, False)
   88         else pure (T.intercalate "\n" (successes <> failures), True)
   89 
   90     applyFile :: PatchFile -> IO (Either Text Text)
   91     applyFile pf = do
   92       let path = _pf_path pf
   93       pathResult <- mkSafePath root (T.unpack path)
   94       case pathResult of
   95         Left pe -> pure (Left (path <> ": " <> T.pack (show pe)))
   96         Right sp -> do
   97           readResult <- try @SomeException (_fh_readFile fh sp)
   98           case readResult of
   99             Left e -> pure (Left (path <> ": " <> T.pack (show e)))
  100             Right bs -> case TE.decodeUtf8' bs of
  101               Left _ -> pure (Left (path <> ": cannot patch binary file"))
  102               Right content -> do
  103                 case applyHunks content (_pf_hunks pf) of
  104                   Left err -> pure (Left (path <> ": " <> err))
  105                   Right newContent -> do
  106                     writeResult <- try @SomeException
  107                       (_fh_writeFile fh sp (TE.encodeUtf8 newContent))
  108                     case writeResult of
  109                       Left e -> pure (Left (path <> ": " <> T.pack (show e)))
  110                       Right () -> pure (Right ("Patched " <> path))
  111 
  112     partitionResults :: [Either Text Text] -> ([Text], [Text])
  113     partitionResults = foldr go ([], [])
  114       where
  115         go (Right s) (ss, fs) = (s:ss, fs)
  116         go (Left  f) (ss, fs) = (ss, f:fs)
  117 
  118     parseInput :: Value -> Parser Text
  119     parseInput = withObject "PatchInput" $ \o -> o .: "patch"

  121 -- | Apply a list of hunks to file content, sequentially.
  122 applyHunks :: Text -> [PatchHunk] -> Either Text Text
  123 applyHunks content [] = Right content
  124 applyHunks content (h:hs) =
  125   case applyHunk content h of
  126     Left err -> Left err
  127     Right content' -> applyHunks content' hs
  128 
  129 -- | Apply a single hunk to file content.
  130 -- Uses context lines to locate the position, then removes the
  131 -- specified lines and inserts the additions.
  132 applyHunk :: Text -> PatchHunk -> Either Text Text
  133 applyHunk content hunk =
  134   let contentLines = T.lines content
  135       -- Build the search pattern: context lines followed by removal lines
  136       searchPat = _ph_context hunk <> _ph_removals hunk
  137   in case findPattern searchPat contentLines of
  138     Nothing ->
  139       -- Try without context (just removals)
  140       case findPattern (_ph_removals hunk) contentLines of
  141         Nothing -> Left "hunk failed: could not find target lines"
  142         Just idx ->
  143           let before = take idx contentLines
  144               after  = drop (idx + length (_ph_removals hunk)) contentLines
  145               result = before <> _ph_additions hunk <> after
  146           in Right (T.unlines result)
  147     Just idx ->
  148       let ctxLen = length (_ph_context hunk)
  149           removeLen = length (_ph_removals hunk)
  150           before = take (idx + ctxLen) contentLines  -- keep context
  151           after  = drop (idx + ctxLen + removeLen) contentLines
  152           result = before <> _ph_additions hunk <> after
  153       in Right (T.unlines result)
  154 
  155 -- | Find the starting index of a pattern (list of lines) within
  156 -- the content lines. Returns the index of the first match.
  157 findPattern :: [Text] -> [Text] -> Maybe Int
  158 findPattern [] _ = Just 0
  159 findPattern _ [] = Nothing
  160 findPattern pat contentLines = go 0 contentLines
  161   where
  162     patLen = length pat
  163     go _ remaining | length remaining < patLen = Nothing
  164     go idx remaining
  165       | take patLen remaining `linesMatch` pat = Just idx
  166       | otherwise = go (idx + 1) (drop 1 remaining)
  167 
  168     linesMatch :: [Text] -> [Text] -> Bool
  169     linesMatch as bs = length as == length bs && all (uncurry matchLine) (zip as bs)
  170 
  171     matchLine :: Text -> Text -> Bool
  172     matchLine a b = T.strip a == T.strip b

  174 -- | Parse a unified diff into a list of 'PatchFile'.
  175 parsePatch :: Text -> Either Text [PatchFile]
  176 parsePatch input =
  177   let ls = T.lines input
  178   in parseFiles ls
  179 
  180 parseFiles :: [Text] -> Either Text [PatchFile]
  181 parseFiles [] = Right []
  182 parseFiles ls =
  183   case dropWhile (not . isFileLine) ls of
  184     [] -> Right []
  185     rest ->
  186       case parseOneFile rest of
  187         Left err -> Left err
  188         Right (pf, remaining) -> do
  189           more <- parseFiles remaining
  190           Right (pf : more)
  191 
  192 isFileLine :: Text -> Bool
  193 isFileLine l = "--- " `T.isPrefixOf` l
  194 
  195 parseOneFile :: [Text] -> Either Text (PatchFile, [Text])
  196 parseOneFile [] = Left "unexpected end of patch"
  197 parseOneFile (_minusLine:rest) =
  198   case rest of
  199     [] -> Left "missing +++ line after ---"
  200     (plusLine:rest') ->
  201       let path = extractPath plusLine
  202       in case parseHunks rest' of
  203         (hunks, remaining) -> Right (PatchFile path hunks, remaining)
  204 
  205 extractPath :: Text -> Text
  206 extractPath line =
  207   let stripped = T.strip line
  208       -- Remove "+++ " or "--- " prefix
  209       withoutPrefix = T.drop 4 stripped
  210       -- Remove "a/" or "b/" prefix if present
  211   in if "a/" `T.isPrefixOf` withoutPrefix || "b/" `T.isPrefixOf` withoutPrefix
  212     then T.drop 2 withoutPrefix
  213     else withoutPrefix
  214 
  215 parseHunks :: [Text] -> ([PatchHunk], [Text])
  216 parseHunks [] = ([], [])
  217 parseHunks ls@(l:_)
  218   | "@@" `T.isPrefixOf` l =
  219       let (hunk, rest) = parseOneHunk (drop 1 ls)  -- skip @@ line
  220           (more, remaining) = parseHunks rest
  221       in (hunk : more, remaining)
  222   | isFileLine l = ([], ls)  -- next file
  223   | otherwise = parseHunks (drop 1 ls)  -- skip non-hunk lines
  224 
  225 parseOneHunk :: [Text] -> (PatchHunk, [Text])
  226 parseOneHunk = go [] [] []
  227   where
  228     go ctx rmv add [] = (PatchHunk rmv add ctx, [])
  229     go ctx rmv add (l:ls)
  230       | "@@" `T.isPrefixOf` l = (PatchHunk rmv add ctx, l:ls)  -- next hunk
  231       | isFileLine l          = (PatchHunk rmv add ctx, l:ls)  -- next file
  232       | "-" `T.isPrefixOf` l && not ("---" `T.isPrefixOf` l) =
  233           go ctx (rmv <> [T.drop 1 l]) add ls
  234       | "+" `T.isPrefixOf` l && not ("+++" `T.isPrefixOf` l) =
  235           go ctx rmv (add <> [T.drop 1 l]) ls
  236       | " " `T.isPrefixOf` l =
  237           if null rmv && null add
  238             then go (ctx <> [T.drop 1 l]) rmv add ls  -- pre-change context
  239             else go ctx rmv add ls  -- post-change context (skip)
  240       | otherwise = go ctx rmv add ls  -- skip other lines
