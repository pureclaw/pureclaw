    1 -- |
    2 -- Module      : PureClaw.Tabs.Relay
    3 -- Description : Streaming-aware per-conversation output relay (Tabs-as-View, #79).
    4 --
    5 -- When a tab (named by its 'TabRef') emits a 'ChannelEvent', 'relayEvent' fans
    6 -- that event out to every conversation according to /that conversation's/
    7 -- effective 'RelayMode' (its per-conversation override, else the global
    8 -- default). This replaces the single-focus @ChannelOut@ gate (@shouldEmit@): a
    9 -- tab can be foreground for conversation A and background for B at the same
   10 -- time, so the decision is taken per conversation rather than once for the
   11 -- whole output (design §9.3).
   12 --
   13 -- Unlike the earlier whole-@Text@ relay, this engine is __streaming-aware__:
   14 -- it sinks a 'ChannelEvent', so a focused conversation receives every
   15 -- @StreamStart@\/@ChunkOf@\/@StreamEnd@ verbatim (full provider streaming),
   16 -- while a background 'ActivityDigest' conversation gets __at most one__
   17 -- breadcrumb ping per /burst/ — one logical message. The burst is identified
   18 -- by a 'BurstKey': all framing events of one stream share @'BurstStream' sid@,
   19 -- and a one-shot @FullMsg@ is its own @'BurstFull'@ burst. The pinged-set is
   20 -- keyed on @('ConversationKey', 'BurstKey')@ so a 100-chunk stream pings once,
   21 -- not 100 times (§4).
   22 --
   23 -- The engine is a __single writer__: the dispatcher invokes it from one
   24 -- thread, and it iterates conversations in a deterministic (sorted-key) order,
   25 -- calling the injected sink in turn. There is no concurrency inside the
   26 -- engine — ordering across the per-conversation sinks is preserved exactly as
   27 -- @ChannelOut@ guaranteed.
   28 --
   29 -- == Per-conversation decision (effective mode @m@ for key @k@)
   30 --
   31 -- Let @focused@ mean the conversation's cursor names the source tab.
   32 --
   33 --   * __Focused__ (any mode) — forward the event, MERGING a one-shot
   34 --     @\"\/N \<model\>: \"@ speaker prefix into the burst-start when more than
   35 --     one tab exists (prepended to a @FullMsg@, injected as the first @ChunkOf@
   36 --     after a @StreamStart@; mid-burst\/banner events pass verbatim). Also clear
   37 --     /every/ @(k, _)@ entry from the pinged-set: the conversation is looking at
   38 --     the tab, so a future background burst should ping it again.
   39 --   * __FocusedOnly__ background — nothing.
   40 --   * __Firehose__ background — forward the event verbatim.
   41 --   * __ActivityDigest__ background — emit at most one ping per burst. A
   42 --     'BannerLine' source event is skipped entirely (banners are
   43 --     dispatcher-class, never relayed here): no ping, set untouched. For a
   44 --     stream\/full event, if @(k, burstKey)@ is not already pinged AND the
   45 --     source ref resolves in the 'TabList', deliver a 'BannerLine' ping
   46 --     (@\"\<name\> (\/N) has new output\"@, where @\<name\>@ is the source
   47 --     tab's friendly name and @N@ its /current/ display slot) and remember
   48 --     @(k, burstKey)@. A 'Dead' tombstone source still pings (§8) — only a
   49 --     source ref /absent/ from the list yields no ping (no name\/slot).
   50 --
   51 -- The set of conversations considered is every key present in the
   52 -- 'CursorState' — the union of the cursor-map keys and the relay-override-map
   53 -- keys — so a conversation that has only set a 'RelayMode' (no cursor yet) is
   54 -- still routed.
   55 module PureClaw.Tabs.Relay

   59   ) where

   79 -- | The /burst/ a 'ChannelEvent' belongs to — one logical message. All framing
   80 -- events of a single provider stream (@StreamStart@\/@ChunkOf@\/@StreamEnd@)
   81 -- share the same @'BurstStream' sid@; a one-shot @FullMsg@ is its own
   82 -- @'BurstFull'@ burst. 'ActivityDigest' de-dups its background ping on
   83 -- @('ConversationKey', 'BurstKey')@, so a multi-chunk stream pings once.
   84 data BurstKey
   85   = BurstStream !StreamId
   86   | BurstFull
   87   deriving stock (Eq, Ord, Show)
   88 
   89 -- | The relay engine's only side-effect seam: a conversation-addressed output
   90 -- sink. @'_rl_sink' k e@ delivers event @e@ to conversation @k@. Injected so
   91 -- the engine can be unit-tested against a recording sink.
   92 newtype RelayDeps = RelayDeps
   93   { _rl_sink :: ConversationKey -> ChannelEvent -> IO ()
   94   }
   95 
   96 -- | The burst a 'ChannelEvent' belongs to, or 'Nothing' for a 'BannerLine'
   97 -- (banners are dispatcher-class and never relayed to background conversations).
   98 burstKeyOf :: ChannelEvent -> Maybe BurstKey
   99 burstKeyOf = \case
  100   StreamStart sid _ -> Just (BurstStream sid)
  101   ChunkOf sid _     -> Just (BurstStream sid)
  102   StreamEnd sid     -> Just (BurstStream sid)
  103   FullMsg _ _       -> Just BurstFull
  104   BannerLine _      -> Nothing
  105 
  106 -- | Fan one tab's 'ChannelEvent' out to every conversation per its effective
  107 -- 'RelayMode'. See the module header for the full per-conversation decision.
  108 --
  109 -- @globalDefault@ is the fallback 'RelayMode' for conversations with no
  110 -- override. @pinged@ is the set of @('ConversationKey', 'BurstKey')@ pairs
  111 -- already activity-pinged (burst de-dup for 'ActivityDigest'); the updated set
  112 -- is returned.
  113 relayEvent
  114   :: RelayDeps
  115   -> CursorState
  116   -> RelayMode
  117   -> TabList
  118   -> Maybe Text
  119      -- ^ The source tab's model name, used to render the focused speaker prefix
  120      --   (@\/N \<model\>: @). 'Nothing' when unknown (e.g. a harness tab), in
  121      --   which case the prefix is slot-only (@\/N: @).
  122   -> Set (ConversationKey, BurstKey)
  123   -> TabRef
  124   -> ChannelEvent
  125   -> IO (Set (ConversationKey, BurstKey))
  126 relayEvent deps cs globalDefault tl srcModel pinged src event =
  127     go pinged conversationKeys
  128   where
  129     -- Every conversation the dispatcher knows about: keys with a cursor or a
  130     -- relay override. Sorted so the single-writer emission order is testable.
  131     conversationKeys :: [ConversationKey]
  132     conversationKeys =
  133       Set.toAscList
  134         (Set.fromList (Map.keys (_cs_cursors cs))
  135            `Set.union` Set.fromList (Map.keys (_cs_relay cs)))
  136 
  137     -- The source tab's current display slot (0-based), if it is in the list.
  138     srcSlot :: Maybe Int
  139     srcSlot = unTabIndex <$> lookupRef src tl
  140 
  141     -- A slot-keyed activity ping for the source tab, as a 'BannerLine'.
  142     -- 'Nothing' when the source ref is absent (no slot). Tabs carry no label of
  143     -- their own anymore, so the source is named by its current slot.
  144     activityPing :: Maybe ChannelEvent
  145     activityPing = do
  146       slot <- srcSlot
  147       pure (BannerLine ("/" <> T.pack (show slot) <> " has new output"))
  148 
  149     -- The speaker prefix to MERGE into a focused burst-start, or 'Nothing' when
  150     -- the source ref has no slot. Rendered as @\/N \<model\>: @: the slot the
  151     -- source tab currently occupies, then the source model when known (omitted
  152     -- for a harness tab / unresolved model), then a @": "@ separator before the
  153     -- content. Every focused burst is labelled — single-tab sessions included —
  154     -- so the speaker is always identifiable (reverses the gb7 single-tab
  155     -- exemption). Tabs carry no label of their own, so the source is named by
  156     -- its current slot.
  157     speakerPrefix :: Maybe Text
  158     speakerPrefix = do
  159       slot <- srcSlot
  160       pure ("/" <> T.pack (show slot) <> maybe "" (" " <>) srcModel <> ": ")
  161 
  162     deliver :: ConversationKey -> ChannelEvent -> IO ()
  163     deliver = _rl_sink deps
  164 
  165     -- Forward the event to a focused conversation, MERGING a one-shot speaker
  166     -- prefix into the START of every burst. The prefix is prepended to a
  167     -- 'FullMsg' and injected as the first 'ChunkOf' right after a 'StreamStart'
  168     -- — so a streamed reply renders the speaker inline, and a non-streaming
  169     -- channel buffers it into the one flushed message. Mid-burst
  170     -- 'ChunkOf'\/'StreamEnd' and 'BannerLine' events forward verbatim, so a
  171     -- multi-chunk reply is prefixed exactly once. Single-tab sessions are
  172     -- labelled too (the always-label decision, reversing the gb7 exemption).
  173     deliverFocused :: ConversationKey -> IO ()
  174     deliverFocused k = mapM_ (deliver k) (focusedEvents event)
  175 
  176     -- The event(s) to deliver to a focused conversation for @ev@: the
  177     -- burst-start carrying the merged speaker prefix when 'speakerPrefix' is
  178     -- 'Just', else the event verbatim. A 'StreamStart' becomes the start plus an
  179     -- injected prefix chunk; a 'FullMsg' has the prefix prepended to its text.
  180     -- 'ChunkOf'\/'StreamEnd'\/'BannerLine' are always forwarded verbatim.
  181     focusedEvents :: ChannelEvent -> [ChannelEvent]
  182     focusedEvents ev = case ev of
  183       StreamStart sid tix -> case speakerPrefix of
  184         Just pfx -> [StreamStart sid tix, ChunkOf sid pfx]
  185         Nothing  -> [ev]
  186       FullMsg tix t -> case speakerPrefix of
  187         Just pfx -> [FullMsg tix (pfx <> t)]
  188         Nothing  -> [ev]
  189       ChunkOf {}    -> [ev]
  190       StreamEnd {}  -> [ev]
  191       BannerLine {} -> [ev]
  192 
  193     go
  194       :: Set (ConversationKey, BurstKey)
  195       -> [ConversationKey]
  196       -> IO (Set (ConversationKey, BurstKey))
  197     go acc []       = pure acc
  198     go acc (k : ks) = do
  199       acc' <-
  200         if Map.lookup k (_cs_cursors cs) == Just src
  201           -- Focused on the source tab: forward the event verbatim in every
  202           -- mode, and clear ALL of this conversation's burst entries (it is
  203           -- now looking at the tab, so a future background burst should ping).
  204           then deliverFocused k >> pure (clearKey k acc)
  205           -- Background: behaviour depends on the conversation's effective mode.
  206           else case relayModeFor k globalDefault cs of
  207             FocusedOnly    -> pure acc
  208             Firehose       -> deliver k event >> pure acc
  209             ActivityDigest -> backgroundDigest acc k
  210       go acc' ks
  211 
  212     -- Drop every @(k, _)@ burst entry for conversation @k@.
  213     clearKey
  214       :: ConversationKey
  215       -> Set (ConversationKey, BurstKey)
  216       -> Set (ConversationKey, BurstKey)
  217     clearKey k = Set.filter ((/= k) . fst)
  218 
  219     -- ActivityDigest for a background conversation: one name-first ping per
  220     -- burst. A 'BannerLine' source is skipped (no burst, set untouched).
  221     -- Otherwise suppressed when @(k, burstKey)@ was already pinged, or when the
  222     -- source ref is absent from the list (no name/slot to render).
  223     backgroundDigest
  224       :: Set (ConversationKey, BurstKey)
  225       -> ConversationKey
  226       -> IO (Set (ConversationKey, BurstKey))
  227     backgroundDigest acc k = case burstKeyOf event of
  228       Nothing -> pure acc
  229       Just bk
  230         | (k, bk) `Set.member` acc -> pure acc
  231         | otherwise -> case activityPing of
  232             Just p  -> deliver k p >> pure (Set.insert (k, bk) acc)
  233             Nothing -> pure acc
