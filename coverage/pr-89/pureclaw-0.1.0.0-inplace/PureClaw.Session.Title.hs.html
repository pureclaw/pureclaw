    1 -- | Canonical displayed title for a session — the single source of truth
    2 -- shared by both the web frontend and the TUI so the two compute a
    3 -- session's name identically.
    4 --
    5 -- This module is a leaf: it depends only on the serializable session and
    6 -- transcript types ('PureClaw.Session.Types', 'PureClaw.Transcript.Types')
    7 -- and must NOT import "PureClaw.Frontend.API" (which would create a cycle).
    8 module PureClaw.Session.Title
    9   ( sessionTitle
   10   , firstMessageSnippet
   11   , snippetCharBudget
   12   ) where

   30 -- | Canonical displayed title: override -> model summary -> first-message
   31 -- snippet -> agent -> session-id prefix. The single source of truth for web
   32 -- AND TUI so defaults are identical.
   33 sessionTitle :: FilePath -> SessionMeta -> IO Text
   34 sessionTitle baseDir meta =
   35   case _sm_description meta of
   36     Just d | not (T.null (T.strip d)) -> pure d
   37     _ -> case _sm_autoSummary meta of
   38       Just s | not (T.null (T.strip s)) -> pure s
   39       _ -> do
   40         mSnip <- firstMessageSnippet baseDir meta
   41         pure $ case mSnip of
   42           Just s | not (T.null s) -> s
   43           _ -> fallbackLabel meta
   44 
   45 -- | Last-resort label when there is no description, summary, or
   46 -- first-message snippet: the agent name, else a prefix of the session id.
   47 fallbackLabel :: SessionMeta -> Text
   48 fallbackLabel meta = case _sm_agent meta of
   49   Just a  -> unAgentName a
   50   Nothing -> T.take 12 (unSessionId (_sm_id meta))
   51 
   52 -- | Cheap fallback for display: read just the first line of the
   53 -- session's @transcript.jsonl@, decode it as a 'TranscriptEntry', and
   54 -- extract a short snippet of the first user message. Returns 'Nothing'
   55 -- when there's no transcript, the first entry isn't a request, or any
   56 -- decoding step fails. The returned string is at most
   57 -- 'snippetCharBudget' chars with newlines normalised to spaces.
   58 --
   59 -- Cost is one bounded read per call (we don't load the whole transcript).
   60 firstMessageSnippet :: FilePath -> SessionMeta -> IO (Maybe Text)
   61 firstMessageSnippet baseDir meta = do
   62   let path = baseDir </> T.unpack (unSessionId (_sm_id meta)) </> "transcript.jsonl"
   63   result <- try @IOException $ withFile path ReadMode BSC.hGetLine
   64   case result of
   65     Left _    -> pure Nothing
   66     Right line -> case Aeson.eitherDecodeStrict' line :: Either String TranscriptEntry of
   67       Left _ -> pure Nothing
   68       Right entry
   69         | _te_direction entry /= Request -> pure Nothing
   70         | otherwise -> pure (snippetFromPayload (_te_payload entry))
   71 
   72 snippetCharBudget :: Int
   73 snippetCharBudget = 120
   74 
   75 -- | Extract a display snippet from a request payload. Two shapes:
   76 -- (1) JSON object with a "messages" array (provider request) — pull
   77 --     text out of the first message; (2) plain text (harness send) —
   78 --     use the payload directly. Trimmed, newline-normalised, and
   79 --     truncated to 'snippetCharBudget' characters.
   80 snippetFromPayload :: Text -> Maybe Text
   81 snippetFromPayload raw = trimAndTruncate <$>
   82   case Aeson.decodeStrict (TE.encodeUtf8 raw) of
   83     Just (Aeson.Object o)
   84       | Just (Aeson.Array msgs) <- KM.lookup "messages" o
   85       , not (V.null msgs)
   86       -> messageText (V.unsafeHead msgs)
   87     _ -> Just raw
   88   where
   89     -- Anthropic-style message content: either a plain string or an
   90     -- array of {type:"text", text:"..."} blocks. We just want a
   91     -- human-readable lead.
   92     messageText :: Aeson.Value -> Maybe Text
   93     messageText (Aeson.Object m) = case KM.lookup "content" m of
   94       Just (Aeson.String s) -> Just s
   95       Just (Aeson.Array bs) -> Just (T.intercalate " " [t | Aeson.Object b <- V.toList bs
   96                                                           , Just (Aeson.String t) <- [KM.lookup "text" b]])
   97       _                     -> Nothing
   98     messageText _ = Nothing
   99 
  100     trimAndTruncate :: Text -> Text
  101     trimAndTruncate t =
  102       let normalized = T.unwords (T.words t)  -- collapse whitespace incl. newlines
  103       in  if T.length normalized > snippetCharBudget
  104             then T.take (snippetCharBudget - 1) normalized <> "\x2026"
  105             else normalized
