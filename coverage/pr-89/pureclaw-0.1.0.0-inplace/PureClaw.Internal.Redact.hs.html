    2 -- Module      : PureClaw.Internal.Redact
    3 -- Description : Redaction helpers for backend error \/ exception 'Show' (WU2).
    4 --
    5 -- All hand-written 'Show' instances in @PureClaw.Handles.Backend@
    6 -- (notably for 'BackendError' and 'BackendException') route through
    7 -- this module so the rendered output never leaks raw hostnames, IPv4
    8 -- addresses, absolute filesystem paths, identity-file basenames, or
    9 -- ssh stderr fragments.
   10 --
   11 -- The matching is intentionally pure-'Text', no regex dependency:
   12 -- every replacement is a left-to-right scan that consumes a token at
   13 -- a time. This makes the redactor easy to property-test (see
   14 -- @test\/Internal\/RedactSpec.hs@) and keeps it usable from the
   15 -- redacted 'Show' instances (no 'IO', no transitive deps).
   16 --
   17 -- See @docs\/terminal-backend-abstractions.md@ § \"Error Model\" and
   18 -- § \"Information Disclosure / Redaction\".

   52 -- | Redact a 'SomeException' to a 'Text' safe for logs and 'Show'.
   53 --
   54 -- The pipeline is, in order: ssh stderr fragments → IPv4 addresses
   55 -- → hostnames → absolute filesystem paths. Each step is idempotent
   56 -- and operates on whatever the previous step produced. Order matters:
   57 -- ssh-stderr replacement first so its embedded hostnames are wiped
   58 -- as part of the fixed fragment, not split mid-token.
   59 redactErr :: SomeException -> Text
   60 redactErr e = redactText (T.pack (Exception.displayException e))
   61 
   62 -- | The shared pipeline used by 'redactErr' and 'redactBackendException'.
   63 --
   64 -- Composition order matters: paths run FIRST (greedy on @/seg/seg/seg@),
   65 -- otherwise a path whose segment names contain dots (e.g.
   66 -- @/var/log/app.example/x@) would be partially eaten by
   67 -- 'redactHostname' before 'redactPathText' ever sees it. After paths
   68 -- are replaced, hostnames and IPv4 are safe to apply (their patterns
   69 -- do not match the @\<path\>@ placeholder), and ssh-stderr fragments
   70 -- are matched last so any hostname / IPv4 embedded in their fixed
   71 -- vocabulary is already gone.
   72 redactText :: Text -> Text
   73 redactText = redactSshStderr . redactIPv4 . redactHostname . redactPathText
   74 
   75 -- | Run an already-stringified payload (e.g. produced by 'show' on a
   76 -- leaf ADT) through the same redaction pipeline 'redactErr' uses.
   77 --
   78 -- Exposed via the @.hs-boot@ so 'PureClaw.Handles.Backend'\'s
   79 -- hand-written 'Show' instances for 'BackendError' \/
   80 -- 'BackendException' can route through the redactor without
   81 -- re-introducing the import cycle that pattern-matching on those
   82 -- types here would create.
   83 redactedShowString :: String -> Text
   84 redactedShowString = redactText . T.pack
   85 
   86 -- | Redact a 'BackendError' by emitting a redaction-safe summary for
   87 -- each constructor. The inner ADT payloads are already redaction-safe
   88 -- (closed sums or short fixed-vocabulary 'InvalidOptionDetail' text),
   89 -- so it is enough to render the constructor name plus the inner
   90 -- 'Show'.
   91 redactBackendError :: BackendError -> Text
   92 redactBackendError e = case e of
   93   BackendBinaryNotFound c ->
   94     T.pack ("BackendBinaryNotFound " <> show c)
   95   BackendPtyAllocFailed f ->
   96     T.pack ("BackendPtyAllocFailed " <> show f)
   97   BackendSshConnectFailed f ->
   98     T.pack ("BackendSshConnectFailed " <> show f)
   99   BackendTmuxTargetMissing t ->
  100     -- 'TmuxTargetRef' carries the (validated) session/window names; the
  101     -- contained Text was produced by the WU10 smart constructors and
  102     -- never contains shell metacharacters, but the names themselves
  103     -- could resemble hostnames (e.g. @prod-db.example.com@). Run the
  104     -- redaction pipeline over the Show form for defence in depth.
  105     redactText (T.pack ("BackendTmuxTargetMissing " <> show t))
  106   BackendInvalidOption d ->
  107     T.pack ("BackendInvalidOption " <> show d)
  108   BackendBufferQuotaExceeded n ->
  109     T.pack ("BackendBufferQuotaExceeded " <> show n)
  110   BackendBrokenTmuxTarget t ->
  111     redactText (T.pack ("BackendBrokenTmuxTarget " <> show t))
  112 
  113 -- | Redact a 'BackendException' by formatting the context tag and
  114 -- pushing the wrapped 'SomeException' through 'redactErr'.
  115 redactBackendException :: BackendException -> Text
  116 redactBackendException be =
  117   "BackendException { _be_context = "
  118     <> showContext (_be_context be)
  119     <> ", _be_cause = "
  120     <> redactErr (_be_cause be)
  121     <> " }"
  122   where
  123     showContext :: BackendContext -> Text
  124     showContext = T.pack . show

  130 -- | Replace each ssh stderr fragment (case-insensitive substring
  131 -- match) with @\<ssh-error\>@. The fixed list intentionally covers
  132 -- the OpenSSH messages most likely to embed hostnames or paths.
  133 redactSshStderr :: Text -> Text
  134 redactSshStderr =
  135   replaceAnyCI sshStderrFragments "<ssh-error>"
  136 
  137 -- | Patterns recognised as ssh stderr — matched case-insensitively
  138 -- as plain substrings (not regex).
  139 sshStderrFragments :: [Text]
  140 sshStderrFragments =
  141   [ "Host key verification failed"
  142   , "Could not resolve hostname"
  143   , "Network is unreachable"
  144   , "Operation timed out"
  145   , "Connection timed out"
  146   , "Connection refused"
  147   , "Connection reset"
  148   , "No route to host"
  149   , "Permission denied"
  150   ]

  152 -- | Replace dotted-quad IPv4 addresses with @\<ipv4\>@. Tokens with
  153 -- four 1–3-digit groups separated by literal dots.
  154 redactIPv4 :: Text -> Text
  155 redactIPv4 = scanReplace stepIPv4
  156 
  157 stepIPv4 :: Text -> Maybe (Int, Text)
  158 stepIPv4 t = do
  159   (g1, r1) <- consumeOctet t
  160   r1'      <- T.stripPrefix "." r1
  161   (g2, r2) <- consumeOctet r1'
  162   r2'      <- T.stripPrefix "." r2
  163   (g3, r3) <- consumeOctet r2'
  164   r3'      <- T.stripPrefix "." r3
  165   (g4, _)  <- consumeOctet r3'
  166   let consumed = g1 + 1 + g2 + 1 + g3 + 1 + g4
  167   pure (consumed, "<ipv4>")
  168   where
  169     consumeOctet :: Text -> Maybe (Int, Text)
  170     consumeOctet s =
  171       let (ds, rest) = T.span Char.isDigit s
  172           n         = T.length ds
  173       in if n >= 1 && n <= 3 then Just (n, rest) else Nothing
  174 
  175 -- | Replace tokens that look like RFC-1123 hostnames (have at least
  176 -- one internal dot, are not purely numeric, and contain only
  177 -- letters/digits/@.@/@-@) with @\<host\>@.
  178 --
  179 -- Tokens like @tmux@ (no dot) or @0@ (numeric) are left alone.
  180 redactHostname :: Text -> Text
  181 redactHostname = scanReplace stepHostname
  182 
  183 stepHostname :: Text -> Maybe (Int, Text)
  184 stepHostname t =
  185   let (tok, _) = T.span isHostByte t
  186       n        = T.length tok
  187   in if looksLikeHostname tok && n > 0
  188        then Just (n, "<host>")
  189        else Nothing
  190   where
  191     isHostByte :: Char -> Bool
  192     isHostByte c = Char.isAscii c && (Char.isAlphaNum c || c == '.' || c == '-')
  193 
  194 -- | A token is a hostname-y string when it
  195 --
  196 -- * has length >= 3,
  197 -- * contains at least one @.@,
  198 -- * has neither leading nor trailing @.@ or @-@,
  199 -- * is not purely composed of digits and dots (that\'s an IPv4-ish
  200 --   string already handled by 'redactIPv4'),
  201 -- * every dot-separated label is non-empty and starts\/ends with an
  202 --   alnum.
  203 looksLikeHostname :: Text -> Bool
  204 looksLikeHostname tok
  205   | T.length tok < 3 = False
  206   | T.head tok == '.' || T.head tok == '-' = False
  207   | T.last tok == '.' || T.last tok == '-' = False
  208   | not (T.any (== '.') tok) = False
  209   | T.all (\c -> Char.isDigit c || c == '.') tok = False
  210   | otherwise = all labelOk (T.split (== '.') tok)
  211   where
  212     labelOk :: Text -> Bool
  213     labelOk lbl =
  214       not (T.null lbl)
  215         && Char.isAlphaNum (T.head lbl)
  216         && Char.isAlphaNum (T.last lbl)
  217 
  218 -- | Replace absolute filesystem paths (tokens beginning with @\/@
  219 -- followed by one or more path-safe characters) with @\<path\>@.
  220 --
  221 -- The path-safe charset is @[A-Za-z0-9_.\/-]@, matching the
  222 -- regex-equivalent @\/[A-Za-z0-9_.\/-]+@.
  223 redactPath :: FilePath -> Text
  224 redactPath = redactPathText . T.pack
  225 
  226 redactPathText :: Text -> Text
  227 redactPathText = scanReplace stepPath
  228 
  229 stepPath :: Text -> Maybe (Int, Text)
  230 stepPath t = case T.uncons t of
  231   Just ('/', rest) ->
  232     let (body, _) = T.span isPathByte rest
  233         n         = T.length body
  234     in if n > 0 then Just (n + 1, "<path>") else Nothing
  235   _ -> Nothing
  236   where
  237     isPathByte :: Char -> Bool
  238     isPathByte c =
  239       Char.isAlphaNum c || c == '_' || c == '.' || c == '/' || c == '-'

  241 -- ---------------------------------------------------------------------------
  242 -- Generic scan-replace machinery
  243 -- ---------------------------------------------------------------------------
  244 
  245 -- | Left-to-right scan: at each position, the step function either
  246 -- recognises a token (returning its byte length and the replacement
  247 -- text) or yields 'Nothing', in which case the current character is
  248 -- copied through and the scan advances by one.
  249 scanReplace :: (Text -> Maybe (Int, Text)) -> Text -> Text
  250 scanReplace step = go mempty
  251   where
  252     go acc t
  253       | T.null t = acc
  254       | otherwise = case step t of
  255           Just (n, replacement) ->
  256             go (acc <> replacement) (T.drop n t)
  257           Nothing -> case T.uncons t of
  258             Just (c, rest) -> go (acc <> T.singleton c) rest
  259             Nothing        -> acc
  260 
  261 -- | Replace any (case-insensitive) substring from the needle list with
  262 -- the given replacement. First match wins per scan step.
  263 replaceAnyCI :: [Text] -> Text -> Text -> Text
  264 replaceAnyCI needles replacement = scanReplace step
  265   where
  266     lowered :: [(Text, Int)]
  267     lowered = [(T.toLower n, T.length n) | n <- needles]
  268     step :: Text -> Maybe (Int, Text)
  269     step t =
  270       let lt = T.toLower t
  271           tryOne (n, len)
  272             | n `T.isPrefixOf` lt = Just (len, replacement)
  273             | otherwise = Nothing
  274       in firstJust (map tryOne lowered)
  275 
  276     firstJust :: [Maybe a] -> Maybe a
  277     firstJust = foldr (<|>) Nothing
  278

  283 -- | Scrub credential prompts in a 'ByteString' chunk.
  284 --
  285 -- The scrubber scans left-to-right for any of the known prompt
  286 -- patterns (case-insensitive on @password:@ \/ @passphrase:@; literal
  287 -- on the other two). When a match is found, the matched trigger is
  288 -- preserved verbatim and everything from the byte immediately
  289 -- following the trigger up to the next newline (or end of chunk) is
  290 -- replaced with the literal sentinel @[REDACTED]@.
  291 --
  292 -- This is stateless across calls; the overlap-window carry-forward
  293 -- across chunk boundaries lives in the drainer (WU7).
  294 credentialPromptScrubber :: ByteString -> ByteString
  295 credentialPromptScrubber = go
  296   where
  297     go :: ByteString -> ByteString
  298     go bs = case firstPromptMatch bs of
  299       Nothing -> bs
  300       Just (start, len) ->
  301         let (before, fromMatch) = BS.splitAt start bs
  302             (matched, afterMatch) = BS.splitAt len fromMatch
  303             (line, rest) = BS.break (== 0x0A) afterMatch
  304             redactedLine
  305               | BS.null line = BS.empty
  306               | otherwise = sentinel
  307         in before <> matched <> redactedLine <> go rest
  308 
  309     sentinel :: ByteString
  310     sentinel = BSC.pack "[REDACTED]"
  311 
  312 -- | Locate the first credential prompt occurrence: returns its
  313 -- starting offset and the length of the matched trigger.
  314 firstPromptMatch :: ByteString -> Maybe (Int, Int)
  315 firstPromptMatch bs = bestOf
  316   [ findCI (BSC.pack "password:")          bs
  317   , findCI (BSC.pack "passphrase:")        bs
  318   , findLit (BSC.pack "[sudo] password for ") bs
  319   , findLit (BSC.pack "Sorry, try again")  bs
  320   ]
  321   where
  322     bestOf :: [Maybe (Int, Int)] -> Maybe (Int, Int)
  323     bestOf = foldr keepEarliest Nothing
  324       where
  325         keepEarliest Nothing acc = acc
  326         keepEarliest (Just x) Nothing = Just x
  327         keepEarliest (Just x@(a, _)) (Just y@(b, _))
  328           | a <= b = Just x
  329           | otherwise = Just y
  330 
  331     findLit :: ByteString -> ByteString -> Maybe (Int, Int)
  332     findLit needle hay =
  333       let (pre, post) = BS.breakSubstring needle hay
  334       in if BS.null post
  335            then Nothing
  336            else Just (BS.length pre, BS.length needle)
  337 
  338     findCI :: ByteString -> ByteString -> Maybe (Int, Int)
  339     findCI needle hay =
  340       let lneedle = lowerAscii needle
  341           lhay    = lowerAscii hay
  342           (pre, post) = BS.breakSubstring lneedle lhay
  343       in if BS.null post
  344            then Nothing
  345            else Just (BS.length pre, BS.length needle)
  346 
  347     lowerAscii :: ByteString -> ByteString
  348     lowerAscii = BS.map toLowerByte
  349 
  350     toLowerByte :: Word8 -> Word8
  351     toLowerByte w
  352       | w >= 0x41 && w <= 0x5A = w + 0x20
  353       | otherwise = w
