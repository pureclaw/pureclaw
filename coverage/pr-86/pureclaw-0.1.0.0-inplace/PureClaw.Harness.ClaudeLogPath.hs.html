    1 -- | A validated, canonical path to a spawned @claude-code@ harness's on-disk
    2 -- JSONL session-log file. This module is the ONLY way to obtain such a path.
    3 --
    4 -- == Why this exists
    5 --
    6 -- A spawned @claude-code@ harness writes its conversation to a JSONL file under
    7 -- @<base>/projects/<sanitized-cwd>/<uuid>.jsonl@. PureClaw wants an optional,
    8 -- read-only, high-fidelity view of that log (see @docs/harness-jsonl-capture.md@).
    9 -- Reading an attacker-controllable path would be a security hole, so — exactly
   10 -- like 'PureClaw.Security.Path.SafePath' — the value constructor is unexported
   11 -- and the smart constructor enforces filesystem-containment, symlink, and
   12 -- owner\/mode invariants before any caller can read the file (WU5 re-opens it).
   13 --
   14 -- == How the path is located (the WU0 refinement)
   15 --
   16 -- The WU0 Phase-0 spike (@docs/harness-jsonl-capture-spike.md@) established that
   17 -- claude-code derives the project directory name by canonicalizing the cwd and
   18 -- then replacing every non-alphanumeric character with @-@. That rule is an
   19 -- undocumented, version-fragile claude internal. We deliberately do NOT
   20 -- reconstruct it. Instead we exploit the fact that PureClaw mints a
   21 -- /globally-unique/ uuid for every harness: we glob
   22 -- @<base>/projects/*/<uuid>.jsonl@. Because the uuid is unique there is at most
   23 -- one hit. Zero hits is a typed not-found error; more than one hit (which should
   24 -- be impossible) is a typed ambiguity error — we never silently pick one.
   25 --
   26 -- == Security checks (mirror 'SafePath' + a net-new @O_NOFOLLOW@ open)
   27 --
   28 --   1. 'canonicalizePath' the globbed candidate and verify it is contained under
   29 --      the /canonical/ @<base>/projects@ root. A symlink whose target escapes
   30 --      that root is rejected.
   31 --   2. Open the FINAL path component with @O_NOFOLLOW@ ('System.Posix.IO.openFd'
   32 --      with @'PIO.nofollow' = True@). 'System.Posix.Files.getFileStatus' FOLLOWS
   33 --      symlinks, so the existing 'PureClaw.Security.Path.checkModeAndOwner' is
   34 --      insufficient against a symlinked leaf; we do a real @O_NOFOLLOW@ open and
   35 --      @fstat@ the resulting fd. The fd is always closed (bracketed) — this WU
   36 --      only validates; WU5 re-opens the validated path to read.
   37 --   3. Reject if the file owner is not the effective uid, or if the mode is
   38 --      group\/other-writable.
   39 module PureClaw.Harness.ClaudeLogPath
   40   ( -- * Validated log path (constructor intentionally NOT exported)
   41     SafeClaudeLogPath
   42   , getSafeClaudeLogPath
   43     -- * Base directory (injectable for testability)
   44   , ClaudeBase
   45   , mkClaudeBase
   46   , getClaudeBase
   47   , resolveClaudeBase
   48   , chooseBase
   49     -- * Errors
   50   , ClaudeLogPathError (..)
   51     -- * Smart constructor (the ONLY way to obtain a 'SafeClaudeLogPath')

   75 -- | A filesystem path validated to point at a contained, owner-only
   76 -- @claude-code@ JSONL session log. The value constructor is intentionally NOT
   77 -- exported — the only way to obtain a 'SafeClaudeLogPath' is through
   78 -- 'mkSafeClaudeLogPath'. The 'Show' instance is REDACTED so the concrete path
   79 -- never leaks into general logs (mirrors WU1's 'ClaudeSessionUuid').
   80 newtype SafeClaudeLogPath = SafeClaudeLogPath { getSafeClaudeLogPath :: FilePath }
   81   deriving stock (Eq, Ord)
   82 
   83 -- | Redacted on purpose. See 'SafeClaudeLogPath' haddock. (D2.5)
   84 instance Show SafeClaudeLogPath where
   85   show _ = "SafeClaudeLogPath <redacted>"

   87 -- | The claude-code config base directory: @CLAUDE_CONFIG_DIR@ if set, else
   88 -- @~/.claude@. Wrapped so the env\/home resolution ('resolveClaudeBase') is a
   89 -- thin IO shell over a value that tests can inject directly via 'mkClaudeBase'
   90 -- (a temp dir), keeping the validation core testable without mutating global
   91 -- process env.
   92 newtype ClaudeBase = ClaudeBase { getClaudeBase :: FilePath }
   93   deriving stock (Eq, Show)
   94 
   95 -- | Wrap a base directory. Exposed so tests can point the smart constructor at
   96 -- a temp dir without touching @CLAUDE_CONFIG_DIR@ or @$HOME@.
   97 mkClaudeBase :: FilePath -> ClaudeBase
   98 mkClaudeBase = ClaudeBase

  100 -- | Resolve the claude-code base directory: @CLAUDE_CONFIG_DIR@ if present and
  101 -- non-empty in the environment, otherwise @~/.claude@ (HOME expanded). This is
  102 -- the only env\/home-touching part; it reads the two IO inputs and defers the
  103 -- decision to the pure 'chooseBase' so the choice logic is unit-testable
  104 -- without mutating process env.
  105 resolveClaudeBase :: IO ClaudeBase
  106 resolveClaudeBase = do
  107   mEnv <- Env.lookupEnv "CLAUDE_CONFIG_DIR"
  108   chooseBase mEnv <$> Dir.getHomeDirectory
  109 
  110 -- | Pure base-directory choice (exported for unit testing). Use
  111 -- @CLAUDE_CONFIG_DIR@ when it is set AND non-empty; otherwise fall back to
  112 -- @<home>/.claude@. An empty env value is treated as unset (some shells export
  113 -- empty strings).
  114 chooseBase :: Maybe FilePath -> FilePath -> ClaudeBase
  115 chooseBase mEnv home =
  116   case mEnv of
  117     Just dir | not (null dir) -> ClaudeBase dir
  118     _ -> ClaudeBase (home </> ".claude")

  120 -- | Reasons a 'SafeClaudeLogPath' cannot be obtained.
  121 data ClaudeLogPathError
  122   = -- | No @<base>/projects/*/<uuid>.jsonl@ matched the uuid. Carries the
  123     -- canonical projects-root that was searched (or the raw root when it did
  124     -- not exist).
  125     ClaudeLogNotFound FilePath
  126   | -- | More than one candidate matched the (supposedly unique) uuid. Carries
  127     -- every matching path. We never silently pick one.
  128     ClaudeLogAmbiguous [FilePath]
  129   | -- | The canonicalized candidate escaped the canonical projects root
  130     -- (symlink escape). Carries @(canonicalProjectsRoot, canonicalCandidate)@.
  131     ClaudeLogEscapesRoot FilePath FilePath
  132   | -- | The @O_NOFOLLOW@-opened file failed the owner\/mode check. Carries
  133     -- @(path, actualMode masked to 0o777, fileOwner)@.
  134     ClaudeLogInsecure FilePath FileMode UserID
  135   deriving stock (Show, Eq)
  136 
  137 --------------------------------------------------------------------------------
  138 -- Pure owner/mode predicate (unit-testable in isolation, D2.4)
  139 
  140 -- | The pure security predicate factored out of the @O_NOFOLLOW@ open path so
  141 -- it can be unit-tested with synthetic values (you cannot @chown@ to a foreign
  142 -- uid in CI). Accept iff the file is owned by the effective uid AND is not
  143 -- group- or other-writable.
  144 --
  145 -- @ownerModeOk euid mode owner@:
  146 --
  147 --   * @owner == euid@
  148 --   * @mode .&. 0o022 == 0@  (no group-write @0o020@, no other-write @0o002@)
  149 ownerModeOk :: UserID -> FileMode -> UserID -> Bool
  150 ownerModeOk euid mode owner =
  151   owner == euid && (mode `PF.intersectFileModes` 0o022) == 0

  154 -- Smart constructor
  155 
  156 -- | The ONLY way to obtain a 'SafeClaudeLogPath'.
  157 --
  158 -- @mkSafeClaudeLogPath base uuid mExpectedCwd@:
  159 --
  160 --   1. Glob @<base>/projects/*/<uuid>.jsonl@ (D2.2). Zero hits ⇒
  161 --      'ClaudeLogNotFound'; >1 hit ⇒ 'ClaudeLogAmbiguous' (never silently
  162 --      pick).
  163 --   2. 'canonicalizePath' the single hit and verify containment under the
  164 --      canonical projects root (D2.3) ⇒ 'ClaudeLogEscapesRoot' on escape.
  165 --   3. @O_NOFOLLOW@-open the leaf, @fstat@ it, and apply 'ownerModeOk' (D2.4)
  166 --      ⇒ 'ClaudeLogInsecure' on failure. The fd is bracketed (never leaks).
  167 --
  168 -- The optional @mExpectedCwd@ (the persisted canonical spawn-cwd from WU6) is
  169 -- reserved for a future cross-check\/fallback — the PRIMARY derivation is the
  170 -- uuid-glob, which sidesteps reproducing claude's fragile cwd→dirname rule.
  171 mkSafeClaudeLogPath
  172   :: ClaudeBase
  173   -> ClaudeSessionUuid
  174   -> Maybe FilePath
  175   -- ^ persisted canonical spawn-cwd (cross-check\/fallback; currently unused —
  176   -- accepted now so the signature is stable for WU6).
  177   -> IO (Either ClaudeLogPathError SafeClaudeLogPath)
  178 mkSafeClaudeLogPath (ClaudeBase base) uuid _mExpectedCwd = do
  179   let projectsRoot = base </> "projects"
  180       fileName = T.unpack (unClaudeSessionUuid uuid) <> ".jsonl"
  181   rootExists <- Dir.doesDirectoryExist projectsRoot
  182   if not rootExists
  183     then pure (Left (ClaudeLogNotFound projectsRoot))
  184     else do
  185       hits <- globUuidLogs projectsRoot fileName
  186       case hits of
  187         [] -> pure (Left (ClaudeLogNotFound projectsRoot))
  188         (_ : _ : _) -> pure (Left (ClaudeLogAmbiguous hits))
  189         [candidate] -> validateCandidate projectsRoot candidate
  190 
  191 -- | Find every @<projectsRoot>/<subdir>/<fileName>@ that exists. We use
  192 -- 'listDirectory' + 'doesPathExist' rather than adding a @Glob@ dependency.
  193 -- Subdirectory entries that are not directories are simply skipped (their
  194 -- @</fileName>@ child cannot exist).
  195 globUuidLogs :: FilePath -> FilePath -> IO [FilePath]
  196 globUuidLogs projectsRoot fileName = do
  197   entries <- Dir.listDirectory projectsRoot
  198   let candidates = [projectsRoot </> e </> fileName | e <- entries]
  199   Monad.filterM Dir.doesPathExist candidates
  200 
  201 -- | Canonical-containment check (D2.3) followed by the @O_NOFOLLOW@ owner\/mode
  202 -- check (D2.4).
  203 --
  204 -- Two distinct paths are used on purpose:
  205 --
  206 --   * Containment (D2.3) is checked on the CANONICAL candidate (symlinks
  207 --     resolved), so a symlinked project directory or a leaf symlink whose
  208 --     /target/ escapes the projects root is rejected.
  209 --   * The @O_NOFOLLOW@ open (D2.4) is performed on the ORIGINAL (literal,
  210 --     un-canonicalized) candidate leaf. Canonicalizing first would resolve a
  211 --     leaf symlink away before the open, defeating @O_NOFOLLOW@; opening the
  212 --     literal leaf means a symlinked leaf — even one whose target is in-root —
  213 --     fails the open (ELOOP) and is rejected.
  214 --
  215 -- The stored 'SafeClaudeLogPath' carries the CANONICAL path: it is the stable,
  216 -- fully-resolved location WU5 re-opens to read.
  217 validateCandidate
  218   :: FilePath
  219   -> FilePath
  220   -> IO (Either ClaudeLogPathError SafeClaudeLogPath)
  221 validateCandidate projectsRoot candidate = do
  222   canonRoot <- Dir.canonicalizePath projectsRoot
  223   canonCandidate <- Dir.canonicalizePath candidate
  224   if not (isContainedUnder canonRoot canonCandidate)
  225     then pure (Left (ClaudeLogEscapesRoot canonRoot canonCandidate))
  226     else checkOwnerModeNoFollow candidate canonCandidate
  227 
  228 -- | True iff @child@ is @root@ itself or strictly below it. We compare on
  229 -- 'System.FilePath.splitDirectories'-style component prefixes rather than raw
  230 -- string 'List.isPrefixOf' so that @\/a\/bc@ is NOT considered under @\/a\/b@.
  231 isContainedUnder :: FilePath -> FilePath -> Bool
  232 isContainedUnder root child =
  233   let rootC = splitNonEmpty root
  234       childC = splitNonEmpty child
  235   in rootC `List.isPrefixOf` childC
  236   where
  237     splitNonEmpty :: FilePath -> [String]
  238     splitNonEmpty = filter (not . null) . splitOnSlash
  239     splitOnSlash :: String -> [String]
  240     splitOnSlash s = case break (== '/') s of
  241       (h, []) -> [h]
  242       (h, _ : t) -> h : splitOnSlash t
  243 
  244 -- | Open the LITERAL leaf (@openPath@) with @O_NOFOLLOW@, @fstat@ the fd, apply
  245 -- the pure 'ownerModeOk' predicate, and ALWAYS close the fd (bracket). On
  246 -- success store @canonPath@ (the canonical, fully-resolved location WU5 reads).
  247 --
  248 -- If the literal leaf is a symlink, @O_NOFOLLOW@ makes 'PIO.openFd' fail with
  249 -- @ELOOP@; we surface that (and any other open failure) as 'ClaudeLogInsecure'
  250 -- rather than letting the exception escape, because a symlinked\/unopenable leaf
  251 -- is by definition an insecure candidate.
  252 --
  253 -- 'getFileStatus' (used by the existing 'PureClaw.Security.Path.checkModeAndOwner')
  254 -- FOLLOWS symlinks, so it cannot defeat a symlinked leaf — that is precisely why
  255 -- we re-open the LITERAL leaf with @O_NOFOLLOW@ and @fstat@ the resulting fd.
  256 checkOwnerModeNoFollow
  257   :: FilePath
  258   -- ^ literal candidate leaf to open with @O_NOFOLLOW@
  259   -> FilePath
  260   -- ^ canonical path to store on success
  261   -> IO (Either ClaudeLogPathError SafeClaudeLogPath)
  262 checkOwnerModeNoFollow openPath canonPath = do
  263   euid <- PU.getEffectiveUserID
  264   let flags = PIO.defaultFileFlags { PIO.nofollow = True }
  265   opened <- Exc.try (PIO.openFd openPath PIO.ReadOnly flags)
  266   case opened of
  267     Left (_ :: IOException) ->
  268       -- Open failed (e.g. ELOOP because the leaf is a symlink). Treat as
  269       -- insecure; report a zeroed mode\/owner since we never obtained a status.
  270       pure (Left (ClaudeLogInsecure canonPath 0 0))
  271     Right fd ->
  272       Exc.bracket (pure fd) PIO.closeFd $ \openedFd -> do
  273         status <- PF.getFdStatus openedFd
  274         let actualMode = PF.fileMode status `PF.intersectFileModes` 0o777
  275             owner = PF.fileOwner status
  276         pure $
  277           if ownerModeOk euid actualMode owner
  278             then Right (SafeClaudeLogPath canonPath)
  279             else Left (ClaudeLogInsecure canonPath actualMode owner)
