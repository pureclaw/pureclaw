    2 -- Module      : PureClaw.Routing.TabDispatch
    3 -- Description : Per-conversation inbound router + flat tab commands (8b.4).
    4 --
    5 -- This is the per-conversation routing brain of the Tabs-as-View cutover
    6 -- (GitHub #79; spike §8). 'handleInbound' takes one inbound message tagged
    7 -- with its 'ConversationKey' and routes it against /this/ conversation's
    8 -- cursor and the global 'TabRegistry':
    9 --
   10 --   1. __Wizard interception__ — if the conversation is mid-@\/tab@, its next
   11 --      message is consumed by the wizard ('stepWizard') /before/ the grammar
   12 --      sees it (design §9.2 / §11). A completed pick binds a new tab; a
   13 --      slash-prefixed reply cancels the wizard and re-dispatches that command.
   14 --   2. __Flat tab verbs__ — @\/new@, @\/nt@, @\/close@, @\/tabs@, @\/rename@,
   15 --      @\/relay@, @\/tab@ — recognised directly on the trimmed text (design
   16 --      §7). These are __not__ routed through the legacy @\/tab \<sub\>@
   17 --      grammar.
   18 --   3. __Routing grammar__ — anything else goes through
   19 --      'PureClaw.Routing.Parse.parseInput': @\/N@ switch, @\/N \<text\>@
   20 --      inject, default text to the active tab, or a non-tab slash command
   21 --      handed to '_td_fallthrough'.
   22 --
   23 -- Every effect is injected through 'TabDispatchDeps' (no real provider /
   24 -- harness / Exec), so the whole router is unit-testable with recording fakes
   25 -- and the pinned design §14 copy is asserted byte-for-byte. This module is
   26 -- /additive/ — it is wired into the live loop at 8c, replacing the old
   27 -- @dispatchOne@.

  168 -- | The deps + the conversation being served, bundled so every helper takes a
  169 -- single context argument. This keeps each effect call a saturated 2-arg
  170 -- application (e.g. @emit ctx msg@) — passing @deps@ and @convKey@ as two

  182 -- | Route one inbound message for a conversation. See the module haddock for
  183 -- the three-stage order (wizard → flat verbs → routing grammar).
  184 handleInbound :: TabDispatchDeps -> ConversationKey -> Text -> IO ()
  185 handleInbound deps convKey raw = do
  186   let ctx = Ctx deps convKey
  187   wiz <- readIORef (_td_wizard deps)
  188   case Map.lookup convKey wiz of
  189     Just st -> handleWizardReply ctx st raw
  190     Nothing -> handleNonWizard ctx raw
  191 
  192 -- ---------------------------------------------------------------------------
  193 -- Parsed-command entry point
  194 -- ---------------------------------------------------------------------------
  195 
  196 -- | Execute a parsed @\/tab@ command against the shared tab subsystem, with an
  197 -- explicit conversation key (the web path supplies a synthetic key; the
  198 -- TUI\/channel path supplies the real one). Reuses the same helpers the
  199 -- flat-verb handlers use, so every surface produces identical results.
  200 --
  201 -- This is /additive/: it does not change the flat-verb handlers
  202 -- ('cmdClose' \/ 'cmdRename' \/ …) and is not yet wired into the live loop —
  203 -- a later task connects it to @executeSlashCommand@.
  204 runTabCommand :: TabDispatchDeps -> ConversationKey -> TabSlashCommand -> IO ()
  205 runTabCommand deps conv cmd =
  206   let ctx = Ctx deps conv
  207       -- The parser guarantees a non-negative decimal index ('parseDecimalIndex'
  208       -- rejects non-digits), and 'mkTabIndex' only checks the floor (n >= 0), so
  209       -- the 'Nothing' arm is genuinely unreachable here. (An over-cap index is
  210       -- NOT rejected at parse time — it resolves to a graceful "no tab" message
  211       -- at registry-lookup time inside the handlers.) The impossible 'Nothing'
  212       -- is folded into 'error' per this module's convention.
  213       idx n = Maybe.fromMaybe
  214                 (impossible "runTabCommand: non-negative parser index unexpectedly rejected")
  215                 (mkTabIndex n)
  216   in case cmd of
  217        TabListCmd             -> cmdTabs   ctx
  218        TabRenameCmd n name    -> doRename  ctx (idx n) name
  219        TabCloseCmd _ ForceYes -> emit ctx closeForceMsg   -- mirror /close's --force rejection
  220        TabCloseCmd n ForceNo  -> closeSlot ctx (idx n)
  221        TabFocusCmd n          -> doSwitch  ctx (idx n)
  222        TabResumeCmd sid       -> bindNewTab ctx (BoundSession sid) "attached"
  223        TabNewCmd mKind mArg   -> cmdTabNew ctx (kindWords mKind ++ argWords mArg)
  224   where
  225     -- Reconstruct the @[Text]@ 'cmdTabNew' re-parses: the kind keyword (if any)
  226     -- followed by the remaining argument words.
  227     kindWords = maybe [] (\k -> [tabKindArgText k])
  228     argWords  = maybe [] T.words

  230 -- | Build the '_env_runTabCommand' seam closure: route a parsed @\/tab@ command
  231 -- through 'runTabCommand' over the shared dispatch deps, sending the dispatcher
  232 -- reply to the caller-supplied channel (so the web capture channel receives it),
  233 -- and substituting the web placeholder conversation key when none is supplied.
  234 --
  235 -- This is the named, testable extraction of the inline lambda that
  236 -- "PureClaw.CLI.Commands" assigns to '_env_runTabCommand'. The behaviour is
  237 -- identical: the '_td_emit' field is overridden so the dispatcher's reply goes
  238 -- to @chan@ (the per-request capture channel) instead of the conversation's
  239 -- live sink, and a missing 'ConversationKey' falls back to @placeholderKey@.
  240 mkRunTabCommandSeam
  241   :: TabDispatchDeps
  242   -> ConversationKey
  243      -- ^ placeholder key used when the caller passes 'Nothing' (the web path)
  244   -> (ChannelHandle -> Maybe ConversationKey -> TabSlashCommand -> IO ())
  245 mkRunTabCommandSeam deps placeholderKey chan mConv =
  246   runTabCommand
  247     (deps { _td_emit = \_ t -> _ch_send chan (OutgoingMessage t) })
  248     (Maybe.fromMaybe placeholderKey mConv)

  250 -- | Render a 'TabKindArg' back to the lowercase keyword 'cmdTabNew' recognises
  251 -- (the inverse of 'PureClaw.Agent.SlashCommands.parseTabKindArg').
  252 tabKindArgText :: TabKindArg -> Text
  253 tabKindArgText TkaAi       = "ai"
  254 tabKindArgText TkaProvider = "provider"
  255 tabKindArgText TkaHarness  = "harness"
  256 tabKindArgText TkaShell    = "shell"
  257 tabKindArgText TkaSsh      = "ssh"
  258 tabKindArgText TkaTmux     = "tmux"

  264 -- | Feed a reply to an in-flight wizard and act on the resulting 'WizardStep'.
  265 handleWizardReply :: Ctx -> WizardState -> Text -> IO ()
  266 handleWizardReply ctx st raw = do
  267   (mNext, step) <- stepWizard (WizardEnv (_td_liveHarness (_ctx_deps ctx))) st raw
  268   case step of
  269     Done target -> do
  270       clearWizard ctx
  271       bindWizardTarget ctx target
  272     Cancelled -> do
  273       clearWizard ctx
  274       emit ctx "wizard cancelled"
  275     RunCommand cmd -> do
  276       clearWizard ctx
  277       handleNonWizard ctx cmd
  278     Reprompt msg -> do
  279       -- Invalid reply or vanished target: persist the (possibly refreshed)
  280       -- state and re-show the notice. 'Prompt' is the open-time render and is
  281       -- never produced by 'stepWizard'; it is folded in here so the match is
  282       -- total — both carry a notice, both re-prompt.
  283       setWizard ctx mNext
  284       emit ctx msg
  285     Prompt msg -> do
  286       setWizard ctx mNext
  287       emit ctx msg
  288 
  289 -- | Bind a completed wizard pick to a new tab: append (or switch to an
  290 -- already-bound ref), ensure its runtime, focus the conversation, confirm.
  291 bindWizardTarget :: Ctx -> WizardTarget -> IO ()
  292 bindWizardTarget ctx target =
  293   bindNewTab ctx ref "attached"
  294   where
  295     ref = case target of
  296       AttachHarness h -> BoundHarness h
  297       ReopenSession s -> BoundSession s

  303 -- | Recognise flat tab verbs on the trimmed text; otherwise fall to the
  304 -- routing grammar.
  305 handleNonWizard :: Ctx -> Text -> IO ()
  306 handleNonWizard ctx raw =
  307   case T.words trimmed of
  308     ("/new"    : _)    -> cmdNew    ctx
  309     ("/nt"     : _)    -> cmdNt     ctx
  310     ("/close"  : args) -> cmdClose  ctx args
  311     ("/tabs" : sub@(_:_)) -> handleNonWizard ctx (T.unwords ("/tab" : sub))  -- /tabs <sub> == /tab <sub>
  312     ["/tabs"]          -> cmdTabs   ctx   -- bare /tabs lists
  313     ("/rename" : args) -> cmdRename ctx args
  314     ("/relay"  : args) -> cmdRelay  ctx args
  315     ("/tab" : "new" : rest)    -> cmdTabNew ctx rest
  316     ("/tab" : "close" : rest)  -> cmdClose  ctx rest
  317     ("/tab" : "rename" : rest) -> cmdRename ctx rest
  318     ("/tab"    : args) -> cmdTab    ctx args
  319     _                  -> routeGrammar ctx raw
  320   where
  321     trimmed = T.strip raw

  323 -- ---------------------------------------------------------------------------
  324 -- /new — reset the active tab
  325 -- ---------------------------------------------------------------------------
  326 
  327 -- | @\/new@: reset the conversation's __active__ tab to a fresh default
  328 -- session (same slot). With no active tab, behaves like @\/nt@.
  329 cmdNew :: Ctx -> IO ()
  330 cmdNew ctx = do
  331   cs <- readIORef (_td_cursors (_ctx_deps ctx))
  332   tl <- readTabs (_td_tabs (_ctx_deps ctx))
  333   case resolveCursorSlot (_ctx_conv ctx) cs tl of
  334     Nothing   -> cmdNt ctx   -- no active tab → create one
  335     Just slot -> resetActiveTab ctx slot
  336 
  337 -- | Reset the tab at @slot@: mint a fresh session, rebind the slot to it
  338 -- (releasing the old ref's runtime, ensuring the new), keep the slot, point
  339 -- the cursor at the new ref, confirm. The previous session persists on disk
  340 -- (I4) — only the tab's binding moves.
  341 resetActiveTab :: Ctx -> TabIndex -> IO ()
  342 resetActiveTab ctx slot = do
  343   mNew <- _td_newDefaultSession (_ctx_deps ctx)
  344   case mNew of
  345     Left msg     -> emit ctx msg
  346     Right newRef -> do
  347       -- The slot is present (the caller resolved it from a live cursor) and
  348       -- @newRef@ is freshly minted, so it is never already bound: 'rebindSlot'
  349       -- always succeeds and 'registryLookupSlot' is always 'Just' here. The
  350       -- structurally-impossible branches are folded into 'error' (mirroring
  351       -- "PureClaw.Routing.Parse"'s 'boundedTabIndex') so the HPC report has no
  352       -- dead defensive arm to flag.
  353       mOld <- registryLookupSlot (_td_tabs (_ctx_deps ctx)) slot
  354       tl   <- readTabs (_td_tabs (_ctx_deps ctx))
  355       let oldRef = _tab_ref (Maybe.fromMaybe (impossible "resetActiveTab: slot vanished") mOld)
  356           tl'    = either (impossible "resetActiveTab: fresh ref already bound")
  357                           id
  358                           (rebindSlot slot newRef tl)
  359       writeTabs ctx tl'
  360       release ctx oldRef
  361       ensure ctx newRef
  362       setCursorTo ctx newRef
  363       notifyChanged ctx
  364       emit ctx ("reset tab /" <> slotChar slot)
  365 
  366 -- ---------------------------------------------------------------------------
  367 -- /nt — new tab with a fresh default session
  368 -- ---------------------------------------------------------------------------
  369 
  370 -- | @\/nt@: create a fresh default session, append a tab at the next slot,
  371 -- ensure its runtime, switch the conversation's cursor to it.
  372 cmdNt :: Ctx -> IO ()
  373 cmdNt ctx = do
  374   mNew <- _td_newDefaultSession (_ctx_deps ctx)
  375   case mNew of
  376     Left msg     -> emit ctx msg
  377     Right newRef -> bindNewTab ctx newRef "new tab"

  379 -- ---------------------------------------------------------------------------
  380 -- Shared: append-or-switch a new tab + confirm
  381 -- ---------------------------------------------------------------------------
  382 
  383 -- | Append a tab binding @ref@, ensure its runtime, focus the conversation on
  384 -- it, and emit @\<verb\> \/\<slot\>@. If @ref@ is already bound (I2), switch to
  385 -- the existing tab instead of duplicating. Slot exhaustion emits the §14 copy
  386 -- with no state change. Tabs carry no label of their own — the display title
  387 -- derives from the bound session (or a harness label) at render time.
  388 bindNewTab :: Ctx -> TabRef -> Text -> IO ()
  389 bindNewTab ctx ref verb = do
  390   res <- registryAppend (_td_tabs (_ctx_deps ctx)) ref
  391   case res of
  392     Left SlotsFull          -> emit ctx slotsFullMsg
  393     Left (AlreadyBound cur) -> do
  394       -- Ref already bound (I2): switch to the existing tab, do not duplicate.
  395       -- The error payload IS the current slot, so no re-lookup is needed.
  396       setCursorTo ctx ref
  397       emit ctx ("switched to /" <> slotChar cur)
  398     Right slot              -> do
  399       ensure ctx ref
  400       setCursorTo ctx ref
  401       notifyChanged ctx
  402       emit ctx (verb <> " /" <> slotChar slot)
  403 
  404 -- ---------------------------------------------------------------------------
  405 -- /close [N]
  406 -- ---------------------------------------------------------------------------
  407 
  408 -- | @\/close [N]@: drop a tab view (never harms ground truth). @--force@ is
  409 -- rejected with the §14 copy; @N@ (or the active tab) is removed + compacted
  410 -- and its runtime released.
  411 cmdClose :: Ctx -> [Text] -> IO ()
  412 cmdClose ctx args = case args of
  413   ("--force" : _) -> emit ctx closeForceMsg
  414   (a : _)         -> case slotArg a of
  415     Just slot -> closeSlot ctx slot
  416     Nothing   -> emit ctx closeBadArgMsg
  417   []              -> do
  418     cs <- readIORef (_td_cursors (_ctx_deps ctx))
  419     tl <- readTabs (_td_tabs (_ctx_deps ctx))
  420     case resolveCursorSlot (_ctx_conv ctx) cs tl of
  421       Just slot -> closeSlot ctx slot
  422       Nothing   -> emit ctx closeNoTargetMsg
  423 
  424 -- | Remove the tab at @slot@ (compacting), release its runtime, and clear the
  425 -- cursors that pointed at it (compaction renumbers others automatically).
  426 closeSlot :: Ctx -> TabIndex -> IO ()
  427 closeSlot ctx slot = do
  428   mTab <- registryLookupSlot (_td_tabs (_ctx_deps ctx)) slot
  429   case mTab of
  430     Nothing  -> emit ctx closeNoTargetMsg
  431     Just tab -> do
  432       let ref = _tab_ref tab
  433       cs <- readIORef (_td_cursors (_ctx_deps ctx))
  434       registryRemove (_td_tabs (_ctx_deps ctx)) slot
  435       release ctx ref
  436       -- Drop every cursor focused on the removed ref (this conversation + any
  437       -- others sharing the tab); valid cursors survive via TabRef identity.
  438       mapM_ (modifyCursor ctx . clearCursor) (conversationsOn ref cs)
  439       notifyChanged ctx
  440       emit ctx ("closed /" <> slotChar slot)
  441 
  442 -- ---------------------------------------------------------------------------
  443 -- /tabs
  444 -- ---------------------------------------------------------------------------
  445 
  446 -- | @\/tabs@: list every tab (@\/\<slot\>  \<name\>  \<kind\>  \<status\>@,
  447 -- dead tombstones included) plus a final line naming this conversation's
  448 -- relay mode.
  449 cmdTabs :: Ctx -> IO ()
  450 cmdTabs ctx = do
  451   tl     <- readTabs (_td_tabs (_ctx_deps ctx))
  452   cs     <- readIORef (_td_cursors (_ctx_deps ctx))
  453   titles <- _td_recentSessions (_ctx_deps ctx)
  454   let focused    = resolveCursorSlot (_ctx_conv ctx) cs tl
  455       titleOf sid = lookup sid titles
  456       rows       = map (tabRow focused titleOf) (toList tl)
  457       relay      = relayLine (relayModeFor ctx cs)
  458   emit ctx (T.intercalate "\n" (rows ++ [relay]))
  459 
  460 -- | Render one @\/tabs@ row. The row for @focused@ (this conversation's cursor
  461 -- slot, if any) carries a trailing @(focused)@ marker so the user can see at a
  462 -- glance where plain text will go.
  463 --
  464 -- Tabs no longer carry a label of their own — a row's display title derives
  465 -- from the bound session (resolved via @titleOf@) for a 'BoundSession', or
  466 -- falls back to the slot for a 'BoundHarness' (or a session with no recorded
  467 -- title). The fallback keeps every row non-blank.
  468 tabRow :: Maybe TabIndex -> (SessionId -> Maybe Text) -> Tab -> Text
  469 tabRow focused titleOf t =
  470   "/" <> slotChar (_tab_slot t)
  471     <> "  " <> title
  472     <> "  " <> kindLabel (_tab_ref t)
  473     <> "  " <> statusLabel (_tab_status t)
  474     <> (if focused == Just (_tab_slot t) then "  (focused)" else "")
  475   where
  476     slot  = "/" <> slotChar (_tab_slot t)
  477     title = case _tab_ref t of
  478       BoundSession sid -> Maybe.fromMaybe slot (titleOf sid)
  479       BoundHarness _   -> slot
  480 
  481 -- | The trailing relay-mode line shown by @\/tabs@. Phrased as a full sentence
  482 -- so a first-time user understands which tabs' output reaches them without
  483 -- having to learn the @relay@ jargon.
  484 relayLine :: RelayMode -> Text
  485 relayLine FocusedOnly    = "Relay mode: focused — you only see output from this tab."
  486 relayLine ActivityDigest = "Relay mode: activity — this tab in full, plus activity pings from others."
  487 relayLine Firehose       = "Relay mode: all — full output from every tab."

  491 -- ---------------------------------------------------------------------------
  492 
  493 -- | @\/rename [N] \<name\>@: relabel the tab at @N@ (or the active tab) using
  494 -- 'Parse.sanitizeTabName'.
  495 cmdRename :: Ctx -> [Text] -> IO ()
  496 cmdRename ctx args = case args of
  497   (a : rest@(_ : _))
  498     | Just slot <- slotArg a -> doRename ctx slot (T.unwords rest)
  499   _ -> do
  500     -- No leading slot arg: rename the active tab to the whole argument blob.
  501     cs <- readIORef (_td_cursors (_ctx_deps ctx))
  502     tl <- readTabs (_td_tabs (_ctx_deps ctx))
  503     case resolveCursorSlot (_ctx_conv ctx) cs tl of
  504       Just slot -> doRename ctx slot (T.unwords args)
  505       Nothing   -> emit ctx renameNoTargetMsg
  506 
  507 -- | Sanitize @name@ and set the bound entity's canonical name.
  508 --
  509 -- For a @BoundSession@ slot this sets the /session's/ description through the
  510 -- injected '_td_setSessionDescription' seam (the canonical, cross-process name)
  511 -- rather than relabelling the per-process tab field — so a rename syncs to every
  512 -- surface. For a @BoundHarness@ slot, rename is out of scope (the harness has no
  513 -- equivalent canonical name here): emit a clear note and do nothing.
  514 doRename :: Ctx -> TabIndex -> Text -> IO ()
  515 doRename ctx slot name = do
  516   mTab <- registryLookupSlot (_td_tabs (_ctx_deps ctx)) slot
  517   case mTab of
  518     Nothing  -> emit ctx renameNoTargetMsg
  519     Just tab -> case _tab_ref tab of
  520       BoundHarness _   -> emit ctx renameHarnessMsg
  521       BoundSession sid -> case Parse.sanitizeTabName name of
  522         Left e      -> emit ctx (renameBadNameMsg e)
  523         Right clean -> do
  524           res <- _td_setSessionDescription (_ctx_deps ctx) sid (Just clean)
  525           case res of
  526             Right () -> emit ctx ("renamed /" <> slotChar slot <> " " <> clean)
  527             Left err -> emit ctx err

  529 -- ---------------------------------------------------------------------------
  530 -- /relay <mode>
  531 -- ---------------------------------------------------------------------------
  532 
  533 -- | @\/relay \<mode\>@: set this conversation's relay mode
  534 -- (@focused@\/@activity@\/@all@). No arg shows the current mode (§14 copy).
  535 cmdRelay :: Ctx -> [Text] -> IO ()
  536 cmdRelay ctx args = case args of
  537   []      -> do
  538     cs <- readIORef (_td_cursors (_ctx_deps ctx))
  539     emit ctx (relayCurrentMsg (relayModeFor ctx cs))
  540   (a : _) -> case parseRelayMode a of
  541     Just m  -> do
  542       modifyCursor ctx (setRelay (_ctx_conv ctx) m)
  543       emit ctx ("relay mode: " <> relayWord m)
  544     Nothing -> emit ctx relayBadModeMsg

  546 -- ---------------------------------------------------------------------------
  547 -- /tab — open the attach wizard
  548 -- ---------------------------------------------------------------------------
  549 
  550 -- | @\/tab [query]@: snapshot the running harnesses + recent sessions
  551 -- (filtered by @query@), store a 'WizardState' for the conversation, and emit
  552 -- the rendered menu.
  553 cmdTab :: Ctx -> [Text] -> IO ()
  554 cmdTab ctx args = do
  555   harnesses <- _td_recentHarnesses (_ctx_deps ctx)
  556   sessions  <- _td_recentSessions (_ctx_deps ctx)
  557   let query = T.unwords args
  558       hs    = filterCandidates query harnesses
  559       ss    = filterCandidates query sessions
  560       st    = mkWizardSnapshot hs ss
  561   setWizard ctx (Just st)
  562   emit ctx (renderMenu st)

  564 -- ---------------------------------------------------------------------------
  565 -- /tab new <kind> — create a tab of the named kind
  566 -- ---------------------------------------------------------------------------
  567 
  568 -- | @\/tab new [\<kind\>]@: create a new tab of the requested kind. This is
  569 -- distinct from bare @\/tab@ (the attach wizard) and is dispatched /before/ it,
  570 -- so @\/tab new@ never opens the wizard.
  571 --
  572 -- The default-provider session kinds (@ai@\/@provider@\/bare) mint a fresh
  573 -- session; @harness@ (optionally with an explicit flavour, e.g.
  574 -- @\/tab new harness claude-code@) spawns a real harness via '_td_spawnHarness'
  575 -- (WU-B). The remaining shell-family kinds emit a "not yet supported"
  576 -- placeholder; an unrecognised keyword emits a short usage hint.
  577 cmdTabNew :: Ctx -> [Text] -> IO ()
  578 cmdTabNew ctx rest = case kindKeyword of
  579   -- bare / "ai" / "provider": mint a default-provider session (mirrors 'cmdNt').
  580   Nothing         -> mintDefault
  581   Just "ai"       -> mintDefault
  582   Just "provider" -> mintDefault
  583   Just "harness"  -> spawnHarness ctx flavourArgs
  584   Just "shell"    -> emit ctx (tabNewUnsupportedMsg "shell")
  585   Just "ssh"      -> emit ctx (tabNewUnsupportedMsg "ssh")
  586   Just "tmux"     -> emit ctx (tabNewUnsupportedMsg "tmux")
  587   Just _          -> emit ctx tabNewUsageMsg
  588   where
  589     kindKeyword = case rest of
  590       (k : _) -> Just (T.toLower k)
  591       []      -> Nothing
  592     -- Args following the @harness@ keyword (the optional flavour word).
  593     flavourArgs = drop 1 rest
  594     mintDefault = do
  595       mNew <- _td_newDefaultSession (_ctx_deps ctx)
  596       case mNew of
  597         Left msg     -> emit ctx msg
  598         Right newRef -> bindNewTab ctx newRef "new tab"
  599 
  600 -- | @\/tab new harness [\<flavour\>]@: build a default 'HarnessSpec' (flavour
  601 -- from the optional argument, defaulting to @claude-code@; local backend; no
  602 -- cwd\/args\/ids) and spawn it through '_td_spawnHarness'. On success bind the
  603 -- returned 'TabRef' into a new tab; on failure emit the error.
  604 spawnHarness :: Ctx -> [Text] -> IO ()
  605 spawnHarness ctx flavourArgs = do
  606   result <- _td_spawnHarness (_ctx_deps ctx) spec
  607   case result of
  608     Left msg        -> emit ctx msg
  609     Right (ref, _label) -> bindNewTab ctx ref "new harness"
  610   where
  611     flavour = case flavourArgs of
  612       (f : _) -> fixedFlavourLookup (T.toLower f)
  613       []      -> HClaudeCode
  614     spec = HarnessSpec
  615       { _h_flavour           = flavour
  616       , _h_backend           = TbLocal
  617       , _h_cwd               = Nothing
  618       , _h_args              = []
  619       , _h_harnessId         = Nothing
  620       , _h_claudeSessionUuid = Nothing
  621       , _h_canonicalCwd      = Nothing
  622       }

  624 -- ---------------------------------------------------------------------------
  625 -- Stage 3: routing grammar

  628 -- | Route via 'Parse.parseInput': switch, inject, default text, or non-tab
  629 -- slash command.
  630 routeGrammar :: Ctx -> Text -> IO ()
  631 routeGrammar ctx raw =
  632   case Parse.parseInput (_td_routingConfig (_ctx_deps ctx)) raw of
  633     Left _    -> emit ctx parseErrorMsg
  634     Right pin -> case pin of
  635       RT.Switch idx         -> doSwitch ctx idx
  636       RT.Inject idx text    -> doInject ctx idx text
  637       RT.Default text       -> doDefault ctx text
  638       RT.ParsedSlashCmd cmd -> _td_fallthrough (_ctx_deps ctx) (_ctx_conv ctx) cmd
  639 
  640 -- | @\/N@: focus the tab at slot @N@ (out-of-range → §14 copy).
  641 doSwitch :: Ctx -> TabIndex -> IO ()
  642 doSwitch ctx idx = withSlot ctx idx $ \tab -> do
  643   setCursorTo ctx (_tab_ref tab)
  644   emit ctx ("switched to /" <> slotChar idx)
  645 
  646 -- | @\/N \<text\>@: focus slot @N@ and route text to it (out-of-range → copy).
  647 doInject :: Ctx -> TabIndex -> Text -> IO ()
  648 doInject ctx idx text = withSlot ctx idx $ \tab -> do
  649   let ref = _tab_ref tab
  650   setCursorTo ctx ref
  651   deliver ctx ref text
  652 
  653 -- | Route @text@ to @ref@'s runtime, surfacing a delivery failure as the §14
  654 -- copy. Shared by 'doInject', 'doDefault', and the auto-start path so the
  655 -- send-then-handle-error shape lives in exactly one place.
  656 deliver :: Ctx -> TabRef -> Text -> IO ()
  657 deliver ctx ref text = do
  658   result <- sendTo ctx ref text
  659   case result of
  660     Right () -> pure ()
  661     Left err -> emit ctx ("couldn't deliver your message — " <> unPublicTabError (toPublicTabError err))
  662 
  663 -- | Resolve @idx@ to a present tab or emit the out-of-range §14 copy.
  664 withSlot :: Ctx -> TabIndex -> (Tab -> IO ()) -> IO ()
  665 withSlot ctx idx k = do
  666   tl   <- readTabs (_td_tabs (_ctx_deps ctx))
  667   mTab <- registryLookupSlot (_td_tabs (_ctx_deps ctx)) idx
  668   case mTab of
  669     Just tab -> k tab
  670     Nothing  -> emit ctx (outOfRangeMsg idx (length (toList tl)))
  671 
  672 -- | Plain text: route to the conversation's active tab. No active tab →
  673 -- auto-start a fresh default session and route the text to it (the implicit
  674 -- "just works" session — matches a fresh conversation's pre-Tabs behaviour); a
  675 -- 'Dead' tombstone → deferred death warning + clear + drop (§8).
  676 doDefault :: Ctx -> Text -> IO ()
  677 doDefault ctx text = do
  678   cs <- readIORef (_td_cursors (_ctx_deps ctx))
  679   tl <- readTabs (_td_tabs (_ctx_deps ctx))
  680   case resolveCursorSlot (_ctx_conv ctx) cs tl of
  681     Nothing   -> autoStartDefault ctx text
  682     Just slot -> do
  683       -- 'resolveCursorSlot' returns the slot by resolving the cursor's ref
  684       -- against the list, so the tab is guaranteed present here; the impossible
  685       -- absence is folded into 'error' (no dead defensive arm for HPC).
  686       tab <- Maybe.fromMaybe (impossible "doDefault: resolved slot vanished")
  687                <$> registryLookupSlot (_td_tabs (_ctx_deps ctx)) slot
  688       if _tab_status tab == Dead
  689         then do
  690           -- §8 deferred death warning: warn, clear the cursor, DROP the text
  691           -- (never route it into the dead harness). Tabs carry no label of
  692           -- their own anymore, so the warning names the tab by its slot.
  693           emit ctx (deferredDeathMsg ("/" <> slotChar (_tab_slot tab)))
  694           modifyCursor ctx (clearCursor (_ctx_conv ctx))
  695         else deliver ctx (_tab_ref tab) text
  696 
  697 -- | No active tab + plain text: mint a fresh default session, append a tab for
  698 -- it, focus the conversation on it, and route the text — restoring the implicit
  699 -- "just works" default session. No @new tab \/N@ confirmation is emitted (the
  700 -- user sent a chat message, not @\/nt@); the new tab still becomes visible via
  701 -- 'notifyChanged'. @'Left' msg@ (no provider\/model configured) surfaces the
  702 -- existing setup guidance, exactly like 'cmdNt'.
  703 autoStartDefault :: Ctx -> Text -> IO ()
  704 autoStartDefault ctx text = do
  705   mNew <- _td_newDefaultSession (_ctx_deps ctx)
  706   case mNew of
  707     Left msg     -> emit ctx msg
  708     Right ref    -> do
  709       res <- registryAppend (_td_tabs (_ctx_deps ctx)) ref
  710       case res of
  711         -- Slot exhaustion: surface the §14 copy with no state change.
  712         Left SlotsFull -> emit ctx slotsFullMsg
  713         -- 'Right' (the freshly-bound slot) and 'Left (AlreadyBound _)' (which is
  714         -- structurally impossible for a freshly-minted ref) both route the text
  715         -- to the ref — focus + deliver — rather than crash on the latter.
  716         _              -> routeToNew ctx ref text
  717 
  718 -- | Focus the conversation on a freshly-bound @ref@, ensure its runtime, notify
  719 -- listeners, and route @text@ to it. Shared by the two reachable arms of
  720 -- 'autoStartDefault'.
  721 routeToNew :: Ctx -> TabRef -> Text -> IO ()
  722 routeToNew ctx ref text = do
  723   ensure ctx ref
  724   setCursorTo ctx ref
  725   notifyChanged ctx
  726   deliver ctx ref text

  748 -- | Apply a pure 'CursorState' update to the shared cursor ref.

  751   atomicModifyIORef' (_td_cursors (_ctx_deps ctx)) (\cs -> (f cs, ()))

  769   atomicModifyIORef' (_td_wizard (_ctx_deps ctx)) $ \m ->

  797 -- | Parse a single @[0-9a-z]@ slot argument into a 'TabIndex'.
  798 slotArg :: Text -> Maybe TabIndex
  799 slotArg t = case T.unpack t of
  800   [c] -> tabIndexChar c >>= mkTabIndex
  801   _   -> Nothing
  802 
  803 -- | The @[0-9a-z]@ → index mapping (digits @0..9@, letters @10..35@).
  804 tabIndexChar :: Char -> Maybe Int
  805 tabIndexChar c
  806   | Char.isDigit c      = Just (fromEnum c - fromEnum '0')
  807   | Char.isAsciiLower c = Just (10 + fromEnum c - fromEnum 'a')
  808   | otherwise           = Nothing
  809 
  810 -- | Render a 'TabIndex' back to its single @[0-9a-z]@ display char.
  811 slotChar :: TabIndex -> Text
  812 slotChar ti =
  813   let n = unTabIndex ti
  814   in if n < 10
  815        then T.singleton (toEnum (fromEnum '0' + n))
  816        else T.singleton (toEnum (fromEnum 'a' + n - 10))
  817 
  818 -- | Parse a relay-mode word.
  819 parseRelayMode :: Text -> Maybe RelayMode
  820 parseRelayMode t = case T.toLower t of
  821   "focused"  -> Just FocusedOnly
  822   "activity" -> Just ActivityDigest
  823   "all"      -> Just Firehose
  824   _          -> Nothing

  828 relayWord FocusedOnly    = "focused"

  846 -- | @\/N@ out of range (§14). @n@ = current tab count.
  847 outOfRangeMsg :: TabIndex -> Int -> Text
  848 outOfRangeMsg idx n =
  849   "/" <> slotChar idx <> ": out of range — you have " <> tshow n
  850     <> " tabs (/0–/" <> lastSlotChar n <> "); /tabs to list"
  851 
  852 -- | The high slot char of an @n@-tab list (@n-1@). For @n <= 0@ (an
  853 -- out-of-range switch against an empty list) the high coordinate is @0@. For
  854 -- @n >= 1@, @n-1 >= 0@ so 'mkTabIndex' always succeeds; the impossible
  855 -- 'Nothing' is folded into 'error'.
  856 lastSlotChar :: Int -> Text
  857 lastSlotChar n
  858   | n <= 0    = "0"
  859   | otherwise =
  860       slotChar (Maybe.fromMaybe (impossible "lastSlotChar: negative index")
  861                                 (mkTabIndex (n - 1)))
