    1 {-# LANGUAGE PatternSynonyms #-}
    2 -- |
    3 -- Module      : PureClaw.Handles.Tab
    4 -- Description : Tabbed Chat type layer (WU1) — user-facing primitive: a tab.
    5 --
    6 -- A 'TabHandle' is the universal Handle-pattern carrier for one tab in
    7 -- the Tabbed Chat surface. Each tab carries a 'TabKind' discriminator
    8 -- that classifies it as either a /session-backed/ tab
    9 -- (@'TkSession' sk@, where @sk@ is a 'PureClaw.Session.Kind.SessionKind')
   10 -- or a /raw terminal-backend/ tab
   11 -- (@'TkRawShell' tb@, where @tb@ is a 'PureClaw.Session.Kind.TerminalBackend').
   12 -- The dispatcher routes user input to the focused tab and per-tab
   13 -- output to the channel (focus-gated).
   14 --
   15 -- == Spec-vs-runtime layering (T6)
   16 --
   17 -- 'PureClaw.Session.Kind.SessionKind' and its component types
   18 -- ('ProviderSpec', 'HarnessSpec', 'TerminalBackend', etc.) are
   19 -- /serialisation-safe specs/ — they carry configuration data that can
   20 -- be round-tripped through JSON or TOML without IO.
   21 --
   22 -- Runtime-validated types ('SafePath', 'SshTarget', 'AuthorizedCommand',
   23 -- etc.) live in the factory and transport layers
   24 -- ('PureClaw.Tab.Ai', 'PureClaw.Tab.Backend', 'PureClaw.Tab.Harness')
   25 -- and are never stored in 'TabKind'. This module sits at the boundary:
   26 -- 'TabKind' references the spec layer but never exposes the runtime
   27 -- layer.
   28 --
   29 -- == Scope of WU1
   30 --
   31 -- This module currently lands the type layer only: 'TabHandle' record
   32 -- shape, smart constructors, error ADTs (with the manually-written
   33 -- redacted 'Show' instance for 'TabError' required by H14), and the
   34 -- per-kind factory signatures. Factory bodies are stubbed with
   35 -- @'error' \"not implemented\"@ — the real bodies land in WU6 ('Tab.Ai'),
   36 -- WU7 ('Tab.Harness'), and WU8 ('Tab.Backend').
   37 --
   38 -- == Field-naming convention
   39 --
   40 -- The handle record uses the @_tabHandle_*@ field-name prefix (NOT the
   41 -- terser @_th_*@ used in some older draft notes). This was decided
   42 -- during WU0 planning to keep the prefix self-describing when the field
   43 -- name is grepped in isolation.
   44 --
   45 -- See @docs\/tabbed-chat.md@ for the full design (sections
   46 -- \"TabHandle abstraction (H-series)\" and \"Channel output\").

   51   , unTabIndex

   98 -- TabIndex
   99 -- ---------------------------------------------------------------------------
  100 
  101 -- | Validated tab index. The data constructor is intentionally NOT
  102 -- exported; obtain values via 'mkTabIndex' (which bounds-checks against
  103 -- the configured @_rc_maxTabs@ ceiling) or by pattern-matching on the
  104 -- newtype via 'unTabIndex'.
  105 --
  106 -- The underlying 'Int' is non-negative on every value produced by
  107 -- 'mkTabIndex' (@0 <= n@). Bounds-checking against @_rc_maxTabs@ is
  108 -- performed at the parser layer (WU2) using the 'RoutingConfig' in
  109 -- scope; the smart constructor here enforces the floor only.
  110 newtype TabIndex = TabIndex { unTabIndex :: Int }
  111   deriving stock (Eq, Ord, Show)
  112 
  113 -- | The only way to construct a 'TabIndex'. Returns 'Nothing' for
  114 -- negative inputs.
  115 mkTabIndex :: Int -> Maybe TabIndex
  116 mkTabIndex n
  117   | n < 0     = Nothing
  118   | otherwise = Just (TabIndex n)

  122 -- TabKind
  123 -- ---------------------------------------------------------------------------
  124 
  125 -- | Kind of tab — two-level discriminator (WU-5 refactor, T4).
  126 --
  127 -- * @'TkSession' sk@ — session-backed tab. The 'SessionKind' payload
  128 --   ('SkProvider' or 'SkHarness') is a serialisation-safe /spec/
  129 --   (declared in "PureClaw.Session.Kind"); runtime-validated types
  130 --   ('SafePath', 'SshTarget', etc.) live in factory\/transport layers.
  131 --   This module sits at the boundary: 'TabKind' references the spec
  132 --   layer but never exposes the runtime layer (T6).
  133 --
  134 -- * @'TkRawShell' tb@ — raw terminal-backend tab (local shell, SSH,
  135 --   or tmux). The 'TerminalBackend' payload carries connection
  136 --   coordinates; the backend factory validates and connects.
  137 data TabKind
  138   = TkSession  !SessionKind
  139   | TkRawShell !TerminalBackend
  140   deriving stock (Eq, Show)
  141 
  142 -- | @KindAi@ matches any provider-backed session tab. When used as
  143 -- an expression it constructs a 'TkSession' with a placeholder
  144 -- 'ProviderSpec' — production code should prefer the explicit
  145 -- constructor with a real spec.
  146 pattern KindAi :: TabKind
  147 pattern KindAi <- TkSession (SkProvider {})
  148   where KindAi = TkSession (SkProvider defaultProviderSpec)
  149 
  150 -- | @KindHarness@ matches any harness-backed session tab. When used
  151 -- as an expression it constructs a 'TkSession' with a placeholder
  152 -- 'HarnessSpec'.
  153 pattern KindHarness :: TabKind
  154 pattern KindHarness <- TkSession (SkHarness {})
  155   where KindHarness = TkSession (SkHarness defaultHarnessSpec)
  156 
  157 -- | @KindShell@ matches a local-shell backend tab.
  158 pattern KindShell :: TabKind

  160

  227 -- ---------------------------------------------------------------------------
  228 -- TabStatus
  229 -- ---------------------------------------------------------------------------
  230 
  231 -- | Runtime status of a tab.
  232 --
  233 -- * 'Active' — a provider call or tool execution is in flight, OR a
  234 --   backend recv is waiting for output. The dashboard does not
  235 --   distinguish these sub-cases.
  236 -- * 'Idle' — the tab is waiting for input; carries the timestamp of
  237 --   the last observed input (for sort / display).
  238 -- * 'Crashed' — the tab loop hit a synchronous exception and exited;
  239 --   the wrapped 'PublicTabError' is the channel-safe summary. The
  240 --   registry entry persists until @\/tab close N@ or
  241 --   @\/tab resume \<session\>@.
  242 data TabStatus
  243   = Active
  244   | Idle !UTCTime
  245   | Crashed !PublicTabError
  246   deriving stock (Eq, Show)

  249 -- ---------------------------------------------------------------------------
  250 -- CloseMode
  251 -- ---------------------------------------------------------------------------
  252 
  253 -- | Close semantics requested by the caller of '_tabHandle_close'.
  254 --
  255 -- * 'CloseGraceful' — kind-specific graceful close: for AI sessions
  256 --   ('TkSession (SkProvider _)') archives the session via @_sh_save@;
  257 --   for non-AI kinds runs the underlying destructive close (e.g.
  258 --   @_bh_close@, @_hh_stop@).
  259 -- * 'CloseForce' — for AI sessions skips the archive (transcript
  260 --   deleted from disk); for non-AI kinds is a no-op distinct from
  261 --   'CloseGraceful' because close is already destructive there.
  262 data CloseMode
  263   = CloseGraceful
  264   | CloseForce
  265   deriving stock (Eq, Show)

  268 -- ---------------------------------------------------------------------------
  269 -- TabRunner
  270 -- ---------------------------------------------------------------------------
  271 
  272 -- | The cancel + wait pair returned by the @_env_fork@ test seam.
  273 --
  274 -- Cannot be an 'Control.Concurrent.Async.Async' directly because:
  275 -- (a) the synchronous test variant runs the body inline and has no
  276 -- live async to wrap; (b) the IORef-Bool cancel-observability primitive
  277 -- used by tests needs a uniform shape across sync and async variants.
  278 --
  279 -- See the @TabRunner bootstrapping during spawn@ section of the design
  280 -- doc and the WU0 scaffold in @Test.Fake.TabFactory@.
  281 data TabRunner = TabRunner
  282   { _trun_cancel :: IO ()
  283     -- ^ Idempotent cancel. Safe to call multiple times; safe to call
  284     --   on a runner whose body has already exited.
  285   , _trun_wait   :: IO ()
  286     -- ^ Block until the tab loop's body exits. In the synchronous
  287     --   test variant returns immediately because the body has
  288     --   already run inline.
  289   }

  292 -- ---------------------------------------------------------------------------
  293 -- TabHandle
  294 -- ---------------------------------------------------------------------------
  295 
  296 -- | Universal Handle-pattern carrier for one tab.
  297 --
  298 -- Pure fields ('_tabHandle_index', '_tabHandle_name', '_tabHandle_kind')
  299 -- are bound at construction time. IO-action fields are kind-specific:
  300 -- 'TkSession (SkProvider _)' enqueues into a per-tab @TBQueue InputEvent@; non-AI kinds
  301 -- write to a 'PureClaw.Handles.Backend.BackendHandle' or
  302 -- 'PureClaw.Handles.Harness.HarnessHandle'.
  303 --
  304 -- '_tabHandle_close' is idempotent and never throws (parallel to the
  305 -- @_bh_close@ contract on 'PureClaw.Handles.Backend.BackendHandle').
  306 data TabHandle = TabHandle
  307   { _tabHandle_index         :: !TabIndex
  308     -- ^ Pure: tab's slot in the registry (H12-adjacent).
  309   , _tabHandle_name          :: !TabName
  310     -- ^ Pure: redacted friendly label (H11). Constructed via
  311     --   @sanitizeTabName@ (WU2) at the factory site.
  312   , _tabHandle_kind          :: !TabKind
  313     -- ^ Pure: H12. Read without IO.
  314   , _tabHandle_status        :: IO TabStatus
  315     -- ^ H5: returns 'Active' \/ 'Idle' \/ 'Crashed'.
  316   , _tabHandle_send          :: Text -> IO (Either TabError ())
  317     -- ^ H4: enqueue a 'UserText' input event. Bounded by
  318     --   @_rc_inputQueueBound@; overflow returns
  319     --   'Left' (e.g. 'TabConcurrencyLimit') so the dispatcher never
  320     --   blocks.
  321   , _tabHandle_close         :: CloseMode -> IO ()
  322     -- ^ H6 \/ H7 \/ H8 \/ H9 \/ H10: idempotent, never throws,
  323     --   kind-specific semantics. The 'CloseMode' selects graceful
  324     --   vs forced archive behaviour for AI sessions.
  325   }

  367 -- ---------------------------------------------------------------------------
  368 -- TabError
  369 -- ---------------------------------------------------------------------------
  370 
  371 -- | Construction-time and runtime errors on a tab.
  372 --
  373 -- Every constructor's payload is itself a redaction-safe ADT or a
  374 -- bounded primitive — no constructor carries a free 'Data.Text.Text',
  375 -- a raw 'Control.Exception.SomeException' message, or a raw filesystem
  376 -- path. The hand-written 'Show' below (H14) elides every payload value
  377 -- so accidental @show@ of a 'TabError' in a log line cannot leak
  378 -- hostnames, paths, or stderr fragments.
  379 --
  380 -- The exhaustive constructor list is enforced by H3.
  381 data TabError
  382   = TabIndexInUse !TabIndex
  383   | TabIndexOutOfRange !Int
  384   | TabLimitExceeded !Int
  385   | TabBackendConstructFailed !Backend.BackendError
  386   | TabSessionCreateFailed !SessionError
  387   | TabSpawnAuthDenied !PublicAuthError
  388   | TabNotFound !Int
  389   | TabConcurrencyLimit !Int
  390   | TabInvalidName !NameError
  391   deriving stock (Eq)
  392 
  393 -- | Hand-written 'Show' for 'TabError' per H14: constructor names only,
  394 -- payload values elided. Adding a new constructor here requires adding
  395 -- a new branch in this 'Show' instance (H3 enumerates the exhaustive
  396 -- set), so accidentally regressing to @deriving Show@ would surface as
  397 -- a build error from the @-Wincomplete-patterns@ flag.
  398 instance Show TabError where
  399   show e = case e of
  400     TabIndexInUse{}             -> "TabIndexInUse"
  401     TabIndexOutOfRange{}        -> "TabIndexOutOfRange"
  402     TabLimitExceeded{}          -> "TabLimitExceeded"
  403     TabBackendConstructFailed{} -> "TabBackendConstructFailed"
  404     TabSessionCreateFailed{}    -> "TabSessionCreateFailed"
  405     TabSpawnAuthDenied{}        -> "TabSpawnAuthDenied"
  406     TabNotFound{}               -> "TabNotFound"
  407     TabConcurrencyLimit{}       -> "TabConcurrencyLimit"
  408     TabInvalidName{}            -> "TabInvalidName"

  411 -- ---------------------------------------------------------------------------
  412 -- PublicTabError
  413 -- ---------------------------------------------------------------------------
  414 
  415 -- | Channel-safe projection of a 'TabError'.
  416 --
  417 -- The wrapped 'Data.Text.Text' is short, fixed-vocabulary, and
  418 -- contains no hostnames, paths, or secrets. This is what reaches the
  419 -- user via 'PureClaw.Handles.Channel.ChannelHandle' when a tab
  420 -- operation fails.
  421 newtype PublicTabError = PublicTabError { unPublicTabError :: Text }
  422   deriving stock (Eq, Show)
  423 
  424 -- | Project a 'TabError' to its user-visible form.
  425 --
  426 -- The mapping is intentionally coarse: similar failure modes collapse
  427 -- to the same short label so a future audit of channel-bound strings
  428 -- has a small surface to grep.
  429 toPublicTabError :: TabError -> PublicTabError
  430 toPublicTabError e = PublicTabError $ case e of
  431   TabIndexInUse{}             -> "tab: index already in use"
  432   TabIndexOutOfRange{}        -> "tab: index out of range"
  433   TabLimitExceeded{}          -> "tab: maximum tab count reached"
  434   TabBackendConstructFailed{} -> "tab: backend construction failed"
  435   TabSessionCreateFailed{}    -> "tab: session creation failed"
  436   TabSpawnAuthDenied{}        -> "tab: spawn authorization denied"
  437   TabNotFound{}               -> "tab: not found"
  438   TabConcurrencyLimit{}       -> "tab: input queue full"
  439   TabInvalidName{}            -> "tab: invalid name"
