   20 -- | Security policy governing what an agent can do.
   21 -- Policy evaluation is pure — no IO, fully testable with QuickCheck.
   22 --
   23 -- Local subprocess authorization consults '_sp_allowedCommands'.
   24 -- Remote subprocess authorization (e.g. ssh-dispatched commands)
   25 -- consults '_sp_allowedRemoteCommands'; the two allowlists are
   26 -- independent so granting "git" locally does not silently grant
   27 -- "git" remotely.
   28 --
   29 -- __Construction-site enforcement:__ adding a new field to
   30 -- 'SecurityPolicy' is intentionally a breaking change. With
   31 -- @-Werror -Wall@ (which implies @-Wmissing-fields@) any brace-form
   32 -- construction site that forgets the new field fails to compile.
   33 -- Positional construction sites (e.g. @SecurityPolicy AllowAll Full _@)
   34 -- fail to compile because the arity changes. This is the type-level
   35 -- migration safety net referenced in 'docs\/SECURITY_PRACTICES.md' §5.2
   36 -- and 'docs\/terminal-backend-abstractions.md' § \"SecurityPolicy
   37 -- field migration\".
   38 data SecurityPolicy = SecurityPolicy
   39   { _sp_allowedCommands       :: AllowList CommandName
   40   , _sp_autonomy              :: AutonomyLevel
   41   , _sp_allowedRemoteCommands :: AllowList CommandName
   42   }
   43   deriving stock (Show, Eq)
   44 
   45 -- | Default policy: deny everything (local and remote). Start here and
   46 -- open up explicitly.
   47 defaultPolicy :: SecurityPolicy
   48 defaultPolicy = SecurityPolicy
   49   { _sp_allowedCommands       = AllowList Set.empty
   50   , _sp_autonomy              = Deny
   51   , _sp_allowedRemoteCommands = AllowList Set.empty
   52   }
   53 
   54 -- | Add a command to the local allowed set.
   55 -- If the policy already uses 'AllowAll', this is a no-op.
   56 allowCommand :: CommandName -> SecurityPolicy -> SecurityPolicy
   57 allowCommand cmd policy =
   58   case _sp_allowedCommands policy of
   59     AllowAll    -> policy
   60     AllowList s -> policy { _sp_allowedCommands = AllowList (Set.insert cmd s) }
   61 
   62 -- | Remove a command from the local allowed set.
   63 -- If the policy uses 'AllowAll', this has no effect (you cannot deny
   64 -- individual commands from an AllowAll policy — switch to explicit list first).
   65 denyCommand :: CommandName -> SecurityPolicy -> SecurityPolicy
   66 denyCommand cmd policy =
   67   case _sp_allowedCommands policy of
   68     AllowAll    -> policy
   69     AllowList s -> policy { _sp_allowedCommands = AllowList (Set.delete cmd s) }
   70 
   71 -- | Add a command to the remote allowed set.
   72 -- If the policy already uses 'AllowAll' for remote commands, this is a no-op.
   73 allowRemoteCommand :: CommandName -> SecurityPolicy -> SecurityPolicy
   74 allowRemoteCommand cmd policy =
   75   case _sp_allowedRemoteCommands policy of
   76     AllowAll    -> policy
   77     AllowList s -> policy { _sp_allowedRemoteCommands = AllowList (Set.insert cmd s) }
   78 
   79 -- | Remove a command from the remote allowed set.
   80 -- If the policy uses 'AllowAll' for remote commands, this has no effect
   81 -- (switch to an explicit list first).
   82 denyRemoteCommand :: CommandName -> SecurityPolicy -> SecurityPolicy
   83 denyRemoteCommand cmd policy =
   84   case _sp_allowedRemoteCommands policy of
   85     AllowAll    -> policy
   86     AllowList s -> policy { _sp_allowedRemoteCommands = AllowList (Set.delete cmd s) }
   87 
   88 -- | Set the autonomy level on a policy.
   89 withAutonomy :: AutonomyLevel -> SecurityPolicy -> SecurityPolicy
   90 withAutonomy level policy = policy { _sp_autonomy = level }
   91 
   92 -- | Check whether a command is allowed for local execution by this policy. Pure.
   93 isCommandAllowed :: SecurityPolicy -> CommandName -> Bool
   94 isCommandAllowed policy = isAllowed (_sp_allowedCommands policy)
   95 
   96 -- | Check whether a command is allowed for remote execution by this policy. Pure.
   97 isRemoteCommandAllowed :: SecurityPolicy -> CommandName -> Bool
   98 isRemoteCommandAllowed policy = isAllowed (_sp_allowedRemoteCommands policy)
