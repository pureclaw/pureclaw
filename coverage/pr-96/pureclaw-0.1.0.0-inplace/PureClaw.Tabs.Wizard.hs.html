    1 -- |
    2 -- Module      : PureClaw.Tabs.Wizard
    3 -- Description : The @\/tab@ attach-wizard state machine (Tabs-as-View, WU6).
    4 --
    5 -- A small, pure-ish state machine for the @\/tab@ attach wizard (design §11).
    6 -- When @\/tab@ runs, the dispatcher snapshots the currently-running harnesses
    7 -- and the most-recent sessions ('mkWizardSnapshot'), numbers them with
    8 -- single-character keys drawn from 'wizardKeys' (@[1-9a-z]@ — @0@ is reserved
    9 -- for /cancel/), renders the menu ('renderMenu'), and shows it. The
   10 -- conversation then sits in a transient 'WizardState' until the user replies.
   11 --
   12 -- Each reply is fed to 'stepWizard', which decides the next state and the
   13 -- visible 'WizardStep':
   14 --
   15 --   * A bare key present in the snapshot binds the __exact__ id captured at
   16 --     snapshot time — never re-resolved by list position (§11). For a harness
   17 --     target the injected liveness probe ('_wz_live') is consulted; if the
   18 --     harness has vanished since the snapshot, the wizard re-prompts with a
   19 --     refreshed list rather than attaching to nothing. Sessions live on disk,
   20 --     so a session pick is always valid (no probe).
   21 --   * @0@\/@cancel@ exits; a @\/@-prefixed reply cancels the wizard and asks
   22 --     the caller to run that command instead (no modal lock-in, §9.2); any
   23 --     other reply re-prompts with the state unchanged.
   24 --
   25 -- The /only/ IO is the injected 'WizardEnv' liveness probe, so the engine is
   26 -- driven deterministically in unit tests. Dispatcher interception (consuming a
   27 -- conversation's next message before @parseInput@) is __not__ here — that is
   28 -- WU8. This module is purely additive.
   29 --
   30 -- Leaf module: depends on 'PureClaw.Core.Types' ('SessionId') and
   31 -- 'PureClaw.Harness.Registry' ('HarnessId') only.

   54 -- ---------------------------------------------------------------------------
   55 -- Types
   56 -- ---------------------------------------------------------------------------
   57 
   58 -- | What a completed wizard binds to: an existing running harness, or a past
   59 -- session to reopen (continue\/append). The dispatcher (WU8) turns this into a
   60 -- new tab at the next free slot.
   61 data WizardTarget
   62   = AttachHarness !HarnessId
   63   | ReopenSession !SessionId
   64   deriving stock (Eq, Show)
   65 
   66 -- | A wizard snapshot: the numbered options as shown to the user. The numbering
   67 -- is stable for the lifetime of this state — a reply binds the id captured
   68 -- here, never re-resolved by position (§11).
   69 newtype WizardState = WizardState
   70   { _wz_options :: [(Char, WizardTarget)]
   71   }
   72   deriving stock (Eq, Show)
   73 
   74 -- | The visible outcome of one wizard turn.
   75 data WizardStep
   76   = Prompt !Text
   77     -- ^ The rendered menu to show when the wizard first opens.
   78   | Done !WizardTarget
   79     -- ^ The user picked a still-valid target; bind it.
   80   | Cancelled
   81     -- ^ The user cancelled (@0@ or @cancel@).
   82   | Reprompt !Text
   83     -- ^ Invalid reply, or the picked target vanished: re-render the menu with a
   84     --   one-line notice.
   85   | RunCommand !Text
   86     -- ^ The reply began with @\/@: cancel the wizard and run that command.
   87   deriving stock (Eq, Show)
   88 
   89 -- | The wizard's only effect seam: probe whether a harness is still running.
   90 -- Injected so the engine is unit-testable without a live tmux server.
   91 newtype WizardEnv = WizardEnv
   92   { _wz_live :: HarnessId -> IO Bool
   93   }
   94 
   95 -- ---------------------------------------------------------------------------
   96 -- Key namespace
   97 -- ---------------------------------------------------------------------------
   98 
   99 -- | The single-character option keys, in assignment order: @1..9@ then @a..z@.
  100 -- @0@ is deliberately excluded — it is reserved for /cancel/ (§11), so a bare
  101 -- @0@ is never ambiguous. This yields 35 assignable options; the snapshot is
  102 -- capped to this length.
  103 wizardKeys :: [Char]
  104 wizardKeys = ['1' .. '9'] ++ ['a' .. 'z']
  105 
  106 -- ---------------------------------------------------------------------------
  107 -- Building & rendering
  108 -- ---------------------------------------------------------------------------
  109 
  110 -- | Build a wizard snapshot from the running harnesses and recent sessions,
  111 -- each paired with its display label. Options are numbered with 'wizardKeys' —
  112 -- harnesses first, then sessions — and the total is capped at @length
  113 -- 'wizardKeys'@ (harnesses are kept first, sessions truncated). Overflow beyond
  114 -- the cap is reachable via 'filterCandidates' before snapshotting (§11).
  115 mkWizardSnapshot :: [(HarnessId, Text)] -> [(SessionId, Text)] -> WizardState
  116 mkWizardSnapshot harnesses sessions =
  117   WizardState (zip wizardKeys targets)
  118   where
  119     targets =
  120       map (AttachHarness . fst) harnesses
  121         ++ map (ReopenSession . fst) sessions
  122     -- 'zip' against the finite 'wizardKeys' truncates to the cap automatically,
  123     -- keeping harnesses (which come first) ahead of sessions.
  124 
  125 -- | Render the numbered menu plus a @0  cancel@ line (the user-facing prompt).
  126 -- 'mkWizardSnapshot' keeps only the binding-relevant '(Char, WizardTarget)' and
  127 -- discards friendly labels, so this engine-level renderer shows the target id
  128 -- text. The dispatcher (WU8) may render a richer label menu from the raw
  129 -- candidate lists; this label-free renderer exists so the wizard is
  130 -- self-describing in tests.
  131 renderMenu :: WizardState -> Text
  132 renderMenu (WizardState opts) =
  133   T.unlines (header : map line opts ++ [cancelLine])
  134   where
  135     header     = "Attach a tab — reply with a number:"
  136     cancelLine = " 0  cancel"
  137     line (k, t) = " " <> T.singleton k <> "  " <> targetLabel t
  138 
  139 -- | A terse label for a target, used by the engine-level 'renderMenu'.
  140 targetLabel :: WizardTarget -> Text
  141 targetLabel (AttachHarness h) = "harness " <> tshow h
  142 targetLabel (ReopenSession s) = "session " <> tshow s
  143 
  144 -- ---------------------------------------------------------------------------
  145 -- Driving the wizard
  146 -- ---------------------------------------------------------------------------
  147 
  148 -- | Process one wizard reply against the current snapshot. Returns the next
  149 -- state (@Nothing@ ⇒ the wizard ends) and the visible 'WizardStep'.
  150 --
  151 -- Resolution order (the reply is trimmed first):
  152 --
  153 --   1. A @\/@-prefixed reply ⇒ @(Nothing, 'RunCommand' reply)@ — cancel + run it.
  154 --   2. @0@ or @cancel@ (case-insensitive) ⇒ @(Nothing, 'Cancelled')@.
  155 --   3. A single key present in '_wz_options':
  156 --
  157 --        * 'AttachHarness' whose '_wz_live' is False ⇒ the target vanished:
  158 --          @('Just' st', 'Reprompt' notice)@ where @st'@ drops that harness.
  159 --        * otherwise ⇒ @(Nothing, 'Done' target)@.
  160 --
  161 --   4. Anything else (unknown key, empty, multi-char non-command) ⇒
  162 --      @('Just' st, 'Reprompt' …)@ — invalid; re-prompt, state unchanged.
  163 stepWizard :: WizardEnv -> WizardState -> Text -> IO (Maybe WizardState, WizardStep)
  164 stepWizard env st@(WizardState opts) raw
  165   | "/" `T.isPrefixOf` reply = pure (Nothing, RunCommand reply)
  166   | isCancel reply           = pure (Nothing, Cancelled)
  167   | Just target <- singleKeyTarget = resolvePick env st target
  168   | otherwise                = pure (Just st, Reprompt invalidNotice)
  169   where
  170     reply = T.strip raw
  171 
  172     isCancel r = r == "0" || T.toLower r == "cancel"
  173 
  174     -- A single-character reply that names an option.
  175     singleKeyTarget = case T.unpack reply of
  176       [c] -> lookup c opts
  177       _   -> Nothing
  178 
  179 -- | Resolve a picked target: harnesses are liveness-checked (a dead one
  180 -- re-prompts with a refreshed, harness-dropped snapshot); sessions are always
  181 -- valid (they live on disk — no probe).
  182 resolvePick :: WizardEnv -> WizardState -> WizardTarget -> IO (Maybe WizardState, WizardStep)
  183 resolvePick env st target = case target of
  184   ReopenSession _ -> pure (Nothing, Done target)
  185   AttachHarness h -> do
  186     alive <- _wz_live env h
  187     if alive
  188       then pure (Nothing, Done target)
  189       else pure (Just (dropTarget target st), Reprompt vanishedNotice)
  190 
  191 -- | Drop a vanished target from a snapshot and re-key the survivors with
  192 -- 'wizardKeys' (so the refreshed menu stays contiguous). Used when a picked
  193 -- harness has died.
  194 dropTarget :: WizardTarget -> WizardState -> WizardState
  195 dropTarget gone (WizardState opts) =
  196   WizardState (zip wizardKeys [ t | (_, t) <- opts, t /= gone ])
  197 
  198 -- ---------------------------------------------------------------------------
  199 -- Overflow query filter
  200 -- ---------------------------------------------------------------------------
  201 
  202 -- | A sanitised, case-insensitive substring filter on candidate labels, used by
  203 -- the @\/tab \<query\>@ overflow path (§11) before 'mkWizardSnapshot'. The query
  204 -- is trimmed and lowercased; an empty\/whitespace-only query keeps everything.
  205 filterCandidates :: Text -> [(a, Text)] -> [(a, Text)]
  206 filterCandidates query cands
  207   | T.null q  = cands
  208   | otherwise = [ c | c@(_, label) <- cands, q `T.isInfixOf` T.toLower label ]
  209   where
  210     q = T.toLower (T.strip query)

  216 -- | The vanished-target notice (§14).
  217 vanishedNotice :: Text
  218 vanishedNotice = "that target is gone — list refreshed"
  219 
  220 -- | The invalid-reply notice.
  221 invalidNotice :: Text
  222 invalidNotice = "not a valid choice — reply with a number, or 0 to cancel"
