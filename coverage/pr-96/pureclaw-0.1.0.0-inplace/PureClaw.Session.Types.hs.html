   43 -- | Validated session prefix. Used as the human-readable leading segment
   44 -- of a 'PureClaw.Core.Types.SessionId'. Same character rules as
   45 -- 'PureClaw.Agent.AgentDef.AgentName' plus a reserved-word denylist.
   46 newtype SessionPrefix = SessionPrefix { unSessionPrefix :: Text }
   47   deriving stock (Show, Eq, Ord)
   48 
   49 -- | Reasons a raw 'Text' cannot be promoted to a 'SessionPrefix'.
   50 data SessionPrefixError
   51   = PrefixEmpty
   52   | PrefixTooLong
   53   | PrefixInvalidChars Text
   54   | PrefixLeadingDot
   55   | PrefixReserved Text
   56   deriving stock (Show, Eq)
   57 
   58 -- | Maximum allowed length for a session prefix.
   59 sessionPrefixMaxLength :: Int
   60 sessionPrefixMaxLength = 64
   61 
   62 -- | Valid character predicate: ASCII letters, digits, underscore, hyphen.
   63 -- Mirrors 'PureClaw.Agent.AgentDef.isValidAgentNameChar'.
   64 isValidPrefixChar :: Char -> Bool
   65 isValidPrefixChar c =
   66   Char.isAsciiUpper c
   67     || Char.isAsciiLower c
   68     || Char.isDigit c
   69     || c == '_'
   70     || c == '-'
   71 
   72 -- | Reserved tokens that look like prefixes but collide with CLI verbs.
   73 -- Currently just @"new"@, which is the literal argument to
   74 -- @\/session new@ and would create ambiguous resume targets if allowed
   75 -- as a prefix.
   76 reservedPrefixes :: [Text]
   77 reservedPrefixes = ["new"]
   78 
   79 -- | Smart constructor. Same validation as 'mkAgentName' (non-empty,
   80 -- max 64 chars, no leading dot, only @[a-zA-Z0-9_-]@), plus rejection
   81 -- of reserved tokens like @"new"@.
   82 mkSessionPrefix :: Text -> Either SessionPrefixError SessionPrefix
   83 mkSessionPrefix raw
   84   | T.null raw = Left PrefixEmpty
   85   | T.length raw > sessionPrefixMaxLength = Left PrefixTooLong
   86   | T.head raw == '.' = Left PrefixLeadingDot
   87   | not (T.all isValidPrefixChar raw) = Left (PrefixInvalidChars raw)
   88   | raw `elem` reservedPrefixes = Left (PrefixReserved raw)
   89   | otherwise = Right (SessionPrefix raw)
   90 
   91 -- | Custom 'Aeson.FromJSON' routes through 'mkSessionPrefix' so corrupted
   92 -- on-disk JSON cannot bypass the smart constructor.
   93 instance Aeson.FromJSON SessionPrefix where
   94   parseJSON = Aeson.withText "SessionPrefix" $ \t ->
   95     case mkSessionPrefix t of
   96       Right p -> pure p
   97       Left e -> fail ("invalid SessionPrefix: " ++ show e)

   99 -- | Pure session ID generator. Encodes a 'UTCTime' as
  100 -- @YYYYMMDD-HHMMSS-mmm@ (milliseconds zero-padded to three digits)
  101 -- and appends the optional 'SessionPrefix' after it, separated by a
  102 -- hyphen. The timestamp leads so that lexicographic sorting of session
  103 -- IDs (and their on-disk directory names) is also chronological.
  104 newSessionId :: Maybe SessionPrefix -> UTCTime -> SessionId
  105 newSessionId mPrefix time =
  106   let hms     = formatTime defaultTimeLocale "%Y%m%d-%H%M%S" time
  107       -- Extract milliseconds from the fractional second.
  108       picoDay = diffTimeToPicoseconds (utctDayTime time)
  109       millis  = (picoDay `div` 1000000000) `mod` 1000
  110       timeStr = T.pack hms <> "-" <> T.justifyRight 3 '0' (T.pack (show millis))
  111       full    = case mPrefix of
  112         Nothing -> timeStr
  113         Just p  -> timeStr <> "-" <> unSessionPrefix p
  114   in SessionId full

  139 -- | Persistent metadata for a single session. Stored as @session.json@
  140 -- inside the session's on-disk directory.
  141 data SessionMeta = SessionMeta
  142   { _sm_id                :: SessionId
  143   , _sm_agent             :: Maybe AgentName
  144   , _sm_kind              :: SessionKind
  145   , _sm_model             :: Text
  146   , _sm_channel           :: Text
  147   , _sm_createdAt         :: UTCTime
  148   , _sm_lastActive        :: UTCTime
  149   , _sm_bootstrapConsumed :: Bool
  150   , _sm_archived          :: Bool
  151     -- ^ User-controllable flag that suppresses this session from
  152     -- prominent UI surfaces (e.g. the "Recent Sessions" sidebar).
  153     -- Pure display state — archiving NEVER removes the session
  154     -- directory or transcript from disk. Defaults to 'False'.
  155   , _sm_description       :: Maybe Text
  156     -- ^ Optional user-provided description / title for this session.
  157     -- Display surfaces prefer this when set; otherwise they fall back
  158     -- through '_sm_autoSummary', a transcript-derived snippet, the
  159     -- agent name, and finally the session id. 'Nothing' means "no
  160     -- user choice — use a fallback."
  161   , _sm_autoSummary       :: Maybe Text
  162     -- ^ Optional model-generated short summary of the session,
  163     -- cached here so it doesn't have to be recomputed on every
  164     -- recent-sessions poll. Populated lazily by a separate
  165     -- summarization path (not yet wired up). Defaults to 'Nothing'
  166     -- on new sessions and after disk loads of older session.json
  167     -- files.
  168   , _sm_source            :: Maybe MessageSource
  169     -- ^ Session origin: the 'MessageSource' of the FIRST inbound
  170     -- message of this session (set-once). 'Nothing' for legacy
  171     -- sessions written before this field existed, and for sessions
  172     -- that have not yet received an inbound message.
  173     --
  174     -- SECURITY: this is attacker-asserted provenance — the sender id
  175     -- is unauthenticated. It MUST NOT feed any access-control or
  176     -- trust decision. Routing/authz key on the dispatcher's
  177     -- allow-list, never on this value.
  178   } deriving stock (Show, Eq, Generic)

  180 -- Hand-written JSON so we don't depend on a 'ToJSON' instance for
  181 -- 'AgentName' (which 'PureClaw.Agent.AgentDef' deliberately does not
  182 -- export). The agent is serialized as its bare 'unAgentName' string and
  183 -- deserialized via the existing 'FromJSON AgentName' instance, which
  184 -- routes through 'mkAgentName'.
  185 instance Aeson.ToJSON SessionMeta where
  186   toJSON s = Aeson.object $
  187     [ "id"                 .= _sm_id s
  188     , "kind"               .= _sm_kind s
  189     , "model"              .= _sm_model s
  190     , "channel"            .= _sm_channel s
  191     , "created_at"         .= _sm_createdAt s
  192     , "last_active"        .= _sm_lastActive s
  193     , "bootstrap_consumed" .= _sm_bootstrapConsumed s
  194     , "archived"           .= _sm_archived s
  195     , "description"        .= _sm_description s
  196     , "auto_summary"       .= _sm_autoSummary s
  197     ] <> (case _sm_agent s of
  198       Just n  -> ["agent" .= unAgentName n]
  199       Nothing -> [])
  200       <> (case _sm_source s of
  201       Just src -> ["source" .= src]
  202       Nothing  -> [])

  234 -- | Parse the session kind from the JSON object. Accepts three formats:
  235 --
  236 --   1. New format: @"kind"@ key present -> parse as 'SessionKind' directly.
  237 --   2. Legacy format: @"runtime"@ key present ->
  238 --      * @"provider"@ -> 'SkProvider' with inferred provider from model.
  239 --      * @"harness:\<name\>"@ -> 'SkHarness' with 'fixedFlavourLookup'.
  240 --   3. Neither present -> default to 'SkProvider' with inferred provider.
  241 parseSessionKind
  242   :: Aeson.Object
  243   -> Text              -- ^ model text (for inferring provider)
  244   -> Maybe AgentName   -- ^ agent (for ProviderSpec)
  245   -> Parser SessionKind
  246 parseSessionKind o model agent = do
  247   mKind    <- o .:? "kind"    :: Parser (Maybe Aeson.Value)
  248   mRuntime <- o .:? "runtime" :: Parser (Maybe Text)
  249   case mKind of
  250     Just kindVal -> Aeson.parseJSON kindVal
  251     Nothing -> case mRuntime of
  252       Just "provider" ->
  253         pure (SkProvider (ProviderSpec (inferProviderId model) (ModelId model) agent))
  254       Just rt | Just name <- T.stripPrefix "harness:" rt ->
  255         pure (SkHarness (HarnessSpec
  256           (fixedFlavourLookup name)
  257           (TmuxConfig name name Nothing)
  258           Nothing
  259           []
  260           Nothing
  261           Nothing
  262           Nothing))
  263       Just _ ->
  264         -- Unknown runtime text: default to provider
  265         pure (SkProvider (ProviderSpec (inferProviderId model) (ModelId model) agent))
  266       Nothing ->
  267         -- No "kind" and no "runtime": default to provider
  268         pure (SkProvider (ProviderSpec (inferProviderId model) (ModelId model) agent))
