    1 -- |
    2 -- Module      : PureClaw.Tabs.Persist
    3 -- Description : Persist the tab view to @~\/.pureclaw\/state\/tabs.json@.
    4 --
    5 -- The tab /view/ — the ordered 'TabList', the per-'ConversationKey' cursors,
    6 -- and the per-conversation 'RelayMode' overrides — is machine-local mutable
    7 -- runtime state, kept apart from the version-controllable session\/agent
    8 -- ground truth. It lives under @state\/@ (mode @0700@) as @tabs.json@ (mode
    9 -- @0600@). See the Tabs-as-View design §10 (GitHub #79).
   10 --
   11 -- Codec policy (project convention): the on-disk shape is a __hand-written__
   12 -- Aeson codec, never generic deriving, so the wire format is reviewable and
   13 -- stable. Maps with tuple keys ('ConversationKey') serialize as an /array of
   14 -- objects/ (@{"channel":…,"conversation":…,…}@), not as JSON object keys, so
   15 -- the key structure is explicit. A 'TabRef' is a tag-discriminated object
   16 -- (@session@\/@harness@), mirroring the @TerminalBackend@ codec in
   17 -- "PureClaw.Session.Kind".
   18 --
   19 -- Security (§2): the file stores __no secrets__ — no tokens, API keys,
   20 -- passwords, or absolute filesystem paths. Only tab refs (opaque ids),
   21 -- statuses, channel kinds, conversation ids, cursors, and relay modes.
   22 --
   23 -- IO seams are injected through 'PersistDeps' so boot reconcile is testable
   24 -- without a real tmux\/harness: liveness probing and the "one discovery pass
   25 -- completed" gate are both functions in the record.
   26 module PureClaw.Tabs.Persist
   27   ( -- * Injected dependencies
   28     PersistDeps (..)
   29     -- * Save \/ load
   30   , saveTabs
   31   , loadTabs

   82 -- | The IO seams 'loadTabs' needs, injected so boot reconcile is deterministic
   83 -- in tests (no real tmux\/harness).
   84 data PersistDeps = PersistDeps
   85   { _pd_stateDir       :: !FilePath
   86     -- ^ The @state\/@ directory (created @0700@ if missing on save).
   87   , _pd_harnessLive    :: !(HarnessId -> IO Bool)
   88     -- ^ Probe whether a harness-backed tab's ground truth is still alive.
   89   , _pd_discoveryReady :: !(IO ())
   90     -- ^ Block until at least one harness-discovery\/reconcile pass has
   91     --   completed, so a transiently-absent live harness is not wrongly
   92     --   dropped at boot. Awaited /before/ any harness pruning.
   93   , _pd_sessionExists  :: !(SessionId -> IO Bool)
   94     -- ^ Probe whether a session-backed tab's ground truth still exists on
   95     --   disk. A 'BoundSession' tab whose session.json has been removed
   96     --   out-of-band is dropped during reconcile (analogous to the harness
   97     --   liveness probe for 'BoundHarness' tabs).
   98   }
   99 
  100 -- | The on-disk file name under '_pd_stateDir'.
  101 tabsFileName :: FilePath
  102 tabsFileName = "tabs.json"
  103 
  104 -- ---------------------------------------------------------------------------
  105 -- Save
  106 -- ---------------------------------------------------------------------------
  107 
  108 -- | Write the tab view to @<stateDir>\/tabs.json@. The @state\/@ directory is
  109 -- created @0700@ if missing (via 'ensureRuntimeRoot').
  110 --
  111 -- Security-by-construction (§2): the file is created @0600@ /atomically/ — the
  112 -- payload is written to a unique temp file in the same directory, created at
  113 -- mode @0600@ by 'mkstemp' (POSIX), then 'rename'd over the target. This
  114 -- avoids both the world-readable window of a write-then-chmod (the file is
  115 -- never momentarily @0644@) and the re-exposure window when overwriting an
  116 -- existing @0600@ file. The same-directory rename is atomic, so a crash mid-
  117 -- write never leaves a torn @tabs.json@; the temp artifact is removed on any
  118 -- failure.
  119 saveTabs :: FilePath -> TabList -> CursorState -> IO ()
  120 saveTabs stateDir tabs cursors = do
  121   _ <- ensureRuntimeRoot stateDir
  122   let path    = stateDir </> tabsFileName
  123       tmplate = stateDir </> (tabsFileName <> ".")
  124       payload = BL.toStrict (Aeson.encode (encodeState tabs cursors))
  125   E.bracketOnError
  126     (Temp.mkstemp tmplate)
  127     (uncurry cleanupTemp)
  128     (\(tmpPath, h) -> do
  129         BS.hPut h payload
  130         IO.hClose h
  131         PF.rename tmpPath path)
  132 
  133 -- | Best-effort cleanup of a temp file after a failed save: close the handle
  134 -- and remove the file, swallowing any error so the original failure (e.g. a
  135 -- disk-full write) is the one that propagates.
  136 cleanupTemp :: FilePath -> IO.Handle -> IO ()
  137 cleanupTemp tmpPath h = do
  138   ignore (IO.hClose h)
  139   ignore (removeFile tmpPath)
  140   where
  141     ignore :: IO () -> IO ()
  142     ignore act = act `E.catch` \(_ :: SomeException) -> pure ()
  143 
  144 -- ---------------------------------------------------------------------------
  145 -- Load
  146 -- ---------------------------------------------------------------------------
  147 
  148 -- | Load and reconcile the tab view from @<stateDir>\/tabs.json@:
  149 --
  150 --   1. Read + decode the file. A missing file, an IO error, or any decode
  151 --      failure yields a __fresh__ @('emptyTabs', 'emptyCursors')@ — never an
  152 --      exception (Tabs-as-View design §10.2\/§10.3).
  153 --   2. Await one discovery pass ('_pd_discoveryReady') before pruning.
  154 --   3. Drop harness-backed tabs whose '_pd_harnessLive' reports False
  155 --      (silently — the documented I5 boot-drop exception); provider-session
  156 --      tabs are kept as-is.
  157 --   4. Prune any cursor whose 'TabRef' no longer resolves (I3).
  158 loadTabs :: PersistDeps -> IO (TabList, CursorState)
  159 loadTabs deps = do
  160   decoded <- readPersisted (_pd_stateDir deps </> tabsFileName)
  161   case decoded of
  162     Nothing                -> pure (emptyTabs, emptyCursors)
  163     Just (tabs, cursors)   -> do
  164       _pd_discoveryReady deps
  165       liveTabs <- reconcileTabs deps tabs
  166       pure (liveTabs, pruneDangling liveTabs cursors)
  167 
  168 -- | Read and decode the persisted state, returning 'Nothing' on any failure
  169 -- (missing file, IO error, or malformed JSON) so the caller starts fresh.
  170 --
  171 -- The read is __strict__ ('BS.readFile') and the decode strict
  172 -- ('Aeson.eitherDecodeStrict''), both inside the 'try'. Lazy
  173 -- 'Data.ByteString.Lazy.readFile' would defer the actual read syscalls until
  174 -- the bytes are forced by the decoder, /outside/ the guard — a read-time
  175 -- 'IOException' (truncation or removal mid-read, disk @EIO@, a stale NFS
  176 -- handle) would then escape 'loadTabs' and crash boot. Reading and decoding
  177 -- strictly forces every read syscall and the full parse within the guarded
  178 -- scope, so any IO error or decode error degrades to a fresh start.
  179 readPersisted :: FilePath -> IO (Maybe (TabList, CursorState))
  180 readPersisted path = do
  181   result <- try (BS.readFile path) :: IO (Either SomeException BS.ByteString)
  182   case result of
  183     Left _    -> pure Nothing
  184     Right raw -> case Aeson.eitherDecodeStrict' raw of
  185       Left _   -> pure Nothing
  186       Right ps -> pure (Just (psTabs ps, psCursors ps))
  187 
  188 -- | A tab's persisted content, slot excluded — the slot is always re-derived
  189 -- (I1) by 'rebuildTabs', never trusted from disk.
  190 type TabSpec = (TabRef, TabStatus)
  191 
  192 -- | Project a 'Tab' onto its persisted content (dropping the slot).
  193 tabSpec :: Tab -> TabSpec
  194 tabSpec t = (_tab_ref t, _tab_status t)
  195 
  196 -- | Drop harness-backed tabs whose ground truth is gone. Provider tabs are
  197 -- always retained; the surviving tabs are re-appended in order so the result
  198 -- satisfies I1 (contiguous slots) via 'rebuildTabs'.
  199 reconcileTabs :: PersistDeps -> TabList -> IO TabList
  200 reconcileTabs deps tabs = do
  201   survivors <- filterM keepTab (toList tabs)
  202   pure (rebuildTabs (map tabSpec survivors))
  203   where
  204     keepTab :: Tab -> IO Bool
  205     keepTab t = case _tab_ref t of
  206       BoundSession s -> _pd_sessionExists deps s
  207       BoundHarness h -> _pd_harnessLive deps h
  208 
  209 -- | Re-append a list of tab specs into a fresh 'TabList', stamping contiguous
  210 -- slots (I1) and carrying each tab's original 'TabStatus' through (so a 'Dead'
  211 -- tombstone survives). The input satisfies I2 (unique refs) — for decoded
  212 -- input this is enforced by 'appendTab' itself silently de-duplicating, which
  213 -- is the desired tolerant-load behaviour.
  214 rebuildTabs :: [TabSpec] -> TabList
  215 rebuildTabs = foldl step emptyTabs
  216   where
  217     step tl (ref, status) = case appendTab ref tl of
  218       Right (_, tl') -> setStatus ref status tl'
  219       Left _         -> tl
  220 
  221 -- | A small local 'filterM' to avoid pulling in @Control.Monad@ for one use.
  222 filterM :: Applicative m => (a -> m Bool) -> [a] -> m [a]
  223 filterM p = foldr step (pure [])
  224   where
  225     step x acc = (\keep rest -> if keep then x : rest else rest) <$> p x <*> acc

  231 -- | The decoded on-disk payload.
  232 data PersistedState = PersistedState
  233   { psTabs    :: !TabList
  234   , psCursors :: !CursorState
  235   }
  236 
  237 -- | Build the top-level JSON 'Aeson.Value' for the tab view.
  238 encodeState :: TabList -> CursorState -> Aeson.Value
  239 encodeState tabs cursors =
  240   Aeson.object
  241     [ "tabs"    .= map encodeTab (toList tabs)
  242     , "cursors" .= map encodeCursorEntry (Map.toList (_cs_cursors cursors))
  243     , "relay"   .= map encodeRelayEntry (Map.toList (_cs_relay cursors))
  244     ]
  245 
  246 instance Aeson.FromJSON PersistedState where
  247   parseJSON = Aeson.withObject "PersistedState" $ \o -> do
  248     tabRows    <- o .: "tabs"
  249     cursorRows <- o .: "cursors"
  250     relayRows  <- o .: "relay"
  251     tabs       <- rebuildTabs <$> traverse parseTab tabRows
  252     cursorMap  <- Map.fromList <$> traverse parseCursorEntry cursorRows
  253     relayMap   <- Map.fromList <$> traverse parseRelayEntry relayRows
  254     pure (PersistedState tabs (CursorState cursorMap relayMap))
  255 
  256 -- Tab -----------------------------------------------------------------------
  257 
  258 encodeTab :: Tab -> Aeson.Value
  259 encodeTab t =
  260   Aeson.object
  261     [ "ref"    .= encodeRef (_tab_ref t)
  262     , "status" .= statusToText (_tab_status t)
  263     ]
  264 
  265 -- | Parse a tab row into a slot-less 'TabSpec'; the slot is re-derived on load
  266 -- (I1) by 'rebuildTabs', never read from disk.
  267 --
  268 -- Back-compat: a legacy @tabs.json@ may carry a per-tab @"name"@ key (tabs no
  269 -- longer have their own label — the title derives from the bound session). The
  270 -- key is ignored on load; Aeson tolerates the extra field, so older files keep
  271 -- loading and re-encode without it.
  272 parseTab :: Aeson.Value -> Aeson.Parser TabSpec
  273 parseTab = Aeson.withObject "Tab" $ \o -> do
  274   ref    <- o .: "ref" >>= parseRef
  275   status <- o .: "status" >>= statusFromText
  276   pure (ref, status)
  277 
  278 -- TabRef --------------------------------------------------------------------
  279 
  280 encodeRef :: TabRef -> Aeson.Value
  281 encodeRef (BoundSession (SessionId s)) =
  282   Aeson.object ["tag" .= ("session" :: Text), "sessionId" .= s]
  283 encodeRef (BoundHarness h) =
  284   Aeson.object ["tag" .= ("harness" :: Text), "harnessId" .= harnessIdToText h]
  285 
  286 parseRef :: Aeson.Value -> Aeson.Parser TabRef
  287 parseRef = Aeson.withObject "TabRef" $ \o -> do
  288   tag <- o .: "tag" :: Aeson.Parser Text
  289   case tag of
  290     "session" -> do
  291       raw <- o .: "sessionId"
  292       if isValidSessionId raw
  293         then pure (BoundSession (SessionId raw))
  294         else fail "invalid sessionId in tabs.json"
  295     "harness" -> do
  296       raw <- o .: "harnessId"
  297       case parseHarnessId raw of
  298         Just h  -> pure (BoundHarness h)
  299         Nothing -> fail "invalid harnessId in tabs.json"
  300     _ -> fail "unknown TabRef tag"
  301 
  302 -- TabStatus -----------------------------------------------------------------
  303 
  304 statusToText :: TabStatus -> Text
  305 statusToText Live = "live"
  306 statusToText Dead = "dead"
  307 
  308 statusFromText :: Text -> Aeson.Parser TabStatus
  309 statusFromText "live" = pure Live
  310 statusFromText "dead" = pure Dead
  311 statusFromText _      = fail "unknown tab status"
  312 
  313 -- Cursor entry --------------------------------------------------------------
  314 
  315 encodeCursorEntry :: (ConversationKey, TabRef) -> Aeson.Value
  316 encodeCursorEntry ((ch, conv), ref) =
  317   Aeson.object
  318     [ "channel"      .= channelKindToText ch
  319     , "conversation" .= conversationIdToText conv
  320     , "ref"          .= encodeRef ref
  321     ]
  322 
  323 parseCursorEntry :: Aeson.Value -> Aeson.Parser (ConversationKey, TabRef)
  324 parseCursorEntry = Aeson.withObject "CursorEntry" $ \o -> do
  325   key <- parseKey o
  326   ref <- o .: "ref" >>= parseRef
  327   pure (key, ref)
  328 
  329 -- Relay entry ---------------------------------------------------------------
  330 
  331 encodeRelayEntry :: (ConversationKey, RelayMode) -> Aeson.Value
  332 encodeRelayEntry ((ch, conv), mode) =
  333   Aeson.object
  334     [ "channel"      .= channelKindToText ch
  335     , "conversation" .= conversationIdToText conv
  336     , "mode"         .= relayModeToText mode
  337     ]
  338 
  339 parseRelayEntry :: Aeson.Value -> Aeson.Parser (ConversationKey, RelayMode)
  340 parseRelayEntry = Aeson.withObject "RelayEntry" $ \o -> do
  341   key  <- parseKey o
  342   mode <- o .: "mode" >>= relayModeFromText
  343   pure (key, mode)
  344 
  345 -- ConversationKey -----------------------------------------------------------
  346 
  347 parseKey :: Aeson.Object -> Aeson.Parser ConversationKey
  348 parseKey o = do
  349   ch   <- channelKindFromText <$> o .: "channel"
  350   conv <- ConversationId <$> o .: "conversation"
  351   pure (ch, conv)
  352 
  353 conversationIdToText :: ConversationId -> Text
  354 conversationIdToText (ConversationId t) = t

  358 relayModeToText :: RelayMode -> Text
  359 relayModeToText FocusedOnly    = "focused"
  360 relayModeToText ActivityDigest = "activity"

  363 relayModeFromText :: Text -> Aeson.Parser RelayMode
  364 relayModeFromText "focused"  = pure FocusedOnly
  365 relayModeFromText "activity" = pure ActivityDigest
