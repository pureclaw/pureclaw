    1 -- |
    2 -- Module      : PureClaw.Backend.Tmux
    3 -- Description : Tmux Attach backend factory (WU10).
    4 --
    5 -- == Overview
    6 --
    7 -- 'mkTmuxBackendHandle' produces a 'Pty'-kind 'BackendHandle' that
    8 -- attaches to a __pre-existing__ tmux window — either via a local
    9 -- @tmux attach-session@ inside a PTY or via @ssh -tt user\@host tmux
   10 -- attach-session@ inside a PTY. The factory never creates the window;
   11 -- if the target session\/window does not exist at construction time it
   12 -- returns @Left (BackendTmuxTargetMissing _)@ and spawns no further
   13 -- subprocesses.
   14 --
   15 -- == Two phases at construction
   16 --
   17 -- 1. __Pin-resolve__ (TOCTOU mitigation, design doc § "Pinning to
   18 --    stable ids"). Run @tmux display-message -p -t \'sess:win\'
   19 --    \'#{window_id}\'@ as a one-shot (NOT a PTY); validate the stdout
   20 --    against @^@[0-9]+$@. The pinned @@@@ID@ is what subsequent
   21 --    @attach-session@ targets use. If a human later destroys and
   22 --    re-creates the same-named window, the pinned @@@@ID@ will be
   23 --    stale and the optional mid-session probe (DoD #21) surfaces a
   24 --    'BackendBrokenTmuxTarget' rather than silently writing to the
   25 --    new window.
   26 -- 2. __Attach__. Spawn the tmux client inside a PTY via
   27 --    'mkDrainerBackendHandle'. For 'LocalHost' the spawned program is
   28 --    the local tmux binary; for 'RemoteHost' the spawned program is
   29 --    ssh, with the remote tmux argv composed via
   30 --    'PureClaw.Backend.SSH.buildSshArgv'.
   31 --
   32 -- == Detach-only close
   33 --
   34 -- '_bh_close' writes the tmux detach key sequence (Ctrl-B @d@ — bytes
   35 -- @[0x02, 0x64]@) into the existing PTY and then calls the inner
   36 -- handle's close. It does NOT open a new ssh hop, NOT issue
   37 -- @kill-window@\/@kill-session@\/@kill-pane@. A pre-existing window
   38 -- survives close. Idempotent and never throws.
   39 --
   40 -- == Mid-session destruction (DoD #21)
   41 --
   42 -- When '_to_brokenTargetCheck' is 'True', the wrapper around
   43 -- '_bh_recv' re-resolves the pinned @@@@ID@ after a settled recv; if
   44 -- the resolution returns a different @@@@ID@ (window destroyed and
   45 -- re-created by some other actor) the next caller-visible operation
   46 -- raises 'BackendException' with @_be_context = 'BcTmuxDetach'@ whose
   47 -- @_be_cause@ wraps a 'BackendBrokenTmuxTarget' payload (via the
   48 -- internal 'BrokenTmuxTargetException' carrier — recovered by tests
   49 -- via 'fromException').
   50 --
   51 -- The check is OFF by default (the production cost of an extra
   52 -- @display-message@ probe per recv is real); tests turn it on
   53 -- explicitly. See @docs\/terminal-backend-abstractions.md@ DoD #21.

  135 -- | A validated tmux session name.
  136 --
  137 -- The constructor is intentionally NOT exported. Obtain a value only
  138 -- via 'mkTmuxSession', which enforces the charset rules below.
  139 --
  140 -- Validation (matches 'mkTmuxWindow' \/ 'mkTmuxPane'):
  141 --
  142 -- 1. Empty rejected.
  143 -- 2. Leading @-@ rejected (defends against argv injection at the tmux
  144 --    binary).
  145 -- 3. NUL byte rejected.
  146 -- 4. Any non-ASCII byte rejected.
  147 -- 5. Length must be <= 200.
  148 -- 6. Permitted charset: @[A-Za-z0-9_./@:=+-]@.
  149 newtype TmuxSession = TmuxSession Text
  150   deriving stock (Eq, Ord, Show)

  194 -- | Shared validator for 'mkTmuxSession' \/ 'mkTmuxWindow' \/
  195 -- 'mkTmuxPane'. The first argument is the prefix used in the
  196 -- 'InvalidOptionDetail' message.
  197 validateTmuxName :: Text -> Text -> Either InvalidOptionDetail Text
  198 validateTmuxName ctx t
  199   | T.null t =
  200       reject "empty name"
  201   | T.head t == '-' =
  202       reject "leading dash"
  203   | T.any (== '\NUL') t =
  204       reject "NUL byte"
  205   | T.any (not . isAscii) t =
  206       reject "non-ASCII"
  207   | T.length t > 200 =
  208       reject "too long (> 200)"
  209   | T.any (not . allowed) t =
  210       reject "disallowed character"
  211   | otherwise =
  212       Right t
  213   where
  214     reject :: Text -> Either InvalidOptionDetail Text
  215     reject msg = Left (InvalidOptionDetail (ctx <> ": " <> msg))
  216 
  217     allowed :: Char -> Bool
  218     allowed c =
  219       isAsciiLower c
  220         || isAsciiUpper c
  221         || isDigit c
  222         || c == '_'
  223         || c == '.'
  224         || c == '/'
  225         || c == '@'
  226         || c == ':'
  227         || c == '='
  228         || c == '+'
  229         || c == '-'

  242 -- | Construction options for the tmux backend.
  243 --
  244 -- '_to_socketPath' pins the tmux server socket (@tmux -S \<path\>@);
  245 -- when 'Nothing', the system-default socket is used (a v1 limitation on

  248 -- '_to_brokenTargetCheck' is the opt-in flag for the DoD #21 mid-session

  251 -- as 'BackendException' carrying a 'BackendBrokenTmuxTarget' payload.

  253 -- (This field is a v1 superset over the active plan's WU10 spec —

  256 -- enables it explicitly.)

  276 -- | Type-enforced authorization split for the two operating modes.
  277 --
  278 -- * 'LocalHost' carries a local 'AuthorizedCommand' (the @tmux@ binary).
  279 -- * 'RemoteHost' carries: the 'SshTarget' (user, host, identity), the
  280 --   local 'AuthorizedCommand' (the @ssh@ binary, allowlisted via
  281 --   'PureClaw.Security.Command.authorize'), AND the 'RemoteCommand' (the
  282 --   remote @tmux@ binary, allowlisted via
  283 --   'PureClaw.Backend.SSH.authorizeRemote'). Both authorizations are
  284 --   required to construct the value — the factory cannot be called
  285 --   with only one or with a local 'AuthorizedCommand' standing in for
  286 --   the remote tmux.
  287 data SshLocation
  288   = LocalHost !AuthorizedCommand
  289   | RemoteHost !SshTarget !AuthorizedCommand !RemoteCommand

  308 -- | Injection seam for the pin-resolve probe.
  309 --
  310 -- Production uses 'realTmuxIO', which shells out to @tmux
  311 -- display-message -p -t ... '#{window_id}'@ (or its remote analogue
  312 -- via @ssh user\@host -- tmux ...@). Tests inject a fake that returns
  313 -- a scripted 'Right' \/ 'Left' for each call.
  314 newtype TmuxIO = TmuxIO
  315   { _tio_resolvePin :: SshLocation -> TmuxTarget -> Maybe SafeRuntimePath -> IO (Either BackendError Text)
  316   }
  317 
  318 -- | Production 'TmuxIO': spawns the pin-resolve subprocess via
  319 -- @typed-process@'s 'P.readProcessStdout_'.
  320 --
  321 -- For 'LocalHost' the program is the local tmux binary; for
  322 -- 'RemoteHost' it is ssh, with the remote tmux argv built via
  323 -- 'PureClaw.Backend.SSH.buildSshArgv' (shell-quoted on the remote
  324 -- side).
  325 --
  326 -- The pin-resolve probe is NEVER spawned in a PTY — a one-shot pipe
  327 -- is sufficient (and avoids burning a PTY on each construction).
  328 realTmuxIO :: TmuxIO
  329 realTmuxIO = TmuxIO
  330   { _tio_resolvePin = resolvePinReal
  331   }
  332 
  333 -- | Spawn the pin-resolve probe as a one-shot subprocess and validate
  334 -- the stdout against the expected @@@@\<digits\>@ regex.
  335 --
  336 -- A non-zero exit, an empty stdout, or a stdout that does not match
  337 -- the regex collapses to @Left (BackendTmuxTargetMissing _)@. This is
  338 -- intentional: from the caller\'s point of view, "the window does not
  339 -- exist" and "tmux returned malformed output" are indistinguishable
  340 -- — both mean "do not attach".
  341 resolvePinReal
  342   :: SshLocation
  343   -> TmuxTarget
  344   -> Maybe SafeRuntimePath
  345   -> IO (Either BackendError Text)
  346 resolvePinReal loc tgt mSock = do
  347   let (prog, argv) = buildPinResolveArgv loc tgt mSock
  348       config = P.setStdin (P.byteStringInput BL.empty)
  349              $ P.setStderr P.nullStream
  350              $ P.proc prog argv
  351   r <- try @SomeException (P.readProcessStdout config)
  352   let missing = Left (BackendTmuxTargetMissing (tmuxTargetRef tgt))
  353   case r of
  354     Left _ -> pure missing
  355     Right (ExitSuccess, bsLazy) ->
  356       let bs   = BL.toStrict bsLazy
  357           line = T.strip (TE.decodeUtf8 bs)
  358       in if isPinId line then pure (Right line) else pure missing
  359     Right (ExitFailure _, _) -> pure missing
  360 
  361 -- | Predicate: does the supplied 'Text' match the @^@[0-9]+$@ regex
  362 -- that tmux\'s @#{window_id}@ format string produces?
  363 --
  364 -- We deliberately accept @@@@<digits>@ rather than the pane-id form
  365 -- @%@<digits>@ — the pin we cache is the window id, even when the
  366 -- caller also specified a pane.
  367 isPinId :: Text -> Bool
  368 isPinId t = case T.unpack t of
  369   '@' : rest@(_:_) -> all isAsciiDigit rest
  370   _                -> False
  371   where
  372     isAsciiDigit c = isAscii c && isDigit c
  373

  378 -- | Build the argv list for the pin-resolve subprocess.
  379 --
  380 -- Exported as the test seam. The returned tuple is @(program, args)@,
  381 -- ready to splice into a 'P.proc' call. For 'LocalHost' the program
  382 -- is the local tmux binary path (from the local 'AuthorizedCommand');
  383 -- for 'RemoteHost' the program is ssh, and the tmux invocation is
  384 -- spliced into the argv via 'PureClaw.Backend.SSH.buildSshArgv'
  385 -- (which is also the only legal way to construct the hardened ssh
  386 -- prefix).
  387 --
  388 -- The probe always asks for @#{window_id}@ specifically (NOT pane
  389 -- id), even when '_tt_pane' is set: the pin is the window-level id.
  390 buildPinResolveArgv
  391   :: SshLocation
  392   -> TmuxTarget
  393   -> Maybe SafeRuntimePath
  394   -> (FilePath, [String])
  395 buildPinResolveArgv loc tgt mSock = case loc of
  396   LocalHost authCmd ->
  397     let prog = getCommandProgram authCmd
  398         argv = socketArgs mSock
  399             <> [ "display-message"
  400                , "-p"
  401                , "-t", T.unpack targetExpr
  402                , "#{window_id}"
  403                ]
  404     in (prog, argv)
  405   RemoteHost sshTgt sshAuthCmd remoteCmd ->
  406     let prog       = getCommandProgram sshAuthCmd
  407         remoteProg = getRemoteProgram remoteCmd
  408         remoteArgs = map T.pack (socketArgs mSock)
  409                   <> [ "display-message"
  410                      , "-p"
  411                      , "-t", targetExpr
  412                      , "#{window_id}"
  413                      ]
  414         argv = buildSshArgvFromParts sshTgt remoteProg remoteArgs
  415                  Nothing Nothing False
  416     in (prog, argv)
  417   where
  418     targetExpr :: Text
  419     targetExpr =
  420       getTmuxSession (_tt_session tgt)
  421         <> ":"
  422         <> getTmuxWindow (_tt_window tgt)
  423 
  424 -- | Build the argv for the attach-session step.
  425 --
  426 -- Returned as @(program, args)@ — ready for 'PtyOpenSpec'. The pin
  427 -- argument is the @@@\<id\>@ produced by the pin-resolve step; it is
  428 -- spliced as the @-t@ target so the attach is bound to the stable id,
  429 -- not the user-facing @session:window@ name. (Mid-session window
  430 -- destruction is therefore detectable: the @@@\<id\>@ probe and the
  431 -- attach diverge.)
  432 --
  433 -- For 'RemoteHost' the program path comes from the local @ssh@
  434 -- 'AuthorizedCommand'; the remote tmux invocation is spliced into the
  435 -- ssh argv via 'buildSshArgv', so the program-path and arg quoting
  436 -- (DoD #11) goes through 'PureClaw.Internal.ShellQuote.shellQuote' on
  437 -- the remote half.
  438 buildAttachArgv
  439   :: SshLocation
  440   -> Text               -- ^ the pinned @@@\<id\>@
  441   -> Maybe SafeRuntimePath
  442   -> (FilePath, [String])
  443 buildAttachArgv loc pinned mSock = case loc of
  444   LocalHost authCmd ->
  445     let prog = getCommandProgram authCmd
  446         argv = socketArgs mSock
  447             <> [ "attach-session"
  448                , "-t", T.unpack pinned
  449                ]
  450     in (prog, argv)
  451   RemoteHost sshTgt sshAuthCmd remoteCmd ->
  452     let prog       = getCommandProgram sshAuthCmd
  453         remoteProg = getRemoteProgram remoteCmd
  454         remoteArgs = map T.pack (socketArgs mSock)
  455                   <> [ "attach-session"
  456                      , "-t", pinned
  457                      ]
  458         argv = buildSshArgvFromParts sshTgt remoteProg remoteArgs
  459                  Nothing Nothing True
  460     in (prog, argv)
  461 
  462 -- | The @-S \<path\>@ prefix when '_to_socketPath' is set.
  463 socketArgs :: Maybe SafeRuntimePath -> [String]
  464 socketArgs Nothing  = []
  465 socketArgs (Just p) = ["-S", getSafeRuntimePath p]

  499 -- | Construct a tmux Attach backend, using 'realPtyIO' for the attach
  500 -- step and 'realTmuxIO' for the pin-resolve probe.
  501 --
  502 -- See 'mkTmuxBackendHandleWith' for the full semantics; this is the
  503 -- production entry point.
  504 mkTmuxBackendHandle
  505   :: PtyIO
  506   -> SshLocation
  507   -> TmuxTarget
  508   -> TmuxOpts
  509   -> IO (Either BackendError BackendHandle)
  510 mkTmuxBackendHandle = mkTmuxBackendHandleWith realTmuxIO
  511 
  512 -- | Construct a tmux Attach backend with caller-supplied 'TmuxIO' and
  513 -- 'PtyIO'. Test seam.
  514 --
  515 -- Steps:
  516 --
  517 -- 1. Run the pin-resolve probe via @_tio_resolvePin@. On failure
  518 --    surface the error as-is (typically
  519 --    @Left (BackendTmuxTargetMissing _)@; the SSH-hop case may surface
  520 --    @Left (BackendSshConnectFailed _)@ from the fake or production
  521 --    'TmuxIO').
  522 -- 2. Compose the attach argv via 'buildAttachArgv' against the pinned
  523 --    @@@\<id\>@ from step 1.
  524 -- 3. Delegate to 'mkDrainerBackendHandle' to spawn the PTY + drainer.
  525 -- 4. Wrap the resulting handle: substitute '_bh_close' for the
  526 --    detach-key write, optionally re-arm the mid-session probe in
  527 --    '_bh_recv', and apply 'withConcurrentUseGuard'.
  528 mkTmuxBackendHandleWith
  529   :: TmuxIO
  530   -> PtyIO
  531   -> SshLocation
  532   -> TmuxTarget
  533   -> TmuxOpts
  534   -> IO (Either BackendError BackendHandle)
  535 mkTmuxBackendHandleWith tio pio loc tgt opts = do
  536   pinResult <- _tio_resolvePin tio loc tgt (_to_socketPath opts)
  537   case pinResult of
  538     Left e -> pure (Left e)
  539     Right pinned -> do
  540       let (prog, argv) = buildAttachArgv loc pinned (_to_socketPath opts)
  541           ptyOpts = _to_pty opts
  542           spec = PtyOpenSpec
  543             { _pos_program = prog
  544             , _pos_args    = map T.pack argv
  545             , _pos_env     = _pto_env ptyOpts
  546             , _pos_cwd     = _pto_cwd ptyOpts
  547             , _pos_cols    = _pto_cols ptyOpts
  548             , _pos_rows    = _pto_rows ptyOpts
  549             }
  550       inner <- mkDrainerBackendHandle pio ptyOpts spec Pty
  551       case inner of
  552         Left e   -> pure (Left e)
  553         Right bh -> do
  554           wrapped <- wrapTmuxHandle tio loc tgt opts pinned bh
  555           Right <$> withConcurrentUseGuard wrapped
  556 
  557 -- | Wrap a freshly-constructed PTY 'BackendHandle' with the tmux-
  558 -- specific invariants:
  559 --
  560 -- * @_bh_close@ writes Ctrl-B @d@ into the existing PTY, then calls
  561 --   the inner close. Idempotent (MVar-guarded), never throws.
  562 -- * @_bh_recv@ optionally re-runs the pin-resolve probe after a
  563 --   settled recv when @_to_brokenTargetCheck@ is 'True'. If the
  564 --   probe returns a different @@@\<id\>@, the next recv (or send)
  565 --   surfaces 'BackendException' carrying
  566 --   'BrokenTmuxTargetException'.
  567 wrapTmuxHandle
  568   :: TmuxIO
  569   -> SshLocation
  570   -> TmuxTarget
  571   -> TmuxOpts
  572   -> Text                     -- ^ the pinned @@@\<id\>@ from construction
  573   -> BackendHandle
  574   -> IO BackendHandle
  575 wrapTmuxHandle tio loc tgt opts pinnedAtCtor bh = do
  576   closeLock <- newMVar ()
  577   closedRef <- newIORef False
  578   brokenRef <- newIORef False     -- latches True once probe disagrees
  579   let trackedRecv mIdle = do
  580         -- Pre-check: if a prior probe already detected divergence,
  581         -- raise immediately (don't even talk to the PTY).
  582         wasBroken <- readIORef brokenRef
  583         if wasBroken
  584           then throwBroken
  585           else do
  586             res <- _bh_recv bh mIdle
  587             postCheck res
  588             pure res
  589 
  590       postCheck :: RecvResult ByteString -> IO ()
  591       postCheck r
  592         | _to_brokenTargetCheck opts = case r of
  593             RecvSettled _ -> do
  594               probe <- _tio_resolvePin tio loc tgt (_to_socketPath opts)
  595               case probe of
  596                 Right newPin
  597                   | newPin /= pinnedAtCtor -> do
  598                       writeIORef brokenRef True
  599                       throwBroken
  600                   | otherwise -> pure ()
  601                 Left _ -> do
  602                   -- Probe failed entirely — treat as broken.
  603                   writeIORef brokenRef True
  604                   throwBroken
  605             _ -> pure ()
  606         | otherwise = pure ()
  607 
  608       trackedSend bs = do
  609         wasBroken <- readIORef brokenRef
  610         if wasBroken
  611           then throwBroken
  612           else _bh_send bh bs
  613 
  614       performClose :: TmuxCloseAction -> IO ()
  615       performClose TmuxDetach = do
  616         -- Ctrl-B (0x02) + 'd' (0x64). Tmux's default prefix +
  617         -- detach-client binding. Write failures are non-fatal:
  618         -- the transport may already be dead. We swallow the
  619         -- exception to honour the "_bh_close never throws"
  620         -- contract.
  621         _ <- try @SomeException (_bh_send bh (BS.pack [0x02, 0x64]))
  622         -- Defensive: the inner close is documented total today, but
  623         -- the contract on this wrapper is "MUST NOT throw". Swallow
  624         -- any exception escaping the inner close so a future
  625         -- regression cannot silently break the wrapper contract.
  626         _ <- try @SomeException (_bh_close bh)
  627         pure ()
  628 
  629       detachClose = modifyMVar_ closeLock $ \() -> do
  630         wasClosed <- readIORef closedRef
  631         if wasClosed
  632           then pure ()
  633           else do
  634             writeIORef closedRef True
  635             performClose TmuxDetach
  636 
  637       throwBroken :: IO a
  638       throwBroken = throwIO $ BackendException
  639         BcTmuxDetach
  640         (toException (BrokenTmuxTargetException (tmuxTargetRef tgt)))
  641 
  642   pure bh
  643     { _bh_name  = "tmux-attach"
  644     , _bh_send  = trackedSend
  645     , _bh_recv  = trackedRecv
  646     , _bh_close = detachClose
  647     }
