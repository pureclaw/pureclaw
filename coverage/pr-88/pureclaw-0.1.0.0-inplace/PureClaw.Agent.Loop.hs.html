   34 -- | Maximum turns for a @\/bg@ background task (provider + tool-call cycles).
   35 backgroundMaxTurns :: Int
   36 backgroundMaxTurns = 20
   37 
   38 -- | Run a @\/bg@ prompt in a fresh background session and push the result
   39 -- directly to the channel (issue #52).
   40 --
   41 -- The background turn runs in its OWN session — a brand-new
   42 -- 'PureClaw.Session.Handle.SessionHandle' created under the same sessions
   43 -- directory as the foreground session and wired to the same
   44 -- 'PureClaw.Frontend.StreamBroker.StreamBroker'. The provider is wrapped
   45 -- with 'mkTranscriptProvider' so every request/response is recorded to the
   46 -- session's @transcript.jsonl@ (and broadcast to the broker). This is what
   47 -- makes the background conversation appear in the frontend UI like any
   48 -- other conversation: the frontend enumerates session directories on disk
   49 -- and subscribes to broker events. The fresh session also means the
   50 -- background turn does NOT leak into the foreground conversation history.
   51 --
   52 -- It uses the process default provider/model and the effective tool
   53 -- registry (built-ins + connected MCP servers), runs to completion via
   54 -- 'runSubAgent' (non-streaming, so it does not interleave with the
   55 -- foreground), and emits a single @[bg done] …@ message via '_ch_send'.
   56 -- This intentionally does NOT go through the tab-output queue
   57 -- ('_env_tabOutQ') — the background turn emits directly so its result is
   58 -- delivered even without the relay-writer thread in scope.
   59 --
   60 -- Provider/model-absent and provider-failure cases each emit a short,
   61 -- redacted @[bg] …@ message rather than throwing (the caller forks this
   62 -- with '_env_fork' and does not observe its result).
   63 runBackgroundTurn :: AgentEnv -> Text -> IO ()
   64 runBackgroundTurn env prompt = do
   65   let channel = _env_channel env
   66   mProvider <- readIORef (_env_provider env)
   67   mModel    <- readIORef (_env_model env)
   68   case (mProvider, mModel) of
   69     (Nothing, _) ->
   70       _ch_send channel (OutgoingMessage "[bg] Cannot run: no provider configured.")
   71     (_, Nothing) ->
   72       _ch_send channel (OutgoingMessage "[bg] Cannot run: no model configured.")
   73     (Just provider, Just model) -> do
   74       outcome <- try @SomeException (runBackgroundSession env provider model prompt)
   75       case outcome of
   76         Left e -> do
   77           _lh_logError (_env_logger env) $
   78             "Background task error: " <> T.pack (show e)
   79           _ch_send channel (OutgoingMessage
   80             "[bg] Something went wrong running the background task.")
   81         Right text ->
   82           let body = if T.null (T.strip text) then "(no response)" else text
   83           in _ch_send channel (OutgoingMessage ("[bg done] " <> body))

   85 -- | Create the fresh background session, run the prompt against a
   86 -- transcript-recording provider, persist + close the session, and return
   87 -- the final assistant text. Any exception propagates to 'runBackgroundTurn'
   88 -- (which reports a redacted failure); the session is saved + closed
   89 -- regardless via 'finally'.
   90 runBackgroundSession :: AgentEnv -> SomeProvider -> ModelId -> Text -> IO Text
   91 runBackgroundSession env provider model prompt = do
   92   registry  <- backgroundRegistry env
   93   bgSession <- mkBackgroundSession env model prompt
   94   let transcript = Session._sh_transcript bgSession
   95       -- Recording wrapper: each provider call writes a request/response
   96       -- pair to the session transcript (and fans out to the broker).
   97       provider'  = mkTranscriptProvider transcript (unModelId model) Nothing provider
   98   runSubAgent provider' model registry (_env_systemPrompt env)
   99               prompt backgroundMaxTurns
  100     `finally` do
  101       ignoreExc (Session._sh_save bgSession)
  102       ignoreExc (Transcript._th_close transcript)

  104 -- | Build a fresh on-disk session for a @\/bg@ turn, rooted under the same
  105 -- sessions directory as the foreground session (so the frontend — which
  106 -- scans that directory — discovers it) and wired to '_env_broker' (so its
  107 -- transcript writes broadcast live).
  108 mkBackgroundSession :: AgentEnv -> ModelId -> Text -> IO Session.SessionHandle
  109 mkBackgroundSession env model prompt = do
  110   now         <- getCurrentTime
  111   sessionsDir <- backgroundSessionsDir env
  112   createDirectoryIfMissing True sessionsDir
  113   let modelTxt = unModelId model
  114       mAgent   = AgentDef._ad_name <$> _env_agentDef env
  115       meta = SessionTypes.SessionMeta
  116         { SessionTypes._sm_id    = SessionTypes.newSessionId Nothing now
  117         , SessionTypes._sm_agent = mAgent
  118         , SessionTypes._sm_kind  = SessionTypes.SkProvider
  119             (SessionTypes.ProviderSpec
  120               (SessionTypes.inferProviderId modelTxt) model mAgent)
  121         , SessionTypes._sm_model   = modelTxt
  122         , SessionTypes._sm_channel = "bg"
  123         , SessionTypes._sm_createdAt  = now
  124         , SessionTypes._sm_lastActive = now
  125         , SessionTypes._sm_bootstrapConsumed = False
  126         , SessionTypes._sm_archived = False
  127         , SessionTypes._sm_description = Just (backgroundDescription prompt)
  128         , SessionTypes._sm_autoSummary = Nothing
  129         , SessionTypes._sm_source = Nothing
  130         }
  131   Session.mkSessionHandle (_env_broker env) (_env_logger env) sessionsDir meta

  133 -- | The sessions directory a @\/bg@ session should be created under: the
  134 -- parent of the foreground session's directory (which is exactly the
  135 -- directory the frontend enumerates). Falls back to 'getSessionsDir' when
  136 -- the foreground session has no on-disk directory (e.g. a no-op handle).
  137 backgroundSessionsDir :: AgentEnv -> IO FilePath
  138 backgroundSessionsDir env = do
  139   sh <- readIORef (_env_session env)
  140   let dir = Session._sh_dir sh
  141   if null dir then getSessionsDir else pure (takeDirectory dir)

  143 -- | A short sidebar label for a @\/bg@ session.
  144 backgroundDescription :: Text -> Text
  145 backgroundDescription prompt = "/bg: " <> T.take 80 (T.strip prompt)

  147 -- | Run an IO action and swallow its exception. Used to make the
  148 -- background session's save/close best-effort so a flush failure does not
  149 -- mask the turn's own result. Although this catches 'SomeException', it is
  150 -- only ever invoked inside 'finally's finalizer (which runs with async
  151 -- exceptions masked), so in practice it swallows synchronous IO failures
  152 -- from save/close — not an asynchronous 'AsyncCancelled'.
  153 ignoreExc :: IO () -> IO ()
  154 ignoreExc m = m `catch` \(_ :: SomeException) -> pure ()

  156 -- | The effective tool registry for a background turn: built-in tools
  157 -- merged with any connected MCP server tools.
  158 backgroundRegistry :: AgentEnv -> IO ToolRegistry
  159 backgroundRegistry env = do
  160   servers <- readIORef (_env_mcpServers env)
  161   let base = _env_registry env
  162   pure $ if Map.null servers
  163            then base
  164            else mergeRegistries base (mcpRegistry (Map.elems servers))
