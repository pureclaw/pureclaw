   22 -- | Logging capability. Functions that only receive a 'LogHandle' cannot
   23 -- shell out, read files, or access the network — they can only log.
   24 data LogHandle = LogHandle
   25   { _lh_logInfo  :: Text -> IO ()
   26   , _lh_logWarn  :: Text -> IO ()
   27   , _lh_logError :: Text -> IO ()
   28   , _lh_logDebug :: Text -> IO ()
   29   }
   30 
   31 -- | Minimum severity threshold for logging. Constructor order encodes
   32 -- severity, so @LlDebug < LlInfo < LlWarn < LlError@ and filtering is a
   33 -- comparison. Constructors are prefixed to avoid clashing with unqualified
   34 -- names like 'Data.Aeson.Error' at import sites.
   35 data LogLevel = LlDebug | LlInfo | LlWarn | LlError
   36   deriving stock (Show, Eq, Ord, Bounded, Enum)
   37 
   38 -- | Parse a log level from a CLI string. Case-insensitive; 'Nothing' for
   39 -- anything that is not one of @debug@, @info@, @warn@, @error@.
   40 parseLogLevel :: String -> Maybe LogLevel
   41 parseLogLevel s = case map Char.toLower s of
   42   "debug" -> Just LlDebug
   43   "info"  -> Just LlInfo
   44   "warn"  -> Just LlWarn
   45   "error" -> Just LlError
   46   _       -> Nothing
   47 
   48 -- | Should a message at the given level be emitted under the configured
   49 -- threshold? True when the message is at least as severe as the threshold.
   50 shouldLog :: LogLevel -> LogLevel -> Bool
   51 shouldLog threshold msgLevel = msgLevel >= threshold
   52 
   53 -- | Log to stderr with ISO 8601 timestamps and level prefixes, emitting every
   54 -- level. Equivalent to @'mkStderrLogHandleAt' 'LlInfo'@.
   55 mkStderrLogHandle :: IO LogHandle
   56 mkStderrLogHandle = mkStderrLogHandleAt LlInfo
   57 
   58 -- | Like 'mkStderrLogHandle', but only emits messages at or above the given
   59 -- threshold. Below-threshold calls are silent no-ops. Uses an 'MVar' to
   60 -- serialize writes so concurrent threads don't interleave.
   61 mkStderrLogHandleAt :: LogLevel -> IO LogHandle
   62 mkStderrLogHandleAt threshold = do
   63   lock <- newMVar ()
   64   let logWithLevel :: LogLevel -> Text -> Text -> IO ()
   65       logWithLevel msgLevel label msg
   66         | shouldLog threshold msgLevel = withMVar lock $ \() -> do
   67             now <- getCurrentTime
   68             let timestamp = T.pack (formatTime defaultTimeLocale "%Y-%m-%dT%H:%M:%S" now)
   69             TIO.hPutStrLn stderr $ "[" <> timestamp <> "] [" <> label <> "] " <> msg
   70         | otherwise = pure ()
   71   pure LogHandle
   72     { _lh_logInfo  = logWithLevel LlInfo  "INFO"
   73     , _lh_logWarn  = logWithLevel LlWarn  "WARN"
   74     , _lh_logError = logWithLevel LlError "ERROR"
   75     , _lh_logDebug = logWithLevel LlDebug "DEBUG"
   76     }
   77 
   78 -- | No-op log handle. All operations silently succeed.
   79 mkNoOpLogHandle :: LogHandle
   80 mkNoOpLogHandle = LogHandle
   81   { _lh_logInfo  = \_ -> pure ()
   82   , _lh_logWarn  = \_ -> pure ()
   83   , _lh_logError = \_ -> pure ()
   84   , _lh_logDebug = \_ -> pure ()
   85   }
