   21 -- | Create a search_files tool backed by ripgrep.
   22 -- This tool does NOT go through SecurityPolicy because it is a read-only
   23 -- internal capability — the agent cannot use it to execute arbitrary commands.
   24 searchFilesTool :: WorkspaceRoot -> (ToolDefinition, ToolHandler)
   25 searchFilesTool (WorkspaceRoot root) = (def, handler)
   26   where
   27     def = ToolDefinition
   28       { _td_name        = "search_files"
   29       , _td_description = T.unlines
   30           [ "Search file contents or file names using ripgrep."
   31           , "Modes:"
   32           , "  content (default) — regex search over file contents, returns matching lines with line numbers"
   33           , "  files_only — return only file paths matching the pattern or glob"
   34           , "  count — return match counts per file"
   35           ]
   36       , _td_inputSchema = object
   37           [ "type" .= ("object" :: Text)
   38           , "properties" .= object
   39               [ "pattern" .= object
   40                   [ "type" .= ("string" :: Text)
   41                   , "description" .= ("Regex pattern to search for (ripgrep syntax)" :: Text)
   42                   ]
   43               , "path" .= object
   44                   [ "type" .= ("string" :: Text)
   45                   , "description" .= ("Directory or file to search in, relative to workspace root (default: \".\")" :: Text)
   46                   ]
   47               , "file_glob" .= object
   48                   [ "type" .= ("string" :: Text)
   49                   , "description" .= ("Glob filter for file names, e.g. \"*.hs\" or \"*.{ts,tsx}\"" :: Text)
   50                   ]
   51               , "output_mode" .= object
   52                   [ "type" .= ("string" :: Text)
   53                   , "enum" .= (["content", "files_only", "count"] :: [Text])
   54                   , "description" .= ("Output mode (default: content)" :: Text)
   55                   ]
   56               , "context" .= object
   57                   [ "type" .= ("integer" :: Text)
   58                   , "description" .= ("Number of context lines before and after each match (default: 0)" :: Text)
   59                   ]
   60               , "limit" .= object
   61                   [ "type" .= ("integer" :: Text)
   62                   , "description" .= ("Maximum number of matches to return (default: 50)" :: Text)
   63                   ]
   64               , "case_insensitive" .= object
   65                   [ "type" .= ("boolean" :: Text)
   66                   , "description" .= ("Case-insensitive search (default: false)" :: Text)
   67                   ]
   68               ]
   69           , "required" .= (["pattern"] :: [Text])
   70           ]
   71       }
   72 
   73     handler = ToolHandler $ \input ->
   74       case parseEither parseInput input of
   75         Left err -> pure (T.pack err, True)
   76         Right si -> runSearch si
   77 
   78     runSearch :: SearchInput -> IO (Text, Bool)
   79     runSearch si = do
   80       let args = buildRgArgs si
   81           config = P.setWorkingDir root
   82                  $ P.proc "rg" args
   83       result <- try @IOException $ P.readProcess config
   84       case result of
   85         Left e
   86           | isDoesNotExistError e ->
   87               pure ("search_files requires ripgrep (rg) but it is not installed or not on PATH", True)
   88           | otherwise ->
   89               pure ("search_files error: " <> T.pack (show e), True)
   90         Right (exitCode, outLazy, errLazy) ->
   91           let out = TE.decodeUtf8Lenient (BL.toStrict outLazy)
   92               err = TE.decodeUtf8Lenient (BL.toStrict errLazy)
   93           in case exitCode of
   94             ExitSuccess   -> pure (truncateOutput out, False)
   95             ExitFailure 1 -> pure ("No matches found", False)  -- rg returns 1 for no matches
   96             ExitFailure _ -> pure ("search_files error: " <> err, True)
   97 
   98     truncateOutput :: Text -> Text
   99     truncateOutput t
  100       | T.length t > 100000 =
  101           T.take 100000 t <> "\n[...output truncated at 100000 chars]"
  102       | otherwise = t
  103 
  104     buildRgArgs :: SearchInput -> [String]
  105     buildRgArgs si =
  106       let modeArgs = case _si_outputMode si of
  107             "files_only" -> ["--files-with-matches"]
  108             "count"      -> ["--count"]
  109             _            -> ["--line-number"]  -- content mode (default)
  110           globArgs = case _si_fileGlob si of
  111             Nothing -> []
  112             Just g  -> ["--glob", T.unpack g]
  113           ctxArgs = case _si_context si of
  114             Nothing -> []
  115             Just n  -> ["--context", show n]
  116           limitArgs = ["--max-count", show (_si_limit si)]
  117           caseArgs = ["--ignore-case" | _si_caseInsensitive si]
  118           searchPath = T.unpack (_si_path si)
  119       in concat
  120           [ ["--no-heading", "--color", "never"]
  121           , modeArgs
  122           , globArgs
  123           , ctxArgs
  124           , limitArgs
  125           , caseArgs
  126           , ["--", T.unpack (_si_pattern si), searchPath]
  127           ]
  128 
  129 data SearchInput = SearchInput
  130   { _si_pattern         :: Text
  131   , _si_path            :: Text
  132   , _si_fileGlob        :: Maybe Text
  133   , _si_outputMode      :: Text
  134   , _si_context         :: Maybe Int
  135   , _si_limit           :: Int
  136   , _si_caseInsensitive :: Bool
  137   }
  138 
  139 parseInput :: Value -> Parser SearchInput
  140 parseInput = withObject "SearchFilesInput" $ \o ->
  141   SearchInput
  142     <$> o .:  "pattern"
  143     <*> o .:? "path"             .!= "."
  144     <*> o .:? "file_glob"
  145     <*> o .:? "output_mode"      .!= "content"
  146     <*> o .:? "context"
  147     <*> o .:? "limit"            .!= 50
  148     <*> o .:? "case_insensitive" .!= False
