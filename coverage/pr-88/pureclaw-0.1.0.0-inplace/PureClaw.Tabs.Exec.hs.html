    3 -- |
    4 -- Module      : PureClaw.Tabs.Exec
    5 -- Description : Refcounted per-'TabRef' runtime registry (8b.3b).
    6 --
    7 -- The Tabs-as-View cutover (GitHub #79) moves the execution unit from
    8 -- /per-tab/ to /per-ground-truth-ref/ (spike §3): a single live 'Runtime'
    9 -- drives each bound 'TabRef', shared by every tab/conversation that resolves
   10 -- it. This module owns that mapping as a refcounted registry — the SAME
   11 -- refcount discipline as "PureClaw.Tabs.SessionPool", only it holds started
   12 -- 'Runtime' values instead of @SessionHandle@s:
   13 --
   14 --   * 'ensure' starts a runtime (via the injected '_ex_startRuntime') on the
   15 --     FIRST bind of a ref and bumps the refcount on every subsequent bind.
   16 --   * 'release' decrements; on the LAST release it '_rt_stop's the runtime
   17 --     and removes the entry. Releasing an absent / already-zero ref is a safe
   18 --     no-op.
   19 --   * 'sendTo' routes one input to a ref's runtime via '_rt_send'; an absent
   20 --     ref returns @'Left' 'TabNotFound'@ (no such runtime) rather than
   21 --     throwing.
   22 --   * 'stopAll' stops every live runtime and clears the registry.
   23 --
   24 -- == Injected construction (8c wires the real bodies)
   25 --
   26 -- Runtime CONSTRUCTION is injected through 'ExecDeps' ('_ex_startRuntime'),
   27 -- so this module is fully unit-testable with a fake. The real provider /
   28 -- harness runtime bodies (enqueue-to-worker for providers, @_hh_send@ for
   29 -- harnesses) are built later in 8c against the live @AgentEnv@; this module
   30 -- only owns the refcount + lifecycle machinery and never performs real
   31 -- provider / tmux IO itself.
   32 --
   33 -- == Concurrency
   34 --
   35 -- The internal @'Data.Map.Strict.Map' 'TabRef' (Int, 'Runtime')@ (refcount +
   36 -- runtime) lives behind an 'IORef' mutated only with 'atomicModifyIORef''.
   37 -- Crucially, /no IO runs inside the atomic section/: starting and stopping a
   38 -- runtime are real side effects (forking workers / drainers, cancelling
   39 -- threads), so they are performed __outside__ 'atomicModifyIORef''. This
   40 -- mirrors 'PureClaw.Tabs.SessionPool' exactly:
   41 --
   42 --   * 'ensure' uses a bump-if-present, else start-outside-the-lock,
   43 --     then-commit-if-still-absent pattern. On a lost race it adopts the
   44 --     winner's runtime and stops its own redundant one.
   45 --   * 'release' atomically decrements; the orphaned runtime (at refcount 0)
   46 --     is captured and stopped outside the lock.
   47 --   * 'sendTo' atomically reads the runtime out (no IO in the atomic
   48 --     section), then calls '_rt_send' outside it.
   49 --   * 'stopAll' atomically swaps the map for empty and stops every captured
   50 --     runtime outside the lock.
   51 --
   52 -- See the Tabs-as-View execution-binding spike (§3/§6) for the design.
   53 module PureClaw.Tabs.Exec
   54   ( -- * Started runtime
   55     Runtime (..)
   56     -- * Injected IO seam
   57   , ExecDeps (..)
   58     -- * The registry
   59   , Exec (..)
   60   , newExec
   61     -- * Lifecycle / routing
   62   , ensure
   63   , release
   64   , sendTo
   65   , stopAll
   66   ) where
   67 
   68 import Data.IORef (IORef, atomicModifyIORef', newIORef)
   69 import Data.Map.Strict (Map)
   70 import Data.Map.Strict qualified as Map
   71 import Data.Text (Text)

   73 import PureClaw.Handles.Tab (TabError (..))

   76 -- | A started runtime: how to feed it input and how to stop it.
   77 --
   78 -- For a provider ref, '_rt_send' enqueues a user message onto the worker
   79 -- queue (the worker serializes turns); for a harness ref it forwards to
   80 -- @_hh_send@. '_rt_stop' is idempotent (cancels worker / drainer threads)
   81 -- and never throws. Both bodies are injected by '_ex_startRuntime' — this
   82 -- module never constructs them itself.
   83 data Runtime = Runtime
   84   { _rt_send :: Text -> IO (Either TabError ())
   85     -- ^ Route one input to this runtime.
   86   , _rt_stop :: IO ()
   87     -- ^ Idempotent stop; never throws.
   88   }
   89 
   90 -- | The injected runtime-construction seam. The real implementation builds +
   91 -- starts a provider or harness runtime for a ref against the live
   92 -- @AgentEnv@ (wired in 8c); tests supply a recording fake so the registry's
   93 -- refcount machinery can be driven without a live LLM or tmux.
   94 newtype ExecDeps = ExecDeps
   95   { _ex_startRuntime :: TabRef -> IO Runtime
   96     -- ^ Build + start a runtime for a ref. Called at most once per live ref
   97     -- — on the FIRST 'ensure' (and once more on the losing side of a start
   98     -- race, whose redundant runtime is then stopped).
   99   }
  100 
  101 -- | The refcounted registry: a map from each live 'TabRef' to its current
  102 -- refcount and the single shared 'Runtime'. A ref is present in the map iff
  103 -- its refcount is @>= 1@; entries are removed when the count reaches zero, so
  104 -- absence is indistinguishable from "never ensured".
  105 newtype Exec = Exec (IORef (Map TabRef (Int, Runtime)))
  106 
  107 -- | Create a fresh, empty registry.
  108 newExec :: IO Exec
  109 newExec = Exec <$> newIORef Map.empty
  110 
  111 -- | Ensure a runtime is live for a 'TabRef': start it on the first 'ensure'
  112 -- and bump the refcount on every subsequent one. Two ensures of the same ref
  113 -- always share the SAME runtime.
  114 --
  115 -- The start (a real side effect — forking workers / drainers) is performed
  116 -- outside the atomic section: we first try to bump an existing entry
  117 -- atomically; only if the ref is absent do we start a new runtime and then
  118 -- atomically commit it, stopping the redundant runtime if a concurrent
  119 -- ensure won the race.
  120 ensure :: ExecDeps -> Exec -> TabRef -> IO ()
  121 ensure deps (Exec ref) tref = do
  122   -- Step 1: fast path — bump an existing entry's refcount atomically.
  123   present <- atomicModifyIORef' ref $ \m ->
  124     case Map.lookup tref m of
  125       Just (n, rt) -> (bump tref n rt m, True)
  126       Nothing      -> (m, False)
  127   if present
  128     then pure ()
  129     else do
  130       -- Step 2: ref absent — start OUTSIDE the atomic section.
  131       fresh <- _ex_startRuntime deps tref
  132       -- Step 3: commit only if still absent; otherwise adopt the winner's
  133       -- runtime (bumping it) and stop our now-redundant fresh one.
  134       redundant <- atomicModifyIORef' ref $ \m ->
  135         case Map.lookup tref m of
  136           Just (n, rt) -> (bump tref n rt m, Just fresh)
  137           Nothing      -> (bump tref 0 fresh m, Nothing)
  138       mapM_ _rt_stop redundant
  139 
  140 -- | Strictly store @tref -> (n + 1, rt)@. The refcount and runtime are forced
  141 -- (bang patterns) before insertion so the registry never accumulates a chain
  142 -- of refcount thunks and the stored runtime is always in WHNF. Used both to
  143 -- add a brand-new entry (@n = 0@) and to increment an existing one.
  144 bump
  145   :: TabRef
  146   -> Int
  147   -> Runtime
  148   -> Map TabRef (Int, Runtime)
  149   -> Map TabRef (Int, Runtime)
  150 bump tref !n !rt = Map.insert tref (n + 1, rt)
  151 
  152 -- | Release one hold on a 'TabRef'. Decrements its refcount; when the count
  153 -- reaches zero the entry is removed and its runtime stopped. Releasing a ref
  154 -- that is absent (never ensured, or already fully released) is a safe no-op:
  155 -- nothing is decremented and nothing is stopped.
  156 --
  157 -- The stop (a real side effect) is performed outside the atomic section.
  158 release :: Exec -> TabRef -> IO ()
  159 release (Exec ref) tref = do
  160   toStop <- atomicModifyIORef' ref $ \m ->
  161     case Map.lookup tref m of
  162       Nothing -> (m, Nothing)
  163       Just (n, rt)
  164         | n <= 1    -> (Map.delete tref m, Just rt)
  165         | otherwise -> (unbump tref n rt m, Nothing)
  166   mapM_ _rt_stop toStop
  167 
  168 -- | Strictly store @tref -> (n - 1, rt)@ for the not-last release. The
  169 -- decremented refcount and runtime are forced before insertion (same
  170 -- rationale as 'bump'). Only called on the @n > 1@ path, so the result is
  171 -- always @>= 1@.
  172 unbump
  173   :: TabRef
  174   -> Int
  175   -> Runtime
  176   -> Map TabRef (Int, Runtime)
  177   -> Map TabRef (Int, Runtime)
  178 unbump tref !n !rt = Map.insert tref (n - 1, rt)
  179 
  180 -- | Route one input to a ref's runtime. The runtime is read out of the map
  181 -- atomically (no IO in the atomic section); '_rt_send' is then called outside
  182 -- it. An absent ref (no live runtime) returns @'Left' ('TabNotFound' 0)@ —
  183 -- the "no such runtime" signal — rather than throwing, so the dispatcher can
  184 -- surface it as a user-visible error and keep routing.
  185 sendTo :: Exec -> TabRef -> Text -> IO (Either TabError ())
  186 sendTo (Exec ref) tref t = do
  187   mRt <- atomicModifyIORef' ref $ \m ->
  188     (m, fmap snd (Map.lookup tref m))
  189   case mRt of
  190     Nothing -> pure (Left (TabNotFound 0))
  191     Just rt -> _rt_send rt t
  192 
  193 -- | Stop every live runtime and clear the registry. The map is atomically
  194 -- swapped for empty (capturing all runtimes); each captured runtime is then
  195 -- '_rt_stop'ped outside the atomic section. Idempotent: a second 'stopAll'
  196 -- on the now-empty registry stops nothing.
  197 stopAll :: Exec -> IO ()
  198 stopAll (Exec ref) = do
  199   runtimes <- atomicModifyIORef' ref $ \m ->
  200     (Map.empty, map snd (Map.elems m))
  201   mapM_ _rt_stop runtimes
