    1 -- |
    2 -- Module      : PureClaw.Routing.TabDispatch
    3 -- Description : Per-conversation inbound router + flat tab commands (8b.4).
    4 --
    5 -- This is the per-conversation routing brain of the Tabs-as-View cutover
    6 -- (GitHub #79; spike §8). 'handleInbound' takes one inbound message tagged
    7 -- with its 'ConversationKey' and routes it against /this/ conversation's
    8 -- cursor and the global 'TabRegistry':
    9 --
   10 --   1. __Wizard interception__ — if the conversation is mid-@\/tab@, its next
   11 --      message is consumed by the wizard ('stepWizard') /before/ the grammar
   12 --      sees it (design §9.2 / §11). A completed pick binds a new tab; a
   13 --      slash-prefixed reply cancels the wizard and re-dispatches that command.
   14 --   2. __Flat tab verbs__ — @\/new@, @\/nt@, @\/close@, @\/tabs@, @\/rename@,
   15 --      @\/relay@, @\/tab@ — recognised directly on the trimmed text (design
   16 --      §7). These are __not__ routed through the legacy @\/tab \<sub\>@
   17 --      grammar.
   18 --   3. __Routing grammar__ — anything else goes through
   19 --      'PureClaw.Routing.Parse.parseInput': @\/N@ switch, @\/N \<text\>@
   20 --      inject, default text to the active tab, or a non-tab slash command
   21 --      handed to '_td_fallthrough'.
   22 --
   23 -- Every effect is injected through 'TabDispatchDeps' (no real provider /
   24 -- harness / Exec), so the whole router is unit-testable with recording fakes
   25 -- and the pinned design §14 copy is asserted byte-for-byte. This module is
   26 -- /additive/ — it is wired into the live loop at 8c, replacing the old
   27 -- @dispatchOne@.
   28 module PureClaw.Routing.TabDispatch
   29   ( -- * Injected dependencies
   30     TabDispatchDeps (..)
   31     -- * The router
   32   , handleInbound

  101 -- | Everything 'handleInbound' needs, injected so the router is testable
  102 -- against fakes (no live provider / harness / 'PureClaw.Tabs.Exec'). The
  103 -- runtime-binding seams ('_td_ensure' \/ '_td_release' \/ '_td_sendTo') are the
  104 -- 'PureClaw.Tabs.Exec' operations partially applied to the live registry at
  105 -- 8c; here they are plain functions a test can record.
  106 data TabDispatchDeps = TabDispatchDeps
  107   { _td_tabs    :: !TabRegistry
  108     -- ^ The global ordered tab registry (I1\/I2).
  109   , _td_cursors :: !(IORef CursorState)
  110     -- ^ Per-conversation cursors + relay overrides (I3).
  111   , _td_wizard  :: !(IORef (Map ConversationKey WizardState))
  112     -- ^ In-flight @\/tab@ wizard state, keyed by conversation (runtime-only).
  113   , _td_ensure  :: !(TabRef -> IO ())
  114     -- ^ Start (or ++refcount) the runtime backing a ref.
  115   , _td_release :: !(TabRef -> IO ())
  116     -- ^ --refcount; stop the runtime on last release.
  117   , _td_sendTo  :: !(TabRef -> Text -> IO (Either TabError ()))
  118     -- ^ Route input text to a ref's runtime.
  119   , _td_emit    :: !(ConversationKey -> Text -> IO ())
  120     -- ^ Reply / banner to a conversation's output sink.
  121   , _td_newDefaultSession :: !(IO (Either Text TabRef))
  122     -- ^ Create a fresh default-provider session, returning its 'BoundSession'
  123     --   ref. @'Left' msg@ when no default provider is configured.
  124   , _td_recentHarnesses :: !(IO [(HarnessId, Text)])
  125     -- ^ Running harnesses (id + label) for the @\/tab@ wizard menu.
  126   , _td_recentSessions  :: !(IO [(SessionId, Text)])
  127     -- ^ Recent sessions (id + label) for the @\/tab@ wizard menu.
  128   , _td_liveHarness     :: !(HarnessId -> IO Bool)
  129     -- ^ Liveness probe used by the wizard when a harness pick is resolved.
  130   , _td_relayDefault    :: !RelayMode
  131     -- ^ Global default relay mode (shown by @\/tabs@ / @\/relay@).
  132   , _td_routingConfig   :: !RT.RoutingConfig
  133     -- ^ Routing config for 'Parse.parseInput' (index bounds etc.).
  134   , _td_fallthrough     :: !(ConversationKey -> Slash.SlashCommand -> IO ())
  135     -- ^ Handler for non-tab slash commands (e.g. @\/bg@).
  136   , _td_onTabsChanged   :: !(IO ())

  142   , _td_spawnHarness    :: !(HarnessSpec -> IO (Either Text (TabRef, Text)))
  143     -- ^ Spawn a fresh harness, persist + link its session, and return the new

  146     -- 'PureClaw.Agent.Env._env_startHarness' in "PureClaw.Tabs.Wiring".

  149 -- ---------------------------------------------------------------------------
  150 -- Per-call context
  151 -- ---------------------------------------------------------------------------
  152 
  153 -- | The deps + the conversation being served, bundled so every helper takes a
  154 -- single context argument. This keeps each effect call a saturated 2-arg
  155 -- application (e.g. @emit ctx msg@) — passing @deps@ and @convKey@ as two
  156 -- separate arguments at each site leaves an argument-position tick box that
  157 -- HPC reports as un-entered even when the call runs.
  158 data Ctx = Ctx
  159   { _ctx_deps :: !TabDispatchDeps
  160   , _ctx_conv :: !ConversationKey
  161   }

  163 -- ---------------------------------------------------------------------------
  164 -- Entry point
  165 -- ---------------------------------------------------------------------------
  166 
  167 -- | Route one inbound message for a conversation. See the module haddock for
  168 -- the three-stage order (wizard → flat verbs → routing grammar).
  169 handleInbound :: TabDispatchDeps -> ConversationKey -> Text -> IO ()
  170 handleInbound deps convKey raw = do
  171   let ctx = Ctx deps convKey
  172   wiz <- readIORef (_td_wizard deps)
  173   case Map.lookup convKey wiz of
  174     Just st -> handleWizardReply ctx st raw
  175     Nothing -> handleNonWizard ctx raw
  176 
  177 -- ---------------------------------------------------------------------------
  178 -- Stage 1: wizard interception
  179 -- ---------------------------------------------------------------------------
  180 
  181 -- | Feed a reply to an in-flight wizard and act on the resulting 'WizardStep'.
  182 handleWizardReply :: Ctx -> WizardState -> Text -> IO ()
  183 handleWizardReply ctx st raw = do
  184   (mNext, step) <- stepWizard (WizardEnv (_td_liveHarness (_ctx_deps ctx))) st raw
  185   case step of
  186     Done target -> do
  187       clearWizard ctx
  188       bindWizardTarget ctx target
  189     Cancelled -> do
  190       clearWizard ctx
  191       emit ctx "wizard cancelled"
  192     RunCommand cmd -> do
  193       clearWizard ctx
  194       handleNonWizard ctx cmd
  195     Reprompt msg -> do
  196       -- Invalid reply or vanished target: persist the (possibly refreshed)
  197       -- state and re-show the notice. 'Prompt' is the open-time render and is
  198       -- never produced by 'stepWizard'; it is folded in here so the match is
  199       -- total — both carry a notice, both re-prompt.
  200       setWizard ctx mNext
  201       emit ctx msg
  202     Prompt msg -> do
  203       setWizard ctx mNext
  204       emit ctx msg
  205 
  206 -- | Bind a completed wizard pick to a new tab: append (or switch to an
  207 -- already-bound ref), ensure its runtime, focus the conversation, confirm.
  208 bindWizardTarget :: Ctx -> WizardTarget -> IO ()
  209 bindWizardTarget ctx target =
  210   bindNewTab ctx ref (targetLabel target) "attached"
  211   where
  212     ref = case target of
  213       AttachHarness h -> BoundHarness h
  214       ReopenSession s -> BoundSession s
  215 
  216 -- | A terse default label for a freshly-attached wizard target.
  217 targetLabel :: WizardTarget -> Text
  218 targetLabel (AttachHarness _) = "harness"
  219 targetLabel (ReopenSession _) = "session"
  220 
  221 -- ---------------------------------------------------------------------------
  222 -- Stage 2 + 3: non-wizard dispatch
  223 -- ---------------------------------------------------------------------------
  224 
  225 -- | Recognise flat tab verbs on the trimmed text; otherwise fall to the
  226 -- routing grammar.
  227 handleNonWizard :: Ctx -> Text -> IO ()
  228 handleNonWizard ctx raw =
  229   case T.words trimmed of
  230     ("/new"    : _)    -> cmdNew    ctx
  231     ("/nt"     : _)    -> cmdNt     ctx
  232     ("/close"  : args) -> cmdClose  ctx args
  233     ("/tabs"   : _)    -> cmdTabs   ctx
  234     ("/rename" : args) -> cmdRename ctx args
  235     ("/relay"  : args) -> cmdRelay  ctx args
  236     ("/tab" : "new" : rest)    -> cmdTabNew ctx rest
  237     ("/tab" : "close" : rest)  -> cmdClose  ctx rest
  238     ("/tab" : "rename" : rest) -> cmdRename ctx rest
  239     ("/tab"    : args) -> cmdTab    ctx args
  240     _                  -> routeGrammar ctx raw
  241   where
  242     trimmed = T.strip raw
  243 
  244 -- ---------------------------------------------------------------------------
  245 -- /new — reset the active tab
  246 -- ---------------------------------------------------------------------------
  247 
  248 -- | @\/new@: reset the conversation's __active__ tab to a fresh default
  249 -- session (same slot). With no active tab, behaves like @\/nt@.
  250 cmdNew :: Ctx -> IO ()
  251 cmdNew ctx = do
  252   cs <- readIORef (_td_cursors (_ctx_deps ctx))
  253   tl <- readTabs (_td_tabs (_ctx_deps ctx))
  254   case resolveCursorSlot (_ctx_conv ctx) cs tl of
  255     Nothing   -> cmdNt ctx   -- no active tab → create one
  256     Just slot -> resetActiveTab ctx slot
  257 
  258 -- | Reset the tab at @slot@: mint a fresh session, rebind the slot to it
  259 -- (releasing the old ref's runtime, ensuring the new), keep the slot, point
  260 -- the cursor at the new ref, confirm. The previous session persists on disk
  261 -- (I4) — only the tab's binding moves.
  262 resetActiveTab :: Ctx -> TabIndex -> IO ()
  263 resetActiveTab ctx slot = do
  264   mNew <- _td_newDefaultSession (_ctx_deps ctx)
  265   case mNew of
  266     Left msg     -> emit ctx msg
  267     Right newRef -> do
  268       -- The slot is present (the caller resolved it from a live cursor) and
  269       -- @newRef@ is freshly minted, so it is never already bound: 'rebindSlot'
  270       -- always succeeds and 'registryLookupSlot' is always 'Just' here. The
  271       -- structurally-impossible branches are folded into 'error' (mirroring
  272       -- "PureClaw.Routing.Parse"'s 'boundedTabIndex') so the HPC report has no
  273       -- dead defensive arm to flag.
  274       mOld <- registryLookupSlot (_td_tabs (_ctx_deps ctx)) slot
  275       tl   <- readTabs (_td_tabs (_ctx_deps ctx))
  276       let oldRef = _tab_ref (Maybe.fromMaybe (impossible "resetActiveTab: slot vanished") mOld)
  277           tl'    = either (impossible "resetActiveTab: fresh ref already bound")
  278                           id
  279                           (rebindSlot slot newRef defaultSessionName tl)
  280       writeTabs ctx tl'
  281       release ctx oldRef
  282       ensure ctx newRef
  283       setCursorTo ctx newRef
  284       notifyChanged ctx
  285       emit ctx ("reset tab /" <> slotChar slot)
  286 
  287 -- ---------------------------------------------------------------------------
  288 -- /nt — new tab with a fresh default session
  289 -- ---------------------------------------------------------------------------
  290 
  291 -- | @\/nt@: create a fresh default session, append a tab at the next slot,
  292 -- ensure its runtime, switch the conversation's cursor to it.
  293 cmdNt :: Ctx -> IO ()
  294 cmdNt ctx = do
  295   mNew <- _td_newDefaultSession (_ctx_deps ctx)
  296   case mNew of
  297     Left msg     -> emit ctx msg
  298     Right newRef -> bindNewTab ctx newRef defaultSessionName "new tab"
  299 
  300 -- ---------------------------------------------------------------------------
  301 -- Shared: append-or-switch a new tab + confirm
  302 -- ---------------------------------------------------------------------------
  303 
  304 -- | Append a tab binding @ref@ (label @name@), ensure its runtime, focus the
  305 -- conversation on it, and emit @\<verb\> \/\<slot\>@. If @ref@ is already
  306 -- bound (I2), switch to the existing tab instead of duplicating. Slot
  307 -- exhaustion emits the §14 copy with no state change.
  308 bindNewTab :: Ctx -> TabRef -> Text -> Text -> IO ()
  309 bindNewTab ctx ref name verb = do
  310   res <- registryAppend (_td_tabs (_ctx_deps ctx)) ref name
  311   case res of
  312     Left SlotsFull          -> emit ctx slotsFullMsg
  313     Left (AlreadyBound cur) -> do
  314       -- Ref already bound (I2): switch to the existing tab, do not duplicate.
  315       -- The error payload IS the current slot, so no re-lookup is needed.
  316       setCursorTo ctx ref
  317       emit ctx ("switched to /" <> slotChar cur)
  318     Right slot              -> do
  319       ensure ctx ref
  320       setCursorTo ctx ref
  321       notifyChanged ctx
  322       emit ctx (verb <> " /" <> slotChar slot)
  323 
  324 -- ---------------------------------------------------------------------------
  325 -- /close [N]
  326 -- ---------------------------------------------------------------------------
  327 
  328 -- | @\/close [N]@: drop a tab view (never harms ground truth). @--force@ is
  329 -- rejected with the §14 copy; @N@ (or the active tab) is removed + compacted
  330 -- and its runtime released.
  331 cmdClose :: Ctx -> [Text] -> IO ()
  332 cmdClose ctx args = case args of
  333   ("--force" : _) -> emit ctx closeForceMsg
  334   (a : _)         -> case slotArg a of
  335     Just slot -> closeSlot ctx slot
  336     Nothing   -> emit ctx closeBadArgMsg
  337   []              -> do
  338     cs <- readIORef (_td_cursors (_ctx_deps ctx))
  339     tl <- readTabs (_td_tabs (_ctx_deps ctx))
  340     case resolveCursorSlot (_ctx_conv ctx) cs tl of
  341       Just slot -> closeSlot ctx slot
  342       Nothing   -> emit ctx closeNoTargetMsg
  343 
  344 -- | Remove the tab at @slot@ (compacting), release its runtime, and clear the
  345 -- cursors that pointed at it (compaction renumbers others automatically).
  346 closeSlot :: Ctx -> TabIndex -> IO ()
  347 closeSlot ctx slot = do
  348   mTab <- registryLookupSlot (_td_tabs (_ctx_deps ctx)) slot
  349   case mTab of
  350     Nothing  -> emit ctx closeNoTargetMsg
  351     Just tab -> do
  352       let ref = _tab_ref tab
  353       cs <- readIORef (_td_cursors (_ctx_deps ctx))
  354       registryRemove (_td_tabs (_ctx_deps ctx)) slot
  355       release ctx ref
  356       -- Drop every cursor focused on the removed ref (this conversation + any
  357       -- others sharing the tab); valid cursors survive via TabRef identity.
  358       mapM_ (modifyCursor ctx . clearCursor) (conversationsOn ref cs)
  359       notifyChanged ctx
  360       emit ctx ("closed /" <> slotChar slot)
  361 
  362 -- ---------------------------------------------------------------------------
  363 -- /tabs
  364 -- ---------------------------------------------------------------------------
  365 
  366 -- | @\/tabs@: list every tab (@\/\<slot\>  \<name\>  \<kind\>  \<status\>@,
  367 -- dead tombstones included) plus a final line naming this conversation's
  368 -- relay mode.
  369 cmdTabs :: Ctx -> IO ()
  370 cmdTabs ctx = do
  371   tl <- readTabs (_td_tabs (_ctx_deps ctx))
  372   cs <- readIORef (_td_cursors (_ctx_deps ctx))
  373   let focused = resolveCursorSlot (_ctx_conv ctx) cs tl
  374       rows    = map (tabRow focused) (toList tl)
  375       relay   = relayLine (relayModeFor ctx cs)
  376   emit ctx (T.intercalate "\n" (rows ++ [relay]))
  377 
  378 -- | Render one @\/tabs@ row. The row for @focused@ (this conversation's cursor
  379 -- slot, if any) carries a trailing @(focused)@ marker so the user can see at a
  380 -- glance where plain text will go.
  381 tabRow :: Maybe TabIndex -> Tab -> Text
  382 tabRow focused t =
  383   "/" <> slotChar (_tab_slot t)
  384     <> "  " <> _tab_name t
  385     <> "  " <> kindLabel (_tab_ref t)
  386     <> "  " <> statusLabel (_tab_status t)
  387     <> (if focused == Just (_tab_slot t) then "  (focused)" else "")
  388 
  389 -- | The trailing relay-mode line shown by @\/tabs@. Phrased as a full sentence
  390 -- so a first-time user understands which tabs' output reaches them without
  391 -- having to learn the @relay@ jargon.
  392 relayLine :: RelayMode -> Text
  393 relayLine FocusedOnly    = "Relay mode: focused — you only see output from this tab."
  394 relayLine ActivityDigest = "Relay mode: activity — this tab in full, plus activity pings from others."
  395 relayLine Firehose       = "Relay mode: all — full output from every tab."
  396 
  397 -- ---------------------------------------------------------------------------
  398 -- /rename [N] <name>
  399 -- ---------------------------------------------------------------------------
  400 
  401 -- | @\/rename [N] \<name\>@: relabel the tab at @N@ (or the active tab) using
  402 -- 'Parse.sanitizeTabName'.
  403 cmdRename :: Ctx -> [Text] -> IO ()
  404 cmdRename ctx args = case args of
  405   (a : rest@(_ : _))
  406     | Just slot <- slotArg a -> doRename ctx slot (T.unwords rest)
  407   _ -> do
  408     -- No leading slot arg: rename the active tab to the whole argument blob.
  409     cs <- readIORef (_td_cursors (_ctx_deps ctx))
  410     tl <- readTabs (_td_tabs (_ctx_deps ctx))
  411     case resolveCursorSlot (_ctx_conv ctx) cs tl of
  412       Just slot -> doRename ctx slot (T.unwords args)
  413       Nothing   -> emit ctx renameNoTargetMsg
  414 
  415 -- | Sanitize @name@ and rewrite the tab at @slot@'s label in place.
  416 doRename :: Ctx -> TabIndex -> Text -> IO ()
  417 doRename ctx slot name =
  418   case Parse.sanitizeTabName name of
  419     Left e      -> emit ctx (renameBadNameMsg e)
  420     Right clean -> do
  421       mTab <- registryLookupSlot (_td_tabs (_ctx_deps ctx)) slot
  422       case mTab of
  423         Nothing  -> emit ctx renameNoTargetMsg
  424         Just tab -> do
  425           -- Rebinding a slot to the ref it already holds always succeeds
  426           -- (relabel in place); the impossible 'Left' is folded into 'error'.
  427           tl <- readTabs (_td_tabs (_ctx_deps ctx))
  428           let tl' = either (impossible "doRename: in-place rebind rejected")
  429                            id
  430                            (rebindSlot slot (_tab_ref tab) clean tl)
  431           writeTabs ctx tl'
  432           notifyChanged ctx
  433           emit ctx ("renamed /" <> slotChar slot <> " " <> clean)
  434 
  435 -- ---------------------------------------------------------------------------
  436 -- /relay <mode>
  437 -- ---------------------------------------------------------------------------
  438 
  439 -- | @\/relay \<mode\>@: set this conversation's relay mode
  440 -- (@focused@\/@activity@\/@all@). No arg shows the current mode (§14 copy).
  441 cmdRelay :: Ctx -> [Text] -> IO ()
  442 cmdRelay ctx args = case args of
  443   []      -> do
  444     cs <- readIORef (_td_cursors (_ctx_deps ctx))
  445     emit ctx (relayCurrentMsg (relayModeFor ctx cs))
  446   (a : _) -> case parseRelayMode a of
  447     Just m  -> do
  448       modifyCursor ctx (setRelay (_ctx_conv ctx) m)
  449       emit ctx ("relay mode: " <> relayWord m)
  450     Nothing -> emit ctx relayBadModeMsg

  452 -- ---------------------------------------------------------------------------
  453 -- /tab — open the attach wizard
  454 -- ---------------------------------------------------------------------------
  455 
  456 -- | @\/tab [query]@: snapshot the running harnesses + recent sessions
  457 -- (filtered by @query@), store a 'WizardState' for the conversation, and emit
  458 -- the rendered menu.
  459 cmdTab :: Ctx -> [Text] -> IO ()
  460 cmdTab ctx args = do
  461   harnesses <- _td_recentHarnesses (_ctx_deps ctx)
  462   sessions  <- _td_recentSessions (_ctx_deps ctx)
  463   let query = T.unwords args
  464       hs    = filterCandidates query harnesses
  465       ss    = filterCandidates query sessions
  466       st    = mkWizardSnapshot hs ss
  467   setWizard ctx (Just st)
  468   emit ctx (renderMenu st)

  470 -- ---------------------------------------------------------------------------
  471 -- /tab new <kind> — create a tab of the named kind
  472 -- ---------------------------------------------------------------------------
  473 
  474 -- | @\/tab new [\<kind\>]@: create a new tab of the requested kind. This is
  475 -- distinct from bare @\/tab@ (the attach wizard) and is dispatched /before/ it,
  476 -- so @\/tab new@ never opens the wizard.
  477 --
  478 -- The default-provider session kinds (@ai@\/@provider@\/bare) mint a fresh
  479 -- session; @harness@ (optionally with an explicit flavour, e.g.
  480 -- @\/tab new harness claude-code@) spawns a real harness via '_td_spawnHarness'
  481 -- (WU-B). The remaining shell-family kinds emit a "not yet supported"
  482 -- placeholder; an unrecognised keyword emits a short usage hint.
  483 cmdTabNew :: Ctx -> [Text] -> IO ()
  484 cmdTabNew ctx rest = case kindKeyword of
  485   -- bare / "ai" / "provider": mint a default-provider session (mirrors 'cmdNt').
  486   Nothing         -> mintDefault
  487   Just "ai"       -> mintDefault
  488   Just "provider" -> mintDefault
  489   Just "harness"  -> spawnHarness ctx flavourArgs
  490   Just "shell"    -> emit ctx (tabNewUnsupportedMsg "shell")
  491   Just "ssh"      -> emit ctx (tabNewUnsupportedMsg "ssh")
  492   Just "tmux"     -> emit ctx (tabNewUnsupportedMsg "tmux")
  493   Just _          -> emit ctx tabNewUsageMsg
  494   where
  495     kindKeyword = case rest of
  496       (k : _) -> Just (T.toLower k)
  497       []      -> Nothing
  498     -- Args following the @harness@ keyword (the optional flavour word).
  499     flavourArgs = drop 1 rest
  500     mintDefault = do
  501       mNew <- _td_newDefaultSession (_ctx_deps ctx)
  502       case mNew of
  503         Left msg     -> emit ctx msg
  504         Right newRef -> bindNewTab ctx newRef defaultSessionName "new tab"
  505 
  506 -- | @\/tab new harness [\<flavour\>]@: build a default 'HarnessSpec' (flavour
  507 -- from the optional argument, defaulting to @claude-code@; local backend; no
  508 -- cwd\/args\/ids) and spawn it through '_td_spawnHarness'. On success bind the
  509 -- returned 'TabRef' into a new tab; on failure emit the error.
  510 spawnHarness :: Ctx -> [Text] -> IO ()
  511 spawnHarness ctx flavourArgs = do
  512   result <- _td_spawnHarness (_ctx_deps ctx) spec
  513   case result of
  514     Left msg          -> emit ctx msg
  515     Right (ref, label) -> bindNewTab ctx ref label "new harness"
  516   where
  517     flavour = case flavourArgs of
  518       (f : _) -> fixedFlavourLookup (T.toLower f)
  519       []      -> HClaudeCode
  520     spec = HarnessSpec
  521       { _h_flavour           = flavour
  522       , _h_backend           = TbLocal
  523       , _h_cwd               = Nothing
  524       , _h_args              = []
  525       , _h_harnessId         = Nothing
  526       , _h_claudeSessionUuid = Nothing
  527       , _h_canonicalCwd      = Nothing
  528       }

  530 -- ---------------------------------------------------------------------------
  531 -- Stage 3: routing grammar
  532 -- ---------------------------------------------------------------------------
  533 
  534 -- | Route via 'Parse.parseInput': switch, inject, default text, or non-tab
  535 -- slash command.
  536 routeGrammar :: Ctx -> Text -> IO ()
  537 routeGrammar ctx raw =
  538   case Parse.parseInput (_td_routingConfig (_ctx_deps ctx)) raw of
  539     Left _    -> emit ctx parseErrorMsg
  540     Right pin -> case pin of
  541       RT.Switch idx         -> doSwitch ctx idx
  542       RT.Inject idx text    -> doInject ctx idx text
  543       RT.Default text       -> doDefault ctx text
  544       RT.ParsedSlashCmd cmd -> _td_fallthrough (_ctx_deps ctx) (_ctx_conv ctx) cmd
  545 
  546 -- | @\/N@: focus the tab at slot @N@ (out-of-range → §14 copy).
  547 doSwitch :: Ctx -> TabIndex -> IO ()
  548 doSwitch ctx idx = withSlot ctx idx $ \tab -> do
  549   setCursorTo ctx (_tab_ref tab)
  550   emit ctx ("switched to /" <> slotChar idx)
  551 
  552 -- | @\/N \<text\>@: focus slot @N@ and route text to it (out-of-range → copy).
  553 doInject :: Ctx -> TabIndex -> Text -> IO ()
  554 doInject ctx idx text = withSlot ctx idx $ \tab -> do
  555   let ref = _tab_ref tab
  556   setCursorTo ctx ref
  557   deliver ctx ref text
  558 
  559 -- | Route @text@ to @ref@'s runtime, surfacing a delivery failure as the §14
  560 -- copy. Shared by 'doInject', 'doDefault', and the auto-start path so the
  561 -- send-then-handle-error shape lives in exactly one place.
  562 deliver :: Ctx -> TabRef -> Text -> IO ()
  563 deliver ctx ref text = do
  564   result <- sendTo ctx ref text
  565   case result of
  566     Right () -> pure ()
  567     Left err -> emit ctx ("couldn't deliver your message — " <> unPublicTabError (toPublicTabError err))
  568 
  569 -- | Resolve @idx@ to a present tab or emit the out-of-range §14 copy.
  570 withSlot :: Ctx -> TabIndex -> (Tab -> IO ()) -> IO ()
  571 withSlot ctx idx k = do
  572   tl   <- readTabs (_td_tabs (_ctx_deps ctx))
  573   mTab <- registryLookupSlot (_td_tabs (_ctx_deps ctx)) idx
  574   case mTab of
  575     Just tab -> k tab
  576     Nothing  -> emit ctx (outOfRangeMsg idx (length (toList tl)))
  577 
  578 -- | Plain text: route to the conversation's active tab. No active tab →
  579 -- auto-start a fresh default session and route the text to it (the implicit
  580 -- "just works" session — matches a fresh conversation's pre-Tabs behaviour); a
  581 -- 'Dead' tombstone → deferred death warning + clear + drop (§8).
  582 doDefault :: Ctx -> Text -> IO ()
  583 doDefault ctx text = do
  584   cs <- readIORef (_td_cursors (_ctx_deps ctx))
  585   tl <- readTabs (_td_tabs (_ctx_deps ctx))
  586   case resolveCursorSlot (_ctx_conv ctx) cs tl of
  587     Nothing   -> autoStartDefault ctx text
  588     Just slot -> do
  589       -- 'resolveCursorSlot' returns the slot by resolving the cursor's ref
  590       -- against the list, so the tab is guaranteed present here; the impossible
  591       -- absence is folded into 'error' (no dead defensive arm for HPC).
  592       tab <- Maybe.fromMaybe (impossible "doDefault: resolved slot vanished")
  593                <$> registryLookupSlot (_td_tabs (_ctx_deps ctx)) slot
  594       if _tab_status tab == Dead
  595         then do
  596           -- §8 deferred death warning: warn, clear the cursor, DROP the text
  597           -- (never route it into the dead harness).
  598           emit ctx (deferredDeathMsg (_tab_name tab))
  599           modifyCursor ctx (clearCursor (_ctx_conv ctx))
  600         else deliver ctx (_tab_ref tab) text
  601 
  602 -- | No active tab + plain text: mint a fresh default session, append a tab for
  603 -- it, focus the conversation on it, and route the text — restoring the implicit
  604 -- "just works" default session. No @new tab \/N@ confirmation is emitted (the
  605 -- user sent a chat message, not @\/nt@); the new tab still becomes visible via
  606 -- 'notifyChanged'. @'Left' msg@ (no provider\/model configured) surfaces the
  607 -- existing setup guidance, exactly like 'cmdNt'.
  608 autoStartDefault :: Ctx -> Text -> IO ()
  609 autoStartDefault ctx text = do
  610   mNew <- _td_newDefaultSession (_ctx_deps ctx)
  611   case mNew of
  612     Left msg     -> emit ctx msg
  613     Right ref    -> do
  614       res <- registryAppend (_td_tabs (_ctx_deps ctx)) ref defaultSessionName
  615       case res of
  616         -- Slot exhaustion: surface the §14 copy with no state change.
  617         Left SlotsFull -> emit ctx slotsFullMsg
  618         -- 'Right' (the freshly-bound slot) and 'Left (AlreadyBound _)' (which is
  619         -- structurally impossible for a freshly-minted ref) both route the text
  620         -- to the ref — focus + deliver — rather than crash on the latter.
  621         _              -> routeToNew ctx ref text
  622 
  623 -- | Focus the conversation on a freshly-bound @ref@, ensure its runtime, notify
  624 -- listeners, and route @text@ to it. Shared by the two reachable arms of
  625 -- 'autoStartDefault'.
  626 routeToNew :: Ctx -> TabRef -> Text -> IO ()
  627 routeToNew ctx ref text = do
  628   ensure ctx ref
  629   setCursorTo ctx ref
  630   notifyChanged ctx
  631   deliver ctx ref text

  702 -- | Parse a single @[0-9a-z]@ slot argument into a 'TabIndex'.
  703 slotArg :: Text -> Maybe TabIndex
  704 slotArg t = case T.unpack t of
  705   [c] -> tabIndexChar c >>= mkTabIndex
  706   _   -> Nothing
  707 
  708 -- | The @[0-9a-z]@ → index mapping (digits @0..9@, letters @10..35@).
  709 tabIndexChar :: Char -> Maybe Int
  710 tabIndexChar c
  711   | Char.isDigit c      = Just (fromEnum c - fromEnum '0')
  712   | Char.isAsciiLower c = Just (10 + fromEnum c - fromEnum 'a')
  713   | otherwise           = Nothing
  714 
  715 -- | Render a 'TabIndex' back to its single @[0-9a-z]@ display char.
  716 slotChar :: TabIndex -> Text
  717 slotChar ti =
  718   let n = unTabIndex ti
  719   in if n < 10
  720        then T.singleton (toEnum (fromEnum '0' + n))
  721        else T.singleton (toEnum (fromEnum 'a' + n - 10))
  722 
  723 -- | Parse a relay-mode word.

  756 -- | @\/N@ out of range (§14). @n@ = current tab count.
  757 outOfRangeMsg :: TabIndex -> Int -> Text
  758 outOfRangeMsg idx n =
  759   "/" <> slotChar idx <> ": out of range — you have " <> tshow n
  760     <> " tabs (/0–/" <> lastSlotChar n <> "); /tabs to list"
  761 
  762 -- | The high slot char of an @n@-tab list (@n-1@). For @n <= 0@ (an
  763 -- out-of-range switch against an empty list) the high coordinate is @0@. For
  764 -- @n >= 1@, @n-1 >= 0@ so 'mkTabIndex' always succeeds; the impossible
  765 -- 'Nothing' is folded into 'error'.
  766 lastSlotChar :: Int -> Text
  767 lastSlotChar n
  768   | n <= 0    = "0"
  769   | otherwise =
  770       slotChar (Maybe.fromMaybe (impossible "lastSlotChar: negative index")
  771                                 (mkTabIndex (n - 1)))
