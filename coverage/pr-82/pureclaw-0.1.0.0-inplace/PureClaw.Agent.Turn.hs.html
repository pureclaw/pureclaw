    3 -- |
    4 -- Module      : PureClaw.Agent.Turn
    5 -- Description : Env-light, single-turn provider execution with tool cycling.
    6 --
    7 -- == Tabs-as-View
    8 --
    9 -- This module is the env-light extraction of the tool-call cycle. It is the
   10 -- reusable, per-tab core that the tabbed loop ('runTabbedLoop' via the tab
   11 -- runtimes) drives, replacing the original env-coupled completion handler that
   12 -- wrote @_env_channel@ directly and tail-recursed into a singleton loop.
   13 --
   14 -- 'runTurnWithTools' performs the logical cycle — build request, stream,
   15 -- capture response, run any tool calls, re-complete until none remain — but
   16 -- over an injected 'TurnDeps' record. Every effect (the provider stream, the
   17 -- tool executor, the streaming sink, the transcript append, the 'StreamId'
   18 -- allocator) is a field, so the function unit-tests fully with fakes: no live
   19 -- provider, no 'PureClaw.Tools.Registry.ToolRegistry', no @AgentEnv@.
   20 --
   21 -- The streaming sink is a small 'TurnEvent' (start\/chunk\/end framed by a
   22 -- 'StreamId') that carries no 'PureClaw.Handles.Tab.TabIndex' or slot
   23 -- identity — the runtime layer (8b.3) maps 'TurnEvent' to
   24 -- 'PureClaw.Routing.Types.ChannelEvent'. Keeping that mapping out of here is
   25 -- what decouples 'PureClaw.Agent.Turn' from the slot\/relay machinery.
   26 module PureClaw.Agent.Turn
   27   ( -- * Output events
   28     TurnEvent (..)
   29     -- * Dependencies
   30   , TurnDeps (..)
   31     -- * Running a turn
   32   , runTurnWithTools
   33     -- * Stream identifiers (re-exported for convenience)
   34   , StreamId
   35   , mkStreamId
   36   ) where

   51 -- ---------------------------------------------------------------------------
   52 -- Output events
   53 -- ---------------------------------------------------------------------------
   54 
   55 -- | A single streaming-output event for one provider turn.
   56 --
   57 -- One logical streamed message is framed @'TurnStart' sid@, a sequence of
   58 -- @'TurnChunk' sid chunk@, then @'TurnEnd' sid@. The 'StreamId' is allocated
   59 -- once per provider completion (so a tool cycle that re-completes produces a
   60 -- fresh 'StreamId' per leg). Deliberately env-light: no slot\/tab identity —
   61 -- the runtime maps these to 'PureClaw.Routing.Types.ChannelEvent' in 8b.3.
   62 --
   63 -- @'TurnError' sid msg@ surfaces a provider failure as a visible message. The
   64 -- the original completion handler caught a throwing stream and sent the user a
   65 -- @TemporaryError "Something went wrong. Please try again."@; on the tabbed
   66 -- path the throw must likewise be surfaced (pureclaw-9d0) rather than silently
   67 -- swallowed, leaving the user with an empty 'TurnStart'\/'TurnEnd' pair.
   68 data TurnEvent
   69   = TurnStart !StreamId
   70   | TurnChunk !StreamId !Text
   71   | TurnEnd   !StreamId
   72   | TurnError !StreamId !Text
   73   deriving stock (Eq, Show)

   79 -- | The injected effects 'runTurnWithTools' needs. Production wires:
   80 --
   81 --   * '_turn_stream'       — @'P.completeStream' someProvider@ (wrapped with
   82 --                            the per-session transcript provider).
   83 --   * '_turn_execTool'     — a closure over the session's
   84 --                            'PureClaw.Tools.Registry.ToolRegistry' that runs
   85 --                            one tool call and returns the tool-result
   86 --                            'P.Message'.
   87 --   * '_turn_emit'         — the per-conversation streaming sink.
   88 --   * '_turn_record'       — the per-session transcript append.
   89 --   * '_turn_nextStreamId' — a monotonic 'StreamId' allocator.
   90 --
   91 -- Tests inject fakes for all of these.
   92 data TurnDeps = TurnDeps
   93   { _turn_stream       :: P.CompletionRequest -> (P.StreamEvent -> IO ()) -> IO ()
   94     -- ^ Injected provider stream (prod: @'P.completeStream' someProvider@).
   95   , _turn_execTool     :: ToolCallId -> Text -> Value -> IO P.Message
   96     -- ^ Injected tool exec — runs one call, yields a tool-result 'P.Message'.
   97   , _turn_emit         :: TurnEvent -> IO ()
   98     -- ^ Injected streaming sink.
   99   , _turn_record       :: P.Message -> IO ()
  100     -- ^ Injected transcript append (user, assistant, and tool-result msgs).
  101   , _turn_nextStreamId :: IO StreamId
  102     -- ^ Allocates a fresh 'StreamId' per completion leg.
  103   , _turn_model        :: ModelId
  104   , _turn_systemPrompt :: Maybe Text
  105     -- ^ Overrides the context's system prompt when @'Just'@; otherwise the
  106     -- request falls back to 'Ctx.contextSystemPrompt' (matching Loop).
  107   , _turn_tools        :: [P.ToolDefinition]
  108   , _turn_maxTokens    :: Maybe Int
  109   , _turn_onStreamDone :: IO ()
  110     -- ^ A one-shot-style hook fired on EVERY provider 'P.StreamDone' (in
  111     -- addition to capturing the response). Idempotency is the action's
  112     -- responsibility — production wires a read-and-clear @fireOnce@ over
  113     -- @_env_onFirstStreamDone@ so @'markBootstrapConsumed'@ runs exactly once,
  114     -- restoring the the original completion handler behaviour the tabbed path
  115     -- dropped (pureclaw-8g4). Tests default it to @pure ()@.
  116   }

  118 -- ---------------------------------------------------------------------------
  119 -- Running a turn
  120 -- ---------------------------------------------------------------------------
  121 
  122 -- | Run one user turn WITH the provider's tool-call cycle, env-light.
  123 --
  124 -- Mirrors the original completion handler but over 'TurnDeps':
  125 --
  126 --   1. Append a user 'P.Message' (the text) to the 'Context' and record it.
  127 --   2. Cycle: build a 'P.CompletionRequest' from the current context plus the
  128 --      injected model\/system\/tools\/maxTokens; allocate a 'StreamId'; emit
  129 --      'TurnStart'; stream — 'P.StreamText' becomes a 'TurnChunk',
  130 --      'P.StreamWarning' is ignored, 'P.StreamDone' captures the response;
  131 --      emit 'TurnEnd'.
  132 --   3. Append the assistant message and record it. If 'P.toolUseCalls' is
  133 --      empty, return the context. Otherwise run every call through
  134 --      '_turn_execTool', append + record each tool-result message, and loop.
  135 --   4. On a provider error (the stream throws, or yields no 'P.StreamDone'),
  136 --      do not crash: 'TurnEnd' is still emitted (it is emitted unconditionally
  137 --      after the stream attempt) and the context — carrying at least the user
  138 --      message — is returned. When the stream /throws/, a 'TurnError' is also
  139 --      emitted (after 'TurnEnd') so the failure is surfaced as a visible
  140 --      message — restoring the legacy @TemporaryError@ visibility (pureclaw-9d0).
  141 --      The no-'P.StreamDone' path stays a graceful, silent no-op.
  142 runTurnWithTools :: TurnDeps -> Context -> Text -> IO Context
  143 runTurnWithTools deps ctx0 userText = do
  144   let userMsg = P.textMessage P.User userText
  145       ctx1    = Ctx.addMessage userMsg ctx0
  146   _turn_record deps userMsg
  147   cycleTurn deps ctx1
  148 
  149 -- | One completion leg plus the tool cycle. Recurses when the response
  150 -- carries tool calls. Each call to 'cycleTurn' performs exactly one provider
  151 -- completion; a tool cycle of @n@ legs invokes 'cycleTurn' @n@ times.
  152 cycleTurn :: TurnDeps -> Context -> IO Context
  153 cycleTurn deps ctx = do
  154   let req = P.CompletionRequest
  155         { P._cr_model        = _turn_model deps
  156         , P._cr_messages     = Ctx.contextMessages ctx
  157         , P._cr_systemPrompt = case _turn_systemPrompt deps of
  158             Just sys -> Just sys
  159             Nothing  -> Ctx.contextSystemPrompt ctx
  160         , P._cr_maxTokens    = _turn_maxTokens deps
  161         , P._cr_tools        = _turn_tools deps
  162         , P._cr_toolChoice   = Nothing
  163         }
  164   sid <- _turn_nextStreamId deps
  165   _turn_emit deps (TurnStart sid)
  166   responseRef <- Ref.newIORef (Nothing :: Maybe P.CompletionResponse)
  167   streamedRef <- Ref.newIORef False
  168   -- Mirror Loop's @try@ around the stream so a throwing provider does not
  169   -- bring the turn down. The sink, not @_env_channel@, receives framing.
  170   streamResult <- E.try @E.SomeException $
  171     _turn_stream deps req $ \case
  172       P.StreamText t   -> Ref.writeIORef streamedRef True >> _turn_emit deps (TurnChunk sid t)
  173       P.StreamWarning _ -> pure ()      -- non-fatal; logged by prod sink elsewhere
  174       P.StreamDone resp -> do
  175         Ref.writeIORef responseRef (Just resp)
  176         -- Fire the one-shot-style hook on every StreamDone (mirrors the legacy
  177         -- the original completion handler StreamDone handler that ran
  178         -- @_env_onFirstStreamDone@). Idempotency is the action's responsibility
  179         -- — production wires a read-and-clear @fireOnce@ so
  180         -- @markBootstrapConsumed@ runs exactly once (pureclaw-8g4).
  181         _turn_onStreamDone deps
  182       _                 -> pure ()
  183   -- Non-streaming providers (e.g. Ollama with @stream = False@) emit only a
  184   -- single 'P.StreamDone' carrying the full response and no incremental
  185   -- 'P.StreamText', so nothing was relayed as a 'TurnChunk' above. Mirror Loop's
  186   -- fallback (Loop.hs: @unless wasStreaming ... _ch_send fullText@) by emitting
  187   -- the whole response text as one chunk so it is displayed.
  188   streamed <- Ref.readIORef streamedRef
  189   mResp0   <- Ref.readIORef responseRef
  190   case mResp0 of
  191     Just resp | not streamed ->
  192       let txt = P.responseText resp
  193       in M.unless (T.null (T.strip txt)) $ _turn_emit deps (TurnChunk sid txt)
  194     _ -> pure ()
  195   -- 'TurnEnd' is emitted unconditionally — success, throw, or no-StreamDone —
  196   -- so the burst is always closed for the downstream relay.
  197   _turn_emit deps (TurnEnd sid)
  198   case streamResult of
  199     Left _  -> do
  200       -- Provider threw: surface a VISIBLE error (mirrors the legacy
  201       -- the original completion handler @TemporaryError@), then stop and keep the
  202       -- context. Emitted after 'TurnEnd' so it lands as its own message rather
  203       -- than as part of the (closed) stream burst (pureclaw-9d0).
  204       _turn_emit deps (TurnError sid "Something went wrong. Please try again.")
  205       pure ctx
  206     Right () -> do
  207       mResp <- Ref.readIORef responseRef
  208       case mResp of
  209         Nothing   -> pure ctx        -- no StreamDone captured: stop gracefully
  210         Just resp -> handleResponse deps ctx resp
  211 
  212 -- | Fold a captured response into the context: append + record the assistant
  213 -- message, then either finish (no tool calls) or run the calls and re-complete
  214 -- (mirrors Loop.hs:251-268).
  215 handleResponse :: TurnDeps -> Context -> P.CompletionResponse -> IO Context
  216 handleResponse deps ctx resp = do
  217   let assistantMsg = P.Message P.Assistant (P._crsp_content resp)
  218       ctx'         = Ctx.recordUsage (P._crsp_usage resp)
  219                    $ Ctx.addMessage assistantMsg ctx
  220       calls        = P.toolUseCalls resp
  221   _turn_record deps assistantMsg
  222   if null calls
  223     then pure ctx'
  224     else do
  225       ctx'' <- M.foldM runOneCall ctx' calls
  226       cycleTurn deps ctx''
  227   where
  228     runOneCall :: Context -> (ToolCallId, Text, Value) -> IO Context
  229     runOneCall c (callId, name, input) = do
  230       resultMsg <- _turn_execTool deps callId name input
  231       _turn_record deps resultMsg
  232       pure (Ctx.addMessage resultMsg c)
