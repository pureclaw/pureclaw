  161 -- | Send to a named window through the WU3 tmux seam, reporting whether the
  162 -- window was found. tmux exits non-zero when @-t@ targets a missing window.
  163 --
  164 -- C3 input hygiene (§8 C3): the payload is sent LITERALLY (@send-keys -l --@)
  165 -- so embedded tmux key tokens (@C-c@, @Enter@, @;@, …) are typed verbatim rather
  166 -- than interpreted, then a SEPARATE @Enter@ submits the line (so a launch
  167 -- command still executes). The two-step preserves this function's @IO Bool@
  168 -- contract: found-ness is determined by the LITERAL send (the targeting step);
  169 -- the Enter is only attempted when that succeeded.
  170 realSendNamed :: Text -> Text -> ByteString -> IO Bool
  171 realSendNamed session windowName _input
  172   -- WU8 Part A (§8 C3/C4): fail-closed on the adopted/discovered path — an
  173   -- invalid session/window identifier is refused (False) WITHOUT any tmux op.
  174   | not (validateTmuxIdent session && validateTmuxIdent windowName) = pure False
  175 realSendNamed session windowName input = do
  176   mPath <- findTmux
  177   case mPath of
  178     Nothing -> pure False
  179     Just tmuxBin -> do
  180       let run args = P.runProcess
  181             $ P.setStdin P.closed
  182             $ P.setStdout P.nullStream
  183             $ P.setStderr P.nullStream
  184             $ tmuxProc (authorizeTmuxCommand tmuxBin (map T.pack args))
  185       litCode <- run (sendKeysNamedArgs session windowName input)
  186       case litCode of
  187         ExitSuccess -> do
  188           _ <- run (sendEnterNamedArgs session windowName)
  189           pure True
  190         ExitFailure _ -> pure False
  191 
  192 -- | Capture from a named window through the WU3 tmux seam. 'Nothing' signals a
  193 -- missing window (tmux exit non-zero); 'Just' carries the (ANSI-stripped) bytes.
  194 realCaptureNamed :: Text -> Text -> Int -> IO (Maybe ByteString)
  195 realCaptureNamed session windowName _lineCount
  196   -- WU8 Part A (§8 C3/C4): fail-closed — an invalid identifier is refused
  197   -- (Nothing, the missing-window value) WITHOUT issuing capture-pane.
  198   | not (validateTmuxIdent session && validateTmuxIdent windowName) = pure Nothing
  199 realCaptureNamed session windowName lineCount = do
  200   mPath <- findTmux
  201   case mPath of
  202     Nothing -> pure Nothing
  203     Just tmuxBin -> do
  204       let args = captureNamedArgs session windowName lineCount
  205           config = P.setStdin P.closed
  206                  $ P.setStdout P.byteStringOutput
  207                  $ P.setStderr P.nullStream
  208                  $ tmuxProc (authorizeTmuxCommand tmuxBin (map T.pack args))
  209       (exitCode, stdout, _stderr) <- P.readProcess config
  210       pure $ case exitCode of
  211         ExitSuccess   -> Just (LBS.toStrict stdout)
  212         ExitFailure _ -> Nothing

  214 -- | Window-status check by name through the WU3 tmux seam: a present window is
  215 -- 'HarnessRunning'; a missing one is 'HarnessExited'.
  216 realStatus :: Text -> Text -> IO HarnessStatus
  217 realStatus session windowName
  218   -- WU8 Part A (§8 C3/C4): fail-closed — an invalid identifier is treated as a
  219   -- missing/unusable window ('HarnessExited') WITHOUT issuing list-windows.
  220   | not (validateTmuxIdent session && validateTmuxIdent windowName) =
  221       pure (HarnessExited (ExitFailure 1))
  222 realStatus session windowName = do
  223   mTmux <- findTmux
  224   case mTmux of
  225     Nothing -> pure (HarnessExited (ExitFailure 127))
  226     Just tmuxBin -> do
  227       exitCode <- P.runProcess
  228         $ P.setStdin P.closed
  229         $ P.setStdout P.nullStream
  230         $ P.setStderr P.nullStream
  231         $ tmuxProc (authorizeTmuxCommand tmuxBin
  232             ["list-windows", "-t", T.pack (windowTarget session windowName)
  233             , "-F", "#{window_name}"])
  234       pure $ case exitCode of
  235         ExitSuccess      -> HarnessRunning
  236         ExitFailure code -> HarnessExited (ExitFailure code)

  244 -- On success it has stamped the durable @\@pcl_id@ identity marker, enabled

  249   :: SecurityPolicy

  253   -> Int              -- ^ tmux window index (placement hint for the new window)
  254   -> Maybe FilePath   -- ^ optional working directory
  255   -> [Text]           -- ^ extra CLI arguments (e.g. --dangerously-skip-permissions)
  256   -> HarnessRegistry  -- ^ the durable registry to register the spawned entry into
  257   -> IO (Either HarnessError (HarnessId, HarnessHandle))

  260 -- | Testable variant with injectable dependencies ('ClaudeCodeDeps').
  261 mkClaudeCodeHarnessWith
  262   :: ClaudeCodeDeps
  263   -> SecurityPolicy
  264   -> TranscriptHandle
  265   -> Text             -- ^ tmux session name
  266   -> Text             -- ^ tmux window name
  267   -> Int              -- ^ tmux window index
  268   -> Maybe FilePath   -- ^ optional working directory
  269   -> [Text]           -- ^ extra CLI arguments
  270   -> HarnessRegistry
  271   -> IO (Either HarnessError (HarnessId, HarnessHandle))
  272 mkClaudeCodeHarnessWith deps policy th session windowName _windowIdx mWorkDir extraArgs reg =
  273   -- Step 1: Pre-check authorization (pure, no IO needed).
  274   case preAuthorize policy of
  275     Left cmdErr -> pure (Left (HarnessNotAuthorized cmdErr))
  276     Right () -> do
  277       -- Step 2: Check tmux availability
  278       tmuxResult <- _ccd_checkTmux deps
  279       case tmuxResult of
  280         Left err -> pure (Left err)
  281         Right () -> do
  282           -- Step 3: Find claude binary
  283           mClaudePath <- _ccd_findClaude deps
  284           case mClaudePath of
  285             Nothing -> pure (Left (HarnessBinaryNotFound "claude"))
  286             Just claudePath ->
  287               -- Step 4: Authorize the full command path
  288               case authorize policy claudePath [] of
  289                 Left cmdErr -> pure (Left (HarnessNotAuthorized cmdErr))
  290                 Right authorizedCmd -> do
  291                   let program = getCommandProgram authorizedCmd
  292                   -- Step 5: Start tmux session (idempotent)
  293                   sessionResult <- _ccd_startSession deps session
  294                   case sessionResult of
  295                     Left err -> pure (Left err)
  296                     Right () -> do
  297                       -- Step 6: Add harness window (named)
  298                       windowResult <- _ccd_addWindow deps session windowName program extraArgs mWorkDir
  299                       case windowResult of
  300                         Left err -> pure (Left err)
  301                         Right () -> do
  302                           -- Step 6b: If unsafe mode, auto-confirm the safety prompt
  303                           when (hasUnsafeFlag extraArgs) $ do
  304                             threadDelay 2000000  -- 2s for the prompt to appear
  305                             _ <- _ccd_sendNamed deps session windowName ""
  306                             pure ()
  307                           -- Step 7: Establish identity (D4.2).
  308                           hid <- _ccd_newId deps
  309                           _ccd_setMarker deps session windowName (Reg.harnessIdToText hid)
  310                           _ccd_setRemain deps session windowName
  311                           mShellPid <- _ccd_panePidOf deps session windowName
  312                           mHarnessPid <- case mShellPid of
  313                             Just shellPid -> _ccd_harnessPidOf deps shellPid claudeComm
  314                             Nothing       -> pure Nothing
  315                           -- Step 8: Wire up the cached-coordinate handle (D4.3).
  316                           -- Spawn baseline = 0: PureClaw created this window
  317                           -- fresh, so there is no pre-existing backlog to
  318                           -- exclude (B3 / D3.2 — behavior unchanged). WU4
  319                           -- (adopt) instead calls
  320                           -- 'mkClaudeCodeHandleWithBaseline' with the window's
  321                           -- current scrollback line count so an adopted
  322                           -- window's prior backlog never enters the transcript.
  323                           handle <- mkClaudeCodeHandleWithBaseline deps reg hid th session 0
  324                           -- Step 9: Register the Spawned entry (D4.2).
  325                           let entry = Reg.HarnessEntry
  326                                 { Reg._he_id          = hid
  327                                 , Reg._he_session     = session
  328                                 , Reg._he_windowName  = windowName
  329                                 , Reg._he_shellPid    = mShellPid
  330                                 , Reg._he_harnessPid  = mHarnessPid
  331                                 , Reg._he_origin      = Reg.OriginSpawned
  332                                 , Reg._he_liveness    = Reg.LivenessIdle
  333                                 , Reg._he_extModified = False
  334                                 , Reg._he_stale       = False
  335                                 , Reg._he_sessionId   = Nothing
  336                                 , Reg._he_label       = windowName
  337                                 , Reg._he_orphanedTicks = 0
  338                                 , Reg._he_handle      = Just handle
  339                                 }
  340                           Reg.insertEntry reg entry
  341                           pure (Right (hid, handle))

  369 -- | Build a cached-coordinate 'HarnessHandle' for an already-registered harness
  370 -- with an explicit capture /baseline/ (B3 mechanism, D5).
  371 --
  372 -- The baseline is the number of pre-existing scrollback lines to EXCLUDE from
  373 -- captured\/recorded output (see 'dropBaselineLines'). 'harnessReceive' drops
  374 -- the first @baseline@ lines of every full capture before extraction, so
  375 -- pre-baseline backlog can never reach the transcript NOR the WS broadcast
  376 -- (D3.3 — both derive from the baseline-stripped capture).
  377 --
  378 -- The spawn path ('mkClaudeCodeHarnessWith') passes @0@: a freshly-created
  379 -- window has no pre-existing backlog, so behavior is unchanged (D3.2). WU4
  380 -- (adopt) is the intended other caller: it passes the adopted window's CURRENT
  381 -- scrollback line count so the window's prior backlog is excluded from the
  382 -- adoption point forward.
  383 mkClaudeCodeHandleWithBaseline
  384   :: ClaudeCodeDeps
  385   -> HarnessRegistry
  386   -> HarnessId
  387   -> TranscriptHandle
  388   -> Text            -- ^ tmux session name (for '_hh_session')
  389   -> Int             -- ^ initial capture baseline: pre-existing lines to exclude
  390   -> IO HarnessHandle
  391 mkClaudeCodeHandleWithBaseline deps reg hid th session baseline = do
  392   baselineRef <- newIORef baseline
  393   pure HarnessHandle
  394     { _hh_send    = harnessSend deps reg hid th baselineRef
  395     , _hh_receive = harnessReceive deps reg hid th baselineRef
  396     , _hh_name    = "Claude Code"
  397     , _hh_session = session
  398     , _hh_status  = harnessStatus deps reg hid
  399     , _hh_stop    = harnessStop deps reg hid

  406 -- | Adopt an EXTERNAL (discovered, PureClaw-UNMARKED) tmux window into the
  407 -- registry. This is the spawn path MINUS window creation: the window already
  408 -- exists, so we only stamp identity, set the capture baseline, register, and
  409 -- link a @session.json@.
  410 --
  411 -- The first argument is an 'AdoptedHarness' capability TOKEN. Its value
  412 -- constructor is unexported, so the ONLY way to obtain one is
  413 -- 'PureClaw.Security.Adoption.authorizeAdoption', which enforces the consent +
  414 -- allow-list gate. Requiring the token as a parameter makes it impossible BY
  415 -- CONSTRUCTION to adopt a window without first passing the gate (D4.3) — there
  416 -- is no token-free adopt path. The tmux session name rides in the token
  417 -- ('adoptedSession'); only the window name is supplied separately.
  418 --
  419 -- Steps (mirror 'mkClaudeCodeHarnessWith', omitting tmux session\/window
  420 -- creation):
  421 --
  422 --   1. Generate a fresh 'HarnessId'.
  423 --   2. Stamp the durable @\@pcl_id@ marker and enable @remain-on-exit@ on the
  424 --      adopted coordinates (the C4 trust anchor is established AT adopt time).
  425 --   3. Measure the window's CURRENT scrollback line count and build the handle
  426 --      with that as its capture baseline (B3 \/ D5): the pre-existing backlog
  427 --      is excluded from the transcript and the WS broadcast from the adoption
  428 --      point forward.
  429 --   4. Record the shell PID (best-effort) and derive the harness PID
  430 --      best-effort (may be 'Nothing' for a non-flavour window — OK).
  431 --   5. Register an @OriginAdopted@ 'Reg.HarnessEntry' (with the handle) into the
  432 --      registry.
  433 --   6. Create\/link a harness @session.json@ that persists the 'HarnessId' and
  434 --      the adopted tmux coordinates (mirrors the spawn persistence).
  435 --
  436 -- Does NOT create the window and does NOT capture the backlog. The legacy
  437 -- '_fe_harnesses' map sync (D-ADD-2) is the ENDPOINT's responsibility (it has
  438 -- the map); this mechanism returns the @(HarnessId, HarnessHandle)@ the caller
  439 -- inserts.
  440 adoptExternalWindow
  441   :: ClaudeCodeDeps
  442   -> HarnessRegistry
  443   -> TranscriptHandle
  444   -> FilePath            -- ^ sessions base directory (where @session.json@ is linked)
  445   -> AdoptedHarness      -- ^ capability token (REQUIRED — type-enforced, D4.3)
  446   -> Text                -- ^ tmux window name to adopt
  447   -> IO (Either HarnessError (HarnessId, HarnessHandle))
  448 adoptExternalWindow deps reg th sessionsDir token windowName = do
  449   let session = adoptedSession token
  450   -- WU8 Part A (§8 C3/C4 defense-in-depth): the adopted coordinates originate
  451   -- from the server-wide sweep (attacker-writable) and the allow-list (a literal
  452   -- pattern can mint any session string). Validate BOTH identifiers FAIL-CLOSED
  453   -- BEFORE any tmux mutation — a leading @-@ / @:@ / control char never reaches
  454   -- set-option, capture-pane, or registration. The argv defense already holds;
  455   -- this refuses the most-exposed path outright (no marker stamped, nothing
  456   -- registered, no session.json written).
  457   if not (validateTmuxIdent session && validateTmuxIdent windowName)
  458     then pure (Left (HarnessNotAuthorized
  459       (CommandNotAllowed ("invalid tmux identifier for adoption: "
  460         <> session <> ":" <> windowName))))
  461     else adoptValidated deps reg th sessionsDir session windowName
  462 
  463 -- | The adopt mechanism proper, reached only AFTER both tmux identifiers have
  464 -- passed 'validateTmuxIdent' (WU8 Part A). Split out so the guard in
  465 -- 'adoptExternalWindow' provably precedes every tmux op below.
  466 adoptValidated
  467   :: ClaudeCodeDeps
  468   -> HarnessRegistry
  469   -> TranscriptHandle
  470   -> FilePath
  471   -> Text                -- ^ validated tmux session name
  472   -> Text                -- ^ validated tmux window name
  473   -> IO (Either HarnessError (HarnessId, HarnessHandle))
  474 adoptValidated deps reg th sessionsDir session windowName = do
  475   -- Step 1: fresh identity.
  476   hid <- _ccd_newId deps
  477   -- Step 2: resolve the user-picked window to its INDEX via a fresh scan. The
  478   -- scan reads @#{window_name}@ as DATA, so a name containing @.@ (or a purely
  479   -- numeric name) cannot be misparsed here. We pick the FIRST UNMARKED window
  480   -- whose name matches (mirrors discovery's adoptable filter — avoids grabbing
  481   -- an already-ours duplicate-named window). If none matches, the window is
  482   -- gone\/already adopted: refuse with NO mutation (maps to 503).
  483   rows <- _ccd_sweep deps session
  484   case find (\r -> _twr_windowName r == windowName && _twr_pclId r == "") rows of
  485     Nothing -> pure (Left (HarnessTmuxNotAvailable
  486       ("adopt: no unmarked window named " <> windowName
  487         <> " in session " <> session)))
  488     Just row -> do
  489       let idx = _twr_windowIndex row
  490           -- Step 3: a SAFE canonical name. The HarnessId UUID is hex + hyphens,
  491           -- so the result never contains @.@\/@:@ and passes 'validateTmuxIdent'.
  492           -- We DO NOT reuse the user's name (which may contain a @.@ that breaks
  493           -- the @session:name@ targeting every downstream op relies on — the
  494           -- root-cause of the §8 C4 "no corroborating PID" eviction).
  495           safeName = "adopted-" <> T.take 8 (Reg.harnessIdToText hid)
  496       -- Step 4: rename the window BY INDEX (@session:<int>@ resolves by window
  497       -- index — unambiguous, immune to the @.@ misparse). After this, the window
  498       -- name is safe and @session:safeName@ targeting is correct.
  499       _ccd_renameWindow deps session (T.pack (show idx)) safeName
  500       -- From here EVERYTHING addresses the window by the SAFE name.
  501       -- Step 5: establish identity on the renamed window (C4 anchor at adopt time).
  502       _ccd_setMarker deps session safeName (Reg.harnessIdToText hid)
  503       _ccd_setRemain deps session safeName
  504       -- Step 6: measure the window's current scrollback and set the baseline so
  505       -- the pre-existing backlog is excluded (B3 / D5). A full capture
  506       -- (lineCount 0) mirrors the receive path's full-screen capture; we count
  507       -- its lines.
  508       fullCapture <- fromMaybe "" <$> _ccd_captureNamed deps session safeName 0
  509       let baseline = countCaptureLines fullCapture
  510       -- Step 7: PID provenance (both best-effort).
  511       mShellPid <- _ccd_panePidOf deps session safeName
  512       mHarnessPid <- case mShellPid of
  513         Just shellPid -> _ccd_harnessPidOf deps shellPid claudeComm
  514         Nothing       -> pure Nothing
  515       -- Step 8: build the cached-coordinate handle with the adoption baseline.
  516       handle <- mkClaudeCodeHandleWithBaseline deps reg hid th session baseline
  517       -- Step 9: link a harness session.json (mirrors the spawn persistence;
  518       -- persists the HarnessId + adopted coords like WU6 / Phase 2).
  519       now <- getCurrentTime
  520       let sid = newSessionId Nothing now
  521           entry = Reg.HarnessEntry
  522             { Reg._he_id          = hid
  523             , Reg._he_session     = session
  524             , Reg._he_windowName  = safeName
  525             , Reg._he_shellPid    = mShellPid
  526             , Reg._he_harnessPid  = mHarnessPid
  527             , Reg._he_origin      = Reg.OriginAdopted
  528             , Reg._he_liveness    = Reg.LivenessIdle
  529             , Reg._he_extModified = False
  530             , Reg._he_stale       = False
  531             , Reg._he_sessionId   = Just (unSessionId sid)
  532             , Reg._he_label       = safeName
  533             , Reg._he_orphanedTicks = 0
  534             , Reg._he_handle      = Just handle
  535             }
  536           meta = SessionMeta
  537             { _sm_id                = sid
  538             , _sm_agent             = Nothing
  539             , _sm_kind              = SkHarness HarnessSpec
  540                 { _h_flavour   = HClaudeCode
  541                 , _h_backend   = TbTmux TmuxConfig
  542                     { _tc_session = session
  543                     , _tc_window  = safeName
  544                     , _tc_pane    = Nothing
  545                     }
  546                 , _h_cwd       = Nothing
  547                 , _h_args      = []
  548                 , _h_harnessId = Just hid
  549                 -- Adopted harnesses are NOT spawned by PureClaw, so we never
  550                 -- minted a --session-id for them and cannot correlate their
  551                 -- JSONL log (WU6: claude-code log capture is spawned-only).
  552                 -- Both Nothing.
  553                 , _h_claudeSessionUuid = Nothing
  554                 , _h_canonicalCwd      = Nothing
  555                 }
  556             , _sm_model             = ""
  557             , _sm_channel           = "web"
  558             , _sm_createdAt         = now
  559             , _sm_lastActive        = now
  560             , _sm_bootstrapConsumed = True
  561             , _sm_archived          = False
  562             , _sm_description       = Nothing
  563             , _sm_autoSummary       = Nothing
  564             , _sm_source            = Nothing
  565             }
  566       Reg.insertEntry reg entry
  567       writeSessionMeta sessionsDir meta
  568       pure (Right (hid, handle))

  570 -- | Count the newline-delimited lines in a raw capture, matching the splitting
  571 -- 'dropBaselineLines' \/ 'extractLastResponse' use (split on @0x0A@). An empty
  572 -- capture is 0 lines; otherwise it is one more than the number of @0x0A@ bytes.
  573 -- This is the adoption baseline: the count of pre-existing scrollback lines to
  574 -- exclude from the adoption point forward (B3).
  575 countCaptureLines :: ByteString -> Int
  576 countCaptureLines bs
  577   | BS.null bs = 0
  578   | otherwise  = length (BS.split 0x0A bs)

  580 -- | Atomically write a session's @session.json@ (mirrors
  581 -- @PureClaw.Session.Handle.saveMeta@: tmp-file + 'Dir.renameFile'). Kept local
  582 -- so the adopt mechanism does not depend on the higher broker-aware
  583 -- 'PureClaw.Session.Handle' layer. Creates the session directory if needed.
  584 writeSessionMeta :: FilePath -> SessionMeta -> IO ()
  585 writeSessionMeta baseDir meta = do
  586   let dir    = baseDir </> T.unpack (unSessionId (_sm_id meta))
  587       finalP = dir </> "session.json"
  588       tmpP   = finalP <> ".tmp"
  589       txP    = dir </> "transcript.jsonl"
  590   Dir.createDirectoryIfMissing True dir
  591   -- Mirror the spawn path's session creation: 'mkSessionHandle' opens
  592   -- @transcript.jsonl@ with @O_CREAT 0o600@ at session-creation time, so a
  593   -- freshly-spawned session always has the file. The adopt path persists the
  594   -- meta by hand (it does not go through 'mkSessionHandle'), so without this an
  595   -- adopted session has only @session.json@ — and the FIRST
  596   -- @POST /api/sessions/<sid>/send@ is rejected 404 by 'handleSend', whose
  597   -- existence check is @doesFileExist transcript.jsonl@. Create the (empty)
  598   -- transcript here, non-destructively (never truncates an existing one) and
  599   -- with the same 0o600 mode the transcript layer uses.
  600   txExists <- Dir.doesFileExist txP
  601   unless txExists $ do
  602     LBS.writeFile txP LBS.empty
  603     PFiles.setFileMode txP 0o600
  604   LBS.writeFile tmpP (Aeson.encode meta)
  605   -- Best-effort persistence: a failed rename is SWALLOWED (swallow-and-continue),
  606   -- not surfaced. The adopt mechanism has already stamped identity and registered
  607   -- the OriginAdopted entry by this point, so a session.json write fault must not
  608   -- fail the adoption — the in-memory registry entry still routes; the persisted
  609   -- metadata is a reconnect convenience that the next reconcile/save can rebuild.
  610   result <- try (Dir.renameFile tmpP finalP) :: IO (Either IOError ())
  611   case result of
  612     Right () -> pure ()
  613     Left _   -> pure ()

  619 -- | Read the current cached coordinate @(session, windowName)@ for a harness
  620 -- from its registry entry. The handle does NOT close over a frozen window name
  621 -- (design §4): it reads the live entry on every I\/O so the reconcile loop
  622 -- (WU5) can update the coordinate underneath it.
  623 currentCoord :: HarnessRegistry -> HarnessId -> IO (Maybe (Text, Text))
  624 currentCoord reg hid = do
  625   mEntry <- Reg.lookupById reg hid
  626   pure (fmap (\e -> (Reg._he_session e, Reg._he_windowName e)) mEntry)

  628 -- | Re-resolve a harness's window after a tmux \"no such window\": sweep the
  629 -- session, find the row whose @\@pcl_id@ matches this harness's id, update the
  630 -- cached '_he_windowName', and return the new name. 'Nothing' if no marked
  631 -- window matches (the full reconcile loop is WU5; this is the minimal
  632 -- one-shot heal).
  633 reResolve :: ClaudeCodeDeps -> HarnessRegistry -> HarnessId -> Text -> IO (Maybe Text)
  634 reResolve deps reg hid session = do
  635   rows <- _ccd_sweep deps session
  636   let idText = Reg.harnessIdToText hid
  637   case find ((== idText) . _twr_pclId) rows of
  638     Nothing  -> pure Nothing
  639     Just row -> do
  640       let newName = _twr_windowName row
  641       mEntry <- Reg.lookupById reg hid
  642       case mEntry of
  643         Nothing -> pure ()
  644         Just e  -> Reg.insertEntry reg e { Reg._he_windowName = newName }
  645       pure (Just newName)

  647 -- | Send input to the harness, logging the request. Reads the cached coordinate;
  648 -- on a tmux \"no such window\" it re-resolves once by @\@pcl_id@ and retries.
  649 harnessSend :: ClaudeCodeDeps -> HarnessRegistry -> HarnessId -> TranscriptHandle -> IORef Int -> ByteString -> IO ()
  650 harnessSend deps reg hid th _baselineRef input = do
  651   entryId <- UUID.toText <$> UUID.nextRandom
  652   now <- getCurrentTime
  653   let entry = TranscriptEntry
  654         { _te_id            = entryId
  655         , _te_timestamp     = now
  656         , _te_harness       = Just "claude-code"
  657         , _te_model         = Nothing
  658         , _te_direction     = Request
  659         , _te_payload       = encodePayload input
  660         , _te_durationMs    = Nothing
  661         , _te_correlationId = entryId
  662         , _te_metadata      = Map.empty
  663         }
  664   _th_record th entry
  665   mCoord <- currentCoord reg hid
  666   case mCoord of
  667     Nothing -> pure ()  -- entry vanished; nothing to target
  668     Just (session, windowName) -> do
  669       found <- _ccd_sendNamed deps session windowName input
  670       unless found $ do
  671         mNew <- reResolve deps reg hid session
  672         case mNew of
  673           Nothing      -> pure ()  -- could not heal; give up (WU5 owns reconcile)
  674           Just newName -> do
  675             _ <- _ccd_sendNamed deps session newName input
  676             pure ()

  678 -- | Poll the harness window until idle, then capture and extract the last
  679 -- response. Re-resolves the coordinate once on a tmux \"no such window\".
  680 harnessReceive :: ClaudeCodeDeps -> HarnessRegistry -> HarnessId -> TranscriptHandle -> IORef Int -> IO ByteString
  681 harnessReceive deps reg hid th baselineRef = do
  682   mCoord <- currentCoord reg hid
  683   case mCoord of
  684     Nothing -> pure ""
  685     Just (session, windowName0) -> do
  686       -- Ensure the window is reachable; re-resolve once if not.
  687       windowName <- do
  688         probe <- _ccd_captureNamed deps session windowName0 1
  689         case probe of
  690           Just _  -> pure windowName0
  691           Nothing -> do
  692             mNew <- reResolve deps reg hid session
  693             pure (fromMaybe windowName0 mNew)
  694       startTime <- getCurrentTime
  695       pollUntilIdle session windowName startTime "" (0 :: Int)
  696       fullCapture <- fromMaybe "" <$> _ccd_captureNamed deps session windowName 0
  697       -- B3: exclude everything before the recorded baseline (the count of
  698       -- pre-existing scrollback lines) so pre-baseline backlog can never be
  699       -- extracted/recorded. Both 'responseText' (the WS-broadcast value) and
  700       -- the transcript entry below derive from the stripped capture (D3.3).
  701       baseline <- readIORef baselineRef
  702       let responseText = extractLastResponse (dropBaselineLines baseline fullCapture)
  703       entryId <- UUID.toText <$> UUID.nextRandom
  704       now <- getCurrentTime
  705       let entry = TranscriptEntry
  706             { _te_id            = entryId
  707             , _te_timestamp     = now
  708             , _te_harness       = Just "claude-code"
  709             , _te_model         = Nothing
  710             , _te_direction     = Response
  711             , _te_payload       = encodePayload responseText
  712             , _te_durationMs    = Nothing
  713             , _te_correlationId = entryId
  714             , _te_metadata      = Map.empty
  715             }
  716       _th_record th entry
  717       pure responseText
  718   where
  719     pollUntilIdle session windowName startTime lastScreen stableCount = do
  720       threadDelay 500000  -- 500ms
  721       current <- fromMaybe "" <$> _ccd_captureNamed deps session windowName 300
  722       now <- getCurrentTime
  723       let elapsed = diffUTCTime now startTime
  724           screenText = TE.decodeUtf8Lenient current
  725       if elapsed > 120
  726         then pure ()
  727         else if isIdle screenText
  728           then if current == lastScreen
  729             then if stableCount >= 2
  730               then pure ()
  731               else pollUntilIdle session windowName startTime current (stableCount + 1)
  732             else pollUntilIdle session windowName startTime current 0
  733           else pollUntilIdle session windowName startTime "" 0

  755 -- | Check if Claude Code is idle (showing prompt, not busy).
  756 isIdle :: Text -> Bool
  757 isIdle screen =
  758   let hasPrompt = T.isInfixOf "\x276F" screen   -- ❯
  759       isBusy    = T.isInfixOf "\x280B" screen    -- ⠋ (spinner)
  760                 || T.isInfixOf "Thinking" screen
  761                 || T.isInfixOf "Running" screen
  762   in hasPrompt && not isBusy

  764 -- | Drop the first @n@ scrollback lines from a raw capture (B3 baseline
  765 -- mechanism). The baseline is the number of pre-existing lines to EXCLUDE.
  766 --
  767 -- Total: @n <= 0@ → identity; @n >= line count@ → empty; otherwise drops
  768 -- exactly @n@ leading newline-delimited lines and re-joins the remainder with
  769 -- @0x0A@. Splitting on @0x0A@ mirrors 'extractLastResponse' so the line indices
  770 -- agree.
  771 dropBaselineLines :: Int -> ByteString -> ByteString
  772 dropBaselineLines n capture
  773   | n <= 0    = capture
  774   | otherwise = BS.intercalate (BS.singleton 0x0A) (drop n (BS.split 0x0A capture))

  776 -- | Extract the last response block from Claude Code scrollback.
  777 extractLastResponse :: ByteString -> ByteString
  778 extractLastResponse capture =
  779   let allLines  = map TE.decodeUtf8Lenient (BS.split 0x0A capture)
  780       markerIdxs = [ i | (i, line) <- zip [0..] allLines
  781                        , isResponseMarker line ]
  782   in case markerIdxs of
  783     [] -> ""  -- no response found
  784     _  ->
  785       let startIdx   = last markerIdxs
  786           response   = takeWhile (not . isUiBoundary)
  787                      $ drop startIdx allLines
  788           cleaned    = case response of
  789             (firstLine : rest) -> stripMarker firstLine : rest
  790             []                 -> []
  791       in TE.encodeUtf8 (T.intercalate "\n" cleaned)
  792 
  793 -- | Lines starting with ⏺ (U+23FA, BLACK CIRCLE FOR RECORD) are response markers.
  794 isResponseMarker :: Text -> Bool
  795 isResponseMarker line =
  796   T.isPrefixOf "\x23FA" (T.stripStart line)
  797   || T.isPrefixOf "\x2B24" (T.stripStart line)  -- ⬤ alternate marker
  798 
  799 -- | UI boundaries that terminate response extraction.
  800 isUiBoundary :: Text -> Bool
  801 isUiBoundary line =
  802   let stripped = T.stripStart line
  803   in T.isPrefixOf "\x276F" stripped         -- ❯ input prompt
  804   || T.isPrefixOf "?" stripped
  805      && T.isInfixOf "shortcut" line         -- "? for shortcuts"
  806   || T.isInfixOf "\x2580\x2580" line        -- ▀▀ top bar
  807   || T.isInfixOf "\x2584\x2584" line        -- ▄▄ bottom bar
  808   || T.isInfixOf "\x2500\x2500\x2500" line  -- ─── horizontal rule
  809 
  810 -- | Strip the response marker prefix (⏺ or ⬤) from a line.
  811 stripMarker :: Text -> Text
  812 stripMarker line =
  813   let stripped = T.stripStart line
  814   in if T.isPrefixOf "\x23FA" stripped || T.isPrefixOf "\x2B24" stripped
  815      then T.stripStart (T.drop 1 stripped)
  816      else line

  900 -- | Capture the full scrollback buffer (not just the visible pane) by name,
  901 -- through the WU3 tmux seam. Returns empty on a missing window or tmux absence.
  902 captureFullScrollbackNamed :: Text -> Text -> IO ByteString
  903 captureFullScrollbackNamed session windowName
  904   -- WU8 Part A (§8 C3/C4): fail-closed on the discovered receive path — an
  905   -- invalid identifier yields the empty capture WITHOUT issuing capture-pane.
  906   | not (validateTmuxIdent session && validateTmuxIdent windowName) = pure ""
  907 captureFullScrollbackNamed session windowName = do
  908   mPath <- findTmux
  909   case mPath of
  910     Nothing -> pure ""
  911     Just tmuxBin -> do
  912       let config = P.setStdin P.closed
  913                  $ P.setStdout P.byteStringOutput
  914                  $ P.setStderr P.nullStream
  915                  $ tmuxProc (authorizeTmuxCommand tmuxBin
  916                      [ "capture-pane", "-t", T.pack (windowTarget session windowName)
  917                      , "-p", "-S", "-", "-E", "-"
  918                      ])
  919       (exitCode, stdout, _stderr) <- P.readProcess config
  920       pure $ case exitCode of
  921         ExitSuccess   -> LBS.toStrict stdout
  922         ExitFailure _ -> ""
