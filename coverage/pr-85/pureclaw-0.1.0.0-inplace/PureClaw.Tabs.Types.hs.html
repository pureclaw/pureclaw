    1 -- |
    2 -- Module      : PureClaw.Tabs.Types
    3 -- Description : First-class tab registry — core value types & pure operations.
    4 --
    5 -- A /tab/ is a pure binding over ground truth: it names a 'TabRef' (a live
    6 -- 'SessionId' or 'HarnessId'), the slot it currently occupies, a friendly
    7 -- label, and a liveness 'TabStatus'. The ordered collection of tabs is a
    8 -- 'TabList' that maintains two invariants:
    9 --
   10 --   * __I1 (contiguity)__ — the slots of a 'TabList' are exactly
   11 --     @[0 .. n-1]@ at all times. 'removeSlot' compacts (tmux-window style):
   12 --     every tab above the removed slot shifts down by one.
   13 --   * __I2 (uniqueness)__ — no two tabs bind the same 'TabRef'. 'appendTab'
   14 --     of an already-bound ref is rejected with @'Left' ('AlreadyBound' i)@,
   15 --     reporting the ref's /current/ slot.
   16 --
   17 -- There is also a hard cap of 36 slots (@0..35@), matching the validated
   18 -- 'TabIndex' range. Appending past it returns @'Left' 'SlotsFull'@.
   19 --
   20 -- This module is pure (no IO). The thin IORef handle that mutates a 'TabList'
   21 -- lives in "PureClaw.Tabs". The 0–35 'TabIndex' arithmetic is reused from
   22 -- "PureClaw.Handles.Tab" ('TabIndex'\/'mkTabIndex'\/'unTabIndex') rather than
   23 -- redefined, so there is a single validated index type across the codebase.
   24 --
   25 -- See the Tabs-as-View refactor (GitHub #79) for the design context.

   72 -- TabRef / TabStatus / Tab

   75 -- | The ground-truth entity a tab binds to: either an LLM-backed
   76 -- 'SessionId' or an external 'HarnessId'. A tab is a /view/ over one of
   77 -- these; it owns no lifecycle of its own.
   78 data TabRef
   79   = BoundSession !SessionId
   80   | BoundHarness !HarnessId
   81   deriving stock (Eq, Ord, Show)
   82 
   83 -- | Liveness of a tab's bound ground truth.
   84 --
   85 -- * 'Live' — the bound session\/harness is believed alive; messages route.
   86 -- * 'Dead' — the bound harness has exited (a tombstone). The tab persists
   87 --   in the list (visible in @\/tabs@) until it is explicitly removed; routing
   88 --   to a 'Dead' tab is handled by the dispatcher (deferred warning + drop).
   89 data TabStatus
   90   = Live
   91   | Dead
   92   deriving stock (Eq, Show)
   93 
   94 -- | A single tab: a pure binding over ground truth.
   95 data Tab = Tab
   96   { _tab_slot   :: !TabIndex
   97     -- ^ The tab's position in the 'TabList'. Maintained contiguous (I1);
   98     --   never set directly by callers — derived by 'appendTab'\/'removeSlot'.
   99   , _tab_ref    :: !TabRef
  100     -- ^ The bound ground-truth entity (unique across the list, I2).
  101   , _tab_status :: !TabStatus
  102     -- ^ Liveness of the binding.
  103   }
  104   deriving stock (Eq, Show)

  106 -- ---------------------------------------------------------------------------
  107 -- TabList
  108 -- ---------------------------------------------------------------------------
  109 
  110 -- | An ordered list of tabs maintaining invariants I1 (contiguous slots
  111 -- @[0..n-1]@) and I2 (unique 'TabRef's). The constructor is not exported;
  112 -- all mutation goes through the smart operations below, which preserve the
  113 -- invariants by construction.
  114 newtype TabList = TabList [Tab]
  115   deriving stock (Eq, Show)
  116 
  117 -- | Why a 'TabList' mutation was rejected.
  118 data TabsError
  119   = SlotsFull
  120     -- ^ All 36 slots are occupied; the next append would be slot 36.
  121   | AlreadyBound !TabIndex
  122     -- ^ The 'TabRef' is already bound; the payload is its /current/ slot.
  123   deriving stock (Eq, Show)
  124 
  125 -- | The hard slot cap. Slots are @0 .. 'maxTabSlots' - 1@ (i.e. 36 slots,
  126 -- matching the validated 'TabIndex' range used elsewhere).
  127 maxTabSlots :: Int
  128 maxTabSlots = 36
  129 
  130 -- | The empty tab list.
  131 emptyTabs :: TabList
  132 emptyTabs = TabList []
  133 
  134 -- | The tabs in slot order (slot @0@ first). Total; never partial.
  135 toList :: TabList -> [Tab]
  136 toList (TabList ts) = ts
  137 
  138 -- | Append a new tab binding @ref@ at the next free slot.
  139 --
  140 -- * If @ref@ is already bound, returns @'Left' ('AlreadyBound' i)@ where @i@
  141 --   is its current slot (I2 — no duplicate refs).
  142 -- * If all 36 slots are taken, returns @'Left' 'SlotsFull'@.
  143 -- * Otherwise the new tab lands at slot @length@ (preserving I1) with status
  144 --   'Live', and the new slot index is returned alongside the updated list.
  145 --
  146 -- Tabs no longer carry a friendly label of their own — a tab's display title
  147 -- derives from its bound session (or a harness label fallback) at render time.
  148 appendTab :: TabRef -> TabList -> Either TabsError (TabIndex, TabList)
  149 appendTab ref tl@(TabList ts) =
  150   case lookupRef ref tl of
  151     Just i  -> Left (AlreadyBound i)
  152     Nothing ->
  153       let n = length ts
  154       in if n >= maxTabSlots
  155            then Left SlotsFull
  156            else
  157              -- @0 <= n < 36@ here, so slot @n@ is valid. Append the new tab
  158              -- (with a provisional slot 'reindex' overwrites) and 'reindex' to
  159              -- stamp contiguous slots @0..n@ (I1). The new tab is last, so its
  160              -- slot is the last stamped index, read back via 'NE.last'.
  161              let placed = NE.fromList (reindex (ts ++ [Tab firstSlot ref Live]))
  162              in Right (_tab_slot (NE.last placed), TabList (NE.toList placed))
  163 
  164 -- | The infinite stream of valid slots @0, 1, 2, …@ as a 'NonEmpty'. Built
  165 -- via the comprehension-filter trick so the impossible @'mkTabIndex' n ==
  166 -- Nothing@ (n negative) case introduces no uncoverable alternative; the
  167 -- 'NonEmpty' shape makes 'firstSlot' total (no partial 'head').
  168 slotStream :: NonEmpty TabIndex
  169 slotStream = NE.fromList [ s | n <- [0 :: Int ..], Just s <- [mkTabIndex n] ]
  170 
  171 -- | The first slot (@0@), used as the provisional slot for a freshly-appended
  172 -- tab before 'reindex' stamps the real one.
  173 firstSlot :: TabIndex
  174 firstSlot = NE.head slotStream
  175 
  176 -- | Remove the tab at @slot@ (if present) and compact: every tab above the
  177 -- removed slot shifts down by one, so the result still satisfies I1. A no-op
  178 -- when no tab occupies @slot@.
  179 removeSlot :: TabIndex -> TabList -> TabList
  180 removeSlot slot (TabList ts) =
  181   TabList (reindex (filter ((/= unTabIndex slot) . unTabIndex . _tab_slot) ts))
  182 
  183 -- | Re-stamp a list of tabs with contiguous slots @0..n-1@ in their current
  184 -- order. Used after a removal to restore I1. Total: 'slotStream' supplies a
  185 -- valid 'TabIndex' for every position, so 'zipWith' stamps every tab.
  186 reindex :: [Tab] -> [Tab]
  187 reindex = zipWith (\s t -> t { _tab_slot = s }) (NE.toList slotStream)
  188 
  189 -- | Look up the tab currently at @slot@.
  190 lookupSlot :: TabIndex -> TabList -> Maybe Tab
  191 lookupSlot slot (TabList ts) =
  192   case filter ((== unTabIndex slot) . unTabIndex . _tab_slot) ts of
  193     (t : _) -> Just t
  194     []      -> Nothing
  195 
  196 -- | Look up the slot currently occupied by @ref@ (I2 guarantees at most one).
  197 lookupRef :: TabRef -> TabList -> Maybe TabIndex
  198 lookupRef ref (TabList ts) =
  199   case filter ((== ref) . _tab_ref) ts of
  200     (t : _) -> Just (_tab_slot t)
  201     []      -> Nothing
  202 
  203 -- | Set the 'TabStatus' of the tab bound to @ref@. A no-op when @ref@ is not
  204 -- bound. Slots are unaffected.
  205 setStatus :: TabRef -> TabStatus -> TabList -> TabList
  206 setStatus ref status (TabList ts) =
  207   TabList (map upd ts)
  208   where
  209     upd t =
  210       if _tab_ref t == ref
  211         then t { _tab_status = status }
  212         else t
  213 
  214 -- | Rebind the tab at @slot@ to a new @ref@, /in place/ — the slot, and the
  215 -- slots of every other tab, are unchanged (I1 preserved). The rebound tab's
  216 -- status is reset to 'Live'. This backs @\/new@'s "reset the active tab to a
  217 -- fresh session" semantics (design §6.1).
  218 --
  219 -- The new @ref@ is dedup-checked against I2:
  220 --
  221 --   * If @ref@ is already bound to a /different/ slot, the rebind is rejected
  222 --     with @'Left' ('AlreadyBound' i)@ (@i@ = that other slot) and the list is
  223 --     unchanged.
  224 --   * Rebinding a slot to the ref it /already/ holds is allowed — it is just a
  225 --     relabel (and a status reset) in place.
  226 --
  227 -- If no tab occupies @slot@, this is a no-op returning the list unchanged
  228 -- (@'Right'@) — callers that already verified an active tab never hit this,
  229 -- but it keeps the function total.
  230 rebindSlot :: TabIndex -> TabRef -> TabList -> Either TabsError TabList
  231 rebindSlot slot ref tl@(TabList ts) =
  232   case lookupSlot slot tl of
  233     Nothing -> Right tl
  234     Just _  ->
  235       case lookupRef ref tl of
  236         Just other | other /= slot -> Left (AlreadyBound other)
  237         _ -> Right (TabList (map upd ts))
  238   where
  239     upd t =
  240       if unTabIndex (_tab_slot t) == unTabIndex slot
  241         then t { _tab_ref = ref, _tab_status = Live }
  242         else t

  244 -- ---------------------------------------------------------------------------
  245 -- Per-conversation cursors & relay (I3)
  246 -- ---------------------------------------------------------------------------

  248 -- | The identity of a conversation: its originating 'ChannelKind' paired with
  249 -- a server-derived 'ConversationId'. Two messages sharing a key belong to the
  250 -- same conversation and therefore share a tab cursor and a 'RelayMode'.
  251 type ConversationKey = (ChannelKind, ConversationId)
  252 
  253 -- | How a conversation's output relay fans tab output out to it. Persisted
  254 -- per 'ConversationKey'; the global default lives in config (see the
  255 -- Tabs-as-View design §9.3).
  256 --
  257 -- * 'FocusedOnly' (default) — only the conversation's focused tab is relayed.
  258 -- * 'ActivityDigest' — focused tab full; other live tabs get an activity ping.
  259 -- * 'Firehose' — full content from all live tabs.
  260 data RelayMode
  261   = FocusedOnly
  262   | ActivityDigest
  263   | Firehose
  264   deriving stock (Eq, Show)
  265 
  266 -- | The per-conversation routing state the dispatcher owns: which 'TabRef'
  267 -- each conversation is focused on ('_cs_cursors'), and any per-conversation
  268 -- 'RelayMode' override ('_cs_relay'). Both maps are keyed by 'ConversationKey'.
  269 --
  270 -- Cursors key by 'TabRef', not by slot — this is invariant __I3__: a cursor
  271 -- survives slot compaction ('removeSlot') because it names ground truth, not a
  272 -- position. 'resolveCursorSlot' resolves the ref to its /current/ slot at read
  273 -- time.
  274 data CursorState = CursorState
  275   { _cs_cursors :: !(Map ConversationKey TabRef)
  276     -- ^ The 'TabRef' each conversation is currently focused on.
  277   , _cs_relay   :: !(Map ConversationKey RelayMode)
  278     -- ^ Per-conversation 'RelayMode' overrides (absent ⇒ use the global default).
  279   }
  280   deriving stock (Eq, Show)
  281 
  282 -- | The empty cursor state: no cursors, no relay overrides.
  283 emptyCursors :: CursorState
  284 emptyCursors = CursorState Map.empty Map.empty
  285 
  286 -- | Focus a conversation on a 'TabRef', replacing any prior cursor for that key.
  287 -- The relay map is untouched.
  288 setCursor :: ConversationKey -> TabRef -> CursorState -> CursorState
  289 setCursor k ref cs = cs { _cs_cursors = Map.insert k ref (_cs_cursors cs) }
  290 
  291 -- | Drop a conversation's cursor (a no-op if it had none). The relay map is
  292 -- untouched.
  293 clearCursor :: ConversationKey -> CursorState -> CursorState
  294 clearCursor k cs = cs { _cs_cursors = Map.delete k (_cs_cursors cs) }
  295 
  296 -- | Resolve a conversation's cursor to the slot its bound 'TabRef' currently
  297 -- occupies (__I3__). Returns 'Nothing' when the key has no cursor or when the
  298 -- cursor's ref is no longer present in the 'TabList' (a dangling cursor).
  299 resolveCursorSlot :: ConversationKey -> CursorState -> TabList -> Maybe TabIndex
  300 resolveCursorSlot k cs tl =
  301   case Map.lookup k (_cs_cursors cs) of
  302     Nothing  -> Nothing
  303     Just ref -> lookupRef ref tl
  304 
  305 -- | All conversation keys whose cursor is focused on the given 'TabRef'.
  306 conversationsOn :: TabRef -> CursorState -> [ConversationKey]
  307 conversationsOn ref cs =
  308   [ k | (k, r) <- Map.toList (_cs_cursors cs), r == ref ]
  309 
  310 -- | Drop every cursor whose 'TabRef' is no longer present in the 'TabList'.
  311 -- Valid cursors and the entire relay map are preserved.
  312 pruneDangling :: TabList -> CursorState -> CursorState
  313 pruneDangling tl cs =
  314   cs { _cs_cursors = Map.filter present (_cs_cursors cs) }
  315   where
  316     present ref = case lookupRef ref tl of
  317       Just _  -> True
  318       Nothing -> False
  319 
  320 -- | The 'RelayMode' for a conversation: its per-conversation override if one is
  321 -- set, otherwise the supplied global default.
  322 relayModeFor :: ConversationKey -> RelayMode -> CursorState -> RelayMode
  323 relayModeFor k def cs = Map.findWithDefault def k (_cs_relay cs)

  329 -- | A redacted enumeration of @TabKind@ as it appears in the @/tab new@
  330 -- command. Local to this module to avoid an import cycle through
  331 -- 'PureClaw.Handles.Tab' (which imports this module for 'SlashCommand'
  332 -- in its 'TabUnsupportedCommand' constructor). WU2 introduces this type
  333 -- alongside the @/tab@ command family; downstream WUs that need a
  334 -- 'PureClaw.Handles.Tab.TabKind' translate via a trivial total
  335 -- conversion at the handler layer (WU9).
  336 data TabKindArg
  337   = TkaAi
  338   | TkaProvider
  339   | TkaHarness
  340   | TkaShell
  341   | TkaSsh
  342   | TkaTmux
  343   deriving stock (Show, Eq, Ord, Bounded, Enum)

  345 -- | Whether @/tab close N@ was passed the @--force@ flag.
  346 --
  347 -- For @KindAi@ tabs, @ForceYes@ skips the session archive on close
  348 -- (transcript deleted from disk). For non-AI tabs the close path is
  349 -- already destructive so the flag is a no-op semantically — the
  350 -- distinction is preserved here so 'executeSlashCommand' (WU9) can
  351 -- still echo what the user asked for.
  352 data ForceMode
  353   = ForceNo
  354   | ForceYes
  355   deriving stock (Show, Eq, Ord, Bounded, Enum)

  357 -- | The @/tab@ command family (introduced by WU2 of Tabbed Chat #51).
  358 --
  359 -- Tab indices are stored as plain 'Int' rather than as
  360 -- 'PureClaw.Handles.Tab.TabIndex' to avoid an import cycle. The parser
  361 -- ('PureClaw.Routing.Parse.parseInput') validates the index against
  362 -- @_rc_maxTabs@ via @mkTabIndex@ before constructing these values, so
  363 -- callers may treat the contained 'Int' as well-formed for the
  364 -- configured cap; downstream handlers (WU9) are still expected to
  365 -- re-wrap via @mkTabIndex@ when crossing into 'TabIndex'-typed APIs.
  366 --
  367 -- @TabResumeCmd@ carries the validated 'SessionId' produced by
  368 -- @mkSessionId@ (WU2 smart constructor).
  369 --
  370 -- /tmux-style packing update:/ @TabNewCmd@ no longer carries an
  371 -- explicit target index — new tabs are always allocated at the lowest
  372 -- free slot. The constructor's payload is therefore just the optional
  373 -- kind keyword and the optional argument-text remainder.
  374 data TabSlashCommand
  375   = TabNewCmd !(Maybe TabKindArg) !(Maybe Text)
  376     -- ^ @\/tab new [\<kind\> [\<arg-text\>]]@. Index is allocated at
  377     --   the lowest free slot by the handler (tmux-style packing). The
  378     --   second field is the remainder of the line after the kind,
  379     --   captured as a single 'Text' for the handler to split further.
  380   | TabListCmd
  381     -- ^ @\/tab list@ (and the @\/tabs@ alias).
  382   | TabCloseCmd !Int !ForceMode
  383     -- ^ @\/tab close \<N\> [--force]@. Remaining tabs are renumbered
  384     --   down by one starting at @N+1@ so the registry is always packed
  385     --   in the lowest slots (tmux @renumber-windows on@ model).
  386   | TabFocusCmd !Int
  387     -- ^ @\/tab focus \<N\>@ (functional alias of @\/N@).
  388   | TabResumeCmd !SessionId
  389     -- ^ @\/tab resume \<session-id\>@. Validation lives in
  390     --   @mkSessionId@; rejection surfaces as
  391     --   @ParseErrorInvalidSessionId@.
  392   | TabRenameCmd !Int !Text
  393     -- ^ @\/tab rename \<N\> \<name\>@. Parser captures the requested
  394     --   name verbatim; @sanitizeTabName@ runs at handler time per
  395     --   S10 so the user sees the rejection reason when applicable.
  396   deriving stock (Show, Eq)
