   47 -- | Errors from the OpenAI API.
   48 data OpenAIError

   60 openAIComplete :: OpenAIProvider -> CompletionRequest -> IO CompletionResponse
   61 openAIComplete provider req = do
   62   initReq <- HTTP.parseRequest (_oai_baseUrl provider)
   63   let httpReq = initReq
   64         { HTTP.method = "POST"
   65         , HTTP.requestBody = HTTP.RequestBodyLBS (encodeRequest req)
   66         , HTTP.requestHeaders =
   67             [ ("Authorization", "Bearer " <> withApiKey (_oai_apiKey provider) id)
   68             , ("content-type", "application/json")
   69             ]
   70         }
   71   resp <- HTTP.httpLbs httpReq (_oai_manager provider)
   72   let status = Status.statusCode (HTTP.responseStatus resp)
   73   if status /= 200
   74     then throwIO (OpenAIAPIError status (BL.toStrict (HTTP.responseBody resp)))
   75     else case decodeResponse (HTTP.responseBody resp) of
   76       Left err -> throwIO (OpenAIParseError (T.pack err))
   77       Right response -> pure response

   79 -- | Encode a completion request as OpenAI Chat Completions JSON.
   80 encodeRequest :: CompletionRequest -> BL.ByteString
   81 encodeRequest req = encode $ object $
   82   [ "model"    .= unModelId (_cr_model req)
   83   , "messages" .= encodeMessages req
   84   ]
   85   ++ maybe [] (\mt -> ["max_tokens" .= mt]) (_cr_maxTokens req)
   86   ++ ["tools" .= map encodeTool (_cr_tools req) | not (null (_cr_tools req))]
   87   ++ maybe [] (\tc -> ["tool_choice" .= encodeToolChoice tc]) (_cr_toolChoice req)
   88 
   89 -- | OpenAI puts system prompt as a system message in the messages array.
   90 encodeMessages :: CompletionRequest -> [Value]
   91 encodeMessages req =
   92   maybe [] (\s -> [object ["role" .= ("system" :: Text), "content" .= s]]) (_cr_systemPrompt req)
   93   ++ map encodeMsg (_cr_messages req)
   94 
   95 encodeMsg :: Message -> Value
   96 encodeMsg msg = case _msg_content msg of
   97   [TextBlock t] ->
   98     -- Simple text message — use string content for compatibility
   99     object ["role" .= roleToText (_msg_role msg), "content" .= t]
  100   blocks ->
  101     object [ "role"    .= roleToText (_msg_role msg)
  102            , "content" .= map encodeContentBlock blocks
  103            ]
  104 
  105 encodeContentBlock :: ContentBlock -> Value
  106 encodeContentBlock (TextBlock t) = object
  107   [ "type" .= ("text" :: Text), "text" .= t ]
  108 encodeContentBlock (ImageBlock mediaType imageData) = object
  109   [ "type" .= ("image_url" :: Text)
  110   , "image_url" .= object
  111       [ "url" .= ("data:" <> mediaType <> ";base64," <> TE.decodeUtf8 imageData) ]
  112   ]
  113 encodeContentBlock (ToolUseBlock callId name input) = object
  114   [ "type" .= ("function" :: Text)
  115   , "id"   .= unToolCallId callId
  116   , "function" .= object ["name" .= name, "arguments" .= TE.decodeUtf8 (BL.toStrict (encode input))]
  117   ]
  118 encodeContentBlock (ToolResultBlock callId parts _) = object
  119   [ "type"         .= ("tool_result" :: Text)
  120   , "tool_call_id" .= unToolCallId callId
  121   , "content"      .= T.intercalate "\n" [t | TRPText t <- parts]
  122   ]
  123 
  124 encodeTool :: ToolDefinition -> Value
  125 encodeTool td = object
  126   [ "type" .= ("function" :: Text)
  127   , "function" .= object
  128       [ "name"        .= _td_name td
  129       , "description" .= _td_description td
  130       , "parameters"  .= _td_inputSchema td
  131       ]
  132   ]
  133 
  134 encodeToolChoice :: ToolChoice -> Value
  135 encodeToolChoice AutoTool = String "auto"
  136 encodeToolChoice AnyTool = String "required"
  137 encodeToolChoice (SpecificTool name) = object
  138   [ "type" .= ("function" :: Text)
  139   , "function" .= object ["name" .= name]
  140   ]

  142 -- | Decode an OpenAI Chat Completions response.
  143 decodeResponse :: BL.ByteString -> Either String CompletionResponse
  144 decodeResponse bs = eitherDecode bs >>= parseEither parseResp
  145   where
  146     parseResp :: Value -> Parser CompletionResponse
  147     parseResp = withObject "OpenAIResponse" $ \o -> do
  148       choices <- o .: "choices"
  149       case choices of
  150         [] -> fail "No choices in response"
  151         (firstChoice : _) -> do
  152           msg <- firstChoice .: "message"
  153           blocks <- parseMessage msg
  154           modelText <- o .: "model"
  155           usageObj <- o .:? "usage"
  156           usage <- case usageObj of
  157             Nothing -> pure Nothing
  158             Just u -> do
  159               inToks <- u .: "prompt_tokens"
  160               outToks <- u .: "completion_tokens"
  161               pure (Just (Usage inToks outToks))
  162           pure CompletionResponse
  163             { _crsp_content = blocks
  164             , _crsp_model   = ModelId modelText
  165             , _crsp_usage   = usage
  166             }
  167 
  168     parseMessage :: Value -> Parser [ContentBlock]
  169     parseMessage = withObject "Message" $ \m -> do
  170       contentVal <- m .:? "content"
  171       toolCalls <- m .:? "tool_calls" .!= ([] :: [Value])
  172       let textBlocks = case contentVal of
  173             Just (String t) | not (T.null t) -> [TextBlock t]
  174             _ -> []
  175       toolBlocks <- mapM parseToolCall toolCalls
  176       pure (textBlocks ++ toolBlocks)
  177 
  178     parseToolCall :: Value -> Parser ContentBlock
  179     parseToolCall = withObject "ToolCall" $ \tc -> do
  180       callId <- tc .: "id"
  181       fn <- tc .: "function"
  182       name <- fn .: "name"
  183       argsStr <- fn .: "arguments"
  184       let input = fromMaybe (object []) (decode (BL.fromStrict (TE.encodeUtf8 argsStr)))
  185       pure (ToolUseBlock (ToolCallId callId) name input)

  187 -- | OpenAI Models listing endpoint.
  188 openAIModelsUrl :: String
  189 openAIModelsUrl = "https://api.openai.com/v1/models"
  190 
  191 -- | List available models for the authenticated OpenAI account, filtered
  192 -- to chat-completion-eligible IDs. Returns an empty list on any error.
  193 openAIListModels :: OpenAIProvider -> IO [ModelId]
  194 openAIListModels provider = do
  195   result <- try @SomeException $ do
  196     initReq <- HTTP.parseRequest openAIModelsUrl
  197     let httpReq = initReq
  198           { HTTP.method = "GET"
  199           , HTTP.requestHeaders =
  200               [ ("Authorization", "Bearer " <> withApiKey (_oai_apiKey provider) id)
  201               , ("content-type", "application/json")
  202               ]
  203           , HTTP.responseTimeout = HTTP.responseTimeoutMicro (30 * 1000000)
  204           }
  205     resp <- HTTP.httpLbs httpReq (_oai_manager provider)
  206     let status = Status.statusCode (HTTP.responseStatus resp)
  207     if status /= 200
  208       then pure []
  209       else case eitherDecode (HTTP.responseBody resp) of
  210         Left  _   -> pure []
  211         Right val -> pure (filter (isChatEligibleOpenAIModel . unModelId) (parseOpenAIModelIds val))
  212   case result of
  213     Left  _   -> pure []
  214     Right ids -> pure ids
  215 
  216 -- | Extract model IDs from an OpenAI /v1/models response body.
  217 -- Expected shape: @{"object":"list","data":[{"id":"gpt-4o","object":"model",...},...]}@
  218 parseOpenAIModelIds :: Value -> [ModelId]
  219 parseOpenAIModelIds = fromMaybe [] . parseMaybe parseList
  220   where
  221     parseList :: Value -> Parser [ModelId]
  222     parseList = withObject "OpenAIModelsResponse" $ \o -> do
  223       arr <- o .: "data"
  224       mapM (withObject "Model" (\m -> ModelId <$> m .: "id")) arr

  226 -- | Filter to chat-completion-eligible OpenAI model IDs.
  227 -- The /v1/models endpoint returns hundreds of entries spanning embeddings,
  228 -- audio, image, moderation, and fine-tuning models — most of which can't
  229 -- be used with /chat/completions. We keep IDs that start with a known
  230 -- chat-family prefix and don't contain a known non-chat substring.
  231 isChatEligibleOpenAIModel :: Text -> Bool
  232 isChatEligibleOpenAIModel mid =
  233   any (`T.isPrefixOf` mid) chatPrefixes
  234     && not (any (`T.isInfixOf` mid) nonChatSubstrings)
  235   where
  236     chatPrefixes = ["gpt-", "o1", "o3", "o4", "chatgpt-"]
  237     nonChatSubstrings =
  238       [ "-instruct"
  239       , "audio"
  240       , "realtime"
  241       , "tts"
  242       , "whisper"
  243       , "dall-e"
  244       , "embedding"
  245       , "moderation"
  246       , "search-preview"
  247       , "transcribe"
  248       , "image"
  249       ]
