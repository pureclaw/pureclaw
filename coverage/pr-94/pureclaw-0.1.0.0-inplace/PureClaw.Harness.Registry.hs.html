    1 -- | Durable harness identity registry (Harness Registry & Lifecycle, Phase 1).
    2 --
    3 -- This is the source of truth for harness identity and health. A harness is
    4 -- keyed by a PureClaw-assigned 'HarnessId' (a UUID), NOT by its mutable tmux
    5 -- window name (which the user can rename) — see @docs\/harness-registry.md@ §3.
    6 --
    7 -- The registry itself is a 'TVar' (STM) because a background reconcile loop
    8 -- performs compound read-modify-write concurrently with HTTP\/slash handlers
    9 -- (design decision K2b). The critical invariant is that 'mergeReconcile' merges
   10 -- tmux-observed fields into existing entries /by key/ inside a single
   11 -- 'atomically', recomputing from the CURRENT 'TVar' contents — so an entry
   12 -- inserted by another transaction between a naive read and write is never
   13 -- clobbered (the lost-update-safe path, §4).
   14 --
   15 -- Leaf module: it may import 'PureClaw.Handles.Harness' (for 'HarnessHandle')
   16 -- and standard libraries only — never @Agent@\/@Frontend@\/@CLI@.
   17 module PureClaw.Harness.Registry
   18   ( -- * Identity (re-exported from PureClaw.Harness.Id)
   19     HarnessId (..)
   20   , newHarnessId
   21   , parseHarnessId
   22   , harnessIdToText
   23     -- * Entry types
   24   , HarnessOrigin (..)
   25   , Liveness (..)
   26   , HarnessEntry (..)
   27     -- * The registry

   61 -- Entry types

   64 -- | How PureClaw came to manage this harness.
   65 --
   66 -- Phase 1 produces 'OriginSpawned' (we launched it) and 'OriginDiscovered'
   67 -- (boot-reconstructed from a window carrying our @\@pcl_id@). 'OriginAdopted'
   68 -- (an external window we took over) is defined here but exercised in Phase 3.
   69 data HarnessOrigin
   70   = OriginSpawned
   71   | OriginDiscovered
   72   | OriginAdopted
   73   deriving stock (Eq, Show)
   74 
   75 -- | Liveness of a harness (design §5 / decision K7).
   76 --
   77 -- Note that @ExternallyModified@ and @Unknown@ are intentionally NOT liveness
   78 -- states: the former is the orthogonal '_he_extModified' flag, and the latter
   79 -- maps to the '_he_stale' flag (hold last-known liveness).
   80 data Liveness
   81   = LivenessIdle           -- ^ Window present, harness running, not actively working.
   82   | LivenessThinking       -- ^ Harness actively working (screen-capture heuristic).
   83   | LivenessAwaitingInput  -- ^ Window present, harness blocked on an approval/menu prompt.
   84   | LivenessExited         -- ^ Harness PID gone or @pane_dead@, window still present.
   85   | LivenessOrphaned       -- ^ No live window+PID for this id.
   86   deriving stock (Eq, Show)
   87 
   88 -- | A registry entry: the durable identity plus the reconciled health\/coordinate
   89 -- cache (design §4). The cached @(session, windowName)@ coordinate lets handles
   90 -- target tmux without a per-I\/O @\@pcl_id@ sweep; the reconcile loop is the sole
   91 -- owner that refreshes it.
   92 --
   93 -- Note: '_he_handle' is a 'HarnessHandle', which is a record of @IO@ actions and
   94 -- therefore has no 'Eq'\/'Show'. Consequently 'HarnessEntry' does NOT derive
   95 -- 'Eq'\/'Show'; tests assert via field projections instead.
   96 data HarnessEntry = HarnessEntry
   97   { _he_id          :: !HarnessId
   98     -- ^ Canonical identity (the map key).
   99   , _he_session     :: !Text
  100     -- ^ Cached tmux session name (coordinate; revalidated by reconcile, K3).
  101   , _he_windowName  :: !Text
  102     -- ^ Cached tmux window name (coordinate; revalidated by reconcile, K3).
  103   , _he_shellPid    :: !(Maybe Int)
  104     -- ^ Recorded pane shell PID (@#{pane_pid}@) — provenance + liveness.
  105   , _he_harnessPid  :: !(Maybe Int)
  106     -- ^ Recorded harness-process PID (the agent binary) — trust anchor.
  107   , _he_origin      :: !HarnessOrigin
  108     -- ^ How PureClaw came to manage this harness.
  109   , _he_flavour     :: !Kind.HarnessFlavour
  110     -- ^ The harness tool flavour (Claude Code, Codex, …). Selects the
  111     --   per-flavour 'PureClaw.Harness.Observer.HarnessObserver' the reconcile
  112     --   loop uses to classify the screen capture.
  113   , _he_liveness    :: !Liveness
  114     -- ^ Current reconciled liveness.
  115   , _he_extModified :: !Bool
  116     -- ^ Orthogonal flag: a matched window's name\/session changed out-of-band.
  117   , _he_stale       :: !Bool
  118     -- ^ The last sweep\/capture failed; hold last-known liveness, don't repaint.
  119   , _he_sessionId   :: !(Maybe Text)
  120     -- ^ The PureClaw session this harness backs (the join to @session.json@).
  121   , _he_label       :: !Text
  122     -- ^ Legacy window-name label, used for name->id resolution ('lookupByLabel').
  123   , _he_orphanedTicks :: !Int
  124     -- ^ Consecutive reconcile ticks this entry has been classified Orphaned.
  125     --   0 whenever the entry is live (Idle\/Thinking\/Exited); the reconcile
  126     --   loop increments it each tick the corroborated window stays absent and
  127     --   auto-evicts the entry once it reaches 'PureClaw.Harness.Reconcile.defaultOrphanGraceTicks'
  128     --   (the wall-clock-free grace policy, design §5\/§10 Q2). Eviction drops
  129     --   the entry from the registry + the legacy harness map but never touches
  130     --   @session.json@ (the session reappears in Recent Sessions).
  131   , _he_handle      :: !(Maybe HarnessHandle)
  132     -- ^ The live handle, if we have one. 'Nothing' for a boot-discovered entry
  133     --   that has not yet had a handle attached.
  134   }

  136 -- ---------------------------------------------------------------------------
  137 -- The registry
  138 -- ---------------------------------------------------------------------------

  140 -- | The durable harness registry: a strict 'Map' from 'HarnessId' to
  141 -- 'HarnessEntry', held in a 'TVar' for safe concurrent read-modify-write.
  142 newtype HarnessRegistry = HarnessRegistry
  143   { unHarnessRegistry :: TVar (Map HarnessId HarnessEntry) }
  144 
  145 -- | Create a new, empty registry.
  146 newRegistry :: IO HarnessRegistry
  147 newRegistry = HarnessRegistry <$> newTVarIO Map.empty
  148 
  149 -- ---------------------------------------------------------------------------
  150 -- CRUD
  151 -- ---------------------------------------------------------------------------
  152 
  153 -- | Insert (or overwrite) an entry by its 'HarnessId'.
  154 insertEntry :: HarnessRegistry -> HarnessEntry -> IO ()
  155 insertEntry reg e =
  156   atomically (modifyTVar' (unHarnessRegistry reg) (Map.insert (_he_id e) e))
  157 
  158 -- | Look up an entry by its canonical id.
  159 lookupById :: HarnessRegistry -> HarnessId -> IO (Maybe HarnessEntry)
  160 lookupById reg hid =
  161   atomically (Map.lookup hid <$> readTVar (unHarnessRegistry reg))
  162 
  163 -- | Resolve an entry by its label (legacy window-name). Returns the first
  164 -- match (label is not guaranteed unique on a shared server; the canonical key
  165 -- is the id). Used by name->id resolution for legacy\/CLI routing surfaces.
  166 lookupByLabel :: HarnessRegistry -> Text -> IO (Maybe HarnessEntry)
  167 lookupByLabel reg lbl = do
  168   m <- readTVarIO (unHarnessRegistry reg)
  169   pure (firstMatch (Map.elems m))
  170   where
  171     firstMatch []       = Nothing
  172     firstMatch (e : es)
  173       | _he_label e == lbl = Just e
  174       | otherwise          = firstMatch es
  175 
  176 -- | Remove an entry by id (no-op if absent).
  177 deleteEntry :: HarnessRegistry -> HarnessId -> IO ()
  178 deleteEntry reg hid =
  179   atomically (modifyTVar' (unHarnessRegistry reg) (Map.delete hid))
  180 
  181 -- | Apply a pure function to the entry with the given id, in-place, as a
  182 -- SINGLE-STM read-modify-write (no-op if the id is absent). This is the
  183 -- race-safe way to flip a flag on one entry (e.g. clearing '_he_extModified'
  184 -- for Acknowledge): unlike a 'lookupById' followed by 'insertEntry' — two
  185 -- separate transactions — the whole adjust commits atomically, so a concurrent
  186 -- 'mergeReconcile' tick cannot interleave and clobber the change (a lost
  187 -- update). 'Map.adjust' touches only the named key, leaving every other entry
  188 -- intact.
  189 modifyEntry :: HarnessRegistry -> HarnessId -> (HarnessEntry -> HarnessEntry) -> IO ()
  190 modifyEntry reg hid f =
  191   atomically (modifyTVar' (unHarnessRegistry reg) (Map.adjust f hid))
  192 
  193 -- | Snapshot all current entries.
  194 snapshot :: HarnessRegistry -> IO [HarnessEntry]
  195 snapshot reg =
  196   atomically (Map.elems <$> readTVar (unHarnessRegistry reg))
  197 
  198 -- ---------------------------------------------------------------------------
  199 -- Reconciliation
  200 -- ---------------------------------------------------------------------------
  201 
  202 -- | The tmux-observed fields the reconcile loop merges into an existing entry,
  203 -- keyed by 'HarnessId'. This is deliberately a small projection of
  204 -- 'HarnessEntry' — it carries only what a server sweep observes, never the
  205 -- handle or the durable provenance/origin, which the registry owns.
  206 data ObservedHarness = ObservedHarness
  207   { _oh_id          :: !HarnessId
  208   , _oh_session     :: !Text
  209   , _oh_windowName  :: !Text
  210   , _oh_shellPid    :: !(Maybe Int)
  211   , _oh_harnessPid  :: !(Maybe Int)
  212   , _oh_liveness    :: !Liveness
  213   , _oh_extModified :: !Bool
  214   , _oh_stale       :: !Bool
  215   , _oh_orphanedTicks :: !Int
  216     -- ^ The entry's new consecutive-orphaned-tick count for this observation:
  217     --   @old + 1@ when classified Orphaned, @0@ when live. Rides the
  218     --   'mergeReconcile' path so the counter lives durably on the entry.
  219   }
  220   deriving stock (Eq, Show)
  221 
  222 -- | Merge tmux-observed fields into existing entries BY KEY, inside a single
  223 -- 'atomically'. This is the lost-update-safe reconcile path (design §4):
  224 --
  225 --   * The new map is computed from the CURRENT 'TVar' contents inside the
  226 --     transaction, so a concurrent insert that landed before this transaction
  227 --     commits is preserved.
  228 --   * Only keys present in the observed set are updated; every other key
  229 --     (including newly-inserted ones) is left intact.
  230 --   * An observed row whose id is NOT already in the registry is ignored — an
  231 --     observation alone never resurrects\/creates an entry (registration is the
  232 --     caller's job; this defends against, e.g., a spoofed marker).
  233 mergeReconcile :: HarnessRegistry -> [ObservedHarness] -> IO ()
  234 mergeReconcile reg observed =
  235   atomically (modifyTVar' (unHarnessRegistry reg) merge)
  236   where
  237     obsMap :: Map HarnessId ObservedHarness
  238     obsMap = Map.fromList [ (_oh_id o, o) | o <- observed ]
  239 
  240     -- Map over the CURRENT contents, updating only keys we observed and
  241     -- leaving all others (incl. concurrently-inserted) untouched.
  242     merge :: Map HarnessId HarnessEntry -> Map HarnessId HarnessEntry
  243     merge = Map.mapWithKey applyObserved
  244 
  245     applyObserved :: HarnessId -> HarnessEntry -> HarnessEntry
  246     applyObserved hid e =
  247       case Map.lookup hid obsMap of
  248         Nothing -> e
  249         Just o  -> e
  250           { _he_session     = _oh_session o
  251           , _he_windowName  = _oh_windowName o
  252           , _he_shellPid    = _oh_shellPid o
  253           , _he_harnessPid  = _oh_harnessPid o
  254           , _he_liveness    = _oh_liveness o
  255           , _he_extModified = _oh_extModified o
  256           , _he_stale       = _oh_stale o
  257           , _he_orphanedTicks = _oh_orphanedTicks o
  258           }
