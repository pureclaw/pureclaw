   42 -- ---------------------------------------------------------------------------
   43 -- TabSnapshot
   44 -- ---------------------------------------------------------------------------
   45 
   46 -- | A point-in-time snapshot of a single tab, pre-resolved to
   47 -- JSON-friendly text values. The snapshot callback in 'FrontendEnv'
   48 -- produces these; the API layer simply serializes them.
   49 data TabSnapshot = TabSnapshot
   50   { _ts_index     :: !Int
   51   , _ts_kind      :: !Text
   52     -- ^ @\"provider\"@, @\"harness\"@, or @\"raw_shell\"@.
   53   , _ts_label     :: !(Maybe Text)
   54     -- ^ Harness label fallback so a harness tab (whose session id can be
   55     -- 'Nothing') never renders blank. 'Nothing' for session-backed tabs —
   56     -- their display title derives from the session (via '_ts_sessionId').
   57   , _ts_status    :: !Text
   58     -- ^ Liveness word: @\"running\"@, @\"idle\"@, @\"exited\"@, or
   59     -- @\"orphaned\"@ (Phase 2 split Exited\/Orphaned; see
   60     -- 'livenessToTabStatus').
   61   , _ts_sessionId :: !(Maybe Text)
   62     -- ^ Session ID for session-backed tabs; 'Nothing' for raw shells.
   63   , _ts_extModified :: !Bool
   64     -- ^ The harness window was renamed out-of-band (the §7 ⚠ \"edited\"
   65     -- pill). Orthogonal to liveness.
   66   , _ts_stale :: !Bool
   67     -- ^ Health could not be refreshed this cycle; the frontend holds the
   68     -- last-known icon with a dimmed cue (§7).
   69   , _ts_origin :: !Text
   70     -- ^ How the harness entered the registry: @\"spawned\"@,
   71     -- @\"discovered\"@, or @\"adopted\"@ (the §7 origin pill).
   72   , _ts_attachCommand :: !(Maybe Text)
   73     -- ^ Copyable @tmux attach@ command for live harness rows;
   74     -- 'Nothing' for non-harness tabs.
   75   }
   76   deriving stock (Show, Eq)
   77 
   78 -- | Serialize a 'TabSnapshot'. The per-tab @name@ key is GONE — a tab's
   79 -- display title now derives from its bound session (frontend resolves it from
   80 -- @session_id@). Harness tabs keep a @label@ fallback so they never render
   81 -- blank. The Phase-2 health fields (@ext_modified@\/@stale@\/@origin@\/
   82 -- @attach_command@) remain. Consumers that ignore unknown keys keep working.
   83 instance ToJSON TabSnapshot where
   84   toJSON ts = object
   85     [ "index"          .= _ts_index ts
   86     , "kind"           .= _ts_kind ts
   87     , "label"          .= _ts_label ts
   88     , "status"         .= _ts_status ts
   89     , "session_id"     .= _ts_sessionId ts
   90     , "ext_modified"   .= _ts_extModified ts
   91     , "stale"          .= _ts_stale ts
   92     , "origin"         .= _ts_origin ts
   93     , "attach_command" .= _ts_attachCommand ts
   94     ]

   96 -- ---------------------------------------------------------------------------
   97 -- Vocabulary helpers
   98 -- ---------------------------------------------------------------------------
   99 
  100 -- | Map a registry 'Registry.Liveness' to the @TabSnapshot@ status vocabulary.
  101 -- Phase 2 (§7) SPLITS the former \"crashed\" bucket: 'Registry.LivenessExited'
  102 -- (the process exited; offer Restart\/Dismiss) and 'Registry.LivenessOrphaned'
  103 -- (the window vanished out-of-band; greyed, offer Dismiss) now map to distinct
  104 -- words so the frontend can render the state→visual table.
  105 livenessToTabStatus :: Registry.Liveness -> Text
  106 livenessToTabStatus lv = case lv of
  107   Registry.LivenessIdle          -> "idle"
  108   Registry.LivenessThinking      -> "running"
  109   Registry.LivenessAwaitingInput -> "running"  -- distinct state shown via the activity dot, not the tab badge
  110   Registry.LivenessExited        -> "exited"
  111   Registry.LivenessOrphaned      -> "orphaned"
  112 
  113 -- | Map a registry 'Registry.HarnessOrigin' to the @TabSnapshot@ origin
  114 -- vocabulary (the §7 origin pill): @\"spawned\"@ (we launched it),
  115 -- @\"discovered\"@ (boot-reconstructed from a tagged window), or
  116 -- @\"adopted\"@ (taken over from another controller).
  117 harnessOriginToText :: Registry.HarnessOrigin -> Text
  118 harnessOriginToText o = case o of
  119   Registry.OriginSpawned    -> "spawned"
  120   Registry.OriginDiscovered -> "discovered"
  121   Registry.OriginAdopted    -> "adopted"
  122 
  123 -- ---------------------------------------------------------------------------
  124 -- Pure projection
  125 -- ---------------------------------------------------------------------------
  126 
  127 -- | Project the tab registry and the harness-entry lookup into a list of
  128 -- 'TabSnapshot' values, one per tab, in slot order.
  129 --
  130 -- The lookup argument is injected so this function is fully pure and
  131 -- testable without IO:
  132 --
  133 -- * @harnOf hid@ — look up the live 'Registry.HarnessEntry' for a harness-
  134 --   backed tab. 'Nothing' means the entry has vanished from the registry
  135 --   since the tab was created; the tab is projected as @exited\/stale@.
  136 --
  137 -- Provider tabs ('BoundSession') are projected purely from the 'Tab' record
  138 -- itself — no session-meta lookup is needed. This is a structural guarantee:
  139 -- the function takes no session-meta input, so no session metadata
  140 -- (@_sm_source@, channelUserId, credentials, etc.) can appear in a
  141 -- 'TabSnapshot' by construction.
  142 tabSnapshotsFromRegistry
  143   :: TabList
  144   -> (Registry.HarnessId -> Maybe Registry.HarnessEntry)
  145   -> [TabSnapshot]
  146 tabSnapshotsFromRegistry tl harnOf = map project (toList tl)
  147   where
  148     project :: Tab -> TabSnapshot
  149     project tab = case _tab_ref tab of
  150       BoundSession sid ->
  151         TabSnapshot
  152           { _ts_index         = unTabIndex (_tab_slot tab)
  153           , _ts_kind          = "provider"
  154           , _ts_label         = Nothing
  155           , _ts_status        = case _tab_status tab of
  156                                   Live -> "idle"
  157                                   Dead -> "exited"
  158           , _ts_sessionId     = Just (unSessionId sid)
  159           , _ts_extModified   = False
  160           , _ts_stale         = False
  161           , _ts_origin        = ""
  162           , _ts_attachCommand = Nothing
  163           }
  164       BoundHarness hid ->
  165         case harnOf hid of
  166           Just e ->
  167             TabSnapshot
  168               { _ts_index         = unTabIndex (_tab_slot tab)
  169               , _ts_kind          = "harness"
  170               , _ts_label         = Just (Registry._he_label e)
  171               , _ts_status        = livenessToTabStatus (Registry._he_liveness e)
  172               , _ts_sessionId     = Registry._he_sessionId e
  173               , _ts_extModified   = Registry._he_extModified e
  174               , _ts_stale         = Registry._he_stale e
  175               , _ts_origin        = harnessOriginToText (Registry._he_origin e)
  176               , _ts_attachCommand = Just
  177                   ("tmux attach -t "
  178                     <> Registry._he_session e
  179                     <> ":"
  180                     <> Registry._he_windowName e)
  181               }
  182           Nothing ->
  183             TabSnapshot
  184               { _ts_index         = unTabIndex (_tab_slot tab)
  185               , _ts_kind          = "harness"
  186               , _ts_label         = Nothing
  187               , _ts_status        = "exited"
  188               , _ts_sessionId     = Nothing
  189               , _ts_extModified   = False
  190               , _ts_stale         = True
  191               , _ts_origin        = ""
  192               , _ts_attachCommand = Nothing
  193               }
