    1 -- | Leaf module for 'HarnessId' — a UUID-backed harness identity.
    2 --
    3 -- This module exists purely to break the import cycle between
    4 -- 'PureClaw.Harness.Registry' (which defines 'HarnessEntry' and may import
    5 -- 'PureClaw.Session.Kind') and 'PureClaw.Session.Kind' (which embeds
    6 -- 'HarnessId' in 'HarnessSpec'). By isolating 'HarnessId' here, both modules
    7 -- can import this leaf without creating a cycle.
    8 --
    9 -- Depends ONLY on @uuid@, @text@, @aeson@, and @base@. Never imports any
   10 -- project module.
   11 module PureClaw.Harness.Id
   12   ( -- * Identity
   13     HarnessId (..)
   14   , parseHarnessId
   15   , harnessIdToText
   16   ) where
   17 
   18 import Data.Aeson (FromJSON (..), ToJSON (..))
   19 import Data.Aeson qualified as Aeson
   20 import Data.Text (Text)
   21 import Data.UUID (UUID)
   22 import Data.UUID qualified as UUID

   24 -- | A PureClaw-assigned, UUID-backed harness identity. This is the canonical
   25 -- key for the registry and the durable anchor that survives tmux window
   26 -- rename\/move and PureClaw restart (persisted in @session.json@).
   27 newtype HarnessId = HarnessId { unHarnessId :: UUID }
   28   deriving stock (Eq, Ord, Show)
   29 
   30 -- | Parse a 'HarnessId' from its canonical UUID text representation.
   31 -- Returns 'Nothing' for any non-UUID input.
   32 parseHarnessId :: Text -> Maybe HarnessId
   33 parseHarnessId = fmap HarnessId . UUID.fromText
   34 
   35 -- | Render a 'HarnessId' as its canonical UUID text representation.
   36 harnessIdToText :: HarnessId -> Text
   37 harnessIdToText = UUID.toText . unHarnessId
   38 
   39 -- | Round-trippable JSON: a 'HarnessId' is encoded as the canonical UUID
   40 -- string (D2.4). We hand-write the codec rather than @deriving newtype@ so the
   41 -- on-the-wire shape (a plain string) is explicit and decode rejects malformed
   42 -- UUIDs with a clear error.
   43 instance ToJSON HarnessId where
   44   toJSON = Aeson.String . harnessIdToText
   45 
   46 instance FromJSON HarnessId where
   47   parseJSON = Aeson.withText "HarnessId" $ \t ->
   48     case parseHarnessId t of
   49       Just hid -> pure hid
   50       Nothing  -> fail ("invalid HarnessId (not a UUID): " <> show t)
