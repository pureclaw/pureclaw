    3 -- |
    4 -- Module      : PureClaw.Tabs.Runtimes
    5 -- Description : The real per-'TabRef' runtime constructors (Tabs-as-View 8c.1).
    6 --
    7 -- "PureClaw.Tabs.Exec" owns a refcounted registry of started
    8 -- @'Runtime' { '_rt_send', '_rt_stop' }@ values, one per live 'TabRef', but
    9 -- defers their CONSTRUCTION to an injected seam (@Exec._ex_startRuntime@). This
   10 -- module supplies the two real constructors that seam is wired to at the 8c.2
   11 -- flip:
   12 --
   13 --   * 'mkProviderRuntime' — an LLM-backed session. A single forked worker
   14 --     serialises turns so it stays the /only/ writer of the session 'Context'
   15 --     (invariant E5). Input text arrives on a bounded queue; each turn runs the
   16 --     env-light 'PureClaw.Agent.Turn.runTurnWithTools' and streams output to an
   17 --     injected ref-tagged sink.
   18 --   * 'mkHarnessRuntime' — an external tmux harness. A forked /writer/ drains
   19 --     queued input to @_hh_send@; a forked /drainer/ polls @_hh_receive@,
   20 --     sanitises each non-empty capture, and emits it as a 'FullMsg'.
   21 --
   22 -- == Injected seams (no live LLM / tmux here)
   23 --
   24 -- Every effect is a field of 'ProviderRuntimeDeps' \/ 'HarnessRuntimeDeps' —
   25 -- the provider stream, the tool executor, the transcript append, the context
   26 -- seed, the 'HarnessHandle', and the fork primitive (the @_env_fork@ seam). So
   27 -- both constructors unit-test fully with fakes (see @test\/Tabs\/RuntimesSpec@):
   28 -- a scripted stream, a recording harness handle, a recording emit sink, and a
   29 -- real async fork, synchronised by 'MVar' handshakes rather than sleeps.
   30 --
   31 -- == TurnEvent -> ChannelEvent mapping
   32 --
   33 -- 'runTurnWithTools' emits the env-light 'PureClaw.Agent.Turn.TurnEvent'
   34 -- (start\/chunk\/end framed by a 'StreamId', carrying no slot identity). The
   35 -- provider worker maps each to a 'ChannelEvent' and tags it with the runtime's
   36 -- own 'TabRef' before calling '_prd_emit':
   37 --
   38 -- @
   39 --   'TurnStart' sid    -> 'StreamStart' sid placeholderSlot
   40 --   'TurnChunk' sid t   -> 'ChunkOf' sid t
   41 --   'TurnEnd'   sid     -> 'StreamEnd' sid
   42 --   'TurnError' _sid t  -> 'FullMsg' placeholderSlot t
   43 -- @
   44 --
   45 -- 'TurnError' carries a provider-failure message (pureclaw-9d0); it maps to a
   46 -- whole-message 'FullMsg' (not stream framing) so the error reaches every
   47 -- channel — including the non-streaming Signal\/Telegram path.
   48 --
   49 -- The 'Tab.TabIndex' embedded in 'StreamStart' is __vestigial__: the relay now
   50 -- routes purely by 'TabRef' (spike §4), so the slot is a fixed placeholder
   51 -- (@'Tab.mkTabIndex' 0@). Downstream consumers must not read it.
   52 --
   53 -- == Stop semantics
   54 --
   55 -- '_rt_stop' cancels the runtime's worker\/drainer\/writer threads and is
   56 -- idempotent (it forwards to the async-cancel '_trun_cancel', itself
   57 -- idempotent) and never throws. For a harness, '_rt_stop' deliberately does
   58 -- __not__ call @_hh_stop@: detaching a tab leaves the harness running in tmux,
   59 -- re-attachable later (spike §6).
   60 module PureClaw.Tabs.Runtimes
   61   ( -- * Provider-session runtime
   62     ProviderRuntimeDeps (..)
   63   , mkProviderRuntime
   64     -- * Harness runtime
   65   , HarnessRuntimeDeps (..)
   66   , mkHarnessRuntime
   67   ) where

  114 -- | The injected effects 'mkProviderRuntime' needs. Production wires:
  115 --
  116 --   * '_prd_emit'    — tag + enqueue onto the relay-writer queue.
  117 --   * '_prd_stream'  — @'P.completeStream' someProvider@ (transcript-wrapped).
  118 --   * '_prd_execTool'— a closure over the session's tool registry.
  119 --   * '_prd_record'  — @messageToEntry >>> _th_record@ for the session.
  120 --   * '_prd_seedCtx' — @loadRecentMessages + replaceMessages@ (or 'emptyContext').
  121 --   * '_prd_fork'    — the @_env_fork@ seam.
  122 --
  123 -- Tests inject fakes for all of these.
  124 data ProviderRuntimeDeps = ProviderRuntimeDeps
  125   { _prd_ref          :: TabRef
  126     -- ^ This runtime's tab ref (a @BoundSession@). Every emitted event is
  127     --   tagged with it.
  128   , _prd_emit         :: TabRef -> ChannelEvent -> IO ()
  129     -- ^ Output sink: tags + enqueues to the relay-writer queue.
  130   , _prd_stream       :: P.CompletionRequest -> (P.StreamEvent -> IO ()) -> IO ()
  131     -- ^ Provider stream (prod: @'P.completeStream' someProvider@).
  132   , _prd_execTool     :: ToolCallId -> Text -> Value -> IO P.Message
  133     -- ^ Injected tool exec — runs one call, yields a tool-result message.
  134   , _prd_record       :: P.Message -> IO ()
  135     -- ^ Per-session transcript append.
  136   , _prd_seedCtx      :: IO Context
  137     -- ^ Seed the conversation 'Context' (prod: from the session transcript).
  138   , _prd_model        :: ModelId
  139   , _prd_systemPrompt :: Maybe Text
  140   , _prd_tools        :: IO [P.ToolDefinition]
  141     -- ^ Advertised tool definitions, read PER TURN (so tools connected after
  142     --   the tab was created — e.g. via @\/mcp connect@ — are still advertised
  143     --   to the LLM, matching the legacy loop's per-turn @effectiveRegistry@).
  144   , _prd_maxTokens    :: Maybe Int
  145   , _prd_onStreamDone :: IO ()
  146     -- ^ Fired on EVERY provider 'P.StreamDone' (threaded into 'TurnDeps' as
  147     --   '_turn_onStreamDone'). Production wires a read-and-clear @fireOnce@ over
  148     --   @_env_onFirstStreamDone@ so @'markBootstrapConsumed'@ runs exactly once
  149     --   (pureclaw-8g4); idempotency is the action's responsibility.
  150   , _prd_fork         :: IO () -> IO Tab.TabRunner
  151     -- ^ The @_env_fork@ seam.
  152   , _prd_inputBound   :: Int
  153     -- ^ Input queue capacity.
  154   }
  155 
  156 -- | Build + start a provider-session runtime.
  157 --
  158 -- Seeds the 'Context' via '_prd_seedCtx' into an 'IORef' owned by the single
  159 -- forked worker (so the worker is the sole Context writer). The worker loops
  160 -- forever reading one user text off a bounded 'TBQueue', running
  161 -- 'runTurnWithTools', and writing the resulting 'Context' back. '_rt_send'
  162 -- non-blockingly enqueues onto that queue (returns @'Left' …@ when full, like
  163 -- the legacy per-tab loop); '_rt_stop' cancels the worker.
  164 mkProviderRuntime :: ProviderRuntimeDeps -> IO Runtime
  165 mkProviderRuntime deps = do
  166   ctx0   <- _prd_seedCtx deps
  167   ctxRef <- newIORef ctx0
  168   inputQ <- newTBQueueIO (fromIntegral (max 1 (_prd_inputBound deps)))
  169   sidRef <- newIORef (0 :: Word64)
  170   let mkTurnDeps tools = TurnDeps
  171         { _turn_stream       = _prd_stream deps
  172         , _turn_execTool     = _prd_execTool deps
  173         , _turn_emit         = emitTurnEvent deps
  174         , _turn_record       = _prd_record deps
  175         , _turn_nextStreamId = nextStreamId sidRef
  176         , _turn_model        = _prd_model deps
  177         , _turn_systemPrompt = _prd_systemPrompt deps
  178         , _turn_tools        = tools
  179         , _turn_maxTokens    = _prd_maxTokens deps
  180         , _turn_onStreamDone = _prd_onStreamDone deps
  181         }
  182       worker = forever $ do
  183         userText <- atomically (readTBQueue inputQ)
  184         -- Read the advertised tools PER TURN so tools connected after this tab
  185         -- was created (e.g. via @/mcp connect@) are still seen (pureclaw-2u4).
  186         tools <- _prd_tools deps
  187         ctx   <- readIORef ctxRef
  188         ctx'  <- runTurnWithTools (mkTurnDeps tools) ctx userText
  189         writeIORef ctxRef ctx'
  190   runner <- _prd_fork deps worker
  191   pure Runtime
  192     { _rt_send = \t -> atomically $ do
  193         full <- isFullTBQueue inputQ
  194         if full
  195           then pure (Left (Tab.TabConcurrencyLimit 0))
  196           else do
  197             writeTBQueue inputQ t
  198             pure (Right ())
  199     , _rt_stop = Tab._trun_cancel runner
  200     }

  202 -- | Map one env-light 'TurnEvent' to a ref-tagged 'ChannelEvent' and emit it.
  203 --
  204 -- The 'Tab.TabIndex' in 'StreamStart' is the vestigial 'placeholderSlot' (the
  205 -- relay routes by 'TabRef', not slot).
  206 emitTurnEvent :: ProviderRuntimeDeps -> TurnEvent -> IO ()
  207 emitTurnEvent deps = \case
  208   TurnStart sid   -> emit (StreamStart sid placeholderSlot)
  209   TurnChunk sid t -> emit (ChunkOf sid t)
  210   TurnEnd   sid   -> emit (StreamEnd sid)
  211   -- A provider failure: relay the legacy "Something went wrong" text as a
  212   -- whole-message 'FullMsg' (pureclaw-9d0). 'FullMsg' reaches every channel
  213   -- via @_ch_send@ — including the non-streaming Signal\/Telegram path (ao9) —
  214   -- whereas the stream framing only reaches streaming sinks.
  215   TurnError _sid t -> emit (FullMsg placeholderSlot t)
  216   where
  217     emit = _prd_emit deps (_prd_ref deps)
  218 
  219 -- | The vestigial slot stamped into 'StreamStart'. Built via the
  220 -- comprehension-filter trick (mirroring "PureClaw.Tabs.Types") so the
  221 -- impossible @'Tab.mkTabIndex' 0 == Nothing@ branch introduces no uncoverable
  222 -- alternative and the value is total (no partial @head@\/@fromJust@).
  223 placeholderSlot :: Tab.TabIndex
  224 placeholderSlot = NE.head (NE.fromList (Maybe.catMaybes [Tab.mkTabIndex 0]))
  225 
  226 -- | Allocate a fresh, monotonically-increasing 'StreamId' (one per completion
  227 -- leg). Atomic so concurrent reads (there are none today — single worker — but
  228 -- the contract is allocator-safe) never collide.
  229 nextStreamId :: IORef Word64 -> IO StreamId
  230 nextStreamId ref = do
  231   n <- atomicModifyIORef' ref (\k -> (k + 1, k))
  232   pure (mkStreamId n)

  238 -- | The injected effects 'mkHarnessRuntime' needs.
  239 data HarnessRuntimeDeps = HarnessRuntimeDeps
  240   { _hrd_ref        :: TabRef
  241     -- ^ This runtime's tab ref (a @BoundHarness@). Output is tagged with it.
  242   , _hrd_emit       :: TabRef -> ChannelEvent -> IO ()
  243     -- ^ Output sink: tags + enqueues to the relay-writer queue.
  244   , _hrd_handle     :: HarnessHandle
  245     -- ^ The harness I/O handle (@_hh_send@\/@_hh_receive@\/@_hh_status@).
  246   , _hrd_fork       :: IO () -> IO Tab.TabRunner
  247     -- ^ The @_env_fork@ seam (forks the drainer + writer).
  248   , _hrd_pollMicros :: Int
  249     -- ^ Drainer sleep between @_hh_receive@ polls (microseconds).
  250   , _hrd_sendBound  :: Int
  251     -- ^ Send queue capacity.
  252   }
  253 
  254 -- | Build + start a harness runtime.
  255 --
  256 -- Forks a /writer/ that drains a bounded send queue to @_hh_send@, and a
  257 -- /drainer/ that polls @_hh_receive@: each non-empty capture is sanitised
  258 -- ('sanitizeHarnessOutput', decoded as UTF-8) and emitted as a ref-tagged
  259 -- 'FullMsg'. When @_hh_status@ reports 'HarnessExited' the drainer stops
  260 -- (without throwing). '_rt_send' enqueues the UTF-8 bytes of its argument;
  261 -- '_rt_stop' cancels both threads but does __not__ stop the harness.
  262 mkHarnessRuntime :: HarnessRuntimeDeps -> IO Runtime
  263 mkHarnessRuntime deps = do
  264   let h = _hrd_handle deps
  265   sendQ <- newTBQueueIO (fromIntegral (max 1 (_hrd_sendBound deps)))
  266   let writer = forever $ do
  267         bs <- atomically (readTBQueue sendQ)
  268         _hh_send h bs
  269       drainer = do
  270         status <- _hh_status h
  271         case status of
  272           HarnessExited _ -> pure ()   -- harness gone: stop draining, no throw
  273           HarnessRunning  -> do
  274             out <- _hh_receive h
  275             unless (BS.null out) $ do
  276               let txt = sanitizeHarnessOutput (TE.decodeUtf8 out)
  277               unless (T.null txt) $
  278                 _hrd_emit deps (_hrd_ref deps) (FullMsg placeholderSlot txt)
  279             threadDelay (max 0 (_hrd_pollMicros deps))
  280             drainer
  281   writerRunner  <- _hrd_fork deps writer
  282   drainerRunner <- _hrd_fork deps drainer
  283   pure Runtime
  284     { _rt_send = \t -> atomically $ do
  285         full <- isFullTBQueue sendQ
  286         if full
  287           then pure (Left (Tab.TabConcurrencyLimit 0))
  288           else do
  289             writeTBQueue sendQ (TE.encodeUtf8 t)
  290             pure (Right ())
  291     , _rt_stop = do
  292         -- Cancel both threads; do NOT call _hh_stop (harness keeps running).
  293         Tab._trun_cancel drainerRunner
  294         Tab._trun_cancel writerRunner
  295     }
