   27 -- | A message received from a channel user, tagged with the source that
   28 -- describes where it came from (channel + optional user id + extensible
   29 -- fields). The source is captured at the channel boundary via
   30 -- 'mkMessageSource'.
   31 data IncomingMessage = IncomingMessage
   32   { _im_source  :: MessageSource
   33   , _im_content :: Text
   34   }
   35   deriving stock (Show, Eq)
   36 
   37 -- | Derive the channel user id from a message's source, preserving the
   38 -- legacy '_im_userId' behavior. A source with no user id (@Nothing@)
   39 -- yields the @UserId ""@ sentinel — the same value the no-op channel
   40 -- produced before the 'MessageSource' migration — so dispatcher routing
   41 -- against the allow-list is unchanged.
   42 imUserId :: IncomingMessage -> UserId
   43 imUserId m = fromMaybe (UserId "") (_ms_userId (_im_source m))

   57 -- | Channel communication capability interface. Concrete implementations
   58 -- (CLI, Telegram, Signal) live in @PureClaw.Channels.*@ modules.
   59 --
   60 -- 'sendError' only accepts 'PublicError' — internal errors with stack
   61 -- traces or model names cannot be sent to channel users. This is enforced
   62 -- at the type level.
   63 data ChannelHandle = ChannelHandle
   64   { _ch_receive      :: IO IncomingMessage
   65   , _ch_send         :: OutgoingMessage -> IO ()
   66   , _ch_sendError    :: PublicError -> IO ()
   67   , _ch_sendChunk    :: StreamChunk -> IO ()
   68   , _ch_streaming    :: Bool
   69     -- ^ Whether this channel supports streaming output. When 'True',
   70     -- the agent loop sends text via '_ch_sendChunk' during generation
   71     -- and skips the full '_ch_send'. When 'False', only '_ch_send'
   72     -- is used for the final complete response.
   73   , _ch_readSecret   :: IO Text    -- ^ Read a line without echo (CLI only)
   74   , _ch_prompt       :: Text -> IO Text
   75     -- ^ Display a prompt and read input on the same line (no trailing
   76     -- newline after the prompt text). For non-interactive channels this
   77     -- falls back to send-then-receive.
   78   , _ch_promptSecret :: Text -> IO Text
   79     -- ^ Like '_ch_prompt' but with echo disabled (for passwords / API keys).
   80     -- For non-interactive channels this falls back to '_ch_readSecret'.
   81   }
   82 
   83 -- | No-op channel handle. Receive returns an empty message, send and
   84 -- sendError are silent. readSecret returns empty text.
   85 mkNoOpChannelHandle :: ChannelHandle
   86 mkNoOpChannelHandle = ChannelHandle
   87   { _ch_receive      = pure (IncomingMessage (mkMessageSource (CkOther "noop") noopConversationId Nothing mempty) "")
   88   , _ch_send         = \_ -> pure ()
   89   , _ch_sendError    = \_ -> pure ()
   90   , _ch_sendChunk    = \_ -> pure ()
   91   , _ch_streaming    = False
   92   , _ch_readSecret   = pure ""
   93   , _ch_prompt       = \_ -> pure ""
   94   , _ch_promptSecret = \_ -> pure ""
   95   }
   96 
   97 -- | The conversation id for the no-op channel handle. A constant, since the
   98 -- no-op handle has no real transport and never carries a routable
   99 -- conversation.
  100 noopConversationId :: ConversationId
  101 noopConversationId = ConversationId "noop"

  103 -- | Thrown when a slash command tries to read interactive input through a
  104 -- capture channel (which has no interactive transport). Carries the prompt
  105 -- label so the caller can report which command needs the CLI.
  106 newtype InteractiveUnsupported = InteractiveUnsupported Text
  107   deriving stock (Show, Eq)
  108 
  109 instance Exception InteractiveUnsupported
  110 
  111 -- | Render a 'PublicError' to channel-safe display text (for buffering).
  112 renderPublicError :: PublicError -> Text
  113 renderPublicError (TemporaryError t) = t
  114 renderPublicError RateLimitError     = "Rate limit reached."
  115 renderPublicError NotAllowedError    = "Not authorized."
  116 
  117 -- | A channel handle that buffers all output instead of writing to a
  118 -- transport, and refuses interactive input. Returns the handle plus a reader
  119 -- that yields the accumulated output (messages joined by newline).
  120 mkCaptureChannelHandle :: IO (ChannelHandle, IO Text)
  121 mkCaptureChannelHandle = do
  122   buf <- newIORef []  -- reversed list of emitted fragments
  123   let append t = modifyIORef' buf (t :)
  124       handle = ChannelHandle
  125         -- '_ch_receive'/'_ch_readSecret' take no label, so they throw with a
  126         -- parenthesized sentinel identifying the method (not a user-facing prompt).
  127         { _ch_receive      = throwIO (InteractiveUnsupported "(receive)")
  128         , _ch_send         = \(OutgoingMessage t) -> append t
  129         , _ch_sendError    = append . renderPublicError
  130         , _ch_sendChunk    = \case
  131             ChunkText t -> append t
  132             ChunkDone   -> pure ()
  133         , _ch_streaming    = False
  134         , _ch_readSecret   = throwIO (InteractiveUnsupported "(secret)")
  135         , _ch_prompt       = throwIO . InteractiveUnsupported
  136         , _ch_promptSecret = throwIO . InteractiveUnsupported
  137         }
  138       reader = T.intercalate "\n" . reverse <$> readIORef buf
  139   pure (handle, reader)
