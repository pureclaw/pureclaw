   62 -- | All runtime dependencies for the agent loop, gathered into a single record.
   63 -- This bundles every capability the tabbed loop ('runTabbedLoop') and the
   64 -- slash-command handlers need into one record, making it easy to add new
   65 -- capabilities (e.g. 'VaultHandle') without touching call sites.
   66 --
   67 -- The Tabbed Chat WU3 (Issue #51) added the '_env_routingConfig' and
   68 -- '_env_fork' fields; the live tab subsystem (Tabs-as-View #79) is carried by
   69 -- the '_env_tabRegistry' \/ '_env_cursors' \/ '_env_exec' \/ '_env_relayWriter'
   70 -- \/ '_env_sinks' \/ '_env_wizard' \/ '_env_tabOutQ' fields built by
   71 -- 'newTabSubsystem'.
   72 data AgentEnv = AgentEnv
   73   { _env_provider     :: IORef (Maybe SomeProvider)
   74     -- ^ The LLM provider. 'Nothing' when no credentials are configured yet.
   75   , _env_model        :: IORef (Maybe ModelId)
   76     -- ^ The model to use for completions. 'Nothing' when no model is
   77     -- configured yet (user must set one via @\/target@ or config file).
   78     -- Mutable so slash commands like @\/provider@ and @\/target@ can
   79     -- hot-swap it.
   80   , _env_channel      :: ChannelHandle
   81     -- ^ The channel to read messages from and write responses to.
   82   , _env_logger       :: LogHandle
   83     -- ^ Structured logger for diagnostic output.
   84   , _env_systemPrompt :: Maybe Text
   85     -- ^ Optional system prompt prepended to every conversation.
   86   , _env_registry     :: ToolRegistry
   87     -- ^ All registered tools available for the agent to call.
   88   , _env_vault        :: IORef (Maybe VaultHandle)
   89     -- ^ Optional secrets vault. 'Nothing' if no vault is configured.
   90   , _env_pluginHandle :: PluginHandle
   91     -- ^ Handle for detecting and generating age plugin identities.
   92   , _env_policy :: SecurityPolicy
   93     -- ^ Security policy for command authorization. Needed by harness management.
   94   , _env_harnesses :: IORef (Map Text HarnessHandle)
   95     -- ^ Running harness handles, keyed by name (e.g. "claude-code").
   96   , _env_harnessRegistry :: Registry.HarnessRegistry
   97     -- ^ The durable 'HarnessId' registry — the source of truth for harness
   98     -- identity and health (design @docs\/harness-registry.md@). This is
   99     -- ADDITIVE alongside the legacy '_env_harnesses' map: the map keys by
  100     -- mutable window name, whereas the registry keys by a UUID-backed
  101     -- 'Registry.HarnessId' that survives tmux rename\/move and restart. The
  102     -- two coexist during Phase 1; later work units migrate consumers onto the
  103     -- registry. Shares one underlying 'TVar' with '_fe_harnessRegistry' so
  104     -- the agent and frontend observe the same registry.
  105   , _env_target :: IORef MessageTarget
  106     -- ^ Where incoming user messages are routed. Mutable so @\/target@
  107     -- can hot-swap the destination.
  108   , _env_nextWindowIdx :: IORef Int
  109     -- ^ Monotonically increasing counter for assigning tmux window indices
  110     -- to new harnesses. Starts at 0.
  111   , _env_agentDef :: Maybe AgentDef
  112     -- ^ Currently-selected agent, if any. Populated by the @--agent@ flag
  113     -- or the @default_agent@ config field. Used by agent-aware slash
  114     -- commands; 'Nothing' in the backward-compat no-agent path.
  115   , _env_session :: IORef SessionHandle
  116     -- ^ Current conversation session. Mutable so @\/session new@ and
  117     -- @\/session resume@ can swap the active session in place.
  118   , _env_onFirstStreamDone :: IORef (Maybe (IO ()))
  119     -- ^ One-shot callback fired by the tabbed loop after the first
  120     -- 'StreamDone' it observes. The loop atomically consumes the
  121     -- action (setting the field back to 'Nothing') so a second
  122     -- 'StreamDone' does not re-fire it. In production this is
  123     -- populated with @'markBootstrapConsumed' session@ so the agent
  124     -- marks its bootstrap as consumed exactly once per process start.
  125   , _env_mcpServers :: IORef (Map Text McpServer)
  126     -- ^ Connected MCP servers, keyed by user-assigned name.
  127     -- Mutable so @\/mcp connect@ and @\/mcp disconnect@ can add\/remove
  128     -- servers at runtime. Tools from these servers are merged into the
  129     -- effective registry on each completion request.
  130     --
  131     -- ---- WU3 (Tabbed Chat #51) additions below this line ----
  132   , _env_routingConfig :: RoutingConfig
  133     -- ^ Routing-layer configuration loaded once at process start
  134     -- (WU3 'PureClaw.Routing.Config'). Immutable for the life of the
  135     -- process — runtime @\/config@ mutation is v1.5+.
  136   , _env_fork :: IO () -> IO TabRunner
  137     -- ^ Test-seam-aware fork primitive. The production default
  138     -- ('defaultEnvFork') wraps 'Control.Concurrent.Async.async';
  139     -- tests inject synchronous variants (T4) to make spawn paths
  140     -- deterministic. Tab loops (WU5\/WU6) MUST use this instead of
  141     -- 'Control.Concurrent.forkIO'.
  142   , _env_broker :: Maybe StreamBroker
  143     -- ^ Optional in-process pub\/sub broker for live transcript
  144     -- streaming. When 'Just', every transcript write performed via the
  145     -- session handles owned by this 'AgentEnv' fans out broker events
  146     -- ('EntryRecorded' and 'ActivityChanged'). 'Nothing' preserves the
  147     -- legacy no-broker path (tests, one-off scripts, the CLI before
  148     -- WU3's lifecycle changes land). See "PureClaw.Frontend.StreamBroker"
  149     -- and "PureClaw.Frontend.BroadcastingTranscript".
  150     --
  151     -- ---- Tabs-as-View (GitHub #79) tab subsystem below this line ----
  152     -- These seven fields are the LIVE tab subsystem the tabbed loop routes
  153     -- through ('PureClaw.Agent.Loop.runTabbedLoop' + the
  154     -- 'PureClaw.Routing.TabDispatch' router). They are constructed together
  155     -- by 'newTabSubsystem'.
  156   , _env_tabRegistry :: TabRegistry
  157     -- ^ The ordered first-class tab registry (invariants I1\/I2 —
  158     -- "PureClaw.Tabs").
  159   , _env_cursors :: IORef CursorState
  160     -- ^ Per-conversation tab cursors + relay overrides (invariant I3 —
  161     -- "PureClaw.Tabs.Types").
  162   , _env_exec :: Exec
  163     -- ^ The refcounted per-'TabRef' runtime registry
  164     -- ("PureClaw.Tabs.Exec"). The real @_ex_startRuntime@ closure that
  165     -- drives it is built in "PureClaw.Tabs.Wiring".
  166   , _env_relayWriter :: RelayWriter
  167     -- ^ The output-side relay writer's burst-dedup state
  168     -- ("PureClaw.Tabs.RelayWriter"). Drained by the relay-writer thread the
  169     -- tabbed loop forks.
  170   , _env_sinks :: SinkRegistry
  171     -- ^ Conversation -> 'ChannelHandle' output sink registry
  172     -- ("PureClaw.Tabs.RelayWriter"). The CLI conversation registers its sink
  173     -- here at loop start.
  174   , _env_wizard :: IORef (Map ConversationKey WizardState)
  175     -- ^ In-flight @\/tab@ attach-wizard state, keyed by conversation
  176     -- ("PureClaw.Tabs.Wizard"). Runtime-only; starts empty.
  177   , _env_tabOutQ :: TBQueue (TabRef, ChannelEvent)
  178     -- ^ Ref-tagged tab-output queue: runtimes enqueue @('TabRef',
  179     -- 'ChannelEvent')@ here; the relay-writer thread drains it through
  180     -- 'PureClaw.Tabs.RelayWriter.processOutput'.
  181   , _env_onTabsChanged :: !(IO ())
  182     -- ^ Callback fired once after each chat-side 'TabRegistry' mutation
  183     -- (@\/nt@, @\/new@, @\/close@, @\/rename@, wizard bind). Default is
  184     -- @'pure' ()@. Wired to rebroadcast\/persist the registry in
  185     -- "PureClaw.CLI.Commands" (WU8). Keeping the callback here prevents
  186     -- @Routing\/Tabs@ from depending on @Frontend@.
  187   , _env_startHarness :: !(HarnessSpec -> IO (Either Text (TabRef, Text)))
  188     -- ^ Dispatcher-reachable seam that spawns a fresh harness, persists its
  189     -- session, links it into the registry, and returns the new tab's
  190     -- @('TabRef', label)@ on success (or a user-facing error 'Text'). The
  191     -- dispatcher's @\/tab new harness@ arm
  192     -- ("PureClaw.Routing.TabDispatch.cmdTabNew") routes through this. Mirrors
  193     -- '_env_onTabsChanged': the default ('noStartHarness') is the unwired stub
  194     -- and the real closure is wired in "PureClaw.CLI.Commands" (WU-B) over
  195     -- 'PureClaw.Frontend.API.spawnHarnessSession'. Kept here so @Routing\/Tabs@
  196     -- need not depend on @Frontend@.
  197   , _env_runTabCommand :: !(ChannelHandle -> Maybe ConversationKey -> TabSlashCommand -> IO ())
  198     -- ^ Dispatcher-reachable seam that executes a parsed @\/tab@ command
  199     -- (rename\/close\/list\/new\/focus\/resume) against the shared tab
  200     -- subsystem. The first argument is the caller's reply 'ChannelHandle':
  201     -- the web path captures dispatcher output via a scoped /capture/ channel
  202     -- (NOT '_env_sinks'), so the real wiring overrides @_td_emit@ to send to
  203     -- THIS channel rather than the conversation's sink. @Just k@ supplies the
  204     -- current conversation for the conversation-relative bits (focus cursor,
  205     -- "(focused)" marker, relay, wizard); @Nothing@ (the web path, which has
  206     -- no 'ConversationKey') falls back to the wired web 'ConversationKey'.
  207     -- Mirrors '_env_startHarness': the
  208     -- default ('noRunTabCommand') is the unwired stub; the real closure is
  209     -- wired in "PureClaw.Tabs.Wiring"\/"PureClaw.CLI.Commands" (Task C) over
  210     -- the shared @runTabCommand@. Kept here so @Agent.SlashCommands@ (which
  211     -- defines @executeSlashCommand@) can invoke tab logic without importing
  212     -- @Routing.TabDispatch@.
  213   }

  215 -- | Read the active session's transcript handle.
  216 --
  217 -- The transcript lives inside the session directory (@transcript.jsonl@)
  218 -- and is swapped automatically when @\/session new@ or
  219 -- @\/session resume@ replaces the active session. Callers that previously
  220 -- read @_env_transcript@ should use this accessor instead.
  221 envTranscript :: AgentEnv -> IO TranscriptHandle
  222 envTranscript env = _sh_transcript <$> readIORef (_env_session env)

  224 -- | Default '_env_fork' implementation: wraps 'Control.Concurrent.Async.async'
  225 -- so the returned 'TabRunner' cancels and waits via the standard async
  226 -- combinators.
  227 --
  228 -- '_trun_cancel' is 'Control.Concurrent.Async.cancel' (idempotent — safe to
  229 -- call on a finished async). '_trun_wait' is 'Control.Concurrent.Async.wait'
  230 -- with the result discarded (the tab loop body returns @()@). Tests that
  231 -- need deterministic cancel observability should inject the synchronous
  232 -- variant from 'Test.Fake.TabFactory' instead.
  233 defaultEnvFork :: IO () -> IO TabRunner
  234 defaultEnvFork body = do
  235   a <- Async.async body
  236   pure TabRunner
  237     { _trun_cancel = Async.cancel a
  238     , _trun_wait   = void (Async.wait a)
  239     }

  259 -- | The freshly-constructed tab-subsystem state the 8c.2 flip threads into
  260 -- 'AgentEnv'. Built once at the production site (and at any test that drives
  261 -- the tabbed loop) by 'newTabSubsystem'; its fields map one-for-one onto the
  262 -- seven new 'AgentEnv' fields.
  263 data TabSubsystem = TabSubsystem
  264   { _ts_tabRegistry    :: !TabRegistry
  265   , _ts_cursors        :: !(IORef CursorState)
  266   , _ts_exec           :: !Exec
  267   , _ts_relayWriter    :: !RelayWriter
  268   , _ts_sinks          :: !SinkRegistry
  269   , _ts_wizard         :: !(IORef (Map ConversationKey WizardState))
  270   , _ts_tabOutQ        :: !(TBQueue (TabRef, ChannelEvent))
  271   , _ts_onTabsChanged  :: !(IO ())
  272     -- ^ Default @'pure' ()@; overridable at the construction site (e.g.
  273     -- "PureClaw.CLI.Commands" in WU8) before wiring into 'AgentEnv'.
  274   }

  276 -- | Allocate a complete, empty tab subsystem: an empty tab registry, empty
  277 -- cursors, an empty runtime registry, a fresh relay writer + sink registry,
  278 -- empty wizard state, and a bounded tab-output queue sized by @outQBound@
  279 -- (the production site reuses @'_rc_channelOutQBound'@). The @_ex_startRuntime@
  280 -- closure that the 'Exec' is driven with is NOT part of this bundle — it is
  281 -- built against the live 'AgentEnv' in "PureClaw.Tabs.Wiring" and passed to
  282 -- @ensure@/@release@ at call time.
  283 newTabSubsystem :: Int -> IO TabSubsystem
  284 newTabSubsystem outQBound = do
  285   tabRegistry <- newTabRegistry
  286   cursors     <- newIORef emptyCursors
  287   exec        <- newExec
  288   relayWriter <- newRelayWriter
  289   sinks       <- newSinkRegistry
  290   wizard      <- newIORef Map.empty
  291   tabOutQ     <- newTBQueueIO (fromIntegral (max 1 outQBound))
  292   pure TabSubsystem
  293     { _ts_tabRegistry   = tabRegistry
  294     , _ts_cursors       = cursors
  295     , _ts_exec          = exec
  296     , _ts_relayWriter   = relayWriter
  297     , _ts_sinks         = sinks
  298     , _ts_wizard        = wizard
  299     , _ts_tabOutQ       = tabOutQ
  300     , _ts_onTabsChanged = pure ()
  301     }
