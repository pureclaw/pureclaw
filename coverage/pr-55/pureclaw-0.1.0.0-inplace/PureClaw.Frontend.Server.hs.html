   29 -- | Frontend server configuration.
   30 data FrontendConfig = FrontendConfig
   31   { _fsc_port      :: Int
   32   , _fsc_staticDir :: FilePath
   33   }
   34   deriving stock (Show, Eq)
   35 
   36 -- | Default: port 8080, serving from @frontend\/dist@.
   37 defaultFrontendConfig :: FrontendConfig
   38 defaultFrontendConfig = FrontendConfig
   39   { _fsc_port      = 8080
   40   , _fsc_staticDir = "frontend/dist"
   41   }
   42 
   43 -- | Build Warp settings from the frontend config.
   44 -- Binds to all interfaces; CORS middleware restricts allowed origins
   45 -- to @http:\/\/localhost:\<port\>@.
   46 mkFrontendSettings :: FrontendConfig -> Warp.Settings
   47 mkFrontendSettings cfg =
   48   Warp.setPort (_fsc_port cfg)
   49     Warp.defaultSettings
   50 
   51 -- | CORS middleware that restricts the @Access-Control-Allow-Origin@
   52 -- header to @http:\/\/localhost:\<port\>@ and handles OPTIONS preflight
   53 -- requests with a 200 response.
   54 corsMiddleware :: FrontendConfig -> Middleware
   55 corsMiddleware cfg app req respond
   56   | requestMethod req == "OPTIONS" =
   57       respond $ responseLBS HTTP.status200 corsHeaders ""
   58   | otherwise =
   59       app req $ \resp ->
   60         respond (mapResponseHeaders (corsHeaders ++) resp)
   61   where
   62     origin :: BS.ByteString
   63     origin = "http://localhost:" <> BS8.pack (show (_fsc_port cfg))
   64 
   65     corsHeaders :: [HTTP.Header]
   66     corsHeaders =
   67       [ ("Access-Control-Allow-Origin",  origin)
   68       , ("Access-Control-Allow-Methods", "GET, POST, PUT, DELETE, OPTIONS")
   69       , ("Access-Control-Allow-Headers", "Content-Type")
   70       ]
   71 
   72 -- | Start the frontend server with API endpoints and static file serving.
   73 -- API routes (@\/api\/*@) are handled by 'apiApp'; everything else
   74 -- falls through to the static file server with SPA fallback.
   75 -- Binds to 127.0.0.1 and applies CORS middleware.
   76 runFrontend :: FrontendConfig -> Maybe FrontendEnv -> LogHandle -> IO ()
   77 runFrontend cfg mEnv logger = do
   78   let logInfo = _lh_logInfo logger
   79   logInfo "PureClaw frontend server"
   80   logInfo $ "  Serving: " <> T.pack (_fsc_staticDir cfg)
   81   logInfo $ "  URL:     http://localhost:" <> T.pack (show (_fsc_port cfg))
   82   let settings = mkFrontendSettings cfg
   83       app = corsMiddleware cfg (combinedApp mEnv (_fsc_staticDir cfg))
   84   Warp.runSettings settings app
   85 
   86 -- | Combined WAI application: API routes first, then static files.
   87 combinedApp :: Maybe FrontendEnv -> FilePath -> Application
   88 combinedApp mEnv staticDir req respond =
   89   case pathInfo req of
   90     ("api" : _) -> case mEnv of
   91       Just env -> apiApp env req respond
   92       Nothing  -> respond $ responseLBS HTTP.status503
   93         [(HTTP.hContentType, "application/json")]
   94         "{\"error\":\"API not available in standalone serve mode\"}"
   95     _ -> staticApp staticDir req respond
   96 
   97 -- | WAI application that serves static files with SPA fallback.
   98 staticApp :: FilePath -> Application
   99 staticApp dir req respond = do
  100   let segments = pathInfo req
  101   -- Reject path traversal
  102   if any (\s -> s == ".." || s == ".") segments
  103     then respond $ responseLBS HTTP.status400 [] "Invalid path"
  104     else do
  105       let relPath  = T.unpack (T.intercalate "/" segments)
  106           filePath = dir </> if null relPath then "index.html" else relPath
  107       exists <- doesFileExist filePath
  108       if exists
  109         then serveFile filePath respond
  110         else do
  111           -- SPA fallback: serve index.html for client-side routing
  112           let indexPath = dir </> "index.html"
  113           indexExists <- doesFileExist indexPath
  114           if indexExists
  115             then serveFile indexPath respond
  116             else respond $ responseLBS HTTP.status404 [] "Not found"
  117 
  118 serveFile :: FilePath -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  119 serveFile path respond = do
  120   contents <- LBS.readFile path
  121   let ct = mimeType (takeExtension path)
  122   respond $ responseLBS HTTP.status200 [(HTTP.hContentType, ct)] contents
  123 
  124 -- | Map file extensions to MIME types.
  125 mimeType :: String -> BS.ByteString
  126 mimeType ".html"  = "text/html; charset=utf-8"
  127 mimeType ".js"    = "application/javascript"
  128 mimeType ".css"   = "text/css"
  129 mimeType ".svg"   = "image/svg+xml"
  130 mimeType ".json"  = "application/json"
  131 mimeType ".png"   = "image/png"
  132 mimeType ".jpg"   = "image/jpeg"
  133 mimeType ".jpeg"  = "image/jpeg"
  134 mimeType ".ico"   = "image/x-icon"
  135 mimeType ".woff"  = "font/woff"
  136 mimeType ".woff2" = "font/woff2"
  137 mimeType ".map"   = "application/json"
  138 mimeType _        = "application/octet-stream"
