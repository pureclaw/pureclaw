    2 -- Module      : PureClaw.Backend.SSH
    3 -- Description : SSH backend factories with hardened defaults (WU9).
    4 --
    5 -- == Overview
    6 --
    7 -- Two factories produce a real 'BackendHandle' over an @ssh@ subprocess:
    8 --
    9 -- * 'mkSshBackendHandle' — 'Pty' kind. Conversational; spawns @ssh -tt@
   10 --   inside a local PTY and delegates to
   11 --   'PureClaw.Backend.Pty.mkDrainerBackendHandle' for the drainer + STM
   12 --   idle state machine.
   13 -- * 'mkSshPipeBackendHandle' — 'Pipe' kind. One-shot; mirrors the pipe
   14 --   spawn pattern in "PureClaw.Backend.Local" (OS-level stderr merge,
   15 --   drain to EOF) but composes the hardened @ssh@ argv around the
   16 --   already-authorized remote 'RemoteCommand'.
   17 --
   18 -- == Hardened argv (callers cannot opt out)
   19 --
   20 -- For both factories the @ssh@ command line includes the security
   21 -- defaults enumerated in
   22 -- @docs\/terminal-backend-abstractions.md@ § "SSH Security Defaults":
   23 -- @-F /dev/null@, @StrictHostKeyChecking=accept-new@, @BatchMode=yes@,
   24 -- @IdentitiesOnly=yes@, @ConnectTimeout=10@, @ServerAliveInterval=30@,
   25 -- @ServerAliveCountMax=3@, @ForwardX11=no@, @ForwardX11Trusted=no@,
   26 -- @ForwardAgent=no@, @PermitLocalCommand=no@.
   27 --
   28 -- The remote program path AND every remote argument are
   29 -- shell-quoted via 'PureClaw.Internal.ShellQuote.shellQuote' before
   30 -- being spliced into the argv — the remote login shell re-parses the
   31 -- combined string, so quoting on the local side is the only line of
   32 -- defense against argument-list injection.
   33 --
   34 -- == v1 credential model
   35 --
   36 -- v1 supports only passphrase-less @ssh@ keys; the caller is
   37 -- responsible for writing the key file at 'SafeKeyPath' (mode 0400)
   38 -- before constructing a backend. The Vault-backed credential helper
   39 -- that decrypts a key into an ephemeral 'SafeKeyPath' is out of scope
   40 -- for WU9 — this module accepts an already-validated 'SafeKeyPath'.
   41 --
   42 -- See @docs\/terminal-backend-abstractions.md@ § "v1 credential model
   43 -- (passphrase-less keys)".

  136 --------------------------------------------------------------------------------
  137 -- SshHost
  138 --------------------------------------------------------------------------------
  139 
  140 -- | A validated ssh hostname or IP literal.
  141 --
  142 -- The constructor is intentionally NOT exported. Obtain a value only
  143 -- via 'mkSshHost', which rejects leading @-@ (defends against
  144 -- @-oProxyCommand=evil@-style argv injection), whitespace, NUL, and
  145 -- shell metacharacters.
  146 newtype SshHost = SshHost Text
  147   deriving stock (Eq, Ord)
  148 
  149 -- | Show is intentionally redacted: the host string is per-session
  150 -- routing data, not loggable in user-facing surfaces.
  151 instance Show SshHost where
  152   show _ = "SshHost <redacted>"
  153 
  154 -- | Project an 'SshHost' to its underlying 'Text' for use inside argv
  155 -- construction. Not intended for logging.
  156 getSshHost :: SshHost -> Text
  157 getSshHost (SshHost t) = t
  158 
  159 -- | The ONLY way to construct an 'SshHost'.
  160 --
  161 -- Accepts:
  162 --
  163 --  * RFC 1123 hostname labels: @[A-Za-z0-9]([A-Za-z0-9-]{0,62})?@,
  164 --    dot-separated, total length <= 253.
  165 --  * Dotted-quad IPv4 literals: e.g. @10.0.0.1@.
  166 --  * Bracketed IPv6 literals: e.g. @[::1]@.
  167 --
  168 -- Rejects (with 'BackendInvalidOption'):
  169 --
  170 --  * Empty input.
  171 --  * Leading @-@ (defends @-oProxyCommand=evil@-style injection).
  172 --  * Whitespace, NUL, or any non-printable / non-ASCII byte.
  173 --  * Shell metacharacters: @; | & $ \` \\ \" \' ( ) < > * ? [ ] { } ! # ~@.
  174 --  * Length > 253.
  175 mkSshHost :: Text -> Either BackendError SshHost
  176 mkSshHost t
  177   | T.null t =
  178       reject "empty host"
  179   | T.length t > 253 =
  180       reject "host too long"
  181   | T.head t == '-' =
  182       reject "leading dash"
  183   | T.any (\c -> not (isAscii c) || c == '\NUL') t =
  184       reject "non-ASCII or NUL"
  185   | T.any isWhitespaceLike t =
  186       reject "whitespace"
  187   -- The bracketed IPv6 form legitimately contains '[' and ']', which
  188   -- would otherwise be caught by 'isShellMeta'. Apply the IPv6 fast-
  189   -- path BEFORE the generic meta-char rejection. Other shell
  190   -- metacharacters inside the brackets are rejected by
  191   -- 'allowedIPv6Char'.
  192   | isBracketedIPv6 t =
  193       Right (SshHost t)
  194   | T.any isShellMeta t =
  195       reject "shell metacharacter"
  196   | isDottedIPv4 t =
  197       Right (SshHost t)
  198   | isHostnameLabels t =
  199       Right (SshHost t)
  200   | otherwise =
  201       reject "invalid hostname"
  202   where
  203     reject :: Text -> Either BackendError SshHost
  204     reject msg = Left (BackendInvalidOption (InvalidOptionDetail ("SshHost: " <> msg)))
  205 
  206     isWhitespaceLike c = c == ' ' || c == '\t' || c == '\n' || c == '\r'
  207     isShellMeta c = c `elem` (";|&$`\\\"'()<>*?[]{}!#~" :: String)
  208 
  209     isBracketedIPv6 s =
  210       T.length s >= 4
  211         && T.head s == '['
  212         && T.last s == ']'
  213         && T.all allowedIPv6Char (T.init (T.tail s))
  214     allowedIPv6Char c = isAscii c && (isAlphaNum c || c == ':' || c == '.')
  215 
  216     isDottedIPv4 s =
  217       let parts = T.splitOn "." s
  218       in length parts == 4 && all isOctet parts
  219     isOctet p =
  220       not (T.null p)
  221         && T.length p <= 3
  222         && T.all isDigit p
  223         && case reads (T.unpack p) :: [(Int, String)] of
  224              [(n, "")] -> n >= 0 && n <= 255
  225              _         -> False
  226 
  227     isHostnameLabels s =
  228       let labels = T.splitOn "." s
  229       in not (null labels)
  230            && all isLabel labels
  231            && T.length s <= 253
  232 
  233     isLabel l =
  234       let n = T.length l
  235       in n >= 1
  236            && n <= 63
  237            && labelHead (T.head l)
  238            && T.all labelChar l
  239            && T.last l /= '-'
  240     labelHead c = isAscii c && isAlphaNum c
  241     labelChar c = isAscii c && (isAlphaNum c || c == '-')

  331 --------------------------------------------------------------------------------
  332 -- RemoteCommand
  333 --------------------------------------------------------------------------------
  334 
  335 -- | A remote program + args that has been authorized against the
  336 -- policy's '_sp_allowedRemoteCommands' allowlist.
  337 --
  338 -- The constructor is intentionally NOT exported. The only way to
  339 -- obtain a 'RemoteCommand' is via 'authorizeRemote' — mirroring the
  340 -- non-exported-ctor pattern of 'AuthorizedCommand'.
  341 newtype RemoteCommand = RemoteCommand (FilePath, [Text])
  342 
  343 -- | Project a 'RemoteCommand' to its remote program path. Used when
  344 -- composing the @ssh user@host -- <program> <args>@ tail.
  345 getRemoteProgram :: RemoteCommand -> FilePath
  346 getRemoteProgram (RemoteCommand (p, _)) = p
  347 
  348 -- | Project a 'RemoteCommand' to its remote argument list.
  349 getRemoteArgs :: RemoteCommand -> [Text]
  350 getRemoteArgs (RemoteCommand (_, as)) = as
  351 
  352 -- | Authorize a remote command against the policy's remote allowlist.
  353 --
  354 -- Mirrors 'PureClaw.Security.Command.authorize' but consults
  355 -- '_sp_allowedRemoteCommands' instead of '_sp_allowedCommands'. The
  356 -- two allowlists are independent (design doc § Authorization);
  357 -- granting @git@ locally does not silently grant @git@ remotely.
  358 authorizeRemote
  359   :: SecurityPolicy
  360   -> FilePath
  361   -> [Text]
  362   -> Either CommandError RemoteCommand
  363 authorizeRemote policy cmd args
  364   | _sp_autonomy policy == Core.Deny =
  365       Left CommandInAutonomyDeny
  366   | not (isRemoteAllowed policy (CommandName base)) =
  367       Left (CommandNotAllowed base)
  368   | otherwise =
  369       Right (RemoteCommand (cmd, args))
  370   where
  371     base = T.pack (takeFileName cmd)
  372 
  373 -- | Local re-implementation of @isRemoteCommandAllowed@ to avoid a
  374 -- cyclical import dance with @Security.Policy@; the field accessor is
  375 -- exported as is.
  376 isRemoteAllowed :: SecurityPolicy -> CommandName -> Bool
  377 isRemoteAllowed policy name = case _sp_allowedRemoteCommands policy of
  378   AllowAll    -> True
  379   AllowList s -> Set.member name s

  381 --------------------------------------------------------------------------------
  382 -- ssh argv construction
  383 --------------------------------------------------------------------------------
  384 
  385 -- | Build the argv list for an ssh invocation.
  386 --
  387 -- Exported as the test seam for DoD #5 (argv flag hardening): callers
  388 -- can assert that the produced list contains the required hardening
  389 -- flags without spawning a real subprocess.
  390 --
  391 -- Arguments:
  392 --
  393 --   * 'SshTarget'   — user, host, port, identity file.
  394 --   * 'RemoteCommand' — the remote program + args (post-authorization).
  395 --   * 'Maybe' 'SafeRuntimePath' — optional @UserKnownHostsFile@.
  396 --   * 'Maybe' 'ControlOpts' — optional ControlMaster settings.
  397 --   * 'Bool' — whether to request a remote PTY via @-tt@.
  398 --
  399 -- The output is ordered: hardening flags first, then identity (@-i@),
  400 -- then optional port + control-master flags, then the user@host
  401 -- separator, then @--@, then the shell-quoted remote program and
  402 -- args.
  403 buildSshArgv
  404   :: SshTarget
  405   -> RemoteCommand
  406   -> Maybe SafeRuntimePath
  407   -> Maybe ControlOpts
  408   -> Bool
  409   -> [String]
  410 buildSshArgv tgt remote =
  411   buildSshArgvFromParts tgt (getRemoteProgram remote) (getRemoteArgs remote)
  412 
  413 -- | Lower-level variant of 'buildSshArgv' that takes the remote program
  414 -- and arg list directly, rather than packaged in a 'RemoteCommand'.
  415 --
  416 -- The 'RemoteCommand' constructor is intentionally non-exported, but
  417 -- consumers (notably 'PureClaw.Backend.Tmux') need to splice
  418 -- subcommand-specific argv into the same hardened ssh prefix — e.g.
  419 -- the same authorized remote tmux binary, but with different args for
  420 -- the pin-resolve probe vs the attach. This helper expresses exactly
  421 -- that: take an already-authorized program path and a fresh args list
  422 -- (the caller's responsibility — the auth invariant lives at
  423 -- 'authorizeRemote' time).
  424 --
  425 -- Note: the program path is shell-quoted, so a path like
  426 -- @/opt/my tools/tmux@ round-trips through the remote login shell
  427 -- intact.
  428 buildSshArgvFromParts
  429   :: SshTarget
  430   -> FilePath               -- ^ remote program path (already authorized)
  431   -> [Text]                 -- ^ remote program args
  432   -> Maybe SafeRuntimePath  -- ^ optional @UserKnownHostsFile@
  433   -> Maybe ControlOpts      -- ^ optional ControlMaster settings
  434   -> Bool                   -- ^ request a remote PTY via @-tt@?
  435   -> [String]
  436 buildSshArgvFromParts tgt remoteProg remoteArgs mKnown mCtl wantPty =
  437   hardenedFlags
  438     <> knownHostsFlag
  439     <> identityFlag
  440     <> portFlag
  441     <> controlFlags
  442     <> ptyFlag
  443     <> [ userAtHost, "--" ]
  444     <> remoteQuotedTail
  445   where
  446     hardenedFlags :: [String]
  447     hardenedFlags =
  448       [ "-F", "/dev/null"
  449       , "-o", "StrictHostKeyChecking=accept-new"
  450       , "-o", "BatchMode=yes"
  451       , "-o", "IdentitiesOnly=yes"
  452       , "-o", "ConnectTimeout=10"
  453       , "-o", "ServerAliveInterval=30"
  454       , "-o", "ServerAliveCountMax=3"
  455       , "-o", "ForwardX11=no"
  456       , "-o", "ForwardX11Trusted=no"
  457       , "-o", "ForwardAgent=no"
  458       , "-o", "PermitLocalCommand=no"
  459       ]
  460 
  461     knownHostsFlag :: [String]
  462     knownHostsFlag = case mKnown of
  463       Nothing -> []
  464       Just kh -> [ "-o", "UserKnownHostsFile=" <> getSafeRuntimePath kh ]
  465 
  466     identityFlag :: [String]
  467     identityFlag = [ "-i", getSafeKeyPath (_st_identity tgt) ]
  468 
  469     portFlag :: [String]
  470     portFlag = case _st_port tgt of
  471       Nothing -> []
  472       Just p  -> [ "-p", show p ]
  473 
  474     controlFlags :: [String]
  475     controlFlags = case mCtl of
  476       Nothing -> []
  477       Just c  ->
  478         [ "-o", "ControlMaster=auto"
  479         , "-o", "ControlPath=" <> getSafeRuntimePath (_co_controlPath c)
  480         , "-o", "ControlPersist=" <> show (_co_persistSecs c)
  481         ]
  482 
  483     ptyFlag :: [String]
  484     ptyFlag = [ "-tt" | wantPty ]
  485 
  486     userAtHost :: String
  487     userAtHost = T.unpack (_st_user tgt) <> "@" <> T.unpack (getSshHost (_st_host tgt))
  488 
  489     remoteQuotedTail :: [String]
  490     remoteQuotedTail =
  491       let progQ = T.unpack (shellQuote (T.pack remoteProg))
  492           argsQ = map (T.unpack . shellQuote) remoteArgs
  493       in progQ : argsQ

  495 --------------------------------------------------------------------------------
  496 -- Pty-kind factory
  497 --------------------------------------------------------------------------------
  498 
  499 -- | Construct a 'Pty'-kind ssh backend.
  500 --
  501 -- Builds the hardened ssh argv from the supplied 'SshTarget' and
  502 -- already-authorized 'RemoteCommand', then delegates to
  503 -- 'mkDrainerBackendHandle' for the drainer + STM idle state machine.
  504 --
  505 -- The local 'AuthorizedCommand' supplies the @ssh@ program path
  506 -- (typically @\/usr\/bin\/ssh@). Its own args are ignored — the
  507 -- factory constructs the full argv via 'buildSshArgv'.
  508 --
  509 -- Construction errors land as @Left BackendError@:
  510 --
  511 --  * @Left BackendBufferQuotaExceeded@ from quota oversubscription;
  512 --  * @Left BackendSshConnectFailed SshConnectTimeout@ \/ similar from
  513 --    a fast-fail at spawn time (any exception during open).
  514 --
  515 -- The factory does not block waiting for the ssh handshake — the
  516 -- drainer surfaces handshake failures via @RecvEof@ on the next
  517 -- @_bh_recv@ (e.g. when ConnectTimeout fires after ~10s and ssh
  518 -- exits).
  519 mkSshBackendHandle
  520   :: PtyIO
  521   -> AuthorizedCommand
  522   -> SshTarget
  523   -> RemoteCommand
  524   -> SshOpts
  525   -> IO (Either BackendError BackendHandle)
  526 mkSshBackendHandle pio sshCmd tgt remote opts = do
  527   let argv = buildSshArgv
  528                tgt
  529                remote
  530                (_so_knownHostsFile opts)
  531                (_so_control opts)
  532                True
  533       ptyOpts = _so_pty opts
  534       spec = PtyOpenSpec
  535         { _pos_program = getCommandProgram sshCmd
  536         , _pos_args    = map T.pack argv
  537         , _pos_env     = _pto_env ptyOpts
  538         , _pos_cwd     = _pto_cwd ptyOpts
  539         , _pos_cols    = _pto_cols ptyOpts
  540         , _pos_rows    = _pto_rows ptyOpts
  541         }
  542   -- ssh failures at spawn time (binary missing, fork failure, etc.)
  543   -- collapse to a single user-visible error. 'SshConnectTimeout' is
  544   -- the closest fixed-vocabulary tag for an early-exit ssh — the
  545   -- drainer + idle timeout cover the mid-handshake hang case via
  546   -- 'RecvTimedOut'.
  547   r <- try @SomeException (mkDrainerBackendHandle pio ptyOpts spec Pty)
  548   case r of
  549     Right (Left e)   -> pure (Left e)
  550     Right (Right bh) -> Right <$> wrapSshHandle bh
  551     Left  _          -> pure (Left (BackendSshConnectFailed SshConnectTimeout))
  552 
  553 -- | Wrap a freshly-constructed Pty-kind ssh 'BackendHandle' so it
  554 -- enforces two ssh-specific invariants:
  555 --
  556 -- 1. Once a recv has observed @RecvEof@ (the ssh transport has
  557 --    closed), a subsequent '_bh_send' throws 'BackendException' with
  558 --    @_be_context = 'BcSshWrite'@ rather than silently succeeding
  559 --    against a dead transport. Any @IO@ exception escaping the inner
  560 --    '_bh_send' is also wrapped under 'BcSshWrite'.
  561 -- 2. Concurrent entry into '_bh_send' \/ '_bh_recv' from multiple
  562 --    threads is rejected with 'BcConcurrentUse' (via
  563 --    'withConcurrentUseGuard').
  564 --
  565 -- Wrappers compose: the recv-EOF tracker wraps the underlying handle
  566 -- first; the resulting handle is then passed through
  567 -- 'withConcurrentUseGuard'. The end-user-visible '_bh_send' \/
  568 -- '_bh_recv' therefore goes through both gates.
  569 wrapSshHandle :: BackendHandle -> IO BackendHandle
  570 wrapSshHandle bh = do
  571   eofRef <- newIORef False
  572   let trackedRecv mIdle = do
  573         res <- _bh_recv bh mIdle
  574         case res of
  575           RecvEof _ -> writeIORef eofRef True
  576           _         -> pure ()
  577         pure res
  578 
  579       sshSend bs = do
  580         wasEof <- readIORef eofRef
  581         if wasEof
  582           then throwIO $ BackendException BcSshWrite
  583                  (toException (userError "ssh transport closed"))
  584           else do
  585             r <- try @SomeException (_bh_send bh bs)
  586             case r of
  587               Right ()                                  -> pure ()
  588               Left e | Just be <- castBackendException e -> throwIO be
  589                      | otherwise                         -> throwIO $
  590                          BackendException BcSshWrite e
  591   withConcurrentUseGuard bh
  592     { _bh_send = sshSend
  593     , _bh_recv = trackedRecv
  594     }

  596 -- | If the supplied 'SomeException' already is a 'BackendException',
  597 -- return it; otherwise 'Nothing'. Used by 'wrapSshHandle' to avoid
  598 -- double-wrapping when an inner layer already threw a structured
  599 -- 'BackendException'.
  600 castBackendException :: SomeException -> Maybe BackendException
  601 castBackendException = Exception.fromException

  607 -- | Construct a 'Pipe'-kind ssh backend.
  608 --
  609 -- One-shot: stdin is empty (the caller cannot stream input to a
  610 -- one-shot ssh — use 'mkSshBackendHandle' if you need interactivity);
  611 -- stdout and stderr are merged at the OS level (the OS-level @2>&1@
  612 -- pattern, same as 'PureClaw.Backend.Local.mkLocalBackendHandle'); the
  613 -- accumulator is read until EOF or @_spo_recvBufferCap@.
  614 --
  615 -- Mirrors 'PureClaw.Backend.Local.mkLocalBackendHandle' for the pipe
  616 -- spawn machinery, but composes the hardened ssh argv around the
  617 -- already-authorized remote 'RemoteCommand'.
  618 mkSshPipeBackendHandle
  619   :: AuthorizedCommand
  620   -> SshTarget
  621   -> RemoteCommand
  622   -> SshPipeOpts
  623   -> IO (Either BackendError BackendHandle)
  624 mkSshPipeBackendHandle sshCmd tgt remote opts = do
  625   let argv = buildSshArgv
  626                tgt
  627                remote
  628                (_spo_knownHostsFile opts)
  629                (_spo_control opts)
  630                False
  631   spawnSshArgv sshCmd argv opts
  632 
  633 -- | Internal helper: spawn ssh as a pipe-only subprocess.
  634 --
  635 -- Acquires the per-backend recv buffer quota, opens a single shared
  636 -- pipe (write-end dup'd to both child stdout and stderr), spawns ssh
  637 -- via @typed-process@, then drains to EOF (or the cap) into an
  638 -- 'IORef' accumulator.
  639 --
  640 -- This mirrors the pipe spawn pattern in
  641 -- "PureClaw.Backend.Local"; the only divergence is the constructed
  642 -- argv (the hardened ssh prefix already lives in 'buildSshArgv').
  643 spawnSshArgv
  644   :: AuthorizedCommand
  645   -> [String]
  646   -> SshPipeOpts
  647   -> IO (Either BackendError BackendHandle)
  648 spawnSshArgv sshCmd argv opts = do
  649   let capBytes = _spo_recvBufferCap opts
  650       capMiB   = bytesToMiBCeil capBytes
  651   quotaPtr <- readIORef globalBackendBufferQuota
  652   acq <- acquireBufferQuota quotaPtr capMiB
  653   case acq of
  654     Left e   -> pure (Left e)
  655     Right () -> do
  656       releasedRef <- newIORef False
  657       let releaseQuotaOnce = do
  658             already <- atomicSwapTrue releasedRef
  659             if already
  660               then pure ()
  661               else releaseBufferQuota quotaPtr capMiB
  662 
  663       let prog = getCommandProgram sshCmd
  664           envList = envMapToTuples (_spo_env opts)
  665 
  666       pipeResult <- try @SomeException PosixIO.createPipe
  667       case pipeResult of
  668         Left _ -> do
  669           releaseQuotaOnce
  670           pure (Left (BackendBinaryNotFound (CommandName (T.pack prog))))
  671         Right (readFd, writeFd) -> do
  672           readH  <- PosixIO.fdToHandle readFd
  673           writeH <- PosixIO.fdToHandle writeFd
  674           let baseConfig =
  675                   P.setStdin (P.byteStringInput BL.empty)
  676                 $ P.setStdout (P.useHandleOpen writeH)
  677                 $ P.setStderr (P.useHandleOpen writeH)
  678                 $ P.setEnv envList
  679                 $ P.proc prog argv
  680               withCwd = maybe id P.setWorkingDir (_spo_cwd opts)
  681               config = withCwd baseConfig
  682 
  683           spawnResult <- try @SomeException (P.startProcess config)
  684           case spawnResult of
  685             Left _ -> do
  686               _ <- try @SomeException (hClose writeH)
  687               _ <- try @SomeException (hClose readH)
  688               releaseQuotaOnce
  689               pure (Left (BackendSshConnectFailed SshOtherFailure))
  690             Right proc -> do
  691               _ <- try @SomeException (hClose writeH)
  692 
  693               accRef      <- newIORef BS.empty
  694               truncRef    <- newIORef False
  695               reader      <- Async.async (drainHandleInto capBytes accRef truncRef readH)
  696 
  697               closeLock   <- newMVar ()
  698               closedRef   <- newIORef False
  699               recvDoneRef <- newIORef False
  700 
  701               let doSend _ = pure ()
  702 
  703                   doRecv _mIdle = do
  704                     already <- readIORef recvDoneRef
  705                     if already
  706                       then resultFromAcc accRef truncRef
  707                       else do
  708                         _ <- Async.waitCatch reader
  709                         writeIORef recvDoneRef True
  710                         resultFromAcc accRef truncRef
  711 
  712                   doResize _ _ = pure ()
  713 
  714                   doClose = modifyMVar_ closeLock $ \() -> do
  715                     wasClosed <- readIORef closedRef
  716                     if wasClosed
  717                       then pure ()
  718                       else do
  719                         writeIORef closedRef True
  720                         Async.cancel reader
  721                         _ <- try @SomeException (P.stopProcess proc)
  722                         _ <- try @SomeException (hClose readH)
  723                         releaseQuotaOnce
  724                         pure ()
  725 
  726               guarded <- withConcurrentUseGuard BackendHandle
  727                 { _bh_name        = "ssh-pipe"
  728                 , _bh_kind        = Pipe
  729                 , _bh_defaultIdle = _spo_idle opts
  730                 , _bh_send        = doSend
  731                 , _bh_recv        = doRecv
  732                 , _bh_resize      = doResize
  733                 , _bh_close       = doClose
  734                 }
  735               pure (Right guarded)
  736   where
  737     resultFromAcc accRef truncRef = do
  738       bs  <- readIORef accRef
  739       isT <- readIORef truncRef
  740       pure (if isT then RecvTruncated bs else RecvSettled bs)
  741 
  742 -- | Drain a 'Handle' into the accumulator, latching the truncate flag
  743 -- if the per-backend cap is exceeded. Best-effort: read exceptions
  744 -- are treated as EOF.
  745 drainHandleInto :: Int -> IORef ByteString -> IORef Bool -> Handle -> IO ()
  746 drainHandleInto cap accRef truncRef h = go
  747   where
  748     go = do
  749       r <- try @SomeException (BS.hGetSome h 4096)
  750       case r of
  751         Left _    -> pure ()
  752         Right bs
  753           | BS.null bs -> pure ()
  754           | otherwise -> do
  755               acc <- readIORef accRef
  756               let newSize = BS.length acc + BS.length bs
  757               if newSize > cap
  758                 then writeIORef truncRef True
  759                 else do
  760                   atomicModifyIORef' accRef $ \old -> (old <> bs, ())
  761                   go

  767 -- | Tear down an existing ControlMaster multiplex.
  768 --
  769 -- Runs @ssh -S <controlPath> -O exit dummy@dummy@. The user@host is
  770 -- unused by @-O exit@ — only the socket matters — but ssh still
  771 -- requires the positional argument, so we pass a placeholder. The
  772 -- 'AuthorizedCommand' supplies the @ssh@ program path; the caller
  773 -- should reuse the same 'AuthorizedCommand' they used to open the
  774 -- multiplex (preserves the §5.1 authorize-before-execute invariant).
  775 --
  776 -- Best-effort and idempotent: any spawn or wait failure is swallowed.
  777 -- If the socket file does not exist (already closed, never opened),
  778 -- ssh exits with a non-zero status which this function silently
  779 -- ignores.
  780 closeSshMultiplex :: AuthorizedCommand -> ControlOpts -> IO ()
  781 closeSshMultiplex sshCmd opts = do
  782   let prog = getCommandProgram sshCmd
  783       argv =
  784         [ "-S", getSafeRuntimePath (_co_controlPath opts)
  785         , "-O", "exit"
  786         , "dummy@dummy"
  787         ]
  788       config = P.setStdin P.nullStream
  789              $ P.setStdout P.nullStream
  790              $ P.setStderr P.nullStream
  791              $ P.proc prog argv
  792   _ <- try @SomeException (P.runProcess config)
  793   pure ()

  799 -- | Round a byte count up to whole MiB (1 MiB == 1_048_576 bytes).
  800 -- Zero or negative byte counts produce @0@. Mirrors the helper in
  801 -- "PureClaw.Backend.Pty" / "PureClaw.Backend.Local"; reimplemented
  802 -- locally so the export surface of those modules stays tight.
  803 bytesToMiBCeil :: Int -> Int
  804 bytesToMiBCeil n
  805   | n <= 0    = 0
  806   | otherwise = (n + mib - 1) `div` mib
  807   where mib = 1024 * 1024
  808 
  809 -- | Atomically set an 'IORef' 'Bool' to 'True', returning its previous
  810 -- value. Used to defend against double-release of the buffer quota.
  811 atomicSwapTrue :: IORef Bool -> IO Bool
  812 atomicSwapTrue ref = atomicModifyIORef' ref (True,)
  813 
  814 -- | Render an 'EnvMap' as the @[(String, String)]@ tuple list that
  815 -- @typed-process@\'s 'P.setEnv' expects. Values decode byte-for-byte
  816 -- (Latin-1) so non-ASCII bytes round-trip 1:1 to the subprocess.
  817 envMapToTuples :: EnvMap -> [(String, String)]
  818 envMapToTuples = map render . Map.toAscList
  819   where
  820     render (k, EnvValue v) = (k, map (toEnum . fromIntegral) (BS.unpack v))
