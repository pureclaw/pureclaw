   36 -- | All runtime dependencies for the agent loop, gathered into a single record.
   37 -- This replaces the multi-parameter signature of 'runAgentLoop' and
   38 -- 'executeSlashCommand', making it easy to add new capabilities (e.g.
   39 -- 'VaultHandle') in later work units without touching call sites.
   40 --
   41 -- The Tabbed Chat WU3 (Issue #51) adds seven fields below the original
   42 -- record body: '_env_tabs', '_env_focus', '_env_activeCount',
   43 -- '_env_runners', '_env_channelOutQ', '_env_routingConfig', and
   44 -- '_env_fork'. See @docs\/tabbed-chat.md@ §\"AgentEnv additions\".
   45 data AgentEnv = AgentEnv
   46   { _env_provider     :: IORef (Maybe SomeProvider)
   47     -- ^ The LLM provider. 'Nothing' when no credentials are configured yet.
   48   , _env_model        :: IORef (Maybe ModelId)
   49     -- ^ The model to use for completions. 'Nothing' when no model is
   50     -- configured yet (user must set one via @\/target@ or config file).
   51     -- Mutable so slash commands like @\/provider@ and @\/target@ can
   52     -- hot-swap it.
   53   , _env_channel      :: ChannelHandle
   54     -- ^ The channel to read messages from and write responses to.
   55   , _env_logger       :: LogHandle
   56     -- ^ Structured logger for diagnostic output.
   57   , _env_systemPrompt :: Maybe Text
   58     -- ^ Optional system prompt prepended to every conversation.
   59   , _env_registry     :: ToolRegistry
   60     -- ^ All registered tools available for the agent to call.
   61   , _env_vault        :: IORef (Maybe VaultHandle)
   62     -- ^ Optional secrets vault. 'Nothing' if no vault is configured.
   63   , _env_pluginHandle :: PluginHandle
   64     -- ^ Handle for detecting and generating age plugin identities.
   65   , _env_policy :: SecurityPolicy
   66     -- ^ Security policy for command authorization. Needed by harness management.
   67   , _env_harnesses :: IORef (Map Text HarnessHandle)
   68     -- ^ Running harness handles, keyed by name (e.g. "claude-code").
   69   , _env_target :: IORef MessageTarget
   70     -- ^ Where incoming user messages are routed. Mutable so @\/target@
   71     -- can hot-swap the destination.
   72   , _env_nextWindowIdx :: IORef Int
   73     -- ^ Monotonically increasing counter for assigning tmux window indices
   74     -- to new harnesses. Starts at 0.
   75   , _env_agentDef :: Maybe AgentDef
   76     -- ^ Currently-selected agent, if any. Populated by the @--agent@ flag
   77     -- or the @default_agent@ config field. Used by agent-aware slash
   78     -- commands; 'Nothing' in the backward-compat no-agent path.
   79   , _env_session :: IORef SessionHandle
   80     -- ^ Current conversation session. Mutable so @\/session new@ and
   81     -- @\/session resume@ can swap the active session in place.
   82   , _env_onFirstStreamDone :: IORef (Maybe (IO ()))
   83     -- ^ One-shot callback fired by 'runAgentLoop' after the first
   84     -- 'StreamDone' it observes. The loop atomically consumes the
   85     -- action (setting the field back to 'Nothing') so a second
   86     -- 'StreamDone' does not re-fire it. In production this is
   87     -- populated with @'markBootstrapConsumed' session@ so the agent
   88     -- marks its bootstrap as consumed exactly once per process start.
   89   , _env_mcpServers :: IORef (Map Text McpServer)
   90     -- ^ Connected MCP servers, keyed by user-assigned name.
   91     -- Mutable so @\/mcp connect@ and @\/mcp disconnect@ can add\/remove
   92     -- servers at runtime. Tools from these servers are merged into the
   93     -- effective registry on each completion request.
   94     --
   95     -- ---- WU3 (Tabbed Chat #51) additions below this line ----
   96   , _env_tabs :: IORef (IntMap TabHandle)
   97     -- ^ Tab registry — keyed by 'TabIndex'. Mutated by the dispatcher
   98     -- (WU5) via 'PureClaw.Routing.Registry'. WU3 ships the field empty.
   99   , _env_focus :: IORef (Maybe TabIndex)
  100     -- ^ Currently-focused tab. The dispatcher writes this only between
  101     -- message cycles (E3 invariant). 'Nothing' on first boot, before
  102     -- any tab has been spawned\/auto-spawned.
  103   , _env_activeCount :: TVar Int
  104     -- ^ Number of tabs in 'PureClaw.Handles.Tab.Active' status. Used
  105     -- by the S9 concurrent-active cap. Mutated atomically by tab
  106     -- loops (WU6) under STM together with the status transition.
  107   , _env_runners :: IORef (IntMap (IORef (Maybe TabRunner)))
  108     -- ^ Per-tab 'TabRunner' placeholders. The dispatcher's spawn path
  109     -- (WU5) inserts an @IORef Nothing@ under 'mask' BEFORE calling
  110     -- @_env_fork@, then fills it in once the fork returns — this
  111     -- guarantees 'closeAllTabs' (the bracket cleanup) sees every fork
  112     -- that was started, even if interrupted mid-flight.
  113   , _env_channelOutQ :: TBQueue (OutputSource, ChannelEvent)
  114     -- ^ Process-wide bounded queue feeding the single channel-out
  115     -- writer thread (WU4). Capacity is configured by
  116     -- '_rc_channelOutQBound' at AgentEnv construction time.
  117   , _env_routingConfig :: RoutingConfig
  118     -- ^ Routing-layer configuration loaded once at process start
  119     -- (WU3 'PureClaw.Routing.Config'). Immutable for the life of the
  120     -- process — runtime @\/config@ mutation is v1.5+.
  121   , _env_fork :: IO () -> IO TabRunner
  122     -- ^ Test-seam-aware fork primitive. The production default
  123     -- ('defaultEnvFork') wraps 'Control.Concurrent.Async.async';
  124     -- tests inject synchronous variants (T4) to make spawn paths
  125     -- deterministic. Tab loops (WU5\/WU6) MUST use this instead of
  126     -- 'Control.Concurrent.forkIO'.
  127   }

  129 -- | Read the active session's transcript handle.
  130 --
  131 -- The transcript lives inside the session directory (@transcript.jsonl@)
  132 -- and is swapped automatically when @\/session new@ or
  133 -- @\/session resume@ replaces the active session. Callers that previously
  134 -- read @_env_transcript@ should use this accessor instead.
  135 envTranscript :: AgentEnv -> IO TranscriptHandle
  136 envTranscript env = _sh_transcript <$> readIORef (_env_session env)

  138 -- | Default '_env_fork' implementation: wraps 'Control.Concurrent.Async.async'
  139 -- so the returned 'TabRunner' cancels and waits via the standard async
  140 -- combinators.
  141 --
  142 -- '_trun_cancel' is 'Control.Concurrent.Async.cancel' (idempotent — safe to
  143 -- call on a finished async). '_trun_wait' is 'Control.Concurrent.Async.wait'
  144 -- with the result discarded (the tab loop body returns @()@). Tests that
  145 -- need deterministic cancel observability should inject the synchronous
  146 -- variant from 'Test.Fake.TabFactory' instead.
  147 defaultEnvFork :: IO () -> IO TabRunner
  148 defaultEnvFork body = do
  149   a <- Async.async body
  150   pure TabRunner
  151     { _trun_cancel = Async.cancel a
  152     , _trun_wait   = void (Async.wait a)
  153     }
