   96 -- | Handle for the current conversation session.
   97 --
   98 -- Owns the on-disk session directory (mode @0o700@), the @session.json@
   99 -- metadata file (mode @0o600@), and the per-session transcript handle
  100 -- writing to @transcript.jsonl@ (mode @0o600@).
  101 data SessionHandle = SessionHandle
  102   { _sh_meta       :: IORef SessionMeta
  103     -- ^ Mutable session metadata. Updated by callers (e.g. to bump
  104     -- @last_active@) and persisted by '_sh_save'.
  105   , _sh_transcript :: TranscriptHandle
  106     -- ^ Transcript handle owned by the session.
  107   , _sh_dir        :: FilePath
  108     -- ^ On-disk session directory, i.e. @<baseDir>/<sessionId>/@.
  109   , _sh_save       :: IO ()
  110     -- ^ Persist '_sh_meta' to @session.json@ atomically (write to
  111     -- @session.json.tmp@ then rename).
  112   }
  113 
  114 -- | Reasons a resume attempt can fail.
  115 data ResumeError
  116   = ResumeMissingMetadata FilePath
  117     -- ^ No @session.json@ at the expected path.
  118   | ResumeCorruptedMetadata FilePath String
  119     -- ^ @session.json@ exists but does not parse as 'SessionMeta'.
  120     -- The second field is a human-readable recovery hint.
  121   deriving stock (Show, Eq)
  122 
  123 -- | Lookup result for 'resolveSessionRef'.
  124 data ResolveError
  125   = NotFound
  126   | Ambiguous [SessionId]
  127   deriving stock (Show, Eq)

  129 -- | Result of 'validateRuntime'. 'RuntimeFallback' carries the chosen
  130 -- fallback target plus a warning message suitable for logging.
  131 data ResolvedRuntime
  132   = RuntimeOk MessageTarget
  133   | RuntimeFallback MessageTarget Text
  134   deriving stock (Show, Eq)

  140 -- | Create a brand-new on-disk session handle.
  141 --
  142 -- Creates @\<baseDir\>/\<sessionId\>/@ with mode @0o700@, writes the
  143 -- initial @session.json@ (mode @0o600@) via 'saveMeta', and opens
  144 -- @transcript.jsonl@ (mode @0o600@) through 'mkFileTranscriptHandle'.
  145 mkSessionHandle :: LogHandle -> FilePath -> SessionMeta -> IO SessionHandle
  146 mkSessionHandle logger baseDir meta = do
  147   let sid  = unSessionId (_sm_id meta)
  148       dir  = baseDir </> T.unpack sid
  149       txp  = dir </> "transcript.jsonl"
  150   createDirectoryIfMissing True dir
  151   setFileMode dir 0o700
  152   metaRef <- newIORef meta
  153   saveMeta dir metaRef
  154   tx <- mkFileTranscriptHandle logger txp
  155   pure SessionHandle
  156     { _sh_meta       = metaRef
  157     , _sh_transcript = tx
  158     , _sh_dir        = dir
  159     , _sh_save       = saveMeta dir metaRef
  160     }
  161 
  162 -- | Atomically persist the metadata IORef to @\<dir\>/session.json@ by
  163 -- writing to a sibling @session.json.tmp@ with mode @0o600@ then
  164 -- 'renameFile'-ing it into place. The rename is atomic on POSIX, so a
  165 -- crash mid-write leaves the previous @session.json@ intact.
  166 saveMeta :: FilePath -> IORef SessionMeta -> IO ()
  167 saveMeta dir ref = do
  168   meta <- readIORef ref
  169   let finalP = dir </> "session.json"
  170       tmpP   = finalP <> ".tmp"
  171   LBS.writeFile tmpP (Aeson.encode meta)
  172   setFileMode tmpP 0o600
  173   renameFile tmpP finalP

  241 -- | Reopen an existing session by ID. Reads @session.json@, validates
  242 -- the JSON, and reopens @transcript.jsonl@ for append.
  243 resumeSession
  244   :: LogHandle
  245   -> FilePath
  246   -> SessionId
  247   -> IO (Either ResumeError SessionHandle)
  248 resumeSession logger baseDir sid = do
  249   let dir    = baseDir </> T.unpack (unSessionId sid)
  250       metaP  = dir </> "session.json"
  251       txP    = dir </> "transcript.jsonl"
  252   exists <- doesFileExist metaP
  253   if not exists
  254     then pure (Left (ResumeMissingMetadata metaP))
  255     else do
  256       raw <- LBS.readFile metaP
  257       case Aeson.eitherDecode' raw of
  258         Left err -> pure
  259           (Left (ResumeCorruptedMetadata metaP
  260             ("failed to parse session metadata: " <> err
  261               <> " — recovery hint: inspect or remove " <> metaP)))
  262         Right meta -> do
  263           metaRef <- newIORef (meta :: SessionMeta)
  264           tx <- mkFileTranscriptHandle logger txP
  265           pure (Right SessionHandle
  266             { _sh_meta       = metaRef
  267             , _sh_transcript = tx
  268             , _sh_dir        = dir
  269             , _sh_save       = saveMeta dir metaRef
  270             })

  276 -- | Read all @session.json@ files under @baseDir@, optionally filtering
  277 -- by agent name, sorted by @last_active@ descending, and capped at
  278 -- @limit@ entries.
  279 --
  280 -- Silently skips entries that fail to decode so that a single corrupted
  281 -- metadata file does not hide the remaining sessions from the user.
  282 listSessions :: FilePath -> Maybe AgentName -> Int -> IO [SessionMeta]
  283 listSessions baseDir mAgent limit = do
  284   exists <- doesDirectoryExist baseDir
  285   if not exists
  286     then pure []
  287     else do
  288       entries <- listDirectory baseDir
  289       metas <- fmap catMaybes' (traverse (tryLoad baseDir) entries)
  290       let filtered = case mAgent of
  291             Nothing -> metas
  292             Just a  -> filter (\m -> _sm_agent m == Just a) metas
  293           sorted = sortOn (Down . _sm_lastActive) filtered
  294       pure (take limit sorted)
  295   where
  296     catMaybes' = foldr (\mx acc -> case mx of
  297                           Just x  -> x : acc
  298                           Nothing -> acc) []
  299 
  300 -- | Attempt to load @\<baseDir\>/\<name\>/session.json@. Returns
  301 -- 'Nothing' if the path is not a directory, the metadata file is
  302 -- missing, unreadable, or malformed.
  303 tryLoad :: FilePath -> FilePath -> IO (Maybe SessionMeta)
  304 tryLoad baseDir name = do
  305   let dir   = baseDir </> name
  306       metaP = dir </> "session.json"
  307   isDir <- doesDirectoryExist dir
  308   if not isDir
  309     then pure Nothing
  310     else do
  311       exists <- doesFileExist metaP
  312       if not exists
  313         then pure Nothing
  314         else do
  315           eBytes <- try (LBS.readFile metaP) :: IO (Either IOException LBS.ByteString)
  316           case eBytes of
  317             Left _     -> pure Nothing
  318             Right raw  -> case Aeson.decode' raw of
  319               Just m  -> pure (Just m)
  320               Nothing -> pure Nothing

  322 -- | Resolve a user-supplied reference (exact ID or prefix) to a concrete
  323 -- 'SessionId'. Exact matches win outright; otherwise the match set is
  324 -- computed by 'T.isPrefixOf' over the full ID string.
  325 resolveSessionRef
  326   :: FilePath
  327   -> Text
  328   -> IO (Either ResolveError SessionId)
  329 resolveSessionRef baseDir ref = do
  330   exists <- doesDirectoryExist baseDir
  331   if not exists
  332     then pure (Left NotFound)
  333     else do
  334       entries <- listDirectory baseDir
  335       -- Keep only entries that actually have a session.json — avoids
  336       -- matching stray directories created by unrelated tests or users.
  337       valid <- filterM hasMetadata entries
  338       let ids = map (SessionId . T.pack) valid
  339       if ref `elem` map (\(SessionId t) -> t) ids
  340         then pure (Right (SessionId ref))
  341         else case filter (\(SessionId t) -> ref `T.isPrefixOf` t) ids of
  342           []  -> pure (Left NotFound)
  343           [x] -> pure (Right x)
  344           xs  -> pure (Left (Ambiguous xs))
  345   where
  346     hasMetadata name = do
  347       let metaP = baseDir </> name </> "session.json"
  348       doesFileExist metaP
  349 
  350     -- Local strict filterM to avoid pulling in Control.Monad just for this.
  351     filterM _ []     = pure []
  352     filterM p (x:xs) = do
  353       keep <- p x
  354       rest <- filterM p xs
  355       pure (if keep then x : rest else rest)

  357 -- ----------------------------------------------------------------------------
  358 -- Runtime validation
  359 -- ----------------------------------------------------------------------------
  360 
  361 -- | Validate a 'SessionKind' against the currently-running harnesses.
  362 --
  363 -- * 'SkProvider' always resolves to @RuntimeOk TargetProvider@.
  364 -- * @'SkHarness' spec@ resolves to @RuntimeOk (TargetHarness name)@ if
  365 --   a harness with that name is present in the map (name derived from
  366 --   the harness flavour).
  367 -- * @'SkHarness' spec@ resolves to @RuntimeFallback TargetProvider msg@
  368 --   if the harness is absent, where @msg@ explains the fallback for
  369 --   logging at warn level.
  370 validateRuntime :: Map Text HarnessHandle -> SessionKind -> ResolvedRuntime
  371 validateRuntime _ (SkProvider _) = RuntimeOk TargetProvider
  372 validateRuntime harnesses (SkHarness spec) =
  373   let name = sessionKindHarnessName spec
  374   in if Map.member name harnesses
  375        then RuntimeOk (TargetHarness name)
  376        else let msg = "harness '" <> name <> "' is not running, falling back to provider"
  377             in RuntimeFallback TargetProvider msg
  378 
  379 -- | Extract the harness name from a 'HarnessSpec' by rendering its flavour.
  380 sessionKindHarnessName :: HarnessSpec -> Text
  381 sessionKindHarnessName spec = case _h_flavour spec of
  382   HClaudeCode -> "claude-code"
  383   HCodex      -> "codex"
  384   HOpenCode   -> "opencode"
  385   HHermes     -> "hermes"
  386   HPureClaw   -> "pureclaw"
  387   HCustom n   -> n
  388 
  389 -- | Resolve a resumed session's 'SessionKind' to a concrete
  390 -- 'MessageTarget' given the currently-running harness map, logging a
  391 -- warning if the recorded runtime is no longer available.
  392 --
  393 -- Wraps 'validateRuntime': on 'RuntimeFallback' the provided warning
  394 -- message is routed to @_lh_logWarn@ and the fallback target (always
  395 -- 'TargetProvider') is returned; on 'RuntimeOk' the target is returned
  396 -- without logging.
  397 resolveResumedTarget
  398   :: LogHandle
  399   -> Map Text HarnessHandle
  400   -> SessionKind
  401   -> IO MessageTarget
  402 resolveResumedTarget logger harnesses sk = case validateRuntime harnesses sk of
  403   RuntimeOk tgt -> pure tgt
  404   RuntimeFallback tgt warning -> do
  405     _lh_logWarn logger warning
  406     pure tgt

  412 -- | Flip '_sm_bootstrapConsumed' to 'True' in the session metadata
  413 -- 'IORef' and persist the change to @session.json@ via '_sh_save'.
  414 --
  415 -- Called by the agent loop the first time a provider response completes
  416 -- ('StreamDone'), via the one-shot callback installed on
  417 -- @_env_onFirstStreamDone@. Idempotent: re-invocations re-save the same
  418 -- metadata without error.
  419 markBootstrapConsumed :: SessionHandle -> IO ()
  420 markBootstrapConsumed sh = do
  421   atomicModifyIORef' (_sh_meta sh) $ \m ->
  422     (m { _sm_bootstrapConsumed = True }, ())
  423   _sh_save sh

  429 -- | Reasons 'setArchived' may fail. The session directory and transcript
  430 -- are never touched, so failure leaves on-disk state unchanged.
  431 data SetArchivedError
  432   = SetArchivedSessionMissing
  433     -- ^ @\<baseDir\>/\<sid\>/session.json@ does not exist.
  434   | SetArchivedParseFailed Text
  435     -- ^ session.json was unreadable or did not decode as 'SessionMeta'.
  436   deriving stock (Show, Eq)
  437 
  438 -- | Toggle 'SessionMeta._sm_archived' for a session that may or may not
  439 -- have a live 'SessionHandle'. Operates directly on the on-disk
  440 -- @session.json@: reads, updates the archive flag, writes back via the
  441 -- usual tmp-file + rename atomicity pattern.
  442 --
  443 -- Crucially, this is **display state only** — neither the session
  444 -- directory nor the transcript is removed. Unarchiving restores the
  445 -- session to "Recent Sessions" with all history intact.
  446 setArchived :: FilePath -> SessionId -> Bool -> IO (Either SetArchivedError ())
  447 setArchived baseDir sid archived =
  448   updateSessionMeta baseDir sid (\m -> m { _sm_archived = archived })
  449     SetArchivedSessionMissing SetArchivedParseFailed

  451 -- | Reasons 'setDescription' may fail. Same shape as 'SetArchivedError'.
  452 data SetDescriptionError
  453   = SetDescriptionSessionMissing
  454   | SetDescriptionParseFailed Text
  455   deriving stock (Show, Eq)

  457 -- | Set or clear the user-provided session description. Passing 'Just t'
  458 -- stores the description (trimmed of surrounding whitespace; an empty
  459 -- result is treated as 'Nothing' so callers can pass through what the
  460 -- user typed without preprocessing). Passing 'Nothing' clears it,
  461 -- letting display surfaces fall back to '_sm_autoSummary' or a
  462 -- transcript-derived snippet.
  463 setDescription :: FilePath -> SessionId -> Maybe Text -> IO (Either SetDescriptionError ())
  464 setDescription baseDir sid mDesc =
  465   let normalized = mDesc >>= \t -> let s = T.strip t in if T.null s then Nothing else Just s
  466   in  updateSessionMeta baseDir sid (\m -> m { _sm_description = normalized })
  467         SetDescriptionSessionMissing SetDescriptionParseFailed

  469 -- | Shared read-modify-write helper for the disk-only meta mutators.
  470 -- Reads @session.json@, applies @f@, and writes back via tmp-file +
  471 -- rename. @missing@ and @parseFail@ are the caller's error
  472 -- constructors so each public mutator returns its own error type.
  473 updateSessionMeta
  474   :: FilePath
  475   -> SessionId
  476   -> (SessionMeta -> SessionMeta)
  477   -> e
  478   -> (Text -> e)
  479   -> IO (Either e ())
  480 updateSessionMeta baseDir sid f missing parseFail = do
  481   let dir    = baseDir </> T.unpack (unSessionId sid)
  482       finalP = dir </> "session.json"
  483       tmpP   = finalP <> ".tmp"
  484   exists <- doesFileExist finalP
  485   if not exists
  486     then pure (Left missing)
  487     else do
  488       eBytes <- try (LBS.readFile finalP) :: IO (Either IOException LBS.ByteString)
  489       case eBytes of
  490         Left e -> pure (Left (parseFail (T.pack (show e))))
  491         Right raw -> case Aeson.eitherDecode' raw of
  492           Left  err  -> pure (Left (parseFail (T.pack err)))
  493           Right meta -> do
  494             LBS.writeFile tmpP (Aeson.encode (f meta))
  495             setFileMode tmpP 0o600
  496             renameFile tmpP finalP
  497             pure (Right ())

  499 -- ----------------------------------------------------------------------------
  500 -- Resume context reload
  501 -- ----------------------------------------------------------------------------

  503 -- | Load a bounded window of recent 'Message' values from a session's
  504 -- transcript, oldest-first.
  505 --
  506 -- @loadRecentMessages th maxCount maxTokens@ reads every recorded
  507 -- transcript entry via '_th_query', converts each to a text 'Message'
  508 -- (mapping 'Request' → 'User' and 'Response' → 'Assistant'), keeps at
  509 -- most the last @maxCount@, then walks that window from newest back to
  510 -- oldest accumulating an estimated token budget (computed as
  511 -- @T.length payload `div` 4@ — a rough heuristic, not a true tokenizer).
  512 -- Once the budget is exhausted no further messages are added.
  513 --
  514 -- If the transcript contains a compaction boundary (an entry whose
  515 -- metadata includes the 'compactionMetadataKey'), only entries from the
  516 -- last such boundary onward are considered.  The compaction entry itself
  517 -- is included — its payload carries the summary text that replaces the
  518 -- earlier conversation.  This ensures that after a gateway restart the
  519 -- agent resumes with the compacted view rather than double-counting
  520 -- messages that were folded into the summary.
  521 --
  522 -- Returns the surviving messages in chronological (oldest-first) order
  523 -- so they can be replayed directly into the context.
  524 loadRecentMessages :: TranscriptHandle -> Int -> Int -> IO [Message]
  525 loadRecentMessages th maxCount maxTokens = do
  526   entries <- _th_query th TranscriptFilter
  527     { _tf_harness   = Nothing
  528     , _tf_model     = Nothing
  529     , _tf_direction = Nothing
  530     , _tf_timeRange = Nothing
  531     , _tf_limit     = Nothing
  532     }
  533   let -- Trim to entries from the last compaction boundary onward.
  534       postCompaction = trimToLastCompaction entries
  535       total     = length postCompaction
  536       countWin  = if total > maxCount
  537                     then drop (total - maxCount) postCompaction
  538                     else postCompaction
  539       -- Walk newest→oldest, adding to the budget; stop at the first
  540       -- entry that would push us OVER the limit. Always include at
  541       -- least one entry if the window is non-empty, so a single
  542       -- oversized message is not silently dropped.
  543       reversed  = reverse countWin
  544       budgeted  = goBudget 0 True reversed
  545       goBudget _    _     [] = []
  546       goBudget used first (e:es) =
  547         let cost  = T.length (_te_payload e) `div` 4
  548             used' = used + cost
  549         in if used' > maxTokens && not first
  550              then []
  551              else e : goBudget used' False es
  552   pure (map entryToMessage (reverse budgeted))
  553   where
  554     entryToMessage e =
  555       let role = case _te_direction e of
  556             Request  -> User
  557             Response -> Assistant
  558       in Message role [TextBlock (_te_payload e)]
  559 
  560 -- | If any entry carries the compaction metadata key, return entries
  561 -- from the last such entry onward (inclusive).  Otherwise return the
  562 -- full list unchanged.
  563 trimToLastCompaction :: [TranscriptEntry] -> [TranscriptEntry]
  564 trimToLastCompaction entries =
  565   case lastCompactionIdx of
  566     Nothing -> entries
  567     Just i  -> drop i entries
  568   where
  569     isCompaction e = Map.member compactionMetadataKey (_te_metadata e)
  570     lastCompactionIdx = go Nothing 0 entries
  571     go acc _ []     = acc
  572     go acc n (e:es)
  573       | isCompaction e = go (Just n) (n + 1) es
  574       | otherwise      = go acc      (n + 1) es
  575
