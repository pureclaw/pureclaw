   44 -- | Run the main agent loop. Reads messages from the channel, sends
   45 -- them to the provider (with tool definitions), handles tool call/result
   46 -- cycles, and writes responses back.
   47 --
   48 -- Slash commands (messages starting with '/') are intercepted and
   49 -- handled before being sent to the provider.
   50 --
   51 -- If no provider is configured ('Nothing' in the IORef), chat messages
   52 -- produce a helpful error directing the user to configure credentials.
   53 -- Slash commands always work regardless of provider state.
   54 --
   55 -- Exits cleanly on 'IOException' from the channel (e.g. EOF / Ctrl-D).
   56 -- Provider errors are logged and a 'PublicError' is sent to the channel.
   57 --
   58 -- == WU10 (Tabbed Chat #51) refactor status
   59 --
   60 -- The original WU10 plan called for 'runAgentLoop' to become a thin
   61 -- wrapper around 'PureClaw.Routing.Dispatcher.runDispatcher'. WU10
   62 -- escalated this part of the refactor: the WU6 AI tab loop in
   63 -- 'PureClaw.Tab.Ai' is not yet a drop-in replacement for the
   64 -- single-tab semantics this loop owns (model prefix emission on
   65 -- channel output, tool-call execution cycle, transcript provider
   66 -- wrapping via 'PureClaw.Transcript.Provider.mkTranscriptProvider',
   67 -- '_env_onFirstStreamDone' callback, harness routing via
   68 -- 'PureClaw.Handles.Harness.HarnessHandle'). Migrating all of those
   69 -- features into 'Tab.Ai' is itself a multi-WU refactor and was
   70 -- deferred to WU11\/WU12.
   71 --
   72 -- For now 'runAgentLoop' is the production entry-point for the
   73 -- single-tab CLI flow; tabbed-chat is exercised via
   74 -- 'PureClaw.Routing.Dispatcher.runDispatcher' (which the K-series
   75 -- tests in @test\/Coexistence\/SlashCmdSpec.hs@ drive directly). The
   76 -- two entry points coexist until the Tab.Ai feature gap closes.
   77 runAgentLoop :: AgentEnv -> IO ()
   78 runAgentLoop env = runAgentLoopWith env []

   80 -- | Like 'runAgentLoop' but seeds the initial 'Context' with the given
   81 -- messages (in chronological, oldest-first order). Used by the resume
   82 -- path to replay recent transcript entries so the agent has memory of
   83 -- prior turns.
   84 runAgentLoopWith :: AgentEnv -> [Message] -> IO ()
   85 runAgentLoopWith env initialMessages = do
   86   _lh_logInfo logger "Agent loop started"
   87   let ctx0 = replaceMessages initialMessages (emptyContext (_env_systemPrompt env))
   88   go ctx0
   89   where
   90     channel  = _env_channel env
   91     logger   = _env_logger env
   92     baseRegistry = _env_registry env
   93 
   94     -- | Build the effective registry by merging built-in tools with
   95     -- any connected MCP server tools.
   96     effectiveRegistry :: IO ToolRegistry
   97     effectiveRegistry = do
   98       servers <- readIORef (_env_mcpServers env)
   99       if Map.null servers
  100         then pure baseRegistry
  101         else pure $ mergeRegistries baseRegistry (mcpRegistry (Map.elems servers))
  102 
  103     go ctx = do
  104       receiveResult <- try @IOException (_ch_receive channel)
  105       case receiveResult of
  106         Left _ -> _lh_logInfo logger "Session ended"
  107         Right msg -> do
  108           -- Capture the session origin (set-once) BEFORE any slash/harness/
  109           -- /bg/provider branching, covering the empty-message branch too:
  110           -- origin is about the SENDER, not the content. This is the
  111           -- single-tab loop's sole inbound entry; the tabbed dispatcher
  112           -- runtime is a separate path (out of scope, see design WU3).
  113           --
  114           -- SECURITY: _sm_source is attacker-asserted provenance and MUST
  115           -- NOT feed any access-control decision.
  116           sh <- readIORef (_env_session env)
  117           Session.setSourceIfAbsent sh (_im_source msg)
  118           dispatchMsg
  119           where
  120             stripped = T.strip (_im_content msg)
  121             dispatchMsg
  122               | T.null stripped = go ctx
  123               -- INVARIANT: any message beginning with '/' is handled locally
  124               -- and NEVER forwarded to the provider. Unknown slash commands
  125               -- get an error response rather than silently routing to the LLM.
  126               | "/" `T.isPrefixOf` stripped =
  127                   case parseSlashCommand stripped of
  128                     Just (CmdTab tabCmd) -> do
  129                       -- Tab commands need callbacks (SpawnIO, PromptRenderer,
  130                       -- BannerEmit, etc.) that 'executeSlashCommand' can't wire
  131                       -- without a dependency cycle. Dispatch via the legacy
  132                       -- bridge instead. The bridge invokes the canonical
  133                       -- 'Routing.AutoSpawn' handlers so /tab* commands behave
  134                       -- the same way they would under the new dispatcher.
  135                       _lh_logInfo logger $ "Slash command (tab): " <> stripped
  136                       dispatchLegacyTabCmd env tabCmd
  137                       go ctx
  138                     Just (CmdBg prompt) -> do
  139                       -- /bg runs the prompt in a fresh background session
  140                       -- (issue #52). The single-tab loop has no tabbed-chat
  141                       -- dispatcher, so we fork a self-contained background
  142                       -- turn that emits its result directly to the channel
  143                       -- (the ChannelOut writer is not running in this path).
  144                       -- The foreground loop continues uninterrupted.
  145                       _lh_logInfo logger $ "Slash command (bg): " <> stripped
  146                       _ch_send channel (OutgoingMessage
  147                         "\x1F504 /bg: running in the background \x2014 the result will appear here when ready.")
  148                       _ <- _env_fork env (runBackgroundTurn env prompt)
  149                       go ctx
  150                     Just cmd -> do
  151                       _lh_logInfo logger $ "Slash command: " <> stripped
  152                       ctx' <- executeSlashCommand env cmd ctx
  153                       go ctx'
  154                     Nothing -> do
  155                       _lh_logWarn logger $ "Unrecognized slash command: " <> stripped
  156                       _ch_send channel
  157                         (OutgoingMessage ("Unknown command: " <> stripped
  158                           <> "\nType /status for session info, /help for available commands."))
  159                       go ctx
  160               | otherwise = do
  161                   target <- readIORef (_env_target env)
  162                   case target of
  163                     TargetHarness name -> do
  164                       harnesses <- readIORef (_env_harnesses env)
  165                       case Map.lookup name harnesses of
  166                         Nothing -> do
  167                           _ch_send channel (OutgoingMessage
  168                             ("Harness \"" <> name <> "\" is not running. Use /harness start "
  169                               <> name <> " or /target to switch targets."))
  170                           go ctx
  171                         Just hh -> do
  172                           _lh_logInfo logger $ "Routing to harness: " <> name
  173                           _hh_send hh (TE.encodeUtf8 stripped)
  174                           output <- _hh_receive hh
  175                           let response = sanitizeHarnessOutput (TE.decodeUtf8 output)
  176                           unless (T.null (T.strip response)) $
  177                             _ch_send channel (OutgoingMessage (prefixHarnessOutput name response))
  178                           go ctx
  179                     TargetProvider -> do
  180                       mProvider <- readIORef (_env_provider env)
  181                       case mProvider of
  182                         Nothing -> do
  183                           _ch_send channel (OutgoingMessage noProviderMessage)
  184                           go ctx
  185                         Just provider -> do
  186                           mModel <- readIORef (_env_model env)
  187                           case mModel of
  188                             Nothing -> do
  189                               _ch_send channel (OutgoingMessage noModelMessage)
  190                               go ctx
  191                             Just model -> do
  192                               let userMsg = textMessage User stripped
  193                                   ctx' = addMessage userMsg ctx
  194                               _lh_logDebug logger $
  195                                 "Sending " <> T.pack (show (length (contextMessages ctx'))) <> " messages"
  196                               -- Wrap provider with transcript logging (session owns the transcript)
  197                               th <- envTranscript env
  198                               let provider' = mkTranscriptProvider th (unModelId model) (Just (_im_source msg)) provider
  199                               handleCompletion provider' ctx'
  200 
  201     handleCompletion provider ctx = do
  202       mModel <- readIORef (_env_model env)
  203       registry <- effectiveRegistry
  204       let tools = registryDefinitions registry
  205           model = fromMaybe (ModelId "") mModel
  206           modelName = unModelId model
  207           req = CompletionRequest
  208             { _cr_model        = model
  209             , _cr_messages     = contextMessages ctx
  210             , _cr_systemPrompt = contextSystemPrompt ctx
  211             , _cr_maxTokens    = Just 4096
  212             , _cr_tools        = tools
  213             , _cr_toolChoice   = Nothing
  214             }
  215       responseRef <- newIORef (Nothing :: Maybe CompletionResponse)
  216       streamedRef <- newIORef False
  217       prefixSentRef <- newIORef False
  218       providerResult <- try @SomeException $
  219         completeStream provider req $ \case
  220           StreamText t -> do
  221             -- Emit origin prefix before the first streamed chunk
  222             prefixSent <- readIORef prefixSentRef
  223             unless prefixSent $ do
  224               _ch_sendChunk channel (ChunkText (modelName <> "> "))
  225               writeIORef prefixSentRef True
  226             _ch_sendChunk channel (ChunkText t)
  227             writeIORef streamedRef True
  228           StreamDone resp -> do
  229             writeIORef responseRef (Just resp)
  230             -- Fire and clear the one-shot "first StreamDone" callback
  231             -- atomically so concurrent StreamDone deliveries cannot
  232             -- race and invoke it twice. In production this is used to
  233             -- mark the active session's bootstrap as consumed.
  234             mAction <- atomicModifyIORef' (_env_onFirstStreamDone env)
  235                          (Nothing,)
  236             for_ mAction id
  237           StreamWarning w -> _lh_logWarn logger w
  238           _ -> pure ()
  239       case providerResult of
  240         Left e -> do
  241           _lh_logError logger $ "Provider error: " <> T.pack (show e)
  242           _ch_sendError channel (TemporaryError "Something went wrong. Please try again.")
  243           go ctx
  244         Right () -> do
  245           wasStreaming <- readIORef streamedRef
  246           when wasStreaming $ _ch_sendChunk channel ChunkDone
  247           mResp <- readIORef responseRef
  248           case mResp of
  249             Nothing -> go ctx  -- shouldn't happen
  250             Just response -> do
  251               let calls = toolUseCalls response
  252                   text = responseText response
  253                   ctx' = recordUsage (_crsp_usage response)
  254                        $ addMessage (Message Assistant (_crsp_content response)) ctx
  255               -- Send the full text. For streaming channels, the text was already
  256               -- displayed chunk-by-chunk so we skip the full send to avoid duplicates.
  257               unless (wasStreaming && _ch_streaming channel || T.null (T.strip text)) $
  258                 _ch_send channel (OutgoingMessage (prefixHarnessOutput modelName text))
  259               -- If there are tool calls, execute them and continue
  260               if null calls
  261                 then go ctx'
  262                 else do
  263                   results <- mapM executeCall calls
  264                   let resultMsg = toolResultMessage results
  265                       ctx'' = addMessage resultMsg ctx'
  266                   _lh_logDebug logger $
  267                     "Executed " <> T.pack (show (length results)) <> " tool calls, continuing"
  268                   handleCompletion provider ctx''
  269 
  270     executeCall (callId, name, input) = do
  271       _lh_logInfo logger $ "Tool call: " <> name
  272       registry <- effectiveRegistry
  273       result <- executeTool registry name input
  274       case result of
  275         Nothing -> do
  276           _lh_logWarn logger $ "Unknown tool: " <> name
  277           pure (callId, [TRPText ("Unknown tool: " <> name)], True)
  278         Just (parts, isErr) -> do
  279           when isErr $ _lh_logWarn logger $ "Tool error in " <> name <> ": " <> partsToText parts
  280           pure (callId, parts, isErr)
  281 
  282     partsToText :: [ToolResultPart] -> Text
  283     partsToText parts = T.intercalate "\n" [t | TRPText t <- parts]

  289 -- | Run a @\/bg@ prompt in a fresh background session and push the result
  290 -- directly to the channel (issue #52).
  291 --
  292 -- The background turn runs in its OWN session — a brand-new
  293 -- 'PureClaw.Session.Handle.SessionHandle' created under the same sessions
  294 -- directory as the foreground session and wired to the same
  295 -- 'PureClaw.Frontend.StreamBroker.StreamBroker'. The provider is wrapped
  296 -- with 'mkTranscriptProvider' so every request/response is recorded to the
  297 -- session's @transcript.jsonl@ (and broadcast to the broker). This is what
  298 -- makes the background conversation appear in the frontend UI like any
  299 -- other conversation: the frontend enumerates session directories on disk
  300 -- and subscribes to broker events. The fresh session also means the
  301 -- background turn does NOT leak into the foreground conversation history.
  302 --
  303 -- It uses the process default provider/model and the effective tool
  304 -- registry (built-ins + connected MCP servers), runs to completion via
  305 -- 'runSubAgent' (non-streaming, so it does not interleave with the
  306 -- foreground), and emits a single @[bg done] …@ message via '_ch_send'.
  307 -- This intentionally does NOT go through '_env_channelOutQ' — the
  308 -- single-tab CLI loop does not run the 'PureClaw.Routing.ChannelOut'
  309 -- writer, so a queued event would never be delivered.
  310 --
  311 -- Provider/model-absent and provider-failure cases each emit a short,
  312 -- redacted @[bg] …@ message rather than throwing (the caller forks this
  313 -- with '_env_fork' and does not observe its result).
  314 runBackgroundTurn :: AgentEnv -> Text -> IO ()
  315 runBackgroundTurn env prompt = do
  316   let channel = _env_channel env
  317   mProvider <- readIORef (_env_provider env)
  318   mModel    <- readIORef (_env_model env)
  319   case (mProvider, mModel) of
  320     (Nothing, _) ->
  321       _ch_send channel (OutgoingMessage "[bg] Cannot run: no provider configured.")
  322     (_, Nothing) ->
  323       _ch_send channel (OutgoingMessage "[bg] Cannot run: no model configured.")
  324     (Just provider, Just model) -> do
  325       outcome <- try @SomeException (runBackgroundSession env provider model prompt)
  326       case outcome of
  327         Left e -> do
  328           _lh_logError (_env_logger env) $
  329             "Background task error: " <> T.pack (show e)
  330           _ch_send channel (OutgoingMessage
  331             "[bg] Something went wrong running the background task.")
  332         Right text ->
  333           let body = if T.null (T.strip text) then "(no response)" else text
  334           in _ch_send channel (OutgoingMessage ("[bg done] " <> body))

  336 -- | Create the fresh background session, run the prompt against a
  337 -- transcript-recording provider, persist + close the session, and return
  338 -- the final assistant text. Any exception propagates to 'runBackgroundTurn'
  339 -- (which reports a redacted failure); the session is saved + closed
  340 -- regardless via 'finally'.
  341 runBackgroundSession :: AgentEnv -> SomeProvider -> ModelId -> Text -> IO Text
  342 runBackgroundSession env provider model prompt = do
  343   registry  <- backgroundRegistry env
  344   bgSession <- mkBackgroundSession env model prompt
  345   let transcript = Session._sh_transcript bgSession
  346       -- Recording wrapper: each provider call writes a request/response
  347       -- pair to the session transcript (and fans out to the broker).
  348       provider'  = mkTranscriptProvider transcript (unModelId model) Nothing provider
  349   runSubAgent provider' model registry (_env_systemPrompt env)
  350               prompt backgroundMaxTurns
  351     `finally` do
  352       ignoreExc (Session._sh_save bgSession)
  353       ignoreExc (Transcript._th_close transcript)

  355 -- | Build a fresh on-disk session for a @\/bg@ turn, rooted under the same
  356 -- sessions directory as the foreground session (so the frontend — which
  357 -- scans that directory — discovers it) and wired to '_env_broker' (so its
  358 -- transcript writes broadcast live).
  359 mkBackgroundSession :: AgentEnv -> ModelId -> Text -> IO Session.SessionHandle
  360 mkBackgroundSession env model prompt = do
  361   now         <- getCurrentTime
  362   sessionsDir <- backgroundSessionsDir env
  363   createDirectoryIfMissing True sessionsDir
  364   let modelTxt = unModelId model
  365       mAgent   = AgentDef._ad_name <$> _env_agentDef env
  366       meta = SessionTypes.SessionMeta
  367         { SessionTypes._sm_id    = SessionTypes.newSessionId Nothing now
  368         , SessionTypes._sm_agent = mAgent
  369         , SessionTypes._sm_kind  = SessionTypes.SkProvider
  370             (SessionTypes.ProviderSpec
  371               (SessionTypes.inferProviderId modelTxt) model mAgent)
  372         , SessionTypes._sm_model   = modelTxt
  373         , SessionTypes._sm_channel = "bg"
  374         , SessionTypes._sm_createdAt  = now
  375         , SessionTypes._sm_lastActive = now
  376         , SessionTypes._sm_bootstrapConsumed = False
  377         , SessionTypes._sm_archived = False
  378         , SessionTypes._sm_description = Just (backgroundDescription prompt)
  379         , SessionTypes._sm_autoSummary = Nothing
  380         , SessionTypes._sm_source = Nothing
  381         }
  382   Session.mkSessionHandle (_env_broker env) (_env_logger env) sessionsDir meta

  384 -- | The sessions directory a @\/bg@ session should be created under: the
  385 -- parent of the foreground session's directory (which is exactly the
  386 -- directory the frontend enumerates). Falls back to 'getSessionsDir' when
  387 -- the foreground session has no on-disk directory (e.g. a no-op handle).
  388 backgroundSessionsDir :: AgentEnv -> IO FilePath
  389 backgroundSessionsDir env = do
  390   sh <- readIORef (_env_session env)
  391   let dir = Session._sh_dir sh
  392   if null dir then getSessionsDir else pure (takeDirectory dir)

  394 -- | A short sidebar label for a @\/bg@ session.
  395 backgroundDescription :: Text -> Text
  396 backgroundDescription prompt = "/bg: " <> T.take 80 (T.strip prompt)

  398 -- | Run an IO action and swallow its exception. Used to make the
  399 -- background session's save/close best-effort so a flush failure does not
  400 -- mask the turn's own result. Although this catches 'SomeException', it is
  401 -- only ever invoked inside 'finally's finalizer (which runs with async
  402 -- exceptions masked), so in practice it swallows synchronous IO failures
  403 -- from save/close — not an asynchronous 'AsyncCancelled'.
  404 ignoreExc :: IO () -> IO ()
  405 ignoreExc m = m `catch` \(_ :: SomeException) -> pure ()

  407 -- | The effective tool registry for a background turn: built-in tools
  408 -- merged with any connected MCP server tools. Mirrors the @effectiveRegistry@
  409 -- helper inside 'runAgentLoopWith'.
  410 backgroundRegistry :: AgentEnv -> IO ToolRegistry
  411 backgroundRegistry env = do
  412   servers <- readIORef (_env_mcpServers env)
  413   let base = _env_registry env
  414   pure $ if Map.null servers
  415            then base
  416            else mergeRegistries base (mcpRegistry (Map.elems servers))
