    1 -- |
    2 -- Module      : PureClaw.Internal.FakeClock
    3 -- Description : Deterministic in-process fake clock for tests (WU3).
    4 --
    5 -- A 'FakeClock' is an in-process integer time source measured in
    6 -- milliseconds. It is used by 'PureClaw.Handles.Backend.mkInMemoryBackendHandle'
    7 -- (and, later, by @fakePtyIO@ in WU7) so property tests can advance
    8 -- time without sleeping the test runner.
    9 --
   10 -- The clock starts at @0@. Callers advance it explicitly with
   11 -- 'advanceFakeClock'; there is no automatic wall-clock progression.
   12 --
   13 -- Listeners registered with 'forkClockListener' are invoked
   14 -- synchronously inside 'advanceFakeClock' with the new time. The
   15 -- returned action unregisters the listener (idempotent).
   16 module PureClaw.Internal.FakeClock
   17   ( FakeClock
   18   , newFakeClock
   19   , currentFakeTime
   20   , advanceFakeClock
   21   , forkClockListener
   22   ) where
   23 
   24 import Control.Concurrent.STM (atomically)
   25 import Control.Concurrent.STM.TVar
   26   ( TVar
   27   , modifyTVar'
   28   , newTVarIO
   29   , readTVar
   30   , readTVarIO
   31   , writeTVar
   32   )
   33 import Data.IORef (IORef, atomicModifyIORef', newIORef, readIORef)
   34 
   35 -- | Internal listener identifier; increments monotonically.
   36 newtype ListenerId = ListenerId Int
   37   deriving stock (Eq, Ord)
   38 
   39 -- | A deterministic millisecond clock used for tests.
   40 --
   41 -- The constructor is intentionally NOT exported: callers obtain a
   42 -- clock via 'newFakeClock' and observe it via 'currentFakeTime'.
   43 data FakeClock = FakeClock
   44   { _fc_nowMs     :: !(IORef Int)
   45   , _fc_listeners :: !(TVar [(ListenerId, Int -> IO ())])
   46   , _fc_nextId    :: !(IORef Int)
   47   }
   48 
   49 -- | Construct a fresh fake clock starting at @0@.
   50 newFakeClock :: IO FakeClock
   51 newFakeClock = do
   52   nowRef <- newIORef 0
   53   ls     <- newTVarIO []
   54   idRef  <- newIORef 0
   55   pure FakeClock
   56     { _fc_nowMs     = nowRef
   57     , _fc_listeners = ls
   58     , _fc_nextId    = idRef
   59     }
   60 
   61 -- | The current fake time in milliseconds since the clock was constructed.
   62 currentFakeTime :: FakeClock -> IO Int
   63 currentFakeTime = readIORef . _fc_nowMs
   64 
   65 -- | Advance the clock by the given number of milliseconds. Listeners
   66 -- registered via 'forkClockListener' are fired synchronously, in
   67 -- registration order, with the new time.
   68 --
   69 -- Negative deltas are clamped to zero (the clock never goes backwards).
   70 advanceFakeClock :: FakeClock -> Int -> IO ()
   71 advanceFakeClock clock deltaMs = do
   72   let delta = max 0 deltaMs
   73   newTime <- atomicModifyIORef' (_fc_nowMs clock) $ \t ->
   74     let t' = t + delta in (t', t')
   75   ls <- readTVarIO (_fc_listeners clock)
   76   mapM_ (\(_, cb) -> cb newTime) ls
   77 
   78 -- | Register a listener; returns an action that unregisters it.
   79 --
   80 -- The unregister action is safe to call more than once (subsequent
   81 -- calls are no-ops).
   82 forkClockListener :: FakeClock -> (Int -> IO ()) -> IO (IO ())
   83 forkClockListener clock cb = do
   84   newId <- atomicModifyIORef' (_fc_nextId clock) $ \n ->
   85     let n' = n + 1 in (n', ListenerId n)
   86   atomically $ modifyTVar' (_fc_listeners clock) ((newId, cb) :)
   87   pure $ atomically $ do
   88     cur <- readTVar (_fc_listeners clock)
   89     writeTVar (_fc_listeners clock) (filter ((/= newId) . fst) cur)
