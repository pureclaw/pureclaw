    1 -- |
    2 -- Module      : PureClaw.Tab.Harness
    3 -- Description : Harness tab factory (Tabbed Chat WU7).
    4 --
    5 -- A harness tab is a 'PureClaw.Handles.Tab.TabHandle' that wraps an
    6 -- existing 'PureClaw.Handles.Harness.HarnessHandle' (e.g. a tmux-managed
    7 -- @claude-code@ subprocess). The tab owns two helper threads:
    8 --
    9 --   * a /drainer/ that loops on '_hh_receive' and emits each non-empty
   10 --     output chunk as @'FullMsg' !TabIndex !Text@ via '_env_channelOutQ'
   11 --     (focus-gated at the producer side per D4 and at the writer side
   12 --     per D3);
   13 --   * a /writer/ that drains a bounded outbound queue and forwards
   14 --     bytes to '_hh_send', keeping the public 'TabHandle._tabHandle_send'
   15 --     non-blocking per H4.
   16 --
   17 -- == Factory shape
   18 --
   19 -- @
   20 -- 'mkTabHarness' :: 'PureClaw.Agent.Env.AgentEnv'
   21 --                -> 'PureClaw.Handles.Tab.TabIndex'
   22 --                -> 'PureClaw.Handles.Tab.HarnessSpawnArgs'
   23 --                -> IO (Either 'PureClaw.Handles.Tab.TabError'
   24 --                              'PureClaw.Handles.Tab.TabHandle')
   25 -- @
   26 --
   27 -- Mirrors the WU6 'PureClaw.Tab.Ai.mkTabAi' factory pattern: takes
   28 -- 'AgentEnv', returns either a constructed 'TabHandle' or a
   29 -- redacted 'TabError'.
   30 --
   31 -- == HarnessHandle lookup
   32 --
   33 -- The 'HarnessSpawnArgs._harness_requestedName' field is the lookup
   34 -- key for '_env_harnesses' (which is the existing
   35 -- @IORef (Map Text HarnessHandle)@). Harness lifecycle (start \/ stop)
   36 -- is owned by the existing @\/harness@ slash command surface; the
   37 -- tab factory only /adopts/ a running harness. If the name does not
   38 -- resolve, the factory returns @'Left' ('TabNotFound' ix)@ where @ix@
   39 -- is the requested 'TabIndex' rendered as 'Int' (the public projection
   40 -- is still the channel-safe @\"tab: not found\"@ short label).
   41 --
   42 -- == Slash commands are unsupported (H13 \/ I4)
   43 --
   44 -- '_tabHandle_enqueueSlash' returns @'Left' ('TabUnsupportedCommand'
   45 -- cmd)@ immediately without enqueueing — harness tabs do not have a
   46 -- per-tab 'Context' or slash-command surface. Direct-inject text
   47 -- (@\/N \<text\>@) is forwarded verbatim to the harness via the
   48 -- writer thread so a payload like @\/N \/pwd@ reaches the harness as
   49 -- the literal @\"\/pwd\"@ (I4 — slash-prefix opaque to backend).
   50 --
   51 -- == Close semantics (H8 \/ H9)
   52 --
   53 -- Both 'CloseGraceful' and 'CloseForce' are destructive: they cancel
   54 -- both helper threads and call '_hh_stop' on the underlying
   55 -- 'HarnessHandle'. The harness is /not/ archived (there's no
   56 -- transcript-equivalent to flush). Idempotent + never-throws (H6 \/
   57 -- H7 contract).
   58 --
   59 -- == Async exception discipline
   60 --
   61 -- Helper threads catch 'SomeException' EXCEPT 'AsyncCancelled', which
   62 -- propagates so that '_tabHandle_close' (which cancels the helpers'
   63 -- 'TabRunner's) can unblock a stuck 'threadDelay' or pending
   64 -- '_hh_receive'. Mirrors the WU6 'safelyRunLoop' pattern.
   65 --
   66 -- See @docs\/tabbed-chat.md@ §"H-series" (close lifecycle), §"I4"
   67 -- (opaque slash-prefix), and §"D5" (FullMsg emission).
   68 module PureClaw.Tab.Harness

  148 -- ---------------------------------------------------------------------------
  149 -- HarnessTabState — per-tab mutable state, private to the factory closure
  150 -- ---------------------------------------------------------------------------
  151 
  152 -- | Per-harness-tab mutable state. Fields are exposed so a future
  153 -- diagnostic @tab info@ handler can inspect them; production code never
  154 -- touches the record outside this module.
  155 data HarnessTabState = HarnessTabState
  156   { _hts_harness        :: !HarnessHandle
  157     -- ^ The underlying harness adopted at spawn time. Looked up by
  158     -- name from '_env_harnesses'; the tab does NOT own the harness
  159     -- lifecycle (the existing @\/harness@ command surface does), but
  160     -- '_tabHandle_close' DOES call '_hh_stop' on it per H8 (destructive
  161     -- close — there is no archive equivalent).
  162   , _hts_sendQ          :: !(TBQueue ByteString)
  163     -- ^ Bounded outbound queue. '_tabHandle_send' enqueues here
  164     -- non-blockingly; the writer thread drains and calls '_hh_send'.
  165     -- Bounded by '_rc_inputQueueBound' so a saturating producer
  166     -- surfaces 'TabConcurrencyLimit' rather than blocking the
  167     -- dispatcher (H4).
  168   , _hts_drainRunner    :: !(IORef (Maybe TabRunner))
  169     -- ^ Drainer thread runner (output side: '_hh_receive' →
  170     -- 'FullMsg'). Filled by 'mkTabHarness' post-fork; stays 'Nothing'
  171     -- for the brief window between allocation and fork-fill.
  172   , _hts_writerRunner   :: !(IORef (Maybe TabRunner))
  173     -- ^ Writer thread runner (input side: '_hts_sendQ' →
  174     -- '_hh_send'). Filled by 'mkTabHarness' post-fork.
  175   , _hts_closed         :: !(IORef Bool)
  176     -- ^ Idempotency flag for '_tabHandle_close'. Flipped to 'True' on
  177     -- the first invocation; subsequent invocations are no-ops (H6).
  178   , _hts_statusRef      :: !(IORef TabStatus)
  179     -- ^ Held status backing '_tabHandle_status'. The drainer and
  180     -- writer threads do not transition this on a per-chunk basis (a
  181     -- harness tab is conceptually always available for I\/O); it
  182     -- starts 'Idle' at spawn and transitions to 'Crashed' if a helper
  183     -- thread hits a synchronous exception.
  184   }
  185

  191 -- | Microseconds the drainer sleeps between '_hh_receive' calls.
  192 --
  193 -- '_hh_receive' is a /snapshot/ read (it returns whatever output is
  194 -- currently buffered, possibly empty), so we cannot block on it. A
  195 -- 100 ms cadence keeps idle harness tabs from burning CPU while still
  196 -- producing responsive output on focused tabs.
  197 --
  198 -- Exposed for tests so they can pin the cadence; production callers
  199 -- never reference it directly.
  200 drainerSleepMicros :: Int
  201 drainerSleepMicros = 100000  -- 100 ms

  204 -- ---------------------------------------------------------------------------
  205 -- mkTabHarness — the harness tab factory
  206 -- ---------------------------------------------------------------------------
  207 
  208 -- | Construct a harness tab. Looks up the requested harness from
  209 -- '_env_harnesses', allocates per-tab state, forks the drainer and
  210 -- writer threads via '_env_fork', and returns a 'TabHandle' whose IO
  211 -- actions close over the state.
  212 --
  213 -- The factory itself never throws: it always returns @'Right' h@ or
  214 -- @'Left' e@. Failures inside the forked helper threads are caught by
  215 -- their own outer exception handler and surface as a 'Crashed' status
  216 -- (visible via '_tabHandle_status').
  217 mkTabHarness :: AgentEnv -> TabIndex -> HarnessSpawnArgs
  218              -> IO (Either TabError TabHandle)
  219 mkTabHarness env idx args =
  220   case Parse.sanitizeTabName (_harness_requestedName args) of
  221     Left nameErr -> pure (Left (TabInvalidName nameErr))
  222     Right nameTxt -> do
  223       mHarness <- lookupHarness env (_harness_requestedName args)
  224       case mHarness of
  225         Nothing -> pure (Left (TabNotFound (unTabIndex idx)))
  226         Just hh -> Right <$> spawnTabWithHarness env idx nameTxt hh
  227 
  228 -- | Look up the requested harness from '_env_harnesses' by name. We
  229 -- match by the raw (pre-sanitization) requested name because that's
  230 -- the key used by the existing @\/harness start \<name\>@ surface.
  231 lookupHarness :: AgentEnv -> Text -> IO (Maybe HarnessHandle)
  232 lookupHarness env name = do
  233   hs <- readIORef (_env_harnesses env)
  234   pure (Map.lookup name hs)
  235 
  236 -- | Allocate per-tab state, fork the drainer + writer threads, and
  237 -- build the public 'TabHandle' record.
  238 spawnTabWithHarness
  239   :: AgentEnv -> TabIndex -> Text -> HarnessHandle -> IO TabHandle
  240 spawnTabWithHarness env idx nameTxt hh = do
  241   state <- allocState env hh
  242   -- Sentinel: state starts Idle at allocation; the drainer thread does
  243   -- not own status transitions in WU7 (a harness tab is always
  244   -- available unless a helper thread crashes — see 'safelyRun').
  245   now <- getCurrentTime
  246   writeIORef (_hts_statusRef state) (Idle now)
  247   -- Fork the drainer (output) and writer (input) threads via the
  248   -- env's fork seam. Both runners are captured into the state so
  249   -- '_tabHandle_close' can cancel them.
  250   drainRunner  <- _env_fork env (drainerLoop env idx state)
  251   writeIORef (_hts_drainRunner state) (Just drainRunner)
  252   writerRunner <- _env_fork env (writerLoop state)
  253   writeIORef (_hts_writerRunner state) (Just writerRunner)
  254   pure (mkHandle env idx (TabName nameTxt) state)
  255 
  256 -- | Allocate the per-tab state. The bounded outbound queue's capacity
  257 -- comes from '_rc_inputQueueBound' so the H4 \"non-blocking send\"
  258 -- contract is configurable.
  259 allocState :: AgentEnv -> HarnessHandle -> IO HarnessTabState
  260 allocState env hh = do
  261   let rc = _env_routingConfig env
  262   sendQ        <- newTBQueueIO (fromIntegral (_rc_inputQueueBound rc))
  263   drainRef     <- newIORef Nothing
  264   writerRef    <- newIORef Nothing
  265   closedRef    <- newIORef False
  266   statRef      <- newIORef Active  -- replaced with Idle by spawnTabWithHarness
  267   pure HarnessTabState
  268     { _hts_harness        = hh
  269     , _hts_sendQ          = sendQ
  270     , _hts_drainRunner    = drainRef
  271     , _hts_writerRunner   = writerRef
  272     , _hts_closed         = closedRef
  273     , _hts_statusRef      = statRef
  274     }
  275 
  276 -- | Build the public 'TabHandle' record from the per-tab state.
  277 mkHandle :: AgentEnv -> TabIndex -> TabName -> HarnessTabState -> TabHandle
  278 mkHandle env idx name state =
  279   let harnSpec = HarnessSpec
  280         { _h_flavour = HClaudeCode
  281         , _h_backend = TbLocal
  282         , _h_cwd     = Nothing
  283         , _h_args    = []
  284         }
  285   in TabHandle
  286   { _tabHandle_index        = idx
  287   , _tabHandle_name         = name
  288   , _tabHandle_kind         = TkSession (SkHarness harnSpec)
  289   , _tabHandle_status       = readIORef (_hts_statusRef state)
  290   , _tabHandle_send         = sendBytes state
  291   , _tabHandle_enqueueSlash = enqueueSlashUnsupported
  292   , _tabHandle_close        = closeTabHarness env state
  293   }

  296 -- ---------------------------------------------------------------------------
  297 -- _tabHandle_send / _tabHandle_enqueueSlash
  298 -- ---------------------------------------------------------------------------
  299 
  300 -- | Enqueue a 'Text' payload onto the outbound queue. Non-blocking:
  301 -- returns @'Left' ('TabConcurrencyLimit' 0)@ if the queue is full so
  302 -- the dispatcher's send loop never blocks (H4).
  303 --
  304 -- The text is UTF-8 encoded and forwarded verbatim — including any
  305 -- leading @\/@ — so a direct-inject like @\/N \/pwd@ reaches the
  306 -- harness as the literal @\"\/pwd\"@ (I4: slash-prefix opaque to the
  307 -- backend).
  308 sendBytes :: HarnessTabState -> Text -> IO (Either TabError ())
  309 sendBytes state t = atomically (tryEnqueueSTM (_hts_sendQ state) (TE.encodeUtf8 t))
  310 
  311 -- | STM body of 'sendBytes', factored out so tests can drive it inside
  312 -- a larger transaction if needed.
  313 tryEnqueueSTM :: TBQueue ByteString -> ByteString -> STM (Either TabError ())
  314 tryEnqueueSTM q bs = do
  315   full <- isFullTBQueue q
  316   if full
  317     then pure (Left (TabConcurrencyLimit 0))
  318     else writeTBQueue q bs >> pure (Right ())
  319 
  320 -- | '_tabHandle_enqueueSlash' for a harness tab: slash commands are
  321 -- not supported, so we return @'Left' ('TabUnsupportedCommand' cmd)@
  322 -- immediately without enqueueing (H13 \/ I4 contract).
  323 enqueueSlashUnsupported :: SlashCommand -> IO (Either TabError ())
  324 enqueueSlashUnsupported cmd = pure (Left (TabUnsupportedCommand cmd))

  327 -- ---------------------------------------------------------------------------
  328 -- _tabHandle_close — destructive close for harness tabs
  329 -- ---------------------------------------------------------------------------
  330 
  331 -- | Close a harness tab. Idempotent + never throws (H6, H7).
  332 --
  333 -- Both 'CloseGraceful' and 'CloseForce' are destructive: there is no
  334 -- harness-side archive equivalent to flush. The triad below cancels
  335 -- both helper threads and then calls '_hh_stop'. Each step is wrapped
  336 -- in 'safeIgnore' so a misbehaving harness can't leak an exception out
  337 -- of the never-throws contract.
  338 closeTabHarness :: AgentEnv -> HarnessTabState -> CloseMode -> IO ()
  339 closeTabHarness _env state mode = do
  340   alreadyClosed <- atomicModifyIORef' (_hts_closed state) (True,)
  341   unless alreadyClosed $ do
  342     safeIgnore (cancelMaybeRunner (_hts_drainRunner state))
  343     safeIgnore (cancelMaybeRunner (_hts_writerRunner state))
  344     -- H9: 'CloseGraceful' and 'CloseForce' are semantically identical
  345     -- for non-AI tabs (close is already destructive). The pattern-match
  346     -- below is here for symmetry with the WU6 KindAi factory and so
  347     -- coverage tools see both arms exercised.
  348     case mode of
  349       CloseGraceful -> safeIgnore (_hh_stop (_hts_harness state))
  350       CloseForce    -> safeIgnore (_hh_stop (_hts_harness state))
  351 
  352 -- | Read the captured 'TabRunner' (set by 'mkTabHarness' after fork)
  353 -- and invoke its cancel. Tolerant of the brief allocation\/fork window
  354 -- when the IORef is still 'Nothing'.
  355 cancelMaybeRunner :: IORef (Maybe TabRunner) -> IO ()
  356 cancelMaybeRunner ref = do
  357   mRunner <- readIORef ref
  358   for_ mRunner _trun_cancel
  359 
  360 -- | Best-effort: run an IO action and swallow synchronous failures.
  361 -- The close path MUST be never-throws (H7), so every step is wrapped
  362 -- defensively. 'AsyncCancelled' is also swallowed here because the
  363 -- close handler is invoked from outside the helper threads; any
  364 -- AsyncCancelled bubbling up through cancel itself is benign (the
  365 -- target thread has already received the cancel).
  366 safeIgnore :: IO () -> IO ()
  367 safeIgnore m = do
  368   _ <- try @SomeException m
  369   pure ()

  372 -- ---------------------------------------------------------------------------
  373 -- Drainer thread — _hh_receive → FullMsg via _env_channelOutQ
  374 -- ---------------------------------------------------------------------------
  375 
  376 -- | Drainer-thread top-level. Wraps 'drainerStep' in 'safelyRun' so a
  377 -- synchronous exception transitions the tab to 'Crashed' rather than
  378 -- killing the parent thread, and so 'AsyncCancelled' propagates per
  379 -- the C5 contract.
  380 drainerLoop :: AgentEnv -> TabIndex -> HarnessTabState -> IO ()
  381 drainerLoop env idx state = safelyRun state loop
  382   where
  383     loop = do
  384       drainerStep env idx state
  385       loop
  386 
  387 -- | One iteration of the drainer. Reads '_hh_receive', emits a
  388 -- 'FullMsg' if the result is non-empty and the tab is focused (D4
  389 -- producer-side skip), then sleeps to avoid CPU burn on idle harnesses.
  390 --
  391 -- Also checks '_hh_status': if the harness has exited, we stop the
  392 -- drainer (the writer thread will eventually surface the next send
  393 -- as a no-op since the harness pipe is closed).
  394 drainerStep :: AgentEnv -> TabIndex -> HarnessTabState -> IO ()
  395 drainerStep env idx state = do
  396   -- Check status first so an exited harness terminates the loop
  397   -- quickly. '_hh_status' is documented total in the no-op handle;
  398   -- for the real claude-code variant it inspects tmux output.
  399   st <- _hh_status (_hts_harness state)
  400   case st of
  401     HarnessExited _ ->
  402       -- Harness has exited — drainer's job is done. The tab handle
  403       -- remains valid until '_tabHandle_close' is invoked; closing
  404       -- will be a no-op on the underlying harness which is already
  405       -- gone.
  406       throwIO HarnessExitedSignal
  407     HarnessRunning -> do
  408       raw <- _hh_receive (_hts_harness state)
  409       unless (BS.null raw) $ do
  410         -- Decode + sanitize for display. The sanitizer strips ANSI,
  411         -- decorative Unicode, and trims surrounding blank lines.
  412         let txt = sanitizeHarnessOutput (TE.decodeUtf8 raw)
  413         unless (T.null (T.strip txt)) $ do
  414           -- D4 producer-side focus check: skip the enqueue work when
  415           -- the tab is not focused. The writer thread re-applies the
  416           -- predicate on dequeue (D3), so this is an optimisation
  417           -- only; correctness comes from the writer.
  418           curFocus <- readIORef (_env_focus env)
  419           when (shouldEmit curFocus (SrcTab idx)) $
  420             atomically $ writeTBQueue (_env_channelOutQ env)
  421                            (SrcTab idx, FullMsg idx txt)
  422       threadDelay drainerSleepMicros
  423 
  424 
  425 -- ---------------------------------------------------------------------------
  426 -- Writer thread — _hts_sendQ → _hh_send
  427 -- ---------------------------------------------------------------------------
  428 
  429 -- | Writer-thread top-level. Drains '_hts_sendQ' forever, forwarding
  430 -- each chunk to '_hh_send'. Wrapped in 'safelyRun' for the same
  431 -- AsyncCancelled \/ Crashed semantics as the drainer.
  432 writerLoop :: HarnessTabState -> IO ()
  433 writerLoop state = safelyRun state loop
  434   where
  435     loop = do
  436       bs <- atomically (readTBQueue (_hts_sendQ state))
  437       -- '_hh_send' may block (it writes to a tmux pipe). We tolerate
  438       -- the block because the bounded '_hts_sendQ' keeps the producer
  439       -- side non-blocking; the writer is a single-purpose drainer and
  440       -- nothing else depends on its progress.
  441       _hh_send (_hts_harness state) bs
  442       loop
  443

  445 -- ---------------------------------------------------------------------------
  446 -- safelyRun — shared exception handler for drainer + writer
  447 -- ---------------------------------------------------------------------------
  448 
  449 -- | A private signal raised by the drainer when '_hh_status' reports
  450 -- the harness has exited. Caught by 'safelyRun' so the drainer exits
  451 -- silently (without transitioning the tab to Crashed); other
  452 -- synchronous exceptions DO transition to Crashed.
  453 --
  454 -- 'Show' is hand-written (not derived) so HPC's per-derived-method
  455 -- accounting does not flag the auto-generated instance code as
  456 -- uncovered. The single constructor name suffices for the exception
  457 -- machinery.
  458 data HarnessExitedSignal = HarnessExitedSignal
  459 
  460 instance Show HarnessExitedSignal where
  461   show _ = "HarnessExitedSignal"
  462 
  463 instance Exception HarnessExitedSignal
  464 
  465 -- | Outer exception handler shared by both helper threads.
  466 --
  467 -- * 'AsyncCancelled' propagates (re-raised) so '_tabHandle_close'
  468 --   semantics work correctly per C5.
  469 -- * 'HarnessExitedSignal' is swallowed silently — the harness exit is
  470 --   an expected lifecycle event, not a crash.
  471 -- * Other 'SomeException' transitions the tab to 'Crashed' (carrying
  472 --   a redacted short label) and the helper thread exits.
  473 safelyRun :: HarnessTabState -> IO () -> IO ()
  474 safelyRun state body = body `catch` handler
  475   where
  476     handler :: SomeException -> IO ()
  477     handler e
  478       | Just AsyncCancelled        <- fromException e =
  479           throwIO e  -- propagate AsyncCancelled (C5)
  480       | Just HarnessExitedSignal   <- fromException e =
  481           pure ()    -- expected exit; not a crash
  482       | otherwise =
  483           writeIORef (_hts_statusRef state)
  484                      (Crashed (PublicTabError "tab: harness loop crashed"))
