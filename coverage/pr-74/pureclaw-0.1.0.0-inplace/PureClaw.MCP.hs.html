   58 -- | An active MCP server connection.
   59 data McpServer = McpServer
   60   { _ms_name       :: Text
   61     -- ^ User-assigned name for this server (e.g. "filesystem", "time")
   62   , _ms_command    :: [Text]
   63     -- ^ The command + args used to spawn the server (for display)
   64   , _ms_session    :: ClientSession
   65     -- ^ Active MCP client session
   66   , _ms_tools      :: [Tool]
   67     -- ^ Tools discovered during initialization
   68   , _ms_serverInfo :: Maybe Implementation
   69     -- ^ Server identity from initialization
   70   , _ms_cleanup    :: IO ()
   71     -- ^ Shuts down the process and background threads
   72   }
   73 
   74 -- | Connect to an MCP server by spawning it as a child process.
   75 --
   76 -- The server process communicates over stdio (JSON-RPC). On success,
   77 -- returns the connected server with its discovered tools. On failure,
   78 -- throws 'MCPClientError' or 'IOException'.
   79 connectServer :: LogHandle -> Text -> [Text] -> CreateProcess -> IO McpServer
   80 connectServer logger serverName cmdArgs cp = do
   81   _lh_logInfo logger $ "MCP: connecting to " <> serverName <> "..."
   82 
   83   -- Spawn the child process
   84   let cp' = cp { std_in = CreatePipe, std_out = CreatePipe, std_err = Inherit }
   85   (Just hIn, Just hOut, _hErr, ph) <- createProcess cp'
   86   hSetBuffering hIn LineBuffering
   87   hSetBuffering hOut LineBuffering
   88 
   89   let transport = mkStdioTransport hIn hOut ph
   90 
   91   -- Start a client session (spawns background receive thread).
   92   -- We manage the session manually rather than using withClientSession's
   93   -- bracket because the connection persists across slash commands.
   94   sessionMVar <- newEmptyMVar
   95   tidRef <- newIORef Nothing
   96   tid <- forkIO $
   97     withClientSession transport
   98       defaultClientConfig { config_request_timeout_us = 30_000_000 }
   99       (\session -> do
  100         putMVar sessionMVar (Right session)
  101         -- Block forever until the thread is killed
  102         newEmptyMVar >>= takeMVar :: IO ()
  103       )
  104     `catch` \(e :: SomeException) ->
  105       putMVar sessionMVar (Left e)
  106   writeIORef tidRef (Just tid)
  107 
  108   result <- takeMVar sessionMVar
  109   session <- case result of
  110     Left e  -> do
  111       stopProcess hIn hOut ph
  112       throwIO e
  113     Right s -> pure s
  114 
  115   -- Initialize the MCP protocol
  116   initResult <- initialize session
  117     `catch` \(e :: MCPClientError) -> do
  118       readIORef tidRef >>= mapM_ killThread
  119       stopProcess hIn hOut ph
  120       throwIO e
  121 
  122   let sInfo = Just (serverInfo initResult)
  123 
  124   -- Discover available tools
  125   ListToolsResult { tools = mcpTools } <- MCP.listTools session Nothing
  126 
  127   _lh_logInfo logger $ "MCP: " <> serverName <> " connected ("
  128     <> T.pack (show (length mcpTools)) <> " tools)"
  129 
  130   let cleanup = do
  131         readIORef tidRef >>= mapM_ killThread
  132         stopProcess hIn hOut ph
  133 
  134   pure McpServer
  135     { _ms_name       = serverName
  136     , _ms_command    = cmdArgs
  137     , _ms_session    = session
  138     , _ms_tools      = mcpTools
  139     , _ms_serverInfo = sInfo
  140     , _ms_cleanup    = cleanup
  141     }
  142 
  143 -- | Disconnect from an MCP server, cleaning up the child process.
  144 disconnectServer :: LogHandle -> McpServer -> IO ()
  145 disconnectServer logger server = do
  146   _lh_logInfo logger $ "MCP: disconnecting " <> _ms_name server
  147   _ms_cleanup server
  148     `catch` \(_ :: SomeException) -> pure ()

  150 -- | Get the names of all tools provided by an MCP server.
  151 mcpToolNames :: McpServer -> [Text]
  152 mcpToolNames = map toolName . _ms_tools
  153 
  154 -- | Build a 'ToolRegistry' from a list of connected MCP servers.
  155 --
  156 -- Each MCP tool is registered with its original name. If multiple
  157 -- servers expose tools with the same name, the last server wins.
  158 mcpRegistry :: [McpServer] -> ToolRegistry
  159 mcpRegistry = foldl addServer emptyRegistry
  160   where
  161     addServer :: ToolRegistry -> McpServer -> ToolRegistry
  162     addServer reg server =
  163       foldl (addTool server) reg (_ms_tools server)
  164 
  165     addTool :: McpServer -> ToolRegistry -> Tool -> ToolRegistry
  166     addTool server reg tool =
  167       let toolN = toolName tool
  168           def = ToolDefinition
  169             { _td_name        = toolN
  170             , _td_description = toolDesc tool
  171             , _td_inputSchema = Aeson.toJSON (inputSchema tool)
  172             }
  173           handler = mcpExecuteTool (_ms_session server) toolN
  174       in registerRichTool def handler reg

  190 -- | Execute an MCP tool call, returning PureClaw-compatible results.
  191 mcpExecuteTool :: ClientSession -> Text -> Value -> IO ([ToolResultPart], Bool)
  192 mcpExecuteTool session tName input = do
  193   let argsMap = keyMapToMap input
  194   result <- try @MCPClientError $ MCP.callTool session tName argsMap
  195   case result of
  196     Left e -> pure ([TRPText (T.pack (show e))], True)
  197     Right (CallToolResult { content = blocks, isError = mErr }) ->
  198       let isErr = fromMaybe False mErr
  199           parts = map convertContent blocks
  200       in pure (if null parts then [TRPText ""] else parts, isErr)
  201 
  202 convertContent :: ContentBlock -> ToolResultPart
  203 convertContent (TextBlock (TextContent { text = t })) = TRPText t
  204 convertContent (ImageBlock (ImageContent { data' = d, mimeType = m })) =
  205   TRPImage m (BS.pack (T.unpack d))
  206 convertContent _ = TRPText "[unsupported content type]"
  207 
  208 -- | Convert a JSON object to @Maybe (Map Text Value)@ for callTool.
  209 keyMapToMap :: Value -> Maybe (Map.Map Text Value)
  210 keyMapToMap (Object km) = Just $ Map.fromList
  211   [(Key.toText k, v) | (k, v) <- KM.toList km]
  212 keyMapToMap _ = Nothing

  214 -- ---------------------------------------------------------------------------
  215 -- Internal: stdio transport (manual lifecycle, not bracket-based)
  216 -- ---------------------------------------------------------------------------
  217 
  218 mkStdioTransport :: Handle -> Handle -> ProcessHandle -> Transport
  219 mkStdioTransport hIn hOut ph = Transport
  220   { transport_send = \msg -> do
  221       BSL.hPut hIn (Aeson.encode msg)
  222       BSL.hPut hIn "\n"
  223       hFlush hIn
  224   , transport_receive = do
  225       eof <- hIsEOF hOut
  226       if eof
  227         then pure Nothing
  228         else do
  229           line <- BS.hGetLine hOut
  230           case Aeson.eitherDecodeStrict' line of
  231             Right msg -> pure (Just msg)
  232             Left _    -> transport_receive (mkStdioTransport hIn hOut ph)
  233   , transport_close = stopProcess hIn hOut ph
  234   }
  235 
  236 stopProcess :: Handle -> Handle -> ProcessHandle -> IO ()
  237 stopProcess hIn hOut ph = do
  238   hClose hIn `catch` \(_ :: SomeException) -> pure ()
  239   exited <- timeout 2_000_000 (waitForProcess ph)
  240   case exited of
  241     Just _  -> hClose hOut `catch` \(_ :: SomeException) -> pure ()
  242     Nothing -> cleanupProcess (Nothing, Just hOut, Nothing, ph)
