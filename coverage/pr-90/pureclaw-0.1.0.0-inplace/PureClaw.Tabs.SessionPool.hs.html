    3 -- |
    4 -- Module      : PureClaw.Tabs.SessionPool
    5 -- Description : Refcounted SessionId -> SessionHandle resource pool.
    6 --
    7 -- Since tabs no longer own session handles, and §6.5 sharing means two
    8 -- cursors may resolve the same 'SessionId', there must be exactly __one__
    9 -- resolved 'SessionHandle' per live 'SessionId'. This module owns that
   10 -- mapping as a refcounted pool: it opens a handle when a session is first
   11 -- 'acquire'd and closes it when the last holder 'release's it
   12 -- (superseding the old one-handle-per-tab assumption and the
   13 -- @_env_session@ single-session global).
   14 --
   15 -- The pool is __additive__: it is built and unit-tested here in isolation
   16 -- (WU5) with injected open/close seams ('PoolDeps'), and wired into the
   17 -- agent loop / 'AgentEnv' later (WU8). No real session IO happens in this
   18 -- module — the opener and closer are supplied by the caller.
   19 --
   20 -- == Concurrency
   21 --
   22 -- The internal @'Data.Map.Strict.Map' 'SessionId' (Int, 'SessionHandle')@
   23 -- (refcount + handle) lives behind an 'IORef' mutated only with
   24 -- 'atomicModifyIORef''. Crucially, /no IO runs inside the atomic section/:
   25 -- opening and closing a handle are real side effects, so they are
   26 -- performed __outside__ 'atomicModifyIORef''. 'acquire' uses a
   27 -- check-then-open-then-commit pattern:
   28 --
   29 --   1. Atomically: if the id is already present, bump its refcount and
   30 --      return the existing handle (no open needed).
   31 --   2. Otherwise open a fresh handle (IO, outside the lock), then
   32 --      atomically commit it __only if the slot is still empty__. If a
   33 --      racing thread inserted first, discard (close) our just-opened
   34 --      handle and adopt theirs (bumping its refcount).
   35 --
   36 -- 'release' atomically decrements; when the count reaches zero it removes
   37 -- the entry and hands the now-orphaned handle back to be closed outside
   38 -- the atomic section. Releasing an absent or already-zero id is a safe
   39 -- no-op (nothing to close).
   40 --
   41 -- See the Tabs-as-View refactor (GitHub #79) §13 for the design context.
   42 module PureClaw.Tabs.SessionPool
   43   ( -- * Injected IO seams
   44     PoolDeps (..)
   45     -- * The pool
   46   , SessionPool (..)
   47   , newSessionPool
   48     -- * Acquire / release
   49   , acquire
   50   , release
   51   ) where
   52 
   53 import Data.IORef (IORef, atomicModifyIORef', newIORef)
   54 import Data.Map.Strict (Map)
   55 import Data.Map.Strict qualified as Map
   56 
   57 import PureClaw.Core.Types (SessionId)
   58 import PureClaw.Session.Handle (SessionHandle)
   59 
   60 -- | The injected session-IO seams. Real implementations open / close
   61 -- on-disk session handles; tests supply counting stubs so the pool's
   62 -- refcount machinery can be driven without touching the filesystem.
   63 data PoolDeps = PoolDeps
   64   { _pool_open  :: SessionId -> IO SessionHandle
   65     -- ^ Open (resolve) a fresh 'SessionHandle' for a 'SessionId'. Called
   66     -- at most once per live id — on the FIRST 'acquire'.
   67   , _pool_close :: SessionHandle -> IO ()
   68     -- ^ Close a 'SessionHandle'. Called exactly once per live id — on the
   69     -- LAST 'release' (refcount hits zero), and on the loser of an open
   70     -- race (its redundant handle is discarded).
   71   }
   72 
   73 -- | The refcounted pool: a map from each live 'SessionId' to its current
   74 -- refcount and the single shared 'SessionHandle'. An id is present in the
   75 -- map iff its refcount is @>= 1@; entries are removed when the count
   76 -- reaches zero, so absence is indistinguishable from "never acquired".
   77 newtype SessionPool = SessionPool (IORef (Map SessionId (Int, SessionHandle)))
   78 
   79 -- | Create a fresh, empty pool.
   80 newSessionPool :: IO SessionPool
   81 newSessionPool = SessionPool <$> newIORef Map.empty
   82 
   83 -- | Acquire the shared 'SessionHandle' for a 'SessionId', opening it on
   84 -- the first acquire and bumping the refcount on every subsequent one.
   85 -- Two acquires of the same id always return the SAME handle.
   86 --
   87 -- The open (a real side effect) is performed outside the atomic section:
   88 -- we first try to bump an existing entry atomically; only if the id is
   89 -- absent do we open a new handle and then atomically commit it, closing a
   90 -- redundant handle if a concurrent acquire won the race.
   91 acquire :: PoolDeps -> SessionPool -> SessionId -> IO SessionHandle
   92 acquire deps (SessionPool ref) sid = do
   93   -- Step 1: fast path — bump an existing entry's refcount atomically.
   94   mExisting <- atomicModifyIORef' ref $ \m ->
   95     case Map.lookup sid m of
   96       Just (n, h) -> (bump sid n h m, Just h)
   97       Nothing     -> (m, Nothing)
   98   case mExisting of
   99     Just h  -> pure h
  100     Nothing -> do
  101       -- Step 2: id absent — open OUTSIDE the atomic section.
  102       fresh <- _pool_open deps sid
  103       -- Step 3: commit only if still absent; otherwise adopt the winner's
  104       -- handle (bumping it) and close our now-redundant fresh one.
  105       (winner, redundant) <- atomicModifyIORef' ref $ \m ->
  106         case Map.lookup sid m of
  107           Just (n, h) -> (bump sid n h m, (h, Just fresh))
  108           Nothing     -> (bump sid 0 fresh m, (fresh, Nothing))
  109       mapM_ (_pool_close deps) redundant
  110       pure winner
  111 
  112 -- | Strictly store @sid -> (n + 1, h)@. The refcount and handle are forced
  113 -- (bang patterns) before insertion so the pool never accumulates a chain of
  114 -- refcount thunks and the stored handle is always in WHNF. Used both to add
  115 -- a brand-new entry (@n = 0@) and to increment an existing one.
  116 bump
  117   :: SessionId
  118   -> Int
  119   -> SessionHandle
  120   -> Map SessionId (Int, SessionHandle)
  121   -> Map SessionId (Int, SessionHandle)
  122 bump sid !n !h = Map.insert sid (n + 1, h)
  123 
  124 -- | Release one hold on a 'SessionId'. Decrements its refcount; when the
  125 -- count reaches zero the entry is removed and its handle closed. Releasing
  126 -- an id that is absent (never acquired, or already fully released) is a
  127 -- safe no-op: nothing is decremented and nothing is closed.
  128 --
  129 -- The close (a real side effect) is performed outside the atomic section.
  130 release :: PoolDeps -> SessionPool -> SessionId -> IO ()
  131 release deps (SessionPool ref) sid = do
  132   toClose <- atomicModifyIORef' ref $ \m ->
  133     case Map.lookup sid m of
  134       Nothing -> (m, Nothing)
  135       Just (n, h)
  136         | n <= 1    -> (Map.delete sid m, Just h)
  137         | otherwise -> (unbump sid n h m, Nothing)
  138   mapM_ (_pool_close deps) toClose
  139 
  140 -- | Strictly store @sid -> (n - 1, h)@ for the not-last release. The decremented
  141 -- refcount and handle are forced before insertion (same rationale as 'bump').
  142 -- Only called on the @n > 1@ path, so the result is always @>= 1@.
  143 unbump
  144   :: SessionId
  145   -> Int
  146   -> SessionHandle
  147   -> Map SessionId (Int, SessionHandle)
  148   -> Map SessionId (Int, SessionHandle)
  149 unbump sid !n !h = Map.insert sid (n - 1, h)
