   13 import Data.Text (Text)
   14 
   15 -- | Where the run that is attempting an adoption is being driven from.
   16 --
   17 -- Adoption is a trust-boundary expansion (PureClaw begins managing\/capturing
   18 -- an external, unmanaged tmux window). The consent is the human's explicit
   19 -- /pick/ of a window to take over:
   20 --
   21 --   * In the foreground TUI, picking a session in the New-Tab form IS the
   22 --     consent ('ConsentInteractive').
   23 --   * In the web UI served by the gateway, choosing \"Existing Harness\" and
   24 --     selecting a window in the browser IS the consent ('ConsentWeb'). The
   25 --     @POST \/api\/adopt@ request carries that selection (@consent_confirmed@).
   26 --
   27 -- Only a truly unattended invocation with NO human picking a window — a
   28 -- bot\/cron\/import\/daemon run, or a test default — maps to 'ConsentHeadless'
   29 -- and is denied (fail-closed). That headless-deny is the single, load-bearing
   30 -- remaining control on this gate.
   31 data ConsentChannel
   32   = ConsentInteractive  -- ^ foreground interactive TUI; a human picks the window
   33   | ConsentWeb          -- ^ gateway-served web UI; the browser selection is the consent
   34   | ConsentHeadless     -- ^ no human picking a window (bot\/cron\/import\/test); denied
   35   deriving stock (Show, Eq)
   36 
   37 -- | Why an adoption attempt was refused.
   38 --
   39 -- After the allow-list was dropped (the foreground session pick IS the
   40 -- consent), the ONLY reason the gate refuses is the absence of an interactive
   41 -- consent channel.
   42 data AdoptError
   43   = AdoptNoConsentChannel
   44     -- ^ the run has no interactive consent channel ('ConsentHeadless'); a
   45     -- headless\/gateway\/import\/cron run cannot adopt (fail-closed).
   46   deriving stock (Show, Eq)
   47 
   48 -- | A capability token proving that adopting a particular tmux session has
   49 -- passed the security gate.
   50 --
   51 -- The value constructor is intentionally NOT exported — the ONLY way to obtain
   52 -- an 'AdoptedHarness' is 'authorizeAdoption', which enforces the consent check.
   53 -- Downstream adoption mechanism code REQUIRES an 'AdoptedHarness' argument, so
   54 -- it is impossible by construction to adopt a window without first passing this
   55 -- gate. This mirrors the @AuthorizedCommand@ \/ @SafePath@ precedent in
   56 -- "PureClaw.Security.Command" and "PureClaw.Security.Path".
   57 newtype AdoptedHarness = AdoptedHarness { _unAdoptedHarness :: Text }
   58   deriving stock (Show, Eq)
   59 
   60 -- | The tmux session name that was authorized for adoption.
   61 adoptedSession :: AdoptedHarness -> Text
   62 adoptedSession = _unAdoptedHarness
   63 
   64 -- | The typed, consent-only adoption gate. Pure — no IO.
   65 --
   66 -- The allow-list was deliberately dropped (an explicit security-model
   67 -- relaxation): in the interactive path the user picking a session in the
   68 -- foreground New-Tab form IS the consent, so adoption is gated by consent
   69 -- alone:
   70 --
   71 --   * 'ConsentHeadless' → 'Left' 'AdoptNoConsentChannel' (a bot\/cron\/import\/
   72 --     test run has no human picking a window — fail-closed; the single
   73 --     remaining control).
   74 --   * 'ConsentInteractive' (foreground TUI) and 'ConsentWeb' (gateway web UI)
   75 --     → 'Right' an 'AdoptedHarness' token for ANY session. In both, a human
   76 --     explicitly picked the window to take over, which IS the consent.
   77 --
   78 -- The capability-token construction (unexported ctor) and the downstream adopt
   79 -- mechanism's own identifier hygiene (@send-keys -l --@ + @validateTmuxIdent@,
   80 -- §8 C3) remain unchanged.
   81 authorizeAdoption
   82   :: ConsentChannel
   83   -> Text                -- ^ tmux session name to adopt
   84   -> Either AdoptError AdoptedHarness
   85 authorizeAdoption consent session =
   86   case consent of
   87     ConsentHeadless    -> Left AdoptNoConsentChannel
   88     ConsentInteractive -> Right (AdoptedHarness session)
   89     ConsentWeb         -> Right (AdoptedHarness session)
