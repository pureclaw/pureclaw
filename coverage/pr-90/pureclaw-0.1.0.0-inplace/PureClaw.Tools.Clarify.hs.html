   19 -- | Create a clarify tool that lets the agent ask the user structured questions.
   20 -- Supports open-ended questions and multiple choice (up to 4 options).
   21 clarifyTool :: ChannelHandle -> (ToolDefinition, ToolHandler)
   22 clarifyTool ch = (def, handler)
   23   where
   24     def = ToolDefinition
   25       { _td_name        = "clarify"
   26       , _td_description = T.unlines
   27           [ "Ask the user a clarifying question. Use when the request is ambiguous"
   28           , "or you need more information before proceeding."
   29           , "Supports open-ended questions (just question) or multiple choice"
   30           , "(question + choices array, max 4 options)."
   31           ]
   32       , _td_inputSchema = object
   33           [ "type" .= ("object" :: Text)
   34           , "properties" .= object
   35               [ "question" .= object
   36                   [ "type" .= ("string" :: Text)
   37                   , "description" .= ("The question to ask the user" :: Text)
   38                   ]
   39               , "choices" .= object
   40                   [ "type" .= ("array" :: Text)
   41                   , "items" .= object [ "type" .= ("string" :: Text) ]
   42                   , "maxItems" .= (4 :: Int)
   43                   , "description" .= ("Optional: up to 4 multiple-choice options" :: Text)
   44                   ]
   45               ]
   46           , "required" .= (["question"] :: [Text])
   47           ]
   48       }
   49 
   50     handler = ToolHandler $ \input ->
   51       case parseEither parseClarifyInput input of
   52         Left err -> pure (T.pack err, True)
   53         Right ci -> askUser ci
   54 
   55     askUser :: ClarifyInput -> IO (Text, Bool)
   56     askUser ci = do
   57       let prompt = formatPrompt (_ci_question ci) (_ci_choices ci)
   58       result <- try @SomeException (_ch_prompt ch prompt)
   59       case result of
   60         Left e -> pure ("Failed to get user response: " <> T.pack (show e), True)
   61         Right response -> do
   62           let resolved = resolveChoice response (_ci_choices ci)
   63           pure (encodeResponse (_ci_question ci) (_ci_choices ci) resolved, False)
   64 
   65     formatPrompt :: Text -> Maybe [Text] -> Text
   66     formatPrompt question Nothing = question <> "\n> "
   67     formatPrompt question (Just choices) =
   68       let numbered = zipWith (\i c -> T.pack (show (i :: Int)) <> ". " <> c) [1..] choices
   69       in question <> "\n" <> T.unlines numbered <> T.pack (show (length choices + 1)) <> ". Other\n> "
   70 
   71     resolveChoice :: Text -> Maybe [Text] -> Text
   72     resolveChoice response Nothing = response
   73     resolveChoice response (Just choices) =
   74       case reads (T.unpack (T.strip response)) :: [(Int, String)] of
   75         [(n, "")] | n >= 1 && n <= length choices -> choices !! (n - 1)
   76                   | n == length choices + 1        -> response  -- "Other" selected
   77                   | otherwise                      -> response  -- out of range, treat as freeform
   78         _                                          -> response  -- non-numeric, treat as freeform
   79 
   80     encodeResponse :: Text -> Maybe [Text] -> Text -> Text
   81     encodeResponse question choices answer =
   82       let result = object $
   83             [ "question" .= question
   84             , "user_response" .= answer
   85             ] <> maybe [] (\cs -> ["choices_offered" .= cs]) choices
   86       in case encode result of
   87         bs -> TE.decodeUtf8Lenient (toStrictBS bs)
   88 
   89     toStrictBS :: BL.ByteString -> BS.ByteString
   90     toStrictBS = BL.toStrict
   91 
   92 data ClarifyInput = ClarifyInput
   93   { _ci_question :: Text
   94   , _ci_choices  :: Maybe [Text]
   95   }
   96 
   97 parseClarifyInput :: Value -> Parser ClarifyInput
   98 parseClarifyInput = withObject "ClarifyInput" $ \o ->
   99   ClarifyInput
  100     <$> o .:  "question"
  101     <*> o .:? "choices"
