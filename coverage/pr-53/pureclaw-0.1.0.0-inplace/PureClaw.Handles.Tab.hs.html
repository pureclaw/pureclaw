    1 -- |
    2 -- Module      : PureClaw.Handles.Tab
    3 -- Description : Tabbed Chat type layer (WU1) — user-facing primitive: a tab.
    4 --
    5 -- A 'TabHandle' is the universal Handle-pattern carrier for one tab in
    6 -- the Tabbed Chat surface. Tabs come in several 'TabKind's
    7 -- ('KindAi' for an AI chat tab, 'KindHarness' for a managed harness,
    8 -- and 'KindShell' \/ 'KindSsh' \/ 'KindTmux' for terminal-backend tabs).
    9 -- The dispatcher routes user input to the focused tab and per-tab
   10 -- output to the channel (focus-gated).
   11 --
   12 -- == Scope of WU1
   13 --
   14 -- This module currently lands the type layer only: 'TabHandle' record
   15 -- shape, smart constructors, error ADTs (with the manually-written
   16 -- redacted 'Show' instance for 'TabError' required by H14), and the
   17 -- per-kind factory signatures. Factory bodies are stubbed with
   18 -- @'error' \"not implemented\"@ — the real bodies land in WU6 ('Tab.Ai'),
   19 -- WU7 ('Tab.Harness'), and WU8 ('Tab.Backend').
   20 --
   21 -- == Field-naming convention
   22 --
   23 -- The handle record uses the @_tabHandle_*@ field-name prefix (NOT the
   24 -- terser @_th_*@ used in some older draft notes). This was decided
   25 -- during WU0 planning to keep the prefix self-describing when the field
   26 -- name is grepped in isolation.
   27 --
   28 -- See @docs\/tabbed-chat.md@ for the full design (sections
   29 -- \"TabHandle abstraction (H-series)\" and \"Channel output\").

   72 -- ---------------------------------------------------------------------------
   73 -- TabIndex
   74 -- ---------------------------------------------------------------------------
   75 
   76 -- | Validated tab index. The data constructor is intentionally NOT
   77 -- exported; obtain values via 'mkTabIndex' (which bounds-checks against
   78 -- the configured @_rc_maxTabs@ ceiling) or by pattern-matching on the
   79 -- newtype via 'unTabIndex'.
   80 --
   81 -- The underlying 'Int' is non-negative on every value produced by
   82 -- 'mkTabIndex' (@0 <= n@). Bounds-checking against @_rc_maxTabs@ is
   83 -- performed at the parser layer (WU2) using the 'RoutingConfig' in
   84 -- scope; the smart constructor here enforces the floor only.
   85 newtype TabIndex = TabIndex { unTabIndex :: Int }
   86   deriving stock (Eq, Ord, Show)
   87 
   88 -- | The only way to construct a 'TabIndex'. Returns 'Nothing' for
   89 -- negative inputs.
   90 mkTabIndex :: Int -> Maybe TabIndex
   91 mkTabIndex n
   92   | n < 0     = Nothing
   93   | otherwise = Just (TabIndex n)

  132 -- ---------------------------------------------------------------------------
  133 -- TabStatus
  134 -- ---------------------------------------------------------------------------
  135 
  136 -- | Runtime status of a tab.
  137 --
  138 -- * 'Active' — a provider call or tool execution is in flight, OR a
  139 --   backend recv is waiting for output. The dashboard does not
  140 --   distinguish these sub-cases.
  141 -- * 'Idle' — the tab is waiting for input; carries the timestamp of
  142 --   the last observed input (for sort / display).
  143 -- * 'Crashed' — the tab loop hit a synchronous exception and exited;
  144 --   the wrapped 'PublicTabError' is the channel-safe summary. The
  145 --   registry entry persists until @\/tab close N@ or
  146 --   @\/tab resume \<session\>@.
  147 data TabStatus
  148   = Active
  149   | Idle !UTCTime
  150   | Crashed !PublicTabError
  151   deriving stock (Eq, Show)
  152

  154 -- ---------------------------------------------------------------------------
  155 -- CloseMode
  156 -- ---------------------------------------------------------------------------
  157 
  158 -- | Close semantics requested by the caller of '_tabHandle_close'.
  159 --
  160 -- * 'CloseGraceful' — kind-specific graceful close: for 'KindAi'
  161 --   archives the session via @_sh_save@; for non-AI kinds runs the
  162 --   underlying destructive close (e.g. @_bh_close@, @_hh_stop@).
  163 -- * 'CloseForce' — for 'KindAi' skips the archive (transcript deleted
  164 --   from disk); for non-AI kinds is a no-op distinct from
  165 --   'CloseGraceful' because close is already destructive there.
  166 data CloseMode
  167   = CloseGraceful
  168   | CloseForce
  169   deriving stock (Eq, Show)

  172 -- ---------------------------------------------------------------------------
  173 -- TabRunner
  174 -- ---------------------------------------------------------------------------
  175 
  176 -- | The cancel + wait pair returned by the @_env_fork@ test seam.
  177 --
  178 -- Cannot be an 'Control.Concurrent.Async.Async' directly because:
  179 -- (a) the synchronous test variant runs the body inline and has no
  180 -- live async to wrap; (b) the IORef-Bool cancel-observability primitive
  181 -- used by tests needs a uniform shape across sync and async variants.
  182 --
  183 -- See the @TabRunner bootstrapping during spawn@ section of the design
  184 -- doc and the WU0 scaffold in @Test.Fake.TabFactory@.
  185 data TabRunner = TabRunner
  186   { _trun_cancel :: IO ()
  187     -- ^ Idempotent cancel. Safe to call multiple times; safe to call
  188     --   on a runner whose body has already exited.
  189   , _trun_wait   :: IO ()
  190     -- ^ Block until the tab loop's body exits. In the synchronous
  191     --   test variant returns immediately because the body has
  192     --   already run inline.
  193   }

  196 -- ---------------------------------------------------------------------------
  197 -- TabHandle
  198 -- ---------------------------------------------------------------------------
  199 
  200 -- | Universal Handle-pattern carrier for one tab.
  201 --
  202 -- Pure fields ('_tabHandle_index', '_tabHandle_name', '_tabHandle_kind')
  203 -- are bound at construction time. IO-action fields are kind-specific:
  204 -- 'KindAi' enqueues into a per-tab @TBQueue InputEvent@; non-AI kinds
  205 -- write to a 'PureClaw.Handles.Backend.BackendHandle' or
  206 -- 'PureClaw.Handles.Harness.HarnessHandle'.
  207 --
  208 -- '_tabHandle_close' is idempotent and never throws (parallel to the
  209 -- @_bh_close@ contract on 'PureClaw.Handles.Backend.BackendHandle').
  210 data TabHandle = TabHandle
  211   { _tabHandle_index         :: !TabIndex
  212     -- ^ Pure: tab's slot in the registry (H12-adjacent).
  213   , _tabHandle_name          :: !TabName
  214     -- ^ Pure: redacted friendly label (H11). Constructed via
  215     --   @sanitizeTabName@ (WU2) at the factory site.
  216   , _tabHandle_kind          :: !TabKind
  217     -- ^ Pure: H12. Read without IO.
  218   , _tabHandle_status        :: IO TabStatus
  219     -- ^ H5: returns 'Active' \/ 'Idle' \/ 'Crashed'.
  220   , _tabHandle_send          :: Text -> IO (Either TabError ())
  221     -- ^ H4: enqueue a 'UserText' input event. Bounded by
  222     --   @_rc_inputQueueBound@; overflow returns
  223     --   'Left' (e.g. 'TabConcurrencyLimit') so the dispatcher never
  224     --   blocks.
  225   , _tabHandle_enqueueSlash  :: Slash.SlashCommand -> IO (Either TabError ())
  226     -- ^ H13: enqueue a 'SlashCmd' input event. For 'KindAi' the tab
  227     --   loop runs @executeSlashCommand@ against its own per-tab
  228     --   context; for non-AI kinds the implementation returns
  229     --   @Left ('TabUnsupportedCommand' cmd)@ immediately without
  230     --   enqueueing.
  231   , _tabHandle_close         :: CloseMode -> IO ()
  232     -- ^ H6 \/ H7 \/ H8 \/ H9 \/ H10: idempotent, never throws,
  233     --   kind-specific semantics. The 'CloseMode' selects graceful
  234     --   vs forced archive behaviour for 'KindAi'.
  235   }

  277 -- ---------------------------------------------------------------------------
  278 -- TabError
  279 -- ---------------------------------------------------------------------------
  280 
  281 -- | Construction-time and runtime errors on a tab.
  282 --
  283 -- Every constructor's payload is itself a redaction-safe ADT or a
  284 -- bounded primitive — no constructor carries a free 'Data.Text.Text',
  285 -- a raw 'Control.Exception.SomeException' message, or a raw filesystem
  286 -- path. The hand-written 'Show' below (H14) elides every payload value
  287 -- so accidental @show@ of a 'TabError' in a log line cannot leak
  288 -- hostnames, paths, or stderr fragments.
  289 --
  290 -- The exhaustive constructor list is enforced by H3.
  291 data TabError
  292   = TabIndexInUse !TabIndex
  293   | TabIndexOutOfRange !Int
  294   | TabLimitExceeded !Int
  295   | TabBackendConstructFailed !Backend.BackendError
  296   | TabSessionCreateFailed !SessionError
  297   | TabSpawnAuthDenied !PublicAuthError
  298   | TabNotFound !Int
  299   | TabConcurrencyLimit !Int
  300   | TabInvalidName !NameError
  301   | TabUnsupportedCommand !Slash.SlashCommand
  302   deriving stock (Eq)
  303 
  304 -- | Hand-written 'Show' for 'TabError' per H14: constructor names only,
  305 -- payload values elided. Adding a new constructor here requires adding
  306 -- a new branch in this 'Show' instance (H3 enumerates the exhaustive
  307 -- set), so accidentally regressing to @deriving Show@ would surface as
  308 -- a build error from the @-Wincomplete-patterns@ flag.
  309 instance Show TabError where
  310   show e = case e of
  311     TabIndexInUse{}             -> "TabIndexInUse"
  312     TabIndexOutOfRange{}        -> "TabIndexOutOfRange"
  313     TabLimitExceeded{}          -> "TabLimitExceeded"
  314     TabBackendConstructFailed{} -> "TabBackendConstructFailed"
  315     TabSessionCreateFailed{}    -> "TabSessionCreateFailed"
  316     TabSpawnAuthDenied{}        -> "TabSpawnAuthDenied"
  317     TabNotFound{}               -> "TabNotFound"
  318     TabConcurrencyLimit{}       -> "TabConcurrencyLimit"
  319     TabInvalidName{}            -> "TabInvalidName"
  320     TabUnsupportedCommand{}     -> "TabUnsupportedCommand"
  321

  323 -- ---------------------------------------------------------------------------
  324 -- PublicTabError
  325 -- ---------------------------------------------------------------------------
  326 
  327 -- | Channel-safe projection of a 'TabError'.
  328 --
  329 -- The wrapped 'Data.Text.Text' is short, fixed-vocabulary, and
  330 -- contains no hostnames, paths, or secrets. This is what reaches the
  331 -- user via 'PureClaw.Handles.Channel.ChannelHandle' when a tab
  332 -- operation fails.
  333 newtype PublicTabError = PublicTabError { unPublicTabError :: Text }
  334   deriving stock (Eq, Show)
  335 
  336 -- | Project a 'TabError' to its user-visible form.
  337 --
  338 -- The mapping is intentionally coarse: similar failure modes collapse
  339 -- to the same short label so a future audit of channel-bound strings
  340 -- has a small surface to grep.
  341 toPublicTabError :: TabError -> PublicTabError
  342 toPublicTabError e = PublicTabError $ case e of
  343   TabIndexInUse{}             -> "tab: index already in use"
  344   TabIndexOutOfRange{}        -> "tab: index out of range"
  345   TabLimitExceeded{}          -> "tab: maximum tab count reached"
  346   TabBackendConstructFailed{} -> "tab: backend construction failed"
  347   TabSessionCreateFailed{}    -> "tab: session creation failed"
  348   TabSpawnAuthDenied{}        -> "tab: spawn authorization denied"
  349   TabNotFound{}               -> "tab: not found"
  350   TabConcurrencyLimit{}       -> "tab: input queue full"
  351   TabInvalidName{}            -> "tab: invalid name"
  352   TabUnsupportedCommand{}     -> "tab: command not supported on this tab kind"
