   21 -- | Run a process result through the same output-shaping logic both tools
   22 -- share: stdout + stderr + an "Exit code: N" suffix on failure, with a
   23 -- placeholder for empty output. Returns (text, isError).
   24 shapeResult :: ProcessResult -> (Text, Bool)
   25 shapeResult pr =
   26   let out      = T.pack (BS8.unpack (_pr_stdout pr))
   27       err      = T.pack (BS8.unpack (_pr_stderr pr))
   28       exitInfo = case _pr_exitCode pr of
   29         ExitSuccess     -> ""
   30         ExitFailure n   -> "\nExit code: " <> T.pack (show n)
   31       combined = T.strip (out <> err <> exitInfo)
   32   in  (if T.null combined then "(no output)" else combined, False)

   34 runAuthorized :: ShellHandle -> ExecOptions -> AuthorizedCommand -> IO (Text, Bool)
   35 runAuthorized sh opts cmd = do
   36   result <- try @SomeException (_sh_execute sh opts cmd)
   37   case result of
   38     Left e   -> pure (T.pack (show e), True)
   39     Right pr -> pure (shapeResult pr)

   41 -- | Bash-backed shell tool. The @command@ string is handed verbatim to
   42 -- @bash -c@, so the LLM gets full shell semantics: tilde expansion, glob
   43 -- patterns, env-var substitution, pipes, @&&@, redirects, etc.
   44 --
   45 -- Gated by the policy having @shell@ in its allowed-command set
   46 -- (see 'authorizeShell'). Per-basename allowlisting is the @exec@ tool's
   47 -- job — once bash is in the loop, basename checks are not meaningful.
   48 shellTool :: SecurityPolicy -> ShellHandle -> (ToolDefinition, ToolHandler)
   49 shellTool policy sh = (def, handler)
   50   where
   51     def = ToolDefinition
   52       { _td_name        = "shell"
   53       , _td_description = T.unlines
   54           [ "Run a shell command. The string is executed by /bin/bash -c, so"
   55           , "shell features work: pipes (|), command chaining (&&, ||, ;),"
   56           , "redirects (>, <), tilde expansion (~), glob patterns (*, ?),"
   57           , "and environment variable substitution ($VAR)."
   58           , "Optional: set timeout_ms to limit execution time (returns exit code 124 on timeout)."
   59           , "Optional: set working_dir to run in a specific directory."
   60           , "Returns combined stdout+stderr. Non-zero exit appends 'Exit code: N'."
   61           ]
   62       , _td_inputSchema = object
   63           [ "type" .= ("object" :: Text)
   64           , "properties" .= object
   65               [ "command" .= object
   66                   [ "type" .= ("string" :: Text)
   67                   , "description" .= ("The shell command line. Passed verbatim to bash -c." :: Text)
   68                   ]
   69               , "timeout_ms" .= object
   70                   [ "type" .= ("integer" :: Text)
   71                   , "description" .= ("Timeout in milliseconds. Command is killed after this time (default: no timeout)" :: Text)
   72                   ]
   73               , "working_dir" .= object
   74                   [ "type" .= ("string" :: Text)
   75                   , "description" .= ("Working directory for the command (default: agent workspace)" :: Text)
   76                   ]
   77               ]
   78           , "required" .= (["command"] :: [Text])
   79           ]
   80       }
   81 
   82     handler = ToolHandler $ \input ->
   83       case parseEither parseShellInput input of
   84         Left err -> pure (T.pack err, True)
   85         Right si
   86           | T.null (T.strip (_si_command si)) ->
   87               pure ("Empty command", True)
   88           | otherwise ->
   89               case authorizeShell policy (_si_command si) of
   90                 Left (CommandNotAllowed _) ->
   91                   pure ("shell tool not enabled in security policy (add \"shell\" to allowed_commands, or use the exec tool)", True)
   92                 Left CommandInAutonomyDeny ->
   93                   pure ("All commands denied by security policy", True)
   94                 Right authorized ->
   95                   runAuthorized sh (shellExecOpts si) authorized

   97 -- | Argv-style exec tool. Takes an explicit @program@ name and @args@
   98 -- list — no shell interpretation. Every argument reaches the program as
   99 -- a literal string: @~@ is not expanded, @*@ is not globbed, @|@ is not a
  100 -- pipe.
  101 --
  102 -- Authorized against the policy's per-basename allowlist (see 'authorize').
  103 -- This is the tool to expose when a deployment wants to lock the agent
  104 -- down to an explicit set of executables.
  105 execTool :: SecurityPolicy -> ShellHandle -> (ToolDefinition, ToolHandler)
  106 execTool policy sh = (def, handler)
  107   where
  108     def = ToolDefinition
  109       { _td_name        = "exec"
  110       , _td_description = T.unlines
  111           [ "Execute a program with explicit arguments. No shell is involved:"
  112           , "tilde (~), glob (*), env vars ($VAR), pipes (|), redirects (>),"
  113           , "and chaining (&&) are NOT interpreted — every arg is a literal."
  114           , "Use this for tighter security: the policy allowlist gates each"
  115           , "program by name."
  116           , "Optional: set timeout_ms to limit execution time (returns exit code 124 on timeout)."
  117           , "Optional: set working_dir to run in a specific directory."
  118           , "Returns combined stdout+stderr. Non-zero exit appends 'Exit code: N'."
  119           ]
  120       , _td_inputSchema = object
  121           [ "type" .= ("object" :: Text)
  122           , "properties" .= object
  123               [ "program" .= object
  124                   [ "type" .= ("string" :: Text)
  125                   , "description" .= ("Program to execute (e.g. \"ls\", \"git\"). Looked up via PATH." :: Text)
  126                   ]
  127               , "args" .= object
  128                   [ "type"        .= ("array" :: Text)
  129                   , "items"       .= object [ "type" .= ("string" :: Text) ]
  130                   , "description" .= ("Arguments as literal strings. No shell expansion." :: Text)
  131                   ]
  132               , "timeout_ms" .= object
  133                   [ "type" .= ("integer" :: Text)
  134                   , "description" .= ("Timeout in milliseconds. Command is killed after this time (default: no timeout)" :: Text)
  135                   ]
  136               , "working_dir" .= object
  137                   [ "type" .= ("string" :: Text)
  138                   , "description" .= ("Working directory for the command (default: agent workspace)" :: Text)
  139                   ]
  140               ]
  141           , "required" .= (["program"] :: [Text])
  142           ]
  143       }
  144 
  145     handler = ToolHandler $ \input ->
  146       case parseEither parseExecInput input of
  147         Left err -> pure (T.pack err, True)
  148         Right ei
  149           | T.null (T.strip (_ei_program ei)) ->
  150               pure ("Empty program", True)
  151           | otherwise ->
  152               case authorize policy (T.unpack (_ei_program ei)) (_ei_args ei) of
  153                 Left (CommandNotAllowed c) ->
  154                   pure ("Command not allowed: " <> c, True)
  155                 Left CommandInAutonomyDeny ->
  156                   pure ("All commands denied by security policy", True)
  157                 Right authorized ->
  158                   runAuthorized sh (execExecOpts ei) authorized

  160 -- ── Input parsing ─────────────────────────────────────────────────────

  162 data ShellInput = ShellInput
  163   { _si_command    :: Text
  164   , _si_timeout    :: Maybe Int
  165   , _si_workingDir :: Maybe Text
  166   }
  167 
  168 parseShellInput :: Value -> Parser ShellInput
  169 parseShellInput = withObject "ShellInput" $ \o ->
  170   ShellInput
  171     <$> o .:  "command"
  172     <*> o .:? "timeout_ms"
  173     <*> o .:? "working_dir"
  174 
  175 shellExecOpts :: ShellInput -> ExecOptions
  176 shellExecOpts si = ExecOptions
  177   { _eo_timeout    = _si_timeout si
  178   , _eo_workingDir = fmap T.unpack (_si_workingDir si)
  179   }
  180 
  181 data ExecInput = ExecInput
  182   { _ei_program    :: Text
  183   , _ei_args       :: [Text]
  184   , _ei_timeout    :: Maybe Int
  185   , _ei_workingDir :: Maybe Text
  186   }
  187 
  188 parseExecInput :: Value -> Parser ExecInput
  189 parseExecInput = withObject "ExecInput" $ \o ->
  190   ExecInput
  191     <$> o .:  "program"
  192     <*> o .:? "args" .!= []
  193     <*> o .:? "timeout_ms"
  194     <*> o .:? "working_dir"
  195 
  196 execExecOpts :: ExecInput -> ExecOptions
  197 execExecOpts ei = ExecOptions
  198   { _eo_timeout    = _ei_timeout ei
  199   , _eo_workingDir = fmap T.unpack (_ei_workingDir ei)
  200   }
