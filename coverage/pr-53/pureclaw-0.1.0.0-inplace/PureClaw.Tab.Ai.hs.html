    1 -- |
    2 -- Module      : PureClaw.Tab.Ai
    3 -- Description : AI tab factory + per-tab loop (Tabbed Chat WU6).
    4 --
    5 -- An AI tab is a 'PureClaw.Handles.Tab.TabHandle' whose loop body
    6 -- consumes a per-tab @TBQueue InputEvent@ and either feeds plain text
    7 -- to the configured 'PureClaw.Providers.Class.Provider' or runs the
    8 -- existing slash-command handler against a /per-tab/ 'Context'. The
    9 -- single-writer rule (E5) is structural: the loop is the only mutator
   10 -- of @_ats_context@, and every other producer (the dispatcher,
   11 -- '_tabHandle_send', '_tabHandle_enqueueSlash') goes through the
   12 -- bounded input queue.
   13 --
   14 -- == Factory shape
   15 --
   16 -- @
   17 -- 'mkTabAi' :: 'PureClaw.Agent.Env.AgentEnv'
   18 --           -> 'PureClaw.Handles.Tab.TabIndex'
   19 --           -> 'PureClaw.Handles.Tab.AiSpawnArgs'
   20 --           -> IO (Either 'PureClaw.Handles.Tab.TabError'
   21 --                         'PureClaw.Handles.Tab.TabHandle')
   22 -- @
   23 --
   24 -- Note: this signature takes 'AgentEnv', which is wider than the
   25 -- 2-arg 'PureClaw.Handles.Tab.mkTabAi' stub in 'Handles.Tab'. The
   26 -- stub remains in place (the dispatcher's @defaultTabFactory@ still
   27 -- references it) because the H2 type-shape pin at @Handles/TabSpec.hs@
   28 -- locks the 2-arg signature; the production wiring that calls this
   29 -- 3-arg factory lands in WU9 (auto-spawn UX) alongside the rest of
   30 -- the user-facing tab-creation surface.
   31 --
   32 -- == Async exception discipline
   33 --
   34 -- The loop body catches 'SomeException' EXCEPT 'AsyncCancelled', which
   35 -- propagates so that '_tabHandle_close' (which cancels the loop's
   36 -- 'TabRunner') can unblock a provider call inside 'bracket'. The
   37 -- 'safelyRunLoop' helper re-raises 'AsyncCancelled' before swallowing
   38 -- any other synchronous failure into a 'Crashed' status.
   39 --
   40 -- == LLM-free invariant (P18 \/ I3)
   41 --
   42 -- 'UserText' starting with @\/@ is re-parsed via
   43 -- 'PureClaw.Routing.Parse.parseSlashCommand' (I2). On a hit the loop
   44 -- treats it as a 'SlashCmd' and never reaches the provider. Only
   45 -- non-slash 'UserText' or unparseable slash-input falls through to a
   46 -- provider call.
   47 --
   48 -- See @docs/tabbed-chat.md@ §"AI tab loop body" (I-series) and
   49 -- §"H-series" (close lifecycle).

  134 -- ---------------------------------------------------------------------------
  135 -- AiTabState — per-tab mutable state, private to the factory closure
  136 -- ---------------------------------------------------------------------------
  137 
  138 -- | Per-AI-tab mutable state. The fields are exposed so a future
  139 -- diagnostic 'tab info' handler can inspect them; production code
  140 -- never touches the record outside this module.
  141 --
  142 -- /Single-writer invariant (E5):/ '_ats_context' is mutated only by
  143 -- the per-tab loop. Other producers (dispatcher, channel input, slash
  144 -- enqueue) deposit 'InputEvent's on '_ats_inputQ' and the loop runs
  145 -- the resulting state mutation @readIORef → process → writeIORef@
  146 -- inline.
  147 data AiTabState = AiTabState
  148   { _ats_inputQ    :: !(TBQueue InputEvent)
  149     -- ^ Per-tab input queue. Bounded by '_rc_inputQueueBound' so an
  150     -- overflowing producer surfaces a 'TabConcurrencyLimit' rather
  151     -- than blocking the dispatcher (H4).
  152   , _ats_provider  :: !(IORef (Maybe SomeProvider))
  153     -- ^ Per-tab provider. Mutable so a focused @\/provider@ command
  154     -- (K6.1) can hot-swap it. Initialised from '_env_provider' on
  155     -- spawn.
  156   , _ats_model     :: !(IORef (Maybe ModelId))
  157     -- ^ Per-tab model. Mutable so a focused @\/model@ command (K6.2)
  158     -- can hot-swap it. Initialised from '_env_model' on spawn.
  159   , _ats_target    :: !(IORef MessageTarget)
  160     -- ^ Per-tab message target. Mutable so a focused @\/target@
  161     -- command (K4) can hot-swap it. Initialised from '_env_target'
  162     -- on spawn.
  163   , _ats_context   :: !(IORef Ctx.Context)
  164     -- ^ Per-tab conversation context. Single-writer (the loop).
  165   , _ats_runner    :: !(IORef (Maybe TabRunner))
  166     -- ^ Filled by 'mkTabAi' after '_env_fork' returns. The close
  167     -- handler reads this to cancel the loop. Stays 'Nothing' for the
  168     -- brief window between allocation and fork-fill, during which
  169     -- 'closeAllTabs' tolerates the gap (no-op cancel).
  170   , _ats_closed    :: !(IORef Bool)
  171     -- ^ Idempotency flag for '_tabHandle_close'. Flipped to 'True'
  172     -- on the first invocation; subsequent invocations are no-ops
  173     -- (H6).
  174   , _ats_statusRef :: !(IORef TabStatus)
  175     -- ^ Held status backing '_tabHandle_status'. The loop transitions
  176     -- this between 'Active' (during provider call) and 'Idle'
  177     -- (between events); 'Crashed' is set by the loop's outer
  178     -- exception handler.
  179   , _ats_streamCtr :: !(IORef Word64)
  180     -- ^ Per-tab monotonically-increasing counter for 'StreamId'
  181     -- allocation. Wrap-around is not a practical concern (2^64 ≈
  182     -- 18 quintillion streams).
  183   }

  186 -- ---------------------------------------------------------------------------
  187 -- mkTabAi — the AI tab factory
  188 -- ---------------------------------------------------------------------------
  189 
  190 -- | Construct an AI tab. Allocates the per-tab state, forks the loop
  191 -- via '_env_fork', and returns a 'TabHandle' whose IO actions close
  192 -- over the state.
  193 --
  194 -- The factory itself never throws: it always returns 'Right h' or
  195 -- 'Left e'. Failures inside the forked loop are caught by the loop's
  196 -- own exception handler and surface as a 'Crashed' status (visible via
  197 -- '_tabHandle_status').
  198 mkTabAi :: AgentEnv -> TabIndex -> AiSpawnArgs -> IO (Either TabError TabHandle)
  199 mkTabAi env idx args =
  200   case Parse.sanitizeTabName (_ai_requestedName args) of
  201     Left nameErr -> pure (Left (TabInvalidName nameErr))
  202     Right nameTxt -> do
  203       let rc = _env_routingConfig env
  204       state <- allocState env rc
  205       now <- getCurrentTime
  206       writeIORef (_ats_statusRef state) (Idle now)
  207       -- Fork the loop. The fork seam returns a 'TabRunner' that the
  208       -- close handler will later cancel.
  209       runner <- _env_fork env (loopBody env idx state)
  210       writeIORef (_ats_runner state) (Just runner)
  211       pure (Right (mkHandle env idx (TabName nameTxt) state))
  212 
  213 -- | Allocate the per-tab state. Initialises every IORef from
  214 -- 'AgentEnv' so the per-tab provider\/model\/target track the
  215 -- process-level defaults at spawn time and can drift independently
  216 -- afterwards.
  217 allocState :: AgentEnv -> RoutingConfig -> IO AiTabState
  218 allocState env rc = do
  219   inputQ       <- newTBQueueIO (fromIntegral (_rc_inputQueueBound rc))
  220   curProv      <- readIORef (_env_provider env)
  221   provRef      <- newIORef curProv
  222   curMod       <- readIORef (_env_model env)
  223   modRef       <- newIORef curMod
  224   curTgt       <- readIORef (_env_target env)
  225   tgtRef       <- newIORef curTgt
  226   ctxRef       <- newIORef (Ctx.emptyContext (_env_systemPrompt env))
  227   runRef       <- newIORef Nothing
  228   closedRef    <- newIORef False
  229   -- Status starts 'Active' as a sentinel; 'mkTabAi' replaces it with
  230   -- 'Idle now' before returning.
  231   statRef      <- newIORef Active
  232   streamCtrRef <- newIORef 0
  233   pure AiTabState
  234     { _ats_inputQ    = inputQ
  235     , _ats_provider  = provRef
  236     , _ats_model     = modRef
  237     , _ats_target    = tgtRef
  238     , _ats_context   = ctxRef
  239     , _ats_runner    = runRef
  240     , _ats_closed    = closedRef
  241     , _ats_statusRef = statRef
  242     , _ats_streamCtr = streamCtrRef
  243     }
  244 
  245 -- | Build the public 'TabHandle' record from the per-tab state.
  246 mkHandle :: AgentEnv -> TabIndex -> TabName -> AiTabState -> TabHandle
  247 mkHandle env idx name state = TabHandle
  248   { _tabHandle_index        = idx
  249   , _tabHandle_name         = name
  250   , _tabHandle_kind         = KindAi
  251   , _tabHandle_status       = readIORef (_ats_statusRef state)
  252   , _tabHandle_send         = sendUserText state
  253   , _tabHandle_enqueueSlash = enqueueSlashAi state
  254   , _tabHandle_close        = closeTabAi env state
  255   }

  258 -- ---------------------------------------------------------------------------
  259 -- _tabHandle_send / _tabHandle_enqueueSlash — non-blocking enqueue
  260 -- ---------------------------------------------------------------------------
  261 
  262 -- | Enqueue a 'UserText' event on the per-tab input queue.
  263 -- Non-blocking: returns @Left (TabConcurrencyLimit n)@ if the queue is
  264 -- full so the dispatcher's send loop never blocks (H4).
  265 sendUserText :: AiTabState -> Text -> IO (Either TabError ())
  266 sendUserText state t = tryEnqueue (_ats_inputQ state) (UserText t)
  267 
  268 -- | Enqueue a 'SlashCmd' event. For 'KindAi' this is the legitimate
  269 -- E5 path: the focused-tab slash command is delivered to the loop
  270 -- (which runs 'executeSlashCommand' against the /per-tab/ context).
  271 enqueueSlashAi :: AiTabState -> SlashCommand -> IO (Either TabError ())
  272 enqueueSlashAi state cmd = tryEnqueue (_ats_inputQ state) (SlashCmd cmd)
  273 
  274 -- | Atomic, non-blocking write to a 'TBQueue': if the queue is full
  275 -- right now, return 'Left' with a redacted 'TabConcurrencyLimit'
  276 -- rather than blocking. The capacity is not exposed by the STM API
  277 -- after construction, so we report @0@ as a sentinel — the value is
  278 -- never user-facing (it appears only inside 'TabConcurrencyLimit'
  279 -- which renders as the fixed-vocabulary "tab: input queue full" via
  280 -- 'PureClaw.Handles.Tab.toPublicTabError').
  281 tryEnqueue :: TBQueue InputEvent -> InputEvent -> IO (Either TabError ())
  282 tryEnqueue q ev = atomically (tryEnqueueSTM q ev)
  283 
  284 -- | STM body of 'tryEnqueue', factored out so the tests can drive it
  285 -- inside a larger transaction if needed.
  286 tryEnqueueSTM :: TBQueue InputEvent -> InputEvent -> STM (Either TabError ())
  287 tryEnqueueSTM q ev = do
  288   full <- isFullTBQueue q
  289   if full
  290     then pure (Left (TabConcurrencyLimit 0))
  291     else writeTBQueue q ev >> pure (Right ())

  294 -- ---------------------------------------------------------------------------
  295 -- _tabHandle_close — kind-specific graceful + force semantics
  296 -- ---------------------------------------------------------------------------
  297 
  298 -- | Close an AI tab. Idempotent + never throws (H6, H7).
  299 --
  300 -- * 'CloseGraceful': cancel the loop, archive the session via
  301 --   @_sh_save@, then flush the transcript via the SessionHandle's
  302 --   '_sh_transcript' field.
  303 -- * 'CloseForce': cancel the loop and skip the archive (H9 — the
  304 --   transcript stays on disk per the SessionHandle's append-only
  305 --   contract, but the metadata flush is suppressed).
  306 closeTabAi :: AgentEnv -> AiTabState -> CloseMode -> IO ()
  307 closeTabAi env state mode = do
  308   alreadyClosed <- atomicModifyIORef' (_ats_closed state) (True,)
  309   unless alreadyClosed $ do
  310     -- The cancel + save + transcript-flush triad is intentionally
  311     -- one-shot: rerunning it would attempt to '_sh_save' against a
  312     -- handle whose IORef-backed metadata may have been GCed.
  313     safeIgnore (cancelRunner state)
  314     case mode of
  315       CloseGraceful -> do
  316         safeIgnore (archiveSession env)
  317         safeIgnore (flushTranscript env)
  318       CloseForce ->
  319         -- Force-close intentionally skips the metadata save; the
  320         -- transcript is also NOT closed (per the design's "transcript
  321         -- deleted from disk" semantics, which lands properly in WU9
  322         -- alongside the on-disk teardown).
  323         pure ()
  324 
  325 -- | Read the captured 'TabRunner' (set by 'mkTabAi' after fork) and
  326 -- invoke its cancel. Tolerant of the brief allocation\/fork window
  327 -- when '_ats_runner' is still 'Nothing'.
  328 cancelRunner :: AiTabState -> IO ()
  329 cancelRunner state = do
  330   mRunner <- readIORef (_ats_runner state)
  331   for_ mRunner _trun_cancel
  332 
  333 -- | Persist the session metadata via '_sh_save' on the active
  334 -- 'SessionHandle'.
  335 archiveSession :: AgentEnv -> IO ()
  336 archiveSession env = do
  337   sh <- readIORef (_env_session env)
  338   _sh_save sh
  339 
  340 -- | Flush + close the transcript file descriptor on the active
  341 -- 'SessionHandle'. Per the spec we use '_th_close' on
  342 -- '_sh_transcript', not '_sh_close' (which does not exist).
  343 flushTranscript :: AgentEnv -> IO ()
  344 flushTranscript env = do
  345   sh <- readIORef (_env_session env)
  346   _th_close (_sh_transcript sh)
  347 
  348 -- | Best-effort: run an IO action and swallow synchronous failures.
  349 -- The close path MUST be never-throws (H7), so every step is wrapped
  350 -- defensively. 'AsyncCancelled' is also swallowed here because the
  351 -- close handler is invoked from outside the loop thread; any
  352 -- AsyncCancelled bubbling up through cancel itself is benign (the
  353 -- target thread has already received the cancel).
  354 safeIgnore :: IO () -> IO ()
  355 safeIgnore m = do
  356   _ <- try @SomeException m
  357   pure ()

  360 -- ---------------------------------------------------------------------------
  361 -- Loop body — single-writer Context, async-cancel-aware
  362 -- ---------------------------------------------------------------------------
  363 
  364 -- | Top-level loop body. Runs forever until either the per-tab close
  365 -- cancels it (via 'AsyncCancelled') or a synchronous exception is
  366 -- caught (in which case the tab transitions to 'Crashed' and the
  367 -- loop exits).
  368 loopBody :: AgentEnv -> TabIndex -> AiTabState -> IO ()
  369 loopBody env idx state =
  370   safelyRunLoop state (loop env idx state)
  371 
  372 -- | Outer exception handler: catch 'SomeException' EXCEPT
  373 -- 'AsyncCancelled', which must propagate so that 'bracket'-based
  374 -- close semantics work correctly (per C5).
  375 --
  376 -- On a Crashed exit, the status is updated to 'Crashed' carrying a
  377 -- redacted 'PublicTabError'.
  378 safelyRunLoop :: AiTabState -> IO () -> IO ()
  379 safelyRunLoop state body = body `catch` handler
  380   where
  381     handler :: SomeException -> IO ()
  382     handler e = case fromException e :: Maybe AsyncCancelled of
  383       Just _  -> throwIO e  -- propagate AsyncCancelled (C5)
  384       Nothing -> do
  385         -- Generic crash: surface as Crashed status with a redacted
  386         -- short label so the dispatcher's '/N' switch can emit the
  387         -- H8 banner. The exception message is intentionally
  388         -- discarded to preserve the redacted-error contract.
  389         writeIORef (_ats_statusRef state)
  390                    (Crashed (PublicTabError "tab: ai loop crashed"))
  391 
  392 -- | The main loop. Reads one 'InputEvent' at a time, transitions the
  393 -- tab status across the event, and continues.
  394 loop :: AgentEnv -> TabIndex -> AiTabState -> IO ()
  395 loop env idx state = do
  396   ev <- atomically (readTBQueue (_ats_inputQ state))
  397   handleEvent env idx state ev
  398   loop env idx state
  399 
  400 -- | Dispatch a single 'InputEvent'.
  401 handleEvent :: AgentEnv -> TabIndex -> AiTabState -> InputEvent -> IO ()
  402 handleEvent env idx state ev = case ev of
  403   UserText t
  404     | "/" `T.isPrefixOf` T.strip t ->
  405         -- I2 re-parse path: if the user typed a slash-form via the
  406         -- Inject route, re-classify it as a SlashCmd before deciding
  407         -- whether to feed the provider. Unknown slash inputs fall
  408         -- through to the provider path (matching Agent.Loop's
  409         -- "Unknown command" behaviour).
  410         case Parse.parseSlashCommand (T.strip t) of
  411           Just cmd -> runSlash env idx state cmd
  412           Nothing  -> runUserText env idx state t
  413     | otherwise          -> runUserText env idx state t
  414   SlashCmd cmd           -> runSlash env idx state cmd
  415 
  416 -- | Run a slash command against the per-tab context. The single-writer
  417 -- mutation is @readIORef → executeSlashCommand → writeIORef@.
  418 runSlash :: AgentEnv -> TabIndex -> AiTabState -> SlashCommand -> IO ()
  419 runSlash env _idx state cmd = withActive state $ do
  420   ctx <- readIORef (_ats_context state)
  421   ctx' <- executeSlashCommand env cmd ctx
  422   writeIORef (_ats_context state) ctx'
  423 
  424 -- | Run a non-slash user text turn: append the user message, call the
  425 -- provider once (with chunked streaming when focused), and append the
  426 -- assistant response. Per E5 this is single-writer.
  427 --
  428 -- If no provider is configured the loop emits a dispatcher-side error
  429 -- banner via the channel-out queue (so the user sees feedback even on
  430 -- non-focused tabs).
  431 runUserText :: AgentEnv -> TabIndex -> AiTabState -> Text -> IO ()
  432 runUserText env idx state t = withActive state $ do
  433   mProvider <- readIORef (_ats_provider state)
  434   mModel    <- readIORef (_ats_model state)
  435   case (mProvider, mModel) of
  436     (Just provider, Just model) -> do
  437       ctx <- readIORef (_ats_context state)
  438       let userMsg = textMessage User (T.strip t)
  439           ctx'   = Ctx.addMessage userMsg ctx
  440       writeIORef (_ats_context state) ctx'
  441       runOneTurn env idx state provider model ctx'
  442     _ ->
  443       -- No provider / no model — emit a single dispatcher banner so
  444       -- the user knows why nothing happened. (The legacy
  445       -- 'Agent.Loop' path emits the long-form message; for WU6 we
  446       -- keep it short to avoid duplicating the help-text format.)
  447       emitBanner env "(no provider configured for this tab)"
  448 
  449 -- | Bracket the body in an 'Active' status transition so the
  450 -- dashboard can observe it.
  451 --
  452 -- /WU6 scope note:/ the S9 atomic active-count cap-check belongs to
  453 -- the spawn path (WU9). For WU6 we keep the status transition only;
  454 -- the @_env_activeCount@ wiring lands when the cap is enforced by
  455 -- spawn UX.
  456 withActive :: AiTabState -> IO () -> IO ()
  457 withActive state = bracket_ enter exit
  458   where
  459     enter = writeIORef (_ats_statusRef state) Active
  460     exit = do
  461       now <- getCurrentTime
  462       writeIORef (_ats_statusRef state) (Idle now)

  465 -- ---------------------------------------------------------------------------
  466 -- Single-turn provider call (streaming + tool-call cycle)
  467 -- ---------------------------------------------------------------------------
  468 
  469 -- | Execute one provider turn. Streams chunks into '_env_channelOutQ'
  470 -- as @SrcTab idx@ events (focus-gated by the writer thread per D3),
  471 -- records the response into the per-tab context, and on tool-use
  472 -- continues the turn until no more tool calls are produced.
  473 runOneTurn
  474   :: AgentEnv
  475   -> TabIndex
  476   -> AiTabState
  477   -> SomeProvider
  478   -> ModelId
  479   -> Ctx.Context
  480   -> IO ()
  481 runOneTurn env idx state provider model ctx = do
  482   -- D4 producer-side focus optimisation: if this tab is not focused,
  483   -- we still execute the provider call (the user's input must be
  484   -- processed) but we skip the per-chunk enqueue work since the
  485   -- writer thread would drop them anyway.
  486   curFocus <- readIORef (_env_focus env)
  487   let focusedNow = shouldEmit curFocus (SrcTab idx)
  488   sid <- nextStreamId state
  489   when focusedNow $
  490     atomically $ writeTBQueue (_env_channelOutQ env)
  491                    (SrcTab idx, StreamStart sid idx)
  492   result <- runProviderTurn env idx provider model ctx focusedNow sid
  493   -- Always emit StreamEnd if we emitted StreamStart, so the
  494   -- breadcrumb-state map is GC'd by the writer thread (D5).
  495   when focusedNow $
  496     atomically $ writeTBQueue (_env_channelOutQ env)
  497                    (SrcTab idx, StreamEnd sid)
  498   case result of
  499     Left _e ->
  500       emitBanner env (errorBanner idx)
  501     Right response -> do
  502       let ctx' = Ctx.recordUsage (_crsp_usage response)
  503                $ Ctx.addMessage (Message Assistant (_crsp_content response))
  504                                 ctx
  505       writeIORef (_ats_context state) ctx'
  506       -- Append the (full) assistant text to the transcript via the
  507       -- session's transcript handle. Mirrors what Agent.Loop's
  508       -- mkTranscriptProvider wrapper does.
  509       let assistantText = responseText response
  510       unless (T.null (T.strip assistantText)) $
  511         appendTranscript env assistantText
  512       -- Tool-call continuation: full tool-call cycling is part of
  513       -- the WU10 refactor that ports Agent.Loop's logic wholesale.
  514       -- For WU6 we log the presence of tool calls so the gap is
  515       -- visible in production.
  516       unless (null (toolUseCalls response)) $
  517         _lh_logDebug (_env_logger env)
  518           "tab.ai: tool calls present but not executed in WU6 — \
  519           \deferred to WU10 refactor"
  520 
  521 -- | Invoke the provider exactly once, streaming chunks to the channel
  522 -- output queue when focused.
  523 runProviderTurn
  524   :: AgentEnv
  525   -> TabIndex
  526   -> SomeProvider
  527   -> ModelId
  528   -> Ctx.Context
  529   -> Bool
  530   -> StreamId
  531   -> IO (Either SomeException CompletionResponse)
  532 runProviderTurn env idx provider model ctx focusedNow sid = do
  533   responseRef <- newIORef (Nothing :: Maybe CompletionResponse)
  534   let req = CompletionRequest
  535         { _cr_model        = model
  536         , _cr_messages     = Ctx.contextMessages ctx
  537         , _cr_systemPrompt = Ctx.contextSystemPrompt ctx
  538         , _cr_maxTokens    = Just 4096
  539         , _cr_tools        = registryDefinitions (_env_registry env)
  540         , _cr_toolChoice   = Nothing
  541         }
  542   outcome <- try @SomeException $
  543     completeStream provider req $ \case
  544       StreamText t ->
  545         when focusedNow $
  546           atomically $ writeTBQueue (_env_channelOutQ env)
  547                          (SrcTab idx, ChunkOf sid t)
  548       StreamDone resp ->
  549         writeIORef responseRef (Just resp)
  550       StreamWarning w ->
  551         _lh_logWarn (_env_logger env) w
  552       _ -> pure ()
  553   case outcome of
  554     Left e  -> pure (Left e)
  555     Right () -> do
  556       mResp <- readIORef responseRef
  557       pure $ case mResp of
  558         Just resp -> Right resp
  559         Nothing   -> Right (emptyResponseFor model)
  560 
  561 -- | A stub 'CompletionResponse' returned when the provider's
  562 -- 'completeStream' callback never delivered 'StreamDone'. This
  563 -- preserves the "always returns" contract without crashing the loop.
  564 emptyResponseFor :: ModelId -> CompletionResponse
  565 emptyResponseFor m = CompletionResponse
  566   { _crsp_content = []
  567   , _crsp_model   = m
  568   , _crsp_usage   = Nothing
  569   }

  571 -- | Allocate the next 'StreamId' for this tab.
  572 nextStreamId :: AiTabState -> IO StreamId
  573 nextStreamId state =
  574   atomicModifyIORef' (_ats_streamCtr state)
  575     (\n -> let n' = n + 1 in (n', mkStreamId n'))

  582 -- | Append a single assistant-text entry to the session's transcript.
  583 --
  584 -- /WU6 scope note:/ the legacy 'PureClaw.Agent.Loop' already wraps
  585 -- its provider in 'PureClaw.Transcript.Provider.mkTranscriptProvider'
