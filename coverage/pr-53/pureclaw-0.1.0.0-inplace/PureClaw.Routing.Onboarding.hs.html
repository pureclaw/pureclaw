    2 -- Module      : PureClaw.Routing.Onboarding
    3 -- Description : User-facing onboarding for Tabbed Chat
    4 --               (Issue #51, WU11, O-series).
    5 --
    6 -- Owns the user-discoverable surface area introduced by Tabbed Chat:
    7 --
    8 --   * @\/start@ handler ('handleStart') — Telegram's onboarding
    9 --     convention. Emits a single-message orientation that names the
   10 --     three principal slash-prefix surfaces users need to know about:
   11 --     @\/0@ (the first tab once spawned), @\/tab new shell@ (allocate
   12 --     a shell tab at the next free slot), and @\/tabs@ (dashboard).
   13 --     The message is sent via 'ChannelHandle' @_ch_send@ which works
   14 --     on every channel implementation (CLI, Signal, Telegram).
   15 --     Non-Telegram channels degrade gracefully — they receive the same
   16 --     orientation text even though they have no BotFather autocomplete
   17 --     behind the @\/start@ keyword (O1).
   18 --
   19 --   * @\/help@ \"Tab commands\" subsection ('helpTabSection') — pure
   20 --     text appended by @executeSlashCommand env CmdHelp@ in
   21 --     "PureClaw.Agent.SlashCommands" so users see the new vocabulary
   22 --     enumerated alongside the existing slash commands (O2). The
   23 --     subsection is intentionally lightweight; the full per-command
   24 --     reference still lives in the existing command-spec table.
   25 --
   26 --   * BotFather command registration list ('botFatherCommandList') —
   27 --     the @(command, description)@ tuples Telegram's @setMyCommands@
   28 --     API consumes at channel-startup time. Pinning this as a top-
   29 --     level pure binding lets the O3 test compare it against the
   30 --     golden enumeration in @docs\/tabbed-chat.md@ without spinning
   31 --     up a live channel (O3).
   32 --
   33 -- == Design constraints (per WU11 spec)
   34 --
   35 --   * The module imports only 'AgentEnv', 'ChannelHandle', and 'Text'
   36 --     so it can be imported safely by 'PureClaw.Agent.SlashCommands'
   37 --     (which owns 'CmdStart' and the @\/help@ renderer) without
   38 --     dragging in the dispatcher or any tab factory.
   39 --   * 'handleStart' is non-blocking and has no side effects beyond
   40 --     a single channel emit. It does NOT spawn a tab, mutate
   41 --     '_env_focus', or touch '_env_tabs' — those flows are owned by
   42 --     'PureClaw.Routing.Dispatcher' \/ 'PureClaw.Routing.AutoSpawn'
   43 --     and trigger automatically on the user's next slash command.
   44 --   * No 'forkIO'. The handler runs synchronously on the caller's
   45 --     thread (the dispatcher thread when invoked via
   46 --     'PureClaw.Routing.Dispatcher.dispatchSlash', or whatever thread
   47 --     "PureClaw.Agent.SlashCommands" runs on for the legacy single-
   48 --     tab execution path).
   49 --
   50 -- See @docs\/tabbed-chat.md@ §\"Onboarding (O-series)\" for the
   51 -- normative DoDs (O1, O2, O3).
   52 module PureClaw.Routing.Onboarding
   53   ( -- * @\/start@ handler
   54     handleStart
   55   , onboardingMessage
   56     -- * @\/help@ subsection (consumed by Agent.SlashCommands)
   57   , helpTabSection
   58     -- * BotFather command registration (consumed by Channels.Telegram)
   59   , botFatherCommandList
   60   ) where

   69 -- ---------------------------------------------------------------------------
   70 -- /start handler (O1)
   71 -- ---------------------------------------------------------------------------
   72 
   73 -- | Handle the @\/start@ slash command. Emits the orientation message
   74 -- ('onboardingMessage') via the channel's @_ch_send@ once.
   75 --
   76 -- O1 requires the response to include:
   77 --
   78 --   * a one-line value prop,
   79 --   * @\/0@ (the first tab once spawned),
   80 --   * @\/tab new shell@ (the shell-tab allocation form),
   81 --   * @\/tabs@ (the dashboard alias).
   82 --
   83 -- All four are baked into 'onboardingMessage' so the @\/start@ output
   84 -- is a fixed, golden-comparable string the test can assert verbatim
   85 -- substrings against.
   86 --
   87 -- == Channel portability
   88 --
   89 -- Per the WU11 spec, non-Telegram channels @degrade gracefully@: they
   90 -- still receive the orientation text. The implementation uses only
   91 -- 'ChannelHandle' @_ch_send@ which every channel implements, so
   92 -- nothing special is required for CLI \/ Signal — the user just sees
   93 -- the same help text Telegram users see, minus the BotFather
   94 -- autocomplete affordance.
   95 handleStart :: AgentEnv -> IO ()
   96 handleStart env =
   97   _ch_send (_env_channel env) (OutgoingMessage onboardingMessage)
   98 
   99 -- | The fixed orientation message delivered by 'handleStart'.
  100 --
  101 -- Authored as a single 'Text' so the O1 test can assert against
  102 -- specific substrings without re-deriving them from a builder.
  103 -- Mentions all three slash-prefix surfaces explicitly (value prop,
  104 -- @\/0@, @\/tab new shell@, @\/tabs@). The @\/N@ shortcut accepts
  105 -- any single digit @0-9@ or lowercase letter @a-z@ (36 tabs total).
  106 --
  107 -- Tabs follow a tmux-style packing model: they always occupy the
  108 -- lowest slots, so the first tab is always at @\/0@ and closing tab N
  109 -- renumbers the rest. The orientation copy mentions this so users
  110 -- don't expect to pick a slot themselves.
  111 onboardingMessage :: Text
  112 onboardingMessage = T.intercalate "\n"
  113   [ "Welcome to PureClaw."
  114   , ""
  115   , "Tabbed Chat lets you drive multiple agents from one chat thread."
  116   , ""
  117   , "Quick start:"
  118   , "  hello there        \x2014 just type to talk (an AI tab is auto-spawned)"
  119   , "  /tab new shell     \x2014 open a shell tab at the next free slot"
  120   , "  /0                 \x2014 switch focus to tab 0 (must already exist)"
  121   , "  /tabs              \x2014 list your tabs"
  122   , ""
  123   , "Tabs are always packed in the lowest slots; closing tab N"
  124   , "renumbers the rest (tmux-style)."
  125   , ""
  126   , "Tab shortcut /N accepts any digit 0-9 or lowercase letter a-z"
  127   , "(36 tabs total). /N only works for tabs that already exist \x2014"
  128   , "use /tab new to create one."
  129   , ""
  130   , "See /help for the full command reference."
  131   ]

  138 -- | The @Tab commands@ subsection appended to @\/help@ output by
  139 -- 'PureClaw.Agent.SlashCommands.renderHelpText'. Pure text so the O2
  140 -- assertion can pin the literal substrings @\"Tab commands\"@ and
  141 -- @\"\/tabs\"@ verbatim.
  142 --
  143 -- The format mirrors the existing @\/help@ group layout — two-space
  144 -- indented heading followed by a four-space-indented body — so the
  145 -- subsection visually merges with the surrounding renderer output.
  146 --
  147 -- All Tab-Chat verbs enumerated per O2: @\/N@, @\/N \<payload\>@,
  148 -- @\/tabs@, @\/tab new@, @\/tab close@, @\/tab focus@, @\/tab resume@,
  149 -- @\/tab rename@.
  150 --
  151 -- The leading-line note documents the @\/N@ grammar (single char,
  152 -- digit @0-9@ or lowercase letter @a-z@; 36 tabs maximum).
  153 helpTabSection :: Text
  154 helpTabSection = T.intercalate "\n"
  155   [ ""
  156   , "  Tab commands (N is one char: digit 0-9 or letter a-z; 36 tabs max):"
  157   , "  Tabs are tmux-style packed: they always occupy the lowest slots;"
  158   , "  closing tab N renumbers the rest down by one."
  159   , "    /N                          Switch focus to tab N (error if missing)"
  160   , "    /N <payload>                Direct-inject payload to tab N (focus unchanged)"
  161   , "    /tabs                       List all tabs (alias of /tab list)"
  162   , "    /tab new [<kind>]           Open a new tab at the next free slot (kinds: ai, harness, shell, ssh, tmux)"
  163   , "    /tab close <N> [--force]    Close tab N (--force skips archive on AI tabs); remaining tabs renumber"
  164   , "    /tab focus <N>              Switch focus to tab N (alias of /N)"
  165   , "    /tab resume <session-id>    Resume an archived session into a new tab"
  166   , "    /tab rename <N> <name>      Rename tab N (input is sanitized)"
  167   ]

  170 -- ---------------------------------------------------------------------------
  171 -- BotFather command registration list (O3)
  172 -- ---------------------------------------------------------------------------
  173 
  174 -- | The @(command, description)@ tuples PureClaw's Telegram channel
  175 -- registers with BotFather (via the Telegram @setMyCommands@ API) at
  176 -- channel-startup time.
  177 --
  178 -- Pinning this as a top-level pure value lets the O3 test compare it
  179 -- against the golden enumeration in
  180 -- @docs\/tabbed-chat.md@ §\"Channel autocomplete (Telegram-specific)\"
  181 -- without spinning up a live channel.
  182 --
  183 -- The 36 single-char tab-switch shortcuts (@\/0@..@\/9@, @\/a@..@\/z@)
  184 -- are emitted in their full glory because Telegram's
  185 -- @setMyCommands@ API permits up to 100 entries per bot and visible
  186 -- autocomplete on mobile is the primary discovery affordance for the
  187 -- @\/N@ grammar. The 36 + 3 long-form entries fit comfortably under
  188 -- the limit (39 total).
  189 --
  190 -- Per O3 the descriptions are:
  191 --
  192 --   * @\/0@..@\/9@, @\/a@..@\/z@ → @\"Switch to tab N\"@
  193 --   * @\/tab@      → @\"Tabs: new, list, close, focus, resume, rename\"@
  194 --   * @\/tabs@     → @\"List all tabs\"@
  195 --   * @\/start@    → @\"Tabbed Chat \x2014 see \/help for tab commands\"@
  196 --
  197 -- The tuples are recorded in declaration order so the test asserts a
  198 -- stable list (rather than a set) — the user-visible BotFather
  199 -- ordering is part of the contract.
  200 botFatherCommandList :: [(Text, Text)]
  201 botFatherCommandList =
  202   [ (cmd, "Switch to tab N")
  203   | n <- [0 .. 9 :: Int]
  204   , let cmd = T.pack ('/' : show n)
  205   ]
  206   ++
  207   [ (T.pack ['/', c], "Switch to tab N")
  208   | c <- ['a' .. 'z']
  209   ]
  210   ++
  211   [ ("/tab",   "Tabs: new, list, close, focus, resume, rename")
  212   , ("/tabs",  "List all tabs")
  213   , ("/start", "Tabbed Chat \x2014 see /help for tab commands")
  214   ]
