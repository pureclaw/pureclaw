    1 -- |
    2 -- Module      : PureClaw.Routing.Dashboard
    3 -- Description : Tab dashboard rendering (Tabbed Chat WU9, B1\/B2\/B3).
    4 --
    5 -- Renders the user-facing tab dashboard emitted by the @\/tabs@ (and
    6 -- equivalent @\/tab list@) command.
    7 --
    8 -- == Rendering rules
    9 --
   10 -- * Empty registry (B1) — emits a single helper line:
   11 --
   12 --   @
   13 --   No tabs open. Use \/tab new \<kind\> to create one.
   14 --   @
   15 --
   16 --   Under the tmux packing model, the user does not pick a slot — the
   17 --   factory allocates at the next free index. @\/N@ on a missing tab
   18 --   errors (no auto-spawn), so it is intentionally not advertised
   19 --   here as a way to create tabs.
   20 --
   21 -- * 1 ≤ N \< 8 tabs (B2) — one line per tab, each carrying
   22 --   @index, kind, redacted name, status, asterisk if focused@. The
   23 --   layout uses a leading numeric prefix (@\/0@) followed by a
   24 --   single-space-separated field list so the output renders well in
   25 --   both fixed-width (CLI) and proportional (Telegram \/ Signal) fonts.
   26 --
   27 -- * ≥ 8 tabs (B3) — switches to a bullet-list rendering so the output
   28 --   wraps cleanly on small mobile screens (no fixed-width table).
   29 --
   30 -- == Producer side
   31 --
   32 -- The function returns the formatted body as a single 'Text'; the
   33 -- caller (typically the @\/tabs@ slash-command handler in the
   34 -- dispatcher) wraps the result in 'BannerLine' for emission on
   35 -- '_env_channelOutQ'. Dashboard rendering NEVER reaches the provider
   36 -- and NEVER mutates any tab state (P18 invariant preserved).
   37 --
   38 -- See @docs\/tabbed-chat.md@ §"Dashboard (B-series)" for the spec.
   39 module PureClaw.Routing.Dashboard
   40   ( -- * Render
   41     renderDashboard
   42     -- * Exposed for testing
   43   , bulletThreshold
   44   , emptyDashboardText
   45   ) where

   65 -- | Tab count at or above which the renderer switches from one-line-
   66 -- per-tab to a bullet-style rendering (B3). Exposed for tests so the
   67 -- threshold can be asserted without re-deriving it.
   68 bulletThreshold :: Int
   69 bulletThreshold = 8
   70 
   71 -- | Helper text emitted by the empty-registry branch (B1). Exposed for
   72 -- tests so the exact wording can be pinned without re-deriving it.
   73 emptyDashboardText :: Text
   74 emptyDashboardText =
   75   "No tabs open. Use /tab new <kind> to create one."
   76 
   77 
   78 -- | Render the tab dashboard.
   79 --
   80 -- Reads '_env_tabs' and '_env_focus' atomically (single IORef each)
   81 -- and consults each tab's '_tabHandle_status' to render the status
   82 -- column. The status read is wrapped in a tolerant 'tryStatus' helper
   83 -- so a misbehaving factory cannot crash the dashboard.
   84 --
   85 -- Output is a single 'Text' with embedded @\\n@ separators between
   86 -- per-tab lines; the caller is responsible for emitting it as a
   87 -- single 'BannerLine'.
   88 renderDashboard :: AgentEnv -> IO Text
   89 renderDashboard env = do
   90   tabs   <- readIORef (_env_tabs env)
   91   mFocus <- readIORef (_env_focus env)
   92   if IntMap.null tabs
   93     then pure emptyDashboardText
   94     else do
   95       let entries = IntMap.toAscList tabs
   96           useBullets = length entries >= bulletThreshold
   97       lined <- mapM (renderEntry mFocus useBullets) entries
   98       pure (T.intercalate "\n" lined)
   99 
  100 -- | Render a single registry entry. The format mirrors the design
  101 -- doc's B2 example: @\/I [bullet] kind name status focus-marker@.
  102 renderEntry
  103   :: Maybe TabIndex
  104   -> Bool          -- ^ True ⇒ bullet rendering (B3)
  105   -> (Int, TabHandle)
  106   -> IO Text
  107 renderEntry mFocus useBullets (key, h) = do
  108   let idx        = _tabHandle_index h
  109       kindTxt    = renderKind (_tabHandle_kind h)
  110       nameTxt    = renderName (_tabHandle_name h)
  111       idxTxt     = "/" <> T.pack (show key)
  112       focusMark  = if mFocus == Just idx then " *" else ""
  113       bulletTxt  = if useBullets then " - " else " "
  114   statusTxt <- tryStatus h
  115   pure (idxTxt <> bulletTxt <> kindTxt
  116                 <> " " <> nameTxt
  117                 <> " " <> statusTxt
  118                 <> focusMark)
  119 
  120 -- | Render a 'TabKind' as its short keyword. Mirrors
  121 -- 'PureClaw.Routing.Config.tabKindCodec' so the dashboard and the
  122 -- config file use the same vocabulary.
  123 renderKind :: TabKind -> Text
  124 renderKind k = case k of
  125   KindAi      -> "ai"
  126   KindHarness -> "harness"
  127   KindShell   -> "shell"
  128   KindSsh     -> "ssh"
  129   KindTmux    -> "tmux"
  130 
  131 -- | Project a 'TabName' to its render-safe text. The 'TabName' newtype
  132 -- guarantees H11 (sanitised), so this is just the unwrap.
  133 renderName :: TabName -> Text
  134 renderName = unTabName
  135 
  136 -- | Read '_tabHandle_status' tolerantly. The status field is in IO;
  137 -- a buggy factory could throw. The dashboard collapses any exception
  138 -- into the short label @\"?\"@ so the rendering never crashes the
  139 -- caller.
  140 tryStatus :: TabHandle -> IO Text
  141 tryStatus h = do
  142   r <- try @SomeException (_tabHandle_status h)
  143   pure $ case r of
  144     Left  _ -> "?"
  145     Right s -> renderStatus s
  146 
  147 -- | Render a 'TabStatus' as its short user-visible label.
  148 renderStatus :: TabStatus -> Text
  149 renderStatus s = case s of
  150   Active      -> "active"
  151   Idle _      -> "idle"
  152   Crashed pe  -> "crashed (" <> unPublicTabError pe <> ")"
