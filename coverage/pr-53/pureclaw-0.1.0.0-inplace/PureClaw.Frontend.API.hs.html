  150 -- | WAI application handling @\/api\/*@ routes.
  151 apiApp :: FrontendEnv -> Application
  152 apiApp env req respond = do
  153   let method = requestMethod req
  154       path   = pathInfo req
  155   case (method, path) of
  156     ("GET", ["api", "harnesses"])            -> handleHarnesses env respond
  157     ("GET", ["api", "sessions", "recent"])   -> handleRecentSessions env respond
  158     ("GET", ["api", "sessions", sid, "transcript"]) ->
  159       handleTranscript env sid respond
  160     ("POST", ["api", "sessions", "new"]) ->
  161       handleNewSession env req respond
  162     ("POST", ["api", "sessions", sid, "send"]) ->
  163       handleSend env sid req respond
  164     ("PUT", ["api", "sessions", sid, "prompt"]) ->
  165       handleSetPrompt env sid req respond
  166     ("POST", ["api", "sessions", sid, "archive"]) ->
  167       handleSetArchived env sid True respond
  168     ("POST", ["api", "sessions", sid, "unarchive"]) ->
  169       handleSetArchived env sid False respond
  170     ("PUT", ["api", "sessions", sid, "description"]) ->
  171       handleSetDescription env sid req respond
  172     ("GET", ["api", "agents"])               -> handleAgents env respond
  173     _                                        -> respondNotFound respond

  175 handleHarnesses :: FrontendEnv -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  176 handleHarnesses env respond = do
  177   harnesses <- readIORef (_fe_harnesses env)
  178   infos <- traverse probeHarness (Map.toList harnesses)
  179   respond $ jsonResponse status200 infos

  181 -- | Probe a single harness to determine its activity.
  182 -- Checks process status first, then does a quick screen capture
  183 -- to distinguish thinking from idle.
  184 probeHarness :: (Text, HarnessHandle) -> IO HarnessInfo
  185 probeHarness (name, hh) = do
  186   st <- _hh_status hh
  187   activity <- case st of
  188     HarnessExited _ -> pure HarnessStopped
  189     HarnessRunning  -> probeActivity (_hh_session hh) name
  190   pure HarnessInfo
  191     { _hi_name     = name
  192     , _hi_activity = activity
  193     }

  195 -- | Capture the tmux window and check if the harness is idle or thinking.
  196 probeActivity :: Text -> Text -> IO HarnessActivity
  197 probeActivity tmuxSession windowName = do
  198   let target = tmuxSession <> ":" <> extractWindowIdx windowName
  199   result <- try @SomeException $ captureWindow target 50
  200   case result of
  201     Left _        -> pure HarnessIdle
  202     Right capture -> do
  203       let screenText = TE.decodeUtf8Lenient capture
  204       pure $ if isIdle screenText then HarnessIdle else HarnessThinking
  205 
  206 -- | Extract the window index suffix from a harness name like "claude-code-0".
  207 extractWindowIdx :: Text -> Text
  208 extractWindowIdx name =
  209   case T.splitOn "-" name of
  210     parts | length parts >= 2 -> last parts
  211     _                         -> "0"

  213 handleRecentSessions :: FrontendEnv -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  214 handleRecentSessions env respond = do
  215   let limit   = _fe_recentLimit env
  216       baseDir = _fe_sessionsDir env
  217   -- Over-fetch so the empty-transcript filter below can't shrink the
  218   -- visible count: listSessions already does a full directory scan and
  219   -- sorts the whole metadata set, so a generous bound is essentially
  220   -- free. Empty / archived sessions are dropped here, not erased on
  221   -- disk — archiving is purely a UI hint controlled by the user.
  222   metas    <- listSessions baseDir Nothing (limit * 3)
  223   let visible = filter (not . _sm_archived) metas
  224   nonEmpty <- filterM (hasTranscriptEntries baseDir) visible
  225   let chosen = take limit nonEmpty
  226   -- Read a first-message snippet per session (bounded read per file).
  227   -- This is the cheap display fallback when no user description and no
  228   -- model-generated summary exist yet.
  229   snippets <- traverse (firstMessageSnippet baseDir) chosen
  230   let infos = zipWith toSessionInfo chosen snippets
  231   respond $ jsonResponse status200 infos

  233 -- | Toggle the archive flag on a session. The 'archived' argument is
  234 -- the **target** value (True for /archive, False for /unarchive), so
  235 -- the routes are idempotent: archiving an already-archived session is
  236 -- a no-op success. The session directory and transcript are never
  237 -- removed — see Session.Handle.setArchived for the disk-side contract.
  238 handleSetArchived
  239   :: FrontendEnv
  240   -> Text
  241   -> Bool
  242   -> (Response -> IO ResponseReceived)
  243   -> IO ResponseReceived
  244 handleSetArchived env sidText archived respond = do
  245   let sid = SessionId sidText
  246   result <- setArchived (_fe_sessionsDir env) sid archived
  247   case result of
  248     Right () ->
  249       respond $ jsonResponse status200 (object ["archived" .= archived])
  250     Left SetArchivedSessionMissing ->
  251       respondNotFound respond
  252     Left (SetArchivedParseFailed msg) ->
  253       respond $ jsonResponse status500 (object ["error" .= msg])

  308 -- | Cheap fallback for display: read just the first line of the
  309 -- session's @transcript.jsonl@, decode it as a 'TranscriptEntry', and
  310 -- extract a short snippet of the first user message. Returns 'Nothing'
  311 -- when there's no transcript, the first entry isn't a request, or any
  312 -- decoding step fails. The returned string is at most
  313 -- 'snippetCharBudget' chars with newlines normalised to spaces.
  314 --
  315 -- Cost is one bounded read per call (we don't load the whole transcript).
  316 firstMessageSnippet :: FilePath -> SessionMeta -> IO (Maybe Text)
  317 firstMessageSnippet baseDir meta = do
  318   let path = baseDir </> T.unpack (unSessionId (_sm_id meta)) </> "transcript.jsonl"
  319   result <- try @IOException $ withFile path ReadMode BSC.hGetLine
  320   case result of
  321     Left _    -> pure Nothing
  322     Right line -> case Aeson.eitherDecodeStrict' line :: Either String TranscriptEntry of
  323       Left _ -> pure Nothing
  324       Right entry
  325         | _te_direction entry /= Request -> pure Nothing
  326         | otherwise -> pure (snippetFromPayload (_te_payload entry))
  327 
  328 snippetCharBudget :: Int
  329 snippetCharBudget = 120
  330 
  331 -- | Extract a display snippet from a request payload. Two shapes:
  332 -- (1) JSON object with a "messages" array (provider request) — pull
  333 --     text out of the first message; (2) plain text (harness send) —
  334 --     use the payload directly. Trimmed, newline-normalised, and
  335 --     truncated to 'snippetCharBudget' characters.
  336 snippetFromPayload :: Text -> Maybe Text
  337 snippetFromPayload raw = trimAndTruncate <$>
  338   case Aeson.decodeStrict (TE.encodeUtf8 raw) of
  339     Just (Aeson.Object o)
  340       | Just (Aeson.Array msgs) <- KM.lookup "messages" o
  341       , not (V.null msgs)
  342       -> messageText (V.unsafeHead msgs)
  343     _ -> Just raw
  344   where
  345     -- Anthropic-style message content: either a plain string or an
  346     -- array of {type:"text", text:"..."} blocks. We just want a
  347     -- human-readable lead.
  348     messageText :: Aeson.Value -> Maybe Text
  349     messageText (Aeson.Object m) = case KM.lookup "content" m of
  350       Just (Aeson.String s) -> Just s
  351       Just (Aeson.Array bs) -> Just (T.intercalate " " [t | Aeson.Object b <- V.toList bs
  352                                                           , Just (Aeson.String t) <- [KM.lookup "text" b]])
  353       _                     -> Nothing
  354     messageText _ = Nothing
  355 
  356     trimAndTruncate :: Text -> Text
  357     trimAndTruncate t =
  358       let normalized = T.unwords (T.words t)  -- collapse whitespace incl. newlines
  359       in  if T.length normalized > snippetCharBudget
  360             then T.take (snippetCharBudget - 1) normalized <> "\x2026"
  361             else normalized

  400 -- | Read transcript entries from a session's @transcript.jsonl@ file.
  401 handleTranscript :: FrontendEnv -> Text -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  402 handleTranscript env sid respond = do
  403   -- Reject path traversal
  404   if T.isInfixOf ".." sid || T.isInfixOf "/" sid
  405     then respond $ jsonResponse status400 (object ["error" .= ("Invalid session ID" :: Text)])
  406     else do
  407       let path = _fe_sessionsDir env </> T.unpack sid </> "transcript.jsonl"
  408       exists <- doesFileExist path
  409       if not exists
  410         then respond $ jsonResponse status404 (object ["error" .= ("Session not found" :: Text)])
  411         else do
  412           result <- try @SomeException $ readTranscriptFile path
  413           case result of
  414             Left _ ->
  415               respond $ jsonResponse status500 (object ["error" .= ("Failed to read transcript" :: Text)])
  416             Right entries ->
  417               respond $ jsonResponse status200 (map toTranscriptEntryInfo entries)

  419 -- | Read and parse a JSONL transcript file.
  420 readTranscriptFile :: FilePath -> IO [TranscriptEntry]
  421 readTranscriptFile path = do
  422   contents <- LBS.readFile path
  423   let linesBS = filter (not . LBS.null) (LBS.split 0x0A contents)
  424   pure [e | l <- linesBS, Just e <- [Aeson.decode l]]

  450 -- | Create a new empty session.
  451 handleNewSession :: FrontendEnv -> Request -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  452 handleNewSession env req respond = do
  453   body <- consumeBody req
  454   let parsed = case Aeson.eitherDecode body of
  455         Right r -> r
  456         Left _  -> NewSessionRequest Nothing Nothing
  457       mAgentText = _nsr_agent parsed
  458       mAgentName = mAgentText >>= either (const Nothing) Just . mkAgentName
  459       mPrefix = mAgentText >>= either (const Nothing) Just . mkSessionPrefix
  460   now <- getCurrentTime
  461   mModel <- readIORef (_fe_model env)
  462   let modelText = maybe "" unModelId mModel
  463       sid = newSessionId mPrefix now
  464       meta = SessionMeta
  465         { _sm_id                = sid
  466         , _sm_agent             = mAgentName
  467         , _sm_runtime           = RTProvider
  468         , _sm_model             = modelText
  469         , _sm_channel           = "web"
  470         , _sm_createdAt         = now
  471         , _sm_lastActive        = now
  472         , _sm_bootstrapConsumed = True
  473         , _sm_archived          = False
  474         , _sm_description       = Nothing
  475         , _sm_autoSummary       = Nothing
  476         }
  477   sh <- mkSessionHandle (_fe_logger env) (_fe_sessionsDir env) meta
  478   -- Write custom prompt file if provided
  479   case _nsr_customPrompt parsed of
  480     Just prompt | not (T.null (T.strip prompt)) ->
  481       TIO.writeFile (_sh_dir sh </> "custom-prompt.md") prompt
  482     _ -> pure ()
  483   _sh_save sh
  484   -- New sessions have no transcript yet, so the first-message snippet
  485   -- starts as Nothing.
  486   respond $ jsonResponse status200 (toSessionInfo meta Nothing)

  488 -- | Set or replace the custom prompt for a session, optionally updating
  489 -- the agent name in @session.json@.
  490 handleSetPrompt :: FrontendEnv -> Text -> Request -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  491 handleSetPrompt env sid req respond = do
  492   if T.isInfixOf ".." sid || T.isInfixOf "/" sid
  493     then respond $ jsonResponse status400 (object ["error" .= ("Invalid session ID" :: Text)])
  494     else do
  495       let sessionDir = _fe_sessionsDir env </> T.unpack sid
  496           promptPath = sessionDir </> "custom-prompt.md"
  497           metaPath   = sessionDir </> "session.json"
  498       body <- consumeBody req
  499       case Aeson.eitherDecode body of
  500         Left _ ->
  501           respond $ jsonResponse status400 (object ["error" .= ("Invalid JSON: expected {\"prompt\": \"...\"}" :: Text)])
  502         Right obj -> case Map.lookup ("prompt" :: Text) (obj :: Map.Map Text Text) of
  503           Nothing ->
  504             respond $ jsonResponse status400 (object ["error" .= ("Missing 'prompt' field" :: Text)])
  505           Just prompt -> do
  506             TIO.writeFile promptPath prompt
  507             -- Update agent name in session metadata if a name was provided
  508             let mName = Map.lookup "name" obj
  509             case mName of
  510               Just name | not (T.null name) -> do
  511                 raw <- LBS.readFile metaPath
  512                 case Aeson.eitherDecode' raw of
  513                   Right meta -> do
  514                     let agentName = either (const Nothing) Just (mkAgentName name)
  515                         updated = (meta :: SessionMeta) { _sm_agent = agentName }
  516                         tmpP = metaPath <> ".tmp"
  517                     LBS.writeFile tmpP (Aeson.encode updated)
  518                     renameFile tmpP metaPath
  519                   Left _ -> pure ()
  520               _ -> pure ()
  521             respond $ jsonResponse status200 (object ["ok" .= True])

  530 -- | Send a user message to a session and get a completion.
  531 handleSend :: FrontendEnv -> Text -> Request -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  532 handleSend env sid req respond = do
  533   -- Validate session ID
  534   if T.isInfixOf ".." sid || T.isInfixOf "/" sid
  535     then respond $ jsonResponse status400 (object ["error" .= ("Invalid session ID" :: Text)])
  536     else do
  537       let sessionDir = _fe_sessionsDir env </> T.unpack sid
  538           transcriptPath = sessionDir </> "transcript.jsonl"
  539       exists <- doesFileExist transcriptPath
  540       if not exists
  541         then respond $ jsonResponse status404 (object ["error" .= ("Session not found" :: Text)])
  542         else do
  543           -- Parse request body
  544           body <- consumeBody req
  545           case Aeson.eitherDecode body of
  546             Left _ ->
  547               respond $ jsonResponse status400 (object ["error" .= ("Invalid JSON: expected {\"message\": \"...\"}" :: Text)])
  548             Right (SendRequest userText) -> do
  549               mProvider <- readIORef (_fe_provider env)
  550               mModel <- readIORef (_fe_model env)
  551               case (mProvider, mModel) of
  552                 (Nothing, _) ->
  553                   respond $ jsonResponse status503 (object ["error" .= ("No provider configured" :: Text)])
  554                 (_, Nothing) ->
  555                   respond $ jsonResponse status503 (object ["error" .= ("No model configured" :: Text)])
  556                 (Just provider, Just model) -> do
  557                   result <- try @SomeException $ doCompletion env provider model userText transcriptPath
  558                   case result of
  559                     Left e -> do
  560                       _lh_logError (_fe_logger env) $ "Send error: " <> T.pack (show e)
  561                       respond $ jsonResponse status500 (object ["error" .= ("Completion failed" :: Text)])
  562                     Right respText ->
  563                       respond $ jsonResponse status200 (object ["response" .= respText])
  564 
  565 -- | Run a completion: load context, send to provider, record to transcript.
  566 doCompletion :: FrontendEnv -> SomeProvider -> ModelId -> Text -> FilePath -> IO Text
  567 doCompletion env provider model userText transcriptPath = do
  568   -- Open a transcript handle for recording
  569   th <- mkFileTranscriptHandle (_fe_logger env) transcriptPath
  570   -- Load recent messages for context
  571   history <- loadRecentMessages th 50 100000
  572   -- Check for per-session custom prompt, falling back to global
  573   let sessionDir = takeDirectory transcriptPath
  574       customPromptPath = sessionDir </> "custom-prompt.md"
  575   customExists <- doesFileExist customPromptPath
  576   systemPrompt <- if customExists
  577     then Just <$> TIO.readFile customPromptPath
  578     else pure (_fe_systemPrompt env)
  579   -- Build context with system prompt and history
  580   let ctx0 = foldl (flip addMessage) (emptyContext systemPrompt) history
  581       userMsg = textMessage User userText
  582       ctx = addMessage userMsg ctx0
  583       -- Wrap provider with transcript logging
  584       provider' = mkTranscriptProvider th (unModelId model) provider
  585       req = CompletionRequest
  586         { _cr_model        = model
  587         , _cr_messages     = contextMessages ctx
  588         , _cr_systemPrompt = contextSystemPrompt ctx
  589         , _cr_maxTokens    = Just 4096
  590         , _cr_tools        = []
  591         , _cr_toolChoice   = Nothing
  592         }
  593   resp <- complete provider' req
  594   _th_flush th
  595   _th_close th
  596   pure (responseText resp)
  597 
  598 -- | Consume the full request body.
  599 consumeBody :: Request -> IO LBS.ByteString
  600 consumeBody req = LBS.fromChunks <$> go
  601   where
  602     go = do
  603       chunk <- getRequestBodyChunk req
  604       if BS.null chunk then pure [] else (chunk :) <$> go
  605 
  606 -- | JSON 404 response.
  607 respondNotFound :: (Response -> IO ResponseReceived) -> IO ResponseReceived
  608 respondNotFound respond =
  609   respond $ jsonResponse status404 (object ["error" .= ("Not found" :: Text)])
  610 
  611 -- | Build a JSON response.
  612 jsonResponse :: ToJSON a => Status -> a -> Response
  613 jsonResponse st body =
  614   responseLBS st
  615     [ (hContentType, "application/json")
  616     , ("Access-Control-Allow-Origin", "*")
  617     ]
  618     (Aeson.encode body)
