    1 -- |
    2 -- Module      : PureClaw.Routing.Config
    3 -- Description : TOML loader for 'RoutingConfig' (Tabbed Chat WU3).
    4 --
    5 -- Reads the @[routing]@ section of @~\/.pureclaw\/config.toml@ and
    6 -- projects it onto a 'RoutingConfig'. Missing keys fall back to the
    7 -- field values in 'defaultRoutingConfig'.
    8 --
    9 -- == Scope of WU3
   10 --
   11 --   * 'defaultRoutingConfig' — the source-of-truth defaults; re-used by
   12 --     tests so the field values are not duplicated.
   13 --   * 'loadRoutingConfig' — best-effort TOML loader. Mirrors the
   14 --     forgiving semantics of 'PureClaw.CLI.Config.loadFileConfig':
   15 --     unreadable file or unparseable TOML silently falls back to
   16 --     'defaultRoutingConfig' rather than crashing process start.
   17 --   * 'loadRoutingConfigFromFile' — same as 'loadRoutingConfig' but
   18 --     takes an explicit path (so tests can point at a fixture file
   19 --     without depending on @\$HOME@).
   20 --   * 'routingConfigCodec' — exported for tests of round-tripping
   21 --     behaviour and for a future diagnostic loader.
   22 --
   23 -- See @docs\/tabbed-chat.md@ §\"RoutingConfig\" for the per-field
   24 -- semantics.
   25 module PureClaw.Routing.Config
   26   ( -- * Defaults
   27     defaultRoutingConfig
   28     -- * Loaders
   29   , loadRoutingConfig
   30   , loadRoutingConfigFromFile
   31     -- * Codec

   33   ) where

   94 -- | A partial 'RoutingConfig' with every field optional. The TOML
   95 -- codec produces this; 'overlayRoutingConfig' folds it onto
   96 -- 'defaultRoutingConfig' so missing keys simply fall through.
   97 data PartialRoutingConfig = PartialRoutingConfig
   98   { _prc_defaultKind         :: !(Maybe Tab.TabKind)
   99   , _prc_defaultAi           :: !(Maybe AiDefaults)
  100   , _prc_defaultShell        :: !(Maybe ShellDefaults)
  101   , _prc_switchRecap         :: !(Maybe Int)
  102   , _prc_maxTabs             :: !(Maybe Int)
  103   , _prc_inputQueueBound     :: !(Maybe Int)
  104   , _prc_channelOutQBound    :: !(Maybe Int)
  105   , _prc_spawnRateLimit      :: !(Maybe Int)
  106   , _prc_maxConcurrentActive :: !(Maybe Int)
  107   , _prc_maxNameLen          :: !(Maybe Int)
  108   , _prc_sshIdentityKey      :: !(Maybe Text)
  109   }
  110 
  111 -- | Fold a partial config onto the defaults.
  112 overlayRoutingConfig :: PartialRoutingConfig -> RoutingConfig
  113 overlayRoutingConfig p = RoutingConfig
  114   { _rc_defaultKind         = pick _rc_defaultKind         _prc_defaultKind
  115   , _rc_defaultAi           = pick _rc_defaultAi           _prc_defaultAi
  116   , _rc_defaultShell        = pick _rc_defaultShell        _prc_defaultShell
  117   , _rc_switchRecap         = pick _rc_switchRecap         _prc_switchRecap
  118   , _rc_maxTabs             = pick _rc_maxTabs             _prc_maxTabs
  119   , _rc_inputQueueBound     = pick _rc_inputQueueBound     _prc_inputQueueBound
  120   , _rc_channelOutQBound    = pick _rc_channelOutQBound    _prc_channelOutQBound
  121   , _rc_spawnRateLimit      = pick _rc_spawnRateLimit      _prc_spawnRateLimit
  122   , _rc_maxConcurrentActive = pick _rc_maxConcurrentActive _prc_maxConcurrentActive
  123   , _rc_maxNameLen          = pick _rc_maxNameLen          _prc_maxNameLen
  124   , _rc_sshIdentityKey      = pick _rc_sshIdentityKey      _prc_sshIdentityKey
  125   }
  126   where
  127     pick :: (RoutingConfig -> a) -> (PartialRoutingConfig -> Maybe a) -> a
  128     pick getDef getMb = case getMb p of
  129       Just v  -> v
  130       Nothing -> getDef defaultRoutingConfig

  137 -- | TOML codec for a 'TabKind' encoded as a snake_case string. Used
  138 -- for the @default_kind@ field.
  139 tabKindCodec :: Toml.Key -> TomlCodec Tab.TabKind
  140 tabKindCodec = Toml.textBy showKind parseKind
  141   where
  142     showKind :: Tab.TabKind -> Text
  143     showKind k = case k of
  144       Tab.KindAi      -> "ai"
  145       Tab.KindHarness -> "harness"
  146       Tab.KindShell   -> "shell"
  147       Tab.KindSsh     -> "ssh"
  148       Tab.KindTmux    -> "tmux"
  149 
  150     parseKind :: Text -> Either Text Tab.TabKind
  151     parseKind t = case t of
  152       "ai"      -> Right Tab.KindAi
  153       "harness" -> Right Tab.KindHarness
  154       "shell"   -> Right Tab.KindShell
  155       "ssh"     -> Right Tab.KindSsh
  156       "tmux"    -> Right Tab.KindTmux
  157       _         -> Left ("Unknown tab kind: " <> t)
  158 
  159 -- | TOML codec for 'AiDefaults' (the @[default_ai]@ sub-table inside
  160 -- the @[routing]@ section).
  161 aiDefaultsCodec :: TomlCodec AiDefaults
  162 aiDefaultsCodec = AiDefaults
  163   <$> Toml.diwrap (Toml.text "provider") .= _aid_providerId
  164   <*> Toml.diwrap (Toml.text "model")    .= _aid_modelId
  165 
  166 -- | TOML codec for 'ShellDefaults' (the @[default_shell]@ sub-table).
  167 shellDefaultsCodec :: TomlCodec ShellDefaults
  168 shellDefaultsCodec = ShellDefaults
  169   <$> Toml.text "command" .= _sd_command
  170 
  171 -- | TOML codec for a 'PartialRoutingConfig'. Used internally by
  172 -- 'loadRoutingConfigFromFile'.
  173 partialRoutingCodec :: TomlCodec PartialRoutingConfig
  174 partialRoutingCodec = PartialRoutingConfig
  175   <$> Toml.dioptional (tabKindCodec "default_kind")                   .= _prc_defaultKind
  176   <*> Toml.dioptional (Toml.table aiDefaultsCodec "default_ai")       .= _prc_defaultAi
  177   <*> Toml.dioptional (Toml.table shellDefaultsCodec "default_shell") .= _prc_defaultShell
  178   <*> Toml.dioptional (Toml.int  "switch_recap")                      .= _prc_switchRecap
  179   <*> Toml.dioptional (Toml.int  "max_tabs")                          .= _prc_maxTabs
  180   <*> Toml.dioptional (Toml.int  "input_queue_bound")                 .= _prc_inputQueueBound
  181   <*> Toml.dioptional (Toml.int  "channel_out_q_bound")               .= _prc_channelOutQBound
  182   <*> Toml.dioptional (Toml.int  "spawn_rate_limit")                  .= _prc_spawnRateLimit
  183   <*> Toml.dioptional (Toml.int  "max_concurrent_active")             .= _prc_maxConcurrentActive
  184   <*> Toml.dioptional (Toml.int  "max_name_len")                      .= _prc_maxNameLen
  185   <*> Toml.dioptional (Toml.text "ssh_identity_key")                  .= _prc_sshIdentityKey
  186 
  187 -- | Bidirectional TOML codec for the full 'RoutingConfig'.
  188 --
  189 -- The wrapping 'Toml.table' (@[routing]@) is added by the loader, not
  190 -- here, so this codec works both as a top-level codec (for fixtures
  191 -- that omit the @[routing]@ wrapper) and inside 'Toml.table' (for
  192 -- production @config.toml@ files).
  193 --
  194 -- Implemented as a 'Toml.dimap' over 'partialRoutingCodec': decoding
  195 -- folds onto the defaults via 'overlayRoutingConfig'; encoding lifts
  196 -- every concrete field to 'Just' (round-tripping the full snapshot).
  197 routingConfigCodec :: TomlCodec RoutingConfig
  198 routingConfigCodec = Toml.dimap toPartial overlayRoutingConfig partialRoutingCodec
  199   where
  200     toPartial :: RoutingConfig -> PartialRoutingConfig
  201     toPartial rc = PartialRoutingConfig
  202       { _prc_defaultKind         = Just (_rc_defaultKind rc)
  203       , _prc_defaultAi           = Just (_rc_defaultAi rc)
  204       , _prc_defaultShell        = Just (_rc_defaultShell rc)
  205       , _prc_switchRecap         = Just (_rc_switchRecap rc)
  206       , _prc_maxTabs             = Just (_rc_maxTabs rc)
  207       , _prc_inputQueueBound     = Just (_rc_inputQueueBound rc)
  208       , _prc_channelOutQBound    = Just (_rc_channelOutQBound rc)
  209       , _prc_spawnRateLimit      = Just (_rc_spawnRateLimit rc)
  210       , _prc_maxConcurrentActive = Just (_rc_maxConcurrentActive rc)
  211       , _prc_maxNameLen          = Just (_rc_maxNameLen rc)
  212       , _prc_sshIdentityKey      = Just (_rc_sshIdentityKey rc)
  213       }

  220 -- | Load a 'RoutingConfig' from a single TOML file.
  221 --
  222 -- Best-effort: the file may not exist (in which case
  223 -- 'defaultRoutingConfig' is returned), or it may exist but fail to
  224 -- parse (same fallback). The expected layout is a @[routing]@ table at
  225 -- the top of the file, but a @[routing]@-less document (keys at top
  226 -- level) is also accepted to keep fixture files small.
  227 loadRoutingConfigFromFile :: FilePath -> IO RoutingConfig
  228 loadRoutingConfigFromFile path = do
  229   text <- try @IOError (TIO.readFile path)
  230   pure $ case text of
  231     Left  _    -> defaultRoutingConfig
  232     Right toml ->
  233       case Toml.decode (Toml.table partialRoutingCodec "routing") toml of
  234         Right p -> overlayRoutingConfig p
  235         Left _  ->
  236           -- Fall back to a top-level @routing@-less document.
  237           case Toml.decode partialRoutingCodec toml of
  238             Right p -> overlayRoutingConfig p
  239             Left _  -> defaultRoutingConfig
  240 
  241 -- | Load a 'RoutingConfig' from the standard PureClaw config location
  242 -- (@\<pureclawDir\>\/config.toml@). Returns 'defaultRoutingConfig' if
  243 -- the file is missing or unparseable.
  244 loadRoutingConfig :: FilePath -> IO RoutingConfig
  245 loadRoutingConfig pureclawDir =
  246   loadRoutingConfigFromFile (pureclawDir </> "config.toml")
