    1 -- |
    2 -- Module      : PureClaw.Routing.PromptRenderer
    3 -- Description : Channel-aware spawn-prompt UX (Tabbed Chat WU9, A4 \/ A6).
    4 --
    5 -- When the dispatcher needs to ask the user which 'TabKind' to spawn
    6 -- (because '/tab new' was issued without a kind argument and
    7 -- '_rc_defaultKind' is unset), it emits a short prompt via the channel.
    8 -- Different channels render this differently:
    9 --
   10 --   * Telegram surfaces an inline-keyboard with one button per supported
   11 --     kind ('ai', 'harness', 'shell', 'ssh', 'tmux').
   12 --   * Signal \/ CLI surface a plain-text reply enumerating the same
   13 --     options as numbered choices.
   14 --
   15 -- WU9 ships a single channel-dispatched renderer:
   16 -- 'mkDefaultPromptRenderer' picks the right format based on the
   17 -- supplied 'ChannelHandle'\'s capability hint ('_ch_streaming'); tests
   18 -- inject 'mkTextPromptRenderer' directly when they want to assert on a
   19 -- specific rendering.
   20 --
   21 -- == Why a record, not a typeclass
   22 --
   23 -- Mirrors the Handle pattern used elsewhere in PureClaw — concrete IO
   24 -- actions are easier to substitute in tests than typeclass dictionaries
   25 -- and produce no orphan-instance surface.
   26 --
   27 -- == Buffered payload (A6)
   28 --
   29 -- When '/N <payload>' is issued for a missing index with no default
   30 -- kind, the dispatcher buffers @payload@ alongside the prompt so the
   31 -- (later) kind-selection turn can enqueue @payload@ on the freshly
   32 -- spawned tab. The 'PromptRenderer' carries enough information for the
   33 -- channel to round-trip the index AND the payload back to the
   34 -- dispatcher in the subsequent user reply — for v1 we surface the
   35 -- index \/ payload in the prompt text so the user can re-issue
   36 -- '/tab new <kind> <payload>' verbatim. A v1.5+ refinement may add a
   37 -- proper callback-data layer; the renderer interface admits it without
   38 -- a breaking change.
   39 --
   40 -- See @docs\/tabbed-chat.md@ §"Auto-spawn behavior (A-series)" for the
   41 -- A4 \/ A6 contract and §"Channel emission via ChannelOut writer" for
   42 -- the dispatcher-side emission rules.
   43 module PureClaw.Routing.PromptRenderer

   47   , mkDefaultPromptRenderer
   48   , mkTextPromptRenderer

   61 -- | A record-of-actions abstraction over per-channel spawn-prompt
   62 -- rendering.
   63 --
   64 -- '_pr_renderSpawnPrompt' takes the requested tab index and an
   65 -- optional buffered payload (the @Just txt@ form of A6, where the
   66 -- user typed @\/N \<txt\>@) and returns the user-visible 'Text' to
   67 -- enqueue as a dispatcher banner.
   68 --
   69 -- The renderer does NOT itself touch the channel — the caller (the
   70 -- AutoSpawn module in WU9) drives 'PureClaw.Routing.Dispatcher's
   71 -- @emitDispatcherBanner@ with the rendered string. Keeping the
   72 -- rendering pure-text means tests can pin the exact wording without
   73 -- intercepting channel I\/O.
   74 newtype PromptRenderer = PromptRenderer
   75   { _pr_renderSpawnPrompt :: TabIndex -> Maybe Text -> Text
   76   }
   77 
   78 
   79 -- | Dispatch on the supplied 'ChannelHandle' to pick the renderer
   80 -- that best matches the channel's capabilities.
   81 --
   82 -- The current heuristic is intentionally coarse: a channel reporting
   83 -- '_ch_streaming = True' is treated as a CLI surface (text reply); a
   84 -- channel reporting '_ch_streaming = False' is treated as a Telegram-
   85 -- style surface (inline-keyboard prompt). Real Signal\/Telegram
   86 -- implementations also report '_ch_streaming = False', so both get the
   87 -- inline-keyboard rendering by default — which degrades gracefully to
   88 -- text when the channel layer cannot honour the markup.
   89 --
   90 -- WU11 may refine this dispatch (e.g. by adding a explicit channel-kind
   91 -- discriminator to 'ChannelHandle'); for WU9 the heuristic is
   92 -- sufficient to flip the A4 \/ A6 DoDs green.
   93 mkDefaultPromptRenderer :: ChannelHandle -> PromptRenderer
   94 mkDefaultPromptRenderer ch
   95   | _ch_streaming ch = mkTextPromptRenderer
   96   | otherwise        = mkInlineKeyboardPromptRenderer
   97

   99 -- | A plain-text renderer suitable for CLI \/ Signal: enumerates the
  100 -- supported tab kinds as numbered choices.
  101 --
  102 -- The wording is:
  103 --
  104 -- @
  105 -- Spawn /N as which kind? Reply with: /tab new \<kind\> [\<args\>]
  106 -- where \<kind\> is one of: ai, harness, shell, ssh, tmux
  107 -- @
  108 --
  109 -- @\/N@ in the first half is informational — it shows the user the
  110 -- slot the spawn will land at. The user does NOT type the index when
  111 -- replying (tmux packing model — see @docs\/tabbed-chat.md@).
  112 --
  113 -- When a buffered payload is present (A6), the prompt appends a
  114 -- secondary line so the user can re-issue the full command verbatim.
  115 mkTextPromptRenderer :: PromptRenderer
  116 mkTextPromptRenderer = PromptRenderer renderTextPrompt
  117 
  118 renderTextPrompt :: TabIndex -> Maybe Text -> Text
  119 renderTextPrompt idx mPayload =
  120   let nTxt = tShowIdx idx
  121       header =
  122         "Spawn /" <> nTxt
  123           <> " as which kind? Reply with: /tab new <kind> [<args>]"
  124           <> " where <kind> is one of: "
  125           <> renderKindsAsText
  126       payloadLine = case mPayload of
  127         Nothing -> ""
  128         Just p  -> "\nBuffered text: " <> p
  129   in header <> payloadLine
  130 
  131 -- | A renderer that emits a Telegram-friendly text body. v1 surfaces
  132 -- the inline-keyboard hint as plain text (the channel's underlying
  133 -- 'sendMessage' API discards the markup gracefully); WU11 may upgrade
  134 -- the rendering once 'ChannelHandle' carries an inline-keyboard seam.
  135 --
  136 -- The text body is intentionally compact (≤ 200 chars even with a
  137 -- buffered payload) so it fits well in a mobile chat bubble.
  138 mkInlineKeyboardPromptRenderer :: PromptRenderer
  139 mkInlineKeyboardPromptRenderer = PromptRenderer renderInlineKeyboardPrompt
  140 
  141 renderInlineKeyboardPrompt :: TabIndex -> Maybe Text -> Text
  142 renderInlineKeyboardPrompt idx mPayload =
  143   let nTxt = tShowIdx idx
  144       header =
  145         "Choose a kind for /" <> nTxt
  146           <> ": [ai] [harness] [shell] [ssh] [tmux] — "
  147           <> "reply /tab new <kind>"
  148       payloadLine = case mPayload of
  149         Nothing -> ""
  150         Just p  -> "\nBuffered text: " <> p
  151   in header <> payloadLine
  152 
  153 -- | Comma-separated list of the supported tab-kind keywords. Used by
  154 -- the text renderer; exposed so tests can pin the exact wording.
  155 renderKindsAsText :: Text
  156 renderKindsAsText = T.intercalate ", " ["ai", "harness", "shell", "ssh", "tmux"]
  157 
  158 -- | Format a 'TabIndex' as the user-facing decimal string. Internal
  159 -- helper that mirrors the dispatcher's own 'tShowIdx' (kept local to
  160 -- avoid an import cycle into 'Dispatcher').
  161 tShowIdx :: TabIndex -> Text
  162 tShowIdx = T.pack . show . unTabIndex
