    1 -- |
    2 -- Module      : PureClaw.Routing.ChannelOut
    3 -- Description : Single-writer channel-emission thread for Tabbed Chat (WU4).
    4 --
    5 -- A dedicated writer thread that consumes from
    6 -- 'PureClaw.Agent.Env._env_channelOutQ' and, for each
    7 -- @('OutputSource', 'ChannelEvent')@ pair, decides whether to emit to
    8 -- the underlying 'ChannelHandle' or drop the event based on the
    9 -- current '_env_focus'.
   10 --
   11 -- == Behaviour
   12 --
   13 -- * 'SrcDispatcher' events emit unconditionally (switch confirms,
   14 --   dashboards, errors, breadcrumbs).
   15 -- * 'SrcTab' @n@ events emit only when @_env_focus == Just n@;
   16 --   otherwise they are dropped.
   17 -- * Per the D5 mid-stream switch breadcrumb rule: on the FIRST drop of
   18 --   an AI-tab stream (identified by 'StreamId') the writer emits one
   19 --   @SrcDispatcher BannerLine "/N has new output - /N to view"@.
   20 --   Subsequent drops of the same 'StreamId' are silent. 'StreamEnd'
   21 --   cleans up the breadcrumb state.
   22 -- * Dropped 'FullMsg' events (from non-AI tabs) do NOT emit a
   23 --   breadcrumb — shell\/ssh output is line-oriented and per-line
   24 --   breadcrumbs would be noisy (per D5 backend rule).
   25 --
   26 -- == Public API
   27 --
   28 -- * 'runChannelOutThread' — the writer loop body (runs forever until
   29 --   asynchronously cancelled).
   30 -- * 'startChannelOut' — fork the loop via @_env_fork@ and return the
   31 --   resulting 'TabRunner'.
   32 -- * 'shouldEmit' — producer-side focus pre-check helper exposed so tab
   33 --   loops (WU6\/7\/8) can skip enqueue work when not focused (D4).
   34 -- * 'breadcrumbText' — public for test assertions; the exact wording
   35 --   required by D5.
   36 --
   37 -- See @docs\/tabbed-chat.md@ §"Channel emission via ChannelOut writer"
   38 -- and the D-series DoDs for the full design.
   39 module PureClaw.Routing.ChannelOut
   40   ( -- * Writer thread
   41     runChannelOutThread
   42   , startChannelOut
   43     -- * Producer-side helper (D4)
   44   , shouldEmit
   45     -- * Breadcrumb wording (exposed for tests)
   46   , breadcrumbText
   47     -- * Internal state (exposed for tests)
   48   , BreadcrumbState (..)
   49     -- * Test seam — drive one iteration with an explicit state ref.
   50     -- Exposed so unit tests can run the loop body a finite number of
   51     -- times without spawning a thread that would otherwise never
   52     -- return ('runChannelOutThread' loops forever).
   53   , oneStep
   54   , newBreadcrumbStateRef
   55   ) where

   71 -- ---------------------------------------------------------------------------
   72 -- BreadcrumbState
   73 -- ---------------------------------------------------------------------------
   74 
   75 -- | Per-'StreamId' state tracking whether the writer has already
   76 -- emitted a mid-stream-switch breadcrumb for that stream.
   77 --
   78 -- * 'Pending' — the stream's owner has been focused so far (the writer
   79 --   has emitted at least one event for this 'StreamId' without
   80 --   dropping any). The next drop will produce a breadcrumb.
   81 -- * 'Emitted' — a breadcrumb has already been emitted for this
   82 --   'StreamId'; subsequent drops are silent.
   83 --
   84 -- The map entry is removed when the writer observes 'StreamEnd' for
   85 -- the stream (regardless of emit\/drop outcome).
   86 data BreadcrumbState
   87   = Pending
   88   | Emitted
   89   deriving stock (Eq, Show)
   90 
   91 
   92 -- ---------------------------------------------------------------------------
   93 -- Producer-side helper (D4)
   94 -- ---------------------------------------------------------------------------
   95 
   96 -- | Decide whether a producer should bother enqueueing a channel event
   97 -- given the current focus.
   98 --
   99 -- This is an OPTIMISATION, not the authoritative focus gate: the
  100 -- writer thread re-applies the same predicate on dequeue (D3). Tab
  101 -- loops use this to avoid building large 'ChannelEvent' values for
  102 -- non-focused streams (e.g. a fast-streaming non-focused tab would
  103 -- otherwise saturate the bounded @_env_channelOutQ@).
  104 --
  105 -- The focus argument is the current value of @_env_focus@ (an
  106 -- @'IORef' ('Maybe' 'TabIndex')@); the source argument is the
  107 -- 'OutputSource' the producer is about to enqueue.
  108 --
  109 -- * 'SrcDispatcher' events are ALWAYS eligible to emit (dispatcher
  110 --   breadcrumbs, switch confirms, errors must never be dropped).
  111 -- * 'SrcTab' @n@ events are eligible only when the focus equals
  112 --   @'Just' n@.
  113 shouldEmit :: Maybe TabIndex -> OutputSource -> Bool
  114 shouldEmit _     SrcDispatcher = True
  115 shouldEmit focus (SrcTab n)    = focus == Just n

  118 -- ---------------------------------------------------------------------------
  119 -- Breadcrumb wording (D5)
  120 -- ---------------------------------------------------------------------------
  121 
  122 -- | The exact breadcrumb text emitted on the first drop of an AI-tab
  123 -- stream. Format: @"/N has new output - /N to view"@ where @N@ is the
  124 -- owner tab's index.
  125 --
  126 -- Exposed publicly so tests can pin the wording without re-deriving
  127 -- it. Status-neutral phrasing per D5.
  128 breadcrumbText :: TabIndex -> Text
  129 breadcrumbText idx =
  130   let nTxt = T.pack (show (unTabIndex idx))
  131   in "/" <> nTxt <> " has new output - /" <> nTxt <> " to view"

  134 -- ---------------------------------------------------------------------------
  135 -- Writer thread
  136 -- ---------------------------------------------------------------------------
  137 
  138 -- | Run the channel-out writer loop forever.
  139 --
  140 -- Repeatedly:
  141 --
  142 -- 1. Reads one @('OutputSource', 'ChannelEvent')@ pair off
  143 --    @_env_channelOutQ@ (STM-blocking).
  144 -- 2. Reads @_env_focus@ to decide emit vs drop.
  145 -- 3. Emits via the underlying 'ChannelHandle' or updates the
  146 --    breadcrumb state, per the D-series rules.
  147 --
  148 -- The loop never terminates on its own — it must be cancelled
  149 -- asynchronously via the 'TabRunner' returned by 'startChannelOut'.
  150 runChannelOutThread :: AgentEnv -> IO ()
  151 runChannelOutThread env = do
  152   stateRef <- newBreadcrumbStateRef
  153   forever (oneStep env stateRef)
  154 
  155 -- | Spawn the writer loop using the environment's @_env_fork@ test
  156 -- seam, returning the 'TabRunner' through which the caller can cancel
  157 -- or wait on the writer.
  158 --
  159 -- The writer is intended to be started once at dispatcher boot (WU5)
  160 -- and torn down via the same @bracket@ that cancels the per-tab
  161 -- runners.
  162 startChannelOut :: AgentEnv -> IO TabRunner
  163 startChannelOut env = _env_fork env (runChannelOutThread env)

  166 -- ---------------------------------------------------------------------------
  167 -- One iteration (factored out for clarity + testability)
  168 -- ---------------------------------------------------------------------------
  169 
  170 -- | Allocate a fresh, empty breadcrumb-state 'IORef'. Exposed so unit
  171 -- tests can drive 'oneStep' a finite number of times with their own
  172 -- state ref; production code paths obtain the ref implicitly via
  173 -- 'runChannelOutThread'.
  174 newBreadcrumbStateRef :: IO (IORef (Map StreamId BreadcrumbState))
  175 newBreadcrumbStateRef = newIORef Map.empty
  176 
  177 -- | Process exactly one queue entry.
  178 --
  179 -- Exposed as a test seam ('runChannelOutThread' loops forever and so
  180 -- is awkward to drive deterministically in a unit test). Production
  181 -- code paths invoke this transitively via 'runChannelOutThread'.
  182 oneStep :: AgentEnv -> IORef (Map StreamId BreadcrumbState) -> IO ()
  183 oneStep env stateRef = do
  184   (src, ev) <- atomically (readTBQueue (_env_channelOutQ env))
  185   focus <- readIORef (_env_focus env)
  186   case src of
  187     SrcDispatcher -> emitEvent (_env_channel env) ev
  188     SrcTab n
  189       | focus == Just n -> emitWithStreamTracking (_env_channel env) stateRef ev
  190       | otherwise       -> dropWithBreadcrumb (_env_channel env) stateRef n ev
  191 
  192 -- | Emit a 'ChannelEvent' that originated from the focused tab,
  193 -- updating the breadcrumb-state map for stream lifecycle bookkeeping
  194 -- (so that a later focus change mid-stream is correctly noticed).
  195 emitWithStreamTracking
  196   :: ChannelHandle
  197   -> IORef (Map StreamId BreadcrumbState)
  198   -> ChannelEvent
  199   -> IO ()
  200 emitWithStreamTracking ch stateRef ev = do
  201   case ev of
  202     StreamStart sid _ ->
  203       -- The owner tab is currently focused; register the stream as
  204       -- 'Pending' so a later drop emits exactly one breadcrumb.
  205       atomicModifyIORef' stateRef
  206         (\m -> (Map.insert sid Pending m, ()))
  207     StreamEnd sid ->
  208       -- GC the breadcrumb state on stream end (regardless of focus).
  209       atomicModifyIORef' stateRef
  210         (\m -> (Map.delete sid m, ()))
  211     _ -> pure ()
  212   emitEvent ch ev
  213 
  214 -- | Handle a dropped 'SrcTab' event (focus does not match the source
  215 -- tab). For stream-bearing events on an AI tab we may emit one
  216 -- breadcrumb; for 'FullMsg' (non-AI tabs) we always drop silently.
  217 dropWithBreadcrumb
  218   :: ChannelHandle
  219   -> IORef (Map StreamId BreadcrumbState)
  220   -> TabIndex
  221   -> ChannelEvent
  222   -> IO ()
  223 dropWithBreadcrumb ch stateRef owner ev =
  224   case ev of
  225     StreamStart sid _ -> handleStreamDrop ch stateRef owner sid
  226     ChunkOf sid _     -> handleStreamDrop ch stateRef owner sid
  227     StreamEnd sid     ->
  228       -- Drop AND clean up: even if the stream's body went unread the
  229       -- map entry must not leak.
  230       atomicModifyIORef' stateRef
  231         (\m -> (Map.delete sid m, ()))
  232     FullMsg{}   -> pure ()  -- non-AI tab drops never breadcrumb (D5).
  233     BannerLine{} -> pure () -- BannerLine should only ever arrive on
  234                             -- SrcDispatcher; defensive no-op here.
  235 
  236 -- | Common breadcrumb logic shared by 'StreamStart' and 'ChunkOf'
  237 -- drops. Emits a breadcrumb iff the stream has not already been
  238 -- breadcrumbed; updates the state map in lock-step with the emission
  239 -- decision.
  240 handleStreamDrop
  241   :: ChannelHandle
  242   -> IORef (Map StreamId BreadcrumbState)
  243   -> TabIndex
  244   -> StreamId
  245   -> IO ()
  246 handleStreamDrop ch stateRef owner sid = do
  247   shouldBreadcrumb <- atomicModifyIORef' stateRef $ \m ->
  248     case Map.lookup sid m of
  249       Just Emitted -> (m, False)
  250       _            -> (Map.insert sid Emitted m, True)
  251   when shouldBreadcrumb (emitEvent ch (BannerLine (breadcrumbText owner)))
  252 
  253 -- | Translate a 'ChannelEvent' into the underlying 'ChannelHandle'
  254 -- calls. 'StreamStart' is metadata for the writer's state machine and
  255 -- has no on-channel representation.
  256 emitEvent :: ChannelHandle -> ChannelEvent -> IO ()
  257 emitEvent ch ev = case ev of
  258   StreamStart{}   -> pure ()
  259   ChunkOf _ t     -> _ch_sendChunk ch (ChunkText t)
  260   StreamEnd _     -> _ch_sendChunk ch ChunkDone
  261   FullMsg _ t     -> _ch_send ch (OutgoingMessage t)
  262   BannerLine t    -> _ch_send ch (OutgoingMessage t)
