    1 -- |
    2 -- Module      : PureClaw.Tab.Backend
    3 -- Description : Backend tab factory — KindShell\/KindSsh\/KindTmux (Tabbed Chat WU8).
    4 --
    5 -- A backend tab is a 'PureClaw.Handles.Tab.TabHandle' that wraps a
    6 -- 'PureClaw.Handles.Backend.BackendHandle' constructed at spawn time
    7 -- from a tab kind and a list of user-supplied args. The tab owns two
    8 -- helper threads:
    9 --
   10 --   * a /drainer/ that loops on '_bh_recv' and emits each non-empty
   11 --     output chunk as @'FullMsg' !TabIndex !Text@ via '_env_channelOutQ'
   12 --     (focus-gated at the producer side per D4 and at the writer side
   13 --     per D3 — per the D5 backend rule, no streaming\/no breadcrumbs);
   14 --   * a /writer/ that drains a bounded outbound queue and forwards
   15 --     bytes to '_bh_send', keeping the public 'TabHandle._tabHandle_send'
   16 --     non-blocking per H4.
   17 --
   18 -- == Factory shape
   19 --
   20 -- @
   21 -- 'mkTabBackend' :: 'PureClaw.Agent.Env.AgentEnv'
   22 --                -> 'PureClaw.Handles.Tab.TabIndex'
   23 --                -> 'PureClaw.Handles.Tab.TabKind'
   24 --                -> [Text]
   25 --                -> IO (Either 'PureClaw.Handles.Tab.TabError'
   26 --                              'PureClaw.Handles.Tab.TabHandle')
   27 -- @
   28 --
   29 -- Dispatches on the supplied 'TabKind' to one of three sub-factories:
   30 --
   31 -- * 'KindShell' — parses @[Text]@ via 'PureClaw.Security.Command.authorize'
   32 --   against '_env_policy' (S1); on failure returns
   33 --   @'Left' ('TabSpawnAuthDenied' ...)@. Calls
   34 --   'PureClaw.Backend.Local.mkLocalBackendHandle' (via the injected
   35 --   '_bio_mkShell' seam).
   36 --
   37 -- * 'KindSsh' — parses the first arg as @user\@host@; constructs the host
   38 --   via 'PureClaw.Backend.SSH.mkSshHost' (S2\/S3 — rejects whitespace,
   39 --   leading dashes, NUL bytes, and shell metachars). Parses the remaining
   40 --   args as the remote command via
   41 --   'PureClaw.Backend.SSH.authorizeRemote'. Sources the SSH identity from
   42 --   the Vault slot named by '_rc_sshIdentityKey' (S4 — NO inline
   43 --   identity acceptance); missing slot yields
   44 --   @'Left' ('TabSpawnAuthDenied' ...)@.
   45 --
   46 -- * 'KindTmux' — parses one @session[:window[.pane]]@ argument via the
   47 --   'PureClaw.Backend.Tmux.mkTmuxSession'\/'mkTmuxWindow'\/'mkTmuxPane'
   48 --   smart constructors (S3); the local @tmux@ binary is authorized via
   49 --   'PureClaw.Security.Command.authorize'.
   50 --
   51 -- == Slash commands are unsupported (H13 \/ I4)
   52 --
   53 -- '_tabHandle_enqueueSlash' returns @'Left' ('TabUnsupportedCommand'
   54 -- cmd)@ immediately without enqueueing — backend tabs do not have a
   55 -- per-tab 'Context' or slash-command surface. Direct-inject text
   56 -- (@\/N \<text\>@) is forwarded verbatim to the backend via the writer
   57 -- thread so a payload like @\/N \/pwd@ reaches the backend as the
   58 -- literal @\"\/pwd\"@ (I4 — slash-prefix opaque to backend).
   59 --
   60 -- == Close semantics (H8 \/ H9)
   61 --
   62 -- Both 'CloseGraceful' and 'CloseForce' are destructive for backend
   63 -- tabs: they cancel both helper threads and call '_bh_close' on the
   64 -- underlying 'BackendHandle'. There is no archive equivalent. The close
   65 -- path is idempotent + never-throws (H6 \/ H7 contract).
   66 --
   67 -- == Async exception discipline
   68 --
   69 -- Helper threads catch 'SomeException' EXCEPT 'AsyncCancelled', which
   70 -- propagates so '_tabHandle_close' (which cancels the helpers'
   71 -- 'TabRunner's) can unblock a stuck '_bh_recv' or pending writer.
   72 -- Mirrors the WU6 \/ WU7 'safelyRun' pattern.
   73 --
   74 -- == Test seam ('BackendIO')
   75 --
   76 -- Production code calls 'mkTabBackend', which uses 'realBackendIO'
   77 -- (which in turn calls the real WU8\/WU9\/WU10 backend factories from
   78 -- @PureClaw.Backend.*@). Tests call 'mkTabBackendWith' with a fake
   79 -- 'BackendIO' that returns an in-memory 'BackendHandle' (via
   80 -- 'PureClaw.Handles.Backend.mkInMemoryBackendHandle') so spawn paths
   81 -- are exercised end-to-end without forking real subprocesses.
   82 --
   83 -- See @docs\/tabbed-chat.md@ §"H-series" (close lifecycle), §"I4"
   84 -- (opaque slash-prefix), §"S1"-"S4" (spawn authorization), and §"D5"
   85 -- (FullMsg emission).

  175 -- BackendTabState — per-tab mutable state, private to the factory closure

  177 
  178 -- | Per-backend-tab mutable state. Fields are exposed so a future
  179 -- diagnostic @tab info@ handler can inspect them; production code never
  180 -- touches the record outside this module.
  181 data BackendTabState = BackendTabState
  182   { _bts_backend       :: !BackendHandle
  183     -- ^ The underlying 'BackendHandle' constructed at spawn time. The
  184     -- tab owns its lifecycle; '_tabHandle_close' calls '_bh_close' on
  185     -- the first invocation (H8 — destructive, no archive equivalent).
  186   , _bts_sendQ         :: !(TBQueue ByteString)
  187     -- ^ Bounded outbound queue. '_tabHandle_send' enqueues here
  188     -- non-blockingly; the writer thread drains and calls '_bh_send'.
  189     -- Bounded by '_rc_inputQueueBound' so a saturating producer
  190     -- surfaces 'TabConcurrencyLimit' rather than blocking the
  191     -- dispatcher (H4).
  192   , _bts_drainRunner   :: !(IORef (Maybe TabRunner))
  193     -- ^ Drainer thread runner (output side: '_bh_recv' →
  194     -- 'FullMsg'). Filled by 'mkTabBackendWith' post-fork; stays
  195     -- 'Nothing' for the brief window between allocation and fork-fill.
  196   , _bts_writerRunner  :: !(IORef (Maybe TabRunner))
  197     -- ^ Writer thread runner (input side: '_bts_sendQ' →
  198     -- '_bh_send'). Filled by 'mkTabBackendWith' post-fork.
  199   , _bts_closed        :: !(IORef Bool)
  200     -- ^ Idempotency flag for '_tabHandle_close'. Flipped to 'True' on
  201     -- the first invocation; subsequent invocations are no-ops (H6).
  202   , _bts_statusRef     :: !(IORef TabStatus)
  203     -- ^ Held status backing '_tabHandle_status'. Starts 'Idle' at
  204     -- spawn and transitions to 'Crashed' if a helper thread hits a
  205     -- synchronous exception.
  206   }

  209 -- ---------------------------------------------------------------------------
  210 -- BackendIO seam
  211 -- ---------------------------------------------------------------------------
  212 
  213 -- | Injection seam for the three real backend factories.
  214 --
  215 -- Production callers use 'mkTabBackend', which wires this to
  216 -- 'realBackendIO' (which calls the real WU8\/WU9\/WU10 factories from
  217 -- @PureClaw.Backend.*@). Tests call 'mkTabBackendWith' with a fake
  218 -- 'BackendIO' that returns an in-memory 'BackendHandle' (via
  219 -- 'PureClaw.Handles.Backend.mkInMemoryBackendHandle'), so the spawn,
  220 -- drainer, and writer paths can be exercised end-to-end without
  221 -- forking real subprocesses.
  222 --
  223 -- Note: every sub-factory still receives a fully-validated payload —
  224 -- the smart constructors ('authorize', 'mkSshHost', 'authorizeRemote',
  225 -- 'mkTmuxSession' \/ 'mkTmuxWindow' \/ 'mkTmuxPane') are invoked by
  226 -- 'mkTabBackendWith' BEFORE the seam is called, so the seam is given a
  227 -- security-validated value (the seam is not a way to skip validation).
  228 data BackendIO = BackendIO
  229   { _bio_mkShell
  230       :: AuthorizedCommand
  231       -> IO (Either BackendError BackendHandle)
  232     -- ^ Spawn a local 'Pipe'-kind backend. Production: thin wrapper
  233     -- around 'PureClaw.Backend.Local.mkLocalBackendHandle' with the
  234     -- default 'PureClaw.Handles.Backend.PipeOpts'.
  235   , _bio_mkSsh
  236       :: SSH.SshTarget
  237       -> AuthorizedCommand
  238       -> SSH.RemoteCommand
  239       -> IO (Either BackendError BackendHandle)
  240     -- ^ Spawn an SSH backend. Production: calls
  241     -- 'PureClaw.Backend.SSH.mkSshBackendHandle' with 'realPtyIO' and
  242     -- the default 'PureClaw.Backend.SSH.SshOpts'.
  243   , _bio_mkTmux
  244       :: Tmux.SshLocation
  245       -> Tmux.TmuxTarget
  246       -> IO (Either BackendError BackendHandle)
  247     -- ^ Spawn a tmux Attach backend. Production: calls
  248     -- 'PureClaw.Backend.Tmux.mkTmuxBackendHandle' with 'realPtyIO'
  249     -- and the default 'PureClaw.Backend.Tmux.TmuxOpts'.
  250   }
  251 
  252 -- | Production 'BackendIO': calls the real backend factories from
  253 -- @PureClaw.Backend.*@.
  254 --
  255 -- The SSH and tmux paths use 'realPtyIO' (the @posix-pty@-backed PTY
  256 -- allocator); the local-shell path uses
  257 -- 'PureClaw.Backend.Local.mkLocalBackendHandle' which is pipe-only and
  258 -- needs no PtyIO.
  259 realBackendIO :: BackendIO
  260 realBackendIO = BackendIO
  261   { _bio_mkShell = (`Local.mkLocalBackendHandle` defaultPipeOpts)
  262   , _bio_mkSsh = \tgt sshCmd remote ->
  263       SSH.mkSshBackendHandle realPtyIO sshCmd tgt remote SSH.defaultSshOpts
  264   , _bio_mkTmux = \loc tgt ->
  265       Tmux.mkTmuxBackendHandle realPtyIO loc tgt Tmux.defaultTmuxOpts
  266   }

  269 -- ---------------------------------------------------------------------------
  270 -- mkTabBackend — the backend tab factory
  271 -- ---------------------------------------------------------------------------
  272 
  273 -- | Construct a backend tab (KindShell \/ KindSsh \/ KindTmux).
  274 --
  275 -- Calls 'mkTabBackendWith' with 'realBackendIO'. The factory itself
  276 -- never throws: it always returns @'Right' h@ or @'Left' e@. Failures
  277 -- inside the forked helper threads are caught by their own outer
  278 -- exception handler and surface as a 'Crashed' status (visible via
  279 -- '_tabHandle_status').
  280 mkTabBackend
  281   :: AgentEnv
  282   -> TabIndex
  283   -> TabKind
  284   -> [Text]
  285   -> IO (Either TabError TabHandle)
  286 mkTabBackend = mkTabBackendWith realBackendIO
  287 
  288 -- | Construct a backend tab with a caller-supplied 'BackendIO'. Test
  289 -- seam. See 'mkTabBackend' for the production entry point.
  290 --
  291 -- Dispatches on 'TabKind' to one of three sub-factories. Each
  292 -- sub-factory:
  293 --
  294 -- 1. Runs the kind-specific smart constructors (per S1\/S2\/S3\/S4) on
  295 --    the user-supplied @[Text]@ args. On rejection returns
  296 --    @'Left' ('TabSpawnAuthDenied' ...)@.
  297 -- 2. Calls the corresponding seam action ('_bio_mkShell',
  298 --    '_bio_mkSsh', '_bio_mkTmux') with the validated payload. On
  299 --    'Left' returns @'Left' ('TabBackendConstructFailed' ...)@.
  300 -- 3. On 'Right' wraps the resulting 'BackendHandle' in per-tab state
  301 --    and forks the drainer + writer helper threads via '_env_fork'.
  302 --
  303 -- 'KindAi' and 'KindHarness' are rejected with
  304 -- @'Left' ('TabUnsupportedCommand' ...)@ — those kinds have their own
  305 -- factories ('PureClaw.Tab.Ai.mkTabAi' and
  306 -- 'PureClaw.Tab.Harness.mkTabHarness').
  307 mkTabBackendWith
  308   :: BackendIO
  309   -> AgentEnv
  310   -> TabIndex
  311   -> TabKind
  312   -> [Text]
  313   -> IO (Either TabError TabHandle)
  314 mkTabBackendWith bio env idx kind args = case kind of
  315   KindShell -> mkShellTab bio env idx args
  316   KindSsh   -> mkSshTab   bio env idx args
  317   KindTmux  -> mkTmuxTab  bio env idx args
  318   -- These two kinds are handled by Tab.Ai / Tab.Harness; routing them
  319   -- through Tab.Backend is a programmer error in the dispatcher.
  320   KindAi -> pure (Left (TabSpawnAuthDenied PublicAuthError))
  321   KindHarness -> pure (Left (TabSpawnAuthDenied PublicAuthError))

  324 -- ---------------------------------------------------------------------------
  325 -- KindShell sub-factory (S1)
  326 -- ---------------------------------------------------------------------------
  327 
  328 -- | Spawn a local-shell backend tab.
  329 --
  330 -- Args layout: @[program, arg1, arg2, ...]@ — the first element is the
  331 -- local command path; the rest are passed verbatim as the argv tail.
  332 -- Empty arg list is rejected (no command to run).
  333 --
  334 -- Authorization (S1): the program path is passed to
  335 -- 'PureClaw.Security.Command.authorize' against '_env_policy' BEFORE
  336 -- the seam is invoked. Rejection yields
  337 -- @'Left' ('TabSpawnAuthDenied' ...)@; no subprocess is spawned (the
  338 -- factory short-circuits at the smart-constructor call).
  339 mkShellTab
  340   :: BackendIO -> AgentEnv -> TabIndex -> [Text]
  341   -> IO (Either TabError TabHandle)
  342 mkShellTab bio env idx args = case args of
  343   [] -> pure (Left (TabSpawnAuthDenied PublicAuthError))
  344   (prog : rest) -> case authorize (_env_policy env) (T.unpack prog) rest of
  345     Left _err -> authDenied
  346     Right authCmd -> do
  347       let nameRaw = defaultBackendName "shell" args
  348       case Parse.sanitizeTabName nameRaw of
  349         Left nameErr -> pure (Left (TabInvalidName nameErr))
  350         Right nameTxt -> do
  351           mkResult <- _bio_mkShell bio authCmd
  352           finishSpawn env idx (TabName nameTxt) KindShell mkResult

  355 -- ---------------------------------------------------------------------------
  356 -- KindSsh sub-factory (S2 \/ S3 \/ S4)
  357 -- ---------------------------------------------------------------------------
  358 
  359 -- | Spawn an SSH backend tab.
  360 --
  361 -- Args layout: @[\"user\@host\", program, arg1, ...]@. The first arg is
  362 -- @user\@host@; the remaining args are the remote command (program +
  363 -- argv).
  364 --
  365 -- Validation pipeline:
  366 --
  367 -- 1. The first arg must contain exactly one @\@@ separating a
  368 --    non-empty user from a non-empty host.
  369 -- 2. The host portion passes through
  370 --    'PureClaw.Backend.SSH.mkSshHost' (S2 \/ S3 — rejects whitespace,
  371 --    leading dashes, NUL bytes, shell metacharacters, etc.).
  372 -- 3. The local @ssh@ binary path is hardcoded to @\"ssh\"@ (the
  373 --    'PureClaw.Security.Command.authorize' check confirms it is in
  374 --    '_sp_allowedCommands').
  375 -- 4. The remote command (head + tail) is authorized via
  376 --    'PureClaw.Backend.SSH.authorizeRemote' against
  377 --    '_sp_allowedRemoteCommands' (S2 — independent allowlist).
  378 -- 5. The SSH identity is sourced from the Vault slot named by
  379 --    '_rc_sshIdentityKey' (S4 — NO inline identity acceptance). The
  380 --    Vault returns raw bytes; we materialise them at a 'SafeKeyPath'
  381 --    via 'PureClaw.Security.Path.mkSafeKeyPath'.
  382 --
  383 -- /Implementation note (v1 minimal):/ the v1 Vault → 'SafeKeyPath'
  384 -- translation is currently a missing-feature stub. v1 requires the
  385 -- caller to supply the key path via configuration rather than the
  386 -- Vault payload; the Vault slot lookup happens for S4 audit
  387 -- compliance (the missing-slot path yields
  388 -- @'Left' ('TabSpawnAuthDenied' ...)@ as the design mandates) and
  389 -- the path materialisation is deferred to WU9. For WU8 we surface a
  390 -- clean PublicError; the production path lands in WU9 alongside the
  391 -- auto-spawn UX.
  392 mkSshTab
  393   :: BackendIO -> AgentEnv -> TabIndex -> [Text]
  394   -> IO (Either TabError TabHandle)
  395 mkSshTab bio env idx args = case args of
  396   [] -> authDenied
  397   (userHost : remoteCmd) ->
  398     case parseUserAtHost userHost of
  399       Left _ -> authDenied
  400       Right (user, hostText) ->
  401         case SSH.mkSshHost hostText of
  402           Left _ -> authDenied
  403           Right host -> case remoteCmd of
  404             [] -> authDenied
  405             (remoteProg : remoteArgs) ->
  406               case SSH.authorizeRemote (_env_policy env) (T.unpack remoteProg) remoteArgs of
  407                 Left _ -> authDenied
  408                 Right remote ->
  409                   case authorize (_env_policy env) "ssh" [] of
  410                     Left _ -> authDenied
  411                     Right sshCmd -> do
  412                       mIdent <- resolveSshIdentity env
  413                       case mIdent of
  414                         Left _ -> authDenied
  415                         Right identityPath -> do
  416                           let tgt = SSH.SshTarget
  417                                 { SSH._st_user     = user
  418                                 , SSH._st_host     = host
  419                                 , SSH._st_port     = Nothing
  420                                 , SSH._st_identity = identityPath
  421                                 }
  422                               nameRaw = "ssh tab"
  423                           case Parse.sanitizeTabName nameRaw of
  424                             Left nameErr -> pure (Left (TabInvalidName nameErr))
  425                             Right nameTxt -> do
  426                               mkResult <- _bio_mkSsh bio tgt sshCmd remote
  427                               finishSpawn env idx (TabName nameTxt) KindSsh mkResult

  429 -- | Parse a @user\@host@ string into its components.
  430 --
  431 -- Accepts exactly one @\@@ separating a non-empty user from a non-empty
  432 -- host. Multi-@\@@ inputs (e.g. @\"a\@b\@c\"@) are rejected as
  433 -- malformed — there is no scenario where a user identifier legitimately
  434 -- contains an @\@@.
  435 parseUserAtHost :: Text -> Either () (Text, Text)
  436 parseUserAtHost t = case T.splitOn "@" t of
  437   [u, h] | not (T.null u) && not (T.null h) -> Right (u, h)
  438   _                                          -> Left ()
  439 
  440 -- | Look up the SSH identity from the configured Vault slot.
  441 --
  442 -- Returns @'Right' path@ when the slot exists and the byte payload was
  443 -- successfully materialised to a 'SafeKeyPath'; otherwise 'Left'.
  444 --
  445 -- The v1 path-materialisation step is intentionally a clean failure:
  446 -- 'PureClaw.Security.Path.mkSafeKeyPath' requires a 'KeysRoot' and an
  447 -- on-disk file with mode @0400@, which the v1 Vault payload doesn't
  448 -- yet produce. The WU9 work item that delivers production SSH spawns
  449 -- is the right place to add the materialisation (write-to-temp +
  450 -- chmod + 'mkSafeKeyPath'). For WU8 we focus on the S4 audit invariant
  451 -- (the slot lookup) and surface @'Left' ('TabSpawnAuthDenied' ...)@
  452 -- when materialisation can't complete.
  453 resolveSshIdentity :: AgentEnv -> IO (Either () SafeKeyPath)
  454 resolveSshIdentity env = do
  455   let slot = _rc_sshIdentityKey (_env_routingConfig env)
  456   mVault <- readIORef (_env_vault env)
  457   case mVault of
  458     Nothing    -> pure (Left ())  -- S4: no vault configured
  459     Just vault -> do
  460       r <- _vh_get vault slot
  461       case r of
  462         Left _      -> pure (Left ())  -- S4: missing slot
  463         Right _bytes ->
  464           -- v1 minimal: surface a clean S4-shaped failure rather than
  465           -- materialising the byte payload to disk. The Vault lookup
  466           -- itself happened (S4 audit invariant satisfied — we did
  467           -- read the slot named by _rc_sshIdentityKey). The on-disk
  468           -- materialisation (write-to-temp + chmod + 'mkSafeKeyPath')
  469           -- lands in WU9 alongside the production SSH spawn UX.
  470           pure (Left ())

  473 -- ---------------------------------------------------------------------------
  474 -- KindTmux sub-factory (S3)
  475 -- ---------------------------------------------------------------------------
  476 
  477 -- | Spawn a tmux Attach backend tab.
  478 --
  479 -- Args layout: @[\"session:window[.pane]\"]@. The first arg encodes
  480 -- the tmux target; remaining args are ignored (tmux Attach has no
  481 -- argv tail — the attach key spec is implicit).
  482 --
  483 -- Validation (S3):
  484 --
  485 -- 1. The single arg is split on @\":\"@ and @\".\"@ to extract
  486 --    session, window, and optional pane.
  487 -- 2. Each component passes through its corresponding smart constructor
  488 --    ('mkTmuxSession', 'mkTmuxWindow', 'mkTmuxPane') which reject
  489 --    empty inputs, leading dashes, NUL bytes, non-ASCII, and
  490 --    out-of-charset characters.
  491 -- 3. The local @tmux@ binary is authorized via 'authorize' (with the
  492 --    static @\"tmux\"@ program path).
  493 mkTmuxTab
  494   :: BackendIO -> AgentEnv -> TabIndex -> [Text]
  495   -> IO (Either TabError TabHandle)
  496 mkTmuxTab bio env idx args = case args of
  497   [] -> pure (Left (TabSpawnAuthDenied PublicAuthError))
  498   (targetSpec : _ignored) ->
  499     case parseTmuxTarget targetSpec of
  500       Left _ -> pure (Left (TabSpawnAuthDenied PublicAuthError))
  501       Right tgt ->
  502         case authorize (_env_policy env) "tmux" [] of
  503           Left _ -> authDenied
  504           Right tmuxCmd -> do
  505             let loc = Tmux.LocalHost tmuxCmd
  506                 nameRaw = "tmux tab"
  507             case Parse.sanitizeTabName nameRaw of
  508               Left nameErr -> pure (Left (TabInvalidName nameErr))
  509               Right nameTxt -> do
  510                 mkResult <- _bio_mkTmux bio loc tgt
  511                 finishSpawn env idx (TabName nameTxt) KindTmux mkResult
  512 
  513 -- | Parse a tmux target spec of the form
  514 -- @\"session\"@, @\"session:window\"@, or @\"session:window.pane\"@.
  515 --
  516 -- Each component is validated via its smart constructor; on any
  517 -- rejection the entire parse fails. The window component is required
  518 -- (a bare session has no attachable window in tmux's Attach-mode
  519 -- model); the pane component is optional.
  520 parseTmuxTarget :: Text -> Either InvalidOptionDetail Tmux.TmuxTarget
  521 parseTmuxTarget spec = case T.splitOn ":" spec of
  522   [sessTxt, winSpec] -> do
  523     session <- Tmux.mkTmuxSession sessTxt
  524     case T.splitOn "." winSpec of
  525       [winTxt] -> do
  526         window <- Tmux.mkTmuxWindow winTxt
  527         Right Tmux.TmuxTarget
  528           { Tmux._tt_session = session
  529           , Tmux._tt_window  = window
  530           , Tmux._tt_pane    = Nothing
  531           }
  532       [winTxt, paneTxt] -> do
  533         window <- Tmux.mkTmuxWindow winTxt
  534         pane   <- Tmux.mkTmuxPane paneTxt
  535         Right Tmux.TmuxTarget
  536           { Tmux._tt_session = session
  537           , Tmux._tt_window  = window
  538           , Tmux._tt_pane    = Just pane
  539           }
  540       _ -> Left (InvalidOptionDetail "tmux target: malformed window.pane")
  541   _ -> Left (InvalidOptionDetail "tmux target: expected session:window[.pane]")

  544 -- ---------------------------------------------------------------------------
  545 -- Shared spawn finishing logic (drainer + writer)
  546 -- ---------------------------------------------------------------------------
  547 
  548 -- | Wrap a freshly-constructed 'BackendHandle' in per-tab state, fork
  549 -- the drainer + writer threads, and build the public 'TabHandle'.
  550 --
  551 -- Shared by all three sub-factories (KindShell \/ KindSsh \/ KindTmux)
  552 -- because the post-construction wiring is identical.
  553 finishSpawn
  554   :: AgentEnv -> TabIndex -> TabName -> TabKind
  555   -> Either BackendError BackendHandle
  556   -> IO (Either TabError TabHandle)
  557 finishSpawn _env _idx _name _kind (Left err) =
  558   pure (Left (TabBackendConstructFailed err))
  559 finishSpawn env idx name kind (Right bh) = do
  560   state <- allocState env bh
  561   now <- getCurrentTime
  562   writeIORef (_bts_statusRef state) (Idle now)
  563   drainRunner  <- _env_fork env (drainerLoop env idx state)
  564   writeIORef (_bts_drainRunner state) (Just drainRunner)
  565   writerRunner <- _env_fork env (writerLoop state)
  566   writeIORef (_bts_writerRunner state) (Just writerRunner)
  567   pure (Right (mkHandle env idx name kind state))
  568 
  569 -- | Allocate the per-tab state. The bounded outbound queue's capacity
  570 -- comes from '_rc_inputQueueBound' so the H4 \"non-blocking send\"
  571 -- contract is configurable.
  572 allocState :: AgentEnv -> BackendHandle -> IO BackendTabState
  573 allocState env bh = do
  574   let rc = _env_routingConfig env
  575   sendQ      <- newTBQueueIO (fromIntegral (_rc_inputQueueBound rc))
  576   drainRef   <- newIORef Nothing
  577   writerRef  <- newIORef Nothing
  578   closedRef  <- newIORef False
  579   -- Sentinel: status starts 'Active' at allocation; the spawn finisher
  580   -- writes 'Idle now' before mkTabBackend returns.
  581   statRef    <- newIORef Active
  582   pure BackendTabState
  583     { _bts_backend      = bh
  584     , _bts_sendQ        = sendQ
  585     , _bts_drainRunner  = drainRef
  586     , _bts_writerRunner = writerRef
  587     , _bts_closed       = closedRef
  588     , _bts_statusRef    = statRef
  589     }
  590 
  591 -- | Build the public 'TabHandle' record from the per-tab state.
  592 mkHandle :: AgentEnv -> TabIndex -> TabName -> TabKind -> BackendTabState -> TabHandle
  593 mkHandle env idx name kind state = TabHandle
  594   { _tabHandle_index        = idx
  595   , _tabHandle_name         = name
  596   , _tabHandle_kind         = kind
  597   , _tabHandle_status       = readIORef (_bts_statusRef state)
  598   , _tabHandle_send         = sendBytes state
  599   , _tabHandle_enqueueSlash = enqueueSlashUnsupported
  600   , _tabHandle_close        = closeTabBackend env state
  601   }

  604 -- ---------------------------------------------------------------------------
  605 -- _tabHandle_send / _tabHandle_enqueueSlash
  606 -- ---------------------------------------------------------------------------
  607 
  608 -- | Enqueue a 'Text' payload onto the outbound queue. Non-blocking:
  609 -- returns @'Left' ('TabConcurrencyLimit' 0)@ if the queue is full so
  610 -- the dispatcher's send loop never blocks (H4).
  611 --
  612 -- The text is UTF-8 encoded and forwarded verbatim — including any
  613 -- leading @\/@ — so a direct-inject like @\/N \/pwd@ reaches the
  614 -- backend as the literal @\"\/pwd\"@ (I4: slash-prefix opaque to the
  615 -- backend).
  616 sendBytes :: BackendTabState -> Text -> IO (Either TabError ())
  617 sendBytes state t = atomically (tryEnqueueSTM (_bts_sendQ state) (TE.encodeUtf8 t))
  618 
  619 -- | STM body of 'sendBytes', factored out so tests can drive it inside
  620 -- a larger transaction if needed.
  621 tryEnqueueSTM :: TBQueue ByteString -> ByteString -> STM (Either TabError ())
  622 tryEnqueueSTM q bs = do
  623   full <- isFullTBQueue q
  624   if full
  625     then pure (Left (TabConcurrencyLimit 0))
  626     else writeTBQueue q bs >> pure (Right ())
  627 
  628 -- | '_tabHandle_enqueueSlash' for a backend tab: slash commands are
  629 -- not supported, so we return @'Left' ('TabUnsupportedCommand' cmd)@
  630 -- immediately without enqueueing (H13 \/ I4 contract).
  631 enqueueSlashUnsupported :: SlashCommand -> IO (Either TabError ())
  632 enqueueSlashUnsupported cmd = pure (Left (TabUnsupportedCommand cmd))
  633

  635 -- ---------------------------------------------------------------------------
  636 -- _tabHandle_close — destructive close for backend tabs
  637 -- ---------------------------------------------------------------------------
  638 
  639 -- | Close a backend tab. Idempotent + never throws (H6, H7).
  640 --
  641 -- Both 'CloseGraceful' and 'CloseForce' are destructive (H9 — close is
  642 -- already destructive for non-AI tabs, so @--force@ is a no-op
  643 -- distinct from graceful only at the pattern-match level). The triad
  644 -- below cancels both helper threads and then calls '_bh_close'. Each
  645 -- step is wrapped in 'safeIgnore' so a misbehaving backend can't leak
  646 -- an exception out of the never-throws contract.
  647 closeTabBackend :: AgentEnv -> BackendTabState -> CloseMode -> IO ()
  648 closeTabBackend _env state mode = do
  649   alreadyClosed <- atomicModifyIORef' (_bts_closed state) (True,)
  650   unless alreadyClosed $ do
  651     safeIgnore (cancelMaybeRunner (_bts_drainRunner state))
  652     safeIgnore (cancelMaybeRunner (_bts_writerRunner state))
  653     case mode of
  654       CloseGraceful -> safeIgnore (_bh_close (_bts_backend state))
  655       CloseForce    -> safeIgnore (_bh_close (_bts_backend state))
  656 
  657 -- | Read the captured 'TabRunner' (set by 'mkTabBackendWith' after
  658 -- fork) and invoke its cancel. Tolerant of the brief allocation\/fork
  659 -- window when the IORef is still 'Nothing'.
  660 cancelMaybeRunner :: IORef (Maybe TabRunner) -> IO ()
  661 cancelMaybeRunner ref = do
  662   mRunner <- readIORef ref
  663   for_ mRunner _trun_cancel
  664 
  665 -- | Best-effort: run an IO action and swallow synchronous failures.
  666 -- The close path MUST be never-throws (H7), so every step is wrapped
  667 -- defensively. 'AsyncCancelled' is also swallowed here because the
  668 -- close handler is invoked from outside the helper threads; any
  669 -- AsyncCancelled bubbling up through cancel itself is benign (the
  670 -- target thread has already received the cancel).
  671 safeIgnore :: IO () -> IO ()
  672 safeIgnore m = do
  673   _ <- try @SomeException m
  674   pure ()
  675 
  676 
  677 -- ---------------------------------------------------------------------------
  678 -- Drainer thread — _bh_recv → FullMsg via _env_channelOutQ
  679 -- ---------------------------------------------------------------------------
  680 
  681 -- | Drainer-thread top-level. Wraps 'drainerStep' in 'safelyRun' so a
  682 -- synchronous exception transitions the tab to 'Crashed' rather than
  683 -- killing the parent thread, and so 'AsyncCancelled' propagates per
  684 -- the C5 contract.
  685 drainerLoop :: AgentEnv -> TabIndex -> BackendTabState -> IO ()
  686 drainerLoop env idx state = safelyRun state loop
  687   where
  688     loop = do
  689       keepGoing <- drainerStep env idx state
  690       when keepGoing loop
  691 
  692 -- | One iteration of the drainer. Reads '_bh_recv' (blocking on the
  693 -- handle's default idle policy), emits a 'FullMsg' if the result is
  694 -- non-empty and the tab is focused (D4 producer-side skip), then
  695 -- returns whether the loop should continue.
  696 --
  697 -- The drainer terminates on 'RecvEof' (the backend's underlying pipe
  698 -- closed) or 'RecvTruncated' (the recv-buffer cap was exceeded — the
  699 -- caller cannot pull more bytes through this handle even if more are
  700 -- queued). 'RecvSettled' and 'RecvTimedOut' continue the loop.
  701 drainerStep :: AgentEnv -> TabIndex -> BackendTabState -> IO Bool
  702 drainerStep env idx state = do
  703   result <- _bh_recv (_bts_backend state) Nothing
  704   let bs = recvBytes result
  705   unless (BS.null bs) $ do
  706     let txt = TE.decodeUtf8 bs
  707     unless (T.null (T.strip txt)) $ do
  708       -- D4 producer-side focus check: skip the enqueue work when the
  709       -- tab is not focused. The writer thread re-applies the predicate
  710       -- on dequeue (D3), so this is an optimisation only; correctness
  711       -- comes from the writer.
  712       curFocus <- readIORef (_env_focus env)
  713       when (shouldEmit curFocus (SrcTab idx)) $
  714         atomically $ writeTBQueue (_env_channelOutQ env)
  715                        (SrcTab idx, FullMsg idx txt)
  716   pure $ case result of
  717     RecvSettled   _ -> True
  718     RecvTimedOut  _ -> True
  719     RecvEof       _ -> False
  720     RecvTruncated _ -> False
  721 
  722 
  723 -- ---------------------------------------------------------------------------
  724 -- Writer thread — _bts_sendQ → _bh_send
  725 -- ---------------------------------------------------------------------------
  726 
  727 -- | Writer-thread top-level. Drains '_bts_sendQ' forever, forwarding
  728 -- each chunk to '_bh_send'. Wrapped in 'safelyRun' for the same
  729 -- AsyncCancelled \/ Crashed semantics as the drainer.
  730 writerLoop :: BackendTabState -> IO ()
  731 writerLoop state = safelyRun state loop
  732   where
  733     loop = do
  734       bs <- atomically (readTBQueue (_bts_sendQ state))
  735       -- '_bh_send' may block. We tolerate the block because the
  736       -- bounded '_bts_sendQ' keeps the producer side non-blocking;
  737       -- the writer is a single-purpose drainer and nothing else
  738       -- depends on its progress.
  739       _bh_send (_bts_backend state) bs
  740       loop
  741 
  742 
  743 -- ---------------------------------------------------------------------------
  744 -- safelyRun — shared exception handler for drainer + writer
  745 -- ---------------------------------------------------------------------------
  746 
  747 -- | Outer exception handler shared by both helper threads.
  748 --
  749 -- * 'AsyncCancelled' propagates (re-raised) so '_tabHandle_close'
  750 --   semantics work correctly per C5.
  751 -- * Other 'SomeException' transitions the tab to 'Crashed' (carrying
  752 --   a redacted short label) and the helper thread exits.
  753 safelyRun :: BackendTabState -> IO () -> IO ()
  754 safelyRun state body = body `catch` handler
  755   where
  756     handler :: SomeException -> IO ()
  757     handler e
  758       | Just AsyncCancelled <- fromException e =
  759           throwIO e  -- propagate AsyncCancelled (C5)
  760       | otherwise =
  761           writeIORef (_bts_statusRef state)
  762                      (Crashed (PublicTabError "tab: backend loop crashed"))

  769 -- | Compose a default friendly tab name for a backend tab.
  770 --
  771 -- 'mkShellTab' calls this with the first user-supplied arg as the
  772 -- suffix, producing labels like @\"shell ls\"@. The empty-arg case is
  773 -- handled at the call site, so the function only sees a non-empty
  774 -- arg list.
  775 defaultBackendName :: Text -> [Text] -> Text
  776 defaultBackendName prefix args = case args of
  777   (a : _) -> prefix <> " " <> a
  778   -- Dead arm: callers short-circuit before reaching here. We keep the
  779   -- pattern total so a future call-site change can't silently produce
  780   -- a partial-match runtime crash.
  781   []      -> prefix <> " tab"

  785 authDenied = pure (Left (TabSpawnAuthDenied PublicAuthError))
