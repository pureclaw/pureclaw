   44 -- | Construction-time configuration for 'fakePtyIO'.
   45 --
   46 -- '_fpc_outputScript' is the sequence of chunks the next N
   47 -- @_pio_read@ calls will yield, in order. Once the script is
   48 -- exhausted (and '_fpc_eofAfterBytes' is reached, if set), @_pio_read@
   49 -- returns 'BS.empty' to signal EOF — which the drainer turns into an
   50 -- 'RsEof' on its queue.
   51 data FakePtyConfig = FakePtyConfig
   52   { _fpc_clock         :: !FakeClock
   53   , _fpc_initialOutput :: !ByteString
   54     -- ^ A single chunk emitted from the very first @_pio_read@ call
   55     -- (placed at the head of the read queue at open-time). Empty
   56     -- means: no initial chunk.
   57   , _fpc_eofAfterBytes :: !(Maybe Int)
   58     -- ^ Optional cap: once this many bytes have been read,
   59     -- subsequent @_pio_read@ calls return 'BS.empty' (EOF) regardless
   60     -- of any remaining script.
   61   , _fpc_outputScript  :: ![ByteString]
   62     -- ^ Subsequent chunks emitted on each @_pio_read@ call.
   63   }
   64 
   65 -- | Allocate a 'PtyIO' backed by in-process 'IORef's.
   66 --
   67 -- Each call to 'fakePtyIO' produces a fresh, independent set of
   68 -- 'IORef' state via 'PtyOpenSpec'-time setup in @_pio_open@. The
   69 -- returned 'PtyIO' is safe to share across multiple opens, but the
   70 -- 'IORef' state is per-open — every @_pio_open@ allocates new refs.
   71 fakePtyIO :: FakePtyConfig -> IO PtyIO
   72 fakePtyIO cfg = pure PtyIO
   73   { _pio_open   = fakeOpen
   74   , _pio_read   = fakeRead
   75   , _pio_write  = fakeWrite
   76   , _pio_resize = fakeResize
   77   , _pio_close  = fakeClose
   78   }
   79   where
   80     fakeOpen :: PtyOpenSpec -> IO PtyFds
   81     fakeOpen _spec = do
   82       let initial = _fpc_initialOutput cfg
   83           script  = _fpc_outputScript cfg
   84           chunks  = if BS.null initial then script else initial : script
   85       rq <- newIORef chunks
   86       wl <- newIORef []
   87       rl <- newIORef []
   88       cl <- newIORef False
   89       ea <- newIORef (_fpc_eofAfterBytes cfg)
   90       br <- newIORef 0
   91       pure $ mkFakePtyFds FakePtyState
   92         { _fps_readQueue = rq
   93         , _fps_writeLog  = wl
   94         , _fps_resizeLog = rl
   95         , _fps_closed    = cl
   96         , _fps_eofAfter  = ea
   97         , _fps_bytesRead = br
   98         , _fps_clock     = _fpc_clock cfg
   99         }
  100 
  101     fakeRead :: PtyFds -> Int -> IO ByteString
  102     fakeRead fds _ = do
  103       mResult <- withFakePtyState fds $ \st -> do
  104         closed <- readIORef (_fps_closed st)
  105         if closed
  106           then pure BS.empty
  107           else do
  108             mEof  <- readIORef (_fps_eofAfter st)
  109             sofar <- readIORef (_fps_bytesRead st)
  110             let pastCap = case mEof of
  111                   Just cap -> sofar >= cap
  112                   Nothing  -> False
  113             if pastCap
  114               then pure BS.empty
  115               else do
  116                 next <- atomicModifyIORef' (_fps_readQueue st) $ \case
  117                   []      -> ([], Nothing)
  118                   (x:xs)  -> (xs, Just x)
  119                 case next of
  120                   Just bs -> do
  121                     atomicModifyIORef' (_fps_bytesRead st) $ \n ->
  122                       (n + BS.length bs, ())
  123                     pure bs
  124                   Nothing ->
  125                     -- Script exhausted but no EOF cap reached:
  126                     -- block forever so the drainer\'s read sits
  127                     -- waiting (mirroring a quiet real PTY). Polled
  128                     -- on a coarse 50ms tick so cancellation from the
  129                     -- drainer's 'Async.cancel' lands promptly.
  130                     case mEof of
  131                       Just _  -> pure BS.empty -- EOF cap set: signal EOF
  132                       Nothing -> blockForever
  133       pure (fromMaybe BS.empty mResult)
  134 
  135     blockForever :: IO ByteString
  136     blockForever = do
  137       threadDelay 50_000
  138       blockForever
  139 
  140     fakeWrite :: PtyFds -> ByteString -> IO ()
  141     fakeWrite fds bs = do
  142       _ <- withFakePtyState fds $ \st ->
  143         atomicModifyIORef' (_fps_writeLog st) $ \xs -> (bs : xs, ())
  144       pure ()
  145 
  146     fakeResize :: PtyFds -> Cols -> Rows -> IO ()
  147     fakeResize fds c r = do
  148       _ <- withFakePtyState fds $ \st ->
  149         atomicModifyIORef' (_fps_resizeLog st) $ \xs -> ((c, r) : xs, ())
  150       pure ()
  151 
  152     fakeClose :: PtyFds -> IO ()
  153     fakeClose fds = do
  154       _ <- withFakePtyState fds $ \st ->
  155         writeIORef (_fps_closed st) True
  156       pure ()
  157 
  158 -- | Read the recorded write log (newest-first).
  159 readFakeWriteLog :: FakePtyState -> IO [ByteString]
  160 readFakeWriteLog = readIORef . _fps_writeLog
  161 
  162 -- | Read the recorded resize log (newest-first).
  163 readFakeResizeLog :: FakePtyState -> IO [(Cols, Rows)]
  164 readFakeResizeLog = readIORef . _fps_resizeLog

  166 -- | Has @_pio_close@ been called on this state?
  167 readFakeClosed :: FakePtyState -> IO Bool
  168 readFakeClosed = readIORef . _fps_closed
  169 
  170 -- | Total bytes ever returned from @_pio_read@.
  171 readFakeBytesRead :: FakePtyState -> IO Int
  172 readFakeBytesRead = readIORef . _fps_bytesRead
