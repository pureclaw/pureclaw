   39 -- ---------------------------------------------------------------------------
   40 -- Args denylist (S9)
   41 -- ---------------------------------------------------------------------------
   42 
   43 -- | Container flags that pose a security risk. Any flag in '_h_args'
   44 -- matching one of these strings causes the container factory arm to
   45 -- reject the spawn.
   46 containerArgsDenylist :: [Text]
   47 containerArgsDenylist =
   48   [ "--privileged"
   49   , "--cap-add"
   50   , "--security-opt"
   51   , "--device"
   52   , "--pid=host"
   53   , "--network=host"
   54   , "--userns=host"
   55   , "--uts=host"
   56   , "--ipc=host"
   57   , "-v"
   58   , "--volume"
   59   , "--mount"
   60   ]
   61 
   62 -- | Why container args were rejected.
   63 data ContainerArgsError
   64   = DeniedFlag !Text
   65     -- ^ A flag from 'containerArgsDenylist' was found in '_h_args'.
   66   deriving stock (Show, Eq)
   67 
   68 -- | Check a list of user-supplied args against 'containerArgsDenylist'.
   69 -- Returns the first denied flag found, if any.
   70 checkContainerArgs :: [Text] -> Either ContainerArgsError ()
   71 checkContainerArgs args =
   72   case filter (`elem` containerArgsDenylist) args of
   73     []    -> Right ()
   74     (f:_) -> Left (DeniedFlag f)

   84 containerEngineBinary Podman  = "podman"

   91 
   92 -- | Map a 'HarnessFlavour' to the binary name invoked inside the
   93 -- container. This is the program that appears after the @--@ separator
   94 -- in the container exec argv.
   95 flavourToBinary :: HarnessFlavour -> Text
   96 flavourToBinary HClaudeCode = "claude"
   97 flavourToBinary HCodex      = "codex"
   98 flavourToBinary HOpenCode   = "opencode"
   99 flavourToBinary HHermes     = "hermes"
  100 flavourToBinary HPureClaw   = "pureclaw"
  101 flavourToBinary (HCustom n) = n

  104 -- ---------------------------------------------------------------------------
  105 -- Argv construction (S9)
  106 -- ---------------------------------------------------------------------------
  107 
  108 -- | Build the explicit exec argv for a container backend.
  109 --
  110 -- Structure:
  111 -- @[engine_binary, "exec", "-it", target, "--", harness_binary] ++ args@
  112 --
  113 -- The @--@ separator is MANDATORY to prevent argument injection: without
  114 -- it, a malicious target or harness args could be interpreted as flags
  115 -- to the container engine.
  116 buildContainerExecArgv :: ContainerSpec -> HarnessFlavour -> [Text] -> [Text]
  117 buildContainerExecArgv cs flavour harnessArgs =
  118   [ containerEngineBinary (_cs_engine cs)
  119   , "exec"
  120   , "-it"
  121   , unContainerTarget (_cs_target cs)
  122   , "--"
  123   , flavourToBinary flavour
  124   ] <> harnessArgs

  131 -- | Why a cwd path was rejected.
  132 data CwdError
  133   = CwdPathTraversal
  134     -- ^ The path contains @..@ components (path traversal).
  135   deriving stock (Show, Eq)
  136 
  137 -- | Pure validation of '_h_cwd'. Rejects paths containing @..@
  138 -- components (which could escape the intended working directory).
  139 -- Returns the validated 'FilePath' when present and valid, or 'Nothing'
  140 -- when no cwd was specified.
  141 --
  142 -- This is a lightweight pure check that catches the most common
  143 -- traversal attack. The full 'mkSafePath' IO validation (which follows
  144 -- symlinks and canonicalises) is performed at spawn time by the factory
  145 -- arm when a 'WorkspaceRoot' is available.
  146 validateCwd :: Maybe Text -> Either CwdError (Maybe FilePath)
  147 validateCwd Nothing  = Right Nothing
  148 validateCwd (Just t) =
  149   let path = T.unpack t
  150   in if hasParentTraversal path
  151        then Left CwdPathTraversal
  152        else Right (Just path)
  153 
  154 -- | Check if a path contains @..@ components that could traverse upward.
  155 hasParentTraversal :: FilePath -> Bool
  156 hasParentTraversal path =
  157   let components = splitPath path
  158   in ".." `elem` components
  159   where
  160     -- Simple split on '/' — sufficient for the pure traversal check.
  161     splitPath :: FilePath -> [String]
  162     splitPath = go []
  163       where
  164         go acc [] = [reverse acc]
  165         go acc ('/':rest) = reverse acc : go [] rest
  166         go acc (c:rest) = go (c:acc) rest
