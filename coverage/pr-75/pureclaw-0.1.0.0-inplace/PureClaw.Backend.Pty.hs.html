  189 -- | The injectable PTY-allocation seam.
  190 --
  191 -- The factory modules ('PureClaw.Backend.Local' etc.) consume a value
  192 -- of this type instead of importing @posix-pty@ directly. Tests use
  193 -- 'PureClaw.Backend.Pty.Fake.fakePtyIO'; production uses 'realPtyIO'.
  194 data PtyIO = PtyIO
  195   { _pio_open   :: PtyOpenSpec -> IO PtyFds
  196   , _pio_read   :: PtyFds -> Int -> IO ByteString
  197     -- ^ Read up to N bytes. May block. An empty result means EOF.
  198   , _pio_write  :: PtyFds -> ByteString -> IO ()
  199   , _pio_resize :: PtyFds -> Cols -> Rows -> IO ()
  200   , _pio_close  :: PtyFds -> IO ()

  203 -- | The real, @posix-pty@-backed 'PtyIO'.
  204 --
  205 -- The 'PtyOpenSpec' carries the program + args + env + cwd + geometry;
  206 -- the open action calls 'Posix.Pty.spawnWithPty' inside a real PTY and
  207 -- returns a wrapping 'PtyFds'.
  208 realPtyIO :: PtyIO
  209 realPtyIO = PtyIO
  210   { _pio_open   = realOpen
  211   , _pio_read   = realRead
  212   , _pio_write  = realWrite
  213   , _pio_resize = realResize
  214   , _pio_close  = realClose
  215   }
  216   where
  217     -- HPC:Exclude — kernel-error paths exercised only via integration tests.
  218     realOpen :: PtyOpenSpec -> IO PtyFds
  219     realOpen spec = do
  220       let cols = unCols (_pos_cols spec)
  221           rows = unRows (_pos_rows spec)
  222           env  = envToTuples (_pos_env spec)
  223           mEnv = if null env then Nothing else Just env
  224           prog = _pos_program spec
  225           args = map T.unpack (_pos_args spec)
  226       (pty, ph) <- Posix.Pty.spawnWithPty mEnv True prog args (cols, rows)
  227       pure (RealPtyFds pty (Just ph))
  228 
  229     realRead :: PtyFds -> Int -> IO ByteString
  230     realRead fds _ = case fds of
  231       RealPtyFds pty _ -> do
  232         -- HPC:Exclude — real-PTY error paths
  233         r <- try @SomeException (Posix.Pty.readPty pty)
  234         case r of
  235           Right bs -> pure bs
  236           Left _   -> pure BS.empty -- treat as EOF
  237       FakePtyFds _ -> pure BS.empty
  238       -- ^ unreachable: realPtyIO never produces a FakePtyFds; the
  239       --   total pattern keeps -Wall quiet.
  240 
  241     realWrite :: PtyFds -> ByteString -> IO ()
  242     realWrite fds bs = case fds of
  243       RealPtyFds pty _ -> Posix.Pty.writePty pty bs
  244       FakePtyFds _     -> pure ()
  245 
  246     realResize :: PtyFds -> Cols -> Rows -> IO ()
  247     realResize fds (Cols c) (Rows r) = case fds of
  248       RealPtyFds pty _ -> Posix.Pty.resizePty pty (c, r)
  249       FakePtyFds _     -> pure ()
  250 
  251     realClose :: PtyFds -> IO ()
  252     realClose fds = case fds of
  253       RealPtyFds pty mPh -> do
  254         -- 1. SIGKILL the child FIRST when we have its PID.
  255         --
  256         -- SIGTERM (what 'Proc.terminateProcess' sends on POSIX) is
  257         -- ignored by interactive shells — see bash(1): "In the
  258         -- absence of any traps, an interactive shell ignores
  259         -- SIGTERM". A 'bash -i' child therefore survives the polite
  260         -- termination signal and 'Proc.waitForProcess' blocks
  261         -- forever. SIGKILL cannot be caught or ignored, so it is
  262         -- the only signal that reliably unblocks the close path.
  263         --
  264         -- We do this BEFORE closing the PTY master so the drainer's
  265         -- blocking read (a non-interruptible C call inside
  266         -- @posix-pty@) terminates deterministically — once the child
  267         -- is gone, the slave fd is closed and the master read returns
  268         -- EOF, which lets the drainer Async exit its loop.
  269         case mPh of
  270           Just ph -> do
  271             mPid <- Proc.getPid ph
  272             case mPid of
  273               Just pid ->
  274                 Control.Monad.void $ try @SomeException
  275                   (Posix.Sig.signalProcess Posix.Sig.sigKILL pid)
  276               Nothing  -> pure ()
  277           Nothing -> pure ()
  278         -- 2. Close PTY master. Also a hangup signal for any child
  279         -- that we somehow couldn't address by PID.
  280         _ <- try @SomeException (Posix.Pty.closePty pty)
  281         -- 3. Collect the zombie. SIGKILL above means this should
  282         -- return immediately; bound it to 2s to defend against any
  283         -- pathological case where the child survived.
  284         case mPh of
  285           Just ph -> do
  286             _ <- timeout 2_000_000 (try @SomeException (Proc.waitForProcess ph))
  287             pure ()
  288           Nothing -> pure ()
  289       FakePtyFds _ -> pure ()

  307 -- | A unit pushed onto 'DrainState._ds_queue' by the drainer 'Async'.
  308 data RecvSignal
  309   = RsChunk !ByteString
  310     -- ^ A new chunk of output.
  311   | RsEof
  312     -- ^ The drainer\'s underlying read returned EOF (PTY closed).
  313   | RsTrunc
  314     -- ^ The per-backend recv-buffer cap was reached. After this
  315     -- sentinel, the drainer stops enqueuing chunks and the truncate
  316     -- latch ('_ds_truncated') is set to 'True'.
  317   deriving stock (Eq, Show)
  318 
  319 -- | The STM state shared between the drainer 'Async' and 'drainerRecv'.
  320 data DrainState = DrainState
  321   { _ds_queue       :: !(TBQueue RecvSignal)
  322   , _ds_truncated   :: !(TVar Bool)
  323     -- ^ Latches 'True' on the first 'RsTrunc'; never reset.
  324   , _ds_drainerDone :: !(TVar Bool)
  325     -- ^ Set to 'True' when the drainer 'Async' has finished
  326     -- (EOF, exception, or cancel).
  327   , _ds_bytesQueued :: !(TVar Int)
  328     -- ^ Sum of bytes accepted by the queue (i.e. before truncation).
  329   }
  330 
  331 -- | Allocate a fresh 'DrainState'. The queue capacity is generous: the
  332 -- per-byte cap is enforced by 'runDrainerLoop' against
  333 -- '_ds_bytesQueued', not by the queue\'s slot count.
  334 newDrainState :: IO DrainState
  335 newDrainState = do
  336   q  <- newTBQueueIO 1024
  337   tr <- newTVarIO False
  338   dd <- newTVarIO False
  339   bq <- newTVarIO 0
  340   pure DrainState
  341     { _ds_queue       = q
  342     , _ds_truncated   = tr
  343     , _ds_drainerDone = dd
  344     , _ds_bytesQueued = bq
  345     }

  347 -- | Spawn the drainer loop.
  348 --
  349 -- The loop reads chunks from the 'PtyFds' via @_pio_read@; on each
  350 -- non-empty chunk it accounts for it against the per-backend
  351 -- @recvBufferCap@ (in bytes); on overflow it sets '_ds_truncated' and
  352 -- emits 'RsTrunc', then stops reading. On EOF (empty 'ByteString'
  353 -- from the read) it emits 'RsEof' and exits. Cancelling the returned
  354 -- 'Async' terminates the loop; '_ds_drainerDone' is set on any exit.
  355 runDrainerLoop
  356   :: PtyIO
  357   -> PtyFds
  358   -> Int -- ^ recv-buffer cap, bytes
  359   -> DrainState
  360   -> IO (Async ())
  361 runDrainerLoop pio fds cap st = Async.async (loop `finallyAtomic` markDone)
  362   where
  363     loop :: IO ()
  364     loop = do
  365       truncated <- readTVarIO (_ds_truncated st)
  366       if truncated
  367         then pure ()
  368         else do
  369           chunk <- _pio_read pio fds 4096
  370           if BS.null chunk
  371             then atomically (writeTBQueue (_ds_queue st) RsEof)
  372             else do
  373               cont <- atomically $ do
  374                 queued <- readTVar (_ds_bytesQueued st)
  375                 let newTotal = queued + BS.length chunk
  376                 if newTotal > cap
  377                   then do
  378                     writeTVar (_ds_truncated st) True
  379                     writeTBQueue (_ds_queue st) RsTrunc
  380                     pure False
  381                   else do
  382                     writeTVar (_ds_bytesQueued st) newTotal
  383                     writeTBQueue (_ds_queue st) (RsChunk chunk)
  384                     pure True
  385               when cont loop
  386 
  387     markDone :: IO ()
  388     markDone = atomically (writeTVar (_ds_drainerDone st) True)
  389 
  390     -- A 'finally'-style wrapper that ignores exceptions from the inner
  391     -- action (the drainer is best-effort) but always runs the cleanup.
  392     finallyAtomic :: IO () -> IO () -> IO ()
  393     finallyAtomic action fin = mask $ \restore -> do
  394       r <- try @SomeException (restore action)
  395       fin
  396       case r of
  397         Right () -> pure ()
  398         Left _   -> pure ()

  400 -- | Consume from 'DrainState' according to the supplied 'IdleSpec'.
  401 --
  402 -- Implements the idle state machine from
  403 -- @docs\/terminal-backend-abstractions.md@ § "Idle state machine":
  404 --
  405 -- 1. Wait up to @idleMinFirstByte@ ms (or @idleTimeoutMs@ — whichever
  406 --    bounds the wait first) for the first chunk. If nothing arrives
  407 --    within @idleTimeoutMs@ total, return 'RecvTimedOut' on the
  408 --    bytes-so-far (empty if no chunk ever arrived).
  409 -- 2. After the first chunk, accumulate bytes from subsequent chunks.
  410 --    Each chunk resets a rolling quiet-window timer of length
  411 --    @idleQuietMs@. When the quiet window elapses with no further
  412 --    chunks, return 'RecvSettled'.
  413 -- 3. The truncate latch ('_ds_truncated' @== True@) short-circuits
  414 --    every recv with 'RecvTruncated' on whatever has accumulated.
  415 --    The latch persists across calls until @_bh_close@.
  416 -- 4. An 'RsEof' on the queue (or '_ds_drainerDone' going 'True')
  417 --    yields 'RecvEof' on the bytes-so-far.
  418 -- 5. If @idleTimeoutMs@ elapses at any point (including mid-drain),
  419 --    return 'RecvTimedOut' on whatever has accumulated.
  420 drainerRecv :: DrainState -> IdleSpec -> IO (RecvResult ByteString)
  421 drainerRecv st idle = do
  422   isTrunc <- readTVarIO (_ds_truncated st)
  423   if isTrunc
  424     then drainQueueLatched
  425     else do
  426       totalTimer <- registerDelay (totalMs * 1_000)
  427       phaseWait totalTimer
  428   where
  429     totalMs = idleTimeoutMs idle
  430     quietMs = idleQuietMs idle
  431     firstMs = idleMinFirstByte idle
  432 
  433     -- Phase 0: waiting for the first chunk. Bounded by @firstMs@
  434     -- (or @totalMs@ if @firstMs == 0@), but never longer than the
  435     -- total timer.
  436     phaseWait :: TVar Bool -> IO (RecvResult ByteString)
  437     phaseWait totalTimer = do
  438       let firstWaitMs = if firstMs > 0 then firstMs else totalMs
  439       firstTimer <- registerDelay (firstWaitMs * 1_000)
  440       r <- atomically $
  441         (Right <$> readSignal st)
  442           `orElse` (do
  443               total <- readTVar totalTimer
  444               first <- readTVar firstTimer
  445               if total || first
  446                 then pure (Left ())
  447                 else retry)
  448       case r of
  449         Left ()             -> pure (RecvTimedOut BS.empty)
  450         Right RsEof         -> pure (RecvEof BS.empty)
  451         Right RsTrunc       -> pure (RecvTruncated BS.empty)
  452         Right (RsChunk bs)  -> phaseDrain totalTimer [bs]
  453 
  454     -- Phase 1: draining. Loops on rolling @quietMs@ STM waits, bounded
  455     -- by the same total timer as phase 0. EOF, truncate, total-timeout
  456     -- and quiet-window-elapsed are the four exits.
  457     phaseDrain :: TVar Bool -> [ByteString] -> IO (RecvResult ByteString)
  458     phaseDrain totalTimer acc = do
  459       quietTimer <- registerDelay (quietMs * 1_000)
  460       r <- atomically $
  461         (Right <$> readSignal st)
  462           `orElse` (do
  463               total <- readTVar totalTimer
  464               quiet <- readTVar quietTimer
  465               if total
  466                 then pure (Left True)   -- total-timeout
  467                 else if quiet
  468                   then pure (Left False) -- quiet-elapsed
  469                   else retry)
  470       case r of
  471         Left True             -> pure (RecvTimedOut (BS.concat (reverse acc)))
  472         Left False            -> pure (RecvSettled  (BS.concat (reverse acc)))
  473         Right RsEof           -> pure (RecvEof (BS.concat (reverse acc)))
  474         Right RsTrunc         -> pure (RecvTruncated (BS.concat (reverse acc)))
  475         Right (RsChunk bs)    -> phaseDrain totalTimer (bs : acc)
  476 
  477     -- Latch set: drain whatever is already queued non-blockingly and
  478     -- return 'RecvTruncated'.
  479     drainQueueLatched :: IO (RecvResult ByteString)
  480     drainQueueLatched = do
  481       acc <- drainNonBlocking []
  482       pure (RecvTruncated (BS.concat (reverse acc)))
  483 
  484     drainNonBlocking :: [ByteString] -> IO [ByteString]
  485     drainNonBlocking acc = do
  486       mSig <- atomically (tryReadTBQueue (_ds_queue st))
  487       case mSig of
  488         Nothing           -> pure acc
  489         Just (RsChunk bs) -> drainNonBlocking (bs : acc)
  490         Just RsEof        -> pure acc
  491         Just RsTrunc      -> pure acc
  492 
  493 -- | Read a single 'RecvSignal' atomically. Also wakes on truncate-
  494 -- latch flip or drainer-done so callers can short-circuit promptly.
  495 readSignal :: DrainState -> STM RecvSignal
  496 readSignal st =
  497   readTBQueue (_ds_queue st)
  498     `orElse` (do
  499         tr <- readTVar (_ds_truncated st)
  500         unless tr retry
  501         pure RsTrunc)
  502     `orElse` (do
  503         dd <- readTVar (_ds_drainerDone st)
  504         unless dd retry
  505         pure RsEof)

  511 -- | Wire a 'PtyIO' + 'PtyOpts' into a complete 'BackendHandle' whose
  512 -- @_bh_recv@ talks to a 'drainerRecv' loop.
  513 --
  514 -- The factory:
  515 --
  516 -- 1. Acquires the per-backend recv buffer cap from the process-wide
  517 --    quota (rounded up to whole MiB). On oversubscription returns
  518 --    @Left BackendBufferQuotaExceeded@ — no resources opened.
  519 -- 2. Opens the PTY via @_pio_open@ (which uses the program + args
  520 --    from the supplied 'PtyOpenSpec'-like fields on 'PtyOpts'). On
  521 --    a failure the quota is released and the error propagates.
  522 -- 3. Spawns the drainer 'Async' via 'runDrainerLoop'.
  523 -- 4. Builds a 'BackendHandle' whose @_bh_close@ is idempotent: it
  524 --    flips an internal flag, cancels the drainer, closes the PTY,
  525 --    and releases the buffer quota exactly once.
  526 --
  527 -- WU7 ships the seam and the test-side fake. WU8 \/ WU9 \/ WU10 will
  528 -- compose @AuthorizedCommand@ \/ @RemoteCommand@ \/ tmux selectors
  529 -- into the 'PtyOpenSpec' the factories pass in. For WU7\'s tests, the
  530 -- @open@ closure is called with a dummy 'PtyOpenSpec' (only the
  531 -- geometry matters to fakePtyIO).
  532 mkDrainerBackendHandle
  533   :: PtyIO
  534   -> PtyOpts
  535   -> PtyOpenSpec
  536   -> BackendKind
  537   -> IO (Either BackendError BackendHandle)
  538 mkDrainerBackendHandle pio opts spec kind = do
  539   let capBytes = _pto_recvBufferCap opts
  540       capMiB   = bytesToMiBCeil capBytes
  541   quotaPtr <- readIORef globalBackendBufferQuota
  542   acq <- acquireBufferQuota quotaPtr capMiB
  543   case acq of
  544     Left e  -> pure (Left e)
  545     Right () -> do
  546       fds <- _pio_open pio spec
  547       st  <- newDrainState
  548       drainerAsync <- runDrainerLoop pio fds capBytes st
  549       releasedRef  <- newIORef False
  550       closeLock    <- newMVar ()
  551       let releaseQuotaOnce = do
  552             already <- atomicSwapTrue releasedRef
  553             if already
  554               then pure ()
  555               else releaseBufferQuota quotaPtr capMiB
  556 
  557           doSend = _pio_write pio fds
  558 
  559           doRecv mIdle = drainerRecv st (fromMaybe (_pto_idle opts) mIdle)
  560 
  561           doResize = _pio_resize pio fds
  562 
  563           doClose = modifyMVar_ closeLock $ \() -> do
  564             -- Close FIRST. The drainer is almost always parked in a
  565             -- blocking read inside @posix-pty@ — a non-interruptible
  566             -- C call. 'Async.cancel' queues an async exception, but
  567             -- the FFI call does not return until bytes arrive OR the
  568             -- child dies OR the fd is closed. So cancelling before
  569             -- '_pio_close' makes the subsequent 'Async.waitCatch'
  570             -- block forever.
  571             --
  572             -- '_pio_close' (for 'realPtyIO') sends SIGKILL to the
  573             -- child and closes the master fd, which together force
  574             -- the blocking read to return EOF promptly. The drainer
  575             -- loop then sees an empty 'ByteString', pushes 'RsEof',
  576             -- and exits naturally — and now 'cancel'/'waitCatch'
  577             -- below are fast no-ops.
  578             _ <- try @SomeException (_pio_close pio fds)
  579             Async.cancel drainerAsync
  580             _ <- Async.waitCatch drainerAsync
  581             releaseQuotaOnce
  582             pure ()
  583 
  584       pure $ Right BackendHandle
  585         { _bh_name        = "drainer-pty"
  586         , _bh_kind        = kind
  587         , _bh_defaultIdle = _pto_idle opts
  588         , _bh_send        = doSend
  589         , _bh_recv        = doRecv
  590         , _bh_resize      = doResize
  591         , _bh_close       = doClose
  592         }

  594 -- | Atomically set an 'IORef Bool' to 'True', returning its previous
  595 -- value. Used to defend against double-release of the buffer quota.
  596 atomicSwapTrue :: IORef Bool -> IO Bool
  597 atomicSwapTrue ref = atomicModifyIORef' ref (True,)

  599 -- | Round a byte count up to whole MiB (1 MiB == 1_048_576 bytes).
  600 -- Zero or negative byte counts produce @0@.
  601 bytesToMiBCeil :: Int -> Int
  602 bytesToMiBCeil n
  603   | n <= 0    = 0
  604   | otherwise = (n + mib - 1) `div` mib
  605   where mib = 1024 * 1024
