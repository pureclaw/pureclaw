   23 -- | A command that has been authorized by the security policy.
   24 -- Constructor is intentionally NOT exported — the only way to obtain an
   25 -- 'AuthorizedCommand' is through 'authorize'.
   26 --
   27 -- Note for downstream: 'ShellHandle.execute' is responsible for stripping
   28 -- the subprocess environment (@setEnv (Just [])@) — environment isolation
   29 -- is an execution-time concern, not a policy concern.
   30 newtype AuthorizedCommand = AuthorizedCommand { unAuthorizedCommand :: (FilePath, [Text]) }
   31 
   32 -- | Errors from command authorization.
   33 data CommandError
   34   = CommandNotAllowed Text    -- ^ The command is not in the policy's allowed set
   35   | CommandInAutonomyDeny     -- ^ The policy's autonomy level is 'Deny'
   36   deriving stock (Show, Eq)
   37 
   38 -- | Authorize a command against a security policy. Pure — no IO.
   39 --
   40 -- Checks:
   41 -- 1. Autonomy level is not 'Deny'
   42 -- 2. Command basename is in the policy's allowed command set
   43 authorize :: SecurityPolicy -> FilePath -> [Text] -> Either CommandError AuthorizedCommand
   44 authorize policy cmd args
   45   | _sp_autonomy policy == Deny =
   46       Left CommandInAutonomyDeny
   47   | not (isCommandAllowed policy (CommandName (T.pack (takeFileName cmd)))) =
   48       Left (CommandNotAllowed (T.pack (takeFileName cmd)))
   49   | otherwise =
   50       Right (AuthorizedCommand (cmd, args))
   51 
   52 -- | Authorize a shell-string command for execution via @bash -c@. Pure — no IO.
   53 --
   54 -- Gating is intentionally a single yes/no toggle: the policy must list
   55 -- @shell@ in its allowed-command set. Per-basename allowlisting is
   56 -- meaningless once bash is in the loop (pipes, @&&@, @$()@ can all
   57 -- compose forbidden basenames), so we don't pretend to enforce it here.
   58 -- Deployments that want fine-grained per-program control should disable
   59 -- @shell@ in the policy and use the argv-style @exec@ tool instead.
   60 authorizeShell :: SecurityPolicy -> Text -> Either CommandError AuthorizedCommand
   61 authorizeShell policy cmd
   62   | _sp_autonomy policy == Deny =
   63       Left CommandInAutonomyDeny
   64   | not (isCommandAllowed policy (CommandName "shell")) =
   65       Left (CommandNotAllowed "shell")
   66   | otherwise =
   67       Right (AuthorizedCommand ("bash", ["-c", cmd]))

   69 -- | Construct an 'AuthorizedCommand' for PureClaw's OWN tmux process
   70 -- management. This is the controlled internal seam through which EVERY tmux
   71 -- subprocess invocation flows, so that an unauthorized tmux call is impossible
   72 -- by construction (the 'AuthorizedCommand' value constructor is not exported,
   73 -- and this is the sole tmux-specific path to obtain one).
   74 --
   75 -- Unlike 'authorize'/'authorizeShell', this performs NO 'SecurityPolicy'
   76 -- check and ALWAYS succeeds: tmux invocations here are PureClaw's own
   77 -- harness-lifecycle management (spawn/sweep/capture/kill of windows it owns),
   78 -- NOT user-issued commands. The user-command policy gate governs commands the
   79 -- agent is asked to run on the user's behalf; the manager's internal tmux use
   80 -- is a separate, manager-owned boundary. The in-pane harness command itself
   81 -- (a deliberately shell-interpreted string) is defended separately by
   82 -- 'PureClaw.Internal.ShellQuote' / @escapeForShell@, not by this seam.
   83 --
   84 -- TODO(harness-registry phase 2): audit-log tmux seam invocations once a
   85 -- 'LogHandle' is threaded to the call sites. The design (§8 B1) specifies this
   86 -- seam as "always-permitted-but-logged"; threading a logger through
   87 -- @Harness/Tmux.hs@ would change exported signatures (out of scope for the
   88 -- additive Phase-1 refactor), so logging is deferred. The hard, type-enforced
   89 -- invariant — a single tmux chokepoint — is delivered now.
   90 authorizeTmuxCommand :: FilePath -> [Text] -> AuthorizedCommand
   91 authorizeTmuxCommand tmuxBin args = AuthorizedCommand (tmuxBin, args)

   93 -- | Get the program path from an authorized command.
   94 getCommandProgram :: AuthorizedCommand -> FilePath
   95 getCommandProgram = fst . unAuthorizedCommand

   97 -- | Get the arguments from an authorized command.
   98 getCommandArgs :: AuthorizedCommand -> [Text]
   99 getCommandArgs = snd . unAuthorizedCommand
