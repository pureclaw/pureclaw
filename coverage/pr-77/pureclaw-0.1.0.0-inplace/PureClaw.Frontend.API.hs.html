  275 -- as one injectable seam so the security-critical /corroborate-before-mutate/
  276 -- ordering lives in the handler (not hidden behind the seam) while still being
  277 -- fully testable.

  279 -- @_rt_liveMarker session window@ re-reads the LIVE @\@pcl_id@ of the window at

  290     -- title shows PureClaw has detached. Invoked ONLY on the corroborated
  291     -- branch (right after '_rt_clearMarker'), so we only retitle a window we
  292     -- just confirmed is ours; like clear, it NEVER kills the window.

  301 -- | The production 'ReleaseTmux': the live-marker lookup is a 'Tmux.readMarkers'
  302 -- sweep of the session filtered to the target window (returning its non-empty
  303 -- @\@pcl_id@), and the clear is 'Tmux.clearWindowMarker' (@set-option -wu@).

  320 -- | Pure core of the production live-@\@pcl_id@ lookup: from a session's
  321 -- 'Tmux.readMarkers' sweep, return the non-empty @\@pcl_id@ of the row whose
  322 -- window name matches @window@, or 'Nothing' when the window is absent or its
  323 -- marker is empty\/unset. Empty markers are treated as \"no marker\" so an
  324 -- unmarked\/cleared window never spuriously corroborates.
  325 pickLiveMarker :: Text -> [Tmux.TmuxWindowRow] -> Maybe Text
  326 pickLiveMarker window rows =
  327   Maybe.listToMaybe
  328     [ Tmux._twr_pclId r
  329     | r <- rows
  330     , Tmux._twr_windowName r == window
  331     , not (T.null (Tmux._twr_pclId r))
  332     ]

  339 -- | Resolve the tmux session name a harness should be spawned into from the
  340 -- requested 'HarnessSpec' (WU7, the epic's core fix). The session is read from
  341 -- the spec's 'TbTmux' backend '_tc_session' when it is non-empty; otherwise it
  342 -- falls back to 'defaultHarnessSession' (@"pureclaw"@). A non-tmux backend
  343 -- (the frontend's placeholder @local@ request backend) likewise resolves to the
  344 -- default. The WINDOW is auto-assigned by the spawn path (@canonical-\<idx\>@);
  345 -- honoring a caller-specified '_tc_window' for placement is deferred
  346 -- (pureclaw-jlc), so this reads only the session, never the window.

  348 resolveHarnessSession spec = case _h_backend spec of
  349   TbTmux tc
  350     | not (T.null (_tc_session tc)) -> _tc_session tc
  351   _                                 -> defaultHarnessSession

  413 -- | Shared session-ID validation used by every endpoint that consumes a
  414 -- caller-supplied session id (HTTP @\/transcript@, @\/send@, @\/prompt@
  415 -- and the WS @focus@ op). Rejects @..@ and @\/@ to foreclose path
  416 -- traversal; rejects the empty string. Behavioural surface is intentionally
  417 -- the same across the HTTP and WS paths so D26's "shared helper" property
  418 -- holds: changing the rule here changes it everywhere.
  419 isValidSessionId :: Text -> Bool

  421   | T.null sid = False
  422   | T.isInfixOf ".." sid = False
  423   | T.isInfixOf "/" sid = False
  424   | otherwise = True

  732 -- | @POST \/api\/adopt@ — adopt an external (discovered, UNMARKED) tmux window
  733 -- into the registry (Phase 3, WU4). SECURITY-CRITICAL ordering (SEC-1):
  734 -- 'authorizeAdoption' runs BEFORE any tmux mutation, so a denied request stamps
  735 -- NO @\@pcl_id@ and creates NO registry entry.
  736 --
  737 -- Steps:
  738 --
  739 --   1. Decode the body; require @consent_confirmed == true@ (else @400@).
  740 --   2. 'authorizeAdoption' (_fe_consentChannel) session:
  741 --
  742 --        * @Left AdoptNoConsentChannel@ → @403@ (a headless\/gateway\/import run
  743 --          has no human at the confirm dialog — fail-closed, the SOLE remaining
  744 --          denial; denies even a @consent_confirmed:true@ body).
  745 --        * @Right token@ → run '_fe_adopt' with the token. On success @200@; on
  746 --          a tmux\/adopt failure @500@.
  747 --
  748 -- On a successful adopt the legacy '_fe_harnesses' map is synced (D-ADD-2) so
  749 -- name-keyed routing keeps working until the session.json id is consulted.
  750 handleAdopt :: FrontendEnv -> Request -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  751 handleAdopt env req respond = do
  752   body <- consumeBody req
  753   case Aeson.eitherDecode body of
  754     Left err ->
  755       respond $ jsonResponse status400
  756         (object ["error" .= ("Invalid JSON: " <> T.pack err)])
  757     Right ar
  758       | not (_ar_consentConfirmed ar) ->
  759           respond $ jsonResponse status400
  760             (object ["error" .= ("adoption requires consent_confirmed: true" :: Text)])
  761       | otherwise ->
  762           -- SEC-1: the gate runs BEFORE any tmux mutation. On Left we respond
  763           -- and NEVER call '_fe_adopt', so nothing is stamped/registered.
  764           case authorizeAdoption (_fe_consentChannel env) (_ar_session ar) of
  765             Left AdoptNoConsentChannel ->
  766               respond $ jsonResponse status403
  767                 (object ["error" .= ("adoption requires an interactive consent channel" :: Text)])
  768             Right token -> do
  769               result <- _fe_adopt env token (_ar_window ar)
  770               case result of
  771                 Left err ->
  772                   respond $ harnessErrorResponse err
  773                 Right (hid, hh) -> do
  774                   -- D-ADD-2: sync the legacy name-keyed map so name-fallback
  775                   -- routing reaches the adopted handle immediately.
  776                   modifyIORef' (_fe_harnesses env) (Map.insert (_ar_window ar) hh)
  777                   -- Surface the PureClaw session id created for this harness (it
  778                   -- lives on the registry entry's '_he_sessionId') so the web
  779                   -- "Existing Harness" composer can drop the user into the
  780                   -- conversation and send a first message right after adopting.
  781                   -- 'Nothing' -> null.
  782                   mEntry <- Registry.lookupById (_fe_harnessRegistry env) hid
  783                   broadcastLists env
  784                   respond $ jsonResponse status200
  785                     (object
  786                       [ "adopted"     .= True
  787                       , "harness_id"  .= Registry.harnessIdToText hid
  788                       , "session"     .= _ar_session ar
  789                       , "window"      .= _ar_window ar
  790                       , "session_id"  .= (mEntry >>= Registry._he_sessionId)
  791                       ])

  858 -- | @POST \/api\/tabs\/{index}\/release@ — release an ADOPTED harness: stop
  859 -- managing it and REMOVE PureClaw's @\@pcl_id@ marker, but NEVER kill the
  860 -- window and RETAIN its transcript\/@session.json@ (design §6, §8 C2). Distinct
  861 -- from Dismiss (which removes a dead row with no tmux op) and from Close (which
  862 -- never kills either): Release issues a @set-option -wu@ on a LIVE window — but
  863 -- ONLY after re-corroborating it is still ours (SEC-3 anti-spoof).
  864 --
  865 -- SECURITY-CRITICAL ordering — re-corroboration runs BEFORE any tmux mutation:
  866 --
  867 --   1. Resolve the display index → entry (shared 'tabIndexToEntry' resolver).
  868 --   2. Release applies to a harness of ANY origin (Spawned, Adopted, or
  869 --      Discovered) — the safety control is the corroboration in step 3, not the
  870 --      origin. PureClaw stops managing the window and hands it back.
  871 --   3. Re-read the LIVE @\@pcl_id@ of the entry's @(session, windowName)@ via
  872 --      the injected seam and compare to @harnessIdToText (_he_id entry)@:
  873 --
  874 --        * MATCH → 'Tmux.clearWindowMarker' (@set-option -wu@) + retitle the
  875 --          window via 'Tmux.renameWindowNamed' to @\"… (released)\"@, then
  876 --          deregister from the registry AND the legacy '_fe_harnesses' map;
  877 --          respond @200 {"released":true}@. The window is NOT killed.
  878 --        * MISMATCH \/ window gone (the cached coord now points at a different
  879 --          \/absent window — the PID-reuse\/replacement case) → deregister from
  880 --          BOTH stores WITHOUT issuing ANY tmux @set-option@\/mutation, and log
  881 --          a refusal; respond @200 {"released":true,"note":...}@. The invariant:
  882 --          NEVER @set-option -wu@ a window we cannot confirm is ours.
  883 --
  884 -- The @purge@ body flag is accepted but RESERVED (documented no-op; retention
  885 -- only — design §8 C2).
  886 handleReleaseTab
  887   :: FrontendEnv -> Text -> Request -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  888 handleReleaseTab env tidxText req respond = do
  889   body <- consumeBody req
  890   -- The body is optional; an empty/garbage body decodes to the default
  891   -- (purge = False). The purge flag is reserved regardless, so we never fail the
  892   -- request on a bad body.
  893   let reservedPurge = case Aeson.eitherDecode body of
  894         Right rr -> _rr_purge rr
  895         Left _   -> False
  896   -- RESERVED (Phase-3 MVP = retention only, design §8 C2): accepted but a
  897   -- documented no-op. We log when it is requested so the intent is auditable,
  898   -- but we do NOT purge/delete the transcript or session.json.
  899   when reservedPurge $
  900     _lh_logInfo (_fe_logger env)
  901       "release: purge requested but RESERVED (Phase-3 MVP retains transcript/session.json)"
  902   -- Release applies to ANY harness (spawned OR adopted): PureClaw stops managing
  903   -- it but leaves the tmux window + processes running. The corroborate-before-
  904   -- mutate gate inside 'releaseHarnessEntry' is the safety control, not the
  905   -- origin.
  906   withResolvedTab env tidxText respond $ \e -> releaseHarnessEntry env e respond

  908 -- | The corroborate-then-act core of Release, for a harness of ANY origin.
  909 -- Re-reads the live @\@pcl_id@ BEFORE any mutation (SEC-3) and branches:
  910 -- corroborated → unmark + retitle the window \"… (released)\" + deregister;
  911 -- stale → deregister only (no tmux op) + warn. Either way the entry leaves both
  912 -- stores, the window survives, and the transcript\/@session.json@ are retained
  913 -- (C2).
  914 releaseHarnessEntry
  915   :: FrontendEnv -> Registry.HarnessEntry -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  916 releaseHarnessEntry env e respond = do
  917   let session  = Registry._he_session e
  918       window   = Registry._he_windowName e
  919       expected = Registry.harnessIdToText (Registry._he_id e)
  920       rt       = _fe_releaseTmux env
  921   liveMarker <- _rt_liveMarker rt session window
  922   if liveMarker == Just expected
  923     then do
  924       -- Corroborated: the live window still carries THIS entry's marker, so it
  925       -- is safe to mutate it. Unmark it (@pcl_id), then retitle it so the user
  926       -- can see PureClaw has detached. Neither op kills the window. The window
  927       -- name is PureClaw-managed (spawn: @claude-code-N@; adopt: @adopted-…@),
  928       -- so it is dot-free and the @session:name@ rename target is unambiguous.
  929       _rt_clearMarker rt session window
  930       _rt_renameWindow rt session window (releasedWindowName window)
  931       deregister env e
  932       broadcastLists env
  933       respond $ jsonResponse status200 (object ["released" .= True])
  934     else do
  935       -- Stale/uncorroborated: the cached coord no longer points at our window
  936       -- (mismatched marker or window gone — PID-reuse/replacement). We refuse to
  937       -- `set-option -wu` a window we can't confirm is ours; deregister WITHOUT
  938       -- any tmux mutation and log the refusal.
  939       _lh_logWarn (_fe_logger env) $
  940         "release: window " <> session <> ":" <> window
  941           <> " no longer corroborated (live @pcl_id="
  942           <> Maybe.fromMaybe "<none>" liveMarker
  943           <> ", expected " <> expected
  944           <> "); deregistering WITHOUT unmarking"
  945       deregister env e
  946       broadcastLists env
  947       respond $ jsonResponse status200
  948         (object
  949           [ "released" .= True
  950           , "note"     .= ("window no longer corroborated; deregistered without unmarking" :: Text)
  951           ])

  975 -- | @POST \/api\/tabs\/{index}\/destroy@ — terminate a harness: kill its tmux
  976 -- window (claude-code + the pane shell), ARCHIVE its session (transcript
  977 -- retained — design: destroy is reversible at the data layer), and deregister
  978 -- it from BOTH the registry and the legacy '_fe_harnesses' map. Distinct from
  979 -- Release (never kills), Dismiss (no tmux op, session stays in Recent), and
  980 -- Close.
  981 --
  982 -- Two guards:
  983 --
  984 --   * ADOPTED gate (server-enforced, not UI-trusted): an 'OriginAdopted' entry
  985 --     is rejected with @409@ and NO mutation unless the body carries
  986 --     @confirm_adopted: true@. Spawned\/Discovered entries (our own windows)
  987 --     destroy without confirmation.
  988 --   * SEC-3 corroborate-before-kill: the kill ('_fe_killWindow') fires ONLY
  989 --     when the live @\@pcl_id@ at the entry's cached @(session, window)@ still
  990 --     equals THIS entry's id. On a mismatch\/missing marker (window replaced or
  991 --     gone) we deregister + archive WITHOUT killing and log a refusal — we
  992 --     never kill a window we cannot confirm is ours.
  993 handleDestroyTab
  994   :: FrontendEnv -> Text -> Request -> (Response -> IO ResponseReceived) -> IO ResponseReceived
  995 handleDestroyTab env tidxText req respond = do
  996   body <- consumeBody req
  997   let confirmAdopted = case Aeson.eitherDecode body of
  998         Right dr -> _dr_confirmAdopted dr
  999         Left _   -> False
 1000   withResolvedTab env tidxText respond $ \e ->
 1001     case Registry._he_origin e of
 1002       Registry.OriginAdopted | not confirmAdopted ->
 1003         -- Server-side enforcement of the kill-adopted confirmation: the UI gate
 1004         -- is not trusted. No tmux op, no deregistration.
 1005         respond $ jsonResponse status409
 1006           (object ["error" .= ("adopted harness requires confirm_adopted" :: Text)])
 1007       _ -> destroyHarnessEntry env e respond

 1009 -- | The corroborate-then-kill core of Destroy, for an entry already cleared by
 1010 -- the adopted gate. Re-reads the live @\@pcl_id@ BEFORE killing (SEC-3); kills
 1011 -- only when corroborated, otherwise logs a refusal and skips the kill. Either
 1012 -- way the session is archived (transcript retained) and the entry leaves both
 1013 -- stores.
 1014 destroyHarnessEntry
 1015   :: FrontendEnv -> Registry.HarnessEntry -> (Response -> IO ResponseReceived) -> IO ResponseReceived
 1016 destroyHarnessEntry env e respond = do
 1017   let session  = Registry._he_session e
 1018       window   = Registry._he_windowName e
 1019       expected = Registry.harnessIdToText (Registry._he_id e)
 1020   liveMarker <- _rt_liveMarker (_fe_releaseTmux env) session window
 1021   killed <-
 1022     if liveMarker == Just expected
 1023       then do
 1024         -- Corroborated: the live window still carries THIS entry's marker, so
 1025         -- it is safe to kill.
 1026         _fe_killWindow env session window
 1027         pure True
 1028       else do
 1029         _lh_logWarn (_fe_logger env) $
 1030           "destroy: window " <> session <> ":" <> window
 1031             <> " no longer corroborated (live @pcl_id="
 1032             <> Maybe.fromMaybe "<none>" liveMarker
 1033             <> ", expected " <> expected
 1034             <> "); deregistering WITHOUT killing"
 1035         pure False
 1036   -- Archive the backing session (retain the transcript) if session-backed. A
 1037   -- missing/unparseable session is non-fatal: the window kill + deregistration
 1038   -- still stand, so we log and continue rather than fail the request.
 1039   case Registry._he_sessionId e of
 1040     Just sid -> do
 1041       archiveResult <- setArchived (_fe_sessionsDir env) (SessionId sid) True
 1042       case archiveResult of
 1043         Right () -> pure ()
 1044         Left err -> _lh_logWarn (_fe_logger env) $
 1045           "destroy: could not archive session " <> sid <> ": " <> T.pack (show err)
 1046     Nothing -> pure ()
 1047   deregister env e
 1048   broadcastLists env
 1049   respond $ jsonResponse status200 $ object $
 1050     ("destroyed" .= True)
 1051       : [ "note" .= ("window no longer corroborated; deregistered without killing" :: Text)
 1052         | not killed ]

 1075 handleRecentSessions env respond = do
 1076   let limit   = _fe_recentLimit env
 1077       baseDir = _fe_sessionsDir env
 1078   -- Over-fetch so the empty-transcript filter below can't shrink the
 1079   -- visible count: listSessions already does a full directory scan and
 1080   -- sorts the whole metadata set, so a generous bound is essentially
 1081   -- free. Empty / archived sessions are dropped here, not erased on
 1082   -- disk — archiving is purely a UI hint controlled by the user.
 1083   metas    <- listSessions baseDir Nothing (limit * 3)
 1084   -- Collect session IDs that are currently shown in an active tab so
 1085   -- the same session does not appear in both "Active Tabs" and

 1088       -- Only NON-harness tabs (provider / raw-shell, shown under "Active Tabs")

 1263 -- | Cheap fallback for display: read just the first line of the
 1264 -- session's @transcript.jsonl@, decode it as a 'TranscriptEntry', and
 1265 -- extract a short snippet of the first user message. Returns 'Nothing'
 1266 -- when there's no transcript, the first entry isn't a request, or any
 1267 -- decoding step fails. The returned string is at most
 1268 -- 'snippetCharBudget' chars with newlines normalised to spaces.
 1269 --
 1270 -- Cost is one bounded read per call (we don't load the whole transcript).
 1271 firstMessageSnippet :: FilePath -> SessionMeta -> IO (Maybe Text)
 1272 firstMessageSnippet baseDir meta = do
 1273   let path = baseDir </> T.unpack (unSessionId (_sm_id meta)) </> "transcript.jsonl"
 1274   result <- try @IOException $ withFile path ReadMode BSC.hGetLine
 1275   case result of
 1276     Left _    -> pure Nothing
 1277     Right line -> case Aeson.eitherDecodeStrict' line :: Either String TranscriptEntry of
 1278       Left _ -> pure Nothing
 1279       Right entry
 1280         | _te_direction entry /= Request -> pure Nothing
 1281         | otherwise -> pure (snippetFromPayload (_te_payload entry))
 1282 
 1283 snippetCharBudget :: Int
 1284 snippetCharBudget = 120
 1285 
 1286 -- | Extract a display snippet from a request payload. Two shapes:
 1287 -- (1) JSON object with a "messages" array (provider request) — pull
 1288 --     text out of the first message; (2) plain text (harness send) —
 1289 --     use the payload directly. Trimmed, newline-normalised, and
 1290 --     truncated to 'snippetCharBudget' characters.
 1291 snippetFromPayload :: Text -> Maybe Text
 1292 snippetFromPayload raw = trimAndTruncate <$>
 1293   case Aeson.decodeStrict (TE.encodeUtf8 raw) of
 1294     Just (Aeson.Object o)
 1295       | Just (Aeson.Array msgs) <- KM.lookup "messages" o
 1296       , not (V.null msgs)
 1297       -> messageText (V.unsafeHead msgs)
 1298     _ -> Just raw
 1299   where
 1300     -- Anthropic-style message content: either a plain string or an
 1301     -- array of {type:"text", text:"..."} blocks. We just want a
 1302     -- human-readable lead.
 1303     messageText :: Aeson.Value -> Maybe Text
 1304     messageText (Aeson.Object m) = case KM.lookup "content" m of
 1305       Just (Aeson.String s) -> Just s
 1306       Just (Aeson.Array bs) -> Just (T.intercalate " " [t | Aeson.Object b <- V.toList bs
 1307                                                           , Just (Aeson.String t) <- [KM.lookup "text" b]])
 1308       _                     -> Nothing
 1309     messageText _ = Nothing
 1310 
 1311     trimAndTruncate :: Text -> Text
 1312     trimAndTruncate t =
 1313       let normalized = T.unwords (T.words t)  -- collapse whitespace incl. newlines
 1314       in  if T.length normalized > snippetCharBudget
 1315             then T.take (snippetCharBudget - 1) normalized <> "\x2026"
 1316             else normalized

 1549 -- with 'Nothing' the behaviour is identical to a fresh New-tab session.
 1550 createTab :: FrontendEnv -> TabKind -> Maybe BranchSeed -> (Response -> IO ResponseReceived) -> IO ResponseReceived
 1551 createTab env tabKind mSeed respond = do
 1552   -- Read the slot index for the response; the actual bump is deferred for
 1553   -- the harness path so a failed tmux spawn never consumes a slot (the
 1554   -- count has no decrement). Provider / raw-shell paths bump immediately,
 1555   -- preserving their existing behaviour byte-for-byte.
 1556   curCount <- readIORef (_fe_tabCount env)
 1557   case tabKind of
 1558     -- Harness sessions must actually spawn the tmux harness before the tab
 1559     -- is considered created. A branch seed only ever targets a provider
 1560     -- session (handleNewTab rejects non-provider branch targets), so a
 1561     -- seeded harness cannot occur here.
 1562     TkSession sk@(SkHarness spec) ->
 1563       createHarnessTab env tabKind spec sk curCount respond

 1565       -- Bump tab count
 1566       writeIORef (_fe_tabCount env) (curCount + 1)

 1569       mModel <- readIORef (_fe_model env)

 1571           sid = newSessionId Nothing now

 1573           -- frontend uses it as the display name for the session

 1575           -- definition was already used to build the system prompt.
 1576           -- Without this, the metadata loses the association and
 1577           -- recent-sessions UI falls back to the session id. Harness

 1581             SkProvider ps -> _ps_agent ps

 1583           -- the source meta so the branch's sidebar row matches its
 1584           -- parent; otherwise use the request kind + global model.

 1586             Just seed ->
 1587               let srcMeta = _bseed_sourceMeta seed
 1588               in ( _sm_kind srcMeta, _sm_model srcMeta, _sm_agent srcMeta )
 1589             Nothing -> (sk, modelText, agentName)
 1590           meta = SessionMeta
 1591             { _sm_id                = sid
 1592             , _sm_agent             = metaAgent
 1593             , _sm_kind              = metaKind

 1595             , _sm_channel           = "web"
 1596             , _sm_createdAt         = now
 1597             , _sm_lastActive        = now
 1598             , _sm_bootstrapConsumed = True
 1599             , _sm_archived          = False
 1600             , _sm_description       = Nothing
 1601             , _sm_autoSummary       = Nothing
 1602             , _sm_source            = Nothing
 1603             }
 1604       sh <- mkSessionHandle (_fe_broker env) (_fe_logger env) (_fe_sessionsDir env) meta

 1606       -- For a branch, seed the new transcript with a verbatim copy of the
 1607       -- source prefix (preserving _te_id and order). The frozen system prompt

 1609       -- docs/session-branching.md §9), so the fork does NOT copy the source's

 1627         { _ntresp_tabIndex  = curCount
 1628         , _ntresp_sessionId = Just (unSessionId sid)
 1629         , _ntresp_kind      = tabKindLabel tabKind
 1630         }

 1640 -- | Create a harness-backed tab: persist the session, spawn the tmux
 1641 -- harness via '_fe_startHarness', and only on success persist the real
 1642 -- tmux coordinates into '_sm_kind' and bump the tab count. A failed spawn
 1643 -- removes the just-created session directory and leaves the tab count
 1644 -- untouched (it has no decrement), so a fallible spawn never strands a
 1645 -- session dir or burns a tab slot.
 1646 createHarnessTab
 1647   :: FrontendEnv
 1648   -> TabKind        -- ^ original request kind (for the response label)
 1649   -> HarnessSpec    -- ^ requested harness spec (backend replaced on success)
 1650   -> SessionKind    -- ^ the SkHarness kind to seed _sm_kind with
 1651   -> Int            -- ^ current tab count = response slot index
 1652   -> (Response -> IO ResponseReceived)
 1653   -> IO ResponseReceived
 1654 createHarnessTab env tabKind spec sk curCount respond = do
 1655   now <- getCurrentTime
 1656   let sid  = newSessionId Nothing now
 1657       meta = SessionMeta
 1658         { _sm_id                = sid
 1659         , _sm_agent             = Nothing
 1660         , _sm_kind              = sk
 1661         , _sm_model             = ""
 1662         , _sm_channel           = "web"
 1663         , _sm_createdAt         = now
 1664         , _sm_lastActive        = now
 1665         , _sm_bootstrapConsumed = True
 1666         , _sm_archived          = False
 1667         , _sm_description       = Nothing
 1668         , _sm_autoSummary       = Nothing
 1669         , _sm_source            = Nothing
 1670         }
 1671   sh <- mkSessionHandle (_fe_broker env) (_fe_logger env) (_fe_sessionsDir env) meta
 1672   _sh_save sh
 1673   -- Pass a NON-recording transcript to the harness so it does not
 1674   -- double-record entries: 'handleSend' is the SOLE recorder of harness
 1675   -- Request/Response entries to the session transcript (WU3). Recording
 1676   -- here too would duplicate every fresh-session entry.
 1677   started <- _fe_startHarness env spec mkNoOpTranscriptHandle
 1678   case started of
 1679     Left err -> do
 1680       -- Roll back the session dir created above. A missing dir must not
 1681       -- throw, so swallow IO errors from the cleanup.
 1682       _ <- try @IOException (removeDirectoryRecursive (_sh_dir sh))
 1683       respond $ harnessErrorResponse err
 1684     Right st -> do
 1685       -- Persist the real tmux coordinates AND the durable HarnessId returned by
 1686       -- the spawn (WU7, D7.2). The id is the primary routing anchor (resolved
 1687       -- through the registry by 'harnessKeyFromKind'/'sendToHarness'); the tmux
 1688       -- window/session is dual-written for the name-fallback path. The persisted
 1689       -- '_tc_session' is the resolved session honored by '_fe_startHarness'.
 1690       modifyIORef' (_sh_meta sh) $ \m ->
 1691         m { _sm_kind = SkHarness
 1692               (spec { _h_backend           = TbTmux (_shh_tmux st)
 1693                     , _h_harnessId         = Just (_shh_id st)
 1694                     , _h_claudeSessionUuid = _shh_claudeSessionUuid st
 1695                     , _h_canonicalCwd      = _shh_canonicalCwd st
 1696                     }) }
 1697       _sh_save sh
 1698       -- Link the freshly-spawned harness's registry entry back to THIS session.
 1699       -- The low-level spawn (startHarnessByName) inserts the entry with
 1700       -- '_he_sessionId = Nothing' because it has no session of its own; only this
 1701       -- orchestrator knows both the 'HarnessId' and the 'sid'. Without the link
 1702       -- the tab's session_id is null and the right pane shows "No session
 1703       -- associated yet" (the adopt path already links via '_he_sessionId'). Set
 1704       -- it before 'broadcastLists' so the live lists snapshot already carries it.
 1705       Registry.modifyEntry (_fe_harnessRegistry env) (_shh_id st)
 1706         (\e -> e { Registry._he_sessionId = Just (unSessionId sid) })
 1707       -- Read back the post-spawn meta so the live broadcast carries the real
 1708       -- TbTmux backend (the persisted session.json above is already correct;
 1709       -- the original 'meta' still holds the placeholder backend).
 1710       updatedMeta <- readIORef (_sh_meta sh)
 1711       -- Only a successful spawn consumes a tab slot.
 1712       writeIORef (_fe_tabCount env) (curCount + 1)
 1713       case _fe_broker env of
 1714         Just broker ->
 1715           _streamBroker_publish broker
 1716             (ActivityChanged (_sm_id updatedMeta) (SaSessionCreated updatedMeta))
 1717         Nothing -> pure ()
 1718       broadcastLists env
 1719       respond $ jsonResponse status200 NewTabResponse
 1720         { _ntresp_tabIndex  = curCount
 1721         , _ntresp_sessionId = Just (unSessionId sid)
 1722         , _ntresp_kind      = tabKindLabel tabKind
 1723         }

 1761 -- | Set or replace the custom prompt for a session (WU6, §9.3).
 1762 --
 1763 -- Enforces the @custom-prompt.md ⊕ agent@ invariant: a session may have a
 1764 -- per-session agent /or/ a user-supplied custom prompt, never both. Setting
 1765 -- a custom prompt therefore:
 1766 --
 1767 --   * clears @_sm_agent@ (the custom prompt supersedes the agent);
 1768 --   * stores an optionally-provided @name@ in @_sm_description@ (a friendly
 1769 --     title), /not/ as the agent;
 1770 --   * leaves all other metadata fields untouched.
 1771 --
 1772 -- __Ordering / atomicity:__ the metadata is read, updated, and written
 1773 -- (atomic tmp + rename) /before/ @custom-prompt.md@ is written. If the
 1774 -- existing @session.json@ is missing or undecodable, no @custom-prompt.md@
 1775 -- is written, so a meta failure can never leave a custom prompt sitting
 1776 -- alongside a still-set agent.
 1777 handleSetPrompt :: FrontendEnv -> Text -> Request -> (Response -> IO ResponseReceived) -> IO ResponseReceived
 1778 handleSetPrompt env sid req respond = do
 1779   if not (isValidSessionId sid)
 1780     then respond $ jsonResponse status400 (object ["error" .= ("Invalid session ID" :: Text)])
 1781     else do
 1782       let sessionDir = _fe_sessionsDir env </> T.unpack sid
 1783           promptPath = sessionDir </> "custom-prompt.md"
 1784           metaPath   = sessionDir </> "session.json"
 1785       body <- consumeBody req
 1786       case Aeson.eitherDecode body of
 1787         Left _ ->
 1788           respond $ jsonResponse status400 (object ["error" .= ("Invalid JSON: expected {\"prompt\": \"...\"}" :: Text)])
 1789         Right obj -> case Map.lookup ("prompt" :: Text) (obj :: Map.Map Text Text) of
 1790           Nothing ->
 1791             respond $ jsonResponse status400 (object ["error" .= ("Missing 'prompt' field" :: Text)])
 1792           Just prompt -> do
 1793             -- An optionally-provided friendly title, trimmed; an
 1794             -- all-whitespace value is treated as "no description".
 1795             let mDescription = do
 1796                   name <- Map.lookup "name" obj
 1797                   let trimmed = T.strip name
 1798                   if T.null trimmed then Nothing else Just trimmed
 1799             -- Step 1: read + update + atomically rewrite session.json FIRST.
 1800             -- Clear the agent (invariant) and set the description from name.
 1801             metaExists <- doesFileExist metaPath
 1802             if not metaExists
 1803               then respondNotFound respond
 1804               else do
 1805                 eBytes <- try (LBS.readFile metaPath) :: IO (Either IOException LBS.ByteString)
 1806                 case eBytes of
 1807                   Left e ->
 1808                     respond $ jsonResponse status500
 1809                       (object ["error" .= T.pack (show e)])
 1810                   Right raw -> case Aeson.eitherDecode' raw of
 1811                     Left err ->
 1812                       respond $ jsonResponse status500
 1813                         (object ["error" .= T.pack err])
 1814                     Right meta -> do
 1815                       let updated = (meta :: SessionMeta)
 1816                             { _sm_agent       = Nothing
 1817                             , _sm_description = mDescription
 1818                             }
 1819                           tmpP = metaPath <> ".tmp"
 1820                       LBS.writeFile tmpP (Aeson.encode updated)
 1821                       renameFile tmpP metaPath
 1822                       -- Step 2: only after the meta update succeeds do we
 1823                       -- write the custom prompt file. Now the agent is
 1824                       -- guaranteed cleared, so no contradiction can exist.
 1825                       TIO.writeFile promptPath prompt
 1826                       respond $ jsonResponse status200 (object ["ok" .= True])

 1892 -- | Route a user message to a live tmux harness handle (WU3, defect #2).
 1893 -- This is the harness analogue of 'doCompletion': it is the SOLE recorder
 1894 -- of the harness Request\/Response transcript entries (the harness handle
 1895 -- itself is given a no-op transcript so it does not double-record), it
 1896 -- broadcasts through the same broadcasting transcript handle 'doCompletion'
 1897 -- uses, and it bumps @_sm_lastActive@ and re-broadcasts the lists snapshot
 1898 -- on success.
 1899 --
 1900 -- If no live handle is registered under @key@ (e.g. the harness crashed or
 1901 -- was never reconnected after a restart) it returns a clear 503 — never a
 1902 -- silent provider completion and never a 500.
 1903 --
 1904 -- WU6 makes routing /id-primary with a name fallback/. Resolution order:
 1905 --
 1906 --   1. If @key@ parses as a 'Registry.HarnessId', look it up in the registry:
 1907 --
 1908 --        * Found AND PID-corroborated AND it has a live handle -> route to that
 1909 --          handle (the durable, rename-proof path).
 1910 --        * Found AND corroborated but with NO handle (e.g. boot-discovered, not
 1911 --          yet attached) -> fall through to the name path.
 1912 --        * Found but NOT PID-corroborated -> REFUSE (respond 503, log a
 1913 --          refusal, send NO keystrokes). A spoofable @\@pcl_id@ marker on a
 1914 --          window with no recorded PID is treated as "not ours" (§8 C4 / D6.6).
 1915 --          This case does NOT fall back to the name path.
 1916 --        * Not found in the registry -> fall back to the name path (the registry
 1917 --          is empty until later work units populate it).
 1918 --
 1919 --   2. Name path (legacy, PR #74): look the @fallbackName@ (the dual-written
 1920 --      tmux window name) up in '_fe_harnesses'. On a match, route; on a miss,
 1921 --      respond 503 "not running". When a corroborated registry entry matches
 1922 --      this name by label, lazily back-fill its 'Registry.HarnessId' into
 1923 --      @session.json@ (D6.5) so subsequent sends route by id.
 1924 sendToHarness
 1925   :: FrontendEnv
 1926   -> Text         -- ^ session id
 1927   -> Text         -- ^ routing key (a 'Registry.HarnessId' text, or a window name)
 1928   -> Maybe Text   -- ^ fallback tmux window name (the dual-written legacy key)
 1929   -> Text         -- ^ user message text
 1930   -> FilePath     -- ^ session transcript path
 1931   -> (Response -> IO ResponseReceived)
 1932   -> IO ResponseReceived
 1933 sendToHarness env sid key fallbackName userText transcriptPath respond =
 1934   case Registry.parseHarnessId key of
 1935     Just hid -> do
 1936       mEntry <- Registry.lookupById (_fe_harnessRegistry env) hid
 1937       case mEntry of
 1938         Just entry
 1939           | not (isPidCorroborated entry) -> do
 1940               -- D6.6 (§8 C4): a spoofable marker without recorded PID provenance
 1941               -- is NOT ours. Refuse outright — never send keystrokes, never fall
 1942               -- back to the name map for this case.
 1943               _lh_logError (_fe_logger env) $
 1944                 "Refusing to route session '" <> sid <> "' to harness id '"
 1945                 <> key <> "': registry entry is not PID-corroborated (possible "
 1946                 <> "spoofed @pcl_id marker — §8 C4)."
 1947               respond $ jsonResponse status503 (object
 1948                 [ "error" .= ("harness id is not PID-corroborated; refusing to route" :: Text) ])
 1949           | Just hh <- Registry._he_handle entry ->
 1950               -- Corroborated, with a live handle: the durable id path. The id
 1951               -- is already durable here, so no back-fill is needed.
 1952               routeViaHandle env sid hh userText transcriptPath (pure ()) respond
 1953           | otherwise ->
 1954               -- Corroborated but no attached handle yet: fall through to name.
 1955               routeViaName env sid fallbackName userText transcriptPath respond
 1956         -- Unknown id: the registry has no entry yet -> legacy name fallback.
 1957         Nothing -> routeViaName env sid fallbackName userText transcriptPath respond
 1958     -- Key is not id-shaped: a legacy window-name key. Route by name directly.
 1959     Nothing -> routeViaName env sid (Just key) userText transcriptPath respond

 1961 -- | The legacy name-keyed routing path (PR #74). Looks @mName@ up in
 1962 -- '_fe_harnesses'; on a match, routes via that handle (and lazily back-fills a
 1963 -- 'Registry.HarnessId' if a corroborated entry matches by label — D6.5); on a
 1964 -- miss (or no name available), responds 503 "not running".
 1965 routeViaName
 1966   :: FrontendEnv
 1967   -> Text
 1968   -> Maybe Text
 1969   -> Text
 1970   -> FilePath
 1971   -> (Response -> IO ResponseReceived)
 1972   -> IO ResponseReceived
 1973 routeViaName env sid mName userText transcriptPath respond = do
 1974   handles <- readIORef (_fe_harnesses env)
 1975   case mName >>= (`Map.lookup` handles) of
 1976     Nothing ->
 1977       respond $ jsonResponse status503
 1978         (object ["error" .= ("harness '" <> Maybe.fromMaybe "?" mName

 1991 -- | Route a user message to a resolved 'HarnessHandle': record the
 1992 -- Request\/Response transcript entries (sole writer of this session's
 1993 -- transcript), broadcast, bump @_sm_lastActive@, and respond. Shared by the
 1994 -- id and name resolution paths.
 1995 --
 1996 -- @postSend@ is a synchronous-but-exception-proof action run ONLY after a
 1997 -- successful send (the name path uses it for the D6.5 id back-fill). It runs
 1998 -- synchronously before @respond@ on the success branch — so it can add minor
 1999 -- latency to a successful response but never delays a failed send; callers MUST
 2000 -- make it exception-proof (see 'backfillHarnessId') so it cannot fail the send.
 2001 routeViaHandle
 2002   :: FrontendEnv
 2003   -> Text
 2004   -> HarnessHandle
 2005   -> Text
 2006   -> FilePath
 2007   -> IO ()        -- ^ post-send hook (run only on success; must not throw)
 2008   -> (Response -> IO ResponseReceived)
 2009   -> IO ResponseReceived
 2010 routeViaHandle env sid hh userText transcriptPath postSend respond = do
 2011   -- 'handleSend' is the sole writer of THIS session's transcript: on the
 2012   -- fresh-create path the harness handle was given a no-op transcript
 2013   -- ('createHarnessTab'), and a restart-discovered handle records to the
 2014   -- CLI's own session transcript (a different file), so recording here never
 2015   -- duplicates an entry in this session transcript. 'bracket' guarantees the
 2016   -- transcript fd is flushed + closed even if '_hh_send'/'_hh_receive' throws
 2017   -- (tmux IO), so a failed send cannot leak a file descriptor.
 2018   result <- try @SomeException $
 2019     bracket
 2020       (mkBroadcastingFileTranscriptHandle
 2021          (_fe_broker env) (SessionId sid) (_fe_logger env) transcriptPath)
 2022       (\th -> _th_flush th >> _th_close th)
 2023       (\th -> do
 2024         -- Record the user message as a Request entry (mirrors 'harnesseSend').
 2025         recordHarnessEntry th Request userText
 2026         _hh_send hh (TE.encodeUtf8 userText)
 2027         raw <- _hh_receive hh
 2028         let resp = sanitizeHarnessOutput (TE.decodeUtf8 raw)
 2029         -- Only record a Response entry when the harness produced output;
 2030         -- a blank reply must not leave an empty/duplicate entry.
 2031         unless (T.null (T.strip resp)) $
 2032           recordHarnessEntry th Response resp
 2033         pure resp)
 2034   case result of
 2035     Left e -> do
 2036       _lh_logError (_fe_logger env) $ "Harness send error: " <> T.pack (show e)
 2037       respond $ jsonResponse status500 (object ["error" .= ("Harness send failed" :: Text)])
 2038     Right resp -> do
 2039       touchSessionLastActive (_fe_sessionsDir env) (SessionId sid)
 2040       broadcastLists env
 2041       -- Synchronous-but-exception-proof post-send hook (e.g. the D6.5 id
 2042       -- back-fill). It runs synchronously here, before 'respond' — so it can add
 2043       -- minor latency — but is exception-proof at the source, so it cannot fail
 2044       -- the send the user is about to receive.
 2045       postSend
 2046       respond $ jsonResponse status200 (object ["response" .= resp])

 2048 -- | Lazy migration (D6.5): if a corroborated registry entry's label matches
 2049 -- @windowName@, persist its 'Registry.HarnessId' onto the session's
 2050 -- 'HarnessSpec' in @session.json@ so subsequent sends route by id. Operates
 2051 -- directly on the on-disk meta (read-modify-write with a tmp-file + atomic
 2052 -- rename), mirroring the disk-only meta mutators in "PureClaw.Session.Handle".
 2053 --
 2054 -- Best-effort and exception-proof (D6.5c): a missing\/corrupt @session.json@,
 2055 -- an absent\/uncorroborated entry, or an already-id-bearing spec are all silent
 2056 -- no-ops, and ANY IO fault in the read-modify-write (a TOCTOU between
 2057 -- 'doesFileExist' and 'LBS.readFile', a permission error, a disk-full or
 2058 -- cross-device error on write\/'renameFile', or a lazy-read fault surfacing in
 2059 -- 'Aeson.decode') is caught, logged, and swallowed. Keeping routing fast must
 2060 -- never fail a send the user already received, so this function NEVER throws.
 2061 backfillHarnessId :: FrontendEnv -> Text -> Text -> IO ()
 2062 backfillHarnessId env sid windowName = do
 2063   result <- try @SomeException go
 2064   case result of
 2065     Left e ->
 2066       _lh_logError (_fe_logger env) $
 2067         "harnessId back-fill failed (non-fatal): " <> T.pack (show e)
 2068     Right () -> pure ()
 2069   where
 2070     go :: IO ()
 2071     go = do
 2072       mEntry <- Registry.lookupByLabel (_fe_harnessRegistry env) windowName
 2073       case mEntry of
 2074         Just entry | isPidCorroborated entry -> do
 2075           let p = _fe_sessionsDir env </> T.unpack sid </> "session.json"
 2076           ok <- doesFileExist p
 2077           when ok $ do
 2078             raw <- LBS.readFile p
 2079             case Aeson.decode raw of
 2080               Just meta -> case _sm_kind meta of
 2081                 SkHarness hs
 2082                   | Maybe.isNothing (_h_harnessId hs) -> do
 2083                       let hs'   = hs { _h_harnessId = Just (Registry._he_id entry) }
 2084                           meta' = meta { _sm_kind = SkHarness hs' }
 2085                           -- A back-fill-specific tmp suffix, distinct from the
 2086                           -- @session.json.tmp@ used by 'touchSessionLastActive'
 2087                           -- /'updateSessionMeta', so the post-send back-fill can
 2088                           -- never race that mutator's tmp file on the same dir.
 2089                           tmpP  = p <> ".backfill.tmp"
 2090                       -- tmp-file + rename: a crash mid-write leaves the prior
 2091                       -- session.json intact (same atomicity as 'saveMeta').
 2092                       LBS.writeFile tmpP (Aeson.encode meta')
 2093                       renameFile tmpP p
 2094                 _ -> pure ()
 2095               Nothing -> pure ()
 2096         _ -> pure ()

 2116 -- | Run a completion: load context, send to provider, execute any tool
 2117 -- calls the model emits, and loop until the model produces a turn with
 2118 -- no tool calls (or the iteration cap is hit). The transcript handle is
 2119 -- opened via 'mkBroadcastingFileTranscriptHandle' so each iteration's
 2120 -- provider-wrapper Request\/Response entries reach the broker (D25) when
 2121 -- one is configured on the 'FrontendEnv'.
 2122 doCompletion
 2123   :: FrontendEnv
 2124   -> SessionId
 2125   -> SomeProvider
 2126   -> Maybe Text  -- ^ Request-supplied model (WU5, §9.2), if any.
 2127   -> ModelId     -- ^ Global fallback model (the @_fe_model@ IORef value).
 2128   -> Text
 2129   -> FilePath
 2130   -> IO Text
 2131 doCompletion env sid provider reqModel fallbackModel userText transcriptPath = do
 2132   -- Open a (possibly broadcasting) transcript handle for recording.
 2133   th <- mkBroadcastingFileTranscriptHandle
 2134           (_fe_broker env) sid (_fe_logger env) transcriptPath
 2135   -- Resolve the per-session model with precedence (§9.2): the request's
 2136   -- explicit model, else the most-recent model recorded in the transcript
 2137   -- (the "rides in the transcript" fallback — read from the structured
 2138   -- '_te_model' column, which 'loadRecentMessages' discards), else the
 2139   -- global '_fe_model' fallback. The chosen 'ModelId' flows into
 2140   -- 'mkTranscriptProvider' below so '_te_model' is recorded automatically.
 2141   -- A blank/whitespace request model (e.g. {"model":""}) is treated as
 2142   -- absent so it falls through to the transcript/global fallback rather
 2143   -- than sending an empty model id to the provider.
 2144   model <- case reqModel of
 2145     Just m | not (T.null (T.strip m)) -> pure (ModelId m)
 2146     _ -> do
 2147       mTranscriptModel <- lastRequestModel th
 2148       pure $ maybe fallbackModel ModelId mTranscriptModel
 2149   -- Load recent messages for context
 2150   history <- loadRecentMessages th 50 100000
 2151   -- The system prompt is FROZEN per session: computed once on the first
 2152   -- turn and thereafter replayed from the transcript unchanged (§9.1).
 2153   -- 'frozenSystemPrompt' reads the last recorded Request's system_prompt:
 2154   --   * Just frozen ⇒ a prior turn exists ⇒ reuse it verbatim (including a
 2155   --     frozen-null, which must stay null — do NOT recompute, do NOT
 2156   --     re-read custom-prompt.md).
 2157   --   * Nothing      ⇒ first turn ⇒ compute with precedence
 2158   --                    custom-prompt.md → per-session agent → global.
 2159   let sessionDir = takeDirectory transcriptPath
 2160   frozen <- frozenSystemPrompt th
 2161   systemPrompt <- case frozen of
 2162     Just p  -> pure p
 2163     Nothing -> computeFirstTurnPrompt env sessionDir
 2164   -- Build context with system prompt and history
 2165   let ctx0 = foldl (flip addMessage) (emptyContext systemPrompt) history
 2166       userMsg = textMessage User userText
 2167       ctx = addMessage userMsg ctx0
 2168       -- Wrap provider with transcript logging
 2169       provider' = mkTranscriptProvider th (unModelId model) Nothing provider
 2170       reg = _fe_registry env
 2171       tools = registryDefinitions reg
 2172       cap = _fe_maxToolIterations env
 2173   -- Publish session-thinking activity events so the FE sidebar / chat
 2174   -- area can show a live indicator while the provider call is in flight.
 2175   -- For harness-backed sessions, the equivalent signal comes from the
 2176   -- 2s tmux probe loop in 'PureClaw.Frontend.ActivityProbe'; provider
 2177   -- sessions don't go through the harness map, so the only place we
 2178   -- can hook this is at the request boundary in 'doCompletion'.
 2179   -- 'bracket_' guarantees the Idle event fires even if 'complete' throws.
 2180   let publishStatus s = case _fe_broker env of
 2181         Just broker -> _streamBroker_publish broker
 2182           (ActivityChanged sid (SaHarnessStatus s))
 2183         Nothing -> pure ()
 2184   result <- bracket_
 2185               (publishStatus HarnessThinking)
 2186               (publishStatus HarnessIdle)
 2187               (runCompletionLoop env provider' model tools reg cap ctx)
 2188   _th_flush th
 2189   _th_close th
 2190   -- Bump the session's on-disk _sm_lastActive so the sidebar "age" pill
 2191   -- reflects this completion. The gateway records through a raw
 2192   -- broadcasting transcript handle, which bypasses the touchLastActive
 2193   -- wrapper that live SessionHandles get, so we update the metadata
 2194   -- explicitly here and re-broadcast the lists snapshot to subscribers.
 2195   touchSessionLastActive (_fe_sessionsDir env) sid
 2196   broadcastLists env
 2197   pure result
