    1 -- | On-demand discovery of adoptable (PureClaw-UNMARKED) tmux windows.
    2 --
    3 -- Discovery is the read-only, user-invoked precursor to adoption (Phase 3,
    4 -- design @docs\/harness-registry.md@ §6, §8 B2\/C1). It answers "which external
    5 -- windows COULD I adopt?" and nothing more:
    6 --
    7 --   * __Lists ALL sessions.__ The adoption allow-list was deliberately dropped
    8 --     (the user picking a session in the foreground New-Tab form IS the
    9 --     consent — enforced at adopt time by 'PureClaw.Security.Adoption'). So a
   10 --     scan enumerates every tmux session and surfaces every UNMARKED, live
   11 --     window regardless of session name. Consent gates adoption, not discovery.
   12 --   * __Metadata-only (§8 C1), by construction.__ A scan returns a transient
   13 --     list of 'DiscoverableWindow' values. A 'DiscoverableWindow' carries NO
   14 --     handle and NO capture capability, so a discovered candidate is
   15 --     structurally incapable of having its pane captured. The scan function
   16 --     itself is only given a @readMarkers@-shaped seam (window metadata) and a
   17 --     @listSessions@-shaped seam — it has no way to call @capture-pane@.
   18 --
   19 -- Discovered candidates are NOT registry entries: they are never inserted into
   20 -- the 'PureClaw.Harness.Registry' and are never managed until an explicit,
   21 -- consent-gated adoption (WU4) promotes one to an @OriginAdopted@ entry.
   22 module PureClaw.Harness.Discovery
   23   ( -- * Transient discovery result (no handle, no capture)
   24     DiscoverableWindow (..)
   25     -- * Bounded, metadata-only scan
   26   , scanDiscoverable
   27   , scanDiscoverableIO
   28   ) where
   29 
   30 import Data.Aeson (ToJSON (..), object, (.=))
   31 import Data.Text (Text)
   32 
   33 import PureClaw.Harness.Tmux (TmuxWindowRow (..), listTmuxSessions, readMarkers)
   34 
   35 -- | A single adoptable candidate window surfaced by a discovery scan.
   36 --
   37 -- This is a TRANSIENT, metadata-only value — deliberately NOT a
   38 -- 'PureClaw.Harness.Registry.HarnessEntry'. It carries no 'HarnessHandle' and
   39 -- no capture field, so a discovered candidate cannot be captured or managed
   40 -- (§8 C1 by construction). It is the projection of a tmux window's identifying
   41 -- metadata that the adopt UI needs to present a candidate to the user.
   42 data DiscoverableWindow = DiscoverableWindow
   43   { _dw_session     :: !Text          -- ^ owning tmux session name
   44   , _dw_windowName  :: !Text          -- ^ @#{window_name}@ (display only)
   45   , _dw_windowIndex :: !Int           -- ^ @#{window_index}@ within the session
   46   , _dw_panePid     :: !(Maybe Int)   -- ^ @#{pane_pid}@ (shell PID), if known
   47   }
   48   deriving stock (Eq, Show)
   49 
   50 -- | JSON shape consumed by the discovery endpoint / frontend. Snake-cased
   51 -- keys: @session@, @window_name@, @window_index@, @pane_pid@.
   52 instance ToJSON DiscoverableWindow where
   53   toJSON dw = object
   54     [ "session"      .= _dw_session dw
   55     , "window_name"  .= _dw_windowName dw
   56     , "window_index" .= _dw_windowIndex dw
   57     , "pane_pid"     .= _dw_panePid dw
   58     ]
   59 
   60 -- | Scan for adoptable windows across ALL sessions, METADATA-ONLY.
   61 --
   62 -- Seamed for testability and for the §8 C1 structural guarantee — the function
   63 -- is given only:
   64 --
   65 --   * a @readMarkers@-shaped seam returning per-window metadata rows, and
   66 --   * a @listSessions@-shaped seam returning the server's session names.
   67 --
   68 -- It has NO capture-pane seam, so it is structurally incapable of capturing a
   69 -- pane. Logic:
   70 --
   71 --   1. list all sessions (the allow-list was dropped — consent gates adoption,
   72 --      not discovery);
   73 --   2. for each session, read its window metadata rows;
   74 --   3. keep rows that are UNMARKED (@_twr_pclId == ""@ — not already ours) AND
   75 --      not @pane_dead@;
   76 --   4. project each surviving row into a 'DiscoverableWindow'.
   77 --
   78 -- Pure-ish: all IO is in the injected seams, so the filtering logic is fully
   79 -- unit-testable.
   80 scanDiscoverable
   81   :: (Text -> IO [TmuxWindowRow])      -- ^ @readMarkers@-shaped seam
   82   -> IO [Text]                         -- ^ @listTmuxSessions@-shaped seam
   83   -> IO [DiscoverableWindow]
   84 scanDiscoverable readMarkersSeam listSessionsSeam = do
   85   sessions <- listSessionsSeam
   86   concat <$> mapM scanSession sessions
   87   where
   88     scanSession :: Text -> IO [DiscoverableWindow]
   89     scanSession session = do
   90       rows <- readMarkersSeam session
   91       pure [ toDiscoverable session row | row <- rows, isAdoptable row ]
   92 
   93     isAdoptable :: TmuxWindowRow -> Bool
   94     isAdoptable row = _twr_pclId row == "" && not (_twr_paneDead row)
   95 
   96     toDiscoverable :: Text -> TmuxWindowRow -> DiscoverableWindow
   97     toDiscoverable session row = DiscoverableWindow
   98       { _dw_session     = session
   99       , _dw_windowName  = _twr_windowName row
  100       , _dw_windowIndex = _twr_windowIndex row
  101       , _dw_panePid     = _twr_panePid row
  102       }
  103 
  104 -- | Production wrapper: scan the real tmux server (ALL sessions) using the real
  105 -- metadata-only tmux seams. Wires 'readMarkers' and 'listTmuxSessions' into
  106 -- 'scanDiscoverable'. Like 'scanDiscoverable', it never captures a pane.
  107 scanDiscoverableIO :: IO [DiscoverableWindow]
  108 scanDiscoverableIO = scanDiscoverable readMarkers listTmuxSessions
