   49 -- | A Telegram channel backed by a message queue. Updates are pushed

   57   , _tch_log      :: LogHandle

   61 -- | Create a Telegram channel with an empty inbox.
   62 mkTelegramChannel :: TelegramConfig -> NetworkHandle -> LogHandle -> IO TelegramChannel
   63 mkTelegramChannel config nh lh = do
   64   inbox <- newTQueueIO
   65   chatRef <- newIORef Nothing
   66   pure TelegramChannel
   67     { _tch_config   = config
   68     , _tch_inbox    = inbox
   69     , _tch_network  = nh
   70     , _tch_log      = lh
   71     , _tch_lastChat = chatRef
   72     }

   79 -- | Activate a Telegram channel with allow-list-warning handling.
   80 --
   81 -- Unlike Signal there is no reader thread and no transport to close: Telegram
   82 -- updates are pushed into the inbox by an external webhook. This wrapper is
   83 -- the SINGLE home for the Telegram open-allow-list warning — it fires whenever
   84 -- Telegram is activated; a future gateway boot path routes through it.
   85 -- 'mkTelegramChannel' itself stays warning-free (no test-construction noise).
   86 withTelegramChannel :: TelegramConfig -> NetworkHandle -> LogHandle -> (ChannelHandle -> IO a) -> IO a
   87 withTelegramChannel config nh lh action = do
   88   warnIfOpenAllowList lh telegramAllowListContext (_tc_allowFrom config)
   89   tc <- mkTelegramChannel config nh lh
   90   action (toHandle tc)

   92 instance Channel TelegramChannel where
   93   toHandle tc = ChannelHandle
   94     { _ch_receive      = receiveUpdate tc
   95     , _ch_send         = sendMessage tc
   96     , _ch_sendError    = sendTelegramError tc
   97     , _ch_sendChunk    = \_ -> pure ()  -- Telegram doesn't support streaming
   98     , _ch_streaming    = False
   99     , _ch_readSecret   = ioError (userError "Vault management requires the CLI interface")
  100     , _ch_prompt       = \promptText -> do
  101         sendMessage tc (OutgoingMessage promptText)
  102         _im_content <$> receiveUpdate tc
  103     , _ch_promptSecret = \_ ->
  104         ioError (userError "Vault management requires the CLI interface")
  105     }

  107 -- | Block until an *authorized* Telegram update arrives in the queue.
  108 --
  109 -- Reads one update at a time (consuming it from the inbox). An update passes
  110 -- when its allow-list policy permits either the sender's user id or the
  111 -- conversation's chat id.
  112 --
  113 -- Note: the chat id is matched by wrapping it in the 'UserId' newtype, the same
  114 -- way Signal wraps both phone numbers and UUIDs. This is an intentional
  115 -- conflation for access-control matching — the allow-list gates "may this
  116 -- message in?" against any sender-or-conversation identifier; it is not
  117 -- identity disambiguation. Numeric IDs only (usernames are not parsed or
  118 -- supported). Unauthorized updates are consumed, logged at WARN, and discarded
  119 -- (mirroring Signal's drop-and-log), then we recurse to read the next update.
  120 receiveUpdate :: TelegramChannel -> IO IncomingMessage
  121 receiveUpdate tc = do
  122   update <- atomically $ readTQueue (_tch_inbox tc)
  123   let msg = _tu_message update
  124       userId = T.pack (show (_tu_id (_tm_from msg)))
  125       chatId = _tcht_id (_tm_chat msg)
  126       content = _tm_text msg
  127       flds = Map.singleton "chat_id" (toJSON chatId)
  128       policy = _tc_allowFrom (_tch_config tc)
  129       allowed = isAllowed policy (UserId userId)
  130              || isAllowed policy (UserId (T.pack (show chatId)))
  131   if allowed
  132     then do
  133       writeIORef (_tch_lastChat tc) (Just chatId)
  134       pure IncomingMessage
  135         { _im_source  = mkMessageSource CkTelegram (Just (UserId userId)) flds
  136         , _im_content = content
  137         }
  138     else do
  139       _lh_logWarn (_tch_log tc) $
  140         "Blocked Telegram message from unauthorized sender (user "
  141           <> userId <> ", chat " <> T.pack (show chatId) <> ")"
  142       receiveUpdate tc

  144 -- | Send a message to the last active chat via the Telegram Bot API.
  145 sendMessage :: TelegramChannel -> OutgoingMessage -> IO ()
  146 sendMessage tc msg = do
  147   chatId <- readIORef (_tch_lastChat tc)
  148   case chatId of
  149     Nothing -> _lh_logWarn (_tch_log tc) "No chat_id available for send"
  150     Just cid -> do
  151       result <- try @SomeException (postTelegram tc "sendMessage" cid (_om_content msg))
  152       case result of
  153         Left e -> _lh_logError (_tch_log tc) $ "Telegram send failed: " <> T.pack (show e)
  154         Right resp
  155           | _hr_statusCode resp == 200 -> pure ()
  156           | otherwise ->
  157               _lh_logError (_tch_log tc) $
  158                 "Telegram API error " <> T.pack (show (_hr_statusCode resp))
  159 
  160 -- | Send an error message to the Telegram chat.
  161 sendTelegramError :: TelegramChannel -> PublicError -> IO ()
  162 sendTelegramError tc err = do
  163   chatId <- readIORef (_tch_lastChat tc)
  164   case chatId of
  165     Nothing -> _lh_logWarn (_tch_log tc) "No chat_id available for error send"
  166     Just cid -> do
  167       let errText = case err of
  168             RateLimitError -> "Rate limited. Please try again in a moment."
  169             NotAllowedError -> "Not allowed."
  170             TemporaryError t -> t
  171       result <- try @SomeException (postTelegram tc "sendMessage" cid errText)
  172       case result of
  173         Left e -> _lh_logError (_tch_log tc) $ "Telegram error send failed: " <> T.pack (show e)
  174         Right _ -> pure ()

  176 -- | POST to a Telegram Bot API method with chat_id and text parameters.
  177 postTelegram :: TelegramChannel -> Text -> Int -> Text -> IO HttpResponse
  178 postTelegram tc method chatId text = do
  179   let config = _tch_config tc
  180       url = _tc_apiBase config <> "/bot" <> _tc_botToken config <> "/" <> method
  181       body = "chat_id=" <> URI.urlEncode False (TE.encodeUtf8 (T.pack (show chatId)))
  182           <> "&text=" <> URI.urlEncode False (TE.encodeUtf8 text)
  183   case mkAllowedUrl AllowAll url of
  184     Left e -> throwIO (userError ("Bad Telegram URL: " <> show e))
  185     Right allowed -> _nh_httpPost (_tch_network tc) allowed body

  243 -- | The list of @(command, description)@ tuples PureClaw registers
  244 -- with Telegram's @setMyCommands@ API at channel-startup time.
  245 --
  246 -- Re-exported from "PureClaw.Routing.Onboarding" so the golden file
  247 -- and the @\/help@ subsection live next to each other; the Telegram
  248 -- channel only needs to publish the list, not own it.
  249 --
  250 -- See @docs\/tabbed-chat.md@ §\"Channel autocomplete\" (registration
  251 -- list) and §\"Onboarding (O-series)\" O3 (the golden enumeration).
  252 botFatherCommands :: [(Text, Text)]
  253 botFatherCommands = botFatherCommandList
  254 
  255 -- | Encode 'botFatherCommands' as the JSON payload Telegram's
  256 -- @setMyCommands@ endpoint accepts.
  257 --
  258 -- The endpoint expects @{ \"commands\": [ { \"command\": \"...\",
  259 -- \"description\": \"...\" }, ... ] }@. The leading slash on each
  260 -- command is stripped (Telegram expects bare command words). The
  261 -- resulting payload is a strict 'ByteString' suitable for an
  262 -- @application\/json@ POST body.
  263 encodeBotFatherCommands :: [(Text, Text)] -> ByteString
  264 encodeBotFatherCommands cmds =
  265   BL.toStrict (encode (object ["commands" .= map toEntry cmds]))
  266   where
  267     toEntry (c, d) = object
  268       [ "command"     .= T.dropWhile (== '/') c
  269       , "description" .= d
  270       ]
  271 
  272 -- | Register 'botFatherCommands' with Telegram via the
  273 -- @setMyCommands@ Bot API method.
  274 --
  275 -- POSTs an @application\/json@ payload to
  276 -- @\<api-base\>\/bot\<token\>\/setMyCommands@. Logs (but does not
  277 -- throw) on failure: BotFather autocomplete is a UX enhancement, not
  278 -- a correctness requirement, so a transient failure at startup must
  279 -- not prevent the bot from booting.
  280 --
  281 -- Mirrors the existing 'postTelegram' helper for URL construction
  282 -- (same shape, different endpoint name and Content-Type).
  283 --
  284 -- Per Tabbed Chat O3 this is invoked once at channel startup by the
  285 -- gateway boot path (callers wire it in alongside 'mkTelegramChannel').
  286 registerBotFatherCommands :: TelegramChannel -> IO ()
  287 registerBotFatherCommands tc = do
  288   let config = _tch_config tc
  289       url    = _tc_apiBase config <> "/bot" <> _tc_botToken config
  290             <> "/setMyCommands"
  291       body   = encodeBotFatherCommands botFatherCommands
  292   case mkAllowedUrl AllowAll url of
  293     Left e ->
  294       _lh_logError (_tch_log tc) $
  295         "BotFather setMyCommands URL rejected: " <> T.pack (show e)
  296     Right allowed -> do
  297       result <- try @SomeException
  298                   (_nh_httpPost (_tch_network tc) allowed body)
  299       case result of
  300         Left e ->
  301           _lh_logError (_tch_log tc) $
  302             "BotFather setMyCommands failed: " <> T.pack (show e)
  303         Right resp
  304           | _hr_statusCode resp == 200 ->
  305               _lh_logInfo (_tch_log tc)
  306                 "BotFather setMyCommands registered tab commands"
  307           | otherwise ->
  308               _lh_logError (_tch_log tc) $
  309                 "BotFather setMyCommands HTTP "
  310                   <> T.pack (show (_hr_statusCode resp))
