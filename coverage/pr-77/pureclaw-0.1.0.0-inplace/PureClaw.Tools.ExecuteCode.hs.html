   23 -- | Create an execute_code tool that runs Python (or other language)
   24 -- code in a subprocess. The interpreter must be allowed by the
   25 -- security policy. Output is capped at 50KB stdout / 10KB stderr.
   26 executeCodeTool :: SecurityPolicy -> (ToolDefinition, ToolHandler)
   27 executeCodeTool policy = (def, handler)
   28   where
   29     def = ToolDefinition
   30       { _td_name        = "execute_code"
   31       , _td_description = T.unlines
   32           [ "Execute code in a subprocess. The interpreter (python3, node, etc.)"
   33           , "must be allowed by the security policy."
   34           , "Code is passed via stdin. Output is capped at 50KB stdout / 10KB stderr."
   35           , "Default timeout: 30 seconds (configurable up to 300s)."
   36           ]
   37       , _td_inputSchema = object
   38           [ "type" .= ("object" :: Text)
   39           , "properties" .= object
   40               [ "code" .= object
   41                   [ "type" .= ("string" :: Text)
   42                   , "description" .= ("The code to execute" :: Text)
   43                   ]
   44               , "language" .= object
   45                   [ "type" .= ("string" :: Text)
   46                   , "description" .= ("Language/interpreter: python3, node, ruby, bash (default: python3)" :: Text)
   47                   ]
   48               , "timeout" .= object
   49                   [ "type" .= ("integer" :: Text)
   50                   , "description" .= ("Timeout in seconds (default: 30, max: 300)" :: Text)
   51                   ]
   52               ]
   53           , "required" .= (["code"] :: [Text])
   54           ]
   55       }
   56 
   57     handler = ToolHandler $ \input ->
   58       case parseEither parseCodeInput input of
   59         Left err -> pure (T.pack err, True)
   60         Right ci -> runCode ci
   61 
   62     runCode :: CodeInput -> IO (Text, Bool)
   63     runCode ci = do
   64       let interpreter = resolveInterpreter (_ci_language ci)
   65           timeoutSec = min 300 (fromMaybe 30 (_ci_timeout ci))
   66       case authorize policy (T.unpack interpreter) ["-"] of
   67         Left (CommandNotAllowed _) ->
   68           pure ("Interpreter not allowed: " <> interpreter
   69                <> ". Add it to --allow or set --autonomy full.", True)
   70         Left CommandInAutonomyDeny ->
   71           pure ("All commands denied by security policy", True)
   72         Right _authorized -> do
   73           let config = P.setStdin (P.byteStringInput (BL.fromStrict (TE.encodeUtf8 (_ci_code ci))))
   74                      $ P.setEnv safeEnv
   75                      $ P.proc (T.unpack interpreter) ["-"]
   76           worker <- Async.async $ P.readProcess config
   77           result <- Timeout.timeout (timeoutSec * 1000000) (Async.wait worker)
   78           case result of
   79             Nothing -> do
   80               Async.cancel worker
   81               pure ("Code execution timed out after " <> T.pack (show timeoutSec) <> "s", True)
   82             Just (exitCode, outLazy, errLazy) ->
   83               let outBs = BL.toStrict outLazy
   84                   errBs = BL.toStrict errLazy
   85                   out = truncateOutput 51200 (TE.decodeUtf8Lenient outBs)
   86                   err = truncateOutput 10240 (TE.decodeUtf8Lenient errBs)
   87                   exitInfo = case exitCode of
   88                     ExitSuccess   -> ""
   89                     ExitFailure n -> "\nExit code: " <> T.pack (show n)
   90                   parts = filter (not . T.null) [out, err, exitInfo]
   91                   combined = T.strip (T.intercalate "\n" parts)
   92               in pure (if T.null combined then "(no output)" else combined,
   93                        exitCode /= ExitSuccess)
   94 
   95     resolveInterpreter :: Maybe Text -> Text
   96     resolveInterpreter Nothing          = "python3"
   97     resolveInterpreter (Just "python")  = "python3"
   98     resolveInterpreter (Just "python3") = "python3"
   99     resolveInterpreter (Just "node")    = "node"
  100     resolveInterpreter (Just "ruby")    = "ruby"
  101     resolveInterpreter (Just "bash")    = "bash"
  102     resolveInterpreter (Just "sh")      = "sh"
  103     resolveInterpreter (Just other)     = other
  104 
  105     truncateOutput :: Int -> Text -> Text
  106     truncateOutput maxLen t
  107       | T.length t <= maxLen = t
  108       | otherwise = T.take maxLen t <> "\n[...truncated at " <> T.pack (show maxLen) <> " chars]"
  109 
  110 -- | Minimal safe environment for subprocesses.
  111 safeEnv :: [(String, String)]
  112 safeEnv = [("PATH", "/usr/bin:/bin:/usr/local/bin")]
  113 
  114 data CodeInput = CodeInput
  115   { _ci_code     :: Text
  116   , _ci_language :: Maybe Text
  117   , _ci_timeout  :: Maybe Int
  118   }
  119 
  120 parseCodeInput :: Value -> Parser CodeInput
  121 parseCodeInput = withObject "CodeInput" $ \o ->
  122   CodeInput
  123     <$> o .:  "code"
  124     <*> o .:? "language"
  125     <*> o .:? "timeout"
