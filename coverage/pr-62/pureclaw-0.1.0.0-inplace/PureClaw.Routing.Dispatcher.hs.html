    1 -- |
    2 -- Module      : PureClaw.Routing.Dispatcher
    3 -- Description : The single-threaded routing dispatcher (Tabbed Chat WU5).
    4 --
    5 -- The dispatcher is the one thread that reads from the
    6 -- 'PureClaw.Handles.Channel.ChannelHandle' on the user's side. It
    7 -- classifies every incoming message via 'PureClaw.Routing.Parse.parseInput'
    8 -- and routes it to one of:
    9 --
   10 --   * 'PureClaw.Routing.Types.Switch'         — set '_env_focus' to the named tab
   11 --   * 'PureClaw.Routing.Types.Inject'         — enqueue payload on the named tab's input queue
   12 --   * 'PureClaw.Routing.Types.Default'        — enqueue text on the currently focused tab
   13 --   * 'PureClaw.Routing.Types.ParsedSlashCmd' — handle the slash command (delegated to a hook in WU9\/WU10)
   14 --
   15 -- Per E3 the dispatcher is the only writer of '_env_focus' and runs
   16 -- every routing handler synchronously on its own thread — so there is
   17 -- no TOCTOU between read and write of focused-tab projections.
   18 --
   19 -- == Bootstrapping & cleanup (C4)
   20 --
   21 -- 'runDispatcher' is wrapped in @bracket (newIORef IntMap.empty) cancelAll dispatcherBody@.
   22 -- @cancelAll@ walks '_env_runners' (an @IntMap (IORef (Maybe TabRunner))@,
   23 -- the double-IORef registered by 'spawnTab' BEFORE the fork) and calls
   24 -- '_trun_cancel' on every filled slot. The 'PureClaw.Routing.ChannelOut'
   25 -- writer thread is owned by the same bracket and is cancelled on exit.
   26 --
   27 -- == Async exception discipline
   28 --
   29 -- 'spawnTab' runs under @mask@ so the registry-insert / fork / fill
   30 -- sequence is atomic with respect to async exceptions. 'forkIO' is
   31 -- FORBIDDEN in this module (and downstream tab-related modules): every
   32 -- new thread MUST go through '_env_fork'.
   33 --
   34 -- == Security gates wired here
   35 --
   36 --   * S5 — Crashed tab status is observed by the dispatcher and surfaced
   37 --     to the channel via 'toPublicTabError' (never raw 'TabError' Show).
   38 --   * S7 — per-user spawn-rate token-bucket; exceeding yields a
   39 --     'PublicError' and no spawn.
   40 --   * S8 — the dispatcher reads from '_ch_receive' only; channel-layer
   41 --     allowlist gating is canonical (Signal\/Telegram\/CLI). This module
   42 --     never bypasses the channel.
   43 --
   44 -- == LLM-free invariant (P18)
   45 --
   46 -- Switch \/ Inject \/ ParsedSlashCmd routing NEVER reaches the provider.
   47 -- Only 'PureClaw.Routing.Types.Default' inputs are forwarded to the
   48 -- focused tab's input queue (which may forward to the provider in the
   49 -- AI tab loop case landing in WU6).
   50 --
   51 -- See @docs\/tabbed-chat.md@ §"Dispatcher and Concurrency Model".

  165 -- | The production factory: dispatches by 'TabKind' to the relevant
  166 -- WU6\/WU7\/WU8 factory. Argument parsing for AI \/ Harness \/ Backend
  167 -- spawn arg blobs is shared with 'parseArgsForKind' so the per-kind
  168 -- projection is covered by 'parseArgsForKind' tests alone.

  201 -- ---------------------------------------------------------------------------
  202 -- Rate limit (S7)
  203 -- ---------------------------------------------------------------------------
  204 
  205 -- | Simple per-user token bucket used to defend against close-spawn
  206 -- cycling resource leaks. The bucket holds at most
  207 -- '_rc_spawnRateLimit' tokens; one full bucket refills every minute.
  208 --
  209 -- The representation is intentionally an 'IORef' over a 'Map' rather
  210 -- than 'TVar' — the dispatcher is single-threaded (E3) so STM atomicity
  211 -- is unnecessary, and an 'IORef' keeps the cabal\/test surface small.
  212 data RateLimiter = RateLimiter
  213   { _rl_buckets :: !(IORef (Map UserId (Double, UTCTime)))
  214     -- ^ Map from user id to @(tokens, last-refill-time)@. Tokens are
  215     --   real-valued for fractional refill credit.
  216   , _rl_capacity :: !Double
  217     -- ^ Maximum token count (config: '_rc_spawnRateLimit').
  218   , _rl_refillPerSecond :: !Double
  219     -- ^ Tokens added per second of elapsed real time. Configured so
  220     --   one full bucket refills every minute (cap / 60).
  221   }
  222 
  223 -- | Build a fresh rate limiter sized to the supplied
  224 -- '_rc_spawnRateLimit'. The cap doubles as the refill-per-minute, so
  225 -- @cap = 10@ → 10 spawns/minute (≈ one spawn every 6 seconds).
  226 newRateLimiter :: Int -> IO RateLimiter
  227 newRateLimiter cap = do
  228   ref <- newIORef Map.empty
  229   let capD = fromIntegral (max 0 cap)
  230   pure RateLimiter
  231     { _rl_buckets         = ref
  232     , _rl_capacity        = capD
  233     , _rl_refillPerSecond = capD / 60.0
  234     }
  235 
  236 -- | Atomically attempt to consume one spawn token for the given user
  237 -- at the given wall-clock time. Returns 'True' if a token was available
  238 -- (and consumed), 'False' if the bucket was empty.
  239 --
  240 -- An empty 'UserId' (e.g. a fake channel that supplies no sender) is
  241 -- treated as a single global user — every channel layer's allowlist
  242 -- gating is what assigns identity, so the dispatcher does not need to
  243 -- second-guess it.
  244 tryConsumeSpawnToken :: RateLimiter -> UserId -> UTCTime -> IO Bool
  245 tryConsumeSpawnToken rl uid now = atomicModifyIORef' (_rl_buckets rl) $ \m ->
  246   let (tokens, lastTime) = fromMaybe (_rl_capacity rl, now)
  247                                      (Map.lookup uid m)
  248       elapsed :: Double
  249       elapsed = realToFrac (diffUTCTime now lastTime :: NominalDiffTime)
  250       refilled = min (_rl_capacity rl)
  251                      (tokens + elapsed * _rl_refillPerSecond rl)
  252   in if refilled >= 1.0
  253        then ( Map.insert uid (refilled - 1.0, now) m
  254             , True
  255             )
  256        else ( Map.insert uid (refilled,        now) m
  257             , False
  258             )

  265 -- | Spawn a tab at the lowest free index using the production factory
  266 -- ('defaultTabFactory').
  267 --
  268 -- See 'spawnTabWith' for the parametrised variant used by tests.
  269 spawnTab :: AgentEnv -> TabKind -> [Text] -> IO (Either TabError TabIndex)
  270 spawnTab env = spawnTabWith env (defaultTabFactory env)
  271 
  272 -- | Spawn a tab at the lowest free index, dispatching to the supplied
  273 -- factory. Wrapping in 'mask' guarantees atomicity of the
  274 -- @register-placeholder \/ build-handle \/ insert-registry \/ fork \/ fill-placeholder@
  275 -- sequence with respect to async exceptions:
  276 --
  277 -- 1. Allocate a placeholder @IORef Nothing :: IORef (Maybe TabRunner)@.
  278 -- 2. Insert the placeholder into '_env_runners' under mask — from this
  279 --    point 'closeAllTabs' will see the slot even if we're interrupted
  280 --    before the fork.
  281 -- 3. Run the factory (which itself does not need to fork the tab loop —
  282 --    it constructs and returns a 'TabHandle' record; the per-tab loop
  283 --    is forked by THIS function via '_env_fork').
  284 -- 4. Insert the resulting 'TabHandle' into '_env_tabs'.
  285 -- 5. Fork the tab loop via '_env_fork' (NOT 'forkIO') and write the
  286 --    returned 'TabRunner' into the placeholder.
  287 --
  288 -- Note on the v1 design: in WU5 the factory body still bottoms (WU1
  289 -- stub) so the real per-tab loop is not started here. WU6\/7\/8 will
  290 -- fill the factory bodies and the loop will run for real. The mask
  291 -- discipline below is correct for the eventual real-factory case and
  292 -- intentionally generic — we do NOT start a separate per-tab loop here
  293 -- because the factory's '_tabHandle_close' is what drives the tab's
  294 -- own internal threading.
  295 spawnTabWith
  296   :: AgentEnv
  297   -> TabFactory
  298   -> TabKind
  299   -> [Text]
  300   -> IO (Either TabError TabIndex)
  301 spawnTabWith env factory kind args =
  302   -- mask: the register-then-build sequence below cannot be interrupted
  303   -- mid-flight. If an async exception arrives between insert and fork,
  304   -- 'closeAllTabs' would otherwise leak the in-flight placeholder.
  305   -- Using 'mask' here matches the design's §"Async exception discipline".
  306   mask $ \restore -> do
  307     let rc = _env_routingConfig env
  308     mIdx <- Registry.lowestFreeIndex (_env_tabs env) (_rc_maxTabs rc)
  309     case mIdx of
  310       Nothing  -> pure (Left (TabLimitExceeded (_rc_maxTabs rc)))
  311       Just idx -> doSpawn restore idx
  312   where
  313     doSpawn restore idx = do
  314       runnerRef <- newIORef Nothing
  315       -- Register the placeholder BEFORE running the factory. cancelAll
  316       -- will see this slot (with a Nothing runner — no-op) for the
  317       -- interval between the insert below and the writeIORef at the
  318       -- bottom; that's deliberate per the design.
  319       modifyIORef' (_env_runners env)
  320                    (IntMap.insert (unTabIndex idx) runnerRef)
  321       -- Run the (potentially long) factory inside 'restore' so async
  322       -- exceptions can still arrive at it. On any exception, roll back
  323       -- the placeholder insert and re-raise.
  324       mResult <- restore (try @SomeException (factory kind idx args))
  325                    `onException` rollbackPlaceholder idx
  326       case mResult of
  327         Left e        -> rollbackPlaceholder idx >> rethrowOrLeft e
  328         Right (Left tabErr) -> do
  329           rollbackPlaceholder idx
  330           pure (Left tabErr)
  331         Right (Right th) -> do
  332           ins <- Registry.insertTab (_env_tabs env) idx th
  333           case ins of
  334             Left tabErr -> do
  335               -- Index race (extremely unlikely under single-threaded
  336               -- dispatcher invariant E3; still defend so the registry
  337               -- never gets a double-entry).
  338               rollbackPlaceholder idx
  339               pure (Left tabErr)
  340             Right () ->
  341               -- The placeholder runner stays 'Nothing' for WU5. The
  342               -- WU6/7/8 factory bodies will fork the per-tab loop
  343               -- internally; once they do, the loop's TabRunner can be
  344               -- written into runnerRef. For WU5 we leave it as
  345               -- Nothing so cancelAll is a safe no-op for tabs whose
  346               -- loops were never started.
  347               pure (Right idx)
  348 
  349     rollbackPlaceholder idx =
  350       modifyIORef' (_env_runners env)
  351                    (IntMap.delete (unTabIndex idx))
  352 
  353     rethrowOrLeft :: SomeException -> IO (Either TabError TabIndex)
  354     rethrowOrLeft _e =
  355       -- For C3, an exception from the factory must NOT leave any
  356       -- partially-allocated state. The placeholder was already rolled
  357       -- back; surface the failure as a generic spawn error rather than
  358       -- re-raising — the dispatcher caller treats Left as a
  359       -- 'PublicError' and the underlying 'SomeException' message is
  360       -- intentionally discarded (S5 redaction principle).
  361       pure (Left (TabSessionCreateFailed Tab.SessionError))

  364 -- ---------------------------------------------------------------------------
  365 -- closeAllTabs
  366 -- ---------------------------------------------------------------------------
  367 
  368 -- | Walk '_env_runners' and cancel every started runner. Tolerant of
  369 -- the WU5-era 'Nothing' placeholders (no-op for those).
  370 --
  371 -- This is the bracket cleanup for 'runDispatcher': it fires on BOTH
  372 -- exception AND graceful exit (e.g. '_ch_receive' returning end-of-
  373 -- stream via a thrown exception). The first-pass cancel is fire-and-
  374 -- forget; we then walk the map again to '_trun_wait' on each.
  375 closeAllTabs :: AgentEnv -> IO ()
  376 closeAllTabs env = do
  377   m <- readIORef (_env_runners env)
  378   -- First pass: send cancel to every filled runner. We catch and ignore
  379   -- exceptions from cancel because tab cleanup must never block the
  380   -- dispatcher's own bracket cleanup.
  381   for_ (IntMap.elems m) $ \ref -> do
  382     mr <- readIORef ref
  383     case mr of
  384       Nothing      -> pure ()
  385       Just runner  -> safeIgnore (_trun_cancel runner)
  386   -- Second pass: also drive '_tabHandle_close CloseGraceful' on every
  387   -- registered handle, so per-tab resources (transcript, backend
  388   -- close, harness stop) are released even when the tab loop never
  389   -- fully started. '_tabHandle_close' is documented idempotent and
  390   -- never-throws (H6/H7/H8); we additionally ignore exceptions
  391   -- defensively.
  392   tabs <- readIORef (_env_tabs env)
  393   for_ (IntMap.elems tabs) $ \h ->
  394     safeIgnore (_tabHandle_close h CloseGraceful)
  395 
  396 safeIgnore :: IO () -> IO ()
  397 safeIgnore m = do
  398   _ <- try @SomeException m
  399   pure ()

  406 -- | The pieces of per-run state the dispatcher carries across message
  407 -- cycles. Exposed so 'dispatchOne' can be driven from tests without
  408 -- running 'runDispatcher's infinite loop.
  409 --
  410 -- /WU9 additions:/ '_ds_spawnArgs' records the original
  411 -- @(TabKind, [Text])@ pair for each successful spawn so the X2 \/ X3
  412 -- crashed-tab retry path can replay the exact original arguments
  413 -- (per OQ-6). '_ds_promptRenderer' picks the channel-appropriate
  414 -- spawn-prompt rendering (text on CLI \/ Signal, inline-keyboard
  415 -- friendly on Telegram). '_ds_factory' is the per-kind tab factory
  416 -- the dispatcher will dispatch through for all WU9 spawn paths.
  417 --
  418 -- /WU10 addition:/ '_ds_pendingRetry' records per-user
  419 -- @(TabIndex, SpawnArgs)@ for the most-recently emitted X1 retry
  420 -- prompt. A subsequent \"1\" or \"2\" message from the same user is
  421 -- interpreted as a retry-or-close reply (X1 reply parsing — WU9
  422 -- emitted the prompt but did NOT interpret the reply). The map is
  423 -- single-entry-per-user; a new prompt overrides the previous
  424 -- expectation.
  425 data DispatcherState = DispatcherState
  426   { _ds_rateLimiter    :: !RateLimiter
  427     -- ^ Per-chat-user spawn token bucket (S7).
  428   , _ds_spawnArgs      :: !(IORef (Map Int AutoSpawn.SpawnArgs))
  429     -- ^ Per-tab original spawn args (kind + arg list) for X2 \/ X3
  430     -- retry.
  431   , _ds_promptRenderer :: !Prompt.PromptRenderer
  432     -- ^ Channel-aware spawn-prompt renderer (A4 \/ A6).
  433   , _ds_factory        :: !TabFactory
  434     -- ^ The per-kind factory the dispatcher uses for spawning. Defaults
  435     -- to 'defaultTabFactory env' for production; tests inject a
  436     -- synthetic factory via 'newDispatcherState'.
  437   , _ds_pendingRetry   :: !(IORef (Map UserId (TabIndex, AutoSpawn.SpawnArgs)))
  438     -- ^ Per-user expectation of an X1 retry reply (WU10). Populated
  439     -- by the focus path when it observes a 'Crashed' tab; cleared on
  440     -- reply or on any other intervening user input.
  441   }
  442 
  443 -- | Build a fresh 'DispatcherState' from an 'AgentEnv'.
  444 --
  445 -- The factory argument is retained on the record so WU9's AutoSpawn
  446 -- handlers can drive spawns without re-passing the factory at every
  447 -- call site.
  448 newDispatcherState :: AgentEnv -> TabFactory -> IO DispatcherState
  449 newDispatcherState env factory = do
  450   rl       <- newRateLimiter (_rc_spawnRateLimit (_env_routingConfig env))
  451   argsRef  <- newIORef Map.empty
  452   pendRef  <- newIORef Map.empty
  453   let renderer = Prompt.mkDefaultPromptRenderer (_env_channel env)
  454   pure DispatcherState
  455     { _ds_rateLimiter    = rl
  456     , _ds_spawnArgs      = argsRef
  457     , _ds_promptRenderer = renderer
  458     , _ds_factory        = factory
  459     , _ds_pendingRetry   = pendRef
  460     }

  463 -- ---------------------------------------------------------------------------
  464 -- dispatchOne — process a single parsed input
  465 -- ---------------------------------------------------------------------------
  466 
  467 -- | Process exactly one incoming message:
  468 --
  469 -- 1. Parse with 'Parse.parseInput'. On 'Left' emit a 'PublicError' to
  470 --    the channel-out queue via 'SrcDispatcher' (no handler invocation,
  471 --    no provider call — P18 preserved).
  472 -- 2. On 'Switch': set '_env_focus'.
  473 -- 3. On 'Inject': enqueue the payload onto tab N's input queue via
  474 --    '_tabHandle_send'.
  475 -- 4. On 'Default': forward to the currently focused tab's input queue
  476 --    (or no-op if no focus — auto-spawn UX lives in WU9).
  477 -- 5. On 'ParsedSlashCmd': for /tab* commands the dispatcher handles
  478 --    spawn\/close\/focus\/etc inline (WU9 will extend); other slash
  479 --    commands are deferred to the focused tab's enqueueSlash (E5\/I5).
  480 --
  481 -- The function is intentionally pure-ish (only IO side effects), so
  482 -- tests can drive it one message at a time with deterministic
  483 -- timestamps.
  484 dispatchOne :: AgentEnv -> DispatcherState -> UserId -> Text -> IO ()
  485 dispatchOne = dispatchOneWith
  486 
  487 -- | Like 'dispatchOne' but explicitly named for the test-seam variant
  488 -- — at present they are identical (the 'DispatcherState' record
  489 -- carries every per-run knob WU9 needs); kept as a distinct symbol so
  490 -- future WUs can introduce an even-thinner test-only variant without
  491 -- breaking the public 'dispatchOne' contract.
  492 dispatchOneWith :: AgentEnv -> DispatcherState -> UserId -> Text -> IO ()
  493 dispatchOneWith env ds uid raw = do
  494   -- WU10 — X1 retry-reply intercept: a bare "1" or "2" from a user with
  495   -- a pending crashed-tab retry expectation is interpreted as a
  496   -- retry/close reply. Any other input clears the pending expectation
  497   -- (so a follow-up retry must be initiated by re-focusing the tab).
  498   pendingHit <- tryRetryReply env ds uid raw
  499   if pendingHit
  500     then pure ()
  501     else doDispatchOne env ds uid raw
  502 
  503 -- | Main dispatch body, factored out so the X1 retry-reply intercept
  504 -- in 'dispatchOneWith' can short-circuit without nesting.
  505 doDispatchOne :: AgentEnv -> DispatcherState -> UserId -> Text -> IO ()
  506 doDispatchOne env ds uid raw = do
  507   let rc = _env_routingConfig env
  508   case Parse.parseInput rc raw of
  509     Left _err ->
  510       -- Parser errors carry only bounded primitives; surfacing
  511       -- TemporaryError "input not recognized" keeps the channel clean
  512       -- (no constructor leaks).
  513       emitDispatcherBanner env "input not recognized"
  514 
  515     Right (Switch idx) -> do
  516       -- tmux packing model: a missing /N is user error (no auto-spawn).
  517       AutoSpawn.handleSwitch env
  518         (emitDispatcherBanner env)
  519         idx
  520       -- WU10 — X1 expectation capture: if the focused tab is now
  521       -- Crashed, record the spawn args under the user's pending-retry
  522       -- map so a subsequent "1"/"2" reply lands somewhere.
  523       noteCrashedExpectation env ds uid idx
  524 
  525     Right (Inject idx payload) ->
  526       injectTo env idx payload
  527 
  528     Right (Default text) -> do
  529       mFocus <- readIORef (_env_focus env)
  530       case mFocus of
  531         Nothing  ->
  532           -- L6 \/ K3: empty focus + Default text → implicit auto-spawn
  533           -- of '_rc_defaultKind' at the lowest free index, then
  534           -- forward the text to it.
  535           AutoSpawn.handleDefault env
  536             (ratelimitedSpawn env ds uid)
  537             (emitDispatcherBanner env)
  538             (_ds_spawnArgs ds)
  539             text
  540         Just idx -> injectTo env idx text

  550 -- | Intercept a bare \"1\" or \"2\" reply from a user with a pending
  551 -- retry expectation. Returns 'True' if the input was consumed as a
  552 -- retry reply (and the caller should NOT continue dispatch), 'False'
  553 -- otherwise.
  554 --
  555 -- The pending-retry map is cleared on every non-reply input — sending
  556 -- any other message dismisses the prompt.
  557 tryRetryReply
  558   :: AgentEnv -> DispatcherState -> UserId -> Text -> IO Bool
  559 tryRetryReply env ds uid raw = do
  560   let trimmed = T.strip raw
  561   pending <- readIORef (_ds_pendingRetry ds)
  562   case Map.lookup uid pending of
  563     Nothing -> pure False
  564     Just (idx, sa) -> case trimmed of
  565       "1" -> do
  566         clearPendingRetry ds uid
  567         retryCrashedTab env ds uid idx sa
  568         pure True
  569       "2" -> do
  570         clearPendingRetry ds uid
  571         closeCrashedTab env idx
  572         pure True
  573       _ -> do
  574         -- Any other input dismisses the prompt — the user moved on.
  575         clearPendingRetry ds uid
  576         pure False

  578 -- | Record an X1 retry expectation for the user, if the focused tab
  579 -- is Crashed. No-op for any other status (Active / Idle) so non-crashed
  580 -- focus paths don't leak pending entries.
  581 noteCrashedExpectation
  582   :: AgentEnv -> DispatcherState -> UserId -> TabIndex -> IO ()
  583 noteCrashedExpectation env ds uid idx = do
  584   mTab <- Registry.lookupTab (_env_tabs env) idx
  585   case mTab of
  586     Nothing -> pure ()
  587     Just th -> do
  588       r <- try @SomeException (_tabHandle_status th)
  589       case r of
  590         Right (Crashed _) -> do
  591           argsMap <- readIORef (_ds_spawnArgs ds)
  592           case Map.lookup (unTabIndex idx) argsMap of
  593             Just sa ->
  594               atomicModifyIORef' (_ds_pendingRetry ds)
  595                 (\m -> (Map.insert uid (idx, sa) m, ()))
  596             Nothing ->
  597               -- No spawn args recorded — retry would have nothing to
  598               -- replay. Skip the expectation entry rather than racing
  599               -- the user into a dead-end "1" reply.
  600               pure ()
  601         _ -> pure ()

  603 -- | Retry a crashed tab: close the existing tab handle (idempotent +
  604 -- never-throws) and respawn with the original spawn args (X2 \/ X3).
  605 -- Spawning is rate-limited (S7) as usual.
  606 retryCrashedTab
  607   :: AgentEnv
  608   -> DispatcherState
  609   -> UserId
  610   -> TabIndex
  611   -> AutoSpawn.SpawnArgs
  612   -> IO ()
  613 retryCrashedTab env ds uid idx sa = do
  614   -- Drive the close on the old (crashed) handle so resources are
  615   -- released. The handle's _tabHandle_close is documented idempotent.
  616   mTab <- Registry.lookupTab (_env_tabs env) idx
  617   for_ mTab $ \th ->
  618     safeIgnore (_tabHandle_close th CloseGraceful)
  619   -- Remove the old entry from the registry so spawnTab can re-use
  620   -- the index.
  621   _ <- Registry.removeTab (_env_tabs env) idx
  622   -- Respawn with the original args.
  623   r <- ratelimitedSpawn env ds uid
  624          (AutoSpawn._sa_kind sa) (AutoSpawn._sa_args sa)
  625   case r of
  626     Left tabErr ->
  627       emitDispatcherBanner env
  628         ("/" <> tShowIdx idx <> ": retry failed — "
  629           <> unPublicTabError (toPublicTabError tabErr))
  630     Right newIdx -> do
  631       writeIORef (_env_focus env) (Just newIdx)
  632       emitDispatcherBanner env
  633         ("/" <> tShowIdx newIdx <> ": retried")
  634 
  635 -- | Close a crashed tab without retrying. The handle's
  636 -- '_tabHandle_close' is invoked (idempotent) and the registry entry
  637 -- is removed so the index is free for re-use.
  638 closeCrashedTab :: AgentEnv -> TabIndex -> IO ()
  639 closeCrashedTab env idx = do
  640   mTab <- Registry.lookupTab (_env_tabs env) idx
  641   for_ mTab $ \th -> safeIgnore (_tabHandle_close th CloseGraceful)
  642   _ <- Registry.removeTab (_env_tabs env) idx
  643   -- If the closed tab was focused, clear focus so the next Default
  644   -- input either implicit-spawns or routes correctly.
  645   mFocus <- readIORef (_env_focus env)
  646   when (mFocus == Just idx) $
  647     writeIORef (_env_focus env) Nothing
  648   emitDispatcherBanner env
  649     ("/" <> tShowIdx idx <> ": closed")

  651 -- | Drop the pending retry entry for the user. Idempotent.
  652 clearPendingRetry :: DispatcherState -> UserId -> IO ()
  653 clearPendingRetry ds uid =
  654   atomicModifyIORef' (_ds_pendingRetry ds)
  655     (\m -> (Map.delete uid m, ()))

  658 -- | Spawn a tab via the dispatcher's configured factory, consuming
  659 -- one rate-limit token first (S7). Returns 'Left' with a redacted
  660 -- 'TabError' on either:
  661 --
  662 --   * rate-limit denial → 'TabConcurrencyLimit' (the closest
  663 --     semantically-correct redacted shape — the public projection
  664 --     renders as "tab: input queue full" which is acceptable v1
  665 --     coverage); or
  666 --   * factory failure → propagated as-is from 'spawnTabWith'.
  667 --
  668 -- AutoSpawn handlers consume this as their 'SpawnIO' callback so the
  669 -- spawn-rate limit is enforced uniformly across A-series spawn paths.
  670 ratelimitedSpawn
  671   :: AgentEnv
  672   -> DispatcherState
  673   -> UserId
  674   -> TabKind
  675   -> [Text]
  676   -> IO (Either TabError TabIndex)
  677 ratelimitedSpawn env ds uid kind args = do
  678   now <- getCurrentTime
  679   ok  <- tryConsumeSpawnToken (_ds_rateLimiter ds) uid now
  680   if not ok
  681     then pure (Left (TabConcurrencyLimit 0))
  682     else spawnTabWith env (_ds_factory ds) kind args

  685 -- | Dispatch a slash command. The @\/tab*@ family is routed to the
  686 -- AutoSpawn handlers; every other slash command is delivered to the
  687 -- focused AI tab via '_tabHandle_enqueueSlash' (WU10 — I5 wiring).
  688 --
  689 -- The focused-tab projection rules:
  690 --
  691 --   * Focused 'KindAi' tab → enqueue 'SlashCmd' on the tab's input queue;
  692 --     the tab loop runs @executeSlashCommand env cmd ctx@ against its
  693 --     per-tab context (E5 single-writer invariant).
  694 --   * Focused non-AI tab → emit a redacted PublicError
  695 --     (\"@\/N\: tab kind does not support this command@\") and do NOT
  696 --     enqueue. The 'TabUnsupportedCommand' is the canonical surface
  697 --     mirroring K5's @\/target@ pattern.
  698 --   * No focus → fall back to the in-dispatcher legacy executor against
  699 --     an empty 'Context'. This preserves single-tab behaviour for
  700 --     channels that have not yet auto-spawned a tab and is also what
  701 --     drives the empty-registry K3 implicit-spawn path through the
  702 --     normal channel flow.
  703 dispatchSlash
  704   :: AgentEnv -> DispatcherState -> UserId -> SlashCommand -> IO ()
  705 dispatchSlash env ds uid cmd = case cmd of
  706   CmdTab tabCmd -> dispatchTab env ds uid tabCmd
  707   -- Onboarding (O1): handled at the dispatcher layer because the
  708   -- response is a fixed orientation message — it must not depend on a
  709   -- focused AI tab (a brand-new user has no tabs yet), and it must
  710   -- not be enqueued into the focused tab's slash queue (which would
  711   -- be wrong-channel-of-output for a UX greeting). Delegated to
  712   -- 'Onboarding.handleStart' which emits via @_ch_send@ directly.
  713   CmdStart      -> Onboarding.handleStart env
  714   _             -> routeToFocused env ds uid cmd
  715 
  716 -- | Route a non-@\/tab*@ slash command to the focused tab via
  717 -- '_tabHandle_enqueueSlash' (WU10 — I5 wiring).
  718 --
  719 -- Returns immediately without blocking on the tab loop's processing.
  720 -- Per E5, the tab loop is the single writer to its per-tab context.
  721 routeToFocused
  722   :: AgentEnv -> DispatcherState -> UserId -> SlashCommand -> IO ()
  723 routeToFocused env ds uid cmd = do
  724   -- Consume one rate-limit token (S7) per slash dispatch — exercises the
  725   -- same gate the pre-WU10 'ratelimitedSlashAck' walked.
  726   now <- getCurrentTime
  727   ok  <- tryConsumeSpawnToken (_ds_rateLimiter ds) uid now
  728   if not ok
  729     then emitDispatcherBanner env "rate limit: too many recent spawns"
  730     else do
  731       mFocus <- readIORef (_env_focus env)
  732       case mFocus of
  733         Nothing  -> emitDispatcherBanner env (noFocusBanner cmd)
  734         Just idx -> enqueueOnFocused env idx cmd
  735 
  736 -- | Look up the focused tab and dispatch the slash command per kind.
  737 -- On 'KindAi' the command is enqueued; on any other kind we surface a
  738 -- redacted PublicError without enqueueing (K5 pattern, generalised).
  739 enqueueOnFocused :: AgentEnv -> TabIndex -> SlashCommand -> IO ()
  740 enqueueOnFocused env idx cmd = do
  741   mTab <- Registry.lookupTab (_env_tabs env) idx
  742   case mTab of
  743     Nothing -> emitDispatcherBanner env
  744                  ("/" <> tShowIdx idx <> ": no such tab")
  745     Just th -> case _tabHandle_kind th of
  746       TkSession (SkProvider _) -> do
  747         r <- safeEnqueueSlash th cmd
  748         case r of
  749           Right () -> pure ()
  750           Left  e  -> emitDispatcherBanner env
  751                         ("/" <> tShowIdx idx <> ": "
  752                          <> unPublicTabError (toPublicTabError e))
  753       _ ->
  754         -- Non-AI focused tab: per the K5 generalisation, surface a
  755         -- redacted PublicError and do NOT enqueue.
  756         emitDispatcherBanner env
  757           ("/" <> tShowIdx idx
  758             <> ": tab kind does not support this command")

  760 -- | Banner shown when a non-@\/tab*@ slash command is received but
  761 -- nothing is focused. The cases are too disparate to render a fully
  762 -- contextual message; we emit a short hint to spawn or focus a tab so
  763 -- the slash command can land somewhere.
  764 noFocusBanner :: SlashCommand -> Text
  765 noFocusBanner _ =
  766   "no focused tab — focus one with /N or spawn one with /tab new <kind>"

  768 -- | Wrap '_tabHandle_enqueueSlash' so any synchronous exception
  769 -- becomes a redacted 'TabConcurrencyLimit' rather than a dispatcher
  770 -- crash. Mirrors 'safeEnqueue' for the 'UserText' path.
  771 safeEnqueueSlash :: TabHandle -> SlashCommand -> IO (Either TabError ())
  772 safeEnqueueSlash th cmd = do
  773   r <- try @SomeException (_tabHandle_enqueueSlash th cmd)
  774   pure $ case r of
  775     Left  _e   -> Left (TabConcurrencyLimit 0)
  776     Right inner -> inner

  783 -- | Handle @\/tab resume \<id\>@. The parser already validated the
  784 -- 'SessionId' via 'mkSessionId' (P15a). WU10 wires the L7 chain:
  785 --
  786 --   1. 'Session.resolveSessionRef' — resolve the id under
  787 --      @~\/.pureclaw\/sessions@.
  788 --   2. 'Session.resumeSession' — load metadata + open transcript.
  789 --   3. @writeIORef '_env_session'@ — make the resumed session the
  790 --      active one so the spawned tab's factory snapshots it.
  791 --   4. @'spawnTab' env 'KindAi' []@ — create a new KindAi tab via the
  792 --      dispatcher's configured factory (rate-limited per S7).
  793 --
  794 -- On any failure the dispatcher emits a redacted banner and aborts
  795 -- without touching '_env_session' (the previous active session
  796 -- remains).
  797 handleResumeIntoTab
  798   :: AgentEnv -> DispatcherState -> UserId -> SessionId -> IO ()
  799 handleResumeIntoTab env ds uid sid = do
  800   sessionsDir <- getSessionsDir
  801   resolved <- try @SomeException
  802                 (Session.resolveSessionRef sessionsDir (unSessionId sid))
  803   case resolved of
  804     Left _e ->
  805       emitDispatcherBanner env
  806         ("/tab resume " <> unSessionId sid <> ": resolve failed")
  807     Right (Left Session.NotFound) ->
  808       emitDispatcherBanner env
  809         ("/tab resume " <> unSessionId sid <> ": no such session")
  810     Right (Left (Session.Ambiguous _matches)) ->
  811       emitDispatcherBanner env
  812         ("/tab resume " <> unSessionId sid <> ": ambiguous session id")
  813     Right (Right realSid) -> do
  814       handleResolved env ds uid sessionsDir realSid
  815 
  816 -- | Common tail of 'handleResumeIntoTab': resume the session, swap it
  817 -- into 'AgentEnv', then spawn a new AI tab. Factored out so the
  818 -- top-level handler stays readable.
  819 handleResolved
  820   :: AgentEnv
  821   -> DispatcherState
  822   -> UserId
  823   -> FilePath
  824   -> SessionId
  825   -> IO ()
  826 handleResolved env ds uid sessionsDir realSid = do
  827   resumed <- try @SomeException
  828                (Session.resumeSession
  829                   (_env_broker env) (_env_logger env) sessionsDir realSid)
  830   case resumed of
  831     Left _e ->
  832       emitDispatcherBanner env
  833         ("/tab resume " <> unSessionId realSid <> ": resume failed")
  834     Right (Left _err) ->
  835       emitDispatcherBanner env
  836         ("/tab resume " <> unSessionId realSid <> ": resume failed")
  837     Right (Right newSession) -> do
  838       -- L7 design: replace the active session in place so the spawned
  839       -- tab's factory snapshot picks it up. Prior tabs that already
  840       -- snapshotted '_env_session' keep their handles (v1 limitation
  841       -- noted in design's K7 §). The new tab will own the resumed
  842       -- session.
  843       writeIORef (_env_session env) newSession
  844       let resumeKind = TkSession (SkProvider ProviderSpec
  845                         { _ps_provider = ProviderId "anthropic"
  846                         , _ps_model    = ModelId "placeholder"
  847                         , _ps_agent    = Nothing
  848                         })
  849       spawnResult <- ratelimitedSpawn env ds uid resumeKind []
  850       case spawnResult of
  851         Left tabErr ->
  852           emitDispatcherBanner env
  853             ("/tab resume " <> unSessionId realSid <> ": "
  854               <> unPublicTabError (toPublicTabError tabErr))
  855         Right newIdx -> do
  856           -- Set focus to the newly-spawned tab so the user lands on
  857           -- it immediately.
  858           writeIORef (_env_focus env) (Just newIdx)
  859           emitDispatcherBanner env
  860             ("/tab resume " <> unSessionId realSid <> ": resumed at /"
  861               <> tShowIdx newIdx)

  863 -- | Dispatch a @\/tab*@ command to the AutoSpawn handlers. Each branch
  864 -- maps a 'TabSlashCommand' constructor to the appropriate handler.

  866 -- The @\/tab close N@ branch supplies AutoSpawn with a renumber
  867 -- callback that shifts the dispatcher's side maps ('_ds_pendingRetry'

  907 -- | Shift the keys of the per-user pending-retry map so that any
  908 -- pending @(TabIndex, SpawnArgs)@ payload whose tab index was just
  909 -- renumbered down by one stays pointed at the same tab.
  910 --
  911 -- Entries whose @TabIndex@ equals the closed slot are dropped (the
  912 -- referenced tab no longer exists); entries whose index is greater
  913 -- have the index decremented; entries with smaller indices stay put.
  914 renumberPendingRetry :: DispatcherState -> Int -> IO ()
  915 renumberPendingRetry ds closedN =
  916   atomicModifyIORef' (_ds_pendingRetry ds) $ \m ->
  917     (Map.mapMaybe (shiftEntry closedN) m, ())
  918   where
  919     shiftEntry k (idx, sa) =
  920       let i = unTabIndex idx
  921       in if i == k
  922            then Nothing
  923            else if i > k
  924                   then case Tab.mkTabIndex (i - 1) of
  925                          Just newIdx -> Just (newIdx, sa)
  926                          Nothing     -> Nothing  -- unreachable: i > k >= 0
  927                   else Just (idx, sa)

 1001 -- | Run the dispatcher forever, using the default per-kind factory.
 1002 --
 1003 -- Wraps everything in 'bracket' so a crash inside the loop or a
 1004 -- graceful end-of-stream from '_ch_receive' triggers 'closeAllTabs'
 1005 -- and tears down the channel-out writer.
 1006 runDispatcher :: AgentEnv -> IO ()
 1007 runDispatcher env = runDispatcherWith env (defaultTabFactory env)
 1008 
 1009 -- | Run the dispatcher forever with a caller-supplied factory.
 1010 --
 1011 -- Tests use this to inject 'TabFactory' stubs without touching the
 1012 -- WU1 stub-bottomed production factories.
 1013 runDispatcherWith :: AgentEnv -> TabFactory -> IO ()
 1014 runDispatcherWith env factory =
 1015   -- The outer bracket guarantees:
 1016   --   * the channel-out writer thread is started under mask_ so it
 1017   --     cannot be lost to an async exception;
 1018   --   * 'closeAllTabs' fires on BOTH exception and graceful exit
 1019   --     (per C4).
 1020   bracket
 1021     (acquire env factory)
 1022     (release env)
 1023     (loop env)
 1024   where
 1025     loop e (ds, _outR) = do
 1026       msg <- _ch_receive (_env_channel e)
 1027       dispatchOne e ds (_im_userId msg) (_im_content msg)
 1028       loop e (ds, _outR)
 1029 
 1030 -- | The 'bracket' acquire: starts the channel-out writer, builds the
 1031 -- dispatcher state. Uses 'mask_' to ensure the writer-thread runner is
 1032 -- captured into a field accessible to 'release'.
 1033 acquire :: AgentEnv -> TabFactory -> IO (DispatcherState, TabRunner)
 1034 acquire env factory = mask_ $ do
 1035   ds   <- newDispatcherState env factory
 1036   outR <- ChannelOut.startChannelOut env
 1037   pure (ds, outR)
 1038 
 1039 -- | The 'bracket' release: cancel the channel-out writer thread, then
 1040 -- run 'closeAllTabs' to drive per-tab cleanup. Both steps are
 1041 -- exception-tolerant (per the cleanup contract).
 1042 release :: AgentEnv -> (DispatcherState, TabRunner) -> IO ()
 1043 release env (_ds, outR) = do
 1044   safeIgnore (_trun_cancel outR)
 1045   closeAllTabs env
 1046   -- Drain a debug log line so an explicit shutdown is observable in
 1047   -- the logs. The logger is documented total in mkNoOpLogHandle, but
 1048   -- we wrap defensively.
 1049   safeIgnore (_lh_logDebug (_env_logger env) "dispatcher: closeAll complete")
 1050
